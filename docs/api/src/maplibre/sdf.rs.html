66/// Registers symbol result ingestion, atlas uploads, placement and drawing.
67/// Vector processing supplies the symbol buckets using the same transferable type `T`.
68pub struct SdfPlugin<T>(PhantomData<T>);
69
70impl<T: VectorTransferables> Default for SdfPlugin<T> {
71    fn default() -> Self {
72        Self(Default::default())
73    }
74}
75
76impl<E: Environment, T: VectorTransferables> Plugin<E> for SdfPlugin<T> {
77    fn build(
78        &self,
79        schedule: &mut Schedule,
80        kernel: Rc<Kernel<E>>,
81        world: &mut World,
82        _graph: &mut RenderGraph,
83    ) {
84        let resources = &mut world.resources;
85
86        resources.insert(Eventually::<SymbolPipeline>::Uninitialized);
87        resources.insert(Eventually::<SymbolBufferPool>::Uninitialized);
88        resources.insert(textures::SymbolTextures::default());
89        resources.insert(Eventually::<depth::SymbolDepth>::Uninitialized);
90
91        schedule.add_system_to_stage(
92            RenderStageLabel::Extract,
93            SystemContainer::new(populate_world_system::PopulateWorldSystem::<E, T>::new(
94                &kernel,
95            )),
96        );
97
98        schedule.add_system_to_stage(RenderStageLabel::Prepare, resource_system::resource_system);
99        schedule.add_system_to_stage(RenderStageLabel::Queue, upload_system::upload_system);
100        schedule.add_system_to_stage(RenderStageLabel::Queue, queue_system::queue_system);
101
102        schedule.add_system_to_stage(
103            RenderStageLabel::PhaseSort,
104            SystemContainer::new(collision_system::CollisionSystem::new()),
105        );
106    }
107}

122/// The line a label follows, and where its glyphs sit along the text.
123#[derive(Clone)]
124pub struct LineLabel {
125    /// The polyline of the anchor's line part in tile units, shared by its repeated labels.
126    pub polyline: std::sync::Arc<[[f32; 2]]>,
127    /// Arc length from the start of the polyline to the anchor, in tile units.
128    pub anchor_distance: f32,
129    /// Distance of each glyph's centre from the anchor along the text, in layout pixels.
130    pub glyph_offsets: Vec<f32>,
131    /// Size of each glyph that is an image, in layout pixels at the 24-pixel em; zero for a
132    /// text glyph, which collides by the text size.
133    pub image_sizes: Vec<[f32; 2]>,
134    /// Position in the index buffer where the first glyph's triangles begin; each takes six.
135    pub first_glyph_index: usize,
136}

138/// One label of a symbol bucket.
139pub struct Feature {
140    /// Layout bounds for text, RGBA icon and SDF icon, respectively.
141    pub parts: [Option<placement_geometry::SymbolBounds>; 3],
142    /// Source identity and placement priority.
143    pub data: SymbolFeatureData,
144    /// Pixel bounds relative to the anchor at the layout zoom.
145    pub bbox: Box2D<f32, TileSpace>,
146    /// Positions in the bucket's index buffer that draw the label; empty when the layout does
147    /// not attribute quads to labels.
148    pub indices: Range<usize>,
149    /// Where the label is anchored in tile space.
150    pub text_anchor: Point2D<f32, TileSpace>,
151    /// The text of the label.
152    pub str: String,
153    /// The line the label follows, when its glyphs are placed along one.
154    pub line: Option<LineLabel>,
155    /// Shifts of the text, in layout pixels, that the other anchors of `text-variable-anchor`
156    /// would give it relative to the laid-out one; empty for a label with a single anchor.
157    pub anchor_shifts: Vec<[f32; 2]>,
158    /// Index range of the glyphs of each justification when the text justifies differently
159    /// for its variable anchors; empty when it has one layout.
160    pub text_sets: Vec<std::ops::Range<usize>>,
161    /// The entry of `text_sets` each variable anchor shows.
162    pub anchor_sets: Vec<u8>,
163    /// Index ranges of glyphs that a `format` section colours, with the colour.
164    pub text_colors: Vec<(std::ops::Range<usize>, [f32; 4])>,
165    /// Whether the label is the other way of writing the one before it, which shows only when
166    /// that one finds no place.
167    pub fallback: bool,
168    /// What makes the label one of the repeats its layer places along lines: its text and the
169    /// images it shows. `None` for a label placed once.
170    pub repeat_key: Option<std::sync::Arc<str>>,
171}
