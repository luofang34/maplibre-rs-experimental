34/// Failure reported while preparing or delivering GeoJSON tile geometry.
35#[derive(Error, Debug)]
36pub enum ProcessGeoJsonError {
37    /// A worker result could not be delivered through the supplied reply context.
38    #[error("sending data back through context failed")]
39    SendError(SendError),
40    /// Parser detail for invalid GeoJSON input.
41    #[error("GeoJSON parsing failed: {0}")]
42    Parse(Cow<'static, str>),
43}

45/// Reprojects longitude/latitude in degrees into Web Mercator tile coordinates for a processor.
46/// One tile spans 4096 units. Latitude is clamped to the Mercator range; geometry outside
47/// the requested tile is not clipped. Non-finite input coordinates are rejected.
48pub struct ProjectingTessellator<T> {
49    inner: T,
50    tile_x: i32,
51    tile_y: i32,
52    zoom: u8,
53}
54
55impl<T> ProjectingTessellator<T> {
56    /// Takes ownership of a processor and fixes the tile origin and zoom used for projection.
57    pub fn new(coords: WorldTileCoords, inner: T) -> Self {
58        Self {
59            inner,
60            tile_x: coords.x,
61            tile_y: coords.y,
62            zoom: u8::from(coords.z),
63        }
64    }
65
66    /// Convert geographic lon/lat to tile-local extent coordinates (0–4096).
67    fn project(&self, lon: f64, lat: f64) -> (f64, f64) {
68        let lat = lat.clamp(-85.05112877980659, 85.05112877980659);
69        let scale = (1u64 << self.zoom) as f64;
70        let mx = (180.0 + lon) / 360.0;
71        let my = (180.0 - (180.0 / PI * ((PI / 4.0 + lat * PI / 360.0).tan()).ln())) / 360.0;
72        let x = (mx * scale - self.tile_x as f64) * EXTENT;
73        let y = (my * scale - self.tile_y as f64) * EXTENT;
74        (x, y)
75    }
76
77    /// Returns the wrapped processor with any geometry accumulated through this adapter.
78    pub fn into_inner(self) -> T {
79        self.inner
80    }
81}

83impl<T: GeomProcessor> GeomProcessor for ProjectingTessellator<T> {
84    fn xy(&mut self, x: f64, y: f64, idx: usize) -> geozero::error::Result<()> {
85        if !x.is_finite() || !y.is_finite() {
86            return Err(geozero::error::GeozeroError::Geometry(format!(
87                "non-finite GeoJSON coordinate ({x}, {y}) at index {idx}"
88            )));
89        }
90        let (tx, ty) = self.project(x, y);
91        if !tx.is_finite() || !ty.is_finite() {
92            return Err(geozero::error::GeozeroError::Geometry(format!(
93                "GeoJSON coordinate ({x}, {y}) projected to ({tx}, {ty})"
94            )));
95        }
96        self.inner.xy(tx, ty, idx)
97    }

196/// Request for processing GeoJSON features for a set of style layers.
197pub struct GeoJsonTileRequest {
198    /// Target tile origin and zoom for projected geometry and expression evaluation.
199    pub coords: WorldTileCoords,
200    /// Candidate layers, processed only when their source name matches this request.
201    pub layers: Vec<StyleLayer>,
202    /// Name of the GeoJSON source (used to match style layers by `source` field).
203    pub source_name: String,
204    /// Projection controlling tessellation density and antimeridian policy.
205    pub projection: ProjectionType,
206    /// Glyphs and sprites for the layers' symbols; the bundled Latin fallback when absent.
207    pub atlas: Option<std::sync::Arc<crate::sdf::assets::SymbolAtlas>>,
208    /// The zoom the tile is drawn at when that is past the source's last zoom, so its geometry
209    /// is magnified; zero when the tile is drawn at its own zoom.
210    pub overscaled_zoom: u8,
211    /// Pixels of a 512-pixel tile that the geometry extends past the tile's edge, when the source
212    /// sets it; without it the polygons of an extrusion are not clipped to the tile.
213    pub buffer: Option<u32>,
214}
215
216impl GeoJsonTileRequest {
217    /// The zoom style expressions are evaluated at.
218    fn style_zoom(&self) -> f64 {
219        f64::from(self.overscaled_zoom.max(u8::from(self.coords.z)))
220    }
221
222    /// How many times the tile is magnified past its source's last zoom.
223    fn overscaling(&self) -> f64 {
224        let own = u8::from(self.coords.z);
225        2_f64.powi(i32::from(self.overscaled_zoom.saturating_sub(own)))
226    }
227}

229/// Whether one GeoJSON feature passes a layer filter.
230fn feature_passes(feature: &serde_json::Value, filter: &Filter, zoom: f64) -> bool {
231    let properties = properties_from_json(feature.get("properties"));
232    let geometry_type = feature
233        .get("geometry")
234        .and_then(|geometry| geometry.get("type"))
235        .and_then(serde_json::Value::as_str)
236        .map_or(GeometryType::Unknown, GeometryType::from_geojson);
237    filter.evaluate(&FeatureContext {
238        properties: &properties,
239        geometry_type,
240        id: feature.get("id").map(Value::from_json),
241        zoom,
242    })
243}

245/// The GeoJSON a layer sees after its filter: a collection keeps the passing features, a lone
246/// feature or geometry either stays or becomes an empty collection.
247pub fn filter_geojson(
248    geojson: &serde_json::Value,
249    filter: &Filter,
250    zoom: f64,
251) -> serde_json::Value {
252    let empty = serde_json::json!({"type": "FeatureCollection", "features": []});
253    match geojson.get("type").and_then(serde_json::Value::as_str) {
254        Some("FeatureCollection") => {
255            let features: Vec<serde_json::Value> = geojson
256                .get("features")
257                .and_then(serde_json::Value::as_array)
258                .map(|features| {
259                    features
260                        .iter()
261                        .filter(|feature| feature_passes(feature, filter, zoom))
262                        .cloned()
263                        .collect()
264                })
265                .unwrap_or_default();
266            serde_json::json!({"type": "FeatureCollection", "features": features})
267        }
268        Some("Feature") if feature_passes(geojson, filter, zoom) => geojson.clone(),
269        Some("Feature") => empty,
270        Some(_) => {
271            let feature = serde_json::json!({"type": "Feature", "geometry": geojson});
272            if feature_passes(&feature, filter, zoom) {
273                geojson.clone()
274            } else {
275                empty
276            }
277        }
278        None => geojson.clone(),
279    }
280}

282/// Process inline GeoJSON data and tessellate features for each matching style layer.
283///
284/// This mirrors [`crate::vector::process_vector_tile`] but works with geographic
285/// (lon/lat) coordinates rather than pre-projected MVT tile coordinates.
286///
287/// Matching layers with supported paint are filtered and tessellated independently.
288/// Results use the style-layer ID as their source-layer key. A filter or geometry failure
289/// emits a missing-layer result and processing continues; a delivery failure is returned.
290/// The final tile-completion message is sent after all candidate layers have been processed.
291pub fn process_geojson_features<T: VectorTransferables, C: Context>(
292    geojson_value: &serde_json::Value,
293    request: GeoJsonTileRequest,
294    context: &C,
295) -> Result<(), ProcessGeoJsonError> {
296    let coords = request.coords;
297    let unfiltered = geojson_value.to_string();
298    let with_geometry = request
299        .layers
300        .iter()
301        .any(geometry_property::reads_geometry)
302        .then(|| geometry_property::with_geometry_property(geojson_value));
303
304    for style_layer in &request.layers {
305        let matches_source = style_layer
306            .source
307            .as_deref()
308            .is_some_and(|source| source == request.source_name);
309        if !matches_source {
310            continue;
311        }
312
313        let Some(paint) = &style_layer.paint else {
314            log::warn!("GeoJSON style layer {} has no paint", style_layer.id);
315            continue;
316        };
317
318        let (source_value, unfiltered) = match &with_geometry {
319            Some(annotated) if geometry_property::reads_geometry(style_layer) => {
320                (annotated, annotated.to_string())
321            }
322            _ => (geojson_value, unfiltered.clone()),
323        };
324        let json_str = match &style_layer.filter {
325            Some(filter) => match Filter::parse(filter) {
326                Ok(filter) => {
327                    filter_geojson(source_value, &filter, f64::from(u8::from(coords.z))).to_string()
328                }
329                Err(error) => {
330                    tracing::error!(
331                        layer = %style_layer.id,
332                        %error,
333                        "unsupported filter; the layer renders nothing"
334                    );
335                    context
336                        .send_back(T::LayerMissing::build_from(coords, style_layer.id.clone()))
337                        .map_err(ProcessGeoJsonError::SendError)?;
338                    continue;
339                }
340            },
341            None => unfiltered,
342        };
343
344        match paint {
345            LayerPaint::Fill(_)
346            | LayerPaint::FillExtrusion(_)
347            | LayerPaint::Line(_)
348            | LayerPaint::Background(_)
349            | LayerPaint::Circle(_)
350            | LayerPaint::Heatmap(_) => {
351                let zoom = u8::from(coords.z);
352                let style_zoom = request.style_zoom();
353                let granularity = match paint {
354                    LayerPaint::Fill(_) => granularity_for_zoom(128, 2, zoom),
355                    LayerPaint::Line(_) => granularity_for_zoom(512, 0, zoom),
356                    _ => 1,
357                };
358                let use_globe_geometry = request.projection.uses_globe_rendering(f64::from(zoom));
359                let mut tessellator = match paint {
360                    LayerPaint::Circle(circle) => {
361                        ZeroTessellator::<IndexDataType>::default().with_circles(CircleOptions {
362                            // A circle lying on the map follows the globe's curve.
363                            grid: use_globe_geometry
364                                && circle.circle_pitch_alignment
365                                    == crate::style::circle::CirclePitchAlignment::Map,
366                            ..CircleOptions::for_paint(circle, style_zoom)
367                        })
368                    }
369                    LayerPaint::Heatmap(heatmap) => ZeroTessellator::<IndexDataType>::default()
370                        .with_circles(CircleOptions {
371                            grid: use_globe_geometry,
372                            ..CircleOptions::for_heatmap(heatmap, style_zoom)
373                        }),
374                    LayerPaint::FillExtrusion(extrusion) => {
375                        let tessellator = ZeroTessellator::<IndexDataType>::default()
376                            .with_extrusion(ExtrusionOptions {
377                                // Clipping decides where a pattern's walls start; a plain extrusion
378                                // keeps the copies that other tiles draw of it.
379                                clip_buffer: request
380                                    .buffer
381                                    .filter(|_| extrusion.fill_extrusion_pattern.is_some())
382                                    .map(|buffer| {
383                                        buffer as f32 * crate::coords::EXTENT as f32 / 512.0
384                                    }),
385                                ..ExtrusionOptions::for_paint(extrusion)
386                            });
387                        if use_globe_geometry {
388                            // Roofs and the edges of walls follow the sphere as fills and lines do.
389                            let last_tile =
390                                i64::from(crate::coords::ZOOM_BOUNDS[usize::from(zoom)]) - 1;
391                            tessellator.with_globe_subdivision(
392                                granularity_for_zoom(128, 2, zoom),
393                                zoom == 0,
394                                coords.y == 0,
395                                i64::from(coords.y) == last_tile,
396                            )
397                        } else {
398                            tessellator
399                        }
400                    }
401                    _ if use_globe_geometry => {
402                        let last_tile =
403                            i64::from(crate::coords::ZOOM_BOUNDS[usize::from(zoom)]) - 1;
404                        ZeroTessellator::<IndexDataType>::default().with_globe_subdivision(
405                            granularity,
406                            zoom == 0,
407                            coords.y == 0,
408                            i64::from(coords.y) == last_tile,
409                        )
410                    }
411                    _ => ZeroTessellator::<IndexDataType>::default(),
412                }
413                .with_feature_opacity(paint.opacity(), style_zoom);
414                tessellator.sort_key = crate::vector::tessellation::SortKeys::by(
415                    crate::vector::tessellation::sort_key_of(style_layer),
416                );
417                match paint {
418                    LayerPaint::Fill(p) if p.fill_pattern.is_some() => {
419                        tessellator.fallback_color = [1.0; 4];
420                        tessellator.pattern_property =
421                            crate::vector::pattern::per_feature_pattern(paint);
422                    }
423                    LayerPaint::Fill(p) => {
424                        tessellator.style_property = p.fill_color.clone();
425                        tessellator.outline_property = p.outline_color();
426                    }
427                    LayerPaint::FillExtrusion(p) if p.fill_extrusion_pattern.is_some() => {
428                        tessellator.fallback_color = [1.0; 4];
429                        tessellator.pattern_property =
430                            crate::vector::pattern::per_feature_pattern(paint);
431                    }
432                    LayerPaint::FillExtrusion(p) => {
433                        tessellator.style_property = p.fill_extrusion_color.clone()
434                    }
435                    LayerPaint::Circle(p) => tessellator.style_property = p.circle_color.clone(),
436                    LayerPaint::Line(p) => {
437                        tessellator.pattern_property =
438                            crate::vector::pattern::per_feature_pattern(paint);
439                        tessellator.style_property = p.line_color.clone();
440                        tessellator.is_line_layer = true;
441                        tessellator.line_gradient = p.line_gradient.is_some();
442                        tessellator.split_corners = p.line_gradient.is_some()
443                            || p.line_pattern.is_some()
444                            || p.line_dasharray.is_some();
445                        tessellator.line_feature_style =
446                            crate::vector::tessellation::LineFeatureStyle::for_paint(p, style_zoom);
447                        tessellator.sharp_corner_offset =
448                            crate::vector::tessellation::sharp_corner_offset(request.overscaling());
449                        tessellator.stroke =
450                            crate::style::line_stroke::LineStroke::of_layer(style_layer);
451                        tessellator.join_property =
452                            crate::style::line_stroke::LineStroke::join_property(style_layer);
453                    }
454                    LayerPaint::Background(p) => {
455                        tessellator.style_property = p.background_color.clone()
456                    }
457                    _ => {}
458                }
459
460                let mut projecting = ProjectingTessellator::new(coords, tessellator);
461
462                let mut geojson_src = geozero::geojson::GeoJson(json_str.as_str());
463                if let Err(e) = geojson_src.process(&mut projecting) {
464                    log::warn!(
465                        "GeoJSON tessellation for layer {} failed: {e:?}",
466                        style_layer.id
467                    );
468                    context
469                        .send_back(T::LayerMissing::build_from(coords, style_layer.id.clone()))
470                        .map_err(ProcessGeoJsonError::SendError)?;
471                    continue;
472                }
473
474                let mut inner = projecting.into_inner();
475                // For bare GeoJSON geometries (Polygon, LineString, etc. — not a
476                // FeatureCollection), geozero never calls `feature_end`, so
477                // `feature_indices` stays empty while `buffer.indices` is not.
478                // Manually commit the remaining geometry as a single feature.
479                if inner.feature_indices.is_empty() && !inner.buffer.indices.is_empty() {
480                    let _ = inner.feature_end(0);
481                }
482                inner.apply_sort_keys();
483
484                let synthetic_layer = geozero::mvt::tile::Layer {
485                    version: 2,
486                    name: style_layer.id.clone(),
487                    ..Default::default()
488                };
489
490                let sort_keys = inner.sort_key_values();
491                context
492                    .send_back(T::LayerTessellated::build_from(
493                        coords,
494                        inner.buffer.into(),
495                        inner.feature_indices,
496                        inner.feature_colors,
497                        sort_keys,
498                        synthetic_layer,
499                        style_layer.id.clone(),
500                    ))
501                    .map_err(ProcessGeoJsonError::SendError)?;
502            }
503            LayerPaint::Symbol(symbol_paint) => {
504                let zoom = request.style_zoom();
505                let atlas = request
506                    .atlas
507                    .clone()
508                    .unwrap_or_else(crate::sdf::assets::fallback_atlas);
509                let mut tessellator =
510                    TextTessellator::with_assets(symbol_paint.clone(), zoom, atlas.clone());
511                tessellator.overscaling = request.overscaling();
512                tessellator.source_ids = serde_json::from_str(&json_str)
513                    .map(|filtered| index::feature_ids(&filtered))
514                    .unwrap_or_default();
515                let mut projecting = ProjectingTessellator::new(coords, tessellator);
516
517                let mut geojson_src = geozero::geojson::GeoJson(json_str.as_str());
518                if let Err(e) = geojson_src.process(&mut projecting) {
519                    log::warn!(
520                        "GeoJSON text tessellation for layer {} failed: {e:?}",
521                        style_layer.id
522                    );
523                    context
524                        .send_back(T::LayerMissing::build_from(coords, style_layer.id.clone()))
525                        .map_err(ProcessGeoJsonError::SendError)?;
526                    continue;
527                }
528
529                let mut inner = projecting.into_inner();
530                // A bare geometry never reaches feature_end, so its symbol is committed here.
531                if inner.features.is_empty() {
532                    let _ = inner.feature_end(0);
533                }
534                inner.finish();
535
536                let synthetic_layer = geozero::mvt::tile::Layer {
537                    version: 2,
538                    name: style_layer.id.clone(),
539                    ..Default::default()
540                };
541
542                context
543                    .send_back(T::SymbolLayerTessellated::build_from(
544                        coords,
545                        inner.quad_buffer.into(),
546                        inner.features,
547                        Some(atlas),
548                        synthetic_layer,
549                        style_layer.id.clone(),
550                    ))
551                    .map_err(ProcessGeoJsonError::SendError)?;
552            }
553            _ => {
554                log::trace!(
555                    "GeoJSON layer {} has unsupported paint type, skipping",
556                    style_layer.id
557                );
558            }
559        }
560    }
561
562    query_index::send::<T, C>(
563        geojson_value,
564        &unfiltered,
565        (coords, &request.source_name, request.buffer),
566        context,
567    )?;
568    context
569        .send_back(T::TileTessellated::build_from(coords))
570        .map_err(ProcessGeoJsonError::SendError)?;
571
572    Ok(())
573}
