24/// Polls two fallible futures concurrently and resolves when both succeed or either fails.
25pub(crate) async fn try_join2<A, B, E>(
26    fut_a: impl Future<Output = Result<A, E>>,
27    fut_b: impl Future<Output = Result<B, E>>,
28) -> Result<(A, B), E> {
29    let mut fut_a = std::pin::pin!(fut_a);
30    let mut fut_b = std::pin::pin!(fut_b);
31    let mut out_a: Option<A> = None;
32    let mut out_b: Option<B> = None;
33
34    poll_fn(|cx| {
35        if out_a.is_none() {
36            if let Poll::Ready(res) = fut_a.as_mut().poll(cx) {
37                match res {
38                    Ok(val) => out_a = Some(val),
39                    Err(err) => return Poll::Ready(Err(err)),
40                }
41            }
42        }
43        if out_b.is_none() {
44            if let Poll::Ready(res) = fut_b.as_mut().poll(cx) {
45                match res {
46                    Ok(val) => out_b = Some(val),
47                    Err(err) => return Poll::Ready(Err(err)),
48                }
49            }
50        }
51        if out_a.is_some() && out_b.is_some() {
52            if let (Some(a), Some(b)) = (out_a.take(), out_b.take()) {
53                return Poll::Ready(Ok((a, b)));
54            }
55        }
56        Poll::Pending
57    })
58    .await
59}

61async fn poll_all_dyn<T, R, E>(
62    mut pending: Vec<Option<BoxFut<'_, T>>>,
63    mut step: impl FnMut(&mut Vec<Option<R>>, usize, T) -> Result<(), E>,
64) -> Result<Vec<R>, E> {
65    if pending.is_empty() {
66        return Ok(Vec::new());
67    }
68    let mut results: Vec<Option<R>> = (0..pending.len()).map(|_| None).collect();
69    let mut remaining = pending.len();
70
71    poll_fn(|cx| {
72        for (idx, fut_opt) in pending.iter_mut().enumerate() {
73            if let Some(fut) = fut_opt.as_mut() {
74                if let Poll::Ready(val) = fut.as_mut().poll(cx) {
75                    *fut_opt = None;
76                    if let Err(err) = step(&mut results, idx, val) {
77                        return Poll::Ready(Err(err));
78                    }
79                    remaining -= 1;
80                }
81            }
82        }
83        if remaining == 0 {
84            Poll::Ready(Ok(std::mem::take(&mut results)
85                .into_iter()
86                .flatten()
87                .collect()))
88        } else {
89            Poll::Pending
90        }
91    })
92    .await
93}

95/// Polls all fallible futures in `futs` concurrently, preserving input order without inlining
96/// large future state machines into every caller's poll loop.
97pub(crate) async fn try_join_all_concurrent<'a, T, E, F>(futs: Vec<F>) -> Result<Vec<T>, E>
98where
99    F: Future<Output = Result<T, E>> + MaybeSend + 'a,
100{
101    let pending: Vec<Option<BoxFut<'a, Result<T, E>>>> = futs
102        .into_iter()
103        .map(|f| Some(Box::pin(f) as BoxFut<'a, Result<T, E>>))
104        .collect();
105    poll_all_dyn(pending, |results, idx, res| {
106        results[idx] = Some(res?);
107        Ok(())
108    })
109    .await
110}

112/// Polls all infallible futures concurrently, preserving input order without `FuturesOrdered`.
113pub(crate) async fn join_all_concurrent<'a, T, F>(futs: impl IntoIterator<Item = F>) -> Vec<T>
114where
115    F: Future<Output = T> + MaybeSend + 'a,
116{
117    enum Never {}
118    let pending: Vec<Option<BoxFut<'a, T>>> = futs
119        .into_iter()
120        .map(|f| Some(Box::pin(f) as BoxFut<'a, T>))
121        .collect();
122    match poll_all_dyn::<T, T, Never>(pending, |results, idx, val| {
123        results[idx] = Some(val);
124        Ok(())
125    })
126    .await
127    {
128        Ok(values) => values,
129        Err(never) => match never {},
130    }
