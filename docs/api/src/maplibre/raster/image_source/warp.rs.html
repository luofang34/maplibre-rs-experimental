1//! How an image source's picture is warped onto its corners, as GL JS `calculateImageWarp` and
2//! its raster vertex shader warp it.
3//!
4//! The warp maps the unit square of the picture onto the corners: projectively, as the view of a
5//! plane, while the corners plausibly describe one, blending towards a bilinear rubber sheet as
6//! they stop doing so. GL JS draws it as a mesh whose vertices it places with the warp and whose
7//! texture coordinates it interpolates with the warp's homogeneous weight: two triangles where
8//! that is exact, a parallelogram or a purely projective warp, and a grid of cells otherwise.
9//! The corners are first rounded to the integer units of the tile the picture is centred in, as
10//! GL JS rounds them, which decides whether a near-rectangle counts as a parallelogram.
11
12/// Cells per side of the grid a warp that is neither affine nor projective is drawn with.
13const SUBDIVIDED_QUAD_GRANULARITY: usize = 16;
14
15/// The foreshortening up to which a warp stays purely projective.
16const PROJECTIVE_FORESHORTENING: f64 = 4.0;
17
18/// The foreshortening at which a warp is taken to be purely bilinear.
19const BILINEAR_FORESHORTENING: f64 = 512.0;
20
21/// Units per side of a tile, in which GL JS rounds the corners.
22const EXTENT: f64 = 8192.0;
23
24/// One vertex of the warped mesh: its Mercator position and its texture coordinates multiplied by
25/// its homogeneous weight, with the weight.
26pub(super) type Vertex = ([f64; 2], [f64; 3]);
27
28/// The perspective terms and blend of a warp, GL JS's `RasterImageWarp`.
29#[derive(Clone, Copy, Debug, PartialEq)]
30struct Terms {
31    perspective: [f64; 2],
32    blend: f64,
33}
34
35const BILINEAR: Terms = Terms {
36    perspective: [0.0, 0.0],
37    blend: 1.0,
38};
39
40/// The picture's corners, top left, top right, bottom right and bottom left, in Mercator units
41/// rounded as GL JS rounds them to the tile they are centred in.
42fn rounded_corners(corners: [[f64; 2]; 4]) -> [[f64; 2]; 4] {
43    let fold = |axis: usize| {
44        corners.iter().fold((f64::MAX, f64::MIN), |(lo, hi), c| {
45            (lo.min(c[axis]), hi.max(c[axis]))
46        })
47    };
48    let ((west, east), (north, south)) = (fold(0), fold(1));
49    let largest = (east - west).max(south - north);
50    let zoom = (-largest.log2()).floor().max(0.0);
51    let tiles = 2_f64.powf(zoom);
52    let tile = [
53        ((west + east) / 2.0 * tiles).floor(),
54        ((north + south) / 2.0 * tiles).floor(),
55    ];
56    corners.map(|[x, y]| {
57        let units = [
58            ((x * tiles - tile[0]) * EXTENT).round(),
59            ((y * tiles - tile[1]) * EXTENT).round(),
60        ];
61        [
62            (units[0] / EXTENT + tile[0]) / tiles,
63            (units[1] / EXTENT + tile[1]) / tiles,
64        ]
65    })
66}

68fn is_parallelogram([top_left, top_right, bottom_right, bottom_left]: [[f64; 2]; 4]) -> bool {
69    top_left[0] + bottom_right[0] == top_right[0] + bottom_left[0]
70        && top_left[1] + bottom_right[1] == top_right[1] + bottom_left[1]
71}
72
73/// GL JS `calculateImageWarp` with its default `auto` warp.
74fn terms(corners: [[f64; 2]; 4]) -> Terms {
75    if is_parallelogram(corners) {
76        return BILINEAR;
77    }
78    let [top_left, top_right, bottom_right, bottom_left] = corners;
79    let sum = [
80        top_left[0] - top_right[0] + bottom_right[0] - bottom_left[0],
81        top_left[1] - top_right[1] + bottom_right[1] - bottom_left[1],
82    ];
83    let right = [
84        top_right[0] - bottom_right[0],
85        top_right[1] - bottom_right[1],
86    ];
87    let down = [
88        bottom_left[0] - bottom_right[0],
89        bottom_left[1] - bottom_right[1],
90    ];
91    let determinant = right[0] * down[1] - down[0] * right[1];
92    let perspective = [
93        (sum[0] * down[1] - down[0] * sum[1]) / determinant,
94        (right[0] * sum[1] - sum[0] * right[1]) / determinant,
95    ];
96    let denominators = [
97        1.0,
98        1.0 + perspective[0],
99        1.0 + perspective[0] + perspective[1],
100        1.0 + perspective[1],
101    ];
102    let foreshortening = denominators.iter().copied().fold(f64::MIN, f64::max)
103        / denominators.iter().copied().fold(f64::MAX, f64::min);
104    let blend = ((1.0 - PROJECTIVE_FORESHORTENING / foreshortening)
105        / (1.0 - PROJECTIVE_FORESHORTENING / BILINEAR_FORESHORTENING))
106        .max(0.0);
107    // A degenerate determinant reaches here as an infinite or not-a-number spread.
108    if !(1.0..=BILINEAR_FORESHORTENING).contains(&foreshortening) || blend >= 1.0 {
109        return BILINEAR;
110    }
111    Terms { perspective, blend }
112}

114/// The vertex at `[s, t]` of the picture's unit square, as the raster vertex shader places it.
115fn vertex(corners: [[f64; 2]; 4], terms: Terms, [s, t]: [f64; 2]) -> Vertex {
116    let [top_left, top_right, bottom_right, bottom_left] = corners;
117    let mix = |a: f64, b: f64, by: f64| a + (b - a) * by;
118    let bilinear = [0, 1].map(|axis| {
119        mix(
120            mix(top_left[axis], top_right[axis], s),
121            mix(bottom_left[axis], bottom_right[axis], s),
122            t,
123        )
124    });
125    let [px, py] = terms.perspective;
126    let denominator = px * s + py * t + 1.0;
127    let projective = [0, 1].map(|axis| {
128        let across_top = top_right[axis] - top_left[axis] + px * top_right[axis];
129        let down_left = bottom_left[axis] - top_left[axis] + py * bottom_left[axis];
130        (across_top * s + down_left * t + top_left[axis]) / denominator
131    });
132    let position = [0, 1].map(|axis| mix(projective[axis], bilinear[axis], terms.blend));
133    let weight = mix(1.0 / denominator, 1.0, terms.blend);
134    (position, [s * weight, t * weight, weight])
135}

137/// The triangles the picture is drawn with, each split along its cell's top right to bottom
138/// left diagonal as GL JS's tile mesh splits it.
139pub(super) fn triangles(corners: [[f64; 2]; 4]) -> Vec<[Vertex; 3]> {
140    let corners = rounded_corners(corners);
141    let terms = terms(corners);
142    let cells = if terms.blend > 0.0 && !is_parallelogram(corners) {
143        SUBDIVIDED_QUAD_GRANULARITY
144    } else {
145        1
146    };
147    let at = |x: usize, y: usize| {
148        vertex(
149            corners,
150            terms,
151            [x as f64 / cells as f64, y as f64 / cells as f64],
152        )
153    };
154    let mut triangles = Vec::with_capacity(cells * cells * 2);
155    for y in 0..cells {
156        for x in 0..cells {
157            let (v0, v1, v2, v3) = (at(x, y), at(x + 1, y), at(x, y + 1), at(x + 1, y + 1));
158            triangles.push([v0, v2, v1]);
159            triangles.push([v1, v2, v3]);
160        }
161    }
162    triangles
163}
