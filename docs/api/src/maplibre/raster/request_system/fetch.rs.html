25/// Fetches every raster source of the tile at once, so a tile costs its slowest source's
26/// round trip rather than the sum of them, and reports the outcome once all have landed.
27pub fn fetch_raster_apc<K: OffscreenKernel, T: RasterTransferables, C: Context + Clone + Send>(
28    input: Input,
29    context: C,
30    kernel: K,
31) -> AsyncProcedureFuture {
32    Box::pin(async move {
33        let Input {
34            coords,
35            style,
36            attempt,
37            ..
38        } = input;
39        let context = AttemptContext::new(context, attempt);
40        let client = kernel.source_client();
41        let mut fetches: FuturesUnordered<_> = source_layer_groups(&style, TileKind::Raster)
42            .into_iter()
43            .map(|group| fetch_source::<T, _, _>(&client, group, coords, context.clone()))
44            .collect();
45        let mut retry = false;
46        while let Some(fetched) = fetches.next().await {
47            retry |= fetched?;
48        }
49        context
50            .send_back(TileRequestOutcome {
51                coords,
52                kind: RequestKind::Raster,
53                attempt: Some(attempt),
54                disposition: if retry {
55                    RequestDisposition::Retry
56                } else {
57                    RequestDisposition::Complete
58                },
59            })
60            .map_err(ProcedureError::Send)
61    })
62}

64/// Fetches one source of the tile and sends back its pixels, or that it has none; returns
65/// whether a failed fetch is worth retrying.
66async fn fetch_source<T: RasterTransferables, HC: HttpClient, C: Context + Clone + Send>(
67    client: &SourceClient<HC>,
68    group: SourceLayerGroup,
69    coords: WorldTileCoords,
70    context: C,
71) -> Result<bool, ProcedureError> {
72    let mut retry = false;
73    if let SourceType::Image(image) = &group.source {
74        return send_image_tile::<T, _, _>(
75            (client, &image.source),
76            (coords, RasterSourceId::new(group.source_name.clone())),
77            context,
78        )
79        .await;
80    }
81    if group.source.format(&coords).is_some() {
82        use crate::sdf::assets::tiles::CachedTileFailure;
83        match client
84            .assets()
85            .load_raster_tile(client, &group.source, &coords)
86            .await
87        {
88            Ok(rgba) => {
89                let source = RasterSourceId::new(group.source_name.clone());
90                // The shared image is copied into the reply, which is a full pass over
91                // its pixels.
92                let delivered = run_cpu_bound_replying(context.clone(), move |context| {
93                    ProcessRasterContext::<T, _>::new(context).finish_decoded(
94                        &coords,
95                        source,
96                        (*rgba).clone(),
97                    )
98                })
99                .await;
100                match delivered {
101                    Ok(Err(ProcessRasterError::Send(source))) => {
102                        return Err(ProcedureError::Send(source))
103                    }
104                    Ok(_) => {}
105                    Err(error) => {
106                        tracing::warn!(%coords, source = ?group.source_name, %error, "raster tile not delivered");
107                        context
108                            .send_back(T::LayerRasterMissing::build_from(
109                                coords,
110                                RasterSourceId::new(group.source_name.clone()),
111                            ))
112                            .map_err(ProcedureError::Send)?;
113                    }
114                }
115            }
116            Err(CachedTileFailure::Decode(error)) => {
117                tracing::warn!(%coords, source = ?group.source_name, %error, "invalid raster tile");
118                context
119                    .send_back(T::LayerRasterMissing::build_from(
120                        coords,
121                        RasterSourceId::new(group.source_name.clone()),
122                    ))
123                    .map_err(ProcedureError::Send)?;
124            }
125            Err(CachedTileFailure::NotFound) => {
126                tracing::debug!(
127                    %coords,
128                    source = ?group.source_name,
129                    "no raster tile at the source; the layer is empty"
130                );
131                context
132                    .send_back(<T as RasterTransferables>::LayerRasterMissing::build_from(
133                        coords,
134                        RasterSourceId::new(group.source_name.clone()),
135                    ))
136                    .map_err(ProcedureError::Send)?;
137            }
138            Err(CachedTileFailure::Transport { retryable, message }) => {
139                retry |= retryable;
140                if retryable {
141                    tracing::warn!(
142                        %coords,
143                        source = ?group.source_name,
144                        error = %message,
145                        "raster tile fetch failed"
146                    );
147                } else {
148                    tracing::error!(
149                        %coords,
150                        source = ?group.source_name,
151                        error = %message,
152                        "raster tile fetch failed"
153                    );
154                }
155                context
156                    .send_back(<T as RasterTransferables>::LayerRasterMissing::build_from(
157                        coords,
158                        RasterSourceId::new(group.source_name.clone()),
159                    ))
160                    .map_err(ProcedureError::Send)?;
161            }
162        }
163        return Ok(retry);
164    }
165
166    let data = match client.fetch(&coords, &group.source).await {
167        Ok(data) => data,
168        Err(error) => {
169            let retryable = error.is_retryable();
170            retry |= retryable;
171            if error.is_not_found() {
172                tracing::debug!(
173                    %coords,
174                    source = ?group.source_name,
175                    "no raster tile at the source; the layer is empty"
176                );
177            } else if retryable {
178                tracing::warn!(
179                    %coords,
180                    source = ?group.source_name,
181                    error = %error.describe(),
182                    "raster tile fetch failed"
183                );
184            } else {
185                tracing::error!(
186                    %coords,
187                    source = ?group.source_name,
188                    error = %error.describe(),
189                    "raster tile fetch failed"
190                );
191            }
192
193            context
194                .send_back(<T as RasterTransferables>::LayerRasterMissing::build_from(
195                    coords,
196                    RasterSourceId::new(group.source_name.clone()),
197                ))
198                .map_err(ProcedureError::Send)?;
199            return Ok(retry);
200        }
201    };
202    let request = RasterTileRequest {
203        coords,
204        source: RasterSourceId::new(group.source_name.clone()),
205    };
206    let processed = run_cpu_bound_replying(context.clone(), move |context| {
207        let mut process_context = ProcessRasterContext::<T, _>::new(context);
208        process_raster_tile(&data, request, &mut process_context)
209    })
210    .await
211    .map_err(ProcessRasterError::CpuWork)
212    .and_then(|processed| processed);
213    match processed {
214        Ok(()) => {}
215        Err(ProcessRasterError::Send(source)) => return Err(ProcedureError::Send(source)),
216        Err(error @ (ProcessRasterError::Decoding { .. } | ProcessRasterError::CpuWork(_))) => {
217            tracing::warn!(%coords, source = ?group.source_name, %error, "invalid raster tile");
218            context
219                .send_back(T::LayerRasterMissing::build_from(
220                    coords,
221                    RasterSourceId::new(group.source_name.clone()),
222                ))
223                .map_err(ProcedureError::Send)?;
224        }
225    }
226    Ok(retry)
227}

229/// Sends the tile of an image source back, empty where the picture does not reach it, and
230/// returns whether a failed fetch is worth retrying.
231async fn send_image_tile<
232    T: RasterTransferables,
233    HC: crate::io::source_client::HttpClient,
234    C: Context,
235>(
236    (client, image): (
237        &crate::io::source_client::SourceClient<HC>,
238        &crate::style::source::ImageSource,
239    ),
240    (coords, source): (WorldTileCoords, RasterSourceId),
241    context: C,
242) -> Result<bool, ProcedureError> {
243    use crate::raster::transferables::LayerRaster;
244    match crate::raster::image_source::load_tile(client, image, coords).await {
245        Ok(tile) => {
246            // A tile the picture does not reach has loaded too; it is simply empty.
247            let tile = tile.unwrap_or_else(|| image::RgbaImage::new(1, 1));
248            context
249                .send_back(T::LayerRaster::build_from(coords, source, tile))
250                .map_err(ProcedureError::Send)?;
251            Ok(false)
252        }
253        Err(failure) => {
254            context
255                .send_back(T::LayerRasterMissing::build_from(coords, source))
256                .map_err(ProcedureError::Send)?;
257            Ok(failure.is_retryable())
258        }
259    }
