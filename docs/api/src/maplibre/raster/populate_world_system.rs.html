42            phantom_t: Default::default(),

47impl<E: Environment, T: RasterTransferables> System for PopulateWorldSystem<E, T> {
48    fn name(&self) -> Cow<'static, str> {
49        "populate_world_system".into()
50    }
51
52    fn run(
53        &mut self,
54        MapContext {
55            style,
56            view_state,
57            world,
58            ..
59        }: &mut MapContext,
60    ) -> SystemResult {
61        let circumference = view_state.body().circumference_meters();
62        let messages = self.kernel.apc().receive(|message| {
63            message.has_tag(RequestKind::Raster.message_tag())
64                || message.has_tag(T::LayerRaster::message_tag())
65                || message.has_tag(T::LayerRasterMissing::message_tag())
66        });
67        apply_worker_messages(messages, |message| {
68            apply_raster_message_with_style::<T>(world, Some(style), circumference, message)
69        })?;
70
71        Ok(())
72    }
73}

83        None,

85        message,
86    )

89/// Records a worker's raster or `raster-dem` result on its tile. `raster-dem` sources are stored
90/// once as 2-texel bordered [`LoadedDem`] tiles in [`DemLayersComponent`](crate::terrain::DemLayersComponent)
91/// and mirrored to 3D terrain when shared.
92pub(crate) fn apply_raster_message_with_style<T: RasterTransferables>(
93    world: &mut World,
94    style: Option<&Style>,
95    circumference_meters: f64,
96    message: Message,
97) -> Result<(), MessageError> {
98    if message.has_tag(RequestKind::Raster.message_tag()) {
99        tile_retry::completed(world, *message.into_transferable::<TileRequestOutcome>()?);
100        return Ok(());
101    }
102    let attempt = message.attempt();
103    let (coords, layer) = if message.has_tag(T::LayerRaster::message_tag()) {
104        let message = message.into_transferable::<T::LayerRaster>()?;
105        let coords = message.coords();
106        if !tile_retry::accepts(world, coords, RequestKind::Raster, attempt)
107            || world
108                .tiles
109                .query::<&RasterLayersDataComponent>(coords)
110                .is_none()
111        {
112            return Ok(());
113        }
114        let payload = message.into_dem_or_layer();
115        let source = match &payload {
116            Ok(shared) => shared.source.clone(),
117            Err(layer) => layer.source.clone(),
118        };
119        if let Some((style, unpack)) =
120            style.and_then(|s| dem_unpack_for_source(s, &source).map(|u| (s, u)))
121        {
122            let loaded = match payload {
123                Ok(shared)
124                    if shared.tile.unpack() == unpack
125                        && (shared.circumference_meters - circumference_meters).abs() <= 1e-3 =>
126                {
127                    Some(LoadedDem::from_shared(shared.tile))
128                }
129                Ok(shared) => {
130                    decode_image_to_dem(coords, &shared.fallback, unpack, circumference_meters)
131                }
132                Err(layer) => {
133                    decode_image_to_dem(coords, &layer.image, unpack, circumference_meters)
134                }
135            };
136            match loaded {
137                Some(dem) => {
138                    ingest_source_dem(
139                        &mut world.tiles,
140                        &source,
141                        coords,
142                        dem,
143                        is_terrain_source(style, &source),
144                    );
145                    (
146                        coords,
147                        RasterLayerData::Available(AvailableRasterLayerData {
148                            coords,
149                            source,
150                            image: RgbaImage::new(0, 0),
151                        }),
152                    )
153                }
154                None => (
155                    coords,
156                    RasterLayerData::Missing(MissingRasterLayerData { coords, source }),
157                ),
158            }
159        } else {
160            let available = match payload {
161                Err(layer) => layer,
162                Ok(shared) => AvailableRasterLayerData {
163                    coords: shared.coords,
164                    source: shared.source,
165                    image: Arc::try_unwrap(shared.fallback).unwrap_or_else(|arc| (*arc).clone()),
166                },
167            };
168            (coords, RasterLayerData::Available(available))
169        }
170    } else if message.has_tag(T::LayerRasterMissing::message_tag()) {
171        let message = message.into_transferable::<T::LayerRasterMissing>()?;
172        let coords = message.coords();
173        if !tile_retry::accepts(world, coords, RequestKind::Raster, attempt) {
174            return Ok(());
175        }
176        (coords, RasterLayerData::Missing(message.to_layer()))
177    } else {
178        return Ok(());
179    };
180
181    let source = layer.source().clone();
182    let Some(component) = world
183        .tiles
184        .query_mut::<&mut RasterLayersDataComponent>(coords)
185    else {
186        return Ok(());
187    };
188    if component.record(layer) {
189        if let Some(Eventually::Initialized(raster)) =
190            world.resources.get_mut::<Eventually<RasterResources>>()
191        {
192            raster.tile_data_changed(&source, coords);
193        }
194    }
195    Ok(())
196}
