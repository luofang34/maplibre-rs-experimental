1//! Resamples an `image` source into the tiles that cover it.
2//!
3//! Resampling the picture into raster tiles lets terrain, globe and masks treat it like any other
4//! raster source. The `warp` submodule warps the picture onto its corners as GL JS's default
5//! `auto` warp does; the `perspective` and `flat` warps a GL JS map can be set to are not offered.

13/// Pixels per side of a resampled tile.
14pub const IMAGE_TILE_SIZE: u32 = 512;

16fn mercator([longitude, latitude]: [f64; 2]) -> [f64; 2] {
17    let x = (longitude + 180.0) / 360.0;
18    let y = 0.5
19        - (std::f64::consts::FRAC_PI_4 + latitude.to_radians() / 2.0)
20            .tan()
21            .ln()
22            / std::f64::consts::TAU;
23    [x, y]
24}

26fn sample(image: &RgbaImage, u: f64, v: f64) -> [f32; 4] {
27    let (width, height) = image.dimensions();
28    let x = (u * f64::from(width) - 0.5).clamp(0.0, f64::from(width - 1));
29    let y = (v * f64::from(height) - 0.5).clamp(0.0, f64::from(height - 1));
30    let (x0, y0) = (x.floor() as u32, y.floor() as u32);
31    let (x1, y1) = ((x0 + 1).min(width - 1), (y0 + 1).min(height - 1));
32    let (fx, fy) = ((x - f64::from(x0)) as f32, (y - f64::from(y0)) as f32);
33    let mut premultiplied = [0.0_f32; 4];
34    for ((px, py), weight) in [
35        ((x0, y0), (1.0 - fx) * (1.0 - fy)),
36        ((x1, y0), fx * (1.0 - fy)),
37        ((x0, y1), (1.0 - fx) * fy),
38        ((x1, y1), fx * fy),
39    ] {
40        let [r, g, b, a] = image
41            .get_pixel(px, py)
42            .0
43            .map(|channel| f32::from(channel) / 255.0);
44        for (out, value) in premultiplied.iter_mut().zip([r * a, g * a, b * a, a]) {
45            *out += value * weight;
46        }
47    }
48    premultiplied
49}

51/// Draws one triangle of the warped picture into the tile: each pixel whose centre it covers
52/// takes the picture at the texture coordinates interpolated across it and divided by the
53/// interpolated weight, as GPU interpolation of GL JS's `v_pos0` gives them. A pixel an earlier
54/// triangle or world copy already drew keeps its colour.
55fn fill(
56    tile: &mut RgbaImage,
57    drawn: &mut [bool],
58    image: &RgbaImage,
59    [(a, ta), (b, tb), (c, tc)]: [([f64; 2], [f64; 3]); 3],
60) {
61    let determinant = (b[0] - a[0]) * (c[1] - a[1]) - (c[0] - a[0]) * (b[1] - a[1]);
62    if determinant == 0.0 || !determinant.is_finite() {
63        return;
64    }
65    let size = f64::from(IMAGE_TILE_SIZE);
66    let lo = |values: [f64; 3]| values.into_iter().fold(f64::MAX, f64::min).floor().max(0.0);
67    let hi = |values: [f64; 3]| values.into_iter().fold(f64::MIN, f64::max).ceil().min(size);
68    let (x0, x1) = (lo([a[0], b[0], c[0]]), hi([a[0], b[0], c[0]]));
69    let (y0, y1) = (lo([a[1], b[1], c[1]]), hi([a[1], b[1], c[1]]));
70    for py in y0 as u32..y1 as u32 {
71        for px in x0 as u32..x1 as u32 {
72            let index = (py * IMAGE_TILE_SIZE + px) as usize;
73            if drawn[index] {
74                continue;
75            }
76            let (x, y) = (f64::from(px) + 0.5, f64::from(py) + 0.5);
77            let l1 = ((x - a[0]) * (c[1] - a[1]) - (c[0] - a[0]) * (y - a[1])) / determinant;
78            let l2 = ((b[0] - a[0]) * (y - a[1]) - (x - a[0]) * (b[1] - a[1])) / determinant;
79            let l0 = 1.0 - l1 - l2;
80            if l0 < 0.0 || l1 < 0.0 || l2 < 0.0 {
81                continue;
82            }
83            let texture = [0, 1, 2].map(|i| l0 * ta[i] + l1 * tb[i] + l2 * tc[i]);
84            let [r, g, blue, alpha] =
85                sample(image, texture[0] / texture[2], texture[1] / texture[2]);
86            if alpha > 0.0 {
87                tile.put_pixel(
88                    px,
89                    py,
90                    image::Rgba(
91                        [r / alpha, g / alpha, blue / alpha, alpha]
92                            .map(|c| (c * 255.0).round() as u8),
93                    ),
94                );
95                drawn[index] = true;
96            }
97        }
98    }
99}

101/// Copies of the world an image can wrap onto before the rest are ignored.
102const MAX_WORLD_SHIFTS: usize = 4;
103
104/// The part of the image inside `tile`, or `None` when the image does not touch it.
105pub fn render_tile(
106    image: &RgbaImage,
107    coordinates: [[f64; 2]; 4],
108    tile: WorldTileCoords,
109) -> Option<RgbaImage> {
110    let tile_count = 2_f64.powi(i32::from(u8::from(tile.z)));
111    let corners = coordinates.map(mercator);
112    let (west, east) = corners.iter().fold((f64::MAX, f64::MIN), |(lo, hi), c| {
113        (lo.min(c[0]), hi.max(c[0]))
114    });
115    let (north, south) = corners.iter().fold((f64::MAX, f64::MIN), |(lo, hi), c| {
116        (lo.min(c[1]), hi.max(c[1]))
117    });
118    let (left, top) = (
119        f64::from(tile.x) / tile_count,
120        f64::from(tile.y) / tile_count,
121    );
122    let size = 1.0 / tile_count;
123    // The image may reach past the antimeridian, so the tile also shows it shifted by whole worlds.
124    let shifts: Vec<f64> = ((west - left - size).floor() as i64..=(east - left).ceil() as i64)
125        .map(|shift| shift as f64)
126        .filter(|shift| {
127            let (lo, hi) = (left + shift, left + shift + size);
128            !(east < lo || west > hi)
129        })
130        .take(MAX_WORLD_SHIFTS)
131        .collect();
132    if shifts.is_empty() || south < top || north > top + size {
133        return None;
134    }
135    let mut tile_image = RgbaImage::new(IMAGE_TILE_SIZE, IMAGE_TILE_SIZE);
136    let mut drawn = vec![false; (IMAGE_TILE_SIZE * IMAGE_TILE_SIZE) as usize];
137    let triangles = warp::triangles(corners);
138    // Mercator units to tile pixels, for the copy of the world `shift` whole worlds east.
139    let pixels = f64::from(IMAGE_TILE_SIZE) / size;
140    for shift in &shifts {
141        let to_tile = |[x, y]: [f64; 2]| [(x - left - shift) * pixels, (y - top) * pixels];
142        for triangle in &triangles {
143            let corners = triangle.map(|(position, texture)| (to_tile(position), texture));
144            fill(&mut tile_image, &mut drawn, image, corners);
145        }
146    }
147    Some(tile_image)
148}

150/// Why an image source's picture cannot be drawn.
151#[derive(Debug, thiserror::Error)]
152pub enum PictureError {
153    /// The bytes are not a picture the decoder reads.
154    #[error("the picture cannot be decoded")]
155    Decode(#[source] image::ImageError),
156    /// The picture's embedded colour profile cannot be converted to sRGB.
157    #[error("the picture's colour profile cannot be converted to sRGB")]
158    Profile(#[source] moxcms::CmsError),
159}
160
161/// Decodes an image source's picture into sRGB, converting it from the colour profile it embeds,
162/// as GL JS shows it once the browser has decoded it.
163pub fn decode(bytes: &[u8]) -> Result<RgbaImage, PictureError> {
164    use image::ImageDecoder;
165    let mut decoder = image::ImageReader::new(std::io::Cursor::new(bytes))
166        .with_guessed_format()
167        .map_err(|error| PictureError::Decode(image::ImageError::IoError(error)))?
168        .into_decoder()
169        .map_err(PictureError::Decode)?;
170    let profile = decoder.icc_profile().map_err(PictureError::Decode)?;
171    let picture = image::DynamicImage::from_decoder(decoder)
172        .map_err(PictureError::Decode)?
173        .to_rgba8();
174    // A profile the converter does not understand leaves the stored values as they are.
175    let Some(Ok(source)) = profile.map(|bytes| moxcms::ColorProfile::new_from_slice(&bytes)) else {
176        return Ok(picture);
177    };
178    let transform = source
179        .create_transform_8bit(
180            moxcms::Layout::Rgba,
181            &moxcms::ColorProfile::new_srgb(),
182            moxcms::Layout::Rgba,
183            moxcms::TransformOptions::default(),
184        )
185        .map_err(PictureError::Profile)?;
186    let (width, height) = picture.dimensions();
187    let mut converted = vec![0_u8; picture.as_raw().len()];
188    transform
189        .transform(picture.as_raw(), &mut converted)
190        .map_err(PictureError::Profile)?;
191    Ok(RgbaImage::from_raw(width, height, converted).unwrap_or(picture))
192}

194/// Makes an image source's tiles again after its picture or corners changed. A tile the
195/// picture still reaches keeps showing the old one until the new arrives; a tile it no longer
196/// reaches drops it now.
197pub(crate) fn reload(
198    world: &mut crate::tcs::world::World,
199    style: &crate::style::Style,
200    name: &str,
201) {
202    use crate::{
203        io::tile_retry::{self, RequestKind},
204        raster::{RasterLayersDataComponent, RasterSourceId},
205        render::eventually::Eventually,
206        style::source::Source,
207    };
208    let Some(Source::Image(image)) = style.sources.get(name) else {
209        return;
210    };
211    let source = RasterSourceId::from(name);
212    let coords: Vec<WorldTileCoords> = world.tiles.tiles.values().map(|tile| tile.coords).collect();
213    for coords in coords {
214        if render_tile_bounds_reach(image.coordinates, coords) {
215            continue;
216        }
217        if let Some(component) = world
218            .tiles
219            .query_mut::<&mut RasterLayersDataComponent>(coords)
220        {
221            component.layers.retain(|layer| layer.source() != &source);
222        }
223        if let Some(Eventually::Initialized(raster)) = world
224            .resources
225            .get_mut::<Eventually<crate::raster::resource::RasterResources>>()
226        {
227            raster.remove_source_texture(&source, coords);
228        }
229    }
230    tile_retry::refresh(world, RequestKind::Raster);
231}

233/// Whether the picture stretched over `coordinates` can reach the tile: its corners' bounding box
234/// overlaps the tile.
235fn render_tile_bounds_reach(coordinates: [[f64; 2]; 4], tile: WorldTileCoords) -> bool {
236    let tile_count = 2_f64.powi(i32::from(u8::from(tile.z)));
237    let corners = coordinates.map(mercator);
238    let fold = |axis: usize| {
239        corners.iter().fold((f64::MAX, f64::MIN), |(lo, hi), c| {
240            (lo.min(c[axis]), hi.max(c[axis]))
241        })
242    };
243    let ((west, east), (north, south)) = (fold(0), fold(1));
244    let (left, top) = (
245        f64::from(tile.x) / tile_count,
246        f64::from(tile.y) / tile_count,
247    );
248    let size = 1.0 / tile_count;
249    // A picture past the antimeridian shows in copies of the world, which this does not follow.
250    let wraps = west < 0.0 || east > 1.0;
251    wraps || !(east < left || west > left + size || south < top || north > top + size)
252}

254/// The tile `coords` of an image source, resampled from its picture, which is fetched and
255/// decoded once through the client's shared asset cache. `Ok(None)` when the picture does not
256/// reach the tile.
257pub(crate) async fn load_tile<HC: crate::io::source_client::HttpClient>(
258    client: &crate::io::source_client::SourceClient<HC>,
259    source: &crate::style::source::ImageSource,
260    coords: WorldTileCoords,
261) -> Result<Option<RgbaImage>, crate::sdf::assets::AssetFailure> {
262    use crate::sdf::assets::AssetFailure;
263    let url = source.url.as_str();
264    let picture = client
265        .assets()
266        .load(
267            format!("image-source:{url}#{}", source.generation),
268            || async {
269                let bytes = crate::sdf::assets::fetch(client, url, "image source").await?;
270                let picture = decode(&bytes).map_err(|error| {
271                    tracing::warn!(%url, %error, "invalid image source picture");
272                    AssetFailure::Terminal(format!("invalid image {url}: {error}"))
273                })?;
274                let bytes = picture.as_raw().len();
275                Ok((picture, bytes))
276            },
277        )
278        .await?;
279    Ok(render_tile(&picture, source.coordinates, coords))
280}
