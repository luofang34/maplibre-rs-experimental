39impl<E: Environment, T: RasterTransferables> System for RequestSystem<E, T> {
40    fn name(&self) -> Cow<'static, str> {
41        "raster_request".into()
42    }
43
44    fn run(
45        &mut self,
46        MapContext {
47            style,
48            view_state,
49            world,
50            ..
51        }: &mut MapContext,
52    ) -> SystemResult {
53        tile_retry::stop_cancelled(world, self.kernel.apc());
54        // Missing ancestors and deferred tiles must progress while the camera is stationary.
55        // Each raster source covers the view at its own tile size and rounding, as GL JS's
56        // per-source tile managers do; the tiles of every source are requested together.
57        let multi = world
58            .resources
59            .get::<crate::render::eye_covering::MultiViewExtractCovering>()
60            .filter(|m| m.active)
61            .cloned();
62        let regions = if let Some(multi) = &multi {
63            multi.raster_regions.clone()
64        } else {
65            let computed =
66                raster_source_regions(style, view_state, world, ViewStatePadding::Loose)?;
67            let host_frame = crate::render::frame_signals::host_frame_in_progress(world);
68            let flat: Vec<WorldTileCoords> = computed
69                .iter()
70                .flat_map(|(_, tiles)| tiles.iter().copied())
71                .collect();
72            let cache = world
73                .resources
74                .get_or_init_mut::<crate::render::eye_covering::FrameLooseCovering>();
75            if cache.host_frame != host_frame {
76                *cache = crate::render::eye_covering::FrameLooseCovering {
77                    host_frame,
78                    view_tiles: None,
79                    raster_tiles: None,
80                };
81            }
82            cache.raster_tiles = Some(flat);
83            computed
84        };
85        let mut requested = FxHashSet::default();
86        let mut budget = request_budget(world);
87        let wanted = wanted_tiles(regions, style, world);
88        let sources: Vec<_> = source_layer_groups(style, TileKind::Raster)
89            .into_iter()
90            .map(|group| RasterSourceId::new(group.source_name))
91            .collect();
92        let mut frame_style = FrameStyle::new(style);
93        for coords in wanted {
94            if !requested.insert(coords) {
95                continue;
96            }
97
98            let retry_due = tile_retry::due(world, coords, RequestKind::Raster);
99            if tile_retry::waiting(world, coords, RequestKind::Raster) && !retry_due {
100                continue;
101            }
102            if !retry_due
103                && world
104                    .tiles
105                    .query::<&RasterLayersDataComponent>(coords)
106                    .is_some_and(|component| {
107                        sources
108                            .iter()
109                            .all(|source| component.has_source_result(source))
110                    })
111            {
112                continue;
113            }
114            // The rest wait for a later frame, once tiles in flight have landed.
115            if budget == 0 {
116                continue;
117            }
118            budget -= 1;
119
120            // One rejected tile must not cost the frame its other tiles.
121            if let Err(error) = self.request(coords, &mut frame_style, world) {
122                tracing::warn!(%error, "tile request rejected; it is requested again after a backoff");
123                tile_retry::rejected(world, coords, RequestKind::Raster, None);
124            }
125        }
126        if let Some(multi) = &multi {
127            requested.extend(multi.skipped_wanted_raster.iter().copied());
128        }
129        tile_retry::want(world, RequestKind::Raster, &requested);
130
131        Ok(())
132    }
133}

135impl<E: Environment, T: RasterTransferables> RequestSystem<E, T> {
136    fn request(
137        &self,
138        coords: crate::coords::WorldTileCoords,
139        style: &mut FrameStyle,
140        world: &mut crate::tcs::world::World,
141    ) -> SystemResult {
142        let exists = world
143            .tiles
144            .query::<&RasterLayersDataComponent>(coords)
145            .is_some();
146        let attempt = tile_retry::next_attempt(world);
147        let Some(mut tile) = world.tiles.spawn_mut(coords) else {
148            return Err(SystemError::InvalidTile { coords });
149        };
150        self.kernel
151            .apc()
152            .call(
153                Input::new(coords, style.share(Style::clone), attempt),
154                fetch_raster_apc::<E::OffscreenKernelEnvironment, T, _>,
155            )
156            .map_err(|source| SystemError::TileRequest {
157                kind: "raster",
158                coords,
159                source,
160            })?;
161        if !exists {
162            tile.insert(RasterLayersDataComponent::default());
163        }
164        tile_retry::started(world, coords, RequestKind::Raster, attempt, None);
165        tracing::debug!(%coords, "raster tile request accepted");
166        Ok(())
167    }
168}

170fn coarse_raster_ancestor(coords: WorldTileCoords, minzoom: u8) -> Option<WorldTileCoords> {
171    let target = u8::from(coords.z).saturating_sub(3).max(minzoom);
172    if target >= u8::from(coords.z) {
173        return None;
174    }
175    let mut current = coords;
176    while u8::from(current.z) > target {
177        current = current.get_parent()?;
178    }
179    Some(current)
180}

182fn wanted_tiles(
183    regions: Vec<(RasterSourceId, Vec<WorldTileCoords>)>,
184    style: &crate::style::Style,
185    world: &crate::tcs::world::World,
186) -> Vec<WorldTileCoords> {
187    let mut wanted = Vec::new();
188    let mut coarse_ancestors = Vec::new();
189    let mut seen_coarse = FxHashSet::default();
190    let is_terrain = style.terrain.is_some();
191    for (source, tiles) in regions {
192        let minzoom = match source.name().and_then(|name| style.sources.get(name)) {
193            Some(crate::style::source::Source::Raster(source)) => source.minzoom,
194            Some(crate::style::source::Source::RasterDem(source)) => source.minzoom,
195            _ => None,
196        }
197        .unwrap_or(0);
198        let bounds = match source.name().and_then(|name| style.sources.get(name)) {
199            Some(crate::style::source::Source::Raster(source)) => source.bounds,
200            Some(crate::style::source::Source::Image(image)) => image_bounds(image),
201            _ => None,
202        };
203        for coords in tiles {
204            // Tiles outside the source's declared bounds are never requested.
205            if bounds.is_some_and(|bounds| !crate::io::tile_sources::tile_in_bounds(coords, bounds))
206            {
207                continue;
208            }
209            if is_terrain {
210                if let Some(ancestor) = coarse_raster_ancestor(coords, minzoom) {
211                    if seen_coarse.insert(ancestor) {
212                        coarse_ancestors.push(ancestor);
213                    }
214                }
215            }
216            wanted.push(coords);
217            if let Some(parent) = missing_tile_fallback(coords, minzoom, |coords| {
218                world
219                    .tiles
220                    .query::<&RasterLayersDataComponent>(coords)
221                    .is_some_and(|component| component.source_is_missing(&source))
222            }) {
223                wanted.push(parent);
224            }
225        }
226    }
227    if !coarse_ancestors.is_empty() {
228        // Bracket the local terrain with one coarse raster overview tile first, then fine
229        // tiles, then any remaining coarse ancestors, matching DEM `request_order`.
230        let remaining = coarse_ancestors.split_off(coarse_ancestors.len().min(2));
231        coarse_ancestors.extend(wanted);
232        coarse_ancestors.extend(remaining);
233        return coarse_ancestors;
234    }
235    wanted
236}

238/// The bounds `(west, south, east, north)` of an image's corners, so tiles it cannot reach are
239/// not requested; none for a picture that reaches past the antimeridian, whose copies across it
240/// a tile may show.
241fn image_bounds(image: &crate::style::source::ImageSource) -> Option<(f64, f64, f64, f64)> {
242    let longitudes = image.coordinates.map(|[longitude, _]| longitude);
243    let latitudes = image.coordinates.map(|[_, latitude]| latitude);
244    let fold = |values: [f64; 4]| {
245        values
246            .into_iter()
247            .fold((f64::MAX, f64::MIN), |(lo, hi), value| {
248                (lo.min(value), hi.max(value))
249            })
250    };
251    let ((west, east), (south, north)) = (fold(longitudes), fold(latitudes));
252    (west >= -180.0 && east <= 180.0).then_some((west, south, east, north))
253}
