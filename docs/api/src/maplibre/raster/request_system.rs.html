50impl<E: Environment, T: RasterTransferables> System for RequestSystem<E, T> {
51    fn name(&self) -> Cow<'static, str> {
52        "raster_request".into()
53    }
54
55    fn run(
56        &mut self,
57        MapContext {
58            style,
59            view_state,
60            world,
61            ..
62        }: &mut MapContext,
63    ) -> SystemResult {
64        tile_retry::stop_cancelled(world, self.kernel.apc());
65        // Missing ancestors and deferred tiles must progress while the camera is stationary.
66        // Each raster source covers the view at its own tile size and rounding, as GL JS's
67        // per-source tile managers do; the tiles of every source are requested together.
68        let multi = world
69            .resources
70            .get::<crate::render::eye_covering::MultiViewExtractCovering>()
71            .filter(|m| m.active)
72            .cloned();
73        let regions = if let Some(multi) = &multi {
74            multi.raster_regions.clone()
75        } else {
76            let computed =
77                raster_source_regions(style, view_state, world, ViewStatePadding::Loose)?;
78            let host_frame = crate::render::frame_signals::host_frame_in_progress(world);
79            let flat: Vec<WorldTileCoords> = computed
80                .iter()
81                .flat_map(|(_, tiles)| tiles.iter().copied())
82                .collect();
83            let cache = world
84                .resources
85                .get_or_init_mut::<crate::render::eye_covering::FrameLooseCovering>();
86            if cache.host_frame != host_frame {
87                *cache = crate::render::eye_covering::FrameLooseCovering {
88                    host_frame,
89                    view_tiles: None,
90                    raster_tiles: None,
91                };
92            }
93            cache.raster_tiles = Some(flat);
94            computed
95        };
96        let mut requested = FxHashSet::default();
97        let mut budget = request_budget(world);
98        let wanted = wanted_tiles(regions, style, world);
99        let sources: Vec<_> = source_layer_groups(style, TileKind::Raster)
100            .into_iter()
101            .map(|group| RasterSourceId::new(group.source_name))
102            .collect();
103        let mut frame_style = FrameStyle::new(style);
104        for coords in wanted {
105            if !requested.insert(coords) {
106                continue;
107            }
108
109            let retry_due = tile_retry::due(world, coords, RequestKind::Raster);
110            if tile_retry::waiting(world, coords, RequestKind::Raster) && !retry_due {
111                continue;
112            }
113            if !retry_due
114                && world
115                    .tiles
116                    .query::<&RasterLayersDataComponent>(coords)
117                    .is_some_and(|component| {
118                        sources
119                            .iter()
120                            .all(|source| component.has_source_result(source))
121                    })
122            {
123                continue;
124            }
125            // The rest wait for a later frame, once tiles in flight have landed.
126            if budget == 0 {
127                continue;
128            }
129            budget -= 1;
130
131            // One rejected tile must not cost the frame its other tiles.
132            if let Err(error) = self.request(coords, &mut frame_style, world) {
133                tracing::warn!(%error, "tile request rejected; a later frame requests the tile again");
134                crate::render::frame_signals::keep_animating(world);
135            }
136        }
137        if let Some(multi) = &multi {
138            requested.extend(multi.skipped_wanted_raster.iter().copied());
139        }
140        tile_retry::want(world, RequestKind::Raster, &requested);
141
142        Ok(())
143    }
144}
145pub fn fetch_raster_apc<K: OffscreenKernel, T: RasterTransferables, C: Context + Clone + Send>(
146    input: Input,
147    context: C,
148    kernel: K,
149) -> AsyncProcedureFuture {
150    Box::pin(async move {
151        let Input {
152            coords,
153            style,
154            attempt,
155            ..
156        } = input;
157        let context = AttemptContext::new(context, attempt);
158
159        let client = kernel.source_client();
160        let mut retry = false;
161
162        for group in source_layer_groups(&style, TileKind::Raster) {
163            let context = context.clone();
164            if let SourceType::Image(image) = &group.source {
165                retry |= send_image_tile::<T, _, _>(
166                    (&client, &image.source),
167                    (coords, RasterSourceId::new(group.source_name.clone())),
168                    context,
169                )
170                .await?;
171                continue;
172            }
173            if group.source.format(&coords).is_some() {
174                use crate::sdf::assets::tiles::CachedTileFailure;
175                match client
176                    .assets()
177                    .load_raster_tile(&client, &group.source, &coords)
178                    .await
179                {
180                    Ok(rgba) => {
181                        let source = RasterSourceId::new(group.source_name.clone());
182                        // The shared image is copied into the reply, which is a full pass over
183                        // its pixels.
184                        let delivered = run_cpu_bound_replying(context.clone(), move |context| {
185                            ProcessRasterContext::<T, _>::new(context).finish_decoded(
186                                &coords,
187                                source,
188                                (*rgba).clone(),
189                            )
190                        })
191                        .await;
192                        match delivered {
193                            Ok(Err(ProcessRasterError::Send(source))) => {
194                                return Err(ProcedureError::Send(source))
195                            }
196                            Ok(_) => {}
197                            Err(error) => {
198                                tracing::warn!(%coords, source = ?group.source_name, %error, "raster tile not delivered");
199                                context
200                                    .send_back(T::LayerRasterMissing::build_from(
201                                        coords,
202                                        RasterSourceId::new(group.source_name.clone()),
203                                    ))
204                                    .map_err(ProcedureError::Send)?;
205                            }
206                        }
207                    }
208                    Err(CachedTileFailure::Decode(error)) => {
209                        tracing::warn!(%coords, source = ?group.source_name, %error, "invalid raster tile");
210                        context
211                            .send_back(T::LayerRasterMissing::build_from(
212                                coords,
213                                RasterSourceId::new(group.source_name.clone()),
214                            ))
215                            .map_err(ProcedureError::Send)?;
216                    }
217                    Err(CachedTileFailure::NotFound) => {
218                        tracing::debug!(
219                            %coords,
220                            source = ?group.source_name,
221                            "no raster tile at the source; the layer is empty"
222                        );
223                        context
224                            .send_back(<T as RasterTransferables>::LayerRasterMissing::build_from(
225                                coords,
226                                RasterSourceId::new(group.source_name.clone()),
227                            ))
228                            .map_err(ProcedureError::Send)?;
229                    }
230                    Err(CachedTileFailure::Transport { retryable, message }) => {
231                        retry |= retryable;
232                        if retryable {
233                            tracing::warn!(
234                                %coords,
235                                source = ?group.source_name,
236                                error = %message,
237                                "raster tile fetch failed"
238                            );
239                        } else {
240                            tracing::error!(
241                                %coords,
242                                source = ?group.source_name,
243                                error = %message,
244                                "raster tile fetch failed"
245                            );
246                        }
247                        context
248                            .send_back(<T as RasterTransferables>::LayerRasterMissing::build_from(
249                                coords,
250                                RasterSourceId::new(group.source_name.clone()),
251                            ))
252                            .map_err(ProcedureError::Send)?;
253                    }
254                }
255                continue;
256            }
257
258            let data = match client.fetch(&coords, &group.source).await {
259                Ok(data) => data,
260                Err(error) => {
261                    let retryable = error.is_retryable();
262                    retry |= retryable;
263                    if error.is_not_found() {
264                        tracing::debug!(
265                            %coords,
266                            source = ?group.source_name,
267                            "no raster tile at the source; the layer is empty"
268                        );
269                    } else if retryable {
270                        tracing::warn!(
271                            %coords,
272                            source = ?group.source_name,
273                            error = %error.describe(),
274                            "raster tile fetch failed"
275                        );
276                    } else {
277                        tracing::error!(
278                            %coords,
279                            source = ?group.source_name,
280                            error = %error.describe(),
281                            "raster tile fetch failed"
282                        );
283                    }
284
285                    context
286                        .send_back(<T as RasterTransferables>::LayerRasterMissing::build_from(
287                            coords,
288                            RasterSourceId::new(group.source_name.clone()),
289                        ))
290                        .map_err(ProcedureError::Send)?;
291                    continue;
292                }
293            };
294            let request = RasterTileRequest {
295                coords,
296                source: RasterSourceId::new(group.source_name.clone()),
297            };
298            let processed = run_cpu_bound_replying(context.clone(), move |context| {
299                let mut process_context = ProcessRasterContext::<T, _>::new(context);
300                process_raster_tile(&data, request, &mut process_context)
301            })
302            .await
303            .map_err(ProcessRasterError::CpuWork)
304            .and_then(|processed| processed);
305            match processed {
306                Ok(()) => {}
307                Err(ProcessRasterError::Send(source)) => return Err(ProcedureError::Send(source)),
308                Err(
309                    error @ (ProcessRasterError::Decoding { .. } | ProcessRasterError::CpuWork(_)),
310                ) => {
311                    tracing::warn!(%coords, source = ?group.source_name, %error, "invalid raster tile");
312                    context
313                        .send_back(T::LayerRasterMissing::build_from(
314                            coords,
315                            RasterSourceId::new(group.source_name.clone()),
316                        ))
317                        .map_err(ProcedureError::Send)?;
318                }
319            }
320        }
321
322        context
323            .send_back(TileRequestOutcome {
324                coords,
325                kind: RequestKind::Raster,
326                attempt: Some(attempt),
327                disposition: if retry {
328                    RequestDisposition::Retry
329                } else {
330                    RequestDisposition::Complete
331                },
332            })
333            .map_err(ProcedureError::Send)
334    })
335}

337/// Sends the tile of an image source back, empty where the picture does not reach it, and
338/// returns whether a failed fetch is worth retrying.
339async fn send_image_tile<
340    T: RasterTransferables,
341    HC: crate::io::source_client::HttpClient,
342    C: Context,
343>(
344    (client, image): (
345        &crate::io::source_client::SourceClient<HC>,
346        &crate::style::source::ImageSource,
347    ),
348    (coords, source): (WorldTileCoords, RasterSourceId),
349    context: C,
350) -> Result<bool, ProcedureError> {
351    use crate::raster::transferables::LayerRaster;
352    match crate::raster::image_source::load_tile(client, image, coords).await {
353        Ok(tile) => {
354            // A tile the picture does not reach has loaded too; it is simply empty.
355            let tile = tile.unwrap_or_else(|| image::RgbaImage::new(1, 1));
356            context
357                .send_back(T::LayerRaster::build_from(coords, source, tile))
358                .map_err(ProcedureError::Send)?;
359            Ok(false)
360        }
361        Err(failure) => {
362            context
363                .send_back(T::LayerRasterMissing::build_from(coords, source))
364                .map_err(ProcedureError::Send)?;
365            Ok(failure.is_retryable())
366        }
367    }
368}

370impl<E: Environment, T: RasterTransferables> RequestSystem<E, T> {
371    fn request(
372        &self,
373        coords: crate::coords::WorldTileCoords,
374        style: &mut FrameStyle,
375        world: &mut crate::tcs::world::World,
376    ) -> SystemResult {
377        let exists = world
378            .tiles
379            .query::<&RasterLayersDataComponent>(coords)
380            .is_some();
381        let attempt = tile_retry::next_attempt(world);
382        let Some(mut tile) = world.tiles.spawn_mut(coords) else {
383            return Err(SystemError::InvalidTile { coords });
384        };
385        self.kernel
386            .apc()
387            .call(
388                Input::new(coords, style.share(Style::clone), attempt),
389                fetch_raster_apc::<E::OffscreenKernelEnvironment, T, _>,
390            )
391            .map_err(|source| SystemError::TileRequest {
392                kind: "raster",
393                coords,
394                source,
395            })?;
396        if !exists {
397            tile.insert(RasterLayersDataComponent::default());
398        }
399        tile_retry::started(world, coords, RequestKind::Raster, attempt, None);
400        tracing::debug!(%coords, "raster tile request accepted");
401        Ok(())
402    }
403}

405fn coarse_raster_ancestor(coords: WorldTileCoords, minzoom: u8) -> Option<WorldTileCoords> {
406    let target = u8::from(coords.z).saturating_sub(3).max(minzoom);
407    if target >= u8::from(coords.z) {
408        return None;
409    }
410    let mut current = coords;
411    while u8::from(current.z) > target {
412        current = current.get_parent()?;
413    }
414    Some(current)
415}

417fn wanted_tiles(
418    regions: Vec<(RasterSourceId, Vec<WorldTileCoords>)>,
419    style: &crate::style::Style,
420    world: &crate::tcs::world::World,
421) -> Vec<WorldTileCoords> {
422    let mut wanted = Vec::new();
423    let mut coarse_ancestors = Vec::new();
424    let mut seen_coarse = FxHashSet::default();
425    let is_terrain = style.terrain.is_some();
426    for (source, tiles) in regions {
427        let minzoom = match source.name().and_then(|name| style.sources.get(name)) {
428            Some(crate::style::source::Source::Raster(source)) => source.minzoom,
429            Some(crate::style::source::Source::RasterDem(source)) => source.minzoom,
430            _ => None,
431        }
432        .unwrap_or(0);
433        let bounds = match source.name().and_then(|name| style.sources.get(name)) {
434            Some(crate::style::source::Source::Raster(source)) => source.bounds,
435            Some(crate::style::source::Source::Image(image)) => image_bounds(image),
436            _ => None,
437        };
438        for coords in tiles {
439            // Tiles outside the source's declared bounds are never requested.
440            if bounds.is_some_and(|bounds| !crate::io::tile_sources::tile_in_bounds(coords, bounds))
441            {
442                continue;
443            }
444            if is_terrain {
445                if let Some(ancestor) = coarse_raster_ancestor(coords, minzoom) {
446                    if seen_coarse.insert(ancestor) {
447                        coarse_ancestors.push(ancestor);
448                    }
449                }
450            }
451            wanted.push(coords);
452            if let Some(parent) = missing_tile_fallback(coords, minzoom, |coords| {
453                world
454                    .tiles
455                    .query::<&RasterLayersDataComponent>(coords)
456                    .is_some_and(|component| component.source_is_missing(&source))
457            }) {
458                wanted.push(parent);
459            }
460        }
461    }
462    if !coarse_ancestors.is_empty() {
463        // Bracket the local terrain with one coarse raster overview tile first, then fine
464        // tiles, then any remaining coarse ancestors, matching DEM `request_order`.
465        let remaining = coarse_ancestors.split_off(coarse_ancestors.len().min(2));
466        coarse_ancestors.extend(wanted);
467        coarse_ancestors.extend(remaining);
468        return coarse_ancestors;
469    }
470    wanted
471}

473/// The bounds `(west, south, east, north)` of an image's corners, so tiles it cannot reach are
474/// not requested; none for a picture that reaches past the antimeridian, whose copies across it
475/// a tile may show.
476fn image_bounds(image: &crate::style::source::ImageSource) -> Option<(f64, f64, f64, f64)> {
477    let longitudes = image.coordinates.map(|[longitude, _]| longitude);
478    let latitudes = image.coordinates.map(|[_, latitude]| latitude);
479    let fold = |values: [f64; 4]| {
480        values
481            .into_iter()
482            .fold((f64::MAX, f64::MIN), |(lo, hi), value| {
483                (lo.min(value), hi.max(value))
484            })
485    };
486    let ((west, east), (south, north)) = (fold(longitudes), fold(latitudes));
487    (west >= -180.0 && east <= 180.0).then_some((west, south, east, north))
488}
