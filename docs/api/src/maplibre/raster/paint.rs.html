7/// Bind group layout of a layer's [`RasterUniforms`], group two of the raster pipeline.
8pub fn layout() -> Vec<wgpu::BindGroupLayoutEntry> {
9    vec![wgpu::BindGroupLayoutEntry {
10        binding: 0,
11        visibility: wgpu::ShaderStages::VERTEX_FRAGMENT,
12        ty: wgpu::BindingType::Buffer {
13            ty: wgpu::BufferBindingType::Uniform,
14            has_dynamic_offset: false,
15            min_binding_size: None,
16        },
17        count: None,
18    }]
19}

21/// The translation, in clip units per unit of `w`, that puts the centre of the view on a whole
22/// pixel for raster tiles, as GL JS aligns them while the map is at rest.
23pub fn pixel_alignment(view: &crate::render::view_state::ViewState) -> [f32; 2] {
24    let center = view.camera().position();
25    let angle = view.camera().get_bearing().0;
26    let (cos, sin) = (angle.cos(), angle.sin());
27    let shift_x = (view.width() % 2.0) / 2.0;
28    let shift_y = (view.height() % 2.0) / 2.0;
29    let mut dx = center.x - center.x.round() + cos * shift_x + sin * shift_y;
30    let mut dy = center.y - center.y.round() + cos * shift_y + sin * shift_x;
31    if dx > 0.5 {
32        dx -= 1.0;
33    }
34    if dy > 0.5 {
35        dy -= 1.0;
36    }
37    let matrix = view.view_projection().0;
38    let ndc = |x: f64, y: f64| {
39        let clip = matrix * cgmath::Vector4::new(x, y, 0.0, 1.0);
40        [clip.x / clip.w, clip.y / clip.w]
41    };
42    let (at, moved) = (ndc(center.x, center.y), ndc(center.x + dx, center.y + dy));
43    [(moved[0] - at[0]) as f32, (moved[1] - at[1]) as f32]
44}

46/// Uniforms of the raster fragment stage; the layout mirrors the WGSL struct.
47#[repr(C)]
48#[derive(Copy, Clone, Debug, PartialEq, Pod, Zeroable)]
49pub struct RasterUniforms {
50    /// Hue rotation matrix rows in the form GL JS uses: the three dot-product weights.
51    pub spin_weights: [f32; 4],
52    /// Layer alpha multiplier.
53    pub opacity: f32,
54    /// Weight pulling colours towards their average.
55    pub saturation_factor: f32,
56    /// Scale of colours around middle grey.
57    pub contrast_factor: f32,
58    /// Output for black input.
59    pub brightness_min: f32,
60    /// Output for white input.
61    pub brightness_max: f32,
62    /// One when texels are picked without interpolation (`raster-resampling: nearest`).
63    pub nearest: f32,
64    /// Screen translation in clip units per unit of `w`: the raster is placed as if the
65    /// centre of the view lay on a whole pixel, which keeps texels on pixels.
66    pub align: [f32; 2],
67}
68
69impl RasterUniforms {
70    /// Evaluates the adjustments as GL JS `rasterUniformValues` does.
71    pub fn from_paint(paint: &RasterPaint, zoom: f64) -> Self {
72        let value = |property: &Option<StyleProperty<f32>>, default: f32| {
73            property
74                .as_ref()
75                .and_then(|property| property.evaluate_at_zoom(zoom))
76                .unwrap_or(default)
77        };
78        let hue = value(&paint.raster_hue_rotate, 0.0).to_radians();
79        let (sin, cos) = hue.sin_cos();
80        let root = 3.0_f32.sqrt();
81        let saturation = value(&paint.raster_saturation, 0.0);
82        let contrast = value(&paint.raster_contrast, 0.0);
83        Self {
84            spin_weights: [
85                (2.0 * cos + 1.0) / 3.0,
86                (-root * sin - cos + 1.0) / 3.0,
87                (root * sin - cos + 1.0) / 3.0,
88                0.0,
89            ],
90            opacity: value(&paint.raster_opacity, 1.0).clamp(0.0, 1.0),
91            saturation_factor: if saturation > 0.0 {
92                1.0 - 1.0 / (1.001 - saturation)
93            } else {
94                -saturation
95            },
96            contrast_factor: if contrast > 0.0 {
97                1.0 / (1.0 - contrast)
98            } else {
99                1.0 + contrast
100            },
101            brightness_min: value(&paint.raster_brightness_min, 0.0),
102            brightness_max: value(&paint.raster_brightness_max, 1.0),
103            nearest: f32::from(u8::from(
104                paint
105                    .raster_resampling
106                    .as_ref()
107                    .is_some_and(|resampling| resampling.is_nearest_at(zoom)),
108            )),
109            align: [0.0; 2],
110        }
111    }
112}

114#[cfg(test)]
115mod alignment_tests {
116    use cgmath::Deg;
117
118    use super::pixel_alignment;
119    use crate::{
120        coords::{LatLon, WorldCoords, Zoom},
121        render::view_state::ViewState,
122        window::PhysicalSize,
123    };
124
125    fn view(bearing: f64) -> ViewState {
126        let zoom = Zoom::new(16.0);
127        let mut view = ViewState::new(
128            PhysicalSize::new(512, 256).expect("size"),
129            WorldCoords::from_lat_lon(LatLon::new(52.499167, 13.418056), zoom),
130            zoom,
131            Deg(0.0),
132            cgmath::Rad(0.6435011087932844),
133        );
134        view.camera_mut().set_bearing(Deg(bearing));
135        view
136    }
137
138    /// The map moves on screen by the fractional position of its centre, turned with the map.
139    #[test]
140    fn the_shift_is_the_fraction_of_the_centre_turned_by_the_bearing() {
141        let view = view(0.0);
142        let center = view.camera().position();
143        let fraction = [center.x - center.x.round(), center.y - center.y.round()];
144        let to_pixels = |ndc: [f32; 2]| [ndc[0] * 256.0, -ndc[1] * 128.0];
145        let flat = to_pixels(pixel_alignment(&view));
146        assert!((f64::from(flat[0]) - fraction[0]).abs() < 0.01, "{flat:?}");
147        assert!((f64::from(flat[1]) - fraction[1]).abs() < 0.01, "{flat:?}");
148        // A bearing of 90 degrees puts east at the top, so the shift turns a quarter.
149        let turned = to_pixels(pixel_alignment(&self::view(90.0)));
150        assert!(
151            (f64::from(turned[0]) - fraction[1]).abs() < 0.01,
152            "{turned:?}"
153        );
154        assert!(
155            (f64::from(turned[1]) + fraction[0]).abs() < 0.01,
156            "{turned:?}"
157        );
158    }
159}

161#[cfg(test)]
162mod tests {
163    use super::*;

165    fn paint(json: serde_json::Value) -> RasterPaint {
166        serde_json::from_value(json).expect("valid raster paint")
167    }
168
169    #[test]
170    fn neutral_paint_changes_nothing() {
171        let uniforms = RasterUniforms::from_paint(&paint(serde_json::json!({})), 0.0);
172        assert_eq!(uniforms.opacity, 1.0);
173        assert_eq!(uniforms.saturation_factor, 0.0);
174        assert_eq!(uniforms.contrast_factor, 1.0);
175        assert_eq!(uniforms.brightness_max, 1.0);
176        assert!((uniforms.spin_weights[0] - 1.0).abs() < 1e-6);
177        assert!(uniforms.spin_weights[1].abs() < 1e-6);
178    }
179
180    #[test]
181    fn opacity_follows_zoom() {
182        let paint = paint(serde_json::json!({
183            "raster-opacity": {"stops": [[0, 0.0], [10, 1.0]]}
184        }));
185        assert_eq!(RasterUniforms::from_paint(&paint, 5.0).opacity, 0.5);
186    }
187
188    #[test]
189    fn full_saturation_and_contrast_amplify() {
190        let uniforms = RasterUniforms::from_paint(
191            &paint(serde_json::json!({
192                "raster-saturation": 0.5, "raster-contrast": 0.5, "raster-opacity": 2.0
193            })),
194            0.0,
195        );
196        assert!((uniforms.saturation_factor - (1.0 - 1.0 / 0.501)).abs() < 1e-5);
197        assert_eq!(uniforms.contrast_factor, 2.0);
198        assert_eq!(uniforms.opacity, 1.0);
199    }
