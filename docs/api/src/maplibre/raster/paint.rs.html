7/// Bind group layout of a layer's [`RasterUniforms`], group two of the raster pipeline.
8pub fn layout() -> Vec<wgpu::BindGroupLayoutEntry> {
9    vec![wgpu::BindGroupLayoutEntry {
10        binding: 0,
11        visibility: wgpu::ShaderStages::VERTEX_FRAGMENT,
12        ty: wgpu::BindingType::Buffer {
13            ty: wgpu::BufferBindingType::Uniform,
14            has_dynamic_offset: false,
15            min_binding_size: None,
16        },
17        count: None,
18    }]
19}

21/// The translation, in clip units per unit of `w`, that puts the centre of the view on a whole
22/// pixel for raster tiles, as GL JS aligns them while the map is at rest.
23///
24/// A pitched or host-driven eye has no single pixel scale across the view, so rounding its
25/// centre would only make the imagery jitter by a pixel as it moves; it gets no shift.
26pub fn pixel_alignment(view: &crate::render::view_state::ViewState) -> [f32; 2] {
27    if view.has_external_view()
28        || view.pose_view().is_some()
29        || view.camera().get_pitch().0.abs() > 1e-6
30    {
31        return [0.0; 2];
32    }
33    let center = view.camera().position();
34    let angle = view.camera().get_bearing().0;
35    let (cos, sin) = (angle.cos(), angle.sin());
36    let shift_x = (view.width() % 2.0) / 2.0;
37    let shift_y = (view.height() % 2.0) / 2.0;
38    let mut dx = center.x - center.x.round() + cos * shift_x + sin * shift_y;
39    let mut dy = center.y - center.y.round() + cos * shift_y + sin * shift_x;
40    if dx > 0.5 {
41        dx -= 1.0;
42    }
43    if dy > 0.5 {
44        dy -= 1.0;
45    }
46    let matrix = view.view_projection().0;
47    let ndc = |x: f64, y: f64| {
48        let clip = matrix * cgmath::Vector4::new(x, y, 0.0, 1.0);
49        [clip.x / clip.w, clip.y / clip.w]
50    };
51    let (at, moved) = (ndc(center.x, center.y), ndc(center.x + dx, center.y + dy));
52    [(moved[0] - at[0]) as f32, (moved[1] - at[1]) as f32]
53}

55/// Uniforms of the raster fragment stage; the layout mirrors the WGSL struct.
56#[repr(C)]
57#[derive(Copy, Clone, Debug, PartialEq, Pod, Zeroable)]
58pub struct RasterUniforms {
59    /// Hue rotation matrix rows in the form GL JS uses: the three dot-product weights.
60    pub spin_weights: [f32; 4],
61    /// Layer alpha multiplier.
62    pub opacity: f32,
63    /// Weight pulling colours towards their average.
64    pub saturation_factor: f32,
65    /// Scale of colours around middle grey.
66    pub contrast_factor: f32,
67    /// Output for black input.
68    pub brightness_min: f32,
69    /// Output for white input.
70    pub brightness_max: f32,
71    /// One when texels are picked without interpolation (`raster-resampling: nearest`).
72    pub nearest: f32,
73    /// Screen translation in clip units per unit of `w`: the raster is placed as if the
74    /// centre of the view lay on a whole pixel, which keeps texels on pixels.
75    pub align: [f32; 2],
76}
77
78impl RasterUniforms {
79    /// Evaluates the adjustments as GL JS `rasterUniformValues` does.
80    pub fn from_paint(paint: &RasterPaint, zoom: f64) -> Self {
81        let value = |property: &Option<StyleProperty<f32>>, default: f32| {
82            property
83                .as_ref()
84                .and_then(|property| property.evaluate_at_zoom(zoom))
85                .unwrap_or(default)
86        };
87        let hue = value(&paint.raster_hue_rotate, 0.0).to_radians();
88        let (sin, cos) = hue.sin_cos();
89        let root = 3.0_f32.sqrt();
90        let saturation = value(&paint.raster_saturation, 0.0);
91        let contrast = value(&paint.raster_contrast, 0.0);
92        Self {
93            spin_weights: [
94                (2.0 * cos + 1.0) / 3.0,
95                (-root * sin - cos + 1.0) / 3.0,
96                (root * sin - cos + 1.0) / 3.0,
97                0.0,
98            ],
99            opacity: value(&paint.raster_opacity, 1.0).clamp(0.0, 1.0),
100            saturation_factor: if saturation > 0.0 {
101                1.0 - 1.0 / (1.001 - saturation)
102            } else {
103                -saturation
104            },
105            contrast_factor: if contrast > 0.0 {
106                1.0 / (1.0 - contrast)
107            } else {
108                1.0 + contrast
109            },
110            brightness_min: value(&paint.raster_brightness_min, 0.0),
111            brightness_max: value(&paint.raster_brightness_max, 1.0),
112            nearest: f32::from(u8::from(
113                paint
114                    .raster_resampling
115                    .as_ref()
116                    .is_some_and(|resampling| resampling.is_nearest_at(zoom)),
117            )),
118            align: [0.0; 2],
119        }
120    }
121}

123#[cfg(test)]
124mod alignment_tests {
125    use cgmath::Deg;
126
127    use super::pixel_alignment;
128    use crate::{
129        coords::{LatLon, WorldCoords, Zoom},
130        render::view_state::ViewState,

134    fn view(bearing: f64) -> ViewState {
135        let zoom = Zoom::new(16.0);
136        let mut view = ViewState::new(
137            PhysicalSize::new(512, 256).expect("size"),
138            WorldCoords::from_lat_lon(LatLon::new(52.499167, 13.418056), zoom),
139            zoom,
140            Deg(0.0),
141            cgmath::Rad(0.6435011087932844),
142        );
143        view.camera_mut().set_bearing(Deg(bearing));
144        view
145    }
146
147    /// The map moves on screen by the fractional position of its centre, turned with the map.
148    #[test]
149    fn the_shift_is_the_fraction_of_the_centre_turned_by_the_bearing() {
150        let view = view(0.0);
151        let center = view.camera().position();
152        let fraction = [center.x - center.x.round(), center.y - center.y.round()];
153        let to_pixels = |ndc: [f32; 2]| [ndc[0] * 256.0, -ndc[1] * 128.0];
154        let flat = to_pixels(pixel_alignment(&view));
155        assert!((f64::from(flat[0]) - fraction[0]).abs() < 0.01, "{flat:?}");
156        assert!((f64::from(flat[1]) - fraction[1]).abs() < 0.01, "{flat:?}");
157        // A bearing of 90 degrees puts east at the top, so the shift turns a quarter.
158        let turned = to_pixels(pixel_alignment(&self::view(90.0)));
159        assert!(
160            (f64::from(turned[0]) - fraction[1]).abs() < 0.01,
161            "{turned:?}"
162        );
163        assert!(
164            (f64::from(turned[1]) + fraction[0]).abs() < 0.01,
165            "{turned:?}"
166        );
167    }
168
169    #[test]
170    fn a_pitched_view_is_not_shifted() {
171        let mut view = view(0.0);
172        assert_ne!(
173            pixel_alignment(&view),
174            [0.0; 2],
175            "the level view is shifted"
176        );
177        view.camera_mut().set_pitch(Deg(30.0));
178        assert_eq!(pixel_alignment(&view), [0.0; 2]);
179    }
180
181    #[test]
182    fn an_external_eye_is_not_shifted() {
183        use cgmath::{Matrix4, Rad, SquareMatrix, Vector3};
184
185        use crate::{
186            projection::ProjectionType,
187            render::{
188                camera::EyeFrustum,
189                view_state::{ExternalAnchor, ExternalView},
190            },
191        };
192        let mut view = view(0.0);
193        // Straight down, so only the external eye can explain a missing shift.
194        let external = ExternalView {
195            anchor: ExternalAnchor {
196                position: LatLon::new(52.499167, 13.418056),
197                altitude_meters: 0.0,
198            },
199            view: Matrix4::from_translation(Vector3::new(0.0, 0.0, 400.0))
200                .invert()
201                .expect("eye"),
202            frustum: EyeFrustum::symmetric(Rad(1.0), 2.0, 0.05, 1.0e8),
203        };
204        view.set_external_view(external, &ProjectionType::Mercator)
205            .expect("external view");
206        assert!(view.has_external_view());
207        assert_eq!(pixel_alignment(&view), [0.0; 2]);
208    }

220    fn neutral_paint_changes_nothing() {
221        let uniforms = RasterUniforms::from_paint(&paint(serde_json::json!({})), 0.0);
