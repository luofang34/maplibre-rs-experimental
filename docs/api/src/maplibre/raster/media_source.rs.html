25#[derive(Default)]
26struct DynamicMediaState {
27    /// Cached `(generation, Arc<RgbaImage>)` per source name so panning/zooming between frame
28    /// pushes reuses the decoded buffer without reallocating.
29    decoded: HashMap<String, (u64, Arc<RgbaImage>)>,
30    /// Last `generation` resampled into `world.tiles` at `(source_name, coords)`.
31    tiles: HashMap<(String, WorldTileCoords), u64>,
32}

34/// Returns `true` when `source_name` names a `"video"` or `"canvas"` source in `style`.
35pub(crate) fn is_media_source(style: &Style, source_name: Option<&str>) -> bool {
36    source_name
37        .and_then(|name| style.sources.get(name))
38        .is_some_and(|source| matches!(source, Source::Video(_) | Source::Canvas(_)))
39}

41/// Geographic bounding box `(west, south, east, north)` of a 4-corner quad when it does not wrap
42/// across the antimeridian.
43fn quad_bounds(coordinates: [[f64; 2]; 4]) -> Option<(f64, f64, f64, f64)> {
44    let longitudes = coordinates.map(|[longitude, _]| longitude);
45    let latitudes = coordinates.map(|[_, latitude]| latitude);
46    let fold = |values: [f64; 4]| {
47        values
48            .into_iter()
49            .fold((f64::MAX, f64::MIN), |(lo, hi), value| {
50                (lo.min(value), hi.max(value))
51            })
52    };
53    let ((west, east), (south, north)) = (fold(longitudes), fold(latitudes));
54    (west >= -180.0 && east <= 180.0).then_some((west, south, east, north))
55}

57/// Resamples any visible tiles of `"video"` and `"canvas"` sources whose frame or corner
58/// generation changed since they were last populated, marking their GPU bindings dirty for
59/// [`super::upload_system::upload_system`].
60pub(crate) fn sync_media_sources(style: &Style, view_state: &ViewState, world: &mut World) {
61    let has_active_frame = style.sources.values().any(|source| {
62        matches!(
63            source,
64            Source::Video(VideoSource { frame: Some(_), .. })
65                | Source::Canvas(CanvasSource { frame: Some(_), .. })
66        )
67    });
68    if !has_active_frame {
69        return;
70    }
71
72    let Ok(regions) = raster_source_regions(style, view_state, world, ViewStatePadding::Tight)
73    else {
74        return;
75    };
76
77    let mut state = world
78        .resources
79        .remove::<DynamicMediaState>()
80        .unwrap_or_default();
81
82    // Prune cache entries for sources that left the style.
83    state
84        .decoded
85        .retain(|name, _| is_media_source(style, Some(name.as_str())));
86    state
87        .tiles
88        .retain(|(name, _), _| is_media_source(style, Some(name.as_str())));
89
90    for (source_id, covering_tiles) in regions {
91        let Some(name) = source_id.name() else {
92            continue;
93        };
94        let (frame, coordinates, generation) = match style.sources.get(name) {
95            Some(Source::Video(video)) => {
96                (video.frame.as_ref(), video.coordinates, video.generation)
97            }
98            Some(Source::Canvas(canvas)) => {
99                (canvas.frame.as_ref(), canvas.coordinates, canvas.generation)
100            }
101            _ => continue,
102        };
103        let Some(frame) = frame else {
104            continue;
105        };
106        let Some(picture) = cached_frame_image(&mut state, name, frame, generation) else {
107            continue;
108        };
109        let bounds = quad_bounds(coordinates);
110
111        for coords in covering_tiles {
112            if bounds.is_some_and(|bounds| !crate::io::tile_sources::tile_in_bounds(coords, bounds))
113            {
114                continue;
115            }
116            let has_tile_data = world
117                .tiles
118                .query::<&RasterLayersDataComponent>(coords)
119                .is_some_and(|component| component.has_source_image(&source_id));
120            let key = (name.to_owned(), coords);
121            if has_tile_data && state.tiles.get(&key) == Some(&generation) {
122                continue;
123            }
124
125            let resampled = image_source::render_tile(&picture, coordinates, coords)
126                .unwrap_or_else(|| RgbaImage::new(1, 1));
127            let exists = world
128                .tiles
129                .query::<&RasterLayersDataComponent>(coords)
130                .is_some();
131            let Some(mut tile) = world.tiles.spawn_mut(coords) else {
132                continue;
133            };
134            if !exists {
135                tile.insert(RasterLayersDataComponent::default());
136            }
137            if let Some(component) = world
138                .tiles
139                .query_mut::<&mut RasterLayersDataComponent>(coords)
140            {
141                component.record(RasterLayerData::Available(AvailableRasterLayerData {
142                    coords,
143                    source: source_id.clone(),
144                    image: resampled,
145                }));
146            }
147            if let Some(Eventually::Initialized(raster)) =
148                world.resources.get_mut::<Eventually<RasterResources>>()
149            {
150                raster.tile_data_changed(&source_id, coords);
151            }
152            state.tiles.insert(key, generation);
153        }
154    }
155
156    world.resources.insert(state);
157}
158
159fn cached_frame_image(
160    state: &mut DynamicMediaState,
161    name: &str,
162    frame: &MediaFrame,
163    generation: u64,
164) -> Option<Arc<RgbaImage>> {
165    if let Some((cached_gen, image)) = state.decoded.get(name) {
166        if *cached_gen == generation {
167            return Some(Arc::clone(image));
168        }
169    }
170    let image = Arc::new(RgbaImage::from_raw(
171        frame.width,
172        frame.height,
173        frame.rgba.to_vec(),
174    )?);
175    state
176        .decoded
177        .insert(name.to_owned(), (generation, Arc::clone(&image)));
178    state.tiles.retain(|(src, _), _| src != name);
179    Some(image)
180}
