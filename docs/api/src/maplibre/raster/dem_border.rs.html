1//! One-pixel borders around DEM tiles so slope shading reads across tile edges.
2//!
3//! A tile's border holds the edge samples of its loaded neighbours, as GL JS backfills DEM
4//! tiles, and its own edge where a neighbour is missing. Shaders that read these textures
5//! address the interior at an offset of one texel.
6
7use image::RgbaImage;
8
9/// Texels added on each side of a DEM tile.
10pub const BORDER: u32 = 1;
11
12/// The bit standing for the neighbour at offset (`dx`, `dy`) in a sides mask.
13pub fn side_bit(dx: i32, dy: i32) -> u16 {
14    1 << ((dy + 1) * 3 + (dx + 1))
15}
16
17/// The sides of a border filled from a neighbour of the tile's own size.
18pub fn filled_sides(image: &RgbaImage, neighbours: &[[Option<&RgbaImage>; 3]; 3]) -> u16 {
19    let mut sides = 0;
20    for dy in -1..=1_i32 {
21        for dx in -1..=1_i32 {
22            let usable = neighbours[(dy + 1) as usize][(dx + 1) as usize]
23                .is_some_and(|neighbour| neighbour.dimensions() == image.dimensions());
24            if (dx, dy) != (0, 0) && usable {
25                sides |= side_bit(dx, dy);
26            }
27        }
28    }
29    sides
30}
31
32/// The image with a border of neighbour samples; `neighbours` is indexed `[dy + 1][dx + 1]`
33/// and the centre entry is ignored. A neighbour of another size counts as missing.
34pub fn with_border(image: &RgbaImage, neighbours: &[[Option<&RgbaImage>; 3]; 3]) -> RgbaImage {
35    let (width, height) = image.dimensions();
36    let edge = i64::from(BORDER);
37    RgbaImage::from_fn(width + 2 * BORDER, height + 2 * BORDER, |x, y| {
38        let (column, row) = (i64::from(x) - edge, i64::from(y) - edge);
39        let side = |value: i64, size: u32| match value {
40            v if v < 0 => -1,
41            v if v >= i64::from(size) => 1,
42            _ => 0,
43        };
44        let (dx, dy) = (side(column, width), side(row, height));
45        let wrapped = |value: i64, size: u32| value.rem_euclid(i64::from(size)) as u32;
46        let clamped = |value: i64, size: u32| value.clamp(0, i64::from(size) - 1) as u32;
47        if (dx, dy) != (0, 0) {
48            if let Some(neighbour) = neighbours[(dy + 1) as usize][(dx + 1) as usize]
49                .filter(|neighbour| neighbour.dimensions() == (width, height))
50            {
51                return *neighbour.get_pixel(wrapped(column, width), wrapped(row, height));
52            }
53        }
54        *image.get_pixel(clamped(column, width), clamped(row, height))
55    })
56}

58/// A pixel value for tests that need to tell tiles apart.
59#[cfg(test)]
60fn solid(value: u8) -> RgbaImage {
61    use image::Rgba;
62    RgbaImage::from_pixel(4, 4, Rgba([value, 0, 0, 255]))
63}
64
65#[cfg(test)]
66mod tests {
67    use image::Rgba;
68
69    use super::*;
70
71    #[test]
72    fn the_border_takes_neighbour_edges_and_replicates_where_none_loaded() {
73        let centre = RgbaImage::from_fn(4, 4, |x, y| Rgba([(x * 10 + y) as u8, 0, 0, 255]));
74        let east = solid(200);
75        let mut neighbours: [[Option<&RgbaImage>; 3]; 3] = Default::default();
76        neighbours[1][2] = Some(&east);
77        let bordered = with_border(&centre, &neighbours);
78        assert_eq!(bordered.dimensions(), (6, 6));
79        assert_eq!(
80            bordered.get_pixel(3, 3),
81            centre.get_pixel(2, 2),
82            "interior is offset"
83        );
84        assert_eq!(
85            bordered.get_pixel(5, 3)[0],
86            200,
87            "east border is the east tile"
88        );
89        assert_eq!(
90            bordered.get_pixel(0, 3),
91            centre.get_pixel(0, 2),
92            "an absent neighbour repeats the edge"
93        );
94        assert_eq!(
95            bordered.get_pixel(5, 0),
96            centre.get_pixel(3, 0),
97            "an absent corner repeats the nearest interior texel"
98        );
99    }
100
101    #[test]
102    fn a_neighbour_of_another_size_is_ignored() {
103        let centre = solid(7);
104        let wrong = RgbaImage::from_pixel(2, 2, Rgba([99, 0, 0, 255]));
105        let mut neighbours: [[Option<&RgbaImage>; 3]; 3] = Default::default();
106        neighbours[1][0] = Some(&wrong);
107        assert_eq!(with_border(&centre, &neighbours).get_pixel(0, 2)[0], 7);
108    }
