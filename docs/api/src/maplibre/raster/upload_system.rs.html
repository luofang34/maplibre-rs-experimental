22pub fn upload_system(
23    MapContext {
24        world,
25        renderer: Renderer { device, queue, .. },
26        ..
27    }: &mut MapContext,
28) -> SystemResult {
29    let Some(raster_state) = world.resources.get::<Eventually<RasterResources>>() else {
30        return Ok(());
31    };
32    if !matches!(raster_state, Initialized(_)) {
33        return Err(SystemError::Dependencies);
34    }
35    if crate::render::eye_covering::EyeInFrame::reuses_content(world) {
36        return Ok(());
37    }
38    world.resources.get_or_init_mut::<FrameUploads>();
39    world.resources.get_or_init_mut::<GpuDemPool>();
40    let Some((Initialized(raster_resources), budget, gpu_dem_pool)) = world.resources.query_mut::<(
41        &mut Eventually<RasterResources>,
42        &mut FrameUploads,
43        &mut GpuDemPool,
44    )>() else {
45        return Err(SystemError::Dependencies);
46    };
47
48    // Every loaded tile gets its texture, as GL JS uploads a raster tile when it arrives, as
49    // far as the frame's image floor reaches; coarser tiles go first so an ancestor stand-in is
50    // ready before a burst of finer children exhausts the frame budget.
51    let source_tiles = world
52        .tiles
53        .pyramid_of::<RasterLayersDataComponent>()
54        .coords_coarse_to_fine();
55    let mut deferred = upload_raster_layer(
56        raster_resources,
57        (device, queue),
58        &world.tiles,
59        source_tiles,
60        budget,
61    );
62    if !deferred {
63        let dem_tiles = world
64            .tiles
65            .pyramid_of::<DemLayersComponent>()
66            .coords_coarse_to_fine();
67        deferred |= upload_dem_layers(
68            raster_resources,
69            gpu_dem_pool,
70            (device, queue),
71            &world.tiles,
72            dem_tiles,
73            budget,
74        );
75    }
76    if deferred {
77        crate::render::frame_signals::keep_animating(world);
78    }
79
80    Ok(())
81}

83/// Uploads the imagery textures of loaded tiles that the frame's budget admits; `true` when some
84/// wait for a later frame.
85#[tracing::instrument(skip_all)]
86fn upload_raster_layer(
87    raster_resources: &mut RasterResources,
88    (device, queue): (&wgpu::Device, &crate::render::upload_queue::UploadQueue),
89    tiles: &Tiles,
90    source_tiles: impl IntoIterator<Item = WorldTileCoords>,
91    budget: &mut FrameUploads,
92) -> bool {
93    let mut mipmapped = Vec::new();
94    let mut deferred = false;
95    'tiles: for coords in source_tiles {
96        let Some(layers) = tiles.query::<&RasterLayersDataComponent>(coords) else {
97            continue;
98        };
99        for data in &layers.layers {
100            if let RasterLayerData::Available(data) = data {
101                if raster_resources.has_border(&data.source) && data.image.is_empty() {
102                    continue;
103                }
104                // A result that lands after its source left the style is never drawn.
105                if raster_resources.draws(&data.source)
106                    && raster_resources
107                        .get_bound_texture(&data.source, &coords)
108                        .is_none()
109                {
110                    if !budget.take(UploadKind::Image, data.image.as_raw().len()) {
111                        deferred = true;
112                        break 'tiles;
113                    }
114                    mipmapped.push(upload_imagery(
115                        raster_resources,
116                        device,
117                        queue,
118                        coords,
119                        data,
120                    ));
121                }
122            }
123        }
124    }
125    // One submission fills every new texture's mip chain.
126    raster_resources.generate_mipmaps(device, queue, &mipmapped);
127    deferred
128}

130/// Uploads or reuses shared `(dim + 4) x (dim + 4)` DEM textures in [`GpuDemPool`] for every
131/// `hillshade` and `color-relief` source drawn by the style.
132#[tracing::instrument(skip_all)]
133fn upload_dem_layers(
134    raster_resources: &mut RasterResources,
135    gpu_dem_pool: &mut GpuDemPool,
136    (device, queue): (&wgpu::Device, &crate::render::upload_queue::UploadQueue),
137    tiles: &Tiles,
138    dem_tiles: impl IntoIterator<Item = WorldTileCoords>,
139    budget: &mut FrameUploads,
140) -> bool {
141    let mut deferred = false;
142    'tiles: for coords in dem_tiles {
143        let Some(layers) = tiles.query::<&DemLayersComponent>(coords) else {
144            continue;
145        };
146        for (source, dem) in layers.iter() {
147            if !raster_resources.draws(source)
148                || raster_resources.dem_is_current(source, coords, dem)
149            {
150                continue;
151            }
152            let needs_gpu_write = gpu_dem_pool.needs_upload(source, coords, dem);
153            if needs_gpu_write && !budget.take(UploadKind::Image, dem.tile.pixels().len()) {
154                deferred = true;
155                break 'tiles;
156            }
157            let entry = if needs_gpu_write {
158                gpu_dem_pool.upload(device, queue, source, coords, dem)
159            } else if let Some(existing) = gpu_dem_pool.get(source, coords) {
160                existing
161            } else {
162                gpu_dem_pool.upload(device, queue, source, coords, dem)
163            };
164            raster_resources.bind_dem_texture(
165                device,
166                source,
167                coords,
168                &entry.texture,
169                dem.revision_key(),
170            );
171        }
172    }
173    deferred
174}

176/// Uploads an imagery tile stored at `coords` and returns its `wgpu::Texture` handle for mipmap
177/// generation.
178fn upload_imagery(
179    raster_resources: &mut RasterResources,
180    device: &wgpu::Device,
181    queue: &crate::render::upload_queue::UploadQueue,
182    coords: WorldTileCoords,
183    data: &AvailableRasterLayerData,
184) -> wgpu::Texture {
185    let (width, height) = data.image.dimensions();
186
187    // Raster style colors are sampled in the encoded color space, matching WebGL's default
188    // RGBA upload path. An sRGB texture view would decode the texels to linear values before
189    // writing them to the non-sRGB render target, making imagery too dark.
190    let texture = raster_resources.create_imagery_texture(device, width, height);
191
192    queue.write_texture(
193        wgpu::TexelCopyTextureInfo {
194            aspect: wgpu::TextureAspect::All,
195            texture: &texture.texture,
196            mip_level: 0,
197            origin: wgpu::Origin3d::ZERO,
198        },
199        &data.image,
200        wgpu::TexelCopyBufferLayout {
201            offset: 0,
202            bytes_per_row: Some(4 * width),
203            rows_per_image: Some(height),
204        },
205        texture.size,
206    );
207
208    let mips = texture.texture.clone();
209    raster_resources.bind_texture(device, &data.source, &coords, texture);
210    mips
211}
