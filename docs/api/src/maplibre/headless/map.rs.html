69/// A headless frame advances the frame clock by a nominal 60 Hz interval, so animated
70/// properties progress the same way in every run.
71const HEADLESS_FRAME_INTERVAL: Duration = Duration::from_millis(16);

81    /// Initializes the view from the style and builds plugins in their supplied order.
82    /// The renderer and host kernel must already be initialized.
83    ///
84    /// Sources that name a TileJSON `url` draw nothing until it is resolved; await
85    /// [`resolve_tile_json_sources`] with the kernel's [`Kernel::source_client`] first. It needs
86    /// no blocking executor, so a browser host awaits it in its async startup.
87    pub fn new(
88        mut style: Style,
89        mut renderer: Renderer,
90        kernel: Kernel<HeadlessEnvironment>,
91        plugins: Vec<Box<dyn Plugin<HeadlessEnvironment>>>,
92    ) -> Result<Self, MapError> {
93        style.resolve_global_state();
94        style.log_validation_errors();
95        let window_size = renderer.state().surface().size();
96
97        let view_state = initial_view_state(window_size, &style);
98
99        let mut world = World::default();
100        let mut schedule = Schedule::default();
101        let kernel = Rc::new(kernel);
102
103        for plugin in &plugins {
104            plugin.build(
105                &mut schedule,
106                kernel.clone(),
107                &mut world,
108                &mut renderer.render_graph,
109            );
110        }
111
112        Ok(Self {
113            kernel,
114            map_context: MapContext {
115                style,
116                view_state,
117                world,
118                renderer,
119            },
120            schedule,
121        })
122    }

151    /// Renders decoded source tiles together with DEM tiles for the style's terrain.
152    ///
153    /// DEM images are decoded with the terrain source's encoding; images that cannot be decoded
154    /// are skipped so the mesh falls back to an ancestor tile or sea level.
155    /// Inserts decoded DEM tiles, fills the borders between neighbours and settles the terrain
156    /// index with one schedule pass, so later tile selection sees the elevations they reveal.
157    pub fn load_dem_tiles(
158        &mut self,
159        dem_tiles: Vec<(WorldTileCoords, RgbaImage)>,
160    ) -> Result<(), HeadlessMapOperationError> {
161        let context = &mut self.map_context;
162        let unpack = dem_source(&context.style).map(|dem| dem.unpack);
163        let circumference = context.view_state.body().circumference_meters();
164        let tiles = &mut context.world.tiles;
165        let mut dem_coords = Vec::new();
166        for (coords, image) in dem_tiles {
167            let Some(unpack) = unpack else {
168                break;
169            };
170            let component = dem::dem_component(coords, &image, unpack, circumference);
171            tiles
172                .spawn_mut(coords)
173                .ok_or(HeadlessMapOperationError::InvalidTile { coords })?
174                .insert(component);
175            dem_coords.push(coords);
176        }
177        for coords in dem_coords {
178            backfill_neighbours(tiles, coords);
179        }
180        if let Err(error) = self.schedule.run(context) {
181            tracing::warn!(?error, "terrain index warm-up frame failed");
182        }
183        Ok(())
184    }

186    /// Loads processed vector, raster and DEM data, then runs `frame_count` offscreen frames.
187    /// A zero count is rejected before loading data. DEM decode failures use ancestor fallback;
188    /// invalid tile coordinates and schedule failures are returned with their cause.
189    pub fn render_frames_with_terrain(
190        &mut self,
191        layers: ProcessedLayers,
192        raster_layers: Vec<AvailableRasterLayerData>,
193        dem_tiles: Vec<(WorldTileCoords, RgbaImage)>,
194        frame_count: u8,
195    ) -> Result<(), HeadlessMapOperationError> {
196        if frame_count == 0 {
197            return Err(HeadlessMapOperationError::InvalidFrameCount);
198        }
199        if !dem_tiles.is_empty() {
200            self.load_dem_tiles(dem_tiles)?;
201        }
202        let context = &mut self.map_context;
203        let tiles = &mut context.world.tiles;
204        let ProcessedLayers {
205            vector,
206            symbols,
207            indexes,
208        } = layers;
209        for index in indexes {
210            tiles.geometry_index.index_tile(
211                &index.coords(),
212                index.source(),
213                (*index).to_tile_index(),
214            );
215        }
216
217        let mut layers_by_tile = BTreeMap::new();
218        for layer in vector {
219            layers_by_tile
220                .entry(layer.coords)
221                .or_insert_with(Vec::new)
222                .push(VectorLayerBucket::AvailableLayer(
223                    AvailableVectorLayerBucket {
224                        coords: layer.coords,
225                        source_layer: layer.layer_data.name,
226                        style_layer_id: layer.style_layer_id,
227                        buffer: layer.buffer,
228                        feature_indices: layer.feature_indices,
229                        feature_colors: layer.feature_colors,
230                        feature_sort_keys: layer.feature_sort_keys,
231                    },
232                ));
233        }
234
235        for (coords, layers) in layers_by_tile {
236            tiles
237                .spawn_mut(coords)
238                .ok_or(HeadlessMapOperationError::InvalidTile { coords })?
239                .insert(VectorLayerBucketComponent {
240                    done: true,
241                    failed: false,
242                    layers,
243                    overscaled_zoom: 0,
244                });
245        }
246
247        let mut symbols_by_tile = BTreeMap::new();
248        for layer in symbols {
249            symbols_by_tile
250                .entry(layer.coords())
251                .or_insert_with(Vec::new)
252                .push((*layer).to_bucket());
253        }
254        // A tile whose only layers are symbols is still a delivered tile, as when a worker
255        // reports it finished, or its labels would never be covered.
256        for coords in symbols_by_tile.keys() {
257            if tiles
258                .query::<&VectorLayerBucketComponent>(*coords)
259                .is_none()
260            {
261                tiles
262                    .spawn_mut(*coords)
263                    .ok_or(HeadlessMapOperationError::InvalidTile { coords: *coords })?
264                    .insert(VectorLayerBucketComponent {
265                        done: true,
266                        failed: false,
267                        layers: Vec::new(),
268                        overscaled_zoom: 0,
269                    });
270            }
271        }
272        for (coords, layers) in symbols_by_tile {
273            tiles
274                .spawn_mut(coords)
275                .ok_or(HeadlessMapOperationError::InvalidTile { coords })?
276                .insert(SymbolLayersDataComponent {
277                    layers,
278                    pending_assets: false,
279                });
280        }
281
282        self.load_raster_layers(raster_layers)?;
283        self.advance_frames(frame_count)
284    }

286    fn advance_frames(&mut self, frame_count: u8) -> Result<(), HeadlessMapOperationError> {
287        for _ in 0..frame_count {
288            self.map_context
289                .world
290                .resources
291                .get_or_init_mut::<FrameInput>()
292                .advance(HEADLESS_FRAME_INTERVAL);
293            self.schedule
294                .run(&mut self.map_context)
295                .map_err(|source| HeadlessMapOperationError::Schedule { source })?;
296        }
297        Ok(())
298    }

483        let context = &self.map_context;

500    /// Returns the tiles a raster source must make available for this view: its own covering
501    /// at its tile size and rounding, as the request system asks for them.
502    pub fn required_raster_tile_coords(
503        &self,
504        source_name: &str,
505    ) -> Result<Vec<WorldTileCoords>, ProjectionStateError> {
506        let source = crate::io::tile_sources::source_layer_groups(
507            &self.map_context.style,
508            crate::io::tile_sources::TileKind::Raster,
509        )
510        .into_iter()
511        .find(|group| {
512            group
513                .layers
514                .iter()
515                .any(|layer| layer.source.as_deref() == Some(source_name))
516        })
517        .map(|group| RasterSourceId::new(group.source_name));
518        source.map_or_else(
519            || Ok(Vec::new()),
520            |source| self.required_raster_source_tile_coords(&source),
521        )
522    }

601fn initial_view_state(window_size: crate::window::PhysicalSize, style: &Style) -> ViewState {
602    // The flat world may not be smaller than the viewport is tall, as GL JS constrains the zoom;
603    // a globe may be seen whole.
604    let globe = style
605        .projection
606        .as_ref()
607        .is_some_and(|specification| specification.projection_type.uses_globe_rendering(0.0));
608    let lowest_zoom = if globe {
609        f64::NEG_INFINITY
610    } else {
611        (f64::from(window_size.height()) / crate::coords::TILE_SIZE).log2()
612    };
613    let zoom = Zoom::new(style.zoom.unwrap_or_default().max(lowest_zoom));
614    let center = style.center.unwrap_or_default();
615    let mut center = WorldCoords::from_lat_lon(LatLon::new(center[1], center[0]), zoom);
616    if !globe {
617        // The view may not show past the poles of a flat world, as GL JS constrains the centre.
618        let world = crate::coords::TILE_SIZE * 2_f64.powf(zoom.value());
619        let half = f64::from(window_size.height()) / 2.0;
620        if world > 2.0 * half {
621            center.y = center.y.clamp(half, world - half);
622        }
623    }
624    let mut view_state = ViewState::new(
625        window_size,
626        center,
627        zoom,
628        cgmath::Deg(style.pitch.unwrap_or_default()),
629        cgmath::Rad(0.6435011087932844),
630    );
631    view_state.take_style_center(style);
632    view_state
633        .camera_mut()
634        .set_bearing(cgmath::Deg(style.bearing.unwrap_or_default()));
635    view_state
636        .camera_mut()
637        .set_roll(cgmath::Deg(style.roll.unwrap_or_default()));
638    view_state
639}
