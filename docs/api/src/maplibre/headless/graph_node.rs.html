10/// Node which copies the contents of the GPU-side texture in [`BufferedTextureHead`] to an
11/// unmapped GPU-side buffer. This buffer will be mapped in
12/// [`crate::render::stages::write_surface_buffer_stage::WriteSurfaceBufferStage`].
13#[derive(Default)]
14pub struct CopySurfaceBufferNode;
15
16impl Node for CopySurfaceBufferNode {
17    fn input(&self) -> Vec<SlotInfo> {
18        vec![]
19    }
20
21    fn update(&mut self, _state: &mut RenderResources) {}
22
23    fn run(
24        &self,
25        _graph: &mut RenderGraphContext,
26        RenderContext {
27            command_encoder, ..
28        }: &mut RenderContext,
29        state: &RenderResources,
30        _world: &World,
31    ) -> Result<(), NodeRunError> {
32        let surface = state.surface();
33        match surface.head() {
34            Head::Headed(_) => {}
35            Head::Headless(buffered_texture) => {
36                let size = surface.size();
37                command_encoder.copy_texture_to_buffer(
38                    buffered_texture.copy_texture(),
39                    wgpu::TexelCopyBufferInfo {
40                        buffer: buffered_texture.buffer(),
41                        layout: wgpu::TexelCopyBufferLayout {
42                            offset: 0,
43                            bytes_per_row: Some(buffered_texture.bytes_per_row()),
44                            rows_per_image: None,
45                        },
46                    },
47                    wgpu::Extent3d {
48                        width: size.width(),
49                        height: size.height(),
50                        depth_or_array_layers: 1,
51                    },
52                );
53            }
54        }
55
56        Ok(())
57    }
58}
