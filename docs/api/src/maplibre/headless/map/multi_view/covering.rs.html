44pub(super) fn build_slot_view_state(
45    base: &ViewState,
46    style: &Style,
47    world: &World,
48    slot_size: crate::window::PhysicalSize,
49    scale_factor: f64,
50    (opaque_environment, request_overscan): (bool, f64),
51    view: &SlotView,
52) -> Result<ViewState, super::MultiViewError> {
53    let projection = style
54        .projection
55        .as_ref()
56        .map_or(ProjectionType::Mercator, |spec| {
57            spec.projection_type.clone()
58        });
59    let mut state = base.clone();
60    state.resize(slot_size.to_logical(scale_factor));
61    state.set_opaque_environment(opaque_environment);
62    match *view {
63        SlotView::External(external) => {
64            state
65                .set_external_view(external, &projection)
66                .map_err(|source| super::MultiViewError::InvalidExternalView { source })?;
67            state.set_request_overscan(request_overscan);
68        }
69        SlotView::Camera {
70            center,
71            zoom,
72            bearing_deg,
73            pitch_deg,
74        } => {
75            state.clear_external_view();
76            state
77                .jump_to(CameraOptions {
78                    center: Some(center),
79                    zoom: Some(zoom.value()),
80                    bearing: Some(cgmath::Deg(bearing_deg)),
81                    pitch: Some(cgmath::Deg(pitch_deg)),
82                    roll: Some(cgmath::Deg(0.0)),
83                    elevation: None,
84                })
85                .map_err(|source| super::MultiViewError::InvalidCamera { source })?;
86            crate::terrain::elevation::ground_center(&mut state, world);
87        }
88    }
89    Ok(state)
90}

92/// Computes the unioned [`MultiViewExtractCovering`], [`DrapeRequests`], and initial
93/// [`SlotResidencyFloor`] across all active `Independent` slots and skipped registered slots.
94pub(super) fn build_multi_view_coverings(
95    style: &Style,
96    world: &mut World,
97    registry: &mut BTreeMap<ViewSlotId, SlotState>,
98    active: &[(ViewSlotEntry, ViewState)],
99) -> (MultiViewExtractCovering, SlotResidencyFloor) {
100    let active_ids: FxHashSet<ViewSlotId> = active.iter().map(|(entry, _)| entry.slot).collect();
101    let mut extract = MultiViewExtractCovering {
102        active: true,
103        ..Default::default()
104    };
105    // Every view that holds drapes is granted textures, the views the host ranks first before
106    // the others: those drawn this frame with a covering of their own, and those left out of
107    // it that keep what they last drew.
108    let mut draping: Vec<(i32, ViewSlotId)> = registry
109        .iter()
110        .filter(|(slot, state)| {
111            active.iter().any(|(entry, _)| {
112                entry.slot == **slot && matches!(entry.policy, CoveringPolicy::Independent)
113            }) || (!active_ids.contains(slot) && !state.last_retained_drapes.is_empty())
114        })
115        .map(|(slot, state)| (state.drape_priority, *slot))
116        .collect();
117    draping.sort_by_key(|(priority, slot)| (std::cmp::Reverse(*priority), *slot));
118    let draping: Vec<ViewSlotId> = draping.into_iter().map(|(_, slot)| slot).collect();
119    let allowed = world
120        .resources
121        .get::<crate::render::memory_budget::MemoryBudget>()
122        .copied()
123        .unwrap_or_default()
124        .drape_textures_allowed();
125    world
126        .resources
127        .get_or_init_mut::<DrapeShares>()
128        .grant(&draping, allowed);
129    let mut floor = SlotResidencyFloor::default();
130    let mut vector_seen = FxHashSet::default();
131    let mut terrain_seen = FxHashSet::default();
132    let mut drape_requests_seen = FxHashSet::default();
133    let mut drape_requests = Vec::new();
134    let mut raster_by_source: BTreeMap<
135        RasterSourceId,
136        (FxHashSet<WorldTileCoords>, Vec<WorldTileCoords>),
137    > = BTreeMap::new();
138
139    for (slot_id, slot_state) in registry.iter() {
140        if !active_ids.contains(slot_id) {
141            floor.drawn_count = floor.drawn_count.max(slot_state.last_drawn_count);
142            floor
143                .retained_tiles
144                .extend(slot_state.last_retained_tiles.iter().copied());
145            floor
146                .retained_drapes
147                .extend(slot_state.last_retained_drapes.iter().copied());
148            extract
149                .skipped_wanted_vector
150                .extend(slot_state.last_wanted_vector.iter().copied());
151            extract
152                .skipped_wanted_raster
153                .extend(slot_state.last_wanted_raster.iter().copied());
154            extract
155                .skipped_wanted_dem
156                .extend(slot_state.last_wanted_dem.iter().copied());
157        }
158    }
159
160    let dem_tile_size = dem_source(style).map(|dem| dem.tile_size);
161    let procedural_relief = world
162        .resources
163        .get::<crate::render::settings::TerrainDrapeMode>()
164        .copied()
165        .or_else(|| {
166            match world.resources.get::<crate::render::eventually::Eventually<
167                    crate::terrain::resources::TerrainResources,
168                >>() {
169                    Some(crate::render::eventually::Eventually::Initialized(terrain)) => {
170                        Some(terrain.drape_mode())
171                    }
172                    _ => None,
173                }
174        })
175        .is_some_and(|mode| mode.is_procedural_relief());
176
177    for (entry, slot_view) in active {
178        if matches!(entry.policy, CoveringPolicy::StereoReuse(_)) {
179            continue;
180        }
181        let lod_history = registry
182            .get(&entry.slot)
183            .map(|state| state.lod_history.clone())
184            .unwrap_or_default();
185        world.resources.insert(lod_history);
186        world.resources.insert(DrapeView(Some(entry.slot)));
187
188        let level = slot_view.zoom().zoom_level(DEFAULT_TILE_SIZE);
189        extract.max_view_level = extract.max_view_level.max(u8::from(level));
190
191        let mut slot_wanted_vec = FxHashSet::default();
192        if let Ok(Some(region)) =
193            view_region_for_projection(style, slot_view, world, level, ViewStatePadding::Loose)
194        {
195            for coords in region.iter() {
196                slot_wanted_vec.insert(coords);
197                if vector_seen.insert(coords) {
198                    extract.vector_tiles.push(coords);
199                }
200            }
201        }
202
203        let mut slot_wanted_ras = FxHashSet::default();
204        if let Ok(regions) = raster_source_regions(style, slot_view, world, ViewStatePadding::Loose)
205        {
206            for (source_id, tiles) in regions {
207                let (seen, list) = raster_by_source.entry(source_id).or_default();
208                for coords in tiles {
209                    slot_wanted_ras.insert(coords);
210                    if seen.insert(coords) {
211                        list.push(coords);
212                    }
213                }
214            }
215        }
216
217        let mut slot_wanted_dem = FxHashSet::default();
218        let mut prepared_for_slot = None;
219        if let Some(tile_size) = dem_tile_size {
220            if let Ok(region_opt) =
221                terrain_region(style, slot_view, world, tile_size, ViewStatePadding::Tight)
222            {
223                if let Some(region) = &region_opt {
224                    for coords in region.iter() {
225                        slot_wanted_dem.insert(coords);
226                        if terrain_seen.insert(coords) {
227                            extract.terrain_tiles.push(coords);
228                        }
229                    }
230                }
231                if !procedural_relief {
232                    let key = crate::terrain::coverage::PreparedTerrainKey::capture(
233                        slot_view, world, tile_size,
234                    );
235                    let prepared = crate::terrain::coverage::PreparedTerrainRegion {
236                        key,
237                        region: region_opt,
238                    };
239                    prepared_for_slot = Some(prepared.clone());
240                    world.resources.insert(prepared);
241                }
242            }
243            if !procedural_relief {
244                if let Ok(specs) = target_specs(style, slot_view, world) {
245                    let mut slot_drapes = FxHashSet::default();
246                    for spec in &specs {
247                        slot_drapes.insert(spec.coords);
248                        floor.retained_drapes.insert(spec.coords);
249                        slot_wanted_vec.insert(spec.coords);
250                        slot_wanted_dem.insert(spec.coords);
251                        if drape_requests_seen.insert(spec.coords) {
252                            drape_requests.push(spec.coords);
253                        }
254                    }
255                    if let Some(state) = registry.get_mut(&entry.slot) {
256                        state.last_retained_drapes = slot_drapes;
257                    }
258                }
259            } else if let Some(state) = registry.get_mut(&entry.slot) {
260                state.last_retained_drapes.clear();
261            }
262        }
263
264        let slot_in_use = tiles_in_use_from_loose(
265            world,
266            style,
267            slot_wanted_vec.iter().copied(),
268            slot_wanted_ras.iter().copied(),
269        );
270        floor.retained_tiles.extend(slot_in_use.iter().copied());
271        if let Some(state) = registry.get_mut(&entry.slot) {
272            state.prepared_terrain = prepared_for_slot;
273            state.frame_loose_in_use = slot_in_use;
274            state.last_wanted_vector = slot_wanted_vec;
275            state.last_wanted_raster = slot_wanted_ras;
276            state.last_wanted_dem = slot_wanted_dem;
277        }
278    }
279
280    extract.raster_regions = raster_by_source
281        .into_iter()
282        .map(|(source_id, (_, list))| (source_id, list))
283        .collect::<RasterCoverings>();
284    if !drape_requests.is_empty() {
285        world.resources.insert(DrapeRequests(drape_requests));
286    }
287    (extract, floor)
288}

290/// Captures a slot's post-queue covering, LOD history, stereo snapshot, and residency sets.
291pub(super) fn record_slot_post_render(
292    style: &Style,
293    _slot_view: &ViewState,
294    world: &World,
295    slot_state: &mut SlotState,
296    floor: &mut SlotResidencyFloor,
297) {
298    let shared = world.resources.get::<SharedCovering>().cloned();
299    let surface_tiles = world
300        .resources
301        .get::<SurfaceTiles>()
302        .cloned()
303        .unwrap_or_default();
304    let surface_copies = world
305        .resources
306        .get::<SurfaceCopies>()
307        .cloned()
308        .unwrap_or_default();
309    let terrain_frame = world
310        .resources
311        .get::<TerrainEyeFrame>()
312        .cloned()
313        .unwrap_or_default();
314
315    slot_state.lod_history = FrameLodHistory {
316        view: crate::projection::lod_history::LodHistory::new(
317            shared.as_ref().and_then(|s| s.tiles()).unwrap_or(&[]),
318        ),
319        raster: shared
320            .as_ref()
321            .map(|s| {
322                let _ = s;
323                FxHashMap::default()
324            })
325            .unwrap_or_default(),
326        terrain: crate::projection::lod_history::LodHistory::new(&surface_tiles.0),
327    };
328    if let Some(current_history) = world.resources.get::<FrameLodHistory>() {
329        let _ = current_history;
330    }
331    slot_state.stereo_snapshot = Some(StereoSnapshot {
332        shared,
333        surface_tiles,
334        surface_copies,
335        terrain_frame,
336    });
337    let drawn = drawn_tiles(world);
338    slot_state.last_drawn_count = drawn.len();
339    let mut retained = std::mem::take(&mut slot_state.frame_loose_in_use);
340    retained.extend(drawn);
341    let retained = finish_tiles_in_use(world, style, retained);
342    if let Some(requests) = world.resources.get::<DrapeRequests>() {
343        slot_state.last_retained_drapes = requests.0.iter().copied().collect();
344    }
345    slot_state.last_retained_tiles = retained.clone();
346    floor.drawn_count = floor.drawn_count.max(slot_state.last_drawn_count);
347    floor.retained_tiles.extend(retained);
348    floor
349        .retained_drapes
350        .extend(slot_state.last_retained_drapes.iter().copied());
351}
