1//! What a host of a headless map reads to render on demand.
2
3use super::HeadlessMap;
4use crate::{
5    io::apc::AsyncProcedureCall,
6    render::{
7        frame_signals::{FrameStats, RedrawReasons, ResourceReady},
8        frame_trace::{FrameTrace, FrameTraceSlot},
9    },
10};
11
12impl HeadlessMap {
13    /// Whether the map should draw another frame; see [`crate::context::MapContext::needs_redraw`].
14    /// Also true while worker results wait to be applied, which on the web arrive on this thread
15    /// between frames.
16    pub fn needs_redraw(&self) -> bool {
17        self.redraw_reasons().any()
18    }
19
20    /// What [`Self::needs_redraw`] asks for a frame for; a host logs it when the map does not
21    /// settle.
22    pub fn redraw_reasons(&self) -> RedrawReasons {
23        RedrawReasons {
24            worker_results: self.kernel.apc().has_arrivals(),
25            ..self.map_context.redraw_reasons()
26        }
27    }
28
29    /// Whether the map has work ready to draw right now: dirty or animating frame state, a moved
30    /// camera, an expired retry backoff, or completed worker messages waiting in the APC queue.
31    ///
32    /// Unlike [`Self::needs_redraw`], this stays `false` while remote tile fetches are merely in
33    /// flight over the network with no arrived worker output yet, so VSync-driven hosts (such as
34    /// standalone or embedded EFB displays) do not spin at 120 Hz awaiting I/O.
35    pub fn has_ready_work(&self) -> bool {
36        self.map_context.has_ready_work() || self.kernel.apc().has_arrivals()
37    }
38
39    /// The statistics of the last frame drawn.
40    pub fn last_frame_stats(&self) -> FrameStats {
41        self.map_context.last_frame_stats()
42    }
43
44    /// The resources that became ready since the last call.
45    pub fn take_ready_resources(&mut self) -> Vec<ResourceReady> {
46        self.map_context.take_ready_resources()
47    }
48
49    /// Starts recording a timeline of the last `capacity` frames, each under the number of its
50    /// [`FrameStats`], to which the host adds its own measurements; replaces a running one.
51    pub fn enable_frame_trace(&mut self, capacity: usize) {
52        self.map_context
53            .world
54            .resources
55            .insert(FrameTraceSlot(Some(FrameTrace::new(capacity))));
56    }
57
58    /// Stops recording and drops the timeline.
59    pub fn disable_frame_trace(&mut self) {
60        self.map_context
61            .world
62            .resources
63            .insert(FrameTraceSlot(None));
64    }
65
66    /// The running frame timeline, if one was enabled.
67    pub fn frame_trace_mut(&mut self) -> Option<&mut FrameTrace> {
68        self.map_context
69            .world
70            .resources
71            .get_mut::<FrameTraceSlot>()
72            .and_then(|slot| slot.0.as_mut())
73    }
74}
