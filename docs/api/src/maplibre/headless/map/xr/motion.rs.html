1//! Bounded camera prediction for terrain requests; displayed poses remain unpredicted.
2use std::time::Duration;
3
4use cgmath::{InnerSpace, Matrix3, Matrix4, Quaternion, SquareMatrix, Vector3};
5
6use crate::{
7    projection::ProjectionType,
8    render::view_state::{ExternalView, ViewState},
9};
10
11const SAMPLE_INTERVAL: f64 = 0.1;
12const LOOK_AHEAD: f64 = 0.35;
13const MAX_TURN: f64 = 20.0 * std::f64::consts::PI / 180.0;
14
15#[derive(Default)]
16pub(super) struct MotionPrefetch {
17    sample: Option<(Duration, ExternalView)>,
18    ahead: Option<ViewState>,
19}
20
21impl MotionPrefetch {
22    pub(super) fn update(
23        &mut self,
24        timestamp: Duration,
25        current: ExternalView,
26        base: &ViewState,
27        projection: &ProjectionType,
28    ) -> Option<ViewState> {
29        let Some((time, previous)) = self.sample else {
30            self.sample = Some((timestamp, current));
31            return None;
32        };
33        let dt = timestamp.saturating_sub(time).as_secs_f64();
34        if timestamp > time && dt < SAMPLE_INTERVAL {
35            return self.ahead.clone();
36        }
37        self.sample = Some((timestamp, current));
38        self.ahead = (SAMPLE_INTERVAL..=0.5)
39            .contains(&dt)
40            .then(|| predict(previous, current, dt, base, projection))
41            .flatten();
42        self.ahead.clone()
43    }
44}
45
46fn predict(
47    previous: ExternalView,
48    current: ExternalView,
49    dt: f64,
50    base: &ViewState,
51    projection: &ProjectionType,
52) -> Option<ViewState> {
53    let now = current.view.invert()?;
54    let before = previous.view.invert()?;
55    let scale = now.x.truncate().magnitude();
56    if (before.x.truncate().magnitude() / scale - 1.0).abs() > 0.05 {
57        return None;
58    }
59    let orientation = |matrix: Matrix4<f64>| {
60        Quaternion::from(Matrix3::from_cols(
61            matrix.x.truncate().normalize(),
62            matrix.y.truncate().normalize(),
63            matrix.z.truncate().normalize(),
64        ))
65        .normalize()
66    };
67    let (rotation, old_rotation) = (orientation(now), orientation(before));
68    let angle = 2.0 * rotation.dot(old_rotation).abs().clamp(0.0, 1.0).acos();
69    let turn_factor = (LOOK_AHEAD / dt).min(MAX_TURN / angle.max(1e-9));
70    let ahead_rotation = old_rotation.slerp(rotation, 1.0 + turn_factor).normalize();
71    let offset = anchor_offset(previous, current, base.body().radius_meters)?;
72    let position = now.w.truncate();
73    let travel = (position - before.w.truncate() - offset) * (LOOK_AHEAD / dt);
74    let distance = travel.magnitude();
75    let reach = position.z.abs().max(100.0) * 2.0;
76    let travel = travel * (reach / distance.max(reach));
77    if angle < 0.002 && distance < 1.0 {
78        return None;
79    }
80    let local_from_eye = Matrix4::from_translation(position + travel)
81        * Matrix4::from(ahead_rotation)
82        * Matrix4::from_scale(scale);
83    let mut ahead = base.clone();
84    ahead
85        .set_external_view(
86            ExternalView {
87                view: local_from_eye.invert()?,
88                ..current
89            },
90            projection,
91        )
92        .ok()?;
93    if projection.uses_globe_rendering(ahead.zoom().value()) {
94        return None;
95    }
96    Some(ahead)
97}

99fn anchor_offset(
100    previous: ExternalView,
101    current: ExternalView,
102    radius: f64,
103) -> Option<Vector3<f64>> {
104    let before = previous.anchor.position;
105    let now = current.anchor.position;
106    let longitude = (before.longitude - now.longitude + 180.0).rem_euclid(360.0) - 180.0;
107    if longitude.abs() > 0.5 || (before.latitude - now.latitude).abs() > 0.5 {
108        return None;
109    }
110    let latitude = now.latitude.to_radians();
111    let north = |degrees: f64| degrees.to_radians().tan().asinh();
112    Some(Vector3::new(
113        longitude.to_radians() * radius * latitude.cos(),
114        (north(before.latitude) - north(now.latitude)) * radius * latitude.cos(),
115        previous.anchor.altitude_meters - current.anchor.altitude_meters,
116    ))
117}
