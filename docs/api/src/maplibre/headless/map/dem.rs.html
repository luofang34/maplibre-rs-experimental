16impl HeadlessMap {
17    /// Renders decoded source tiles together with DEM tiles for the style's terrain.
18    ///
19    /// DEM images are decoded with the terrain source's encoding; images that cannot be decoded
20    /// are skipped so the mesh falls back to an ancestor tile or sea level.
21    /// Inserts decoded DEM tiles, fills the borders between neighbours and settles the terrain
22    /// index with one schedule pass, so later tile selection sees the elevations they reveal.
23    pub fn load_dem_tiles(
24        &mut self,
25        dem_tiles: Vec<(WorldTileCoords, RgbaImage)>,
26    ) -> Result<(), HeadlessMapOperationError> {
27        let context = &mut self.map_context;
28        let unpack = dem_source(&context.style).map(|dem| dem.unpack);
29        let circumference = context.view_state.body().circumference_meters();
30        let tiles = &mut context.world.tiles;
31        let mut dem_coords = Vec::new();
32        for (coords, image) in dem_tiles {
33            let Some(unpack) = unpack else {
34                break;
35            };
36            let component = dem_component(coords, &image, unpack, circumference);
37            tiles
38                .spawn_mut(coords)
39                .ok_or(HeadlessMapOperationError::InvalidTile { coords })?
40                .insert(component);
41            dem_coords.push(coords);
42        }
43        for coords in dem_coords {
44            backfill_neighbours(tiles, coords);
45            crate::terrain::dem_store::sync_terrain_dem_to_store(tiles, &context.style, coords);
46        }
47        if let Err(error) = crate::render::frame::run_frame(&mut self.schedule, context) {
48            tracing::warn!(?error, "terrain index warm-up frame failed");
49        }
50        Ok(())
51    }
52}

54/// Decodes one DEM image with seam-spike removal at the tile's ground sample distance.
55///
56/// An unusable image becomes a missing tile, so the mesh falls back to an ancestor tile or sea level.
57pub(super) fn dem_component(
58    coords: WorldTileCoords,
59    image: &RgbaImage,
60    unpack: [f64; 4],
61    circumference_meters: f64,
62) -> DemTileComponent {
63    let sample = sample_metres(coords, image.width(), circumference_meters);
64    match DemTile::from_image_despiked(image, unpack, sample) {
65        Ok(tile) => DemTileComponent::Loaded(LoadedDem::new(tile)),
66        Err(error) => {
67            tracing::warn!(%coords, %error, "DEM tile image is unusable");
68            DemTileComponent::Missing
69        }
70    }
