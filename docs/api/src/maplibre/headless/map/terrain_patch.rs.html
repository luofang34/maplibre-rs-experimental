18/// Immutable CPU + GPU snapshot of a resident terrain surface patch (`129 × 129` vertices)
19/// exported for external volumetric / XVS / custom 3D consumers.
20#[derive(Clone, Debug)]
21pub struct TerrainPatchSnapshot {
22    /// Coordinates of the queried terrain surface patch.
23    pub coords: WorldTileCoords,
24    /// Coordinates of the loaded DEM tile backing `coords` (`dem_coords.z <= coords.z`).
25    pub dem_coords: WorldTileCoords,
26    /// Shared immutable decoded DEM raster grid.
27    pub dem_tile: Arc<DemTile>,
28    /// Number of vertices along each edge of `stitched_heights_meters` (`129`).
29    pub grid_side: u32,
30    /// Row-major `grid_side × grid_side` exaggerated elevations in metres MSL, stitched along
31    /// tile borders to match neighbor LODs without T-junction cracks.
32    pub stitched_heights_meters: Arc<[f32]>,
33    /// Minimum vertex elevation in `stitched_heights_meters` (metres MSL, exaggerated).
34    pub min_elevation_meters: f32,
35    /// Maximum vertex elevation in `stitched_heights_meters` (metres MSL, exaggerated).
36    pub max_elevation_meters: f32,
37    /// Vertical exaggeration factor applied to `stitched_heights_meters`.
38    pub exaggeration: f32,
39    /// `true` when rendered neighbor `EdgeHeights` were applied along the patch boundary.
40    pub stitched_to_neighbors: bool,
41    /// Resident mipmapped drape texture covering `coords` (its own or nearest resident ancestor)
42    /// when [`TerrainDrapeMode::Draped`] is active; `None` in [`TerrainDrapeMode::ProceduralRelief`].
43    pub drape_texture: Option<wgpu::Texture>,
44}
45
46impl TerrainPatchSnapshot {
47    /// Returns the stitched elevation (metres MSL) at `(col, row)` in `0..grid_side`.
48    pub fn height_at_grid(&self, col: u32, row: u32) -> Option<f32> {
49        if col >= self.grid_side || row >= self.grid_side {
50            return None;
51        }
52        let idx = (row as usize) * (self.grid_side as usize) + (col as usize);
53        self.stitched_heights_meters.get(idx).copied()
54    }
55}
56
57impl HeadlessMap {
58    /// Returns the active [`TerrainDrapeMode`] of this map.
59    pub fn terrain_drape_mode(&self) -> TerrainDrapeMode {
60        self.map_context
61            .world
62            .resources
63            .get::<TerrainDrapeMode>()
64            .copied()
65            .unwrap_or_else(|| {
66                self.map_context
67                    .renderer
68                    .settings
69                    .effective_terrain_drape_mode()
70            })
71    }
72
73    /// Switches between baked 2D style drapes ([`TerrainDrapeMode::Draped`]) and zero-drape
74    /// procedural DEM shading ([`TerrainDrapeMode::ProceduralRelief`]).
75    ///
76    /// Switching to `ProceduralRelief` immediately releases any allocated `DrapeScratch` and
77    /// `DrapeCache` textures.
78    pub fn set_terrain_drape_mode(&mut self, mode: TerrainDrapeMode) {
79        self.map_context.renderer.settings.terrain_drape_mode = mode;
80        if let TerrainDrapeMode::Draped { edge_px } = mode {
81            self.map_context.renderer.settings.terrain_drape_size = edge_px;
82        }
83        self.map_context.world.resources.insert(mode);
84        if let Some(Initialized(terrain)) = self
85            .map_context
86            .world
87            .resources
88            .get_mut::<Eventually<TerrainResources>>()
89        {
90            terrain.set_drape_mode(&self.map_context.renderer.device, mode);
91        }
92        crate::render::frame_signals::mark_dirty(&mut self.map_context.world);
93    }
94
95    /// Updates the TAWS reference altitude (metres MSL) in [`TerrainDrapeMode::ProceduralRelief`]
96    /// without reallocating GPU resources. Has no effect when in [`TerrainDrapeMode::Draped`].
97    pub fn set_procedural_relief_reference_altitude(&mut self, altitude_meters: Option<f32>) {
98        if let TerrainDrapeMode::ProceduralRelief(mut style) = self.terrain_drape_mode() {
99            style.reference_altitude_meters = altitude_meters;
100            self.set_terrain_drape_mode(TerrainDrapeMode::ProceduralRelief(style));
101        } else {
102            let _ = ProceduralReliefStyle::default();
103        }
104    }
105
106    /// Samples `(bilinear_msl_meters, cell_max_msl_meters)` in unexaggerated orthometric metres
107    /// at normalized `[0, 1]` tile coordinates `(u, v)` on `coords`'s `128 × 128` terrain mesh
108    /// without allocating a full `129 × 129` [`TerrainPatchSnapshot`].
109    pub fn sample_resident_dem_grid_cell(
110        &self,
111        coords: WorldTileCoords,
112        u: f64,
113        v: f64,
114    ) -> Option<(f64, f64)> {
115        let context = &self.map_context;
116        let fallback_dem =
117            dem_source(&context.style).map(|dem| (dem.minzoom, dem.maxzoom, dem.exaggeration));
118        let empty_index = TerrainCoverageIndex::default();
119        let index = context
120            .world
121            .resources
122            .get::<TerrainCoverageIndex>()
123            .unwrap_or(&empty_index);
124        index.sample_stitched_patch_cell_msl(&context.world.tiles, coords, (u, v), fallback_dem)
125    }
126
127    /// Exports a stitched `129 × 129` (`TERRAIN_MESH_SIZE + 1`) elevation grid and optional
128    /// resident drape texture for `coords` if a covering DEM tile is resident in `World`.
129    pub fn resident_terrain_patch(&self, coords: WorldTileCoords) -> Option<TerrainPatchSnapshot> {
130        let context = &self.map_context;
131        let fallback_dem =
132            dem_source(&context.style).map(|dem| (dem.minzoom, dem.maxzoom, dem.exaggeration));
133        let empty_index = TerrainCoverageIndex::default();
134        let index = context
135            .world
136            .resources
137            .get::<TerrainCoverageIndex>()
138            .unwrap_or(&empty_index);
139        let grid = index.stitched_patch_grid(&context.world.tiles, coords, fallback_dem)?;
140
141        let drape_texture = if self.terrain_drape_mode().is_procedural_relief() {
142            None
143        } else {
144            match context
145                .world
146                .resources
147                .get::<Eventually<TerrainResources>>()
148            {
149                Some(Initialized(terrain)) => {
150                    let mut cur = Some(coords);
151                    let mut found = None;
152                    while let Some(c) = cur {
153                        if let Some(tex) = terrain.drape_texture(c) {
154                            found = Some(tex.texture.clone());
155                            break;
156                        }
157                        cur = c.get_parent();
158                    }
159                    found
160                }
161                _ => None,
162            }
163        };
164
165        Some(TerrainPatchSnapshot {
166            coords,
167            dem_coords: grid.dem_coords,
168            dem_tile: grid.dem_tile,
169            grid_side: TERRAIN_MESH_SIZE + 1,
170            stitched_heights_meters: grid.heights_meters,
171            min_elevation_meters: grid.min_meters,
172            max_elevation_meters: grid.max_meters,
173            exaggeration: grid.exaggeration,
174            stitched_to_neighbors: grid.stitched_to_neighbors,
175            drape_texture,
176        })
177    }
