23/// Why a frame for a head-mounted display could not be drawn.
24#[derive(Error, Debug)]
25pub enum XrFrameError {
26    /// An eye's pose has no inverse, so it looks from nowhere.
27    #[error("eye {index} has a singular pose")]
28    SingularEye {
29        /// Position of the eye in the frame.
30        index: usize,
31    },
32    /// An external view is invalid; the host can keep its complete stereo frame.
33    #[error("eye {index} has an invalid external view")]
34    InvalidView {
35        /// Position of the eye in the frame.
36        index: usize,
37        /// Validation failure from the supplied placement or frustum.
38        #[source]
39        source: crate::render::view_state::ExternalViewError,
40    },
41    /// A stage of the schedule failed while drawing an eye.
42    #[error("eye {index} could not be drawn")]
43    Eye {
44        /// Position of the eye in the frame.
45        index: usize,
46        /// The stage failure.
47        #[source]
48        source: StageError,
49    },
50}

52impl HeadlessMap {
53    /// Resize the render attachments and viewport without discarding supplied tiles. The
54    /// display keeps the pixel ratio [`Self::set_pixel_ratio`] gave it, so the viewport is the
55    /// new size in layout pixels.
56    pub fn resize(&mut self, size: crate::window::PhysicalSize) {
57        let ratio = self
58            .map_context
59            .world
60            .resources
61            .get::<crate::sdf::provided::DisplayPixelRatio>()
62            .map_or(1.0, |ratio| f64::from(ratio.0));
63        self.map_context.resize(size, ratio);
64    }
65
66    /// Draws every eye of the frame into its own target, from the placement the host chose.
67    ///
68    /// Tile ingestion runs once, before the first eye. Later eyes keep its tile content and
69    /// terrain refinement while updating their own projection and render targets.
70    /// An eye without a colour target draws into the map's own texture, which then holds
71    /// the last such eye.
72    pub fn run_xr_frame(&mut self, frame: XrFrame) -> Result<(), XrFrameError> {
73        let views = self.validate_views(&frame)?;
74        self.map_context
75            .view_state
76            .set_opaque_environment(frame.opaque_environment);
77        self.set_motion_prefetch(&frame, views.first().copied());
78        let resources = &mut self.map_context.world.resources;
79        let frame_number = resources
80            .get::<EyeInFrame>()
81            .map_or(0, |eye| eye.frame.wrapping_add(1));
82        for (index, (eye, view)) in frame.eyes.into_iter().zip(views).enumerate() {
83            // The first eye selects the frame's tiles; the others draw the same ones.
84            self.map_context.world.resources.insert(EyeInFrame {
85                index,
86                frame: frame_number,
87            });
88            let input = self.frame_input_mut();
89            input.timestamp = frame.timestamp;
90            input.view = ViewSource::External(view);
91            input.request_overscan = frame.request_overscan;
92
93            let resources = &mut self.map_context.renderer.resources;
94            if let Some(color) = eye.target.color {
95                resources.render_target = Eventually::Initialized(TextureView::from(color));
96            }
97            resources.eye_depth_target = eye.target.depth;
98            let result = if index == 0 {
99                self.schedule.run_once(&mut self.map_context)
100            } else {
101                frame_input_system(&mut self.map_context)
102                    .map_err(StageError::from)
103                    .and_then(|()| {
104                        self.schedule.run_stages(&mut self.map_context, |label| {
105                            label != &RenderStageLabel::Extract as &dyn crate::schedule::StageLabel
106                        })
107                    })
108            };
109            let resources = &mut self.map_context.renderer.resources;
110            resources.eye_depth_target = None;
111            // The graph runner takes the target after a frame; a failed frame must not leave
112            // the eye's texture behind for the next one.
113            resources.render_target.take();
114            if let Err(source) = result {
115                self.map_context.world.resources.insert(EyeInFrame {
116                    index: 0,
117                    frame: frame_number,
118                });
119                return Err(XrFrameError::Eye { index, source });
120            }
121        }
122        self.map_context.world.resources.insert(EyeInFrame {
123            index: 0,
124            frame: frame_number,
125        });
126        Ok(())
127    }
128
129    fn validate_views(
130        &self,
131        frame: &XrFrame,
132    ) -> Result<Vec<crate::render::view_state::ExternalView>, XrFrameError> {
133        let projection = self
134            .map_context
135            .style
136            .projection
137            .as_ref()
138            .map_or(ProjectionType::Mercator, |spec| {
139                spec.projection_type.clone()
140            });
141        let mut checked = self.map_context.view_state.clone();
142        frame
143            .eyes
144            .iter()
145            .enumerate()
146            .map(|(index, eye)| {
147                let view = frame
148                    .placement
149                    .view_from(eye.world_from_eye, eye.frustum)
150                    .ok_or(XrFrameError::SingularEye { index })?;
151                checked
152                    .set_external_view(view, &projection)
153                    .map_err(|source| XrFrameError::InvalidView { index, source })?;
154                Ok(view)
155            })
156            .collect()
157    }
158
159    fn set_motion_prefetch(
160        &mut self,
161        frame: &XrFrame,
162        view: Option<crate::render::view_state::ExternalView>,
163    ) {
164        if frame.prefetch.is_some() {
165            self.map_context
166                .world
167                .resources
168                .insert(motion::MotionPrefetch::default());
169            self.set_prefetch(frame.prefetch.as_ref(), frame.eyes.first());
170            return;
171        }
172        let projection = self
173            .map_context
174            .style
175            .projection
176            .as_ref()
177            .map_or(ProjectionType::Mercator, |spec| {
178                spec.projection_type.clone()
179            });
180        let motion = self
181            .map_context
182            .world
183            .resources
184            .get_or_init_mut::<motion::MotionPrefetch>();
185        let ahead = view.and_then(|view| {
186            motion.update(
187                frame.timestamp,
188                view,
189                &self.map_context.view_state,
190                &projection,
191            )
192        });
193        self.map_context.world.resources.insert(PrefetchView {
194            view_state: ahead,
195            placement: None,
196        });
197    }
198
199    /// Keeps the view the frame's first eye would have from `placement`, so the request
200    /// systems can cover it, or clears it.
201    fn set_prefetch(&mut self, placement: Option<&ScenePlacement>, eye: Option<&XrEye>) {
202        // The anchor's altitude follows the terrain as DEM tiles arrive; a destination
203        // whose only change is that is the same destination.
204        let same_destination = |kept: &ScenePlacement, wanted: &ScenePlacement| {
205            kept.world_from_scene == wanted.world_from_scene
206                && kept.anchor.position == wanted.anchor.position
207        };
208        let resources = &mut self.map_context.world.resources;
209        let kept = resources.get::<PrefetchView>().is_some_and(|prefetch| {
210            match (prefetch.placement.as_ref(), placement) {
211                (Some(kept), Some(wanted)) => same_destination(kept, wanted),
212                (None, None) => true,
213                _ => false,
214            }
215        });
216        if kept {
217            return;
218        }
219        let view = placement
220            .zip(eye)
221            .and_then(|(placement, eye)| placement.view_from(eye.world_from_eye, eye.frustum));
222        let view_state = view.and_then(|view| {
223            let projection = self
224                .map_context
225                .style
226                .projection
227                .as_ref()
228                .map_or(ProjectionType::Mercator, |specification| {
229                    specification.projection_type.clone()
230                });
231            let mut ahead = self.map_context.view_state.clone();
232            ahead.set_external_view(view, &projection).ok()?;
233            Some(ahead)
234        });
235        self.map_context.world.resources.insert(PrefetchView {
236            view_state,
237            placement: placement.copied(),
238        });
239    }

241    /// Terrain elevation in metres at a location, from the DEM tiles loaded so far; `None`
242    /// without terrain or before a tile covering the location arrived. A host stands its
243    /// scene on it, so a height above the anchor is a height above the ground.
244    pub fn terrain_elevation_at(&self, position: LatLon) -> Option<f64> {
245        let world = &self.map_context.world;
246        let index = world.resources.get::<TerrainCoverageIndex>()?;
247        let mercator = WorldCoords::from_lat_lon(position, Zoom::new(0.0));
248        index.loaded_ground_cached(
249            &world.tiles,
250            mercator.x / TILE_SIZE,
251            mercator.y / TILE_SIZE,
252            crate::terrain::DemDetail::Finest,
253        )
254    }
255
256    /// Height of the ground the last frame drew at a location, for converting between the
257    /// screen and the ground: `Ok(None)` without terrain or outside the drawn tiles, and
258    /// [`TerrainUnknown`](crate::terrain::sightline::TerrainUnknown) over a drawn tile whose
259    /// DEM has not loaded. A DEM loaded elsewhere in the cache does not stand in for it.
260    pub fn drawn_ground_cached(
261        &self,
262        position: LatLon,
263    ) -> Result<Option<f64>, crate::terrain::sightline::TerrainUnknown> {
264        self.drawn_terrain(self.map_context.view_state.body())
265            .map_or(Ok(None), |terrain| terrain.ground_at(position))
266    }
