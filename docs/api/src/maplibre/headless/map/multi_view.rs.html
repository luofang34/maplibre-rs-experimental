1//! Heterogeneous multi-viewport rendering (`MultiViewFrame`) over a single [`HeadlessMap`].
2//!
3//! Registering lightweight [`ViewSlotId`]s on one [`HeadlessMap`] allows independent cameras
4//! and resolutions (such as a `1920×1080` forward OTW camera at `60–120 Hz` alongside a
5//! `640×480` nadir/oblique sensor camera at `30 Hz`, or a `ConformalSvs3D` PFD alongside a
6//! `Map2D` MFD) to share a single `World`, `VectorBufferPool`, `DrapeScratch`, `DrapeCache`,
7//! and compiled pipeline set without tile or drape thrashing.
8

41/// Attachment configuration for a registered [`ViewSlotId`].
42#[derive(Clone, Copy, Debug)]
43pub struct ViewSlotConfig {
44    /// Physical pixel dimensions of this viewport's color and depth attachments.
45    pub size: PhysicalSize,
46    /// Whether to resolve a persistent 1× `Depth32Float` readout texture via `DEPTH_COPY`.
47    pub depth_readout: bool,
48}

50/// How a slot selects its tile covering during [`HeadlessMap::run_multi_view_frame`].
51#[derive(Clone, Copy, Debug, PartialEq, Eq)]
52pub enum CoveringPolicy {
53    /// Reuses the tile covering and terrain surface selection prepared for an earlier
54    /// [`CoveringPolicy::Independent`] slot in the same frame (zero-overhead stereo pair).
55    StereoReuse(ViewSlotId),
56    /// Computes this slot's own visible covering and unions its requests/residency with all
57    /// other registered slots.
58    Independent,
59}

61/// Camera pose rendered by a [`ViewSlotEntry`].
62#[derive(Clone, Copy, Debug)]
63pub enum SlotView {
64    /// Host-driven 6-DoF external camera / sensor frustum.
65    External(ExternalView),
66    /// Map camera specified by geographic center, zoom, bearing, and pitch.
67    Camera {
68        /// Geographic center of the viewport.
69        center: LatLon,
70        /// Map zoom level.
71        zoom: Zoom,
72        /// Camera bearing in degrees clockwise from north.
73        bearing_deg: f64,
74        /// Camera pitch in degrees from nadir.
75        pitch_deg: f64,
76    },
77}

79/// One active viewport rendered in a [`MultiViewFrame`].
80///
81/// Registered slots omitted from [`MultiViewFrame::slots`] are skipped for that frame without
82/// evicting their resident tiles or baked terrain drapes.
83#[derive(Clone, Copy, Debug)]
84pub struct ViewSlotEntry {
85    /// Target registered slot.
86    pub slot: ViewSlotId,
87    /// Camera or external eye pose for this slot on this frame.
88    pub view: SlotView,
89    /// Covering policy (`Independent` or `StereoReuse(primary)`).
90    pub policy: CoveringPolicy,
91}

93/// Frame specification rendering one or more active [`ViewSlotId`]s over one [`HeadlessMap`].
94#[derive(Clone, Debug)]
95pub struct MultiViewFrame {
96    /// Host frame timestamp used by animated paint properties and motion prefetch.
97    pub timestamp: Duration,
98    /// Whether globe sky/background draws an opaque environment behind the body.
99    pub opaque_environment: bool,
100    /// Frustum tangent overscan factor (`>= 1.0`) for external-eye tile prefetch.
101    pub request_overscan: f64,
102    /// Active slots to draw in order; omitted registered slots are skipped this frame.
103    pub slots: Vec<ViewSlotEntry>,
104}
105
106impl Default for MultiViewFrame {
107    fn default() -> Self {
108        Self {
109            timestamp: Duration::ZERO,
110            opaque_environment: true,
111            request_overscan: 1.0,
112            slots: Vec::new(),
113        }
114    }
115}

117/// Failure while validating or rendering a [`MultiViewFrame`].
118#[derive(Debug, Error)]
119pub enum MultiViewError {
120    /// Referenced a [`ViewSlotId`] that is not registered on this map.
121    #[error("view slot {slot:?} is not registered")]
122    UnknownSlot {
123        /// Unregistered slot identifier.
124        slot: ViewSlotId,
125    },
126    /// `CoveringPolicy::StereoReuse(primary)` referenced a primary slot that did not precede it
127    /// as an active `Independent` slot in the same frame.
128    #[error("stereo slot {slot:?} references primary {primary:?} which was not drawn earlier as Independent")]
129    InvalidStereoPrimary {
130        /// Dependent stereo slot.
131        slot: ViewSlotId,
132        /// Referenced primary slot.
133        primary: ViewSlotId,
134    },
135    /// External view validation failed for an active slot.
136    #[error("invalid external view on slot")]
137    InvalidExternalView {
138        /// Underlying external-view error.
139        #[source]
140        source: ExternalViewError,
141    },
142    /// Map camera validation failed for an active slot.
143    #[error("invalid camera options on slot")]
144    InvalidCamera {
145        /// Underlying navigation error.
146        #[source]
147        source: NavigationError,
148    },
149    /// A render schedule stage failed while executing a frame or slot.
150    #[error("stage failed for slot {slot:?}")]
151    SlotStage {
152        /// Slot executing when the stage failed, or `None` for shared `Extract`/`Cleanup`.
153        slot: Option<ViewSlotId>,
154        /// Underlying schedule stage failure.
155        #[source]
156        source: StageError,
157    },
158}

197fn swap_slot_resources(slot: &mut SlotState, renderer: &mut Renderer, world: &mut World) {
198    std::mem::swap(&mut slot.surface, &mut renderer.resources.surface);
199    std::mem::swap(
200        &mut slot.depth_readout,
201        &mut renderer.settings.depth_readout,
202    );
203    std::mem::swap(
204        &mut slot.depth_texture,
205        &mut renderer.resources.depth_texture,
206    );
207    std::mem::swap(
208        &mut slot.depth_readout_texture,
209        &mut renderer.resources.depth_readout_texture,
210    );
211    std::mem::swap(
212        &mut slot.multisampling_texture,
213        &mut renderer.resources.multisampling_texture,
214    );
215    if let Some(symbol_depth) = world.resources.get_mut::<Eventually<SymbolDepth>>() {
216        std::mem::swap(&mut slot.symbol_depth, symbol_depth);
217    }
218    #[cfg(feature = "experimental")]
219    if let Some(sun_depth) = world.resources.get_mut::<Eventually<SunDepth>>() {
220        std::mem::swap(&mut slot.sun_depth, sun_depth);
221    }
222    #[cfg(feature = "experimental")]
223    if let Some(overlay) = world.resources.get_mut::<HostOverlay>() {
224        std::mem::swap(&mut slot.overlay, overlay);
225    }
226    #[cfg(feature = "experimental")]
227    if let Some(overlay_gpu) = world.resources.get_mut::<Eventually<OverlayGpu>>() {
228        std::mem::swap(&mut slot.overlay_gpu, overlay_gpu);
229    }
230    #[cfg(feature = "experimental")]
231    if let Some(probe_gpu) = world.resources.get_mut::<Eventually<ProbeGpu>>() {
232        std::mem::swap(&mut slot.probe_gpu, probe_gpu);
233    }
234    std::mem::swap(
235        &mut slot.edge_cache,
236        world
237            .resources
238            .get_or_init_mut::<crate::terrain::queue_system::edges::EdgeCache>(),
239    );
240}

242impl HeadlessMap {
243    /// Registers a new lightweight viewport slot owning its own `(width, height)` attachments,
244    /// LOD history, and optional overlay / depth-probe state.
245    pub fn register_view_slot(&mut self, config: ViewSlotConfig) -> ViewSlotId {
246        let surface = Surface::from_headless_size(
247            &self.map_context.renderer.device,
248            &self.map_context.renderer.adapter,
249            config.size,
250            &self.map_context.renderer.settings,
251        );
252        let registry = self
253            .map_context
254            .world
255            .resources
256            .get_or_init_mut::<MultiViewRegistry>();
257        let id = ViewSlotId::from_raw(registry.next_id);
258        registry.next_id = registry.next_id.wrapping_add(1);
259        registry.slots.insert(
260            id,
261            SlotState {
262                surface,
263                depth_readout: config.depth_readout,
264                depth_texture: Eventually::Uninitialized,
265                depth_readout_texture: Eventually::Uninitialized,
266                multisampling_texture: Eventually::Uninitialized,
267                symbol_depth: Eventually::Uninitialized,
268                #[cfg(feature = "experimental")]
269                sun_depth: Eventually::Uninitialized,
270                #[cfg(feature = "experimental")]
271                overlay: HostOverlay::default(),
272                #[cfg(feature = "experimental")]
273                overlay_gpu: Eventually::Uninitialized,
274                #[cfg(feature = "experimental")]
275                probe_gpu: Eventually::Uninitialized,
276                edge_cache: crate::terrain::queue_system::edges::EdgeCache::default(),
277                lod_history: FrameLodHistory::default(),
278                stereo_snapshot: None,
279                prepared_terrain: None,
280                frame_loose_in_use: FxHashSet::default(),
281                last_retained_tiles: FxHashSet::default(),
282                last_retained_drapes: FxHashSet::default(),
283                last_wanted_vector: FxHashSet::default(),
284                last_wanted_raster: FxHashSet::default(),
285                last_wanted_dem: FxHashSet::default(),
286                last_drawn_count: 0,
287                drape_priority: 0,
288            },
289        );
290        id
291    }

293    /// Resizes a registered view slot; GPU attachments reconfigure lazily on its next active frame.
294    pub fn resize_view_slot(
295        &mut self,
296        slot: ViewSlotId,
297        size: PhysicalSize,
298    ) -> Result<(), MultiViewError> {
299        let state = self.slot_mut(slot)?;
300        state.surface.resize(size);
301        Ok(())
302    }

304    /// Ranks `slot` among the views for the terrain drape textures memory allows the map.
305    ///
306    /// Every view holds its own drapes, and the map's allowance of drape textures is granted
307    /// to the views at the start of each frame, by need, the highest priority first and slots
308    /// of equal priority in the order they were registered. The first view drapes its ground
309    /// at the detail it would have alone; each view after it has the textures that are left,
310    /// never fewer than a small floor, and covers its ground with fewer, coarser drapes when
311    /// they are short. Every slot starts at priority `0`, so the slot registered first is
312    /// served first until the host says otherwise.
313    pub fn set_view_slot_drape_priority(
314        &mut self,
315        slot: ViewSlotId,
316        priority: i32,
317    ) -> Result<(), MultiViewError> {
318        self.slot_mut(slot)?.drape_priority = priority;
319        Ok(())
320    }
321
322    /// How many terrain drapes `slot` showed in the last frame it drew before they were drawn
323    /// from all of their content; see [`Self::pending_drapes`]. Zero for a slot that has not
324    /// drawn, so a host waiting for a settled view draws it first.
325    pub fn view_slot_pending_drapes(&self, slot: ViewSlotId) -> Result<usize, MultiViewError> {
326        let registered = self
327            .map_context
328            .world
329            .resources
330            .get::<MultiViewRegistry>()
331            .is_some_and(|registry| registry.slots.contains_key(&slot));
332        if !registered {
333            return Err(MultiViewError::UnknownSlot { slot });
334        }
335        Ok(self.pending_drapes_of(Some(slot)))
336    }

349    /// Shows or hides the navigation HUD ([`crate::hud::NavigationHudPlugin`]) in `slot`.
350    ///
351    /// The HUD belongs to the map's own view, whose camera its controls steer, so a slot
352    /// draws none until asked to. A slot that shows it lays it out for its own size; pointer
353    /// input is still hit-tested against the map's own viewport.
354    pub fn set_view_slot_navigation_hud(
355        &mut self,
356        slot: ViewSlotId,
357        shown: bool,
358    ) -> Result<(), MultiViewError> {
359        self.slot_mut(slot)?;
360        let resources = &mut self.map_context.world.resources;
361        resources
362            .get_or_init_mut::<crate::hud::HudViewSlots>()
363            .set(slot, shown);
364        Ok(())
365    }

367    /// Unregisters `slot` and releases its per-slot GPU attachments and terrain drapes.
368    pub fn remove_view_slot(&mut self, slot: ViewSlotId) -> bool {
369        let resources = &mut self.map_context.world.resources;
370        let removed = resources
371            .get_mut::<MultiViewRegistry>()
372            .is_some_and(|registry| registry.slots.remove(&slot).is_some());
373        if removed {
374            if let Some(pending) =
375                resources.get_mut::<crate::terrain::queue_system::PendingDrapes>()
376            {
377                pending.forget(Some(slot));
378            }
379            if let Some(hud) = resources.get_mut::<crate::hud::HudViewSlots>() {
380                hud.set(slot, false);
381            }
382            if let Some(Eventually::Initialized(terrain)) =
383                resources.get_mut::<Eventually<crate::terrain::resources::TerrainResources>>()
384            {
385                terrain.release_drapes_of(Some(slot));
386            }
387        }
388        removed
389    }

421    /// Borrows `slot`'s resolved single-sample color [`wgpu::Texture`].

463    fn slot_ref(&self, slot: ViewSlotId) -> Option<&SlotState> {
464        self.map_context
465            .world
466            .resources
467            .get::<MultiViewRegistry>()?
468            .slots

472    fn slot_mut(&mut self, slot: ViewSlotId) -> Result<&mut SlotState, MultiViewError> {
473        self.map_context
474            .world
475            .resources
476            .get_mut::<MultiViewRegistry>()
477            .and_then(|registry| registry.slots.get_mut(&slot))
478            .ok_or(MultiViewError::UnknownSlot { slot })
479    }
