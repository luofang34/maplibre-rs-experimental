1//! Camera settings a host makes on the offscreen map, beyond what its style starts it with.
2
3use super::HeadlessMap;
4
5impl HeadlessMap {
6    /// Puts the camera's center at `meters` above sea level and holds it there, so terrain under
7    /// the center does not move it, as GL JS does once `centerClampedToGround` is off.
8    pub fn pin_center_elevation(&mut self, meters: f64) {
9        let view_state = &mut self.map_context.view_state;
10        view_state.set_center_clamped_to_ground(false);
11        view_state.set_center_altitude(meters);
12    }
13
14    /// Switches how the camera navigates the globe, for the style's projection.
15    pub fn set_navigation_mode(
16        &mut self,
17        mode: crate::render::view_state::NavigationMode,
18    ) -> Result<(), crate::render::view_state::NavigationError> {
19        self.map_context.set_navigation_mode(mode)
20    }
21
22    /// Moves the map at once, as GL JS `jumpTo`, standing a center held by terrain on the
23    /// loaded ground or at sea level, as [`crate::context::MapContext::jump_to`] does.
24    pub fn jump_to(
25        &mut self,
26        options: crate::render::view_state::CameraOptions,
27    ) -> Result<(), crate::render::view_state::NavigationError> {
28        self.map_context.jump_to(options)
29    }
30
31    /// Moves the camera along `curve`, as [`crate::context::MapContext::fly_to`] does; read how
32    /// the move stands through [`Self::context`].
33    pub fn fly_to(
34        &mut self,
35        options: crate::render::view_state::CameraOptions,
36        curve: crate::render::view_state::FlightCurve,
37    ) -> crate::context::FlightId {
38        self.map_context.fly_to(options, curve)
39    }
40
41    /// Stops the camera move under way where the camera is.
42    pub fn stop_flight(&mut self) {
43        self.map_context.stop_flight();
44    }
45
46    /// Restores a stored free-globe pose, for the style's projection.
47    pub fn restore_globe_pose(
48        &mut self,
49        pose: crate::render::view_state::GlobePose,
50    ) -> Result<(), crate::render::view_state::NavigationError> {
51        self.map_context.restore_globe_pose(pose)
52    }
53
54    /// Sets the full vertical field of view, as the GL JS `verticalFieldOfView` map option.
55    pub fn set_vertical_field_of_view(&mut self, field_of_view: cgmath::Deg<f64>) {
56        self.map_context
57            .view_state
58            .set_field_of_view(cgmath::Rad::from(field_of_view));
59    }
60
61    /// Asks for an opaque environment: on a full globe without atmosphere the style's sky
62    /// then fills everything the globe does not cover, as it does for a
63    /// [`crate::headless::MultiViewFrame`] that asks the same.
64    pub fn set_opaque_environment(&mut self, opaque: bool) {
65        self.map_context.view_state.set_opaque_environment(opaque);
66        crate::render::frame_signals::mark_dirty(&mut self.map_context.world);
67    }
68
69    /// Drags a free-globe camera so the ground under `from` moves under `to`, as
70    /// [`crate::render::view_state::ViewState::drag_free_globe`] does, and marks the map for
71    /// redraw when it moved. Returns whether it moved.
72    pub fn drag_globe(&mut self, from: cgmath::Point2<f64>, to: cgmath::Point2<f64>) -> bool {
73        let moved = self.map_context.view_state.drag_free_globe(from, to);
74        if moved {
75            crate::render::frame_signals::mark_dirty(&mut self.map_context.world);
76        }
77        moved
78    }
79
80    /// Zooms a free-globe camera by `zoom_delta` about the ground under `pixel`, as
81    /// [`crate::render::view_state::ViewState::zoom_free_globe`] does, and marks the map for
82    /// redraw when it moved. Returns whether it moved.
83    pub fn zoom_globe(&mut self, pixel: cgmath::Point2<f64>, zoom_delta: f64) -> bool {
84        let moved = self
85            .map_context
86            .view_state
87            .zoom_free_globe(pixel, zoom_delta);
88        if moved {
89            crate::render::frame_signals::mark_dirty(&mut self.map_context.world);
90        }
91        moved
92    }
93
94    /// The sea-level location under a viewport pixel of the globe camera, or `None` where the
95    /// pixel shows space beyond the limb or the view has no globe camera.
96    pub fn screen_to_globe_lat_lon(
97        &self,
98        pixel: cgmath::Point2<f64>,
99    ) -> Option<crate::coords::LatLon> {
100        let camera =
101            crate::render::projection::globe_camera_for_view(&self.map_context.view_state).ok()?;
102        camera
103            .is_point_on_map_surface(pixel)
104            .then(|| camera.screen_point_to_location(pixel))
105            .flatten()
106    }
107
108    /// Where the globe camera draws `location` raised `elevation_meters`, in viewport pixels,
109    /// and whether the planet hides it; `None` when the view has no globe camera.
110    pub fn globe_lat_lon_to_screen(
111        &self,
112        location: crate::coords::LatLon,
113        elevation_meters: f64,
114    ) -> Option<(cgmath::Point2<f64>, bool)> {
115        let camera =
116            crate::render::projection::globe_camera_for_view(&self.map_context.view_state).ok()?;
117        Some((
118            camera.location_to_screen(location, elevation_meters),
119            camera.is_location_occluded(location),
120        ))
121    }
