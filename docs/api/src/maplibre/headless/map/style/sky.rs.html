1//! The sun, its light, the atmosphere and the stars a host sets on the offscreen map in place of the
2//! style's `light` and `sky`, as [`crate::map::Map`] sets its own.
3
4use crate::{
5    celestial::{EcefDirection, TwilightBand},
6    headless::map::HeadlessMap,
7};
8
9impl HeadlessMap {
10    /// Sets the direction towards the sun in ECEF axes, as
11    /// [`crate::context::MapContext::set_sun_direction`] does; `None` returns to the style's light.
12    pub fn set_sun_direction(&mut self, sun: Option<EcefDirection>) {
13        self.map_context.set_sun_direction(sun);
14    }
15
16    /// Sets the solar elevations over which the terrain turns from night to day, as
17    /// [`crate::context::MapContext::set_twilight_band`] does.
18    pub fn set_twilight_band(&mut self, twilight: TwilightBand) {
19        self.map_context.set_twilight_band(twilight);
20    }
21
22    /// Replaces the stars drawn behind the globe, as
23    /// [`crate::context::MapContext::set_stars`] does.
24    pub fn set_stars(&mut self, stars: Vec<crate::celestial::Star>) {
25        self.map_context.set_stars(stars);
26    }
27
28    /// Sets or clears the sun's angular radius, as
29    /// [`crate::context::MapContext::set_sun_angular_radius`] does.
30    pub fn set_sun_angular_radius(&mut self, radius: Option<crate::celestial::AngularRadius>) {
31        self.map_context.set_sun_angular_radius(radius);
32    }
33
34    /// Sets or clears the moon's disc, as [`crate::context::MapContext::set_moon_disc`] does.
35    pub fn set_moon_disc(&mut self, moon: Option<crate::celestial::Disc>) {
36        self.map_context.set_moon_disc(moon);
37    }
38
39    /// Draws the globe's atmosphere without a style `sky`, as
40    /// [`crate::context::MapContext::set_atmosphere`] does.
41    pub fn set_atmosphere(&mut self, atmosphere: bool) {
42        self.map_context.set_atmosphere(atmosphere);
43    }
44}
