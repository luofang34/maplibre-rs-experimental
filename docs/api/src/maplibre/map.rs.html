35/// A window, renderer or scheduled update failed, or the map is in the wrong lifecycle state.
36#[derive(Error, Debug)]
37pub enum MapError {
38    /// Renderer initialization was requested while the map is already ready.
39    #[error("renderer was already set for this map")]
40    RendererAlreadySet,
41    /// The requested operation requires an initialized renderer.
42    #[error("renderer is not fully initialized")]
43    RendererNotReady,
44    /// Render graph construction violated a node or connection contract.
45    #[error("initializing render graph failed")]
46    RenderGraphInit(#[source] RenderGraphError),
47    /// Renderer initialization or presentation surface recovery failed.
48    #[error("initializing device failed")]
49    DeviceInit(#[source] RenderError),
50    /// The host could not create the map window or its event loop.
51    #[error("creating window failed")]
52    Window(#[from] WindowCreateError),
53    /// A scheduled update failed; preceding state changes are not rolled back.
54    #[error("executing map stage failed")]
55    StageError(#[from] StageError),
56    /// A source's data could not be replaced or edited; the source is unchanged.
57    #[error("updating source data failed")]
58    SourceUpdate(#[from] SourceUpdateError),
59    /// A source's features could not be read.
60    #[error("querying source features failed")]
61    SourceQuery(#[from] SourceQueryError),
62    /// A rendered-feature query was rejected.
63    #[error("querying rendered features failed")]
64    Query(#[from] QueryError),
65    /// A style change was refused; the style is unchanged.
66    #[error("changing the style failed")]
67    StyleChange(#[from] StyleMutationError),
68}

102    /// Creates the host window and retains the style and plugins for GPU initialization.
103    /// Window errors are returned; style validation findings are logged.
104    pub fn new(
105        mut style: Style,
106        kernel: Kernel<E>,
107        renderer_builder: RendererBuilder,
108        plugins: Vec<Box<dyn Plugin<E>>>,
109    ) -> Result<Self, MapError> {
110        style.resolve_global_state();
111        style.log_validation_errors();
112        let schedule = Schedule::default();
113
114        let window = kernel.map_window_config().create()?;
115
116        let kernel = Rc::new(kernel);
117
118        let map = Self {
119            kernel,
120            schedule,
121            request_attempts: RequestAttempts::default(),
122            map_context: CurrentMapContext::Pending(Box::new(PendingMapContext {
123                style,
124                renderer_builder,
125                view_state: None,
126            })),
127            window,
128            plugins,
129            max_pitch: DEFAULT_MAX_PITCH,
130        };
131        Ok(map)
132    }

141    /// Creates GPU state, resolves source metadata and builds plugins in their supplied order.
142    /// Uses current logical window dimensions and restores any retained orbit camera.
143    /// Returns [`MapError::RendererAlreadySet`] if ready, or a renderer failure while keeping
144    /// the map pending. TileJSON resolution failures are logged and do not abort initialization.
145    pub async fn initialize_renderer(&mut self) -> Result<(), MapError> {
146        match &mut self.map_context {
147            CurrentMapContext::Ready(_) => Err(MapError::RendererAlreadySet),
148            CurrentMapContext::Pending(pending) => {
149                let PendingMapContext {
150                    style,
151                    renderer_builder,
152                    view_state: retained_view,
153                } = pending.as_mut();
154                let mut renderer = renderer_builder
155                    .clone()
156                    .build()
157                    .initialize_renderer::<E::MapWindowConfig>(&self.window)
158                    .await
159                    .map_err(MapError::DeviceInit)?;
160
161                resolve_tile_json_sources(style, self.kernel.source_client()).await;
162
163                let window_size = self.window.size();
164                renderer.resize_surface(window_size);
165                let mut view_state = match retained_view {
166                    Some(view) => {
167                        let mut view = view.clone();
168                        view.clear_external_view();
169                        view.set_max_pitch(self.max_pitch);
170                        view
171                    }
172                    None => initial_view_state(style, window_size, self.max_pitch),
173                };
174                view_state.resize(window_size.to_logical(self.window.scale_factor()));
175
176                let mut world = World::default();
177                world.resources.insert(self.request_attempts.clone());
178                crate::sdf::provided::set_pixel_ratio(&mut world, self.window.scale_factor());
179
180                for plugin in &self.plugins {
181                    plugin.build(
182                        &mut self.schedule,
183                        self.kernel.clone(),
184                        &mut world,
185                        &mut renderer.render_graph,
186                    );
187                }
188
189                self.map_context = CurrentMapContext::Ready(Box::new(MapContext {
190                    world,
191                    view_state,
192                    style: std::mem::take(style),
193                    renderer,
194                }));
195                Ok(())
196            }
197        }
198    }

217    /// Drops GPU/frame state and clears the schedule while retaining the window, style,
218    /// orbit camera, renderer settings and plugins. External eyes must be supplied for new frames.
219    /// Call [`Self::initialize_renderer`] before drawing again.
220    pub fn reset(&mut self) {
221        self.schedule.clear();
222        match &self.map_context {
223            CurrentMapContext::Ready(c) => {
224                self.map_context = CurrentMapContext::Pending(Box::new(PendingMapContext {
225                    style: c.style.clone(),
226                    view_state: Some(c.view_state.clone()),
227                    renderer_builder: RendererBuilder::new()
228                        .with_renderer_settings(c.renderer.settings)
229                        .with_wgpu_settings(c.renderer.wgpu_settings.clone()),
230                }))
231            }
232            CurrentMapContext::Pending(_) => {}
233        }
234    }

236    /// Runs one scheduled update when initialized, stopping on the first failing stage.
237    /// Unavailable presentation frames are skipped; a lost surface is recreated for a later frame.
238    /// Other failures retain the stage or renderer cause in [`MapError`].
239    #[tracing::instrument(name = "update_and_redraw", skip_all)]
240    pub fn run_schedule(&mut self) -> Result<(), MapError> {
241        match &mut self.map_context {
242            CurrentMapContext::Ready(map_context) => {
243                if let Err(error) = self.schedule.run(map_context) {
244                    use crate::{
245                        render::resource::{Head, SurfaceAcquireError},
246                        tcs::system::SystemError,
247                    };
248                    match error {
249                        StageError::System(SystemError::Render(RenderError::Surface(
250                            SurfaceAcquireError::Timeout
251                            | SurfaceAcquireError::Occluded
252                            | SurfaceAcquireError::Outdated,
253                        ))) => return Ok(()),
254                        StageError::System(SystemError::Render(RenderError::Surface(
255                            SurfaceAcquireError::Lost,
256                        ))) => {
257                            let renderer = &mut map_context.renderer;
258                            if let Head::Headed(surface) = renderer.resources.surface.head_mut() {
259                                surface
260                                    .recreate_surface(&self.window, &renderer.instance)
261                                    .map_err(MapError::DeviceInit)?;
262                                surface.configure(&renderer.device);
263                            }
264                            return Ok(());
265                        }
266                        error => return Err(error.into()),
267                    }
268                }
269                Ok(())
270            }
271            CurrentMapContext::Pending(_) => Err(MapError::RendererNotReady),
272        }
273    }

449fn initial_view_state(
450    style: &Style,
451    window_size: PhysicalSize,
452    max_pitch: cgmath::Deg<f64>,
453) -> ViewState {
454    let center = style.center.unwrap_or_default();
455    let zoom = style.zoom.map(Zoom::new).unwrap_or_default();
456    let mut view = ViewState::new(
457        window_size,
458        WorldCoords::from_lat_lon(LatLon::new(center[1], center[0]), zoom),
459        zoom,
460        cgmath::Deg(style.pitch.unwrap_or_default()),
461        cgmath::Rad(0.6435011087932844),
462    );
463    view.set_max_pitch(max_pitch);
464    view.camera_mut()
465        .set_pitch(cgmath::Deg(style.pitch.unwrap_or_default()));
466    view.camera_mut()
467        .set_bearing(cgmath::Deg(style.bearing.unwrap_or_default()));
468    view.camera_mut()
469        .set_roll(cgmath::Deg(style.roll.unwrap_or_default()));
470    view.take_style_center(style);
471    view
472}
