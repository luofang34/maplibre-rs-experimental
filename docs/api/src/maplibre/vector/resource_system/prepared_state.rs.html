1//! Memoized per-revision style requirements for [`resource_system`](super::resource_system).
2
3use crate::style::{
4    layer::{LayerPaint, StyleLayer},
5    Style,
6};
7
8#[derive(Default)]
9pub(super) struct VectorPreparedState {
10    revision: Option<u64>,
11    terrain_active: bool,
12    pub(super) needs_terrain_lines: bool,
13    pub(super) needs_elevated_lines: bool,
14    pub(super) needs_circle: bool,
15    pub(super) needs_extrusion: bool,
16    pub(super) has_pattern_paint: bool,
17    pub(super) has_dash_or_pattern_line: bool,
18    last_dash_revision: Option<u64>,
19    last_pattern_revision: Option<u64>,
20}
21
22impl VectorPreparedState {
23    /// Updates cached pipeline/layer requirements in a single pass when `style.revision()` changes.
24    pub(super) fn sync_style(&mut self, style: &Style, terrain_active: bool) -> bool {
25        let rev = style.revision();
26        let terrain_needed = style.terrain.is_some() || terrain_active;
27        if self.revision == Some(rev) && self.terrain_active == terrain_active {
28            self.needs_terrain_lines = terrain_needed;
29            return false;
30        }
31        let mut needs_elevated_lines = false;
32        let mut needs_circle = false;
33        let mut needs_extrusion = false;
34        let mut has_pattern_paint = false;
35        let mut has_dash_or_pattern_line = false;
36        for layer in &style.layers {
37            if !needs_elevated_lines
38                && matches!(
39                    crate::vector::structures::kind(layer),
40                    Some(crate::vector::structures::StructureKind::ElevatedLine)
41                )
42            {
43                needs_elevated_lines = true;
44            }
45            if layer.type_ == "circle" {
46                needs_circle = true;
47            } else if layer.type_ == "fill-extrusion" {
48                needs_extrusion = true;
49            }
50            if let Some(paint) = layer.paint.as_ref() {
51                if !has_pattern_paint && layer_uses_pattern(paint) {
52                    has_pattern_paint = true;
53                }
54                if !has_dash_or_pattern_line && line_uses_dash_or_pattern(layer, paint) {
55                    has_dash_or_pattern_line = true;
56                }
57            }
58        }
59        self.revision = Some(rev);
60        self.terrain_active = terrain_active;
61        self.needs_terrain_lines = terrain_needed;
62        self.needs_elevated_lines = needs_elevated_lines;
63        self.needs_circle = needs_circle;
64        self.needs_extrusion = needs_extrusion;
65        self.has_pattern_paint = has_pattern_paint;
66        self.has_dash_or_pattern_line = has_dash_or_pattern_line;
67        true
68    }
69
70    pub(super) fn should_update_dashes(&mut self, style: &Style) -> bool {
71        let rev = style.revision();
72        let changed_rev = self.last_dash_revision != Some(rev);
73        self.last_dash_revision = Some(rev);
74        changed_rev || self.has_dash_or_pattern_line
75    }
76
77    pub(super) fn should_update_patterns(&mut self, style: &Style) -> bool {
78        let rev = style.revision();
79        let changed_rev = self.last_pattern_revision != Some(rev);
80        self.last_pattern_revision = Some(rev);
81        changed_rev || self.has_pattern_paint
82    }
83}
84
85fn layer_uses_pattern(paint: &LayerPaint) -> bool {
86    if crate::vector::pattern::per_feature_pattern(paint).is_some() {
87        return true;
88    }
89    match paint {
90        LayerPaint::Fill(p) => p.fill_pattern.is_some(),
91        LayerPaint::Line(p) => p.line_pattern.is_some(),
92        LayerPaint::FillExtrusion(p) => p.fill_extrusion_pattern.is_some(),
93        _ => false,
94    }
95}
96
97fn line_uses_dash_or_pattern(layer: &StyleLayer, paint: &LayerPaint) -> bool {
98    let LayerPaint::Line(line) = paint else {
99        return false;
100    };
101    line.line_dasharray.is_some()
102        || line.line_pattern.is_some()
103        || (layer.type_ == "line" && crate::vector::pattern::per_feature_pattern(paint).is_some())
104}
