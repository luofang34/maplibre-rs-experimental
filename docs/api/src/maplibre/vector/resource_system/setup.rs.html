13pub(super) struct PipelineSetup<'a> {
14    pub(super) device: &'a wgpu::Device,
15    pub(super) settings: crate::render::settings::RendererSettings,
16    pub(super) format: wgpu::TextureFormat,
17    pub(super) multisampling: bool,
18    pub(super) projection: &'a wgpu::BindGroupLayout,
19    /// The second bind group of every pipeline that draws an image.
20    pub(super) pattern: wgpu::BindGroupLayout,
21}
22
23impl PipelineSetup<'_> {
24    fn layer_color_binding() -> crate::render::resource::UniformFeatureBinding {
25        crate::render::resource::UniformFeatureBinding::LayerMetadata {
26            layer_slot: 2,
27            feature_slot: 3,
28            attribute: wgpu::VertexAttribute {
29                offset: std::mem::offset_of!(shaders::ShaderLayerMetadata, uniform_color) as u64,
30                format: wgpu::VertexFormat::Float32x4,
31                shader_location: 8,
32            },
33        }
34    }
35
36    fn create(
37        &self,
38        name: &'static str,
39        shader: &impl Shader,
40        layouts: &[&wgpu::BindGroupLayout],
41        depth: bool,
42    ) -> crate::render::resource::FeaturePipelinePair {
43        self.create_with_clipping(name, shader, layouts, depth, true)
44    }
45
46    /// `clipped` keeps the pipeline inside its tile's stencil mask.
47    fn create_with_clipping(
48        &self,
49        name: &'static str,
50        shader: &impl Shader,
51        layouts: &[&wgpu::BindGroupLayout],
52        depth: bool,
53        clipped: bool,
54    ) -> crate::render::resource::FeaturePipelinePair {
55        self.describe(name, shader, depth, clipped)
56            .initialize_pair_with_binding(self.device, layouts, Self::layer_color_binding())
57    }
58
59    /// The pipeline of a line on the terrain surface: tested against the terrain depth as a
60    /// structure is, with the terrain tile it lies on bound after the dash atlas. The stencil
61    /// masks lie on the map plane, below a line lifted onto hills, so the line clips itself to
62    /// its tiles instead.
63    fn create_terrain_line(
64        &self,
65        dashes: &wgpu::BindGroupLayout,
66    ) -> crate::render::resource::FeaturePipelinePair {
67        let shader = shaders::TerrainLineShader {
68            format: self.format,
69        };
70        let mut descriptor = self.describe("terrain_line_pipeline", &shader, true, false);
71        descriptor.layout = Some(vec![
72            crate::terrain::resources::TerrainResources::bind_group_layout_entries(),
73        ]);
74        descriptor.initialize_pair_with_binding(
75            self.device,
76            &[self.projection, dashes],
77            Self::layer_color_binding(),
78        )
79    }
80
81    fn describe(
82        &self,
83        name: &'static str,
84        shader: &impl Shader,
85        depth: bool,
86        clipped: bool,
87    ) -> crate::render::resource::RenderPipelineDescriptor {
88        let pipeline = TilePipeline::new(
89            name.into(),
90            self.settings,
91            shader.describe_vertex(),
92            shader.describe_fragment(),
93            crate::render::resource::TilePipelineOptions {
94                depth_stencil_enabled: true,
95                update_stencil: false,
96                debug_stencil: !clipped,
97                wireframe: false,
98                multisampling: self.multisampling,
99                textured: false,
100            },
101        );
102        let pipeline = if depth {
103            pipeline.with_depth_write()
104        } else {
105            pipeline
106        };
107        let mut descriptor = pipeline.describe_render_pipeline();
108        if depth {
109            // Casing and deck are paint layers on the same road surface. Test against
110            // terrain, but let style order composite them without self-occluding edges.
111            if let Some(state) = &mut descriptor.depth_stencil {
112                state.depth_write_enabled = Some(false);
113            }
114        }
115        descriptor
116    }

118    pub(super) fn create_pattern(&self) -> crate::render::resource::FeaturePipelinePair {
119        self.create(
120            "fill_pattern_pipeline",
121            &shaders::FillPatternShader {
122                format: self.format,
123            },
124            &[self.projection, &self.pattern],
125            false,
126        )
127    }

129    fn extrusion_descriptor(
130        &self,
131        pass: shaders::ExtrusionPass,
132        vertex: crate::render::resource::VertexState,
133        fragment: crate::render::resource::FragmentState,
134    ) -> crate::render::resource::RenderPipelineDescriptor {
135        use shaders::ExtrusionPass::{
136            Clear, Color, Depth, PatternColor, TranslucentBack, TranslucentFront,
137        };
138
139        let mut descriptor = TilePipeline::new(
140            match pass {
141                Depth => "extrusion_depth_pipeline",
142                Color => "extrusion_color_pipeline",
143                PatternColor => "extrusion_pattern_pipeline",
144                Clear => "extrusion_clear_pipeline",
145                TranslucentBack => "extrusion_translucent_back_pipeline",
146                TranslucentFront => "extrusion_translucent_front_pipeline",
147            }
148            .into(),
149            self.settings,
150            vertex,
151            fragment,
152            crate::render::resource::TilePipelineOptions {
153                depth_stencil_enabled: true,
154                update_stencil: false,
155                debug_stencil: true,
156                wireframe: false,
157                multisampling: self.multisampling,
158                textured: false,
159            },
160        )
161        .with_depth_write()
162        .describe_render_pipeline();
163        descriptor.primitive.cull_mode = match pass {
164            TranslucentBack => Some(wgpu::Face::Front),
165            TranslucentFront => Some(wgpu::Face::Back),
166            _ => None,
167        };
168        if let Some(state) = &mut descriptor.depth_stencil {
169            state.depth_write_enabled = Some(pass == Depth);
170            state.depth_compare = Some(match pass {
171                Depth | Color | PatternColor | TranslucentBack | TranslucentFront => {
172                    wgpu::CompareFunction::GreaterEqual
173                }
174                Clear => wgpu::CompareFunction::Always,
175            });
176            // Tile references use the low seven bits; the top bit marks the pixels the colour
177            // pass has drawn, so a surface that two tiles both hold is blended once.
178            let marked = wgpu::StencilFaceState {
179                compare: wgpu::CompareFunction::Equal,
180                fail_op: wgpu::StencilOperation::Keep,
181                depth_fail_op: wgpu::StencilOperation::Keep,
182                pass_op: wgpu::StencilOperation::Invert,
183            };
184            let unmark = wgpu::StencilFaceState {
185                compare: wgpu::CompareFunction::Always,
186                fail_op: wgpu::StencilOperation::Keep,
187                depth_fail_op: wgpu::StencilOperation::Keep,
188                pass_op: wgpu::StencilOperation::Zero,
189            };
190            match pass {
191                Depth | TranslucentBack | TranslucentFront => {}
192                Color | PatternColor => state.stencil.front = marked,
193                Clear => state.stencil.front = unmark,
194            }
195            state.stencil.back = state.stencil.front;
196            if matches!(pass, Color | PatternColor | Clear) {
197                state.stencil.read_mask = STENCIL_MARK;
198                state.stencil.write_mask = STENCIL_MARK;
199            } else if matches!(pass, TranslucentBack | TranslucentFront) {
200                state.stencil.read_mask = 0;
201                state.stencil.write_mask = 0;
202            }
203        }
204        descriptor
205    }

207    fn create_extrusion(
208        &self,
209        pass: shaders::ExtrusionPass,
210    ) -> crate::render::resource::FeaturePipelinePair {
211        let shader = shaders::FillExtrusionShader {
212            format: self.format,
213            pass,
214        };
215        let descriptor =
216            self.extrusion_descriptor(pass, shader.describe_vertex(), shader.describe_fragment());
217        let layouts: &[&wgpu::BindGroupLayout] = if pass == shaders::ExtrusionPass::PatternColor {
218            &[self.projection, &self.pattern]
219        } else {
220            &[self.projection]
221        };
222        descriptor.initialize_pair_with_binding(self.device, layouts, Self::layer_color_binding())
223    }
224
225    fn create_compact_extrusion(&self, pass: shaders::ExtrusionPass) -> wgpu::RenderPipeline {
226        let shader = shaders::FillExtrusionCompactShader {
227            format: self.format,
228            pass,
229        };
230        let mut descriptor =
231            self.extrusion_descriptor(pass, shader.describe_vertex(), shader.describe_fragment());
232        if let Some(label) = descriptor.label.take() {
233            descriptor.label = Some(std::borrow::Cow::Owned(format!("{label}_compact")));
234        }
235        descriptor.initialize_with_prefix_layouts(self.device, &[self.projection])
236    }

238    pub(super) fn initialize_extrusion(
239        &self,
240        extrusion: &mut Eventually<ExtrusionPipeline>,
241        needs_pattern: bool,
242        needs_translucent: bool,
243    ) {
244        let ep = extrusion.initialize(|| ExtrusionPipeline {
245            depth: self.create_extrusion(shaders::ExtrusionPass::Depth),
246            color: self.create_extrusion(shaders::ExtrusionPass::Color),
247            clear: self.create_extrusion(shaders::ExtrusionPass::Clear),
248            compact_depth: self.create_compact_extrusion(shaders::ExtrusionPass::Depth),
249            compact_color: self.create_compact_extrusion(shaders::ExtrusionPass::Color),
250            compact_clear: self.create_compact_extrusion(shaders::ExtrusionPass::Clear),
251            pattern: None,
252            translucent_back: None,
253            translucent_front: None,
254            compact_translucent_back: None,
255            compact_translucent_front: None,
256        });
257        if needs_pattern {
258            ep.pattern
259                .get_or_insert_with(|| self.create_extrusion(shaders::ExtrusionPass::PatternColor));
260        }
261        if needs_translucent {
262            ep.translucent_back.get_or_insert_with(|| {
263                self.create_extrusion(shaders::ExtrusionPass::TranslucentBack)
264            });
265            ep.translucent_front.get_or_insert_with(|| {
266                self.create_extrusion(shaders::ExtrusionPass::TranslucentFront)
267            });
268            ep.compact_translucent_back.get_or_insert_with(|| {
269                self.create_compact_extrusion(shaders::ExtrusionPass::TranslucentBack)
270            });
271            ep.compact_translucent_front.get_or_insert_with(|| {
272                self.create_compact_extrusion(shaders::ExtrusionPass::TranslucentFront)
273            });
274        }
275    }

277    pub(super) fn initialize(
278        &self,
279        vector: &mut Eventually<VectorPipeline>,
280        line: &mut Eventually<LinePipeline>,
281        circle: &mut Eventually<CirclePipeline>,
282        dashes: Option<&wgpu::BindGroupLayout>,
283        needs: VectorPipelineNeeds,
284    ) {
285        if needs.fill {
286            vector.initialize(|| {
287                VectorPipeline(self.create(
288                    "vector_pipeline",
289                    &shaders::FillShader {
290                        format: self.format,
291                    },
292                    &[self.projection],
293                    false,
294                ))
295            });
296        }
297        if needs.flat_lines || needs.terrain_lines || needs.elevated_lines {
298            if let Some(dashes) = dashes {
299                let line_shader = shaders::LineShader {
300                    format: self.format,
301                };
302                let line = line.initialize(|| LinePipeline {
303                    flat: None,
304                    structure: None,
305                    elevated: None,
306                    terrain: None,
307                });
308                if needs.flat_lines {
309                    line.flat.get_or_insert_with(|| {
310                        self.create(
311                            "line_pipeline",
312                            &line_shader,
313                            &[self.projection, dashes],
314                            false,
315                        )
316                    });
317                }
318                if needs.terrain_lines {
319                    line.structure.get_or_insert_with(|| {
320                        self.create(
321                            "spatial_line_pipeline",
322                            &line_shader,
323                            &[self.projection, dashes],
324                            true,
325                        )
326                    });
327                    line.terrain
328                        .get_or_insert_with(|| self.create_terrain_line(dashes));
329                }
330                if needs.elevated_lines {
331                    line.elevated.get_or_insert_with(|| {
332                        self.create_with_clipping(
333                            "elevated_line_pipeline",
334                            &line_shader,
335                            &[self.projection, dashes],
336                            true,
337                            false,
338                        )
339                    });
340                }
341            }
342        }
343        if needs.circle {
344            let cp = circle.initialize(|| {
345                // Circles cross tile boundaries, as in GL JS, where each point belongs to one tile.
346                CirclePipeline {
347                    pair: self.create_with_clipping(
348                        "circle_pipeline",
349                        &shaders::CircleShader {
350                            format: self.format,
351                        },
352                        &[self.projection],
353                        false,
354                        false,
355                    ),
356                    feature_stroke: None,
357                    dynamic_radius: None,
358                }
359            });
360            if needs.circle_feature_stroke {
361                cp.feature_stroke.get_or_insert_with(|| {
362                    self.describe(
363                        "circle_feature_stroke_pipeline",
364                        &shaders::CircleFeatureStrokeShader {
365                            format: self.format,
366                        },
367                        false,
368                        false,
369                    )
370                    .initialize_with_prefix_layouts(self.device, &[self.projection])
371                });
372            }
373            if needs.circle_dynamic_radius {
374                cp.dynamic_radius.get_or_insert_with(|| {
375                    self.describe(
376                        "circle_dynamic_radius_pipeline",
377                        &shaders::CircleDynamicRadiusShader {
378                            format: self.format,
379                        },
380                        false,
381                        false,
382                    )
383                    .initialize_with_prefix_layouts(self.device, &[self.projection])
384                });
385            }
386        }
387    }
