1//! Fast-path `earcut` polygon triangulation for simple exterior + hole rings, falling back to
2//! `lyon` when curves, self-intersections, or non-standard multi-ring winding appear.
3

5use lyon::{
6    path::{Event, Path},
7    tessellation::{VertexBuffers, VertexId},

13/// Maximum vertex count per polygon handled on the verified `earcut` fast path; larger coastlines
14/// use `lyon`'s sweep-line tessellator.
15const MAX_SIMPLE_VERTICES: usize = 192;
16
17/// Reusable scratch buffers for zero-allocation per-polygon `earcut` triangulation.
18#[derive(Default)]
19pub(crate) struct PolygonEarcut {
20    earcut: Earcut<f32>,
21    rings: Vec<Vec<[f32; 2]>>,
22    flat_vertices: Vec<[f32; 2]>,
23    hole_indices: Vec<u32>,
24    triangles: Vec<u32>,
25}
26
27impl PolygonEarcut {
28    /// Triangulates `path` directly into `buffer` when `path` is a clean simple polygon (with
29    /// optional strictly interior non-crossing holes) whose fill matches `even_odd` (`false` =
30    /// `FillRule::NonZero`, `true` = `FillRule::EvenOdd`). Returns `false` without modifying
31    /// `buffer` when `lyon` should handle `path`.
32    pub(crate) fn try_triangulate<I, F>(
33        &mut self,
34        path: &Path,
35        even_odd: bool,
36        buffer: &mut VertexBuffers<ShaderVertex, I>,
37        mut make_vertex: F,
38    ) -> bool
39    where
40        I: From<VertexId> + MaxIndex + Copy + Into<u32>,
41        F: FnMut([f32; 2]) -> ShaderVertex,
42    {
43        if !self.extract_rings(path) {
44            return false;
45        }
46        if self.rings.is_empty() {
47            return true;
48        }
49        let total_vertices: usize = self.rings.iter().map(Vec::len).sum();
50        if total_vertices > MAX_SIMPLE_VERTICES {
51            return false;
52        }
53        let Some(last_vertex) = buffer
54            .vertices
55            .len()
56            .checked_add(total_vertices)
57            .and_then(|end| end.checked_sub(1))
58        else {
59            return false;
60        };
61        if last_vertex > I::MAX {
62            return false;
63        }
64
65        if !validate_ring_topology(&self.rings, even_odd) || rings_have_intersections(&self.rings) {
66            return false;
67        }
68
69        self.flat_vertices.clear();
70        self.hole_indices.clear();
71        for (idx, ring) in self.rings.iter().enumerate() {
72            if idx > 0 {
73                self.hole_indices.push(self.flat_vertices.len() as u32);
74            }
75            self.flat_vertices.extend_from_slice(ring);
76        }
77
78        self.earcut.earcut(
79            self.flat_vertices.iter().copied(),
80            &self.hole_indices,
81            &mut self.triangles,
82        );
83
84        let expected_triangles = total_vertices + 2 * self.hole_indices.len() - 2;
85        if self.triangles.len() != expected_triangles * 3 {
86            return false;
87        }
88
89        let base = buffer.vertices.len() as u32;
90        buffer.vertices.reserve(self.flat_vertices.len());
91        for &pt in &self.flat_vertices {
92            buffer.vertices.push(make_vertex(pt));
93        }
94        buffer.indices.reserve(self.triangles.len());
95        for &idx in &self.triangles {
96            buffer.indices.push(I::from(VertexId(base + idx)));
97        }
98        true
99    }
100
101    fn extract_rings(&mut self, path: &Path) -> bool {
102        self.rings.clear();
103        let mut current: Vec<[f32; 2]> = Vec::new();
104        for event in path.iter() {
105            match event {
106                Event::Begin { at } => {
107                    current.clear();
108                    current.push(at.to_array());
109                }
110                Event::Line { to, .. } => {
111                    let pt = to.to_array();
112                    if current.last().copied() != Some(pt) {
113                        current.push(pt);
114                    }
115                }
116                Event::End { .. } => {
117                    if current.len() >= 2 && current.first() == current.last() {
118                        current.pop();
119                    }
120                    if current.is_empty() {
121                        continue;
122                    }
123                    if current.len() < 3 {
124                        return false;
125                    }
126                    self.rings.push(std::mem::take(&mut current));
127                }
128                Event::Quadratic { .. } | Event::Cubic { .. } => return false,
129            }
130        }
131        true
132    }
133}

135fn signed_area(ring: &[[f32; 2]]) -> f32 {
136    ring.iter()
137        .zip(ring.iter().cycle().skip(1))
138        .map(|(a, b)| a[0] * b[1] - b[0] * a[1])
139        .sum()
140}
141
142fn point_in_ring(ring: &[[f32; 2]], [x, y]: [f32; 2]) -> bool {
143    let mut inside = false;
144    for (a, b) in ring.iter().zip(ring.iter().cycle().skip(1)) {
145        if (a[1] > y) != (b[1] > y) && x < (b[0] - a[0]) * (y - a[1]) / (b[1] - a[1]) + a[0] {
146            inside = !inside;
147        }
148    }
149    inside
150}
151
152fn validate_ring_topology(rings: &[Vec<[f32; 2]>], even_odd: bool) -> bool {
153    let outer_area = signed_area(&rings[0]);
154    if !outer_area.is_finite() || outer_area.abs() < 1e-4 {
155        return false;
156    }
157    for (i, hole) in rings.iter().enumerate().skip(1) {
158        let area = signed_area(hole);
159        if !area.is_finite() || area.abs() < 1e-4 {
160            return false;
161        }
162        if !even_odd && outer_area * area > 0.0 {
163            return false;
164        }
165        if !point_in_ring(&rings[0], hole[0]) {
166            return false;
167        }
168        for (j, other_hole) in rings.iter().enumerate().skip(1) {
169            if i != j && point_in_ring(other_hole, hole[0]) {
170                return false;
171            }
172        }
173    }
174    true
175}
176
177fn rings_have_intersections(rings: &[Vec<[f32; 2]>]) -> bool {
178    for (r0, ring_a) in rings.iter().enumerate() {
179        let len_a = ring_a.len();
180        for e0 in 0..len_a {
181            let a0 = ring_a[e0];
182            let a1 = ring_a[(e0 + 1) % len_a];
183            let ax_min = a0[0].min(a1[0]);
184            let ax_max = a0[0].max(a1[0]);
185            let ay_min = a0[1].min(a1[1]);
186            let ay_max = a0[1].max(a1[1]);
187
188            for (r1, ring_b) in rings.iter().enumerate().skip(r0) {
189                let len_b = ring_b.len();
190                let start_e1 = if r0 == r1 { e0 + 1 } else { 0 };
191                for e1 in start_e1..len_b {
192                    if r0 == r1 && (e1 == e0 + 1 || (e0 == 0 && e1 + 1 == len_b)) {
193                        continue;
194                    }
195                    let b0 = ring_b[e1];
196                    let b1 = ring_b[(e1 + 1) % len_b];
197                    if ax_max < b0[0].min(b1[0])
198                        || b0[0].max(b1[0]) < ax_min
199                        || ay_max < b0[1].min(b1[1])
200                        || b0[1].max(b1[1]) < ay_min
201                    {
202                        continue;
203                    }
204                    if segments_touch_or_cross(a0, a1, b0, b1) {
205                        return true;
206                    }
207                }
208            }
209        }
210    }
211    false
212}

214fn cross(o: [f32; 2], a: [f32; 2], b: [f32; 2]) -> f32 {
215    (a[0] - o[0]) * (b[1] - o[1]) - (a[1] - o[1]) * (b[0] - o[0])
216}
217
218fn on_segment(p: [f32; 2], a: [f32; 2], b: [f32; 2]) -> bool {
219    p[0] >= a[0].min(b[0]) - 1e-5
220        && p[0] <= a[0].max(b[0]) + 1e-5
221        && p[1] >= a[1].min(b[1]) - 1e-5
222        && p[1] <= a[1].max(b[1]) + 1e-5
223}
224
225fn segments_touch_or_cross(a0: [f32; 2], a1: [f32; 2], b0: [f32; 2], b1: [f32; 2]) -> bool {
226    let d1 = cross(b0, b1, a0);
227    let d2 = cross(b0, b1, a1);
228    let d3 = cross(a0, a1, b0);
229    let d4 = cross(a0, a1, b1);
230
231    if ((d1 > 0.0 && d2 < 0.0) || (d1 < 0.0 && d2 > 0.0))
232        && ((d3 > 0.0 && d4 < 0.0) || (d3 < 0.0 && d4 > 0.0))
233    {
234        return true;
235    }
236    (d1.abs() <= 1e-5 && on_segment(a0, b0, b1))
237        || (d2.abs() <= 1e-5 && on_segment(a1, b0, b1))
238        || (d3.abs() <= 1e-5 && on_segment(b0, a0, a1))
239        || (d4.abs() <= 1e-5 && on_segment(b1, a0, a1))
240}

253    fn build_path(rings: &[&[[f32; 2]]]) -> Path {
254        let mut builder = Path::builder();
255        for ring in rings {
256            builder.begin(Point::new(ring[0][0], ring[0][1]));
257            for pt in &ring[1..] {
258                builder.line_to(Point::new(pt[0], pt[1]));
259            }
260            builder.end(true);
261        }
262        builder.build()
263    }

265    fn total_unsigned_area(buf: &VertexBuffers<ShaderVertex, u32>) -> f32 {
266        buf.indices
267            .as_chunks::<3>()
268            .0
269            .iter()
270            .map(|tri| {
271                let a = buf.vertices[tri[0] as usize].position;
272                let b = buf.vertices[tri[1] as usize].position;
273                let c = buf.vertices[tri[2] as usize].position;
274                0.5 * ((b[0] - a[0]) * (c[1] - a[1]) - (b[1] - a[1]) * (c[0] - a[0])).abs()
275            })
276            .sum()
277    }

279    #[test]
280    fn concave_and_holed_polygons_match_lyon_area() {
281        let outer: &[[f32; 2]] = &[
282            [0.0, 0.0],
283            [100.0, 0.0],
284            [100.0, 40.0],
285            [40.0, 40.0],
286            [40.0, 100.0],
287            [0.0, 100.0],
288        ];
289        let hole: &[[f32; 2]] = &[[10.0, 10.0], [10.0, 30.0], [30.0, 30.0], [30.0, 10.0]];
290        let path = build_path(&[outer, hole]);
291
292        let mut earcut = PolygonEarcut::default();
293        let mut fast_buf = VertexBuffers::<ShaderVertex, u32>::new();
294        assert!(earcut.try_triangulate(&path, false, &mut fast_buf, |pos| {
295            ShaderVertex::new(pos, [0.0, 0.0])
296        }));
297
298        let mut lyon_buf = VertexBuffers::<ShaderVertex, u32>::new();
299        FillTessellator::new()
300            .tessellate_path(
301                &path,
302                &FillOptions::default().with_fill_rule(FillRule::NonZero),
303                &mut BuffersBuilder::new(&mut lyon_buf, VertexConstructor {}),
304            )
305            .expect("lyon fill");
306
307        assert!((total_unsigned_area(&fast_buf) - total_unsigned_area(&lyon_buf)).abs() < 1e-2);
308    }

310    #[test]
311    fn self_intersecting_and_same_wound_nonzero_rings_fall_back_to_lyon() {
312        let mut earcut = PolygonEarcut::default();
313        let mut buf = VertexBuffers::<ShaderVertex, u32>::new();
314
315        // Bowtie self-intersection must fall back.
316        let bowtie = build_path(&[&[[0.0, 0.0], [100.0, 100.0], [100.0, 0.0], [0.0, 100.0]]]);
317        assert!(!earcut.try_triangulate(&bowtie, false, &mut buf, |pos| {
318            ShaderVertex::new(pos, [0.0, 0.0])
319        }));
320        assert!(buf.vertices.is_empty());
321
322        // Same-wound concentric rings reject NonZero (`even_odd = false`) but accept EvenOdd (`true`).
323        let outer: &[[f32; 2]] = &[[0.0, 0.0], [100.0, 0.0], [100.0, 100.0], [0.0, 100.0]];
324        let inner_same_winding: &[[f32; 2]] =
325            &[[20.0, 20.0], [80.0, 20.0], [80.0, 80.0], [20.0, 80.0]];
326        let concentric = build_path(&[outer, inner_same_winding]);
327        assert!(
328            !earcut.try_triangulate(&concentric, false, &mut buf, |pos| {
329                ShaderVertex::new(pos, [0.0, 0.0])
330            })
331        );
332        assert!(earcut.try_triangulate(&concentric, true, &mut buf, |pos| {
333            ShaderVertex::new(pos, [0.0, 0.0])
334        }));
335        assert!((total_unsigned_area(&buf) - 6400.0).abs() < 1e-2);
336    }
