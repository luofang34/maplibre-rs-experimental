1//! Splits unbounded concave miter vertices emitted by `lyon_tessellation`.
2//!
3//! `lyon_tessellation` evaluates segment-folding heuristics against `half_width` in tile units
4//! (`0.5` by default), whereas line and fill-outline shaders extrude vertex normals in screen
5//! pixels (`line-width` or `1 px`) independent of tile-to-screen zoom. On sharp `< 1°` coastal
6//! zig-zags whose edges exceed `0.5` tile units yet remain shorter than the screen-space miter
7//! reach, `lyon` leaves the concave (`back_side`) corner as an un-folded single miter vertex with
8//! `|normal| = 1 / sin(theta / 2)`, producing long spikes without pinching only if split into two
9//! segment-perpendicular vertices (`inner_join`).
10
11use lyon::tessellation::{VertexBuffers, VertexId};
12
13use crate::render::ShaderVertex;
14
15#[derive(Clone, Copy)]
16struct SplitJoin {
17    spiky_vertex: usize,
18    join_tri_offset: usize,
19    front_prev: usize,
20    front_next: usize,
21    back_is_positive: bool,
22}
23
24/// Replaces any concave join `single_vertex` in `vertex_start..` whose normal magnitude exceeds
25/// `max_normal` with an `inner_join` bevel pair preserving full perpendicular stroke width on both
26/// incident segments.
27pub(super) fn split_unclipped_concave_joins<I>(
28    buffer: &mut VertexBuffers<ShaderVertex, I>,
29    vertex_start: usize,
30    index_start: usize,
31    max_normal: f32,
32) where
33    I: Copy + Into<u32> + From<VertexId>,
34{
35    let vertex_end = buffer.vertices.len();
36    let index_end = buffer.indices.len();
37    if vertex_start >= vertex_end || index_start >= index_end {
38        return;
39    }
40    let max_sq = max_normal * max_normal;
41    let norm_sq = |n: [f32; 2]| n[0] * n[0] + n[1] * n[1];
42    if !buffer.vertices[vertex_start..vertex_end]
43        .iter()
44        .any(|v| norm_sq(v.normal) > max_sq)
45    {
46        return;
47    }
48
49    let mut is_spiky: Vec<bool> = buffer.vertices[vertex_start..vertex_end]
50        .iter()
51        .map(|v| norm_sq(v.normal) > max_sq)
52        .collect();
53    let spiky = |idx: usize, flags: &[bool]| -> bool {
54        idx.checked_sub(vertex_start)
55            .and_then(|s| flags.get(s).copied())
56            .unwrap_or(false)
57    };
58
59    let mut splits: Vec<Option<SplitJoin>> = vec![None; vertex_end - vertex_start];
60    for tri_offset in (index_start..index_end).step_by(3) {
61        let i0 = buffer.indices[tri_offset].into() as usize;
62        let i1 = buffer.indices[tri_offset + 1].into() as usize;
63        let i2 = buffer.indices[tri_offset + 2].into() as usize;
64        let p0 = buffer.vertices[i0].position;
65        if p0 != buffer.vertices[i1].position || p0 != buffer.vertices[i2].position {
66            continue;
67        }
68        let (s0, s1, s2) = (
69            spiky(i0, &is_spiky),
70            spiky(i1, &is_spiky),
71            spiky(i2, &is_spiky),
72        );
73        if u8::from(s0) + u8::from(s1) + u8::from(s2) != 1 {
74            continue;
75        }
76        let (v, b, c) = if s0 {
77            (i0, i1, i2)
78        } else if s1 {
79            (i1, i2, i0)
80        } else {
81            (i2, i0, i1)
82        };
83        if b == c {
84            continue;
85        }
86        let (front_prev, front_next, back_is_positive) =
87            if b < c { (b, c, false) } else { (c, b, true) };
88        splits[v - vertex_start] = Some(SplitJoin {
89            spiky_vertex: v,
90            join_tri_offset: tri_offset,
91            front_prev,
92            front_next,
93            back_is_positive,
94        });
95    }
96
97    let mut back_next_for = vec![0_usize; vertex_end - vertex_start];
98    for (slot, maybe_split) in splits.iter().enumerate() {
99        let Some(split) = maybe_split else {
100            continue;
101        };
102        let prev_front = buffer.vertices[split.front_prev];
103        let next_front = buffer.vertices[split.front_next];
104        let mut back_next_vertex = buffer.vertices[split.spiky_vertex];
105        buffer.vertices[split.spiky_vertex].normal = [-prev_front.normal[0], -prev_front.normal[1]];
106        buffer.vertices[split.spiky_vertex].distance = prev_front.distance;
107        back_next_vertex.normal = [-next_front.normal[0], -next_front.normal[1]];
108        back_next_vertex.distance = next_front.distance;
109        let back_next_idx = buffer.vertices.len();
110        buffer.vertices.push(back_next_vertex);
111        back_next_for[slot] = back_next_idx;
112        is_spiky[slot] = false;
113    }
114
115    for tri_offset in (index_start..index_end).step_by(3) {
116        let tri = [
117            buffer.indices[tri_offset].into() as usize,
118            buffer.indices[tri_offset + 1].into() as usize,
119            buffer.indices[tri_offset + 2].into() as usize,
120        ];
121        let join_match = tri.iter().find_map(|&idx| {
122            idx.checked_sub(vertex_start)
123                .and_then(|s| splits.get(s).copied().flatten())
124                .filter(|split| split.join_tri_offset == tri_offset)
125        });
126        if let Some(split) = join_match {
127            let fp = I::from(VertexId::from_usize(split.front_prev));
128            let fnx = I::from(VertexId::from_usize(split.front_next));
129            let bp = I::from(VertexId::from_usize(split.spiky_vertex));
130            let bn = I::from(VertexId::from_usize(
131                back_next_for[split.spiky_vertex - vertex_start],
132            ));
133            if split.back_is_positive {
134                buffer.indices[tri_offset] = fp;
135                buffer.indices[tri_offset + 1] = bp;
136                buffer.indices[tri_offset + 2] = fnx;
137                buffer.indices.extend([fp, bn, bp]);
138            } else {
139                buffer.indices[tri_offset] = fp;
140                buffer.indices[tri_offset + 1] = fnx;
141                buffer.indices[tri_offset + 2] = bp;
142                buffer.indices.extend([fp, bp, bn]);
143            }
144            continue;
145        }
146
147        for (corner, &idx) in tri.iter().enumerate() {
148            let Some(slot) = idx.checked_sub(vertex_start) else {
149                continue;
150            };
151            let Some(split) = splits.get(slot).copied().flatten() else {
152                continue;
153            };
154            let use_back_next = if split.back_is_positive {
155                tri.contains(&split.front_next)
156            } else {
157                !tri.contains(&split.front_prev)
158            };
159            if use_back_next {
160                buffer.indices[tri_offset + corner] =
161                    I::from(VertexId::from_usize(back_next_for[slot]));
162            }
163        }
164    }
165
166    if is_spiky.iter().any(|&flag| flag) {
167        recover_closed_start_normals(
168            buffer,
169            vertex_start,
170            index_start,
171            index_end,
172            max_normal,
173            &mut is_spiky,
174        );
175    }
176}

178fn recover_closed_start_normals<I>(
179    buffer: &mut VertexBuffers<ShaderVertex, I>,
180    vertex_start: usize,
181    index_start: usize,
182    index_end: usize,
183    max_normal: f32,
184    is_spiky: &mut [bool],
185) where
186    I: Copy + Into<u32>,
187{
188    let spiky = |idx: usize, flags: &[bool]| -> bool {
189        idx.checked_sub(vertex_start)
190            .and_then(|s| flags.get(s).copied())
191            .unwrap_or(false)
192    };
193    for tri_offset in (index_start..index_end).step_by(3) {
194        let tri = [
195            buffer.indices[tri_offset].into() as usize,
196            buffer.indices[tri_offset + 1].into() as usize,
197            buffer.indices[tri_offset + 2].into() as usize,
198        ];
199        for &idx in &tri {
200            if !spiky(idx, is_spiky) {
201                continue;
202            }
203            let pos = buffer.vertices[idx].position;
204            if let Some(&partner) = tri.iter().find(|&&other| {
205                other != idx && !spiky(other, is_spiky) && buffer.vertices[other].position == pos
206            }) {
207                let pn = buffer.vertices[partner].normal;
208                buffer.vertices[idx].normal = [-pn[0], -pn[1]];
209                is_spiky[idx - vertex_start] = false;
210            }
211        }
212    }
213    for (offset, &still_spiky) in is_spiky.iter().enumerate() {
214        if still_spiky {
215            let v = &mut buffer.vertices[vertex_start + offset];
216            let len = (v.normal[0] * v.normal[0] + v.normal[1] * v.normal[1]).sqrt();
217            if len > max_normal {
218                let scale = max_normal / len;
219                v.normal[0] *= scale;
220                v.normal[1] *= scale;
221            }
222        }
223    }
224}
