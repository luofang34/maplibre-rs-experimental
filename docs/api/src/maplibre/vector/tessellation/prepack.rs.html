1//! Incremental worker-side GPU vertex encoding that compacts vertices after each polygon or
2//! stroke instead of materializing multi-megabyte intermediate `Vec<ShaderVertex>` buffers.
3
4use super::{IndexDataType, OverAlignedVertexBuffer, PrepackedGpuGeometry};
5use crate::render::{
6    shaders::{
7        encode_compact_extrusion_words, is_compact_extrusion_vertex, unpack_compact_extrusion_word,
8        GpuShaderVertex, GpuVertexKind,
9    },
10    ShaderVertex,
11};
12
13enum PrepackStorage {
14    CompactExtrusion {
15        words: Vec<GpuShaderVertex>,
16        half: Option<[u32; 2]>,
17        logical_len: usize,
18    },
19    Standard(Vec<GpuShaderVertex>),
20}
21
22/// Streams tessellated [`ShaderVertex`] slices directly into compact 8-byte extrusion pairs or
23/// standard 16-byte [`GpuShaderVertex`] words.
24pub(crate) struct StreamingPrepack {
25    kind: GpuVertexKind,
26    storage: PrepackStorage,
27}
28
29impl StreamingPrepack {
30    /// Creates a streaming packer for `style_type`, using compact 8-byte extrusion words when
31    /// `allow_compact_extrusion` holds and upgrading automatically if any vertex exceeds limits.
32    pub(crate) fn new(style_type: &str, allow_compact_extrusion: bool) -> Option<Self> {
33        let kind = GpuVertexKind::from_layer_type(style_type)?;
34        let storage = if kind == GpuVertexKind::Extrusion && allow_compact_extrusion {
35            PrepackStorage::CompactExtrusion {
36                words: Vec::new(),
37                half: None,
38                logical_len: 0,
39            }
40        } else {
41            PrepackStorage::Standard(Vec::new())
42        };
43        Some(Self { kind, storage })
44    }
45
46    #[inline]
47    pub(crate) fn logical_len(&self) -> usize {
48        match &self.storage {
49            PrepackStorage::CompactExtrusion { logical_len, .. } => *logical_len,
50            PrepackStorage::Standard(verts) => verts.len(),
51        }
52    }
53
54    /// Encodes `vertices` into GPU form, upgrading `CompactExtrusion -> Standard` if needed.
55    pub(crate) fn push_vertices(&mut self, vertices: &[ShaderVertex]) {
56        if vertices.is_empty() {
57            return;
58        }
59        match &mut self.storage {
60            PrepackStorage::CompactExtrusion {
61                words,
62                half,
63                logical_len,
64            } => {
65                for (idx, vertex) in vertices.iter().enumerate() {
66                    if is_compact_extrusion_vertex(vertex) {
67                        let pair = encode_compact_extrusion_words(vertex);
68                        if let Some(first) = half.take() {
69                            words.push(GpuShaderVertex([first[0], first[1], pair[0], pair[1]]));
70                        } else {
71                            *half = Some(pair);
72                        }
73                        *logical_len += 1;
74                    } else {
75                        let mut standard =
76                            Vec::with_capacity(*logical_len + (vertices.len() - idx));
77                        for word in words.iter() {
78                            standard.push(unpack_compact_extrusion_word([word.0[0], word.0[1]]));
79                            standard.push(unpack_compact_extrusion_word([word.0[2], word.0[3]]));
80                        }
81                        if let Some(pending) = half.take() {
82                            standard.push(unpack_compact_extrusion_word(pending));
83                        }
84                        for rem in &vertices[idx..] {
85                            standard.push(self.kind.encode(rem));
86                        }
87                        self.storage = PrepackStorage::Standard(standard);
88                        return;
89                    }
90                }
91            }
92            PrepackStorage::Standard(out) => {
93                out.reserve(vertices.len());
94                for vertex in vertices {
95                    out.push(self.kind.encode(vertex));
96                }
97            }
98        }
99    }
100
101    /// Finalizes packed GPU vertices and narrows indices to aligned `u16` when every index fits.
102    pub(crate) fn finish(
103        self,
104        mut raw_indices: Vec<IndexDataType>,
105    ) -> OverAlignedVertexBuffer<ShaderVertex, IndexDataType> {
106        let logical_len = self.logical_len();
107        if logical_len == 0 {
108            return OverAlignedVertexBuffer::empty();
109        }
110        let (mut vertices, compact_extrusion) = match self.storage {
111            PrepackStorage::CompactExtrusion {
112                mut words, half, ..
113            } => {
114                if let Some(first) = half {
115                    words.push(GpuShaderVertex([first[0], first[1], 0, 0]));
116                }
117                (words, true)
118            }
119            PrepackStorage::Standard(verts) => (verts, false),
120        };
121        vertices.shrink_to_fit();
122
123        let usable = raw_indices.len() as u32;
124        let logical_vertex_count = u32::try_from(logical_len).unwrap_or(u32::MAX);
125        let can_u16 =
126            logical_len <= 65_536 && raw_indices.iter().all(|&idx| idx <= u32::from(u16::MAX));
127        if can_u16 {
128            let padded = raw_indices.len().next_multiple_of(2);
129            let mut u16s = Vec::with_capacity(padded);
130            u16s.extend(raw_indices.iter().map(|&idx| idx as u16));
131            if u16s.len() < padded {
132                u16s.push(0);
133            }
134            OverAlignedVertexBuffer::from_vecs(Vec::new(), Vec::new(), usable).with_prepacked_gpu(
135                PrepackedGpuGeometry {
136                    vertices,
137                    indices_u16: Some(u16s),
138                    compact_extrusion,
139                    logical_vertex_count,
140                },
141            )
142        } else {
143            raw_indices.shrink_to_fit();
144            OverAlignedVertexBuffer::from_vecs(Vec::new(), raw_indices, usable).with_prepacked_gpu(
145                PrepackedGpuGeometry {
146                    vertices,
147                    indices_u16: None,
148                    compact_extrusion,
149                    logical_vertex_count,
150                },
151            )
152        }
153    }
154}

166    #[test]
167    fn streaming_fill_and_line_match_batch_prepack_bit_for_bit() {
168        for (layer_type, is_line) in [("fill", false), ("line", true)] {
169            let mut batch = ZeroTessellator::<IndexDataType> {
170                is_line_layer: is_line,
171                ..Default::default()
172            };
173
174            let mut streaming = ZeroTessellator::<IndexDataType> {
175                is_line_layer: is_line,
176                ..Default::default()
177            };
178            streaming.enable_streaming_prepack(layer_type, false);
179
180            for t in [&mut batch, &mut streaming] {
181                for offset in [100.0_f64, 500.0_f64] {
182                    if is_line {
183                        t.linestring_begin(true, 3, 0).expect("line begin");
184                        t.xy(offset, 100.0, 0).expect("xy0");
185                        t.xy(offset + 80.0, 140.0, 1).expect("xy1");
186                        t.xy(offset + 120.0, 220.0, 2).expect("xy2");
187                        t.linestring_end(true, 0).expect("line end");
188                    } else {
189                        t.polygon_begin(true, 4, 0).expect("poly begin");
190                        t.xy(offset, 100.0, 0).expect("xy0");
191                        t.xy(offset + 80.0, 100.0, 1).expect("xy1");
192                        t.xy(offset + 80.0, 180.0, 2).expect("xy2");
193                        t.xy(offset, 180.0, 3).expect("xy3");
194                        t.polygon_end(true, 0).expect("poly end");
195                    }
196                    t.finish_feature();
197                }
198            }
199
200            assert_eq!(batch.feature_indices, streaming.feature_indices);
201            let mut expected = Vec::new();
202            ShaderVertex::encode_vertices(layer_type, &batch.buffer.vertices, &mut expected);
203            let packed = streaming.into_over_aligned_buffer();
204            let pre = packed.prepacked_gpu().expect("streaming prepacked");
205            assert!(!pre.compact_extrusion);
206            assert_eq!(pre.vertices, expected);
207            assert_eq!(
208                pre.indices_u16.as_ref().expect("u16")[..batch.buffer.indices.len()]
209                    .iter()
210                    .copied()
211                    .map(u32::from)
212                    .collect::<Vec<_>>(),
213                batch.buffer.indices
214            );
215        }
216    }

218    #[test]
219    fn streaming_compact_extrusion_upgrades_cleanly_for_tall_skyscrapers() {
220        let mut t = ZeroTessellator::<IndexDataType> {
221            extrusion: Some(ExtrusionOptions {
222                height: None,
223                base: None,
224                base_datum: None,
225                height_datum: None,
226                part: None,
227                rounded_corner_distance: None,
228                translucent: false,
229                clip_buffer: None,
230                has_pattern: false,
231                overscale: 0,
232            }),
233            ..Default::default()
234        };
235        t.enable_streaming_prepack("fill-extrusion", true);
236
237        t.polygon_begin(true, 4, 0).expect("poly");
238        t.xy(100.0, 100.0, 0).expect("p0");
239        t.xy(150.0, 100.0, 1).expect("p1");
240        t.xy(150.0, 150.0, 2).expect("p2");
241        t.xy(100.0, 150.0, 3).expect("p3");
242        t.polygon_end(true, 0).expect("end");
243        t.finish_feature();
244
245        let compact_buf = t.into_over_aligned_buffer();
246        let compact_pre = compact_buf.prepacked_gpu().expect("compact prepack");
247        assert!(compact_pre.compact_extrusion);
248        assert!(compact_pre.logical_vertex_count > 0);
249        assert_eq!(
250            compact_pre.vertices.len(),
251            (compact_pre.logical_vertex_count as usize).div_ceil(2)
252        );
253
254        let mut packer = StreamingPrepack::new("fill-extrusion", true).expect("packer");
255        let small = ShaderVertex::new([20.0, 30.0], [1.0, 0.0]);
256        let mut tall = ShaderVertex::new([40.0, 50.0], [0.0, 1.0]);
257        tall.elevation = 1500.0; // Exceeds compact 1023.5 m ceiling.
258        packer.push_vertices(&[small, tall]);
259        let upgraded = packer.finish(vec![0, 1, 0]);
260        let up_pre = upgraded.prepacked_gpu().expect("upgraded prepack");
261        assert!(!up_pre.compact_extrusion);
262        assert_eq!(up_pre.logical_vertex_count, 2);
263        assert_eq!(up_pre.vertices.len(), 2);
264        assert_eq!(up_pre.vertices[1], GpuVertexKind::Extrusion.encode(&tall));
265    }
