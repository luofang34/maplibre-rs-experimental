1//! Incremental worker-side GPU vertex encoding that compacts vertices after each polygon or
2//! stroke instead of materializing multi-megabyte intermediate `Vec<ShaderVertex>` buffers.
3
4use super::{IndexDataType, OverAlignedVertexBuffer, PrepackedGpuGeometry};
5use crate::render::{
6    shaders::{
7        encode_compact_extrusion_words, is_compact_extrusion_vertex, unpack_compact_extrusion_word,
8        GpuShaderVertex, GpuVertexKind,
9    },
10    ShaderVertex,
11};

13enum PrepackStorage {
14    CompactExtrusion {
15        words: Vec<GpuShaderVertex>,
16        half: Option<[u32; 2]>,
17        logical_len: usize,
18    },
19    Standard(Vec<GpuShaderVertex>),
20}
21
22/// Streams tessellated [`ShaderVertex`] slices directly into compact 8-byte extrusion pairs or
23/// standard 16-byte [`GpuShaderVertex`] words.
24pub(crate) struct StreamingPrepack {
25    kind: GpuVertexKind,
26    storage: PrepackStorage,
27}
28
29impl StreamingPrepack {
30    /// Creates a streaming packer for `style_type`, using compact 8-byte extrusion words when
31    /// `allow_compact_extrusion` holds and upgrading automatically if any vertex exceeds limits.
32    pub(crate) fn new(style_type: &str, allow_compact_extrusion: bool) -> Option<Self> {
33        let kind = GpuVertexKind::from_layer_type(style_type)?;
34        let storage = if kind == GpuVertexKind::Extrusion && allow_compact_extrusion {
35            PrepackStorage::CompactExtrusion {
36                words: Vec::new(),
37                half: None,
38                logical_len: 0,
39            }
40        } else {
41            PrepackStorage::Standard(Vec::new())
42        };
43        Some(Self { kind, storage })
44    }
45
46    #[inline]
47    pub(crate) fn logical_len(&self) -> usize {
48        match &self.storage {
49            PrepackStorage::CompactExtrusion { logical_len, .. } => *logical_len,
50            PrepackStorage::Standard(verts) => verts.len(),
51        }
52    }
53
54    /// Encodes `vertices` into GPU form, upgrading `CompactExtrusion -> Standard` if needed.
55    pub(crate) fn push_vertices(&mut self, vertices: &[ShaderVertex]) {
56        if vertices.is_empty() {
57            return;
58        }
59        match &mut self.storage {
60            PrepackStorage::CompactExtrusion {
61                words,
62                half,
63                logical_len,
64            } => {
65                for (idx, vertex) in vertices.iter().enumerate() {
66                    if is_compact_extrusion_vertex(vertex) {
67                        let pair = encode_compact_extrusion_words(vertex);
68                        if let Some(first) = half.take() {
69                            words.push(GpuShaderVertex([first[0], first[1], pair[0], pair[1]]));
70                        } else {
71                            *half = Some(pair);
72                        }
73                        *logical_len += 1;
74                    } else {
75                        let mut standard =
76                            Vec::with_capacity(*logical_len + (vertices.len() - idx));
77                        for word in words.iter() {
78                            standard.push(unpack_compact_extrusion_word([word.0[0], word.0[1]]));
79                            standard.push(unpack_compact_extrusion_word([word.0[2], word.0[3]]));
80                        }
81                        if let Some(pending) = half.take() {
82                            standard.push(unpack_compact_extrusion_word(pending));
83                        }
84                        for rem in &vertices[idx..] {
85                            standard.push(self.kind.encode(rem));
86                        }
87                        self.storage = PrepackStorage::Standard(standard);
88                        return;
89                    }
90                }
91            }
92            PrepackStorage::Standard(out) => {
93                out.reserve(vertices.len());
94                for vertex in vertices {
95                    out.push(self.kind.encode(vertex));
96                }
97            }
98        }
99    }
100
101    /// Appends already-encoded 16-byte [`GpuShaderVertex`] words directly into streaming storage.
102    pub(crate) fn push_encoded(&mut self, vertices: &[GpuShaderVertex]) {
103        if vertices.is_empty() {
104            return;
105        }
106        match &mut self.storage {
107            PrepackStorage::Standard(out) => out.extend_from_slice(vertices),
108            PrepackStorage::CompactExtrusion {
109                words,
110                half,
111                logical_len,
112            } => {
113                let mut standard = Vec::with_capacity(*logical_len + vertices.len());
114                for word in words.iter() {
115                    standard.push(unpack_compact_extrusion_word([word.0[0], word.0[1]]));
116                    standard.push(unpack_compact_extrusion_word([word.0[2], word.0[3]]));
117                }
118                if let Some(pending) = half.take() {
119                    standard.push(unpack_compact_extrusion_word(pending));
120                }
121                standard.extend_from_slice(vertices);
122                self.storage = PrepackStorage::Standard(standard);
123            }
124        }
125    }
126
127    /// Finalizes packed GPU vertices and narrows indices to aligned `u16` when every index fits.
128    pub(crate) fn finish(
129        self,
130        mut raw_indices: Vec<IndexDataType>,
131    ) -> OverAlignedVertexBuffer<ShaderVertex, IndexDataType> {
132        let logical_len = self.logical_len();
133        if logical_len == 0 {
134            return OverAlignedVertexBuffer::empty();
135        }
136        let (mut vertices, compact_extrusion) = match self.storage {
137            PrepackStorage::CompactExtrusion {
138                mut words, half, ..
139            } => {
140                if let Some(first) = half {
141                    words.push(GpuShaderVertex([first[0], first[1], 0, 0]));
142                }
143                (words, true)
144            }
145            PrepackStorage::Standard(verts) => (verts, false),
146        };
147        vertices.shrink_to_fit();
148
149        let usable = raw_indices.len() as u32;
150        let logical_vertex_count = u32::try_from(logical_len).unwrap_or(u32::MAX);
151        let can_u16 =
152            logical_len <= 65_536 && raw_indices.iter().all(|&idx| idx <= u32::from(u16::MAX));
153        if can_u16 {
154            let padded = raw_indices.len().next_multiple_of(2);
155            let mut u16s = Vec::with_capacity(padded);
156            u16s.extend(raw_indices.iter().map(|&idx| idx as u16));
157            if u16s.len() < padded {
158                u16s.push(0);
159            }
160            OverAlignedVertexBuffer::from_vecs(Vec::new(), Vec::new(), usable).with_prepacked_gpu(
161                PrepackedGpuGeometry {
162                    vertices,
163                    indices_u16: Some(u16s),
164                    indices_u32: None,
165                    compact_extrusion,
166                    logical_vertex_count,
167                },
168            )
169        } else {
170            raw_indices.shrink_to_fit();
171            OverAlignedVertexBuffer::from_vecs(Vec::new(), Vec::new(), usable).with_prepacked_gpu(
172                PrepackedGpuGeometry {
173                    vertices,
174                    indices_u16: None,
175                    indices_u32: Some(raw_indices),
176                    compact_extrusion,
177                    logical_vertex_count,
178                },
179            )
180        }
181    }
182}

194    #[test]
195    fn streaming_fill_and_line_match_batch_prepack_bit_for_bit() {
196        for (layer_type, is_line) in [("fill", false), ("line", true)] {
197            let mut batch = ZeroTessellator::<IndexDataType> {
198                is_line_layer: is_line,
199                ..Default::default()
200            };
201
202            let mut streaming = ZeroTessellator::<IndexDataType> {
203                is_line_layer: is_line,
204                ..Default::default()
205            };
206            streaming.enable_streaming_prepack(layer_type, false);
207
208            for t in [&mut batch, &mut streaming] {
209                for offset in [100.0_f64, 500.0_f64] {
210                    if is_line {
211                        t.linestring_begin(true, 3, 0).expect("line begin");
212                        t.xy(offset, 100.0, 0).expect("xy0");
213                        t.xy(offset + 80.0, 140.0, 1).expect("xy1");
214                        t.xy(offset + 120.0, 220.0, 2).expect("xy2");
215                        t.linestring_end(true, 0).expect("line end");
216                    } else {
217                        t.polygon_begin(true, 4, 0).expect("poly begin");
218                        t.xy(offset, 100.0, 0).expect("xy0");
219                        t.xy(offset + 80.0, 100.0, 1).expect("xy1");
220                        t.xy(offset + 80.0, 180.0, 2).expect("xy2");
221                        t.xy(offset, 180.0, 3).expect("xy3");
222                        t.polygon_end(true, 0).expect("poly end");
223                    }
224                    t.finish_feature();
225                }
226            }
227
228            assert_eq!(batch.feature_indices, streaming.feature_indices);
229            let mut expected = Vec::new();
230            ShaderVertex::encode_vertices(layer_type, &batch.buffer.vertices, &mut expected);
231            let packed = streaming.into_over_aligned_buffer();
232            let pre = packed.prepacked_gpu().expect("streaming prepacked");
233            assert!(!pre.compact_extrusion);
234            assert_eq!(pre.vertices, expected);
235            assert_eq!(
236                pre.indices_u16.as_ref().expect("u16")[..batch.buffer.indices.len()]
237                    .iter()
238                    .copied()
239                    .map(u32::from)
240                    .collect::<Vec<_>>(),
241                batch.buffer.indices
242            );
243        }
244    }

246    #[test]
247    fn streaming_compact_extrusion_upgrades_cleanly_for_tall_skyscrapers() {
248        let mut t = ZeroTessellator::<IndexDataType> {
249            extrusion: Some(ExtrusionOptions {
250                height: None,
251                base: None,
252                base_datum: None,
253                height_datum: None,
254                part: None,
255                rounded_corner_distance: None,
256                translucent: false,
257                clip_buffer: None,
258                has_pattern: false,
259                overscale: 0,
260            }),
261            ..Default::default()
262        };
263        t.enable_streaming_prepack("fill-extrusion", true);
264
265        t.polygon_begin(true, 4, 0).expect("poly");
266        t.xy(100.0, 100.0, 0).expect("p0");
267        t.xy(150.0, 100.0, 1).expect("p1");
268        t.xy(150.0, 150.0, 2).expect("p2");
269        t.xy(100.0, 150.0, 3).expect("p3");
270        t.polygon_end(true, 0).expect("end");
271        t.finish_feature();
272
273        let compact_buf = t.into_over_aligned_buffer();
274        let compact_pre = compact_buf.prepacked_gpu().expect("compact prepack");
275        assert!(compact_pre.compact_extrusion);
276        assert!(compact_pre.logical_vertex_count > 0);
277        assert_eq!(
278            compact_pre.vertices.len(),
279            (compact_pre.logical_vertex_count as usize).div_ceil(2)
280        );
281
282        let mut packer = StreamingPrepack::new("fill-extrusion", true).expect("packer");
283        let small = ShaderVertex::new([20.0, 30.0], [1.0, 0.0]);
284        let mut tall = ShaderVertex::new([40.0, 50.0], [0.0, 1.0]);
285        tall.elevation = 1500.0; // Exceeds compact 1023.5 m ceiling.
286        packer.push_vertices(&[small, tall]);
287        let upgraded = packer.finish(vec![0, 1, 0]);
288        let up_pre = upgraded.prepacked_gpu().expect("upgraded prepack");
289        assert!(!up_pre.compact_extrusion);
290        assert_eq!(up_pre.logical_vertex_count, 2);
291        assert_eq!(up_pre.vertices.len(), 2);
292        assert_eq!(up_pre.vertices[1], GpuVertexKind::Extrusion.encode(&tall));
293    }
