1//! Where a stroke's path distance starts: where the line first enters the tile's buffer, as in
2//! a source that clips its lines to each tile, so a dash pattern begins at the same place for
3//! a line that starts far outside the tile and stays precise where the distances would be huge.
4use lyon::{
5    path::{Event, Path},
6    tessellation::{StrokeVertex, StrokeVertexConstructor, VertexSource},
7};
8
9use super::line_style::PackedLine;
10use crate::render::ShaderVertex;
11
12/// Tile units either side of a tile that a line clipped for it reaches: the quarter of a tile
13/// that GL JS keeps when it clips the lines of a GeoJSON source.
14const BUFFER: f32 = 1024.0;
15const EXTENT: f32 = 4096.0;
16
17/// The distance along its sub-path at which each endpoint's line enters the buffered tile.
18pub fn entry_distances(path: &Path) -> Vec<f32> {
19    let mut origins = Vec::new();
20    let mut sub_path_start = 0;
21    let mut travelled = 0.0;
22    let mut entered = false;
23    for event in path.iter() {
24        match event {
25            Event::Begin { at } => {
26                sub_path_start = origins.len();
27                travelled = 0.0;
28                entered = inside(at);
29                origins.push(0.0);
30            }
31            Event::Line { from, to } => {
32                let length = (to - from).length();
33                if !entered {
34                    if let Some(at) = entry_parameter(from, to) {
35                        let origin = travelled + at * length;
36                        origins[sub_path_start..].fill(origin);
37                        entered = true;
38                    }
39                }
40                travelled += length;
41                origins.push(origins[sub_path_start]);
42            }
43            _ => {}
44        }
45    }
46    origins
47}
48
49fn inside(point: lyon::math::Point) -> bool {
50    (-BUFFER..=EXTENT + BUFFER).contains(&point.x) && (-BUFFER..=EXTENT + BUFFER).contains(&point.y)
51}
52
53/// The fraction of the segment at which it first lies in the buffered tile, if it does.
54fn entry_parameter(from: lyon::math::Point, to: lyon::math::Point) -> Option<f32> {
55    let (low, high) = (-BUFFER, EXTENT + BUFFER);
56    let (mut enter, mut leave) = (0.0_f32, 1.0_f32);
57    for (start, delta) in [(from.x, to.x - from.x), (from.y, to.y - from.y)] {
58        if delta.abs() < f32::EPSILON {
59            if start < low || start > high {
60                return None;
61            }
62            continue;
63        }
64        let (a, b) = ((low - start) / delta, (high - start) / delta);
65        enter = enter.max(a.min(b));
66        leave = leave.min(a.max(b));
67    }
68    (enter <= leave).then_some(enter)
69}

71/// Builds stroke vertices whose distance counts from where their line enters the tile.
72pub struct StrokeOrigins {
73    /// Entry distance of each path endpoint's sub-path, from [`entry_distances`].
74    pub origins: Vec<f32>,
75    /// What the feature's line style adds to each vertex.
76    pub packed: PackedLine,
77}
78
79impl StrokeVertexConstructor<ShaderVertex> for StrokeOrigins {
80    fn new_vertex(&mut self, vertex: StrokeVertex) -> ShaderVertex {
81        let mut output = ShaderVertex::new(
82            vertex.position_on_path().to_array(),
83            vertex.normal().to_array(),
84        );
85        let endpoint = match vertex.source() {
86            VertexSource::Endpoint { id } => id,
87            VertexSource::Edge { from, .. } => from,
88        };
89        let origin = self
90            .origins
91            .get(endpoint.to_usize())
92            .copied()
93            .unwrap_or(0.0);
94        let distance = vertex.advancement() - origin;
95        // GL JS stores a vertex's distance in whole tile units, so the phase of a dash or pattern
96        // drifts from one line to the next by up to one unit; a gradient keeps its fraction.
97        output.distance = if self.origins.is_empty() {
98            distance
99        } else {
100            distance.floor()
101        };
102        // A stroke has no elevation; the slot instead says which side of the line the vertex is
103        // on, which a pattern needs to draw the image across the line the right way up, and
104        // carries the gap width and blur.
105        output.elevation = self
106            .packed
107            .stroke_elevation(matches!(vertex.side(), lyon::tessellation::Side::Negative));
108        output.edge_distance = self.packed.style;
109        output
110    }
111}

113#[cfg(test)]
114mod tests {
115    use lyon::math::point;
116
117    use super::*;
118
119    #[test]
120    fn a_line_from_far_outside_counts_its_distance_from_where_it_reaches_the_tile() {
121        let mut builder = Path::builder();
122        builder.begin(point(-10_000.0, 100.0));
123        builder.line_to(point(10_000.0, 100.0));
124        builder.end(false);
125        let origins = entry_distances(&builder.build());
126        assert_eq!(origins.len(), 2);
127        assert!((origins[0] - (10_000.0 - BUFFER)).abs() < 0.01);
128        assert_eq!(origins[0], origins[1]);
129    }
130
131    #[test]
132    fn a_line_that_starts_inside_keeps_its_own_start() {
133        let mut builder = Path::builder();
134        builder.begin(point(10.0, 10.0));
135        builder.line_to(point(9_000.0, 10.0));
136        builder.end(false);
137        assert_eq!(entry_distances(&builder.build()), [0.0, 0.0]);
138    }
