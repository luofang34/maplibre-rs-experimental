1//! GPU copy-aligned vertex and index buffers produced by tessellation.
2
3use bytemuck::Pod;
4use lyon::tessellation::VertexBuffers;
5
6/// Geometry with a draw-index count separate from any GPU copy-alignment padding.
7/// Conversion from `VertexBuffers` pads indices and requires copy-aligned vertex bytes.
8#[derive(Clone)]
9pub struct OverAlignedVertexBuffer<V, I> {
10    /// Vertex and index storage, including any upload padding.
11    pub buffer: VertexBuffers<V, I>,
12    /// Number of leading indices to draw, excluding padding.
13    pub usable_indices: u32,
14    uploaded_vertices: usize,
15    icon_bits: Vec<u64>,
16}
17
18impl<V, I> OverAlignedVertexBuffer<V, I> {
19    /// Creates empty geometry with no drawable indices.
20    pub fn empty() -> Self {
21        Self {
22            buffer: VertexBuffers::with_capacity(0, 0),
23            usable_indices: 0,
24            uploaded_vertices: 0,
25            icon_bits: Vec::new(),
26        }
27    }
28
29    /// Wraps pre-allocated, copy-aligned vertex and index vectors without copying.
30    pub fn from_vecs(vertices: Vec<V>, indices: Vec<I>, usable_indices: u32) -> Self {
31        Self {
32            buffer: VertexBuffers { vertices, indices },
33            usable_indices,
34            uploaded_vertices: 0,
35            icon_bits: Vec::new(),
36        }
37    }
38
39    /// Reconstructs buffers without adding padding or validating `usable_indices`.
40    /// Callers must supply copy-aligned storage and a draw count within the index buffer.
41    pub fn from_iters<IV, II>(vertices: IV, indices: II, usable_indices: u32) -> Self
42    where
43        IV: IntoIterator<Item = V>,
44        II: IntoIterator<Item = I>,
45        IV::IntoIter: ExactSizeIterator,
46        II::IntoIter: ExactSizeIterator,
47    {
48        let vertices = vertices.into_iter();
49        let indices = indices.into_iter();
50        let mut buffers = VertexBuffers::with_capacity(vertices.len(), indices.len());
51        buffers.vertices.extend(vertices);
52        buffers.indices.extend(indices);
53        Self {
54            buffer: buffers,
55            usable_indices,
56            uploaded_vertices: 0,
57            icon_bits: Vec::new(),
58        }
59    }
60
61    /// Total GPU vertex count, preserved even after CPU vertex storage is released post-upload.
62    pub fn vertex_count(&self) -> usize {
63        if self.buffer.vertices.is_empty() {
64            self.uploaded_vertices
65        } else {
66            self.buffer.vertices.len()
67        }
68    }
69
70    pub(crate) fn icon_bits_bytes(&self) -> usize {
71        self.icon_bits.capacity() * std::mem::size_of::<u64>()
72    }
73}
74
75impl<I> OverAlignedVertexBuffer<crate::render::shaders::ShaderSymbolVertex, I> {
76    pub(crate) fn is_icon_vertex(&self, index: usize) -> bool {
77        if let Some(vertex) = self.buffer.vertices.get(index) {
78            matches!(vertex.a_data[2], 1 | 2)
79        } else {
80            self.icon_bits
81                .get(index / 64)
82                .is_some_and(|word| ((word >> (index % 64)) & 1) != 0)
83        }
84    }
85
86    /// Compacts per-vertex icon/text kind flags into 1 bit per vertex and frees CPU vertex storage
87    /// once `SymbolBufferPool` holds the GPU copy.
88    #[cfg(target_arch = "wasm32")]
89    pub(crate) fn release_uploaded_symbol_vertices(&mut self) {
90        if self.buffer.vertices.is_empty() {
91            return;
92        }
93        let count = self.buffer.vertices.len();
94        let mut bits = vec![0_u64; count.div_ceil(64)];
95        for (i, vertex) in self.buffer.vertices.iter().enumerate() {
96            if matches!(vertex.a_data[2], 1 | 2) {
97                bits[i / 64] |= 1_u64 << (i % 64);
98            }
99        }
100        self.uploaded_vertices = count;
101        self.icon_bits = bits;
102        self.buffer.vertices = Vec::new();
103    }
104}

106impl<V: Pod, I: Pod> From<VertexBuffers<V, I>> for OverAlignedVertexBuffer<V, I> {
107    fn from(mut buffer: VertexBuffers<V, I>) -> Self {
108        let usable_indices = buffer.indices.len() as u32;
109        buffer.align_vertices();
110        buffer.align_indices();
111        Self {
112            buffer,
113            usable_indices,
114            uploaded_vertices: 0,
115            icon_bits: Vec::new(),
116        }
117    }
118}
119
120trait Align<V: Pod, I: Pod> {
121    fn align_vertices(&mut self);
122    fn align_indices(&mut self);
123}
124
125impl<V: Pod, I: Pod> Align<V, I> for VertexBuffers<V, I> {
126    fn align_vertices(&mut self) {
127        let align = wgpu::COPY_BUFFER_ALIGNMENT;
128        let stride = std::mem::size_of::<V>() as wgpu::BufferAddress;
129        let unpadded_bytes = self.vertices.len() as wgpu::BufferAddress * stride;
130        let padding_bytes = (align - unpadded_bytes % align) % align;
131
132        if padding_bytes != 0 {
133            panic!(
134                "vertices are always aligned to wgpu::COPY_BUFFER_ALIGNMENT \
135                    because GpuVertexUniform is aligned"
136            )
137        }
138    }
139
140    fn align_indices(&mut self) {
141        let align = wgpu::COPY_BUFFER_ALIGNMENT;
142        let stride = std::mem::size_of::<I>() as wgpu::BufferAddress;
143        let unpadded_bytes = self.indices.len() as wgpu::BufferAddress * stride;
144        let padding_bytes = (align - unpadded_bytes % align) % align;
145        let overpad = padding_bytes.div_ceil(stride);
146
147        for _ in 0..overpad {
148            self.indices.push(I::zeroed());
149        }
150    }
