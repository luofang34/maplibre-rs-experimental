10/// Where one feature's geometry sits in the tessellator's buffers, and how it ranks.
11struct Span {
12    key: f32,
13    entries: Range<usize>,
14    indices: Range<usize>,
15}
16
17/// Collects the sort key of each feature as it is tessellated.
18#[derive(Default)]
19pub struct SortKeys {
20    property: Option<StyleProperty<f32>>,
21    spans: Vec<Span>,
22    indices_end: usize,
23}
24
25impl SortKeys {
26    /// Whether a property ranks the features, rather than none or a grouping alone.
27    pub fn is_keyed(&self) -> bool {
28        self.property.is_some()
29    }
30
31    /// Ranks features by `property`; a feature without a key ranks as zero.
32    pub fn by(property: Option<StyleProperty<f32>>) -> Self {
33        Self {
34            property,
35            ..Self::default()
36        }
37    }
38
39    pub(super) fn record(
40        &mut self,
41        properties: &FeatureProperties,
42        zoom: f64,
43        entries: Range<usize>,
44        (indices_end, grouped_by): (usize, Option<f32>),
45    ) {
46        let key = match (&self.property, grouped_by) {
47            (_, Some(group)) => group,
48            (Some(property), None) => property.evaluate_for(properties, zoom).unwrap_or(0.0),
49            (None, None) => return,
50        };
51        self.spans.push(Span {
52            key,
53            entries,
54            indices: self.indices_end..indices_end,
55        });
56        self.indices_end = indices_end;
57    }
58}

60struct Reordered<V, I> {
61    vertices: Vec<V>,
62    indices: Vec<I>,
63    entries: Vec<(u32, Option<[f32; 4]>)>,
64}
65
66impl<I> ZeroTessellator<I>
67where
68    I: std::ops::Add + From<VertexId> + MaxIndex + Copy + Into<u32>,
69{
70    /// The sort key of each entry of `feature_indices` when a property ranks the features;
71    /// empty otherwise.
72    pub fn sort_key_values(&mut self) -> Vec<f32> {
73        if self.sort_key.is_keyed() {
74            std::mem::take(&mut self.entry_sort_keys)
75        } else {
76            Vec::new()
77        }
78    }
79
80    /// Reorders the finished features so that greater sort keys draw later, keeping the order
81    /// of equal keys.
82    pub fn apply_sort_keys(&mut self) {
83        let mut spans = std::mem::take(&mut self.sort_key.spans);
84        self.entry_sort_keys = vec![0.0; self.feature_indices.len()];
85        if spans.windows(2).all(|pair| pair[0].key <= pair[1].key) {
86            for span in &spans {
87                for entry in span.entries.clone() {
88                    if let Some(key) = self.entry_sort_keys.get_mut(entry) {
89                        *key = span.key;
90                    }
91                }
92            }
93            return;
94        }
95        spans.sort_by(|a, b| a.key.total_cmp(&b.key));
96        self.entry_sort_keys = spans
97            .iter()
98            .flat_map(|span| span.entries.clone().map(move |_| span.key))
99            .collect();
100        let mut starts = Vec::with_capacity(self.feature_indices.len());
101        let mut start = 0_u32;
102        for count in &self.feature_indices {
103            starts.push(start);
104            start = start.wrapping_add(*count);
105        }
106        let mut out = Reordered {
107            vertices: Vec::with_capacity(self.buffer.vertices.len()),
108            indices: Vec::with_capacity(self.buffer.indices.len()),
109            entries: Vec::with_capacity(self.feature_indices.len()),
110        };
111        // Each entry's vertices land contiguously, so the index shift is one number per entry.
112        let mut shifts = vec![0_i64; self.feature_indices.len()];
113        for span in &spans {
114            for entry in span.entries.clone() {
115                let (first, count) = (starts[entry] as usize, self.feature_indices[entry] as usize);
116                shifts[entry] = out.vertices.len() as i64 - first as i64;
117                out.vertices
118                    .extend_from_slice(&self.buffer.vertices[first..first + count]);
119                out.entries
120                    .push((count as u32, self.feature_colors.get(entry).copied()));
121            }
122            let entries = span.entries.clone();
123            let ranges: Vec<(u32, u32, i64)> = entries
124                .map(|entry| {
125                    (
126                        starts[entry],
127                        starts[entry] + self.feature_indices[entry],
128                        shifts[entry],
129                    )
130                })
131                .collect();
132            for index in &self.buffer.indices[span.indices.clone()] {
133                let index: u32 = (*index).into();
134                let shift = ranges
135                    .iter()
136                    .find(|(from, to, _)| (*from..*to).contains(&index))
137                    .map_or(0, |(_, _, shift)| *shift);
138                out.indices
139                    .push(I::from(VertexId((i64::from(index) + shift) as u32)));
140            }
141        }
142        self.buffer = VertexBuffers {
143            vertices: out.vertices,
144            indices: out.indices,
145        };
146        self.feature_indices = out.entries.iter().map(|(count, _)| *count).collect();
147        if !self.feature_colors.is_empty() {
148            self.feature_colors = out
149                .entries
150                .iter()
151                .filter_map(|(_, colour)| *colour)
152                .collect();
153        }
154    }
155}

157/// The `*-sort-key` layout property of a fill, line or circle layer.
158pub fn sort_key_of(layer: &crate::style::layer::StyleLayer) -> Option<StyleProperty<f32>> {
159    let name = sort_key_name(&layer.type_)?;
160    layer
161        .unrecognized
162        .layout
163        .get(name)
164        .map(StyleProperty::parse)
165}
166
167/// The sort-key property a layer type accepts, if any.
168pub fn sort_key_name(layer_type: &str) -> Option<&'static str> {
169    match layer_type {
170        "fill" => Some("fill-sort-key"),
171        "line" => Some("line-sort-key"),
172        "circle" => Some("circle-sort-key"),
173        _ => None,
174    }
