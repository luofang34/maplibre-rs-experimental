1//! Circle layers: one screen-aligned quad per point, extruded by the shader.
2//!
3//! Each quad's vertices carry the feature's radius and stroke width in the normal slot; the
4//! shader recovers the corner direction from the vertex index, so both values can be data
5//! driven. Points outside the tile are skipped as GL JS does, since a neighbouring tile draws
6//! them.
7
8use lyon::tessellation::{geometry_builder::MaxIndex, VertexId};
9
10use super::ZeroTessellator;
11use crate::{
12    coords::EXTENT,
13    render::ShaderVertex,
14    style::{circle::CirclePaint, heatmap::HeatmapPaint, layer::StyleProperty},
15};
16
17/// Positions on the eight-by-eight corner grid of a subdivided circle, which is four by four
18/// vertices; GL JS uses these for circles and heatmap kernels that follow the globe.
19pub const CIRCLE_GRID_STEPS: [u32; 4] = [0, 2, 5, 7];
20
21/// Vertex order of a circle quad; corners follow the shader's `vertex_index % 4` decoding.
22pub const CIRCLE_QUAD_INDICES: [u32; 6] = [0, 1, 2, 0, 2, 3];
23
24/// What a circle layer needs to size each feature's quad.
25#[derive(Clone, Debug)]
26pub struct CircleOptions {
27    /// Radius in screen pixels, evaluated per feature.
28    pub radius: StyleProperty<f32>,
29    /// Stroke width in screen pixels, evaluated per feature.
30    pub stroke_width: StyleProperty<f32>,
31    /// What the radius is when the property has no value for a feature.
32    pub radius_default: f32,
33    /// What the stroke-width slot holds when the property has no value for a feature.
34    pub stroke_width_default: f32,
35    /// `circle-blur`, when it varies by feature; otherwise the layer's value serves.
36    pub blur: Option<StyleProperty<f32>>,
37    /// `circle-stroke-opacity`, when it varies by feature.
38    pub stroke_opacity: Option<StyleProperty<f32>>,
39    /// Zoom of the tile, at which zoom-driven properties are evaluated.
40    pub zoom: f64,
41    /// Whether each quad is a grid of vertices, so a circle lying on the globe follows its curve.
42    pub grid: bool,
43}
44
45/// The property when it needs each feature's attributes.
46fn per_feature(property: Option<&StyleProperty<f32>>) -> Option<StyleProperty<f32>> {
47    property
48        .filter(|property| !property.is_feature_constant())
49        .cloned()
50}
51
52impl CircleOptions {
53    /// Options for a circle paint at a tile zoom.
54    pub fn for_paint(paint: &CirclePaint, zoom: f64) -> Self {
55        Self {
56            radius: paint.radius(),
57            radius_default: CirclePaint::DEFAULT_RADIUS,
58            stroke_width: paint.stroke_width(),
59            stroke_width_default: 0.0,
60            blur: per_feature(paint.circle_blur.as_ref()),
61            stroke_opacity: per_feature(paint.circle_stroke_opacity.as_ref()),
62            zoom,
63            grid: false,
64        }
65    }
66
67    /// Options for a heatmap point: a quad whose weight travels in the slot circles use for
68    /// the stroke width. A radius that does not depend on the feature is one value for the
69    /// whole layer, taken from the layer's instance record, so the quad is one unit; a
70    /// per-feature radius is the quad's own size, and the layer's factor is then one. A
71    /// feature without a weight counts once.
72    pub fn for_heatmap(paint: &HeatmapPaint, zoom: f64) -> Self {
73        let per_feature = paint.radius_per_feature();
74        Self {
75            radius: match &paint.heatmap_radius {
76                Some(radius) if per_feature => radius.clone(),
77                _ => StyleProperty::Constant(1.0),
78            },
79            radius_default: if per_feature { 30.0 } else { 1.0 },
80            stroke_width: paint
81                .heatmap_weight
82                .clone()
83                .unwrap_or(StyleProperty::Constant(1.0)),
84            stroke_width_default: 1.0,
85            blur: None,
86            stroke_opacity: None,
87            zoom,
88            grid: false,
89        }
90    }
91}
92
93impl<I> ZeroTessellator<I>
94where
95    I: std::ops::Add + From<VertexId> + MaxIndex + Copy + Into<u32>,
96{
97    /// Turns every point of every feature into a circle quad instead of a path.
98    pub fn with_circles(mut self, options: CircleOptions) -> Self {
99        self.circle = Some(options);
100        self
101    }
102
103    pub(super) fn emit_circle(&mut self, x: f32, y: f32) {
104        let Some(options) = &self.circle else {
105            return;
106        };
107        let extent = EXTENT as f32;
108        if !(0.0..extent).contains(&x) || !(0.0..extent).contains(&y) {
109            return;
110        }
111        let radius = options
112            .radius
113            .evaluate_for(&self.feature_properties, options.zoom)
114            .unwrap_or(options.radius_default)
115            .max(0.0);
116        // A radius that varies by feature and by zoom is stored for the tile's zoom and the one
117        // above it, which the shader blends by the fractional zoom, as GL JS does.
118        let radius_next = if options.radius.is_feature_constant() {
119            radius
120        } else {
121            options
122                .radius
123                .evaluate_for(&self.feature_properties, options.zoom + 1.0)
124                .unwrap_or(options.radius_default)
125                .max(0.0)
126        };
127        let stroke_width = options
128            .stroke_width
129            .evaluate_for(&self.feature_properties, options.zoom)
130            .unwrap_or(options.stroke_width_default)
131            .max(0.0);
132
133        // Negative values tell the shader to keep the layer's own blur and stroke opacity.
134        let per_feature = |property: &Option<StyleProperty<f32>>| {
135            property
136                .as_ref()
137                .and_then(|property| property.evaluate_for(&self.feature_properties, options.zoom))
138                .map_or(-1.0, |value| value.max(0.0))
139        };
140        let (blur, stroke_opacity) = (
141            per_feature(&options.blur),
142            per_feature(&options.stroke_opacity),
143        );
144        let base = self.buffer.vertices.len() as u32;
145        let push = |code: f32, buffer: &mut Vec<ShaderVertex>| {
146            let mut vertex = ShaderVertex::new([x, y], [radius, stroke_width]);
147            vertex.distance = blur;
148            vertex.elevation = stroke_opacity;
149            vertex.edge_distance = radius_next;
150            vertex.corner_code = code;
151            buffer.push(vertex);
152        };
153        if options.grid {
154            for ey in CIRCLE_GRID_STEPS {
155                for ex in CIRCLE_GRID_STEPS {
156                    push(1.0 + (ex * 8 + ey) as f32, &mut self.buffer.vertices);
157                }
158            }
159            for row in 0..3_u32 {
160                for column in 0..3_u32 {
161                    let lower = base + row * 4 + column;
162                    let upper = lower + 4;
163                    for offset in [lower, lower + 1, upper + 1, lower, upper + 1, upper] {
164                        self.buffer.indices.push(I::from(VertexId(offset)));
165                    }
166                }
167            }
168        } else {
169            for _ in 0..4 {
170                push(0.0, &mut self.buffer.vertices);
171            }
172            for offset in CIRCLE_QUAD_INDICES {
173                self.buffer.indices.push(I::from(VertexId(base + offset)));
174            }
175        }
176    }
