1//! Circle layers: one screen-aligned quad per point, extruded by the shader.
2//!
3//! Each quad's vertices carry the feature's radius and stroke width in the normal slot; the
4//! shader recovers the corner direction from the vertex index, so both values can be data
5//! driven. Points outside the tile are skipped as GL JS does, since a neighbouring tile draws
6//! them.
7

18/// Positions on the eight-by-eight corner grid of a subdivided circle, which is four by four
19/// vertices; GL JS uses these for circles and heatmap kernels that follow the globe.
20pub const CIRCLE_GRID_STEPS: [u32; 4] = [0, 2, 5, 7];
21
22/// Vertex order of a circle quad; corners follow the shader's `vertex_index % 4` decoding.
23pub const CIRCLE_QUAD_INDICES: [u32; 6] = [0, 1, 2, 0, 2, 3];
24
25/// The fraction of a circle vertex's `corner_code` that carries a per-feature stroke opacity:
26/// zero keeps the layer's own, and an opacity `o` is `0.25 + 0.5 o`. The integer part stays the
27/// corner, and the elevation slot stays free for the ground the circle stands on.
28fn stroke_opacity_fraction(stroke_opacity: f32) -> f32 {
29    if stroke_opacity < 0.0 {
30        0.0
31    } else {
32        0.25 + 0.5 * stroke_opacity.min(1.0)
33    }
34}
35
36/// Two colours, straight RGBA in 0..=1, in the four lanes of one feature colour: each lane holds
37/// two 8-bit channels as `-(256 a + b) - 1`, which an `f32` keeps exactly, as GL JS packs
38/// colours. The vertex slots of a circle are all taken, so a stroke colour that varies by
39/// feature shares the lanes of the fill's. A plain colour is never negative, so the sign tells
40/// the shader which a vertex holds, whatever the layer's style says now: a tile built before a
41/// paint change keeps drawing as it was built until its replacement arrives.
42pub fn pack_circle_colors(fill: [f32; 4], stroke: [f32; 4]) -> [f32; 4] {
43    let byte = |channel: f32| (channel.clamp(0.0, 1.0) * 255.0).round();
44    let pair = |high: f32, low: f32| -(byte(high) * 256.0 + byte(low)) - 1.0;
45    [
46        pair(fill[0], fill[1]),
47        pair(fill[2], fill[3]),
48        pair(stroke[0], stroke[1]),
49        pair(stroke[2], stroke[3]),
50    ]
51}

53/// What a circle layer needs to size each feature's quad.
54#[derive(Clone, Debug)]
55pub struct CircleOptions {
56    /// Radius in screen pixels, evaluated per feature.
57    pub radius: StyleProperty<f32>,
58    /// Stroke width in screen pixels, evaluated per feature.
59    pub stroke_width: StyleProperty<f32>,
60    /// What the radius is when the property has no value for a feature.
61    pub radius_default: f32,
62    /// What the stroke-width slot holds when the property has no value for a feature.
63    pub stroke_width_default: f32,
64    /// `circle-blur`, when it varies by feature; otherwise the layer's value serves.
65    pub blur: Option<StyleProperty<f32>>,
66    /// `circle-stroke-opacity`, when it varies by feature.
67    pub stroke_opacity: Option<StyleProperty<f32>>,
68    /// `circle-stroke-color`, when it varies by feature; each feature's colour then carries
69    /// both its fill and its stroke, as `pack_circle_colors` lays them out.
70    pub stroke_color: Option<StyleProperty<Color>>,
71    /// Zoom of the tile, at which zoom-driven properties are evaluated.
72    pub zoom: f64,
73    /// Whether each quad is a grid of vertices, so a circle lying on the globe follows its curve.
74    pub grid: bool,
75}
76
77/// The property when it needs each feature's attributes.
78fn per_feature(property: Option<&StyleProperty<f32>>) -> Option<StyleProperty<f32>> {
79    property
80        .filter(|property| !property.is_feature_constant())
81        .cloned()
82}

84impl CircleOptions {
85    /// Options for a circle paint at a tile zoom.
86    pub fn for_paint(paint: &CirclePaint, zoom: f64) -> Self {
87        Self {
88            radius: paint.radius(),
89            radius_default: CirclePaint::DEFAULT_RADIUS,
90            stroke_width: paint.stroke_width(),
91            stroke_width_default: 0.0,
92            blur: per_feature(paint.circle_blur.as_ref()),
93            stroke_opacity: per_feature(paint.circle_stroke_opacity.as_ref()),
94            stroke_color: paint.stroke_color_per_feature().cloned(),
95            zoom,
96            grid: false,
97        }
98    }
99
100    /// Options for a heatmap point: a quad whose weight travels in the slot circles use for
101    /// the stroke width. A radius that does not depend on the feature is one value for the
102    /// whole layer, taken from the layer's instance record, so the quad is one unit; a
103    /// per-feature radius is the quad's own size, and the layer's factor is then one. A
104    /// feature without a weight counts once.
105    pub fn for_heatmap(paint: &HeatmapPaint, zoom: f64) -> Self {
106        let per_feature = paint.radius_per_feature();
107        Self {
108            radius: match &paint.heatmap_radius {
109                Some(radius) if per_feature => radius.clone(),
110                _ => StyleProperty::Constant(1.0),
111            },
112            radius_default: if per_feature { 30.0 } else { 1.0 },
113            stroke_width: paint
114                .heatmap_weight
115                .clone()
116                .unwrap_or(StyleProperty::Constant(1.0)),
117            stroke_width_default: 1.0,
118            blur: None,
119            stroke_opacity: None,
120            stroke_color: None,
121            zoom,
122            grid: false,
123        }
124    }
125}

127impl<I> ZeroTessellator<I>
128where
129    I: std::ops::Add + From<VertexId> + MaxIndex + Copy + Into<u32>,
130{
131    /// Turns every point of every feature into a circle quad instead of a path.
132    pub fn with_circles(mut self, options: CircleOptions) -> Self {
133        self.circle = Some(options);
134        self
135    }
136
137    /// The colour a finished feature's vertices carry: its fill, or its fill and stroke packed
138    /// together when the stroke colour varies by feature.
139    pub(super) fn circle_feature_color(&self, fill: [f32; 4]) -> [f32; 4] {
140        let Some(options) = &self.circle else {
141            return fill;
142        };
143        let Some(stroke) = &options.stroke_color else {
144            return fill;
145        };
146        let stroke = stroke
147            .evaluate_for(&self.feature_properties, options.zoom)
148            .map_or([0.0, 0.0, 0.0, 1.0], |color| color.to_array());
149        pack_circle_colors(fill, stroke)
150    }
151
152    pub(super) fn emit_circle(&mut self, x: f32, y: f32) {
153        let Some(options) = &self.circle else {
154            return;
155        };
156        let extent = EXTENT as f32;
157        if !(0.0..extent).contains(&x) || !(0.0..extent).contains(&y) {
158            return;
159        }
160        let radius = options
161            .radius
162            .evaluate_for(&self.feature_properties, options.zoom)
163            .unwrap_or(options.radius_default)
164            .max(0.0);
165        // A radius that varies by feature and by zoom is stored for the tile's zoom and the one
166        // above it, which the shader blends by the fractional zoom, as GL JS does.
167        let radius_next = if options.radius.is_feature_constant() {
168            radius
169        } else {
170            options
171                .radius
172                .evaluate_for(&self.feature_properties, options.zoom + 1.0)
173                .unwrap_or(options.radius_default)
174                .max(0.0)
175        };
176        let stroke_width = options
177            .stroke_width
178            .evaluate_for(&self.feature_properties, options.zoom)
179            .unwrap_or(options.stroke_width_default)
180            .max(0.0);
181
182        // Negative values tell the shader to keep the layer's own blur and stroke opacity.
183        let per_feature = |property: &Option<StyleProperty<f32>>| {
184            property
185                .as_ref()
186                .and_then(|property| property.evaluate_for(&self.feature_properties, options.zoom))
187                .map_or(-1.0, |value| value.max(0.0))
188        };
189        let (blur, stroke_opacity) = (
190            per_feature(&options.blur),
191            per_feature(&options.stroke_opacity),
192        );
193        let base = self.buffer.vertices.len() as u32;
194        let push = |code: f32, buffer: &mut Vec<ShaderVertex>| {
195            let mut vertex = ShaderVertex::new([x, y], [radius, stroke_width]);
196            vertex.distance = blur;
197            vertex.edge_distance = radius_next;
198            vertex.corner_code = code + stroke_opacity_fraction(stroke_opacity);
199            buffer.push(vertex);
200        };
201        if options.grid {
202            for ey in CIRCLE_GRID_STEPS {
203                for ex in CIRCLE_GRID_STEPS {
204                    push(1.0 + (ex * 8 + ey) as f32, &mut self.buffer.vertices);
205                }
206            }
207            for row in 0..3_u32 {
208                for column in 0..3_u32 {
209                    let lower = base + row * 4 + column;
210                    let upper = lower + 4;
211                    for offset in [lower, lower + 1, upper + 1, lower, upper + 1, upper] {
212                        self.buffer.indices.push(I::from(VertexId(offset)));
213                    }
214                }
215            }
216        } else {
217            for _ in 0..4 {
218                push(0.0, &mut self.buffer.vertices);
219            }
220            for offset in CIRCLE_QUAD_INDICES {
221                self.buffer.indices.push(I::from(VertexId(base + offset)));
222            }
223        }
224    }
225}
