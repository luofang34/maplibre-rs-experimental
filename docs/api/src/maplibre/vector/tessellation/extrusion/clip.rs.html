1//! Ring extraction, hole classification and tile-box clipping for extruded polygons.
2
3use lyon::path::{Event, Path};
4
5use crate::coords::EXTENT;
6
7pub(super) type Ring = Vec<[f32; 2]>;
8
9pub(super) fn rings(path: &Path) -> Vec<Ring> {
10    let mut rings = Vec::new();
11    let mut current = Ring::new();
12    for event in path.iter() {
13        match event {
14            Event::Begin { at } => current = vec![at.to_array()],
15            Event::Line { to, .. } => current.push(to.to_array()),
16            Event::End { .. } => rings.push(std::mem::take(&mut current)),
17            Event::Quadratic { to, .. } | Event::Cubic { to, .. } => current.push(to.to_array()),
18        }
19    }
20    rings
21}

23/// The sum of `(x2 - x1) * (y1 + y2)` over the ring's edges, as geojson-vt measures the winding.
24pub(super) fn trapezoid_area(ring: &Ring) -> f32 {
25    ring.iter()
26        .zip(ring.iter().cycle().skip(1))
27        .map(|(a, b)| (b[0] - a[0]) * (a[1] + b[1]))
28        .sum()
29}
30
31/// Twice the signed area in the raw coordinates; the sign tells which side the inside is on.
32pub(super) fn signed_area(ring: &Ring) -> f32 {
33    ring.iter()
34        .zip(ring.iter().cycle().skip(1))
35        .map(|(a, b)| a[0] * b[1] - b[0] * a[1])
36        .sum()
37}
38
39fn contains(ring: &Ring, [x, y]: [f32; 2]) -> bool {
40    let mut inside = false;
41    for (a, b) in ring.iter().zip(ring.iter().cycle().skip(1)) {
42        if (a[1] > y) != (b[1] > y) && x < (b[0] - a[0]) * (y - a[1]) / (b[1] - a[1]) + a[0] {
43            inside = !inside;
44        }
45    }
46    inside
47}

49/// A ring inside an odd number of others is a hole.
50pub(super) fn is_hole(index: usize, rings: &[Ring]) -> bool {
51    if rings.len() <= 1 {
52        return false;
53    }
54    let Some(&point) = rings[index].first() else {
55        return false;
56    };
57    rings
58        .iter()
59        .enumerate()
60        .filter(|(other, ring)| *other != index && contains(ring, point))
61        .count()
62        % 2
63        == 1
64}

66/// Walls along a tile's clipped edge would show a seam between tiles. When `include_tile_boundary`
67/// is true (polygons clipped tightly to `[0, EXTENT]` for translucent volumes), edges lying right
68/// on `0` or `EXTENT` are also omitted so adjacent tiles share no internal vertical curtain.
69pub(super) fn on_clipped_edge(a: [f32; 2], b: [f32; 2], include_tile_boundary: bool) -> bool {
70    let extent = EXTENT as f32;
71    if include_tile_boundary {
72        (a[0] == b[0] && (a[0] <= 0.0 || a[0] >= extent))
73            || (a[1] == b[1] && (a[1] <= 0.0 || a[1] >= extent))
74    } else {
75        (a[0] == b[0] && (a[0] < 0.0 || a[0] > extent))
76            || (a[1] == b[1] && (a[1] < 0.0 || a[1] > extent))
77    }
78}

80/// `ring` cut to the square from `low` to `high` on both axes, one edge of the square at a time.
81fn clip_ring(ring: &Ring, (low, high): (f32, f32)) -> Ring {
82    let mut clipped = ring.clone();
83    for axis in 0..2 {
84        for keep_above in [true, false] {
85            let bound = if keep_above { low } else { high };
86            let inside = |point: &[f32; 2]| {
87                if keep_above {
88                    point[axis] >= bound
89                } else {
90                    point[axis] <= bound
91                }
92            };
93            let mut next = Ring::new();
94            for (a, b) in clipped.iter().zip(clipped.iter().cycle().skip(1)) {
95                if inside(a) {
96                    next.push(*a);
97                }
98                if inside(a) != inside(b) {
99                    let t = (bound - a[axis]) / (b[axis] - a[axis]);
100                    let mut crossing = [a[0] + t * (b[0] - a[0]), a[1] + t * (b[1] - a[1])];
101                    crossing[axis] = bound;
102                    next.push(crossing);
103                }
104            }
105            clipped = next;
106        }
107    }
108    clipped
109}

111/// `path` with each ring clipped to `buffer` tile units past the tile and wound the way a
112/// GeoJSON source winds them, outer rings counterclockwise on the map and holes the other way;
113/// a ring left with no area is dropped. The wall distance of a pattern runs along that winding.
114pub(in crate::vector::tessellation) fn clipped(path: &Path, buffer: f32) -> Path {
115    let bounds = (-buffer, EXTENT as f32 + buffer);
116    let rings: Vec<Ring> = rings(path)
117        .iter()
118        .map(|ring| clip_ring(ring, bounds))
119        .filter(|ring| ring.len() >= 3 && signed_area(ring) != 0.0)
120        .collect();
121    build_wound_rings(&rings)
122}
123
124/// Normalizes ring winding without clipping so outer walls and hole walls wind consistently for
125/// back/front face culling.
126pub(super) fn normalized_rings(rings: &[Ring]) -> Vec<Ring> {
127    rings
128        .iter()
129        .enumerate()
130        .filter(|(_, ring)| ring.len() >= 3 && signed_area(ring) != 0.0)
131        .map(|(index, ring)| {
132            let mut ring = ring.clone();
133            if (trapezoid_area(&ring) > 0.0) != is_hole(index, rings) {
134                ring[1..].reverse();
135            }
136            ring
137        })
138        .collect()
139}
140
141fn build_wound_rings(rings: &[Ring]) -> Path {
142    let mut builder = Path::builder();
143    for ring in normalized_rings(rings) {
144        builder.begin(lyon::math::point(ring[0][0], ring[0][1]));
145        for point in &ring[1..] {
146            builder.line_to(lyon::math::point(point[0], point[1]));
147        }
148        builder.end(true);
149    }
150    builder.build()
151}
