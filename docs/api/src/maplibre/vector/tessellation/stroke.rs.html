21/// Internal property key storing the fractional start `[0.0, 1.0]` of a clipped `lineMetrics` segment.
22pub(crate) const LINE_CLIP_START_PROPERTY: &str = "\u{1}clip_start";
23/// Internal property key storing the fractional end `[0.0, 1.0]` of a clipped `lineMetrics` segment.
24pub(crate) const LINE_CLIP_END_PROPERTY: &str = "\u{1}clip_end";

26/// Total length of `path`'s segments in tile units.
27pub(super) fn path_length(path: &Path) -> f32 {
28    path.iter()
29        .map(|event| match event {
30            Event::Line { from, to } => (to - from).length(),
31            Event::End {
32                last,
33                first,
34                close: true,
35            } => (first - last).length(),
36            _ => 0.0,
37        })
38        .sum()
39}

41fn corner_miter_ratio(u_in: [f32; 2], u_out: [f32; 2]) -> f32 {
42    let dot = (u_in[0] * u_out[0] + u_in[1] * u_out[1]).clamp(-1.0, 1.0);
43    // `m = 1 / sin((pi - delta) / 2) = 1 / cos(delta / 2) = sqrt(2 / (1 + dot))`.
44    let denom = 1.0 + dot;
45    if denom <= 1e-6 {
46        f32::INFINITY
47    } else {
48        (2.0 / denom).sqrt()
49    }
50}

52/// Returns `true` when `path` has at least one interior join and every join's miter ratio
53/// (`1 / sin(half_angle)`) is at or below `round_limit`, allowing `LineJoin::Round` to collapse to
54/// shallow miter joins with `StrokeOptions::with_miter_limit(round_limit)`.
55pub(crate) fn joins_within_round_limit(path: &Path, round_limit: f32) -> bool {
56    if round_limit < 1.0 {
57        return false;
58    }
59    let mut prev_dir: Option<[f32; 2]> = None;
60    let mut first_dir: Option<[f32; 2]> = None;
61    let mut saw_join = false;
62    for event in path.iter() {
63        match event {
64            Event::Begin { .. } => {
65                prev_dir = None;
66                first_dir = None;
67            }
68            Event::Line { from, to } => {
69                let d = to - from;
70                let len = d.length();
71                if len <= 1e-5 {
72                    continue;
73                }
74                let dir = [d.x / len, d.y / len];
75                if let Some(prev) = prev_dir {
76                    saw_join = true;
77                    if corner_miter_ratio(prev, dir) > round_limit {
78                        return false;
79                    }
80                } else {
81                    first_dir = Some(dir);
82                }
83                prev_dir = Some(dir);
84            }
85            Event::End {
86                last,
87                first,
88                close: true,
89            } => {
90                let d = first - last;
91                let len = d.length();
92                if len > 1e-5 {
93                    let close_dir = [d.x / len, d.y / len];
94                    if let Some(prev) = prev_dir {
95                        saw_join = true;
96                        if corner_miter_ratio(prev, close_dir) > round_limit {
97                            return false;
98                        }
99                    }
100                    if let Some(first_d) = first_dir {
101                        saw_join = true;
102                        if corner_miter_ratio(close_dir, first_d) > round_limit {
103                            return false;
104                        }
105                    }
106                } else if let (Some(prev), Some(first_d)) = (prev_dir, first_dir) {
107                    saw_join = true;
108                    if corner_miter_ratio(prev, first_d) > round_limit {
109                        return false;
110                    }
111                }
112                prev_dir = None;
113                first_dir = None;
114            }
115            _ => {}
116        }
117    }
118    saw_join
119}

121fn clip_metrics(props: &FeatureProperties) -> Option<(f64, f64)> {
122    let start = match props.get(LINE_CLIP_START_PROPERTY)? {
123        Value::Number(n) => *n,
124        _ => return None,
125    };
126    let end = match props.get(LINE_CLIP_END_PROPERTY)? {
127        Value::Number(n) => *n,
128        _ => return None,
129    };
130    (start.is_finite() && end.is_finite() && start >= 0.0 && end > start && end <= 1.0)
131        .then_some((start, end))
132}

134impl<I> ZeroTessellator<I>
135where
136    I: std::ops::Add + From<VertexId> + MaxIndex + Copy + Into<u32>,
137{
138    pub(super) fn tessellate_strokes(&mut self) -> GeoResult<()> {
139        let mut path = self.path_builder.replace(Path::builder()).build();
140        if self.line_elevation.is_some() {
141            path = line_elevation::clip_elevated_path(&path);
142        }
143        if self.split_corners {
144            path = sharp_corners::split(&path, self.sharp_corner_offset);
145        }
146        let seg_len = path_length(&path);
147        let mut gradient_offset = 0.0_f32;
148        if self.line_gradient {
149            if let Some((clip_start, clip_end)) = clip_metrics(self.props()) {
150                if seg_len > 0.0 {
151                    let full_len = (f64::from(seg_len) / (clip_end - clip_start)) as f32;
152                    self.line_length = self.line_length.max(full_len);
153                    gradient_offset = (clip_start * f64::from(full_len)) as f32;
154                }
155            } else {
156                self.line_length = self.line_length.max(seg_len);
157            }
158        } else {
159            self.line_length = self.line_length.max(seg_len);
160        }
161
162        let origins = if self.line_gradient {
163            Vec::new()
164        } else {
165            line_origin::entry_distances(&path)
166        };
167        let (path, base_distances) = if clip::needs_guardband_clip(&path) {
168            let (clipped, bases) =
169                clip::clip_to_guardband(&path, &origins, self.line_gradient, gradient_offset);
170            if bases.is_empty() {
171                return Ok(());
172            }
173            (clipped, Some(bases))
174        } else {
175            (path, None)
176        };
177        let gradient_scale =
178            if self.line_gradient && self.line_length > clip::MAX_LINE_METRIC_LENGTH {
179                let scale = clip::MAX_LINE_METRIC_LENGTH / self.line_length;
180                self.line_length = clip::MAX_LINE_METRIC_LENGTH;
181                scale
182            } else {
183                1.0
184            };
185
186        let constructor = line_origin::StrokeOrigins {
187            origins: if base_distances.is_some() {
188                Vec::new()
189            } else {
190                origins
191            },
192            base_distances,
193            gradient_offset,
194            gradient_scale,
195            is_gradient: self.line_gradient,
196            packed: self
197                .line_feature_style
198                .as_ref()
199                .map_or(PackedLine::LAYER, |style| style.pack(self.props())),
200        };
201
202        let declared_join = self.feature_join();
203        let (effective_join, effective_miter_limit) = if declared_join == LineJoin::Round
204            && joins_within_round_limit(&path, self.stroke.round_limit)
205        {
206            (LineJoin::Miter, self.stroke.round_limit.max(1.0))
207        } else {
208            (declared_join, self.stroke.miter_limit)
209        };
210
211        let vertex_start = self.buffer.vertices.len();
212        let stroke_index_start = self.buffer.indices.len();
213        self.stroke_tessellator
214            .tessellate_path(
215                &path,
216                &StrokeOptions::tolerance(DEFAULT_TOLERANCE)
217                    .with_line_cap(match self.stroke.cap {
218                        LineCap::Butt => lyon::tessellation::LineCap::Butt,
219                        LineCap::Round => lyon::tessellation::LineCap::Round,
220                        LineCap::Square => lyon::tessellation::LineCap::Square,
221                    })
222                    .with_line_join(match effective_join {
223                        LineJoin::Miter => lyon::tessellation::LineJoin::Miter,
224                        LineJoin::Bevel => lyon::tessellation::LineJoin::Bevel,
225                        LineJoin::Round => lyon::tessellation::LineJoin::Round,
226                    })
227                    .with_miter_limit(effective_miter_limit),
228                &mut BuffersBuilder::new(&mut self.buffer, constructor),
229            )
230            .map_err(|error| GeozeroError::Geometry(error.to_string()))?;
231        let max_normal = match effective_join {
232            LineJoin::Miter => (effective_miter_limit * 2.0).max(2.0),
233            LineJoin::Bevel | LineJoin::Round => 2.0,
234        };
235        concave_join::split_unclipped_concave_joins(
236            &mut self.buffer,
237            vertex_start,
238            stroke_index_start,
239            max_normal,
240        );
241        if let Some(options) = &self.line_elevation {
242            let (metres, datum) = options.evaluate(self.props());
243            line_elevation::apply_elevation(
244                &mut self.buffer.vertices[vertex_start..],
245                metres,
246                datum,
247            );
248        }
249        Ok(())
250    }
251
252    fn feature_join(&self) -> LineJoin {
253        self.join_property
254            .as_ref()
255            .and_then(|property| property.evaluate_for(self.props(), self.zoom))
256            .and_then(|name| crate::style::line_stroke::LineStroke::join_named(&name))
257            .unwrap_or(self.stroke.join)
258    }

271    fn stroke_vertex_count(round_limit: f32) -> usize {
272        let layer: StyleLayer = serde_json::from_value(serde_json::json!({
273            "id": "roads",
274            "type": "line",
275            "source": "s",
276            "layout": {
277                "line-join": "round",
278                "line-round-limit": round_limit
279            }
280        }))
281        .expect("layer");
282        let mut tess = ZeroTessellator::<IndexDataType> {
283            is_line_layer: true,
284            ..Default::default()
285        };
286        tess.set_layer_strokes(&layer);
287        // 90-degree corner has miter ratio sqrt(2) ~= 1.4142.
288        tess.linestring_begin(true, 3, 0).expect("begin");
289        tess.xy(100.0, 100.0, 0).expect("p0");
290        tess.xy(300.0, 100.0, 1).expect("p1");
291        tess.xy(300.0, 300.0, 2).expect("p2");
292        tess.linestring_end(true, 0).expect("end");
293        tess.buffer.vertices.len()
294    }
295
296    #[test]
297    fn line_round_limit_converts_round_join_to_miter_when_threshold_exceeds_corner_ratio() {
298        let round_verts = stroke_vertex_count(1.05);
299        let miter_verts = stroke_vertex_count(1.6);
300        assert!(
301            miter_verts < round_verts,
302            "round_limit=1.6 should convert 90-deg round join (ratio ~1.414) to miter ({miter_verts} < {round_verts})"
303        );
304    }
