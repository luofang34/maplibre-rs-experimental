1//! Strokes split along a grid of tile cells: the globe's, so a line follows the curve of the
2//! sphere, and the terrain mesh's, so a line drawn over the terrain follows its faces.
3use geozero::error::{GeozeroError, Result as GeoResult};
4use lyon::geom;
5
6use super::{MaxIndex, ZeroTessellator};
7use crate::{
8    projection::globe::subdivision::subdivide_line_segment_on_grid,
9    style::{layer::StyleLayer, line_stroke::LineStroke},
10    terrain::mesh::TERRAIN_MESH_SIZE,
11};
12
13impl<I> ZeroTessellator<I>
14where
15    I: std::ops::Add + From<lyon::tessellation::VertexId> + MaxIndex + Copy + Into<u32>,
16{
17    /// Strokes features as `layer` lays them out: with its cap and join and, for a line drawn
18    /// over the terrain surface, split at every edge of the triangles of a terrain mesh over
19    /// this tile or any coarser one, so each straight piece lies on one planar face of the
20    /// surface the shader lifts it onto.
21    pub fn set_layer_strokes(&mut self, layer: &StyleLayer) {
22        self.stroke = LineStroke::of_layer(layer);
23        self.join_property = LineStroke::join_property(layer);
24        if crate::vector::structures::is_overlay(layer) {
25            self.subdivision_granularity = self.subdivision_granularity.max(TERRAIN_MESH_SIZE);
26            self.terrain_grid = true;
27        }
28    }
29
30    pub(super) fn append_coordinate(&mut self, coordinate: [f32; 2]) -> GeoResult<()> {
31        if let Some(previous) = self.contour_last {
32            let points = subdivide_line_segment_on_grid(
33                previous,
34                coordinate,
35                self.subdivision_granularity,
36                self.terrain_grid,
37            )
38            .map_err(|error| GeozeroError::Geometry(error.to_string()))?;
39            for point in points {
40                self.path_builder
41                    .borrow_mut()
42                    .line_to(geom::point(point[0], point[1]));
43            }
44        } else {
45            self.path_builder
46                .borrow_mut()
47                .begin(geom::point(coordinate[0], coordinate[1]));
48            self.contour_start = Some(coordinate);
49            self.path_open = true;
50        }
51        self.contour_last = Some(coordinate);
52        Ok(())
53    }
54
55    pub(super) fn subdivide_closing_segment(&mut self) -> GeoResult<()> {
56        let (Some(start), Some(last)) = (self.contour_start, self.contour_last) else {
57            return Ok(());
58        };
59        let mut points = subdivide_line_segment_on_grid(
60            last,
61            start,
62            self.subdivision_granularity,
63            self.terrain_grid,
64        )
65        .map_err(|error| GeozeroError::Geometry(error.to_string()))?;
66        points.pop();
67        for point in points {
68            self.path_builder
69                .borrow_mut()
70                .line_to(geom::point(point[0], point[1]));
71        }
72        Ok(())
73    }
74}
