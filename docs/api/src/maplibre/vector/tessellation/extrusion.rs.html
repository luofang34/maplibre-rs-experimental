1//! Walls and roofs of extruded polygons.
2//!
3//! A vertex carries its place in the fields the fill and line shaders use otherwise: the
4//! wall's lighting normal in `normal`, scaled to length two on the top edge so the shader can
5//! tell it from the bottom edge, the base in `distance`, and the roof height in `elevation`.
6//! A roof vertex has the zero normal.

22/// The paint properties that shape an extrusion.
23#[derive(Clone)]
24pub struct ExtrusionOptions {
25    /// Roof height in metres.
26    pub height: Option<StyleProperty<f32>>,
27    /// Wall base in metres.
28    pub base: Option<StyleProperty<f32>>,
29    /// Tile units past the tile's edge to which the polygons are clipped, for geometry that is
30    /// not clipped already; a vector tile's polygons are.
31    pub clip_buffer: Option<f32>,
32}

34impl ExtrusionOptions {

37        Self {
38            height: paint.fill_extrusion_height.clone(),
39            base: paint.fill_extrusion_base.clone(),
40            clip_buffer: None,
41        }
42    }

59/// The longest distance along a ring in tile units before it starts over, as GL JS stores it.
60const MAX_EDGE_DISTANCE: f32 = 32768.0 / 2.0;
61
62type Ring = Vec<[f32; 2]>;
63
64fn rings(path: &Path) -> Vec<Ring> {
65    let mut rings = Vec::new();
66    let mut current = Ring::new();
67    for event in path.iter() {
68        match event {
69            Event::Begin { at } => current = vec![at.to_array()],
70            Event::Line { to, .. } => current.push(to.to_array()),
71            Event::End { .. } => rings.push(std::mem::take(&mut current)),
72            Event::Quadratic { to, .. } | Event::Cubic { to, .. } => current.push(to.to_array()),
73        }
74    }
75    rings
76}

78/// The sum of `(x2 - x1) * (y1 + y2)` over the ring's edges, as geojson-vt measures the winding.
79fn trapezoid_area(ring: &Ring) -> f32 {
80    ring.iter()
81        .zip(ring.iter().cycle().skip(1))
82        .map(|(a, b)| (b[0] - a[0]) * (a[1] + b[1]))
83        .sum()
84}
85
86/// Twice the signed area in the raw coordinates; the sign tells which side the inside is on.
87fn signed_area(ring: &Ring) -> f32 {
88    ring.iter()
89        .zip(ring.iter().cycle().skip(1))
90        .map(|(a, b)| a[0] * b[1] - b[0] * a[1])
91        .sum()
92}
93
94fn contains(ring: &Ring, [x, y]: [f32; 2]) -> bool {
95    let mut inside = false;
96    for (a, b) in ring.iter().zip(ring.iter().cycle().skip(1)) {
97        if (a[1] > y) != (b[1] > y) && x < (b[0] - a[0]) * (y - a[1]) / (b[1] - a[1]) + a[0] {
98            inside = !inside;
99        }
100    }
101    inside
102}

104/// A ring inside an odd number of others is a hole.
105fn is_hole(index: usize, rings: &[Ring]) -> bool {
106    let Some(&point) = rings[index].first() else {
107        return false;
108    };
109    rings
110        .iter()
111        .enumerate()
112        .filter(|(other, ring)| *other != index && contains(ring, point))
113        .count()
114        % 2
115        == 1
116}

118/// Walls along a tile's clipped edge would show a seam between tiles.
119fn on_clipped_edge(a: [f32; 2], b: [f32; 2]) -> bool {
120    let extent = EXTENT as f32;
121    (a[0] == b[0] && (a[0] < 0.0 || a[0] > extent))
122        || (a[1] == b[1] && (a[1] < 0.0 || a[1] > extent))
123}
124
125/// `ring` cut to the square from `low` to `high` on both axes, one edge of the square at a time.
126fn clip_ring(ring: &Ring, (low, high): (f32, f32)) -> Ring {
127    let mut clipped = ring.clone();
128    for axis in 0..2 {
129        for keep_above in [true, false] {
130            let bound = if keep_above { low } else { high };
131            let inside = |point: &[f32; 2]| {
132                if keep_above {
133                    point[axis] >= bound
134                } else {
135                    point[axis] <= bound
136                }
137            };
138            let mut next = Ring::new();
139            for (a, b) in clipped.iter().zip(clipped.iter().cycle().skip(1)) {
140                if inside(a) {
141                    next.push(*a);
142                }
143                if inside(a) != inside(b) {
144                    let t = (bound - a[axis]) / (b[axis] - a[axis]);
145                    let mut crossing = [a[0] + t * (b[0] - a[0]), a[1] + t * (b[1] - a[1])];
146                    crossing[axis] = bound;
147                    next.push(crossing);
148                }
149            }
150            clipped = next;
151        }
152    }
153    clipped
154}

156/// `path` with each ring clipped to `buffer` tile units past the tile and wound the way a
157/// GeoJSON source winds them, outer rings counterclockwise on the map and holes the other way;
158/// a ring left with no area is dropped. The wall distance of a pattern runs along that winding.
159pub(super) fn clipped(path: &Path, buffer: f32) -> Path {
160    let bounds = (-buffer, EXTENT as f32 + buffer);
161    let rings: Vec<Ring> = rings(path)
162        .iter()
163        .map(|ring| clip_ring(ring, bounds))
164        .filter(|ring| ring.len() >= 3 && signed_area(ring) != 0.0)
165        .collect();
166    let mut builder = Path::builder();
167    for (index, ring) in rings.iter().enumerate() {
168        let mut ring = ring.clone();
169        // An outer ring keeps its order when its area is not positive, a hole when it is, as
170        // geojson-vt winds them.
171        if (trapezoid_area(&ring) > 0.0) != is_hole(index, &rings) {
172            ring[1..].reverse();
173        }
174        builder.begin(lyon::math::point(ring[0][0], ring[0][1]));
175        for point in &ring[1..] {
176            builder.line_to(lyon::math::point(point[0], point[1]));
177        }
178        builder.end(true);
179    }
180    builder.build()
181}

183fn push_wall<I: From<VertexId>>(
184    buffer: &mut VertexBuffers<ShaderVertex, I>,
185    [a, b]: [[f32; 2]; 2],
186    normal: [f32; 2],
187    (base, height): (f32, f32),
188    edge_distances: [f32; 2],
189) {
190    let first = buffer.vertices.len() as u32;
191    let index = |offset: u32| I::from(VertexId(first + offset));
192    for (point, top, distance) in [
193        (a, false, edge_distances[0]),
194        (a, true, edge_distances[0]),
195        (b, true, edge_distances[1]),
196        (b, false, edge_distances[1]),
197    ] {
198        let scale = if top { 2.0 } else { 1.0 };
199        let mut vertex = ShaderVertex::new(point, [normal[0] * scale, normal[1] * scale]);
200        vertex.distance = base;
201        vertex.elevation = height;
202        vertex.edge_distance = distance;
203        buffer.vertices.push(vertex);
204    }
205    buffer.indices.extend([0, 1, 2, 0, 2, 3].map(index));
206}

208/// Appends the roof and walls of the polygons in `path` to `buffer`.
209///
210/// `refine_roof` may split the roof's triangles from the given index on, as a globe tile does so
211/// the roof follows the sphere; the vertices it adds take the roof's base and height.
212pub(super) fn extrude<I>(
213    path: &Path,
214    heights: (f32, f32),
215    buffer: &mut VertexBuffers<ShaderVertex, I>,
216    (tolerance, refine_roof): (
217        f32,
218        impl FnOnce(&mut VertexBuffers<ShaderVertex, I>, usize) -> Result<(), String>,
219    ),
220) -> Result<(), String>
221where
222    I: From<VertexId> + std::ops::Add + MaxIndex,
223{
224    let (base, height) = heights;
225    let roof_start = buffer.indices.len();
226    let vertex_start = buffer.vertices.len();
227    FillTessellator::new()
228        .tessellate_path(
229            path,
230            &FillOptions::tolerance(tolerance).with_fill_rule(FillRule::EvenOdd),
231            &mut BuffersBuilder::new(buffer, RoofVertex { base, height }),
232        )
233        .map_err(|error| error.to_string())?;
234    refine_roof(buffer, roof_start)?;
235    for vertex in &mut buffer.vertices[vertex_start..] {
236        vertex.distance = base;
237        vertex.elevation = height;
238    }
239    let rings = rings(path);
240    for (index, ring) in rings.iter().enumerate() {
241        let area = signed_area(ring);
242        if area == 0.0 {
243            continue;
244        }
245        // The inside of the ring lies on the side `area` points to. GL JS lights a wall by the
246        // normal pointing into the extrusion, so the light meets the walls it shines on the
247        // other way round: the wall of an outer ring takes the inside direction and the wall
248        // of a hole the direction into the ring's outside.
249        let facing = if is_hole(index, &rings) { area } else { -area }.signum();
250        let mut travelled = 0.0;
251        for (a, b) in ring.iter().zip(ring.iter().cycle().skip(1)) {
252            let (dx, dy) = (b[0] - a[0], b[1] - a[1]);
253            let length = dx.hypot(dy);
254            // The distance along the ring counts only the walls drawn, and starts over before it
255            // outgrows what GL JS stores.
256            if on_clipped_edge(*a, *b) {
257                continue;
258            }
259            if travelled + length > MAX_EDGE_DISTANCE {
260                travelled = 0.0;
261            }
262            let start = travelled;
263            travelled += length;
264            if length == 0.0 {
265                continue;
266            }
267            let normal = [facing * dy / length, -facing * dx / length];
268            // GL JS gives the end of an edge the distance before it and the start the distance
269            // after it.
270            push_wall(buffer, [*a, *b], normal, heights, [travelled, start]);
271        }
272    }
273    Ok(())
274}
