44pub fn upload_system(
45    MapContext {
46        world,
47        style,
48        view_state,
49        renderer: Renderer { queue, .. },
50        ..
51    }: &mut MapContext,
52) -> SystemResult {
53    if crate::render::eye_covering::EyeInFrame::reuses_content(world) {
54        return Ok(());
55    }
56    let (paint_frame, previous_paint) = frame_paint(world, style, view_state);
57    let refresh_paint = previous_paint != Some(paint_frame);
58    let Some(Initialized(pattern)) = world.resources.get::<Eventually<WgpuTileViewPattern>>()
59    else {
60        return Err(SystemError::Dependencies);
61    };
62    let source_tiles = if style.terrain.is_some() {
63        sources::terrain_sources(world, style)
64    } else {
65        sources_for_upload(pattern)
66    };
67    let painted_tiles = painted_tiles(pattern, &source_tiles);
68    let (spatial, changed) = super::structures::for_frame(world, style, &painted_tiles);
69    let Some(Initialized(buffer_pool)) = world.resources.get_mut::<Eventually<VectorBufferPool>>()
70    else {
71        return Err(SystemError::Dependencies);
72    };
73    if refresh_paint {
74        refresh_layer_paint(
75            buffer_pool,
76            queue,
77            &painted_tiles,
78            paint_frame,
79            previous_paint,
80        );
81    }
82    if changed {
83        super::structures::refresh_gpu(buffer_pool, queue, &spatial, &world.tiles);
84    }
85    upload_tessellated_layer(
86        buffer_pool,
87        queue,
88        &mut world.tiles,
89        style,
90        source_tiles,
91        &spatial,
92        paint_frame,
93    );
94    Ok(())
95}

97fn frame_paint(
98    world: &mut crate::tcs::world::World,
99    style: &Style,
100    view_state: &crate::render::view_state::ViewState,
101) -> (VectorPaintFrame, Option<VectorPaintFrame>) {
102    let zoom = if style.terrain.is_some() {
103        paint::stabilize_zoom(world, view_state.style_zoom().value()) as f32
104    } else {
105        view_state.style_zoom().level()
106    };
107    let bearing = view_state.camera().get_bearing().0 as f32;
108    let light = style
109        .light
110        .clone()
111        .unwrap_or_default()
112        .extrusion_light(view_state.style_zoom().value(), f64::from(bearing))
113        .unwrap_or_default();
114    let paint_frame = VectorPaintFrame {
115        zoom,
116        bearing,
117        light,
118    };
119    let previous_paint = world.resources.get::<VectorPaintFrame>().copied();
120    world.resources.insert(paint_frame);
121    (paint_frame, previous_paint)
122}

124fn painted_tiles(
125    pattern: &WgpuTileViewPattern,
126    source_tiles: &[crate::coords::WorldTileCoords],
127) -> Vec<crate::coords::WorldTileCoords> {
128    let mut painted_tiles = source_tiles.to_vec();
129    let mut seen: HashSet<_> = painted_tiles.iter().copied().collect();
130    for tile in pattern.iter() {
131        tile.render_kind(TileKind::Vector, |shape| {
132            if seen.insert(shape.coords()) {
133                painted_tiles.push(shape.coords());
134            }
135        });
136    }
137    painted_tiles
138}

140fn sources_for_upload(
141    tile_view_pattern: &WgpuTileViewPattern,
142) -> Vec<crate::coords::WorldTileCoords> {
143    let mut source_tiles = Vec::new();
144    let mut seen = HashSet::new();
145    for view_tile in tile_view_pattern.iter() {
146        // The view tile itself is uploaded whether or not the frame draws it yet: a tile
147        // counts as available only once it is in the pool, so the frame draws a stand-in
148        // until this upload has happened.
149        if seen.insert(view_tile.coords()) {
150            source_tiles.push(view_tile.coords());
151        }
152        view_tile.render_kind(TileKind::Vector, |shape| {
153            if seen.insert(shape.coords()) {
154                source_tiles.push(shape.coords());
155            }
156        });
157    }
158    source_tiles
159}

161fn refresh_layer_paint(
162    buffer_pool: &VectorBufferPool,
163    queue: &crate::render::upload_queue::UploadQueue,
164    source_tiles: &[crate::coords::WorldTileCoords],
165    frame: VectorPaintFrame,
166    previous: Option<VectorPaintFrame>,
167) {
168    let VectorPaintFrame { zoom, .. } = frame;
169    for coords in source_tiles {
170        for entry in buffer_pool
171            .index()
172            .get_layers(*coords)
173            .into_iter()
174            .flatten()
175        {
176            let metadata = metadata_for_layer(&entry.style_layer, *coords, frame);
177            let unchanged = previous.is_some_and(|previous| {
178                bytemuck::bytes_of(&metadata_for_layer(&entry.style_layer, *coords, previous))
179                    == bytemuck::bytes_of(&metadata)
180            });
181            if !unchanged {
182                buffer_pool.update_layer_metadata(queue, entry, metadata);
183            }
184            if let Some(color) = paint::uniform_color(&entry.style_layer, f64::from(zoom)) {
185                if previous
186                    .map(|frame| frame.zoom)
187                    .and_then(|zoom| paint::uniform_color(&entry.style_layer, f64::from(zoom)))
188                    == Some(color)
189                {
190                    continue;
191                }
192                let count = (entry.feature_metadata_buffer_range().end
193                    - entry.feature_metadata_buffer_range().start)
194                    as usize
195                    / std::mem::size_of::<FillShaderFeatureMetadata>();
196                let colors = vec![FillShaderFeatureMetadata { color }; count];
197                buffer_pool.update_feature_metadata(queue, entry, &colors);
198            }
199        }
200    }
201}

203fn upload_tessellated_layer(
204    buffer_pool: &mut VectorBufferPool,
205    queue: &crate::render::upload_queue::UploadQueue,
206    tiles: &mut Tiles,
207    style: &Style,
208    source_tiles: Vec<crate::coords::WorldTileCoords>,
209    spatial: &[super::structures::SpatialBuffer],
210    paint: VectorPaintFrame,
211) {
212    let mut uploaded_tiles = 0;
213    let mut bytes = crate::render::memory_budget::UploadBudget::new(16 << 20);
214    for coords in source_tiles {
215        if uploaded_tiles >= UPLOADS_PER_FRAME {
216            break;
217        }
218        super::content::ensure(tiles, coords);
219        let Some((layers, pending)) = tiles.query_mut::<(
220            &mut VectorLayerBucketComponent,
221            &mut super::content::LayerReplacements,
222        )>(coords) else {
223            continue;
224        };
225        let mut uploaded = false;
226        for style_layer in &style.layers {
227            let replacement = pending
228                .vector
229                .iter()
230                .position(|layer| layer.style_layer_id == style_layer.id);
231            let bucket = replacement.map(|index| &pending.vector[index]).or_else(|| {
232                layers.layers.iter().find_map(|layer| match layer {
233                    VectorLayerBucket::AvailableLayer(layer)
234                        if layer.style_layer_id == style_layer.id =>
235                    {
236                        Some(layer)
237                    }
238                    _ => None,
239                })
240            });
241            let Some(bucket) = bucket else {
242                continue;
243            };
244            let loaded = buffer_pool
245                .get_loaded_style_layers_at(coords)
246                .is_some_and(|layers| layers.contains(style_layer.id.as_str()));
247            if replacement.is_none() && loaded {
248                continue;
249            }
250            if bucket.buffer.buffer.indices.is_empty() {
251                buffer_pool.remove_layer(coords, &style_layer.id);
252            } else {
253                if !bytes.take(upload_size(bucket)) {
254                    return;
255                }
256                if !upload_bucket(buffer_pool, queue, style_layer, bucket, spatial, paint) {
257                    continue;
258                }
259                uploaded = true;
260            }
261            if let Some(index) = replacement {
262                super::content::commit_vector(layers, pending.vector.remove(index));
263            }
264        }
265        if uploaded {
266            uploaded_tiles += 1;
267        }
268    }
269}

271fn upload_size(bucket: &AvailableVectorLayerBucket) -> usize {
272    bucket.buffer.buffer.vertices.len() * size_of::<crate::render::ShaderVertex>()
273        + bucket.buffer.buffer.indices.len() * size_of::<u32>()
274        + bucket
275            .feature_indices
276            .iter()
277            .map(|count| *count as usize)
278            .sum::<usize>()
279            * size_of::<FillShaderFeatureMetadata>()
280}

282fn upload_bucket(
283    buffer_pool: &mut VectorBufferPool,
284    queue: &crate::render::upload_queue::UploadQueue,
285    style_layer: &crate::style::layer::StyleLayer,
286    bucket: &AvailableVectorLayerBucket,
287    spatial: &[super::structures::SpatialBuffer],
288    paint: VectorPaintFrame,
289) -> bool {
290    let VectorPaintFrame { zoom, .. } = paint;
291    let AvailableVectorLayerBucket {
292        coords,
293        buffer,
294        feature_indices,
295        feature_colors,
296        feature_sort_keys,
297        ..
298    } = bucket;
299    let coords = *coords;
300    let color: Option<Vec4f32> = style_layer
301        .paint
302        .as_ref()
303        .and_then(|paint| paint.get_color())
304        .map(|color| color.into());
305
306    // Assign every feature in the layer the color from the style if no parsed feature_color exist.
307    let fallback_color = color.unwrap_or([0.0, 0.0, 0.0, 1.0]);
308
309    let feature_metadata = if let Some(color) = paint::uniform_color(style_layer, f64::from(zoom)) {
310        feature_metadata(feature_indices, &[], color)
311    } else {
312        feature_metadata(feature_indices, feature_colors, fallback_color)
313    };
314
315    let buffer = spatial
316        .iter()
317        .find(|(tile, id, _)| *tile == coords && *id == style_layer.id)
318        .map_or(buffer, |(_, _, spatial)| spatial);
319    let layer_metadata = metadata_for_layer(style_layer, coords, paint);
320    let runs = style_layer
321        .paint
322        .as_ref()
323        .and_then(super::pattern::per_feature_pattern)
324        .map(|_| pattern_runs(feature_indices, feature_colors, &buffer.buffer.indices));
325
326    let sort_runs = (style_layer.type_ == "circle" && !feature_sort_keys.is_empty())
327        .then(|| sort_runs(feature_indices, feature_sort_keys, &buffer.buffer.indices));
328
329    tracing::debug!(%coords, "allocating vector geometry");
330    if let Err(error) = buffer_pool.replace_layer_geometry(
331        queue,
332        coords,
333        style_layer.clone(),
334        buffer,
335        layer_metadata,
336        &feature_metadata,
337    ) {
338        tracing::error!(%coords, %error, "tile geometry upload failed");
339        return false;
340    }
341    buffer_pool.set_pattern_runs(coords, &style_layer.id, runs);
342    buffer_pool.set_sort_runs(coords, &style_layer.id, sort_runs);
343    true
344}

346/// The index range of each run of consecutive features that share a sort key.
347fn sort_runs(
348    feature_indices: &[u32],
349    feature_sort_keys: &[f32],
350    indices: &[u32],
351) -> Vec<(f32, std::ops::Range<u32>)> {
352    let mut runs: Vec<(f32, std::ops::Range<u32>)> = Vec::new();
353    let (mut vertex_end, mut position) = (0_u32, 0_u32);
354    for (feature, count) in feature_indices.iter().enumerate() {
355        vertex_end = vertex_end.wrapping_add(*count);
356        let first = position;
357        while indices
358            .get(position as usize)
359            .is_some_and(|index| *index < vertex_end)
360        {
361            position = position.wrapping_add(1);
362        }
363        if position == first {
364            continue;
365        }
366        let key = feature_sort_keys.get(feature).copied().unwrap_or(0.0);
367        match runs.last_mut() {
368            Some((last, range)) if *last == key && range.end == first => range.end = position,
369            _ => runs.push((key, first..position)),
370        }
371    }
372    runs
373}

375/// The index range of each run of consecutive features that share a pattern key.
376fn pattern_runs(
377    feature_indices: &[u32],
378    feature_colors: &[[f32; 4]],
379    indices: &[u32],
380) -> Vec<(u32, std::ops::Range<u32>)> {
381    let mut runs: Vec<(u32, std::ops::Range<u32>)> = Vec::new();
382    let (mut vertex_end, mut position) = (0_u32, 0_u32);
383    for (feature, count) in feature_indices.iter().enumerate() {
384        vertex_end = vertex_end.wrapping_add(*count);
385        let first = position;
386        while indices
387            .get(position as usize)
388            .is_some_and(|index| *index < vertex_end)
389        {
390            position = position.wrapping_add(1);
391        }
392        let key = feature_colors
393            .get(feature)
394            .and_then(|color| crate::style::pattern_key::key_of_value(color[0]));
395        let Some(key) = key.filter(|_| position > first) else {
396            continue;
397        };
398        match runs.last_mut() {
399            Some((last, range)) if *last == key && range.end == first => range.end = position,
400            _ => runs.push((key, first..position)),
401        }
402    }
403    runs
404}

406fn layer_translate_tile_units(
407    paint: Option<&LayerPaint>,
408    tile_zoom: crate::coords::ZoomLevel,
409    view_zoom: f32,
410    bearing: f32,
411) -> [f32; 2] {
412    let translated = crate::style::translation::layer_translate_pixels(
413        paint,
414        f64::from(view_zoom),
415        f64::from(bearing),
416    )
417    .map(|pixels| pixels as f32);
418    let pixels_to_tile_units = 8.0 * 2.0_f32.powf(f32::from(u8::from(tile_zoom)) - view_zoom);
419    [
420        translated[0] * pixels_to_tile_units,
421        translated[1] * pixels_to_tile_units,
422    ]
423}

428fn metadata_for_layer(
429    style_layer: &crate::style::layer::StyleLayer,
430    coords: crate::coords::WorldTileCoords,
431    frame: VectorPaintFrame,
432) -> ShaderLayerMetadata {
433    let VectorPaintFrame {
434        zoom,
435        bearing,
436        light,
437    } = frame;
438    // Extract line-width from style paint (default 1.0px)
439    let line_width = match &style_layer.paint {
440        Some(LayerPaint::Line(paint)) => paint
441            .line_width
442            .as_ref()
443            .and_then(|w| w.evaluate_at_zoom(f64::from(zoom)))
444            .unwrap_or(1.0),
445        _ => 1.0,
446    };
447    let translate = layer_translate_tile_units(style_layer.paint.as_ref(), coords.z, zoom, bearing);
448    let mut layer_metadata =
449        ShaderLayerMetadata::new(style_layer.index as f32, line_width, translate);
450    if let Some(LayerPaint::Heatmap(paint)) = &style_layer.paint {
451        let zoom = f64::from(zoom);
452        // Radius and intensity are one value per layer, taken at the view zoom.
453        layer_metadata.line_width = if paint.radius_per_feature() {
454            1.0
455        } else {
456            paint.radius_at(zoom)
457        };
458        layer_metadata.circle_params = [paint.intensity_at(zoom), 1.0, 0.0, 0.0];
459    }
460    if let Some(LayerPaint::Line(paint)) = &style_layer.paint {
461        let value = |property: &Option<crate::style::layer::StyleProperty<f32>>| {
462            property
463                .as_ref()
464                .and_then(|property| property.evaluate_at_zoom(f64::from(zoom)))
465                .unwrap_or(0.0)
466        };
467        layer_metadata.circle_params = [
468            value(&paint.line_offset),
469            value(&paint.line_gap_width).max(0.0),
470            value(&paint.line_blur).max(0.0),
471            f32::from(paint.line_gradient.is_some()),
472        ];
473    }
474    if let Some(LayerPaint::Circle(paint)) = &style_layer.paint {
475        let zoom = f64::from(zoom);
476        layer_metadata.stroke_color = paint.stroke_color_rgba();
477        // Fill opacity is already folded into each feature's colour alpha.
478        layer_metadata.circle_params =
479            [1.0, paint.stroke_opacity_at(zoom), paint.blur_at(zoom), 0.0];
480        layer_metadata.circle_flags = [
481            f32::from(paint.circle_pitch_scale == CirclePitchScale::Map),
482            f32::from(paint.circle_pitch_alignment == CirclePitchAlignment::Map),
483            0.0,
484            0.0,
485        ];
486    }
487
488    if let Some(LayerPaint::FillExtrusion(paint)) = &style_layer.paint {
489        let opacity = paint
490            .fill_extrusion_opacity
491            .as_ref()
492            .and_then(|opacity| opacity.evaluate_at_zoom(f64::from(zoom)))
493            .unwrap_or(1.0)
494            .clamp(0.0, 1.0);
495        let gradient = paint.fill_extrusion_vertical_gradient.unwrap_or(true);
496        layer_metadata.stroke_color = [light.color[0], light.color[1], light.color[2], 1.0];
497        layer_metadata.circle_params = [opacity, f32::from(gradient), light.intensity, 0.0];
498        layer_metadata.circle_flags =
499            [light.position[0], light.position[1], light.position[2], 0.0];
500    }
501
502    layer_metadata
503}

505fn feature_metadata(
506    feature_indices: &[u32],
507    feature_colors: &[[f32; 4]],
508    fallback_color: [f32; 4],
509) -> Vec<FillShaderFeatureMetadata> {
510    let mut feature_metadata = Vec::with_capacity(feature_indices.iter().sum::<u32>() as usize);
511    for (idx, &count) in feature_indices.iter().enumerate() {
512        let current_color = feature_colors.get(idx).copied().unwrap_or(fallback_color);
513        for _ in 0..count {
514            feature_metadata.push(FillShaderFeatureMetadata {
515                color: current_color,
516            });
517        }
518    }
519
520    feature_metadata
521}
