39pub fn upload_system(
40    MapContext {
41        world,
42        style,
43        view_state,
44        renderer: Renderer {
45            queue, settings, ..
46        },
47        ..
48    }: &mut MapContext,
49) -> SystemResult {
50    let Some(pool_state) = world.resources.get::<Eventually<VectorBufferPool>>() else {
51        return Ok(());
52    };
53    if !matches!(pool_state, Initialized(_)) {
54        return Err(SystemError::Dependencies);
55    }
56    if settings.buffer_pools.is_empty()
57        || crate::render::eye_covering::EyeInFrame::reuses_content(world)
58    {
59        return Ok(());
60    }
61    let (paint_frame, previous_paint) = frame_paint(world, style, view_state);
62    let refresh_paint = previous_paint != Some(paint_frame);
63    let (source_tiles, painted_tiles) =
64        sources::frame_sources_and_painted(world, style, view_state)?;
65    let (spatial, changed) = super::structures::for_frame(world, style, &painted_tiles);
66    let Some(Initialized(buffer_pool)) = world.resources.get_mut::<Eventually<VectorBufferPool>>()
67    else {
68        return Err(SystemError::Dependencies);
69    };
70    if refresh_paint {
71        refresh_layer_paint(
72            buffer_pool,
73            queue,
74            &painted_tiles,
75            paint_frame,
76            previous_paint,
77        );
78    }
79    if changed {
80        super::structures::refresh_gpu(buffer_pool, queue, &spatial, &world.tiles);
81    }
82    let deferred = upload_tessellated_layer(
83        buffer_pool,
84        queue,
85        &mut world.tiles,
86        style,
87        source_tiles,
88        &spatial,
89        paint_frame,
90    );
91    if deferred {
92        crate::render::frame_signals::keep_animating(world);
93    }
94    Ok(())
95}

97fn frame_paint(
98    world: &mut crate::tcs::world::World,
99    style: &Style,
100    view_state: &crate::render::view_state::ViewState,
101) -> (VectorPaintFrame, Option<VectorPaintFrame>) {
102    let zoom = if style.terrain.is_some() {
103        paint::stabilize_zoom(world, view_state.style_zoom().value()) as f32
104    } else {
105        view_state.style_zoom().level()
106    };
107    let bearing = view_state.camera().get_bearing().0 as f32;
108    let light = style
109        .light
110        .clone()
111        .unwrap_or_default()
112        .extrusion_light(view_state.style_zoom().value(), f64::from(bearing))
113        .unwrap_or_default();
114    let paint_frame = VectorPaintFrame {
115        zoom,
116        bearing,
117        light,
118    };
119    let previous_paint = world.resources.get::<VectorPaintFrame>().copied();
120    world.resources.insert(paint_frame);
121    (paint_frame, previous_paint)
122}

124fn refresh_layer_paint(
125    buffer_pool: &VectorBufferPool,
126    queue: &crate::render::upload_queue::UploadQueue,
127    source_tiles: &[crate::coords::WorldTileCoords],
128    frame: VectorPaintFrame,
129    previous: Option<VectorPaintFrame>,
130) {
131    let VectorPaintFrame { zoom, .. } = frame;
132    for coords in source_tiles {
133        for entry in buffer_pool
134            .index()
135            .get_layers(*coords)
136            .into_iter()
137            .flatten()
138        {
139            let metadata = metadata_for_layer(&entry.style_layer, *coords, frame);
140            let unchanged = previous.is_some_and(|previous| {
141                bytemuck::bytes_of(&metadata_for_layer(&entry.style_layer, *coords, previous))
142                    == bytemuck::bytes_of(&metadata)
143            });
144            if !unchanged {
145                buffer_pool.update_layer_metadata(queue, entry, metadata);
146            }
147            if let Some(color) = paint::uniform_color(&entry.style_layer, f64::from(zoom)) {
148                if previous
149                    .map(|frame| frame.zoom)
150                    .and_then(|zoom| paint::uniform_color(&entry.style_layer, f64::from(zoom)))
151                    == Some(color)
152                {
153                    continue;
154                }
155                let count = (entry.feature_metadata_buffer_range().end
156                    - entry.feature_metadata_buffer_range().start)
157                    as usize
158                    / std::mem::size_of::<FillShaderFeatureMetadata>();
159                let colors = vec![FillShaderFeatureMetadata { color }; count];
160                buffer_pool.update_feature_metadata(queue, entry, &colors);
161            }
162        }
163    }
164}

166fn upload_tessellated_layer(
167    buffer_pool: &mut VectorBufferPool,
168    queue: &crate::render::upload_queue::UploadQueue,
169    tiles: &mut Tiles,
170    style: &Style,
171    source_tiles: Vec<crate::coords::WorldTileCoords>,
172    spatial: &[super::structures::SpatialBuffer],
173    paint: VectorPaintFrame,
174) -> bool {
175    let mut uploaded_tiles = 0;
176    let mut bytes = crate::render::memory_budget::UploadBudget::new(16 << 20);
177    for coords in source_tiles {
178        super::content::ensure(tiles, coords);
179        let Some((layers, pending)) = tiles.query_mut::<(
180            &mut VectorLayerBucketComponent,
181            &mut super::content::LayerReplacements,
182        )>(coords) else {
183            continue;
184        };
185        let mut uploaded = false;
186        for style_layer in &style.layers {
187            let replacement = pending
188                .vector
189                .iter()
190                .position(|layer| layer.style_layer_id == style_layer.id);
191            let bucket = replacement.map(|index| &pending.vector[index]).or_else(|| {
192                layers.layers.iter().find_map(|layer| match layer {
193                    VectorLayerBucket::AvailableLayer(layer)
194                        if layer.style_layer_id == style_layer.id =>
195                    {
196                        Some(layer)
197                    }
198                    _ => None,
199                })
200            });
201            let Some(bucket) = bucket else {
202                continue;
203            };
204            let loaded = buffer_pool
205                .get_loaded_style_layers_at(coords)
206                .is_some_and(|layers| layers.contains(style_layer.id.as_str()));
207            if replacement.is_none() && loaded {
208                continue;
209            }
210            if bucket.buffer.buffer.indices.is_empty() {
211                buffer_pool.remove_layer(coords, &style_layer.id);
212            } else {
213                if uploaded_tiles >= UPLOADS_PER_FRAME || !bytes.take(upload_size(bucket)) {
214                    return true;
215                }
216                if !upload_bucket(buffer_pool, queue, style_layer, bucket, spatial, paint) {
217                    continue;
218                }
219                uploaded = true;
220            }
221            if let Some(index) = replacement {
222                super::content::commit_vector(layers, pending.vector.remove(index));
223            }
224        }
225        if uploaded {
226            uploaded_tiles += 1;
227        }
228    }
229    false
230}

232fn upload_size(bucket: &AvailableVectorLayerBucket) -> usize {
233    bucket.buffer.buffer.vertices.len() * size_of::<crate::render::ShaderVertex>()
234        + bucket.buffer.buffer.indices.len() * size_of::<u32>()
235        + bucket
236            .feature_indices
237            .iter()
238            .map(|count| *count as usize)
239            .sum::<usize>()
240            * size_of::<FillShaderFeatureMetadata>()
241}

243fn upload_bucket(
244    buffer_pool: &mut VectorBufferPool,
245    queue: &crate::render::upload_queue::UploadQueue,
246    style_layer: &crate::style::layer::StyleLayer,
247    bucket: &AvailableVectorLayerBucket,
248    spatial: &[super::structures::SpatialBuffer],
249    paint: VectorPaintFrame,
250) -> bool {
251    let VectorPaintFrame { zoom, .. } = paint;
252    let AvailableVectorLayerBucket {
253        coords,
254        buffer,
255        feature_indices,
256        feature_colors,
257        feature_sort_keys,
258        ..
259    } = bucket;
260    let coords = *coords;
261    let color: Option<Vec4f32> = style_layer
262        .paint
263        .as_ref()
264        .and_then(|paint| paint.get_color())
265        .map(|color| color.into());
266
267    // Assign every feature in the layer the color from the style if no parsed feature_color exist.
268    let fallback_color = color.unwrap_or([0.0, 0.0, 0.0, 1.0]);
269
270    let feature_metadata = if let Some(color) = paint::uniform_color(style_layer, f64::from(zoom)) {
271        feature_metadata(feature_indices, &[], color)
272    } else {
273        feature_metadata(feature_indices, feature_colors, fallback_color)
274    };
275
276    let buffer = spatial
277        .iter()
278        .find(|(tile, id, _)| *tile == coords && *id == style_layer.id)
279        .map_or(buffer, |(_, _, spatial)| spatial);
280    let layer_metadata = metadata_for_layer(style_layer, coords, paint);
281    let runs = style_layer
282        .paint
283        .as_ref()
284        .and_then(super::pattern::per_feature_pattern)
285        .map(|_| pattern_runs(feature_indices, feature_colors, &buffer.buffer.indices));
286
287    let sort_runs = (style_layer.type_ == "circle" && !feature_sort_keys.is_empty())
288        .then(|| sort_runs(feature_indices, feature_sort_keys, &buffer.buffer.indices));
289
290    tracing::debug!(%coords, "allocating vector geometry");
291    if let Err(error) = buffer_pool.replace_layer_geometry(
292        queue,
293        coords,
294        style_layer.clone(),
295        buffer,
296        layer_metadata,
297        &feature_metadata,
298    ) {
299        tracing::error!(%coords, %error, "tile geometry upload failed");
300        return false;
301    }
302    buffer_pool.set_pattern_runs(coords, &style_layer.id, runs);
303    buffer_pool.set_sort_runs(coords, &style_layer.id, sort_runs);
304    true
305}

307/// The index range of each run of consecutive features that share a sort key.
308fn sort_runs(
309    feature_indices: &[u32],
310    feature_sort_keys: &[f32],
311    indices: &[u32],
312) -> Vec<(f32, std::ops::Range<u32>)> {
313    let mut runs: Vec<(f32, std::ops::Range<u32>)> = Vec::new();
314    let (mut vertex_end, mut position) = (0_u32, 0_u32);
315    for (feature, count) in feature_indices.iter().enumerate() {
316        vertex_end = vertex_end.wrapping_add(*count);
317        let first = position;
318        while indices
319            .get(position as usize)
320            .is_some_and(|index| *index < vertex_end)
321        {
322            position = position.wrapping_add(1);
323        }
324        if position == first {
325            continue;
326        }
327        let key = feature_sort_keys.get(feature).copied().unwrap_or(0.0);
328        match runs.last_mut() {
329            Some((last, range)) if *last == key && range.end == first => range.end = position,
330            _ => runs.push((key, first..position)),
331        }
332    }
333    runs
334}

336/// The index range of each run of consecutive features that share a pattern key.
337fn pattern_runs(
338    feature_indices: &[u32],
339    feature_colors: &[[f32; 4]],
340    indices: &[u32],
341) -> Vec<(u32, std::ops::Range<u32>)> {
342    let mut runs: Vec<(u32, std::ops::Range<u32>)> = Vec::new();
343    let (mut vertex_end, mut position) = (0_u32, 0_u32);
344    for (feature, count) in feature_indices.iter().enumerate() {
345        vertex_end = vertex_end.wrapping_add(*count);
346        let first = position;
347        while indices
348            .get(position as usize)
349            .is_some_and(|index| *index < vertex_end)
350        {
351            position = position.wrapping_add(1);
352        }
353        let key = feature_colors
354            .get(feature)
355            .and_then(|color| crate::style::pattern_key::key_of_value(color[0]));
356        let Some(key) = key.filter(|_| position > first) else {
357            continue;
358        };
359        match runs.last_mut() {
360            Some((last, range)) if *last == key && range.end == first => range.end = position,
361            _ => runs.push((key, first..position)),
362        }
363    }
364    runs
365}

367fn layer_translate_tile_units(
368    paint: Option<&LayerPaint>,
369    tile_zoom: crate::coords::ZoomLevel,
370    view_zoom: f32,
371    bearing: f32,
372) -> [f32; 2] {
373    let translated = crate::style::translation::layer_translate_pixels(
374        paint,
375        f64::from(view_zoom),
376        f64::from(bearing),
377    )
378    .map(|pixels| pixels as f32);
379    let pixels_to_tile_units = 8.0 * 2.0_f32.powf(f32::from(u8::from(tile_zoom)) - view_zoom);
380    [
381        translated[0] * pixels_to_tile_units,
382        translated[1] * pixels_to_tile_units,
383    ]
384}
