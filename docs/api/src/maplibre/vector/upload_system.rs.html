36fn layer_is_dynamic_circle(style: &Style, style_layer: &crate::style::layer::StyleLayer) -> bool {
37    #[cfg(feature = "geojson")]
38    {
39        crate::geojson::update::dynamic_points::uses_dynamic_circle_metadata(style, style_layer)
40    }
41    #[cfg(not(feature = "geojson"))]
42    {
43        let _ = (style, style_layer);
44        false
45    }
46}

60pub fn upload_system(
61    MapContext {
62        world,
63        style,
64        view_state,
65        renderer: Renderer {
66            queue, settings, ..
67        },
68        ..
69    }: &mut MapContext,
70) -> SystemResult {
71    let Some(pool_state) = world.resources.get::<Eventually<VectorBufferPool>>() else {
72        return Ok(());
73    };
74    if !matches!(pool_state, Initialized(_)) {
75        return Err(SystemError::Dependencies);
76    }
77    if settings.buffer_pools.is_empty()
78        || crate::render::eye_covering::EyeInFrame::reuses_content(world)
79    {
80        return Ok(());
81    }
82    let (paint_frame, previous_paint) = frame_paint(world, style, view_state);
83    let refresh_paint = previous_paint != Some(paint_frame);
84    let (source_tiles, painted_tiles, uncovered_sources) =
85        sources::frame_sources_and_painted(world, style, view_state)?;
86    let has_pending_uploads = {
87        let stamp = match world.resources.get::<Eventually<VectorBufferPool>>() {
88            Some(Initialized(pool)) => (style.revision(), pool.eviction_generation()),
89            _ => return Err(SystemError::Dependencies),
90        };
91        !source_tiles.iter().all(|&coords| {
92            world
93                .tiles
94                .query::<(
95                    &VectorLayerBucketComponent,
96                    &super::content::LayerReplacements,
97                )>(coords)
98                .is_some_and(|(layers, pending)| {
99                    layers.done && pending.vector.is_empty() && layers.uploaded_stamp == Some(stamp)
100                })
101        })
102    };
103    if has_pending_uploads {
104        crate::render::systems::retention_system::evict_before_upload(
105            world, style, view_state, queue,
106        );
107    }
108    let (spatial, changed) = super::structures::for_frame(world, style, &painted_tiles);
109    world.resources.get_or_init_mut::<FrameUploads>();
110    world.resources.get_or_init_mut::<DynPoints>();
111    world
112        .resources
113        .get_or_init_mut::<paint::PaintRefreshScratch>();
114    #[cfg(feature = "geojson")]
115    crate::geojson::update::dynamic_points::ensure_seeded_for_style(world, style);
116    style_refresh::apply_dirty_style(world, style, queue, paint_frame);
117    let Some((Initialized(buffer_pool), budget, dyn_points, paint_scratch)) =
118        world.resources.query_mut::<(
119            &mut Eventually<VectorBufferPool>,
120            &mut FrameUploads,
121            &mut DynPoints,
122            &mut paint::PaintRefreshScratch,
123        )>()
124    else {
125        return Err(SystemError::Dependencies);
126    };
127    if refresh_paint {
128        paint::refresh_layer_paint(
129            buffer_pool,
130            queue,
131            &world.tiles,
132            &painted_tiles,
133            paint_frame,
134            previous_paint,
135            paint_scratch,
136        );
137    }
138    if changed {
139        super::structures::refresh_gpu(buffer_pool, queue, &spatial, &world.tiles);
140    }
141    let allow_cpu_release = style.terrain.is_none() && settings.buffer_pools.vertices >= 1_000_000;
142    let (deferred, needs_refetch) = upload_tessellated_layer_with_uncovered(
143        buffer_pool,
144        queue,
145        &mut world.tiles,
146        style,
147        (source_tiles, &uncovered_sources),
148        (&spatial, paint_frame),
149        (budget, allow_cpu_release, dyn_points),
150    );
151    #[cfg(feature = "geojson")]
152    crate::geojson::update::dynamic_points::refresh_dynamic_point_layers(
153        world,
154        style,
155        queue,
156        &painted_tiles,
157    );
158    if deferred {
159        crate::render::frame_signals::keep_animating(world);
160    }
161    if needs_refetch {
162        crate::render::frame_signals::mark_dirty(world);
163    }
164    Ok(())
165}

167fn frame_paint(
168    world: &mut crate::tcs::world::World,
169    style: &Style,
170    view_state: &crate::render::view_state::ViewState,
171) -> (VectorPaintFrame, Option<VectorPaintFrame>) {
172    let zoom = if style.terrain.is_some() {
173        paint::stabilize_zoom(world, view_state.style_zoom().value()) as f32
174    } else {
175        (view_state.style_zoom().level() * 256.0).round() / 256.0
176    };
177    let bearing = ((view_state.camera().get_bearing().0.to_degrees() * 4.0).round() * 0.25)
178        .to_radians() as f32;
179    let light = style
180        .light
181        .clone()
182        .unwrap_or_default()
183        .extrusion_light(view_state.style_zoom().value(), f64::from(bearing))
184        .unwrap_or_default();
185    let paint_frame = VectorPaintFrame {
186        zoom,
187        bearing,
188        light,
189    };
190    let previous_paint = world.resources.get::<VectorPaintFrame>().copied();
191    world.resources.insert(paint_frame);
192    (paint_frame, previous_paint)
193}

197fn upload_tessellated_layer(
198    buffer_pool: &mut VectorBufferPool,
199    queue: &crate::render::upload_queue::UploadQueue,
200    tiles: &mut Tiles,
201    style: &Style,
202    source_tiles: Vec<crate::coords::WorldTileCoords>,
203    (spatial, paint): (&[super::structures::SpatialBuffer], VectorPaintFrame),
204    (budget, allow_cpu_release, dyn_points): (&mut FrameUploads, bool, &mut DynPoints),
205) -> (bool, bool) {
206    let empty = rustc_hash::FxHashSet::default();
207    upload_tessellated_layer_with_uncovered(
208        buffer_pool,
209        queue,
210        tiles,
211        style,
212        (source_tiles, &empty),
213        (spatial, paint),
214        (budget, allow_cpu_release, dyn_points),
215    )
216}

218fn upload_tessellated_layer_with_uncovered(
219    buffer_pool: &mut VectorBufferPool,
220    queue: &crate::render::upload_queue::UploadQueue,
221    tiles: &mut Tiles,
222    style: &Style,
223    (source_tiles, uncovered_sources): (
224        Vec<crate::coords::WorldTileCoords>,
225        &rustc_hash::FxHashSet<crate::coords::WorldTileCoords>,
226    ),
227    (spatial, paint): (&[super::structures::SpatialBuffer], VectorPaintFrame),
228    (budget, allow_cpu_release, dyn_points): (&mut FrameUploads, bool, &mut DynPoints),
229) -> (bool, bool) {
230    let stamp = (style.revision(), buffer_pool.eviction_generation());
231    if source_tiles.iter().all(|&coords| {
232        tiles
233            .query::<(
234                &VectorLayerBucketComponent,
235                &super::content::LayerReplacements,
236            )>(coords)
237            .is_some_and(|(layers, pending)| {
238                layers.done && pending.vector.is_empty() && layers.uploaded_stamp == Some(stamp)
239            })
240    }) {
241        return (false, false);
242    }
243    let has_dem_source = crate::terrain::source::dem_source(style).is_some();
244    let mut feature_meta_scratch = Vec::new();
245    let mut needs_refetch = false;
246    let mut any_deferred = false;
247    let protected: rustc_hash::FxHashSet<_> = source_tiles.iter().copied().collect();
248    for coords in source_tiles {
249        let is_uncovered = uncovered_sources.contains(&coords);
250        super::content::ensure(tiles, coords);
251        let Some((layers, pending)) = tiles.query::<(
252            &VectorLayerBucketComponent,
253            &super::content::LayerReplacements,
254        )>(coords) else {
255            continue;
256        };
257        if layers.done && pending.vector.is_empty() && layers.uploaded_stamp == Some(stamp) {
258            continue;
259        }
260        let mut tile_incomplete = false;
261        let mut uploaded = false;
262        let mut pending_cpu_releases: Vec<(String, Option<u64>, bool)> = Vec::new();
263        for style_layer in &style.layers {
264            enum Step {
265                Skip,
266                Exhausted,
267                Saturated,
268                Commit {
269                    replacement: Option<usize>,
270                    did_upload: bool,
271                    release_cpu: Option<Option<u64>>,
272                },
273            }
274            let step = {
275                let Some((layers, pending)) = tiles.query::<(
276                    &VectorLayerBucketComponent,
277                    &super::content::LayerReplacements,
278                )>(coords) else {
279                    break;
280                };
281                let replacement = pending
282                    .vector
283                    .iter()
284                    .position(|layer| layer.style_layer_id == style_layer.id);
285                let bucket = replacement.map(|index| &pending.vector[index]).or_else(|| {
286                    layers.layers.iter().find_map(|layer| match layer {
287                        VectorLayerBucket::AvailableLayer(layer)
288                            if layer.style_layer_id == style_layer.id =>
289                        {
290                            Some(layer)
291                        }
292                        _ => None,
293                    })
294                });
295                let Some(bucket) = bucket else {
296                    continue;
297                };
298                if replacement.is_none() && buffer_pool.has_layer(coords, &style_layer.id) {
299                    continue;
300                }
301                let is_dynamic_circle = layer_is_dynamic_circle(style, style_layer);
302                let is_uniform = !is_dynamic_circle
303                    && paint::uniform_color(style_layer, f64::from(paint.zoom)).is_some();
304                if !bucket.buffer.has_drawable_geometry() {
305                    buffer_pool.remove_layer(coords, &style_layer.id);
306                    Step::Commit {
307                        replacement,
308                        did_upload: false,
309                        release_cpu: None,
310                    }
311                } else if !bucket.buffer.has_cpu_indices() {
312                    if !style_layer.is_hidden() {
313                        needs_refetch = true;
314                    }
315                    tile_incomplete = true;
316                    continue;
317                } else {
318                    let bytes = upload_size(style_layer, bucket, is_uniform, is_dynamic_circle);
319                    if budget.geometry_tiles_exhausted_for(is_uncovered)
320                        || !budget.can_take_geometry(bytes, is_uncovered)
321                    {
322                        Step::Exhausted
323                    } else {
324                        let release_cpu = (allow_cpu_release
325                            && metadata::can_release_cpu_geometry(
326                                style_layer,
327                                bucket,
328                                has_dem_source,
329                            ))
330                        .then(|| bucket.buffer.shared_geometry_key());
331                        let fallback_vertices =
332                            metadata::find_shared_fallback_vertices(bucket, layers, pending);
333                        let paint_store =
334                            tiles.query::<&super::feature_paint::TileFeaturePaintComponent>(coords);
335                        let paint_entries = if replacement.is_some() {
336                            paint_store
337                                .and_then(|s| s.pending.get(&style_layer.id))
338                                .map(Vec::as_slice)
339                        } else {
340                            paint_store
341                                .and_then(|s| s.committed.get(&style_layer.id))
342                                .map(Vec::as_slice)
343                        };
344                        match upload_bucket(
345                            buffer_pool,
346                            queue,
347                            style_layer,
348                            bucket,
349                            fallback_vertices,
350                            (paint_entries, layers.overscaled_zoom, is_dynamic_circle),
351                            (spatial, paint),
352                            (&protected, &mut feature_meta_scratch, dyn_points),
353                        ) {
354                            UploadOutcome::Uploaded => {
355                                budget.take_geometry(bytes, is_uncovered);
356                                Step::Commit {
357                                    replacement,
358                                    did_upload: true,
359                                    release_cpu,
360                                }
361                            }
362                            UploadOutcome::PoolSaturated => Step::Saturated,
363                            UploadOutcome::Failed => Step::Skip,
364                        }
365                    }
366                }
367            };
368            match step {
369                Step::Skip | Step::Saturated => {
370                    tile_incomplete = true;
371                    continue;
372                }
373                Step::Exhausted => {
374                    metadata::flush_safe_cpu_releases(
375                        tiles,
376                        buffer_pool,
377                        coords,
378                        &pending_cpu_releases,
379                    );
380                    any_deferred = true;
381                    tile_incomplete = true;
382                    break;
383                }
384                Step::Commit {
385                    replacement,
386                    did_upload,
387                    release_cpu,
388                } => {
389                    uploaded |= did_upload;
390                    if let Some(index) = replacement {
391                        super::content::commit_pending_vector(tiles, coords, index);
392                    }
393                    if let Some(shared_key) = release_cpu {
394                        let release_colors =
395                            !crate::style::feature_state::layer_paint_reads_feature_state(
396                                style_layer,
397                            ) && !paint::feature_paint_depends_on_zoom(style_layer);
398                        pending_cpu_releases.push((
399                            style_layer.id.clone(),
400                            shared_key,
401                            release_colors,
402                        ));
403                    }
404                }
405            }
406        }
407        metadata::flush_safe_cpu_releases(tiles, buffer_pool, coords, &pending_cpu_releases);
408        if !tile_incomplete {
409            let end_stamp = (style.revision(), buffer_pool.eviction_generation());
410            if let Some((layers, pending)) = tiles.query_mut::<(
411                &mut VectorLayerBucketComponent,
412                &super::content::LayerReplacements,
413            )>(coords)
414            {
415                if layers.done && pending.vector.is_empty() {
416                    layers.uploaded_stamp = Some(end_stamp);
417                }
418            }
419        }
420        if uploaded {
421            budget.count_geometry_tile();
422        }
423    }
424    (any_deferred, needs_refetch)
425}
