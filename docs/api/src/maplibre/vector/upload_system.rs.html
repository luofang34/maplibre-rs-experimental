39pub fn upload_system(
40    MapContext {
41        world,
42        style,
43        view_state,
44        renderer: Renderer {
45            queue, settings, ..
46        },
47        ..
48    }: &mut MapContext,
49) -> SystemResult {
50    let Some(pool_state) = world.resources.get::<Eventually<VectorBufferPool>>() else {
51        return Ok(());
52    };
53    if !matches!(pool_state, Initialized(_)) {
54        return Err(SystemError::Dependencies);
55    }
56    if settings.buffer_pools.is_empty()
57        || crate::render::eye_covering::EyeInFrame::reuses_content(world)
58    {
59        return Ok(());
60    }
61    let (paint_frame, previous_paint) = frame_paint(world, style, view_state);
62    let refresh_paint = previous_paint != Some(paint_frame);
63    let (source_tiles, painted_tiles) =
64        sources::frame_sources_and_painted(world, style, view_state)?;
65    let (spatial, changed) = super::structures::for_frame(world, style, &painted_tiles);
66    world.resources.get_or_init_mut::<FrameUploads>();
67    let Some((Initialized(buffer_pool), budget)) = world
68        .resources
69        .query_mut::<(&mut Eventually<VectorBufferPool>, &mut FrameUploads)>()
70    else {
71        return Err(SystemError::Dependencies);
72    };
73    if refresh_paint {
74        refresh_layer_paint(
75            buffer_pool,
76            queue,
77            &painted_tiles,
78            paint_frame,
79            previous_paint,
80        );
81    }
82    if changed {
83        super::structures::refresh_gpu(buffer_pool, queue, &spatial, &world.tiles);
84    }
85    let deferred = upload_tessellated_layer(
86        buffer_pool,
87        queue,
88        &mut world.tiles,
89        style,
90        source_tiles,
91        (&spatial, paint_frame),
92        budget,
93    );
94    if deferred {
95        crate::render::frame_signals::keep_animating(world);
96    }
97    Ok(())
98}

100fn frame_paint(
101    world: &mut crate::tcs::world::World,
102    style: &Style,
103    view_state: &crate::render::view_state::ViewState,
104) -> (VectorPaintFrame, Option<VectorPaintFrame>) {
105    let zoom = if style.terrain.is_some() {
106        paint::stabilize_zoom(world, view_state.style_zoom().value()) as f32
107    } else {
108        view_state.style_zoom().level()
109    };
110    let bearing = view_state.camera().get_bearing().0 as f32;
111    let light = style
112        .light
113        .clone()
114        .unwrap_or_default()
115        .extrusion_light(view_state.style_zoom().value(), f64::from(bearing))
116        .unwrap_or_default();
117    let paint_frame = VectorPaintFrame {
118        zoom,
119        bearing,
120        light,
121    };
122    let previous_paint = world.resources.get::<VectorPaintFrame>().copied();
123    world.resources.insert(paint_frame);
124    (paint_frame, previous_paint)
125}

127fn refresh_layer_paint(
128    buffer_pool: &VectorBufferPool,
129    queue: &crate::render::upload_queue::UploadQueue,
130    source_tiles: &[crate::coords::WorldTileCoords],
131    frame: VectorPaintFrame,
132    previous: Option<VectorPaintFrame>,
133) {
134    let VectorPaintFrame { zoom, .. } = frame;
135    for coords in source_tiles {
136        for entry in buffer_pool
137            .index()
138            .get_layers(*coords)
139            .into_iter()
140            .flatten()
141        {
142            let metadata = metadata_for_layer(&entry.style_layer, *coords, frame);
143            let unchanged = previous.is_some_and(|previous| {
144                bytemuck::bytes_of(&metadata_for_layer(&entry.style_layer, *coords, previous))
145                    == bytemuck::bytes_of(&metadata)
146            });
147            if !unchanged {
148                buffer_pool.update_layer_metadata(queue, entry, metadata);
149            }
150            if let Some(color) = paint::uniform_color(&entry.style_layer, f64::from(zoom)) {
151                if previous
152                    .map(|frame| frame.zoom)
153                    .and_then(|zoom| paint::uniform_color(&entry.style_layer, f64::from(zoom)))
154                    == Some(color)
155                {
156                    continue;
157                }
158                let count = (entry.feature_metadata_buffer_range().end
159                    - entry.feature_metadata_buffer_range().start)
160                    as usize
161                    / std::mem::size_of::<FillShaderFeatureMetadata>();
162                let colors = vec![FillShaderFeatureMetadata { color }; count];
163                buffer_pool.update_feature_metadata(queue, entry, &colors);
164            }
165        }
166    }
167}

169fn upload_tessellated_layer(
170    buffer_pool: &mut VectorBufferPool,
171    queue: &crate::render::upload_queue::UploadQueue,
172    tiles: &mut Tiles,
173    style: &Style,
174    source_tiles: Vec<crate::coords::WorldTileCoords>,
175    (spatial, paint): (&[super::structures::SpatialBuffer], VectorPaintFrame),
176    budget: &mut FrameUploads,
177) -> bool {
178    let mut protected = std::collections::HashSet::with_capacity(source_tiles.len());
179    for coords in source_tiles {
180        protected.insert(coords);
181        super::content::ensure(tiles, coords);
182        let Some((layers, pending)) = tiles.query_mut::<(
183            &mut VectorLayerBucketComponent,
184            &mut super::content::LayerReplacements,
185        )>(coords) else {
186            continue;
187        };
188        let mut uploaded = false;
189        let mut saturated = false;
190        for style_layer in &style.layers {
191            let replacement = pending
192                .vector
193                .iter()
194                .position(|layer| layer.style_layer_id == style_layer.id);
195            let bucket = replacement.map(|index| &pending.vector[index]).or_else(|| {
196                layers.layers.iter().find_map(|layer| match layer {
197                    VectorLayerBucket::AvailableLayer(layer)
198                        if layer.style_layer_id == style_layer.id =>
199                    {
200                        Some(layer)
201                    }
202                    _ => None,
203                })
204            });
205            let Some(bucket) = bucket else {
206                continue;
207            };
208            let loaded = buffer_pool.has_layer(coords, &style_layer.id);
209            if replacement.is_none() && loaded {
210                continue;
211            }
212            if bucket.buffer.buffer.indices.is_empty() {
213                buffer_pool.remove_layer(coords, &style_layer.id);
214            } else {
215                if budget.geometry_tiles_exhausted()
216                    || !budget.take(UploadKind::Geometry, upload_size(bucket))
217                {
218                    return true;
219                }
220                match upload_bucket(
221                    buffer_pool,
222                    queue,
223                    style_layer,
224                    bucket,
225                    (spatial, paint),
226                    &protected,
227                ) {
228                    UploadOutcome::Uploaded => uploaded = true,
229                    UploadOutcome::PoolSaturated => {
230                        saturated = true;
231                        break;
232                    }
233                    UploadOutcome::Failed => continue,
234                }
235            }
236            if let Some(index) = replacement {
237                super::content::commit_vector(layers, pending.vector.remove(index));
238            }
239        }
240        if uploaded {
241            budget.count_geometry_tile();
242        }
243        if saturated {
244            return false;
245        }
246    }
247    false
248}

250enum UploadOutcome {
251    Uploaded,
252    PoolSaturated,
253    Failed,
254}

256fn upload_size(bucket: &AvailableVectorLayerBucket) -> usize {
257    bucket.buffer.buffer.vertices.len() * size_of::<crate::render::ShaderVertex>()
258        + bucket.buffer.buffer.indices.len() * size_of::<u32>()
259        + bucket
260            .feature_indices
261            .iter()
262            .map(|count| *count as usize)
263            .sum::<usize>()
264            * size_of::<FillShaderFeatureMetadata>()
265}

267fn upload_bucket(
268    buffer_pool: &mut VectorBufferPool,
269    queue: &crate::render::upload_queue::UploadQueue,
270    style_layer: &crate::style::layer::StyleLayer,
271    bucket: &AvailableVectorLayerBucket,
272    (spatial, paint): (&[super::structures::SpatialBuffer], VectorPaintFrame),
273    protected: &std::collections::HashSet<crate::coords::WorldTileCoords>,
274) -> UploadOutcome {
275    let VectorPaintFrame { zoom, .. } = paint;
276    let AvailableVectorLayerBucket {
277        coords,
278        buffer,
279        feature_indices,
280        feature_colors,
281        feature_sort_keys,
282        ..
283    } = bucket;
284    let coords = *coords;
285    let color: Option<Vec4f32> = style_layer
286        .paint
287        .as_ref()
288        .and_then(|paint| paint.get_color())
289        .map(|color| color.into());
290
291    // Assign every feature in the layer the color from the style if no parsed feature_color exist.
292    let fallback_color = color.unwrap_or([0.0, 0.0, 0.0, 1.0]);
293
294    let feature_metadata = if let Some(color) = paint::uniform_color(style_layer, f64::from(zoom)) {
295        feature_metadata(feature_indices, &[], color)
296    } else {
297        feature_metadata(feature_indices, feature_colors, fallback_color)
298    };
299
300    let buffer = spatial
301        .iter()
302        .find(|(tile, id, _)| *tile == coords && *id == style_layer.id)
303        .map_or(buffer, |(_, _, spatial)| spatial);
304    let layer_metadata = metadata_for_layer(style_layer, coords, paint);
305    let runs = style_layer
306        .paint
307        .as_ref()
308        .and_then(super::pattern::per_feature_pattern)
309        .map(|_| pattern_runs(feature_indices, feature_colors, &buffer.buffer.indices));
310
311    let sort_runs = (style_layer.type_ == "circle" && !feature_sort_keys.is_empty())
312        .then(|| sort_runs(feature_indices, feature_sort_keys, &buffer.buffer.indices));
313
314    tracing::debug!(%coords, "allocating vector geometry");
315    if let Err(error) = buffer_pool.replace_layer_geometry_unless(
316        queue,
317        coords,
318        style_layer.clone(),
319        buffer,
320        layer_metadata,
321        &feature_metadata,
322        |candidate| protected.contains(&candidate),
323    ) {
324        if matches!(
325            error,
326            crate::vector::resource::AllocationError::Capacity { .. }
327        ) {
328            tracing::debug!(
329                %coords,
330                %error,
331                "vector buffer pool saturated by higher-priority tiles; skipping peripheral upload"
332            );
333            return UploadOutcome::PoolSaturated;
334        }
335        tracing::error!(%coords, %error, "tile geometry upload failed");
336        return UploadOutcome::Failed;
337    }
338    buffer_pool.set_pattern_runs(coords, &style_layer.id, runs);
339    buffer_pool.set_sort_runs(coords, &style_layer.id, sort_runs);
340    UploadOutcome::Uploaded
341}

343/// The index range of each run of consecutive features that share a sort key.
344fn sort_runs(
345    feature_indices: &[u32],
346    feature_sort_keys: &[f32],
347    indices: &[u32],
348) -> Vec<(f32, std::ops::Range<u32>)> {
349    let mut runs: Vec<(f32, std::ops::Range<u32>)> = Vec::new();
350    let (mut vertex_end, mut position) = (0_u32, 0_u32);
351    for (feature, count) in feature_indices.iter().enumerate() {
352        vertex_end = vertex_end.wrapping_add(*count);
353        let first = position;
354        while indices
355            .get(position as usize)
356            .is_some_and(|index| *index < vertex_end)
357        {
358            position = position.wrapping_add(1);
359        }
360        if position == first {
361            continue;
362        }
363        let key = feature_sort_keys.get(feature).copied().unwrap_or(0.0);
364        match runs.last_mut() {
365            Some((last, range)) if *last == key && range.end == first => range.end = position,
366            _ => runs.push((key, first..position)),
367        }
368    }
369    runs
370}

372/// The index range of each run of consecutive features that share a pattern key.
373fn pattern_runs(
374    feature_indices: &[u32],
375    feature_colors: &[[f32; 4]],
376    indices: &[u32],
377) -> Vec<(u32, std::ops::Range<u32>)> {
378    let mut runs: Vec<(u32, std::ops::Range<u32>)> = Vec::new();
379    let (mut vertex_end, mut position) = (0_u32, 0_u32);
380    for (feature, count) in feature_indices.iter().enumerate() {
381        vertex_end = vertex_end.wrapping_add(*count);
382        let first = position;
383        while indices
384            .get(position as usize)
385            .is_some_and(|index| *index < vertex_end)
386        {
387            position = position.wrapping_add(1);
388        }
389        let key = feature_colors
390            .get(feature)
391            .and_then(|color| crate::style::pattern_key::key_of_value(color[0]));
392        let Some(key) = key.filter(|_| position > first) else {
393            continue;
394        };
395        match runs.last_mut() {
396            Some((last, range)) if *last == key && range.end == first => range.end = position,
397            _ => runs.push((key, first..position)),
398        }
399    }
400    runs
401}

403fn layer_translate_tile_units(
404    paint: Option<&LayerPaint>,
405    tile_zoom: crate::coords::ZoomLevel,
406    view_zoom: f32,
407    bearing: f32,
408) -> [f32; 2] {
409    let translated = crate::style::translation::layer_translate_pixels(
410        paint,
411        f64::from(view_zoom),
412        f64::from(bearing),
413    )
414    .map(|pixels| pixels as f32);
415    let pixels_to_tile_units = 8.0 * 2.0_f32.powf(f32::from(u8::from(tile_zoom)) - view_zoom);
416    [
417        translated[0] * pixels_to_tile_units,
418        translated[1] * pixels_to_tile_units,
419    ]
420}
