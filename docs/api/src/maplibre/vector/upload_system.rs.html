36fn layer_is_dynamic_circle(style: &Style, style_layer: &crate::style::layer::StyleLayer) -> bool {
37    #[cfg(feature = "geojson")]
38    {
39        crate::geojson::update::dynamic_points::uses_dynamic_circle_metadata(style, style_layer)
40    }
41    #[cfg(not(feature = "geojson"))]
42    {
43        let _ = (style, style_layer);
44        false
45    }
46}

60pub fn upload_system(
61    MapContext {
62        world,
63        style,
64        view_state,
65        renderer: Renderer {
66            queue, settings, ..
67        },
68        ..
69    }: &mut MapContext,
70) -> SystemResult {
71    let Some(pool_state) = world.resources.get::<Eventually<VectorBufferPool>>() else {
72        return Ok(());
73    };
74    if !matches!(pool_state, Initialized(_)) {
75        return Err(SystemError::Dependencies);
76    }
77    if settings.buffer_pools.is_empty()
78        || crate::render::eye_covering::EyeInFrame::reuses_content(world)
79    {
80        super::request_system::clear_upload_permits(world);
81        return Ok(());
82    }
83    let (paint_frame, previous_paint) = frame_paint(world, style, view_state);
84    let refresh_paint = previous_paint != Some(paint_frame);
85    let (source_tiles, painted_tiles, uncovered_sources) =
86        sources::frame_sources_and_painted(world, style, view_state)?;
87    let has_pending_uploads = {
88        let stamp = match world.resources.get::<Eventually<VectorBufferPool>>() {
89            Some(Initialized(pool)) => (style.revision(), pool.eviction_generation()),
90            _ => return Err(SystemError::Dependencies),
91        };
92        !source_tiles.iter().all(|&coords| {
93            world
94                .tiles
95                .query::<(
96                    &VectorLayerBucketComponent,
97                    &super::content::LayerReplacements,
98                )>(coords)
99                .is_some_and(|(layers, pending)| {
100                    layers.done && pending.vector.is_empty() && layers.uploaded_stamp == Some(stamp)
101                })
102        })
103    };
104    let permit_source_set =
105        super::request_system::snapshot_permit_sources(world, has_pending_uploads, &source_tiles);
106    let terrain_booting = style.terrain.is_some()
107        && world
108            .resources
109            .get::<crate::terrain::request_system::DrapeRequests>()
110            .is_none_or(|requests| requests.0.is_empty());
111    if has_pending_uploads && !terrain_booting {
112        crate::render::systems::retention_system::evict_before_upload(
113            world,
114            style,
115            view_state,
116            queue,
117            settings.tight_tile_requests,
118        );
119    }
120    let (spatial, changed) = super::structures::for_frame(world, style, &painted_tiles);
121    world.resources.get_or_init_mut::<FrameUploads>();
122    world.resources.get_or_init_mut::<DynPoints>();
123    world
124        .resources
125        .get_or_init_mut::<paint::PaintRefreshScratch>();
126    #[cfg(feature = "geojson")]
127    crate::geojson::update::dynamic_points::ensure_seeded_for_style(world, style);
128    style_refresh::apply_dirty_style(world, style, queue, paint_frame);
129    let retain_loaded_tiles = world
130        .resources
131        .get::<crate::render::systems::retention_system::RetainLoadedTiles>()
132        .is_some();
133    let Some((Initialized(buffer_pool), budget, dyn_points, paint_scratch)) =
134        world.resources.query_mut::<(
135            &mut Eventually<VectorBufferPool>,
136            &mut FrameUploads,
137            &mut DynPoints,
138            &mut paint::PaintRefreshScratch,
139        )>()
140    else {
141        return Err(SystemError::Dependencies);
142    };
143    if refresh_paint {
144        paint_scratch.sync_style(style);
145        paint::refresh_layer_paint(
146            buffer_pool,
147            queue,
148            &world.tiles,
149            &painted_tiles,
150            paint_frame,
151            previous_paint,
152            paint_scratch,
153        );
154    }
155    if changed {
156        super::structures::refresh_gpu(buffer_pool, queue, &spatial, &world.tiles);
157    }
158    let (deferred, needs_refetch) = if has_pending_uploads {
159        let allow_cpu_release = (style.terrain.is_none() || cfg!(target_arch = "wasm32"))
160            && !retain_loaded_tiles
161            && settings.buffer_pools.vertices >= 1_000_000;
162        upload_tessellated_layer_with_uncovered(
163            buffer_pool,
164            queue,
165            &mut world.tiles,
166            style,
167            (source_tiles, &uncovered_sources),
168            (&spatial, paint_frame),
169            (budget, allow_cpu_release, dyn_points),
170        )
171    } else {
172        (false, false)
173    };
174    #[cfg(feature = "geojson")]
175    crate::geojson::update::dynamic_points::refresh_dynamic_point_layers(
176        world,
177        style,
178        queue,
179        &painted_tiles,
180    );
181    if deferred {
182        crate::render::frame_signals::keep_animating(world);
183    }
184    super::request_system::reconcile_upload_permits(
185        world,
186        deferred,
187        style.revision(),
188        permit_source_set.as_ref(),
189    );
190    if needs_refetch {
191        crate::render::frame_signals::mark_dirty(world);
192    }
193    Ok(())
194}

196fn frame_paint(
197    world: &mut crate::tcs::world::World,
198    style: &Style,
199    view_state: &crate::render::view_state::ViewState,
200) -> (VectorPaintFrame, Option<VectorPaintFrame>) {
201    let zoom = if style.terrain.is_some() {
202        paint::stabilize_zoom(world, view_state.style_zoom().value()) as f32
203    } else {
204        (view_state.style_zoom().level() * 256.0).round() / 256.0
205    };
206    let bearing = ((view_state.camera().get_bearing().0.to_degrees() * 4.0).round() * 0.25)
207        .to_radians() as f32;
208    let light = style
209        .light
210        .as_ref()
211        .map_or_else(
212            || {
213                crate::style::light::LightSpecification::default()
214                    .extrusion_light(view_state.style_zoom().value(), f64::from(bearing))
215            },
216            |light| light.extrusion_light(view_state.style_zoom().value(), f64::from(bearing)),
217        )
218        .unwrap_or_default();
219    let paint_frame = VectorPaintFrame {
220        zoom,
221        bearing,
222        light,
223    };
224    let previous_paint = world.resources.get::<VectorPaintFrame>().copied();
225    world.resources.insert(paint_frame);
226    (paint_frame, previous_paint)
227}

231fn upload_tessellated_layer(
232    buffer_pool: &mut VectorBufferPool,
233    queue: &crate::render::upload_queue::UploadQueue,
234    tiles: &mut Tiles,
235    style: &Style,
236    source_tiles: Vec<crate::coords::WorldTileCoords>,
237    (spatial, paint): (&[super::structures::SpatialBuffer], VectorPaintFrame),
238    (budget, allow_cpu_release, dyn_points): (&mut FrameUploads, bool, &mut DynPoints),
239) -> (bool, bool) {
240    let empty = rustc_hash::FxHashSet::default();
241    upload_tessellated_layer_with_uncovered(
242        buffer_pool,
243        queue,
244        tiles,
245        style,
246        (source_tiles, &empty),
247        (spatial, paint),
248        (budget, allow_cpu_release, dyn_points),
249    )
250}

252fn upload_tessellated_layer_with_uncovered(
253    buffer_pool: &mut VectorBufferPool,
254    queue: &crate::render::upload_queue::UploadQueue,
255    tiles: &mut Tiles,
256    style: &Style,
257    (source_tiles, uncovered_sources): (
258        Vec<crate::coords::WorldTileCoords>,
259        &rustc_hash::FxHashSet<crate::coords::WorldTileCoords>,
260    ),
261    (spatial, paint): (&[super::structures::SpatialBuffer], VectorPaintFrame),
262    (budget, allow_cpu_release, dyn_points): (&mut FrameUploads, bool, &mut DynPoints),
263) -> (bool, bool) {
264    let stamp = (style.revision(), buffer_pool.eviction_generation());
265    if source_tiles.iter().all(|&coords| {
266        tiles
267            .query::<(
268                &VectorLayerBucketComponent,
269                &super::content::LayerReplacements,
270            )>(coords)
271            .is_some_and(|(layers, pending)| {
272                layers.done && pending.vector.is_empty() && layers.uploaded_stamp == Some(stamp)
273            })
274    }) {
275        return (false, false);
276    }
277    let has_dem_source = crate::terrain::source::dem_source(style).is_some();
278    let mut feature_meta_scratch = Vec::new();
279    let mut needs_refetch = false;
280    let mut any_deferred = false;
281    let protected: rustc_hash::FxHashSet<_> = source_tiles.iter().copied().collect();
282    for coords in source_tiles {
283        let is_uncovered = uncovered_sources.contains(&coords);
284        super::content::ensure(tiles, coords);
285        let Some((layers, pending)) = tiles.query::<(
286            &VectorLayerBucketComponent,
287            &super::content::LayerReplacements,
288        )>(coords) else {
289            continue;
290        };
291        if layers.done && pending.vector.is_empty() && layers.uploaded_stamp == Some(stamp) {
292            continue;
293        }
294        let mut tile_incomplete = false;
295        let mut uploaded = false;
296        let mut pending_cpu_releases: Vec<(String, Option<u64>, bool)> = Vec::new();
297        for style_layer in &style.layers {
298            enum Step {
299                Skip,
300                Exhausted,
301                Saturated,
302                Commit {
303                    replacement: Option<usize>,
304                    did_upload: bool,
305                    release_cpu: Option<Option<u64>>,
306                },
307            }
308            let step = {
309                let Some((layers, pending)) = tiles.query::<(
310                    &VectorLayerBucketComponent,
311                    &super::content::LayerReplacements,
312                )>(coords) else {
313                    break;
314                };
315                let replacement = pending
316                    .vector
317                    .iter()
318                    .position(|layer| layer.style_layer_id == style_layer.id);
319                let bucket = replacement.map(|index| &pending.vector[index]).or_else(|| {
320                    layers.layers.iter().find_map(|layer| match layer {
321                        VectorLayerBucket::AvailableLayer(layer)
322                            if layer.style_layer_id == style_layer.id =>
323                        {
324                            Some(layer)
325                        }
326                        _ => None,
327                    })
328                });
329                let Some(bucket) = bucket else {
330                    continue;
331                };
332                if replacement.is_none() && buffer_pool.has_layer(coords, &style_layer.id) {
333                    continue;
334                }
335                let is_dynamic_circle = layer_is_dynamic_circle(style, style_layer);
336                let is_uniform = !is_dynamic_circle
337                    && paint::uniform_color(style_layer, f64::from(paint.zoom)).is_some();
338                if !bucket.buffer.has_drawable_geometry() {
339                    buffer_pool.remove_layer(coords, &style_layer.id);
340                    Step::Commit {
341                        replacement,
342                        did_upload: false,
343                        release_cpu: None,
344                    }
345                } else if !bucket.buffer.has_cpu_indices() {
346                    if !style_layer.is_hidden() {
347                        needs_refetch = true;
348                    }
349                    tile_incomplete = true;
350                    continue;
351                } else {
352                    let bytes = upload_size(style_layer, bucket, is_uniform, is_dynamic_circle);
353                    if budget.geometry_tiles_exhausted_for(is_uncovered)
354                        || !budget.can_take_geometry(bytes, is_uncovered)
355                    {
356                        Step::Exhausted
357                    } else {
358                        let release_cpu = (allow_cpu_release
359                            && metadata::can_release_cpu_geometry(
360                                style_layer,
361                                bucket,
362                                has_dem_source,
363                            ))
364                        .then(|| bucket.buffer.shared_geometry_key());
365                        let fallback_vertices =
366                            metadata::find_shared_fallback_vertices(bucket, layers, pending);
367                        let paint_store =
368                            tiles.query::<&super::feature_paint::TileFeaturePaintComponent>(coords);
369                        let paint_entries = if replacement.is_some() {
370                            paint_store
371                                .and_then(|s| s.pending.get(&style_layer.id))
372                                .map(Vec::as_slice)
373                        } else {
374                            paint_store
375                                .and_then(|s| s.committed.get(&style_layer.id))
376                                .map(Vec::as_slice)
377                        };
378                        match upload_bucket(
379                            buffer_pool,
380                            queue,
381                            style_layer,
382                            bucket,
383                            fallback_vertices,
384                            (paint_entries, layers.overscaled_zoom, is_dynamic_circle),
385                            (spatial, paint),
386                            (&protected, &mut feature_meta_scratch, dyn_points),
387                        ) {
388                            UploadOutcome::Uploaded => {
389                                budget.take_geometry(bytes, is_uncovered);
390                                Step::Commit {
391                                    replacement,
392                                    did_upload: true,
393                                    release_cpu,
394                                }
395                            }
396                            UploadOutcome::PoolSaturated => Step::Saturated,
397                            UploadOutcome::Failed => Step::Skip,
398                        }
399                    }
400                }
401            };
402            match step {
403                Step::Skip | Step::Saturated => {
404                    tile_incomplete = true;
405                    continue;
406                }
407                Step::Exhausted => {
408                    metadata::flush_safe_cpu_releases(
409                        tiles,
410                        buffer_pool,
411                        coords,
412                        &pending_cpu_releases,
413                    );
414                    any_deferred = true;
415                    tile_incomplete = true;
416                    break;
417                }
418                Step::Commit {
419                    replacement,
420                    did_upload,
421                    release_cpu,
422                } => {
423                    uploaded |= did_upload;
424                    if let Some(index) = replacement {
425                        super::content::commit_pending_vector(tiles, coords, index);
426                    }
427                    if let Some(shared_key) = release_cpu {
428                        let release_colors =
429                            !crate::style::feature_state::layer_paint_reads_feature_state(
430                                style_layer,
431                            ) && !paint::feature_paint_depends_on_zoom(style_layer);
432                        pending_cpu_releases.push((
433                            style_layer.id.clone(),
434                            shared_key,
435                            release_colors,
436                        ));
437                    }
438                }
439            }
440        }
441        metadata::flush_safe_cpu_releases(tiles, buffer_pool, coords, &pending_cpu_releases);
442        if !tile_incomplete {
443            let end_stamp = (style.revision(), buffer_pool.eviction_generation());
444            if let Some((layers, pending)) = tiles.query_mut::<(
445                &mut VectorLayerBucketComponent,
446                &super::content::LayerReplacements,
447            )>(coords)
448            {
449                if layers.done && pending.vector.is_empty() {
450                    layers.uploaded_stamp = Some(end_stamp);
451                }
452            }
453        }
454        if uploaded {
455            budget.count_geometry_tile();
456        }
457    }
458    (any_deferred, needs_refetch)
459}
