1//! Queues [PhaseItems](crate::render::render_phase::PhaseItem) for rendering.
2use crate::{
3    context::MapContext,
4    io::tile_sources::TileKind,
5    render::{
6        eventually::{Eventually, Eventually::Initialized},
7        render_commands::DrawMasks,
8        render_phase::{
9            DrawState, LayerItem, ProjectionBinding, RenderPhase, SortedRun, TileMaskItem,
10        },
11        tile_view_pattern::WgpuTileViewPattern,
12    },
13    tcs::{
14        system::{SystemError, SystemResult},
15        tiles::Tile,
16    },
17    vector::{
18        render_commands::{
19            DrawCircleTiles, DrawExtrusionClear, DrawExtrusionColor, DrawExtrusionDepth,
20            DrawExtrusionPatternColor, DrawLineTiles, DrawPatternTiles, DrawVectorTiles,
21        },
22        VectorBufferPool,
23    },
24};

26pub fn queue_system(
27    MapContext {
28        style,
29        view_state,
30        world,
31        ..
32    }: &mut MapContext,
33) -> SystemResult {
34    let Some((
35        Initialized(tile_view_pattern),
36        Initialized(buffer_pool),
37        mask_phase,
38        layer_item_phase,
39    )) = world.resources.query_mut::<(
40        &mut Eventually<WgpuTileViewPattern>,
41        &mut Eventually<VectorBufferPool>,
42        &mut RenderPhase<TileMaskItem>,
43        &mut RenderPhase<LayerItem>,
44    )>()
45    else {
46        return Err(SystemError::Dependencies);
47    };
48
49    let buffer_pool_index = buffer_pool.index();
50    let zoom = view_state.zoom().value();
51    let uses_globe = style
52        .projection
53        .as_ref()
54        .is_some_and(|specification| specification.projection_type.uses_globe_rendering(zoom));
55
56    let pattern_layers: std::collections::HashSet<&str> = style
57        .layers
58        .iter()
59        .filter(|layer| {
60            layer.paint.as_ref().is_some_and(|paint| {
61                super::pattern::pattern_name(paint, view_state.style_zoom().value()).is_some()
62                    || super::pattern::per_feature_pattern(paint).is_some()
63            })
64        })
65        .map(|layer| layer.id.as_str())
66        .collect();
67    // Every tile's depth is drawn before any tile's colour, so an extrusion that crosses tiles
68    // shows its nearest surface once instead of once per tile.
69    let mut extrusion_colors = Vec::new();
70    let mut extrusion_clears = Vec::new();
71    for view_tile in tile_view_pattern.iter() {
72        let coords = &view_tile.coords();
73        tracing::trace!("Drawing tile at {coords}");
74
75        // draw tile normal or the source e.g. parent or children
76        view_tile.render_kind(TileKind::Vector, |source_shape| {
77            if uses_globe {
78                mask_phase.add(TileMaskItem {
79                    projection: ProjectionBinding::View,
80                    draw_function: Box::new(DrawState::<TileMaskItem, DrawMasks>::new()),
81                    source_shape: source_shape.clone(),
82                    generate_borders: true,
83                });
84            }
85            mask_phase.add(TileMaskItem {
86                projection: ProjectionBinding::View,
87                draw_function: Box::new(DrawState::<TileMaskItem, DrawMasks>::new()),
88                source_shape: source_shape.clone(),
89                generate_borders: false,
90            });
91
92            if let Some(layer_entries) = buffer_pool_index.get_layers(source_shape.coords()) {
93                for layer_entry in layer_entries {
94                    if !layer_entry
95                        .style_layer
96                        .is_visible_at(view_state.style_zoom().value())
97                    {
98                        continue;
99                    }
100                    if super::pattern::names_missing_image(
101                        layer_entry.style_layer.paint.as_ref(),
102                        style,
103                        view_state.style_zoom().value(),
104                    ) {
105                        continue;
106                    }
107                    let draw_function: Box<dyn crate::render::render_phase::Draw<LayerItem>> =
108                        match layer_entry.style_layer.type_.as_str() {
109                            // The heatmap plugin draws these into a density target instead.
110                            "heatmap" => continue,
111                            "line" => Box::new(DrawState::<LayerItem, DrawLineTiles>::new()),
112                            "circle" => Box::new(DrawState::<LayerItem, DrawCircleTiles>::new()),
113                            "fill-extrusion" => {
114                                extrusion_clears.push(LayerItem {
115                                    projection: ProjectionBinding::View,
116                                    draw_function: Box::new(DrawState::<
117                                        LayerItem,
118                                        DrawExtrusionClear,
119                                    >::new(
120                                    )),
121                                    index: layer_entry.style_layer.index,
122                                    generate_borders: false,
123                                    style_layer: layer_entry.style_layer.id.clone(),
124                                    tile: Tile {
125                                        coords: layer_entry.coords,
126                                    },
127                                    source_shape: source_shape.clone(),
128                                    run: None,
129                                });
130                                let color: Box<dyn crate::render::render_phase::Draw<LayerItem>> =
131                                    if pattern_layers.contains(layer_entry.style_layer.id.as_str())
132                                    {
133                                        Box::new(
134                                            DrawState::<LayerItem, DrawExtrusionPatternColor>::new(
135                                            ),
136                                        )
137                                    } else {
138                                        Box::new(DrawState::<LayerItem, DrawExtrusionColor>::new())
139                                    };
140                                extrusion_colors.push(LayerItem {
141                                    projection: ProjectionBinding::View,
142                                    draw_function: color,
143                                    index: layer_entry.style_layer.index,
144                                    generate_borders: false,
145                                    style_layer: layer_entry.style_layer.id.clone(),
146                                    tile: Tile {
147                                        coords: layer_entry.coords,
148                                    },
149                                    source_shape: source_shape.clone(),
150                                    run: None,
151                                });
152                                Box::new(DrawState::<LayerItem, DrawExtrusionDepth>::new())
153                            }
154                            "fill"
155                                if pattern_layers.contains(layer_entry.style_layer.id.as_str()) =>
156                            {
157                                Box::new(DrawState::<LayerItem, DrawPatternTiles>::new())
158                            }
159                            _ => Box::new(DrawState::<LayerItem, DrawVectorTiles>::new()),
160                        };
161
162                    // Features that draw in the order of a sort key, across tiles, draw a run at
163                    // a time.
164                    let runs = (layer_entry.style_layer.type_ == "circle")
165                        .then(|| {
166                            buffer_pool.sort_runs(layer_entry.coords, &layer_entry.style_layer.id)
167                        })
168                        .flatten();
169                    if let Some(runs) = runs.filter(|runs| !runs.is_empty()) {
170                        for (key, range) in runs {
171                            layer_item_phase.add(LayerItem {
172                                projection: ProjectionBinding::View,
173                                draw_function: Box::new(
174                                    DrawState::<LayerItem, DrawCircleTiles>::new(),
175                                ),
176                                index: layer_entry.style_layer.index,
177                                generate_borders: false,
178                                style_layer: layer_entry.style_layer.id.clone(),
179                                tile: Tile {
180                                    coords: layer_entry.coords,
181                                },
182                                source_shape: source_shape.clone(),
183                                run: Some(SortedRun {
184                                    key: *key,
185                                    range: range.clone(),
186                                }),
187                            });
188                        }
189                        continue;
190                    }
191                    layer_item_phase.add(LayerItem {
192                        projection: ProjectionBinding::View,
193                        draw_function,
194                        index: layer_entry.style_layer.index,
195                        generate_borders: false,
196                        style_layer: layer_entry.style_layer.id.clone(),
197                        tile: Tile {
198                            coords: layer_entry.coords,
199                        },
200                        source_shape: source_shape.clone(),
201                        run: None,
202                    });
203                }
204            }
205        });
206    }
207
208    for item in extrusion_colors.into_iter().chain(extrusion_clears) {
209        layer_item_phase.add(item);
210    }
211
212    Ok(())
213}
