33pub fn queue_system(
34    MapContext {
35        style,
36        view_state,
37        world,
38        renderer,
39        ..
40    }: &mut MapContext,
41) -> SystemResult {
42    if renderer.settings.buffer_pools.is_empty() {
43        return Ok(());
44    }
45    let _ = world.resources.get_or_init_mut::<VectorQueueScratch>();
46    let Some((
47        Initialized(tile_view_pattern),
48        Initialized(buffer_pool),
49        mask_phase,
50        layer_item_phase,
51        scratch,
52    )) = world.resources.query_mut::<(
53        &mut Eventually<WgpuTileViewPattern>,
54        &mut Eventually<VectorBufferPool>,
55        &mut RenderPhase<TileMaskItem>,
56        &mut RenderPhase<LayerItem>,
57        &mut VectorQueueScratch,
58    )>()
59    else {
60        return Err(SystemError::Dependencies);
61    };
62
63    let zoom = view_state.zoom().value();
64    let style_zoom = view_state.style_zoom().value();
65    let uses_globe = style
66        .projection
67        .as_ref()
68        .is_some_and(|specification| specification.projection_type.uses_globe_rendering(zoom));
69    let skip_draped =
70        style.terrain.is_some() && crate::terrain::source::dem_source(style).is_some();
71
72    let layer_state_digest = scratch.sync_style_sets(style, style_zoom);
73    let cache_key = VectorPhaseCacheKey {
74        topology_gen: tile_view_pattern.topology_generation(),
75        buffer_pool_rev: buffer_pool.revision(),
76        style_rev: style.revision(),
77        layer_state_digest,
78        uses_globe,
79        skip_draped,
80    };
81    if let Some(cached) = &scratch.cached_phases {
82        if cached.key == cache_key {
83            mask_phase.extend_from_slice(&cached.masks);
84            layer_item_phase.extend_from_slice(&cached.layers);
85            return Ok(());
86        }
87    }
88
89    let mask_start = mask_phase.size();
90    let layer_start = layer_item_phase.size();
91    let buffer_pool_index = buffer_pool.index();
92    let VectorQueueScratch {
93        pattern_layers,
94        missing_image_layers,
95        translucent_extrusion_layers,
96        unclipped_3d_layers,
97        zero_opacity_layers,
98        drapeable_layers,
99        has_visible_heatmap,
100        active_unclipped_3d_shapes,
101        queued_unclipped_3d,
102        extrusion_colors,
103        extrusion_clears,
104        translucent_fronts,
105        ..
106    } = scratch;
107    active_unclipped_3d_shapes.clear();
108    for view_tile in tile_view_pattern.iter() {
109        view_tile.render_kind(TileKind::Vector, |shape| {
110            if let Some(layer_entries) = buffer_pool_index.get_layers(shape.coords()) {
111                for layer_entry in layer_entries {
112                    if unclipped_3d_layers.contains(&layer_entry.style_layer_id)
113                        && !zero_opacity_layers.contains(&layer_entry.style_layer_id)
114                        && layer_entry.style_layer.is_visible_at(style_zoom)
115                    {
116                        active_unclipped_3d_shapes.insert((
117                            shape.wrap,
118                            shape.coords(),
119                            layer_entry.style_layer.index,
120                        ));
121                    }
122                }
123            }
124        });
125    }
126    // Every tile's depth is drawn before any tile's colour, so an extrusion that crosses tiles
127    // shows its nearest surface once instead of once per tile.
128    extrusion_colors.clear();
129    extrusion_clears.clear();
130    translucent_fronts.clear();
131    queued_unclipped_3d.clear();
132    for view_tile in tile_view_pattern.iter() {
133        let coords = &view_tile.coords();
134        tracing::trace!("Drawing tile at {coords}");
135
136        // draw tile normal or the source e.g. parent or children
137        view_tile.render_kind(TileKind::Vector, |source_shape| {
138            let shape_data = source_shape.draw_data();
139            let mut mask_emitted = false;
140            let mut ensure_mask = |mask_phase: &mut RenderPhase<TileMaskItem>| {
141                if !mask_emitted {
142                    mask_emitted = true;
143                    if uses_globe {
144                        mask_phase.add(TileMaskItem {
145                            projection: ProjectionBinding::View,
146                            draw_function: &DrawState::<TileMaskItem, DrawMasks>::INSTANCE,
147                            source_shape: shape_data,
148                            generate_borders: true,
149                            is_vector: true,
150                        });
151                    }
152                    mask_phase.add(TileMaskItem {
153                        projection: ProjectionBinding::View,
154                        draw_function: &DrawState::<TileMaskItem, DrawMasks>::INSTANCE,
155                        source_shape: shape_data,
156                        generate_borders: false,
157                        is_vector: true,
158                    });
159                }
160            };
161            if !skip_draped || *has_visible_heatmap {
162                ensure_mask(mask_phase);
163            }
164
165            if let Some(layer_entries) = buffer_pool_index.get_layers(source_shape.coords()) {
166                for (slot_idx, layer_entry) in layer_entries.iter().enumerate() {
167                    if (skip_draped && drapeable_layers.contains(&layer_entry.style_layer_id))
168                        || zero_opacity_layers.contains(&layer_entry.style_layer_id)
169                        || missing_image_layers.contains(&layer_entry.style_layer_id)
170                        || !layer_entry.style_layer.is_visible_at(style_zoom)
171                    {
172                        continue;
173                    }
174                    let layer_id = layer_entry.style_layer_id;
175                    let entry_slot = u16::try_from(slot_idx).unwrap_or(UNSET_ENTRY_SLOT);
176                    let is_unclipped_3d = unclipped_3d_layers.contains(&layer_id);
177                    if is_unclipped_3d
178                        && (has_active_ancestor(
179                            active_unclipped_3d_shapes,
180                            source_shape.wrap,
181                            source_shape.coords(),
182                            layer_entry.style_layer.index,
183                        ) || !queued_unclipped_3d.insert((
184                            source_shape.wrap,
185                            source_shape.coords(),
186                            layer_entry.style_layer.index,
187                        )))
188                    {
189                        continue;
190                    }
191                    let mask_group = if is_unclipped_3d {
192                        MaskGroup::Unmasked
193                    } else {
194                        MaskGroup::Vector
195                    };
196                    let draw_function: &'static dyn crate::render::render_phase::Draw<LayerItem> =
197                        match layer_entry.style_layer.type_.as_str() {
198                            // The heatmap plugin draws these into a density target instead.
199                            "heatmap" => continue,
200                            "line" => &DrawState::<LayerItem, DrawLineTiles>::INSTANCE,
201                            "circle" => &DrawState::<LayerItem, DrawCircleTiles>::INSTANCE,
202                            "fill-extrusion"
203                                if translucent_extrusion_layers.contains(&layer_id) =>
204                            {
205                                translucent_fronts.push(LayerItem {
206                                    projection: ProjectionBinding::View,
207                                    draw_function: &DrawState::<
208                                        LayerItem,
209                                        DrawExtrusionTranslucentFront,
210                                    >::INSTANCE,
211                                    index: layer_entry.style_layer.index,
212                                    generate_borders: false,
213                                    style_layer: layer_id,
214                                    entry_slot,
215                                    tile: Tile {
216                                        coords: layer_entry.coords,
217                                    },
218                                    source_shape: shape_data,
219                                    mask_group: MaskGroup::Unmasked,
220                                    run: None,
221                                });
222                                &DrawState::<LayerItem, DrawExtrusionTranslucentBack>::INSTANCE
223                            }
224                            "fill-extrusion" => {
225                                ensure_mask(mask_phase);
226                                if !extrusion_clears
227                                    .iter()
228                                    .any(|item| item.style_layer == layer_id)
229                                {
230                                    extrusion_clears.push(LayerItem {
231                                        projection: ProjectionBinding::View,
232                                        draw_function: &DrawState::<
233                                            LayerItem,
234                                            DrawExtrusionStencilClear,
235                                        >::INSTANCE,
236                                        index: layer_entry.style_layer.index,
237                                        generate_borders: false,
238                                        style_layer: layer_id,
239                                        entry_slot: UNSET_ENTRY_SLOT,
240                                        tile: Tile {
241                                            coords: layer_entry.coords,
242                                        },
243                                        source_shape: shape_data,
244                                        mask_group: MaskGroup::Vector,
245                                        run: None,
246                                    });
247                                }
248                                let color: &'static dyn crate::render::render_phase::Draw<
249                                    LayerItem,
250                                > = if pattern_layers.contains(&layer_id) {
251                                    &DrawState::<LayerItem, DrawExtrusionPatternColor>::INSTANCE
252                                } else {
253                                    &DrawState::<LayerItem, DrawExtrusionColor>::INSTANCE
254                                };
255                                extrusion_colors.push(LayerItem {
256                                    projection: ProjectionBinding::View,
257                                    draw_function: color,
258                                    index: layer_entry.style_layer.index,
259                                    generate_borders: false,
260                                    style_layer: layer_id,
261                                    entry_slot,
262                                    tile: Tile {
263                                        coords: layer_entry.coords,
264                                    },
265                                    source_shape: shape_data,
266                                    mask_group: MaskGroup::Vector,
267                                    run: None,
268                                });
269                                &DrawState::<LayerItem, DrawExtrusionDepth>::INSTANCE
270                            }
271                            "fill" if pattern_layers.contains(&layer_id) => {
272                                &DrawState::<LayerItem, DrawPatternTiles>::INSTANCE
273                            }
274                            _ => &DrawState::<LayerItem, DrawVectorTiles>::INSTANCE,
275                        };
276
277                    // Features that draw in the order of a sort key, across tiles, draw a run at
278                    // a time.
279                    let runs = (layer_entry.style_layer.type_ == "circle")
280                        .then(|| buffer_pool.sort_runs_by_id(layer_entry.coords, layer_id))
281                        .flatten();
282                    if let Some(runs) = runs.filter(|runs| !runs.is_empty()) {
283                        ensure_mask(mask_phase);
284                        for (key, range) in runs {
285                            layer_item_phase.add(LayerItem {
286                                projection: ProjectionBinding::View,
287                                draw_function: &DrawState::<LayerItem, DrawCircleTiles>::INSTANCE,
288                                index: layer_entry.style_layer.index,
289                                generate_borders: false,
290                                style_layer: layer_id,
291                                entry_slot,
292                                tile: Tile {
293                                    coords: layer_entry.coords,
294                                },
295                                source_shape: shape_data,
296                                mask_group: MaskGroup::Vector,
297                                run: Some(SortedRun {
298                                    key: *key,
299                                    start: range.start,
300                                    end: range.end,
301                                }),
302                            });
303                        }
304                        continue;
305                    }
306                    if mask_group == MaskGroup::Vector {
307                        ensure_mask(mask_phase);
308                    }
309                    layer_item_phase.add(LayerItem {
310                        projection: ProjectionBinding::View,
311                        draw_function,
312                        index: layer_entry.style_layer.index,
313                        generate_borders: false,
314                        style_layer: layer_id,
315                        entry_slot,
316                        tile: Tile {
317                            coords: layer_entry.coords,
318                        },
319                        source_shape: shape_data,
320                        mask_group,
321                        run: None,
322                    });
323                }
324            }
325        });
326    }
327
328    for item in extrusion_colors
329        .drain(..)
330        .chain(extrusion_clears.drain(..))
331        .chain(translucent_fronts.drain(..))
332    {
333        layer_item_phase.add(item);
334    }
335
336    mask_phase.sort_tail(mask_start);
337    layer_item_phase.sort_tail(layer_start);
338    scratch.cached_phases = Some(CachedVectorPhases {
339        key: cache_key,
340        masks: mask_phase.items()[mask_start..].to_vec(),
341        layers: layer_item_phase.items()[layer_start..].to_vec(),
342    });
343
344    Ok(())
345}

347fn has_active_ancestor(
348    active: &FxHashSet<(i32, crate::coords::WorldTileCoords, u32)>,
349    wrap: i32,
350    coords: crate::coords::WorldTileCoords,
351    layer_index: u32,
352) -> bool {
353    let mut current = coords.get_parent();
354    while let Some(ancestor) = current {
355        if active.contains(&(wrap, ancestor, layer_index)) {
356            return true;
357        }
358        current = ancestor.get_parent();
359    }
360    false
361}

370    #[test]
371    fn ancestor_without_matching_3d_layer_never_blanks_descendant_airspace() {
372        let overview = WorldTileCoords {
373            x: 0,
374            y: 0,
375            z: ZoomLevel::new(0),
376        };
377        let parent_z8 = WorldTileCoords {
378            x: 75,
379            y: 96,
380            z: ZoomLevel::new(8),
381        };
382        let child_z10 = WorldTileCoords {
383            x: 300,
384            y: 384,
385            z: ZoomLevel::new(10),
386        };
387        let mut active = FxHashSet::default();
388        // Overview 0/0/0 only holds layer 1 (e.g. coastline/ice), NOT layer 9 (3D airspace).
389        active.insert((0, overview, 1));
390        assert!(!has_active_ancestor(&active, 0, child_z10, 9));
391
392        // When parent z8 actively draws the same 3D layer 9, child z10 yields to avoid double alpha.
393        active.insert((0, parent_z8, 9));
394        assert!(has_active_ancestor(&active, 0, child_z10, 9));
395    }

397    #[test]
398    fn scratch_style_sets_reuse_cache_and_refresh_on_revision_or_zoom() {
399        let mut style: crate::style::Style = serde_json::from_str(
400            r#"{
401                "version": 8,
402                "sources": { "v": { "type": "vector", "tiles": ["https://example.invalid/{z}/{x}/{y}.pbf"] } },
403                "layers": [
404                    { "id": "hidden", "type": "fill", "source": "v", "source-layer": "s", "paint": { "fill-opacity": 0.0 } },
405                    { "id": "glass", "type": "fill-extrusion", "source": "v", "source-layer": "s", "paint": { "fill-extrusion-translucent": true } }
406                ]
407            }"#,
408        )
409        .expect("valid style");
410        let mut scratch = super::VectorQueueScratch::default();
411        scratch.sync_style_sets(&style, 12.0);
412        assert!(scratch.zero_opacity_layers.contains(&"hidden".into()));
413        assert!(scratch
414            .translucent_extrusion_layers
415            .contains(&"glass".into()));
416        assert!(scratch.unclipped_3d_layers.contains(&"glass".into()));
417
418        style.layers[0].paint = Some(crate::style::layer::LayerPaint::Fill(Default::default()));
419        // Without bumping revision or changing zoom, cached sets remain untouched.
420        scratch.sync_style_sets(&style, 12.0);
421        assert!(scratch.zero_opacity_layers.contains(&"hidden".into()));
422
423        // Bumping style revision refreshes the cached sets in a single pass.
424        style.bump_revision();
425        let d_after = scratch.sync_style_sets(&style, 12.0);
426        assert!(!scratch.zero_opacity_layers.contains(&"hidden".into()));
427        assert_ne!(d_after, 0);
428    }

432        let style: crate::style::Style = serde_json::from_str(
433            r#"{
