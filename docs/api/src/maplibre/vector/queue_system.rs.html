27pub fn queue_system(
28    MapContext {
29        style,
30        view_state,
31        world,
32        renderer,
33        ..
34    }: &mut MapContext,
35) -> SystemResult {
36    if renderer.settings.buffer_pools.is_empty() {
37        return Ok(());
38    }
39    let Some((
40        Initialized(tile_view_pattern),
41        Initialized(buffer_pool),
42        mask_phase,
43        layer_item_phase,
44    )) = world.resources.query_mut::<(
45        &mut Eventually<WgpuTileViewPattern>,
46        &mut Eventually<VectorBufferPool>,
47        &mut RenderPhase<TileMaskItem>,
48        &mut RenderPhase<LayerItem>,
49    )>()
50    else {
51        return Err(SystemError::Dependencies);
52    };
53
54    let buffer_pool_index = buffer_pool.index();
55    let zoom = view_state.zoom().value();
56    let uses_globe = style
57        .projection
58        .as_ref()
59        .is_some_and(|specification| specification.projection_type.uses_globe_rendering(zoom));
60
61    let pattern_layers: std::collections::HashSet<&str> = style
62        .layers
63        .iter()
64        .filter(|layer| {
65            layer.paint.as_ref().is_some_and(|paint| {
66                super::pattern::pattern_name(paint, view_state.style_zoom().value()).is_some()
67                    || super::pattern::per_feature_pattern(paint).is_some()
68            })
69        })
70        .map(|layer| layer.id.as_str())
71        .collect();
72    let translucent_extrusion_layers: std::collections::HashSet<&str> = style
73        .layers
74        .iter()
75        .filter(|layer| {
76            matches!(
77                &layer.paint,
78                Some(crate::style::layer::LayerPaint::FillExtrusion(paint))
79                    if paint.fill_extrusion_translucent == Some(true)
80            )
81        })
82        .map(|layer| layer.id.as_str())
83        .collect();
84    let unclipped_3d_layers: std::collections::HashSet<&str> = style
85        .layers
86        .iter()
87        .filter(|layer| {
88            translucent_extrusion_layers.contains(layer.id.as_str())
89                || matches!(
90                    super::structures::kind(layer),
91                    Some(super::structures::StructureKind::ElevatedLine)
92                )
93        })
94        .map(|layer| layer.id.as_str())
95        .collect();
96    let style_zoom = view_state.style_zoom().value();
97    let mut active_unclipped_3d_shapes: std::collections::HashSet<(
98        i32,
99        crate::coords::WorldTileCoords,
100        u32,
101    )> = std::collections::HashSet::new();
102    for view_tile in tile_view_pattern.iter() {
103        view_tile.render_kind(TileKind::Vector, |shape| {
104            if let Some(layer_entries) = buffer_pool_index.get_layers(shape.coords()) {
105                for layer_entry in layer_entries {
106                    if unclipped_3d_layers.contains(layer_entry.style_layer.id.as_str())
107                        && layer_entry.style_layer.is_visible_at(style_zoom)
108                    {
109                        active_unclipped_3d_shapes.insert((
110                            shape.wrap,
111                            shape.coords(),
112                            layer_entry.style_layer.index,
113                        ));
114                    }
115                }
116            }
117        });
118    }
119    // Every tile's depth is drawn before any tile's colour, so an extrusion that crosses tiles
120    // shows its nearest surface once instead of once per tile.
121    let mut extrusion_colors = Vec::new();
122    let mut extrusion_clears = Vec::new();
123    let mut translucent_fronts = Vec::new();
124    let mut queued_unclipped_3d: std::collections::HashSet<(
125        i32,
126        crate::coords::WorldTileCoords,
127        u32,
128    )> = std::collections::HashSet::new();
129    for view_tile in tile_view_pattern.iter() {
130        let coords = &view_tile.coords();
131        tracing::trace!("Drawing tile at {coords}");
132
133        // draw tile normal or the source e.g. parent or children
134        view_tile.render_kind(TileKind::Vector, |source_shape| {
135            if uses_globe {
136                mask_phase.add(TileMaskItem {
137                    projection: ProjectionBinding::View,
138                    draw_function: Box::new(DrawState::<TileMaskItem, DrawMasks>::new()),
139                    source_shape: source_shape.clone(),
140                    generate_borders: true,
141                });
142            }
143            mask_phase.add(TileMaskItem {
144                projection: ProjectionBinding::View,
145                draw_function: Box::new(DrawState::<TileMaskItem, DrawMasks>::new()),
146                source_shape: source_shape.clone(),
147                generate_borders: false,
148            });
149
150            if let Some(layer_entries) = buffer_pool_index.get_layers(source_shape.coords()) {
151                for layer_entry in layer_entries {
152                    if !layer_entry
153                        .style_layer
154                        .is_visible_at(view_state.style_zoom().value())
155                    {
156                        continue;
157                    }
158                    if super::pattern::names_missing_image(
159                        layer_entry.style_layer.paint.as_ref(),
160                        style,
161                        view_state.style_zoom().value(),
162                    ) {
163                        continue;
164                    }
165                    if unclipped_3d_layers.contains(layer_entry.style_layer.id.as_str())
166                        && (has_active_ancestor(
167                            &active_unclipped_3d_shapes,
168                            source_shape.wrap,
169                            source_shape.coords(),
170                            layer_entry.style_layer.index,
171                        ) || !queued_unclipped_3d.insert((
172                            source_shape.wrap,
173                            source_shape.coords(),
174                            layer_entry.style_layer.index,
175                        )))
176                    {
177                        continue;
178                    }
179                    let draw_function: Box<dyn crate::render::render_phase::Draw<LayerItem>> =
180                        match layer_entry.style_layer.type_.as_str() {
181                            // The heatmap plugin draws these into a density target instead.
182                            "heatmap" => continue,
183                            "line" => Box::new(DrawState::<LayerItem, DrawLineTiles>::new()),
184                            "circle" => Box::new(DrawState::<LayerItem, DrawCircleTiles>::new()),
185                            "fill-extrusion"
186                                if translucent_extrusion_layers
187                                    .contains(layer_entry.style_layer.id.as_str()) =>
188                            {
189                                translucent_fronts.push(LayerItem {
190                                    projection: ProjectionBinding::View,
191                                    draw_function: Box::new(DrawState::<
192                                        LayerItem,
193                                        DrawExtrusionTranslucentFront,
194                                    >::new()),
195                                    index: layer_entry.style_layer.index,
196                                    generate_borders: false,
197                                    style_layer: layer_entry.style_layer.id.clone(),
198                                    tile: Tile {
199                                        coords: layer_entry.coords,
200                                    },
201                                    source_shape: source_shape.clone(),
202                                    run: None,
203                                });
204                                Box::new(DrawState::<LayerItem, DrawExtrusionTranslucentBack>::new())
205                            }
206                            "fill-extrusion" => {
207                                extrusion_clears.push(LayerItem {
208                                    projection: ProjectionBinding::View,
209                                    draw_function: Box::new(DrawState::<
210                                        LayerItem,
211                                        DrawExtrusionClear,
212                                    >::new(
213                                    )),
214                                    index: layer_entry.style_layer.index,
215                                    generate_borders: false,
216                                    style_layer: layer_entry.style_layer.id.clone(),
217                                    tile: Tile {
218                                        coords: layer_entry.coords,
219                                    },
220                                    source_shape: source_shape.clone(),
221                                    run: None,
222                                });
223                                let color: Box<dyn crate::render::render_phase::Draw<LayerItem>> =
224                                    if pattern_layers.contains(layer_entry.style_layer.id.as_str())
225                                    {
226                                        Box::new(
227                                            DrawState::<LayerItem, DrawExtrusionPatternColor>::new(
228                                            ),
229                                        )
230                                    } else {
231                                        Box::new(DrawState::<LayerItem, DrawExtrusionColor>::new())
232                                    };
233                                extrusion_colors.push(LayerItem {
234                                    projection: ProjectionBinding::View,
235                                    draw_function: color,
236                                    index: layer_entry.style_layer.index,
237                                    generate_borders: false,
238                                    style_layer: layer_entry.style_layer.id.clone(),
239                                    tile: Tile {
240                                        coords: layer_entry.coords,
241                                    },
242                                    source_shape: source_shape.clone(),
243                                    run: None,
244                                });
245                                Box::new(DrawState::<LayerItem, DrawExtrusionDepth>::new())
246                            }
247                            "fill"
248                                if pattern_layers.contains(layer_entry.style_layer.id.as_str()) =>
249                            {
250                                Box::new(DrawState::<LayerItem, DrawPatternTiles>::new())
251                            }
252                            _ => Box::new(DrawState::<LayerItem, DrawVectorTiles>::new()),
253                        };
254
255                    // Features that draw in the order of a sort key, across tiles, draw a run at
256                    // a time.
257                    let runs = (layer_entry.style_layer.type_ == "circle")
258                        .then(|| {
259                            buffer_pool.sort_runs(layer_entry.coords, &layer_entry.style_layer.id)
260                        })
261                        .flatten();
262                    if let Some(runs) = runs.filter(|runs| !runs.is_empty()) {
263                        for (key, range) in runs {
264                            layer_item_phase.add(LayerItem {
265                                projection: ProjectionBinding::View,
266                                draw_function: Box::new(
267                                    DrawState::<LayerItem, DrawCircleTiles>::new(),
268                                ),
269                                index: layer_entry.style_layer.index,
270                                generate_borders: false,
271                                style_layer: layer_entry.style_layer.id.clone(),
272                                tile: Tile {
273                                    coords: layer_entry.coords,
274                                },
275                                source_shape: source_shape.clone(),
276                                run: Some(SortedRun {
277                                    key: *key,
278                                    range: range.clone(),
279                                }),
280                            });
281                        }
282                        continue;
283                    }
284                    layer_item_phase.add(LayerItem {
285                        projection: ProjectionBinding::View,
286                        draw_function,
287                        index: layer_entry.style_layer.index,
288                        generate_borders: false,
289                        style_layer: layer_entry.style_layer.id.clone(),
290                        tile: Tile {
291                            coords: layer_entry.coords,
292                        },
293                        source_shape: source_shape.clone(),
294                        run: None,
295                    });
296                }
297            }
298        });
299    }
300
301    for item in extrusion_colors
302        .into_iter()
303        .chain(extrusion_clears)
304        .chain(translucent_fronts)
305    {
306        layer_item_phase.add(item);
307    }
308
309    Ok(())
310}

312fn has_active_ancestor(
313    active: &std::collections::HashSet<(i32, crate::coords::WorldTileCoords, u32)>,
314    wrap: i32,
315    coords: crate::coords::WorldTileCoords,
316    layer_index: u32,
317) -> bool {
318    let mut current = coords.get_parent();
319    while let Some(ancestor) = current {
320        if active.contains(&(wrap, ancestor, layer_index)) {
321            return true;
322        }
323        current = ancestor.get_parent();
324    }
325    false
326}

328#[cfg(test)]
329mod tests {
330    use std::collections::HashSet;
331
332    use super::has_active_ancestor;
333    use crate::coords::{WorldTileCoords, ZoomLevel};
334
335    #[test]
336    fn ancestor_without_matching_3d_layer_never_blanks_descendant_airspace() {
337        let overview = WorldTileCoords {
338            x: 0,
339            y: 0,
340            z: ZoomLevel::new(0),
341        };
342        let parent_z8 = WorldTileCoords {
343            x: 75,
344            y: 96,
345            z: ZoomLevel::new(8),
346        };
347        let child_z10 = WorldTileCoords {
348            x: 300,
349            y: 384,
350            z: ZoomLevel::new(10),
351        };
352        let mut active = HashSet::new();
353        // Overview 0/0/0 only holds layer 1 (e.g. coastline/ice), NOT layer 9 (3D airspace).
354        active.insert((0, overview, 1));
355        assert!(!has_active_ancestor(&active, 0, child_z10, 9));
356
357        // When parent z8 actively draws the same 3D layer 9, child z10 yields to avoid double alpha.
358        active.insert((0, parent_z8, 9));
359        assert!(has_active_ancestor(&active, 0, child_z10, 9));
360    }
