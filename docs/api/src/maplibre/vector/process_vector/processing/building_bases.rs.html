1//! Unhides stepped `building` podium bases whose `building:part` pieces all start above ground.
2
3use geozero::mvt::tile;
4
5/// Clears `hide_3d` on building footprints whose interior `building:part` pieces all start
6/// above ground (`render_min_height > 0`), so stepped skyscrapers modeled with a `building`
7/// base podium and elevated `building:part` tiers do not float in mid-air.
8pub(super) fn unhide_unreplaced_building_bases(layer: &mut tile::Layer) {
9    let Some(hide_3d_key) = layer
10        .keys
11        .iter()
12        .position(|key| key == "hide_3d")
13        .and_then(|idx| u32::try_from(idx).ok())
14    else {
15        return;
16    };
17    let Some(min_height_key) = layer
18        .keys
19        .iter()
20        .position(|key| key == "render_min_height")
21        .and_then(|idx| u32::try_from(idx).ok())
22    else {
23        return;
24    };
25    let height_key = layer
26        .keys
27        .iter()
28        .position(|key| key == "render_height")
29        .and_then(|idx| u32::try_from(idx).ok());
30
31    struct PartInfo {
32        bbox: [i32; 4],
33        centroids: Vec<[f32; 2]>,
34        min_height: f64,
35    }
36
37    let mut parts = Vec::new();
38    let mut has_hide_3d = false;
39    for feature in &layer.features {
40        if feature.r#type != Some(3) {
41            continue;
42        }
43        if feature_bool_tag(feature, hide_3d_key, &layer.values) == Some(true) {
44            has_hide_3d = true;
45            continue;
46        }
47        let rings = decode_mvt_rings(&feature.geometry);
48        let Some(bbox) = rings_bbox(&rings) else {
49            continue;
50        };
51        let centroids: Vec<[f32; 2]> = rings
52            .iter()
53            .filter_map(|ring| {
54                if ring.is_empty() {
55                    return None;
56                }
57                let n = ring.len() as f32;
58                let sum = ring.iter().fold([0.0_f32, 0.0_f32], |acc, pt| {
59                    [acc[0] + pt[0] as f32, acc[1] + pt[1] as f32]
60                });
61                Some([sum[0] / n, sum[1] / n])
62            })
63            .collect();
64        let min_height = feature_number_tag(feature, min_height_key, &layer.values).unwrap_or(0.0);
65        parts.push(PartInfo {
66            bbox,
67            centroids,
68            min_height,
69        });
70    }
71    if !has_hide_3d || parts.is_empty() {
72        return;
73    }
74
75    for feature_index in 0..layer.features.len() {
76        let feature = &layer.features[feature_index];
77        if feature.r#type != Some(3)
78            || feature_bool_tag(feature, hide_3d_key, &layer.values) != Some(true)
79            || feature_number_tag(feature, min_height_key, &layer.values).unwrap_or(0.0) > 0.0
80        {
81            continue;
82        }
83        let rings = decode_mvt_rings(&feature.geometry);
84        let Some(bbox) = rings_bbox(&rings) else {
85            continue;
86        };
87        let mut min_part_base: Option<f64> = None;
88        for part in &parts {
89            if part.bbox[2] < bbox[0]
90                || part.bbox[0] > bbox[2]
91                || part.bbox[3] < bbox[1]
92                || part.bbox[1] > bbox[3]
93            {
94                continue;
95            }
96            let inside = part.centroids.iter().any(|&point| {
97                rings
98                    .iter()
99                    .filter(|ring| point_in_mvt_ring(ring, point))
100                    .count()
101                    % 2
102                    == 1
103            });
104            if inside {
105                min_part_base =
106                    Some(min_part_base.map_or(part.min_height, |cur| cur.min(part.min_height)));
107            }
108        }
109        let Some(min_base) = min_part_base.filter(|&base| base > 0.0) else {
110            continue;
111        };
112        remove_feature_tag(&mut layer.features[feature_index].tags, hide_3d_key);
113        if let Some(height_key) = height_key {
114            if let Some(height) =
115                feature_number_tag(&layer.features[feature_index], height_key, &layer.values)
116            {
117                if height > min_base {
118                    let val_idx = u32::try_from(layer.values.len()).unwrap_or(u32::MAX);
119                    layer.values.push(tile::Value {
120                        double_value: Some(min_base),
121                        ..Default::default()
122                    });
123                    set_feature_tag(&mut layer.features[feature_index].tags, height_key, val_idx);
124                }
125            }
126        }
127    }
128}

130fn feature_bool_tag(feature: &tile::Feature, key: u32, values: &[tile::Value]) -> Option<bool> {
131    for &[k, v] in feature.tags.as_chunks::<2>().0 {
132        if k == key {
133            return values.get(v as usize)?.bool_value;
134        }
135    }
136    None
137}
138
139fn feature_number_tag(feature: &tile::Feature, key: u32, values: &[tile::Value]) -> Option<f64> {
140    for &[k, v] in feature.tags.as_chunks::<2>().0 {
141        if k == key {
142            let v = values.get(v as usize)?;
143            return v
144                .double_value
145                .or_else(|| v.float_value.map(f64::from))
146                .or_else(|| v.int_value.map(|n| n as f64))
147                .or_else(|| v.uint_value.map(|n| n as f64))
148                .or_else(|| v.sint_value.map(|n| n as f64));
149        }
150    }
151    None
152}

154fn remove_feature_tag(tags: &mut Vec<u32>, key: u32) {
155    let mut kept = Vec::with_capacity(tags.len());
156    for pair in tags.as_chunks::<2>().0 {
157        if pair[0] != key {
158            kept.extend_from_slice(pair);
159        }
160    }
161    *tags = kept;
162}
163
164fn set_feature_tag(tags: &mut [u32], key: u32, value_index: u32) {
165    for pair in tags.as_chunks_mut::<2>().0 {
166        if pair[0] == key {
167            pair[1] = value_index;
168            return;
169        }
170    }
171}

173fn decode_mvt_rings(geometry: &[u32]) -> Vec<Vec<[i32; 2]>> {
174    let mut rings: Vec<Vec<[i32; 2]>> = Vec::new();
175    let (mut cx, mut cy) = (0_i32, 0_i32);
176    let mut idx = 0;
177    while idx < geometry.len() {
178        let cmd_int = geometry[idx];
179        idx += 1;
180        let (cmd, count) = (cmd_int & 7, (cmd_int >> 3) as usize);
181        if cmd == 1 || cmd == 2 {
182            for _ in 0..count {
183                if idx + 1 >= geometry.len() {
184                    return rings;
185                }
186                let raw_dx = geometry[idx];
187                let raw_dy = geometry[idx + 1];
188                idx += 2;
189                let dx = ((raw_dx >> 1) as i32) ^ -((raw_dx & 1) as i32);
190                let dy = ((raw_dy >> 1) as i32) ^ -((raw_dy & 1) as i32);
191                cx = cx.saturating_add(dx);
192                cy = cy.saturating_add(dy);
193                if cmd == 1 {
194                    rings.push(vec![[cx, cy]]);
195                } else if let Some(ring) = rings.last_mut() {
196                    ring.push([cx, cy]);
197                }
198            }
199        }
200    }
201    rings
202}

204fn rings_bbox(rings: &[Vec<[i32; 2]>]) -> Option<[i32; 4]> {
205    let mut iter = rings.iter().flatten();
206    let &[x0, y0] = iter.next()?;
207    let mut bbox = [x0, y0, x0, y0];
208    for &[x, y] in iter {
209        bbox[0] = bbox[0].min(x);
210        bbox[1] = bbox[1].min(y);
211        bbox[2] = bbox[2].max(x);
212        bbox[3] = bbox[3].max(y);
213    }
214    Some(bbox)
215}

217fn point_in_mvt_ring(ring: &[[i32; 2]], [x, y]: [f32; 2]) -> bool {
218    let mut inside = false;
219    for (a, b) in ring.iter().zip(ring.iter().cycle().skip(1)) {
220        let (ax, ay) = (a[0] as f32, a[1] as f32);
221        let (bx, by) = (b[0] as f32, b[1] as f32);
222        if (ay > y) != (by > y) && x < (bx - ax) * (y - ay) / (by - ay) + ax {
223            inside = !inside;
224        }
225    }
226    inside
227}
