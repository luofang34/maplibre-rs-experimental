7/// Decodes a tile, tolerating the zero padding some tile servers append: a zero tag is not a
8/// field, so a strict decoder rejects a tile that GL JS reads.
9pub(crate) fn decode_tile(
10    data: &[u8],
11) -> Result<geozero::mvt::Tile, Box<dyn std::error::Error + Send + Sync>> {
12    // A payload that is not a protobuf tile with layers is tried as a MapLibre Tile.
13    #[cfg(feature = "mlt")]
14    match decode_mvt(data) {
15        Ok(tile) if !tile.layers.is_empty() => Ok(tile),
16        Ok(tile) => Ok(crate::vector::mlt::decode(data).unwrap_or(tile)),
17        Err(error) => crate::vector::mlt::decode(data).map_err(|_| error),
18    }
19    #[cfg(not(feature = "mlt"))]
20    decode_mvt(data)
21}
22
23fn decode_mvt(data: &[u8]) -> Result<geozero::mvt::Tile, Box<dyn std::error::Error + Send + Sync>> {
24    geozero::mvt::Tile::decode(data).or_else(|error| {
25        let end = data
26            .iter()
27            .rposition(|byte| *byte != 0)
28            .map_or(0, |i| i + 1);
29        if end == data.len() {
30            return Err(error.into());
31        }
32        geozero::mvt::Tile::decode(&data[..end]).map_err(|_| error.into())
33    })
34}

36/// Decodes only the layers accepted by `layer_keys`, pruning MVT property tables to referenced
37/// keys where `layer_keys` returns `Some(Some(keys))` and optionally clipping features into a
38/// descendant tile's extent during decoding. Falls back cleanly for non-MVT or malformed tiles.
39pub(crate) fn decode_tile_for_layers_with_keys_rescaled<'a>(
40    data: &[u8],
41    rescale: Option<(
42        crate::coords::WorldTileCoords,
43        crate::coords::WorldTileCoords,
44    )>,
45    mut layer_keys: impl FnMut(&str) -> Option<Option<&'a rustc_hash::FxHashSet<String>>>,
46) -> Result<geozero::mvt::Tile, Box<dyn std::error::Error + Send + Sync>> {
47    if let Some(tile) =
48        mvt_slice::try_decode_mvt_filtered_with_keys_rescaled(data, rescale, &mut layer_keys)
49    {
50        return Ok(tile);
51    }
52    let mut tile = decode_tile(data)?;
53    if let Some((from, to)) = rescale {
54        crate::vector::request_system::rescale::into_descendant(&mut tile, from, to);
55    }
56    Ok(tile)
57}

59/// Decodes raw MVT/MLT bytes into tile-local [`crate::io::geometry_index::IndexedGeometry`]
60/// entries, optionally bringing ancestor MVT tiles into a descendant tile's extent on demand.
61pub(crate) fn index_mvt_bytes_rescaled(
62    mvt_bytes: &[u8],
63    rescale: Option<(
64        crate::coords::WorldTileCoords,
65        crate::coords::WorldTileCoords,
66    )>,
67    promote_id: Option<&crate::style::source::PromoteId>,
68    feature_state: Option<&std::sync::Arc<crate::style::feature_state::SourceFeatureState>>,
69) -> Vec<crate::io::geometry_index::IndexedGeometry<f64>> {
70    #[cfg(feature = "mlt")]
71    if !mvt_slice::looks_like_mvt(mvt_bytes) && crate::vector::mlt::is_valid_mlt(mvt_bytes) {
72        return crate::vector::mlt::index_mlt_bytes(mvt_bytes, promote_id, feature_state);
73    }
74    let Ok(mut tile) = decode_tile(mvt_bytes) else {
75        return Vec::new();
76    };
77    if let Some((from, to)) = rescale {
78        crate::vector::request_system::rescale::into_descendant(&mut tile, from, to);
79    }
80    mvt_slice::index_decoded_mvt_tile(&tile, promote_id, feature_state)
81}

83/// Decodes only the layers of `data` whose names satisfy `keep`, falling back to [`decode_tile`]
84/// when `data` uses non-MVT framing (such as MapLibre Tiles) or needs full wire error reporting.
85pub(crate) fn decode_tile_for_layers(
86    data: &[u8],
87    keep: impl Fn(&str) -> bool,
88) -> Result<geozero::mvt::Tile, Box<dyn std::error::Error + Send + Sync>> {
89    if let Some(tile) = mvt_slice::try_decode_mvt_filtered(data, &keep) {
90        return Ok(tile);
91    }
92    let mut tile = decode_tile(data)?;
93    tile.layers.retain(|mvt| keep(mvt.name.as_str()));
94    Ok(tile)
95}
