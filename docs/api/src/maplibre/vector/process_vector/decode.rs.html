1//! Container-level tile decoding (`MVT` with optional zero-padding and `MLT` fallback).
2
3use geozero::mvt::Message;
4
5use super::mvt_slice;
6
7/// Decodes a tile, tolerating the zero padding some tile servers append: a zero tag is not a
8/// field, so a strict decoder rejects a tile that GL JS reads.
9pub(crate) fn decode_tile(
10    data: &[u8],
11) -> Result<geozero::mvt::Tile, Box<dyn std::error::Error + Send + Sync>> {
12    // A payload that is not a protobuf tile with layers is tried as a MapLibre Tile.
13    #[cfg(feature = "mlt")]
14    match decode_mvt(data) {
15        Ok(tile) if !tile.layers.is_empty() => Ok(tile),
16        Ok(tile) => Ok(crate::vector::mlt::decode(data).unwrap_or(tile)),
17        Err(error) => crate::vector::mlt::decode(data).map_err(|_| error),
18    }
19    #[cfg(not(feature = "mlt"))]
20    decode_mvt(data)
21}
22
23fn decode_mvt(data: &[u8]) -> Result<geozero::mvt::Tile, Box<dyn std::error::Error + Send + Sync>> {
24    geozero::mvt::Tile::decode(data).or_else(|error| {
25        let end = data
26            .iter()
27            .rposition(|byte| *byte != 0)
28            .map_or(0, |i| i + 1);
29        if end == data.len() {
30            return Err(error.into());
31        }
32        geozero::mvt::Tile::decode(&data[..end]).map_err(|_| error.into())
33    })
34}
35
36/// Decodes only the layers of `data` whose names satisfy `keep`, falling back to [`decode_tile`]
37/// when `data` uses non-MVT framing (such as MapLibre Tiles) or needs full wire error reporting.
38pub(crate) fn decode_tile_for_layers(
39    data: &[u8],
40    keep: impl Fn(&str) -> bool,
41) -> Result<geozero::mvt::Tile, Box<dyn std::error::Error + Send + Sync>> {
42    if let Some(tile) = mvt_slice::try_decode_mvt_filtered(data, &keep) {
43        return Ok(tile);
44    }
45    let mut tile = decode_tile(data)?;
46    tile.layers.retain(|mvt| keep(mvt.name.as_str()));
47    Ok(tile)
48}
