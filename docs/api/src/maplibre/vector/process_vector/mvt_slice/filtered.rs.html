13/// Scans raw MVT protobuf wire tags (`Tile.layers = 3`, `Layer.name = 1`) and decodes only layers
14/// accepted by `layer_keys`:
15/// - `None`: skip the layer completely;
16/// - `Some(None)`: decode the full layer including all properties;
17/// - `Some(Some(keys))`: decode geometries and only the MVT keys present in `keys`.
18///
19/// When `rescale` is `Some((from, to))`, each decoded feature is clipped into `to`'s descendant
20/// extent before its referenced properties are materialized. Returns `None` when `data` is empty,
21/// non-MVT (e.g. MLT), or fails wire validation so the caller can fall back cleanly.
22pub(in crate::vector::process_vector) fn try_decode_mvt_filtered_with_keys_rescaled<'a>(
23    data: &[u8],
24    rescale: Option<(WorldTileCoords, WorldTileCoords)>,
25    mut layer_keys: impl FnMut(&str) -> Option<Option<&'a FxHashSet<String>>>,
26) -> Option<geozero::mvt::Tile> {
27    let end = data
28        .iter()
29        .rposition(|byte| *byte != 0)
30        .map_or(0, |idx| idx + 1);
31    let bytes = &data[..end];
32    if bytes.is_empty() {
33        return None;
34    }
35    let mut pos = 0;
36    let mut saw_mvt_layer = false;
37    let mut layers = Vec::new();
38    while pos < bytes.len() {
39        let tag = read_varint(bytes, &mut pos)?;
40        let field_num = tag >> 3;
41        let wire_type = tag & 0x07;
42        if field_num == 3 && wire_type == 2 {
43            let len = usize::try_from(read_varint(bytes, &mut pos)?).ok()?;
44            let slice_end = pos.checked_add(len)?;
45            let layer_bytes = bytes.get(pos..slice_end)?;
46            pos = slice_end;
47            let name = scan_mvt_layer_name(layer_bytes)?;
48            saw_mvt_layer = true;
49            if let Some(needed) = layer_keys(name) {
50                let layer = match needed {
51                    Some(keys) => decode_layer_pruned(layer_bytes, keys, rescale)?,
52                    None => {
53                        let mut layer = tile::Layer::decode(layer_bytes).ok()?;
54                        if let Some((from, to)) = rescale {
55                            let _ = into_descendant_layer(&mut layer, from, to);
56                        }
57                        layer
58                    }
59                };
60                layers.push(layer);
61            }
62        } else if field_num >= 16 {
63            skip_wire_value(bytes, &mut pos, wire_type)?;
64        } else {
65            return None;
66        }
67    }
68    saw_mvt_layer.then_some(geozero::mvt::Tile { layers })
69}

79fn read_bytes<'a>(bytes: &'a [u8], pos: &mut usize) -> Option<&'a [u8]> {
80    let len = usize::try_from(read_varint(bytes, pos)?).ok()?;
81    let end = pos.checked_add(len)?;
82    let slice = bytes.get(*pos..end)?;
83    *pos = end;
84    Some(slice)
85}

87fn scan_mvt_layer_extent(layer_bytes: &[u8]) -> Option<u32> {
88    let mut pos = 0;
89    let mut extent = None;
90    while pos < layer_bytes.len() {
91        let tag = read_varint(layer_bytes, &mut pos)?;
92        let field_num = tag >> 3;
93        let wire_type = tag & 0x07;
94        if field_num == 0 {
95            return None;
96        }
97        if field_num == 5 && wire_type == 0 {
98            extent = Some(u32::try_from(read_varint(layer_bytes, &mut pos)?).ok()?);
99        } else {
100            skip_wire_value(layer_bytes, &mut pos, wire_type)?;
101        }
102    }
103    Some(extent.unwrap_or(4096))
104}

106fn decode_layer_pruned(
107    layer_bytes: &[u8],
108    needed: &FxHashSet<String>,
109    rescale: Option<(WorldTileCoords, WorldTileCoords)>,
110) -> Option<tile::Layer> {
111    let frame = match rescale {
112        Some((from, to)) => {
113            let layer_extent = scan_mvt_layer_extent(layer_bytes)?;
114            DescendantFrame::for_extent(from, to, layer_extent)
115        }
116        None => None,
117    };
118    let mut name: Option<String> = None;
119    let mut version: Option<u32> = None;
120    let mut extent: Option<u32> = None;
121    let mut features: Vec<tile::Feature> = Vec::new();
122    let mut compact_keys: Vec<String> = Vec::with_capacity(needed.len());
123    let mut key_remap: Vec<u32> = Vec::new();
124    let mut value_spans: Vec<(usize, usize)> = Vec::new();
125
126    let mut pos = 0_usize;
127    while pos < layer_bytes.len() {
128        let tag = read_varint(layer_bytes, &mut pos)?;
129        let field_num = tag >> 3;
130        let wire_type = tag & 0x07;
131        match (field_num, wire_type) {
132            (1, 2) => {
133                let slice = read_bytes(layer_bytes, &mut pos)?;
134                name = Some(std::str::from_utf8(slice).ok()?.to_owned());
135            }
136            (2, 2) => {
137                let slice = read_bytes(layer_bytes, &mut pos)?;
138                let mut feature = tile::Feature::decode(slice).ok()?;
139                if !feature.tags.len().is_multiple_of(2) {
140                    return None;
141                }
142                if let Some(ref frame) = frame {
143                    if !frame.cut(&mut feature) {
144                        continue;
145                    }
146                }
147                features.push(feature);
148            }
149            (3, 2) => {
150                let slice = read_bytes(layer_bytes, &mut pos)?;
151                let key = std::str::from_utf8(slice).ok()?;
152                if needed.contains(key) {
153                    let mapped = u32::try_from(compact_keys.len()).ok()?;
154                    compact_keys.push(key.to_owned());
155                    key_remap.push(mapped);
156                } else {
157                    key_remap.push(u32::MAX);
158                }
159            }
160            (4, 2) => {
161                let len = usize::try_from(read_varint(layer_bytes, &mut pos)?).ok()?;
162                let start = pos;
163                let end = pos.checked_add(len)?;
164                if end > layer_bytes.len() {
165                    return None;
166                }
167                pos = end;
168                value_spans.push((start, end));
169            }
170            (5, 0) => {
171                extent = Some(u32::try_from(read_varint(layer_bytes, &mut pos)?).ok()?);
172            }
173            (15, 0) => {
174                version = Some(u32::try_from(read_varint(layer_bytes, &mut pos)?).ok()?);
175            }
176            (1 | 2 | 3 | 4 | 5 | 15, _) | (0, _) => return None,
177            _ => {
178                skip_wire_value(layer_bytes, &mut pos, wire_type)?;
179            }
180        }
181    }
182
183    let name = name?;
184    let version = version?;
185
186    // First compact tags using remapped key indices, temporarily keeping original value indices.
187    let mut val_remap = vec![u32::MAX; value_spans.len()];
188    for feature in &mut features {
189        if !feature.tags.len().is_multiple_of(2) {
190            return None;
191        }
192        let mut write = 0_usize;
193        for read in (0..feature.tags.len()).step_by(2) {
194            let k_idx = feature.tags[read] as usize;
195            let orig_v_idx = feature.tags[read + 1];
196            let &mapped_k = key_remap.get(k_idx)?;
197            let used_slot = val_remap.get_mut(orig_v_idx as usize)?;
198            if mapped_k != u32::MAX {
199                *used_slot = 0;
200                feature.tags[write] = mapped_k;
201                feature.tags[write + 1] = orig_v_idx;
202                write += 2;
203            }
204        }
205        feature.tags.truncate(write);
206    }
207
208    // Decode only the `tile::Value` entries actually referenced by retained tags.
209    let mut compact_values = Vec::new();
210    for (orig_idx, (start, end)) in value_spans.into_iter().enumerate() {
211        if val_remap[orig_idx] == 0 {
212            val_remap[orig_idx] = u32::try_from(compact_values.len()).ok()?;
213            compact_values.push(tile::Value::decode(&layer_bytes[start..end]).ok()?);
214        }
215    }
216
217    // Rewrite kept value indices into the compacted value table.
218    for feature in &mut features {
219        for slot in (1..feature.tags.len()).step_by(2) {
220            let orig_v_idx = feature.tags[slot] as usize;
221            feature.tags[slot] = val_remap[orig_v_idx];
222        }
223    }
224
225    Some(tile::Layer {
226        version,
227        name,
228        features,
229        keys: compact_keys,
230        values: compact_values,
231        extent,
232    })
233}

293    fn selective_decode_prunes_unused_keys_and_values() {
294        let bytes = encode_test_tile();
295        let mut needed = FxHashSet::default();
296        needed.insert("name:en".to_owned());
297
298        let decoded = try_decode_mvt_filtered_with_keys(&bytes, |name| match name {
299            "place" => Some(Some(&needed)),
300            _ => None,
301        })
302        .expect("selective decode succeeds");
303
304        assert_eq!(decoded.layers.len(), 1);
305        let layer = &decoded.layers[0];
306        assert_eq!(layer.name, "place");
307        assert_eq!(layer.keys, vec!["name:en"]);
308        assert_eq!(layer.values.len(), 1);
309        assert_eq!(layer.values[0].string_value.as_deref(), Some("Paris"));
310        assert_eq!(layer.features[0].tags, vec![0, 0]);
311    }

348    fn selective_rescaled_decode_matches_full_decode_then_into_descendant() {
349        use crate::{coords::ZoomLevel, vector::request_system::rescale::into_descendant};
350        let mut inside_city = tile::Feature {
351            id: Some(1),
352            tags: vec![0, 0, 1, 1, 2, 2],
353            r#type: Some(1),
354            geometry: vec![9, 1000, 1000],
355        };
356        // Wait: MVT zigzag encoding: unsigned n -> (n >> 1) ^ -(n & 1). (500 << 1) = 1000 -> (500, 500) in NW quadrant (0..2048).
357        inside_city.geometry = vec![9, 1000, 1000];
358        let outside_point = tile::Feature {
359            id: Some(2),
360            tags: vec![0, 3, 1, 4, 2, 5],
361            r#type: Some(1),

366            id: Some(3),
367            tags: vec![0, 3, 1, 1],
368            r#type: Some(2),
369            // MoveTo(0, 500) -> LineTo(+3000, 0) in zigzag: (9, 0, 1000, 10, 6000, 0)
370            geometry: vec![9, 0, 1000, 10, 6000, 0],
371        };
372        let bytes = geozero::mvt::Tile {
373            layers: vec![tile::Layer {
374                version: 2,
375                name: "place".to_owned(),
376                features: vec![inside_city, outside_point, crossing_line],
377                keys: vec![
378                    "class".to_owned(),
379                    "name:en".to_owned(),
380                    "name:ja".to_owned(),
381                ],
382                values: vec![

384                        string_value: Some("city".to_owned()),
