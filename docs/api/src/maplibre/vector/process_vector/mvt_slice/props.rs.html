13/// Populates `properties` with typed filter values decoded from `feature.tags`, clearing any
14/// previous entries while reusing the map allocation.
15pub(crate) fn fill_feature_properties(
16    layer: &tile::Layer,
17    feature: &tile::Feature,
18    properties: &mut FeatureProperties,
19) {
20    properties.clear();
21    for pair in feature.tags.chunks(2) {
22        let [key_index, value_index] = pair else {
23            continue;
24        };
25        let (Some(key), Some(value)) = (
26            layer.keys.get(*key_index as usize),
27            layer.values.get(*value_index as usize),
28        ) else {
29            continue;
30        };
31        let value = if let Some(text) = &value.string_value {
32            Value::String(text.clone())
33        } else if let Some(number) = value.float_value {
34            Value::Number(f64::from(number))
35        } else if let Some(number) = value.double_value {
36            Value::Number(number)
37        } else if let Some(number) = value.int_value {
38            Value::Number(number as f64)
39        } else if let Some(number) = value.uint_value {
40            Value::Number(number as f64)
41        } else if let Some(number) = value.sint_value {
42            Value::Number(number as f64)
43        } else if let Some(flag) = value.bool_value {
44            Value::Bool(flag)
45        } else {
46            continue;
47        };
48        properties.insert(key.clone(), value);
49    }
50}

52/// Pre-decodes every feature's typed properties in `layer` into shared [`Arc`] maps unless
53/// `state_ctx` is absent and distinct non-empty tag combinations exceed `max_unique` (returning
54/// `None` so high-cardinality layers stream per-feature with a single scratch map).
55pub(in crate::vector::process_vector) fn decode_layer_properties_bounded(
56    layer: &tile::Layer,
57    state_ctx: Option<&FeatureStateSyncContext>,
58    max_unique: usize,
59) -> Option<Vec<Arc<FeatureProperties>>> {
60    let mut out = Vec::with_capacity(layer.features.len());
61    if let Some(ctx) = state_ctx {
62        for feature in &layer.features {
63            if !tags_are_valid(layer, &feature.tags) {
64                return None;
65            }
66            let mut properties = FeatureProperties::default();
67            fill_feature_properties(layer, feature, &mut properties);
68            crate::style::feature_state::sync_properties_feature_state(
69                &mut properties,
70                feature.id,
71                ctx.promote_property.as_deref(),
72                ctx.source_state.as_deref(),
73                ctx.source_layer.as_deref(),
74            );
75            if properties.is_empty() {
76                out.push(Arc::clone(empty_feature_properties()));
77            } else {
78                out.push(Arc::new(properties));
79            }
80        }
81        return Some(out);
82    }
83
84    let empty = empty_feature_properties();
85    let mut memo: FxHashMap<&[u32], Arc<FeatureProperties>> = FxHashMap::default();
86    for feature in &layer.features {
87        if feature.tags.is_empty() {
88            out.push(Arc::clone(empty));
89            continue;
90        }
91        if let Some(existing) = memo.get(feature.tags.as_slice()) {
92            out.push(Arc::clone(existing));
93            continue;
94        }
95        if memo.len() >= max_unique || !tags_are_valid(layer, &feature.tags) {
96            return None;
97        }
98        let mut properties = FeatureProperties::default();
99        fill_feature_properties(layer, feature, &mut properties);
100        let arc = if properties.is_empty() {
101            Arc::clone(empty)
102        } else {
103            Arc::new(properties)
104        };
105        memo.insert(feature.tags.as_slice(), Arc::clone(&arc));
106        out.push(arc);
107    }
108    Some(out)
109}

111/// Pre-decodes every feature's typed properties in `layer` once into `Arc<FeatureProperties>`
112/// (including active `feature-state` entries), memoizing identical MVT tag slices so repeated
113/// attribute combinations share a single heap map. Returns `None` if any feature's tag table is
114/// malformed so callers can fall back to `geozero::mvt::process` error reporting.
115pub(in crate::vector::process_vector) fn decode_layer_properties(
116    layer: &tile::Layer,
117    state_ctx: Option<&FeatureStateSyncContext>,
118) -> Option<Vec<Arc<FeatureProperties>>> {
119    decode_layer_properties_bounded(layer, state_ctx, usize::MAX)
120}

122fn tags_are_valid(layer: &tile::Layer, tags: &[u32]) -> bool {
123    let (pairs, remainder) = tags.as_chunks::<2>();
124    if !remainder.is_empty() {
125        return false;
126    }
127    for &[key_idx, val_idx] in pairs {
128        if layer.keys.get(key_idx as usize).is_none() {
129            return false;
130        }
131        let Some(val) = layer.values.get(val_idx as usize) else {
132            return false;
133        };
134        let has_val = val.string_value.is_some()
135            || val.float_value.is_some()
136            || val.double_value.is_some()
137            || val.int_value.is_some()
138            || val.uint_value.is_some()
139            || val.sint_value.is_some()
140            || val.bool_value.is_some();
141        if !has_val {
142            return false;
143        }
144    }
145    true
146}
