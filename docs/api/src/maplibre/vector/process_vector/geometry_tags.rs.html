1//! Gives each feature of a tile layer its geometry in longitude and latitude, for layers with
2//! `within` or `distance`, and its id and geometry type, for layers whose paint or layout reads

16/// Appends the id and geometry type tags to every feature of `layer`; a feature without an id
17/// gets none, so `["id"]` is null for it.
18pub(super) fn add_identity_tags(layer: &mut tile::Layer) {
19    let key = |layer: &mut tile::Layer, name: &str| {
20        layer.keys.push(name.to_owned());
21        u32::try_from(layer.keys.len() - 1).unwrap_or(u32::MAX)
22    };
23    let (id_key, type_key) = (key(layer, ID_PROPERTY), key(layer, GEOMETRY_TYPE_PROPERTY));
24    // One value per geometry type, shared by the features of that type.
25    let types = [1, 2, 3, 0].map(|kind| {
26        layer.values.push(tile::Value {
27            string_value: Some(GeometryType::from_mvt(kind).name().to_owned()),
28            ..Default::default()
29        });
30        (
31            kind,
32            u32::try_from(layer.values.len() - 1).unwrap_or(u32::MAX),
33        )
34    });
35    for feature in &mut layer.features {
36        let kind = feature.r#type.unwrap_or_default();
37        let type_value = types
38            .iter()
39            .find(|(each, _)| *each == kind)
40            .map_or(types[3].1, |(_, value)| *value);
41        feature.tags.extend([type_key, type_value]);
42        if let Some(id) = feature.id {
43            layer.values.push(tile::Value {
44                uint_value: Some(id),
45                ..Default::default()
46            });
47            let value = u32::try_from(layer.values.len() - 1).unwrap_or(u32::MAX);
48            feature.tags.extend([id_key, value]);
49        }
50    }
51}

53/// Appends the geometry tag to every feature of `layer`.
54pub(super) fn add_geometry_tags(layer: &mut tile::Layer, coords: WorldTileCoords) {
55    let extent = f64::from(layer.extent.unwrap_or(4096)).max(1.0);
56    let key = u32::try_from(layer.keys.len()).unwrap_or(u32::MAX);
57    layer.keys.push(GEOMETRY_PROPERTY.to_owned());
58    for feature in &mut layer.features {
59        let Ok(geometry) = feature.to_geo() else {
60            continue;
61        };
62        let text = lonlat(&geometry, coords, extent).to_geojson().to_string();
63        let value = u32::try_from(layer.values.len()).unwrap_or(u32::MAX);
64        layer.values.push(tile::Value {
65            string_value: Some(text),
66            ..Default::default()
67        });
68        feature.tags.extend([key, value]);
69    }
70}

72fn lonlat(geometry: &Geo<f64>, coords: WorldTileCoords, extent: f64) -> Geometry {
73    let tiles = 2_f64.powi(i32::from(u8::from(coords.z)));
74    let convert = |c: &geo_types::Coord<f64>| {
75        let x = (f64::from(coords.x) + c.x / extent) / tiles;
76        let y = (f64::from(coords.y) + c.y / extent) / tiles;
77        let longitude = x * 360.0 - 180.0;
78        let latitude = (std::f64::consts::PI * (1.0 - 2.0 * y))
79            .sinh()
80            .atan()
81            .to_degrees();
82        [longitude, latitude]
83    };
84    let mut out = Geometry::default();
85    collect(geometry, &convert, &mut out);
86    out
87}

89fn collect(
90    geometry: &Geo<f64>,
91    convert: &dyn Fn(&geo_types::Coord<f64>) -> [f64; 2],
92    out: &mut Geometry,
93) {
94    let line = |line: &geo_types::LineString<f64>| line.0.iter().map(convert).collect::<Vec<_>>();
95    let polygon = |polygon: &geo_types::Polygon<f64>| {
96        std::iter::once(polygon.exterior())
97            .chain(polygon.interiors())
98            .map(line)
99            .collect::<Vec<_>>()
100    };
101    match geometry {
102        Geo::Point(point) => out.points.push(convert(&point.0)),
103        Geo::MultiPoint(points) => out.points.extend(points.0.iter().map(|p| convert(&p.0))),
104        Geo::Line(segment) => out
105            .lines
106            .push(vec![convert(&segment.start), convert(&segment.end)]),
107        Geo::LineString(string) => out.lines.push(line(string)),
108        Geo::MultiLineString(strings) => out.lines.extend(strings.0.iter().map(line)),
109        Geo::Polygon(p) => out.polygons.push(polygon(p)),
110        Geo::MultiPolygon(polygons) => out.polygons.extend(polygons.0.iter().map(polygon)),
111        Geo::GeometryCollection(parts) => {
112            parts.0.iter().for_each(|part| collect(part, convert, out))
113        }
114        Geo::Rect(rect) => out.polygons.push(polygon(&rect.to_polygon())),
115        Geo::Triangle(triangle) => out.polygons.push(polygon(&triangle.to_polygon())),
116    }
117}
