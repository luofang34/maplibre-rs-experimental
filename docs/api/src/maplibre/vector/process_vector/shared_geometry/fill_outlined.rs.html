20/// Returns `true` when an outlined/antialiased fill layer without filters, patterns, sort keys,
21/// or per-feature paint can share a single tessellation with sibling layers on the same source.
22pub(super) fn qualifies_outlined_fill(
23    style: &StyleLayer,
24    fill: &FillPaint,
25    style_zoom: f64,
26) -> bool {
27    let Some(paint) = style.paint.as_ref() else {
28        return false;
29    };
30    style.filter.is_none()
31        && sort_key_of(style).is_none()
32        && fill.fill_pattern.is_none()
33        && crate::vector::pattern::per_feature_pattern(paint).is_none()
34        && fill
35            .fill_color
36            .as_ref()
37            .is_none_or(|prop| prop.is_feature_constant())
38        && fill
39            .fill_opacity
40            .as_ref()
41            .is_none_or(|prop| prop.is_feature_constant())
42        && fill.outline_color().is_some_and(|outline| {
43            outline.is_feature_constant() && outline.evaluate_at_zoom(style_zoom).is_some()
44        })
45}

47/// Tessellates `layer` once for a group of qualifying outlined fill layers and emits a bucket
48/// per sibling layer, sharing pre-packed GPU buffers zero-copy via `Arc`.
49pub(super) fn process_shared_fill_outlined<T: VectorTransferables, C: Context>(
50    layer: &tile::Layer,
51    group: &[&StyleLayer],
52    request: &VectorTileRequest,
53    context: &mut ProcessVectorContext<T, C>,
54) -> Result<(), ProcessVectorError> {
55    let Some(first_style) = group.first().copied() else {
56        return Ok(());
57    };
58    let Some(first_paint) = first_style.paint.as_ref() else {
59        return Ok(());
60    };
61    let allow_prepack = context.allow_prepack_for_style(request.coords, first_style, false, false);
62    let Some(mut tessellator) = tessellate_readable(layer, request, |slice| {
63        let mut tessellator: ZeroTessellator<IndexDataType> =
64            processing::tessellator(first_paint, request);
65        tessellator.coordinate_scale = extent_scale(layer);
66        tessellator.omit_feature_colors = true;
67        tessellator.record_outline_entries = true;
68        if allow_prepack {
69            tessellator.maybe_enable_streaming_prepack(first_style.type_.as_str());
70        }
71        process_mvt_slice(layer, slice, true, &mut tessellator)?;
72        Ok(tessellator)
73    }) else {
74        for _ in group {
75            context.layer_missing(&request.coords, &layer.name)?;
76        }
77        return Ok(());
78    };
79
80    let mut feature_indices = std::mem::take(&mut tessellator.feature_indices);
81    let outline_entries = std::mem::take(&mut tessellator.outline_entries);
82    feature_indices.shrink_to_fit();
83    let shared_buffer = tessellator.into_over_aligned_buffer();
84    let header = layer_header(layer);
85    let style_zoom = request.style_zoom();
86    let mut shared_buffer = Some(shared_buffer);
87    let mut feature_indices = Some(feature_indices);
88    let mut header = Some(header);
89
90    for (idx, &style) in group.iter().enumerate() {
91        let is_last = idx + 1 == group.len();
92        let buffer = if is_last {
93            shared_buffer
94                .take()
95                .unwrap_or_else(OverAlignedVertexBuffer::empty)
96        } else {
97            shared_buffer
98                .as_ref()
99                .cloned()
100                .unwrap_or_else(OverAlignedVertexBuffer::empty)
101        };
102        let indices = if is_last {
103            feature_indices.take().unwrap_or_default()
104        } else {
105            feature_indices.as_ref().cloned().unwrap_or_default()
106        };
107        let layer_data = if is_last {
108            header.take().unwrap_or_else(|| layer_header(layer))
109        } else {
110            header.clone().unwrap_or_else(|| layer_header(layer))
111        };
112        let feature_colors = eval_layer_colors(style, style_zoom, &outline_entries);
113        context.layer_tessellation_finished(
114            &request.coords,
115            buffer,
116            indices,
117            (feature_colors, Vec::new(), Vec::new()),
118            layer_data,
119            style,
120        )?;
121    }
122    Ok(())
123}

125fn eval_layer_colors(
126    style: &StyleLayer,
127    style_zoom: f64,
128    outline_entries: &[bool],
129) -> Vec<[f32; 4]> {
130    if outline_entries.is_empty() {
131        return Vec::new();
132    }
133    #[cfg(any(target_arch = "wasm32", not(test)))]
134    if crate::vector::upload_system::paint::uniform_color(style, style_zoom).is_some() {
135        return Vec::new();
136    }
137    let Some(LayerPaint::Fill(fill)) = style.paint.as_ref() else {
138        return Vec::new();
139    };
140    let opacity = fill
141        .fill_opacity
142        .as_ref()
143        .and_then(|prop| prop.evaluate_at_zoom(style_zoom))
144        .unwrap_or(1.0)
145        .clamp(0.0, 1.0);
146    let fill_base = fill
147        .fill_color
148        .as_ref()
149        .and_then(|prop| prop.evaluate_at_zoom(style_zoom))
150        .map_or([0.0, 0.0, 0.0, 1.0], |c| [c.r, c.g, c.b, c.a]);
151    let fill_rgba = [
152        fill_base[0],
153        fill_base[1],
154        fill_base[2],
155        fill_base[3] * opacity,
156    ];
157    let out_base = fill
158        .outline_color()
159        .and_then(|prop| prop.evaluate_at_zoom(style_zoom))
160        .map_or([0.0, 0.0, 0.0, 1.0], |c| [c.r, c.g, c.b, c.a]);
161    let outline_rgba = [out_base[0], out_base[1], out_base[2], out_base[3] * opacity];
162    outline_entries
163        .iter()
164        .map(|&is_outline| if is_outline { outline_rgba } else { fill_rgba })
165        .collect()
166}

168fn tessellate_readable<T>(
169    layer: &tile::Layer,
170    request: &VectorTileRequest,
171    mut build: impl FnMut(Option<&[usize]>) -> geozero::error::Result<T>,
172) -> Option<T> {
173    let error = match build(None) {
174        Ok(tessellator) => return Some(tessellator),
175        Err(error) => error,
176    };
177    let (readable, dropped) = retain_readable_indices(layer, None);
178    if dropped == 0 {
179        tracing::error!(coords = %request.coords, layer = %layer.name, ?error, "vector tessellation failed");
180        return None;
181    }
182    tracing::warn!(coords = %request.coords, layer = %layer.name, dropped, ?error, "features whose geometry cannot be read are left out");
183    match build(Some(&readable)) {
184        Ok(tessellator) => Some(tessellator),
185        Err(error) => {
186            tracing::error!(coords = %request.coords, layer = %layer.name, ?error, "vector tessellation failed");
187            None
188        }
189    }
190}
