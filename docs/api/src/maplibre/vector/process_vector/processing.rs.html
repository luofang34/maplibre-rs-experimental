13pub(super) fn process_layer<T: VectorTransferables, C: Context>(
14    source_layer: Option<&tile::Layer>,
15    layer_name: &str,
16    predecoded_props: Option<&[Arc<FeatureProperties>]>,
17    style: &StyleLayer,
18    request: &VectorTileRequest,
19    context: &mut ProcessVectorContext<T, C>,
20    atlas: std::sync::Arc<crate::sdf::assets::SymbolAtlas>,
21) -> Result<(), ProcessVectorError> {
22    let empty_layer;
23    let mut layer: Cow<'_, tile::Layer> = match source_layer {
24        Some(existing) => Cow::Borrowed(existing),
25        None => {
26            empty_layer = tile::Layer {
27                version: 2,
28                name: layer_name.to_owned(),
29                extent: Some(4096),
30                ..Default::default()
31            };
32            Cow::Borrowed(&empty_layer)
33        }
34    };
35    if crate::style::expression::layer_reads_geometry(style) && !layer.features.is_empty() {
36        super::geometry_tags::add_geometry_tags(layer.to_mut(), request.coords);
37    }
38    if crate::style::expression::layer_reads_identity(style) && !layer.features.is_empty() {
39        super::geometry_tags::add_identity_tags(layer.to_mut());
40    }
41    if matches!(style.paint, Some(LayerPaint::FillExtrusion(_)))
42        && layer.keys.iter().any(|key| key == "hide_3d")
43    {
44        unhide_unreplaced_building_bases(layer.to_mut());
45    }
46    let shared_props = matches!(layer, Cow::Borrowed(_))
47        .then_some(predecoded_props)
48        .flatten();
49    let selected = match &style.filter {
50        Some(raw_filter) => match Filter::parse(raw_filter) {
51            Ok(filter) => {
52                let state_ctx = context.state_context_for_layer(&layer.name);
53                Some(select_filtered_features(
54                    layer.as_ref(),
55                    &filter,
56                    request.style_zoom(),
57                    state_ctx.as_ref(),
58                    shared_props,
59                ))
60            }
61            Err(error) => {
62                // Rendering either all features or none silently would conceal an invalid style.
63                tracing::error!(layer = %style.id,%error,"unsupported filter; the layer renders nothing");
64                context.layer_missing(&request.coords, &layer.name)?;
65                return Ok(());
66            }
67        },
68        None => None,
69    };
70    match style.paint.as_ref() {
71        Some(
72            paint @ (LayerPaint::Line(_)
73            | LayerPaint::Fill(_)
74            | LayerPaint::FillExtrusion(_)
75            | LayerPaint::Circle(_)
76            | LayerPaint::Heatmap(_)),
77        ) => vector_layer(
78            layer.as_ref(),
79            selected.as_deref(),
80            shared_props,
81            style,
82            paint,
83            request,
84            context,
85        ),
86        Some(LayerPaint::Symbol(paint)) => symbol_layer(
87            layer.as_ref(),
88            selected.as_deref(),
89            style,
90            paint,
91            request,
92            context,
93            atlas,
94        ),
95        _ => {
96            tracing::warn!(layer = %style.id,"unhandled vector style layer type");
97            Ok(())
98        }
99    }
100}

102fn tessellator(paint: &LayerPaint, request: &VectorTileRequest) -> ZeroTessellator<IndexDataType> {
103    let zoom = u8::from(request.coords.z);
104    let style_zoom = request.style_zoom();
105    let granularity = match paint {
106        LayerPaint::Fill(_) => granularity_for_zoom(128, 2, zoom),
107        LayerPaint::Line(_) => granularity_for_zoom(512, 0, zoom),
108        _ => 1,
109    };
110    let mut tessellator = match paint {
111        LayerPaint::Circle(circle) => ZeroTessellator::default().with_circles(CircleOptions {
112            // A circle lying on the map follows the globe's curve.
113            grid: request.projection.uses_globe_rendering(style_zoom)
114                && circle.circle_pitch_alignment == crate::style::circle::CirclePitchAlignment::Map,
115            ..CircleOptions::for_paint(circle, style_zoom)
116        }),
117        LayerPaint::Heatmap(heatmap) => ZeroTessellator::default().with_circles(CircleOptions {
118            grid: request.projection.uses_globe_rendering(style_zoom),
119            ..CircleOptions::for_heatmap(heatmap, style_zoom)
120        }),
121        LayerPaint::FillExtrusion(extrusion) => {
122            let tessellator = ZeroTessellator::default().with_extrusion(
123                ExtrusionOptions::for_paint(extrusion, request.overscaling()),
124            );
125            let min_grid = if extrusion.may_use_vertex_agl() {
126                32
127            } else {
128                2
129            };
130            if request.projection.uses_globe_rendering(style_zoom) {
131                // Roofs and the edges of walls follow the sphere as fills and lines do.
132                let last_tile = i64::from(crate::coords::ZOOM_BOUNDS[usize::from(zoom)]) - 1;
133                tessellator.with_globe_subdivision(
134                    granularity_for_zoom(128, min_grid, zoom),
135                    zoom == 0,
136                    request.coords.y == 0,
137                    i64::from(request.coords.y) == last_tile,
138                )
139            } else if extrusion.may_use_vertex_agl() {
140                tessellator.with_globe_subdivision(32, false, false, false)
141            } else {
142                tessellator
143            }
144        }
145        _ if request.projection.uses_globe_rendering(style_zoom) => {
146            let last_tile = i64::from(crate::coords::ZOOM_BOUNDS[usize::from(zoom)]) - 1;
147            ZeroTessellator::default().with_globe_subdivision(
148                granularity,
149                zoom == 0,
150                request.coords.y == 0,
151                i64::from(request.coords.y) == last_tile,
152            )
153        }
154        _ => ZeroTessellator::default(),
155    }
156    .with_feature_opacity(paint.opacity(), style_zoom);
157    match paint {
158        // An image repeated over a fill takes nothing from the fill's colour but its opacity.
159        LayerPaint::Fill(paint) if paint.fill_pattern.is_some() => {
160            tessellator.fallback_color = [1.0; 4];
161            tessellator.pattern_property =
162                crate::vector::pattern::per_feature_pattern(&LayerPaint::Fill(paint.clone()));
163        }
164        LayerPaint::Fill(paint) => {
165            tessellator.style_property = paint.fill_color.clone();
166            tessellator.outline_property = paint.outline_color();
167        }
168        LayerPaint::FillExtrusion(paint) if paint.fill_extrusion_pattern.is_some() => {
169            tessellator.fallback_color = [1.0; 4];
170            tessellator.pattern_property = crate::vector::pattern::per_feature_pattern(
171                &LayerPaint::FillExtrusion(paint.clone()),
172            );
173        }
174        LayerPaint::FillExtrusion(paint) => {
175            tessellator.style_property = paint.fill_extrusion_color.clone()
176        }
177        LayerPaint::Circle(paint) => tessellator.style_property = paint.circle_color.clone(),
178        LayerPaint::Line(paint) => {
179            tessellator.pattern_property =
180                crate::vector::pattern::per_feature_pattern(&LayerPaint::Line(paint.clone()));
181            tessellator.style_property = paint.line_color.clone();
182            tessellator.is_line_layer = true;
183            tessellator.line_gradient = paint.line_gradient.is_some();
184            tessellator.sharp_corner_offset =
185                crate::vector::tessellation::sharp_corner_offset(request.overscaling());
186            tessellator.split_corners = paint.line_gradient.is_some()
187                || paint.line_pattern.is_some()
188                || paint.line_dasharray.is_some();
189            tessellator.line_feature_style =
190                super::super::tessellation::LineFeatureStyle::for_paint(paint, style_zoom);
191            tessellator.line_elevation =
192                super::super::tessellation::LineElevationOptions::for_paint(paint, style_zoom);
193        }
194        _ => {}
195    }
196    tessellator
197}

199fn vector_layer<T: VectorTransferables, C: Context>(
200    layer: &tile::Layer,
201    selected: Option<&[usize]>,
202    shared_props: Option<&[Arc<FeatureProperties>]>,
203    style: &StyleLayer,
204    paint: &LayerPaint,
205    request: &VectorTileRequest,
206    context: &mut ProcessVectorContext<T, C>,
207) -> Result<(), ProcessVectorError> {
208    let state_ctx = context.state_context_for_layer(&layer.name);
209    let build = |slice: Option<&[usize]>| {
210        let mut tessellator = tessellator(paint, request);
211        tessellator.set_layer_strokes(style);
212        tessellator.coordinate_scale = extent_scale(layer);
213        tessellator.source_ids = source_ids_for_slice(layer, slice);
214        tessellator.shared_properties = shared_props_for_slice(shared_props, slice);
215        tessellator.feature_state_ctx = state_ctx.clone();
216        tessellator.feature_paint_retention =
217            crate::vector::feature_paint::feature_paint_retention(style);
218        tessellator.sort_key = crate::vector::tessellation::SortKeys::by(
219            crate::vector::tessellation::sort_key_of(style),
220        );
221        process_mvt_slice(layer, slice, shared_props.is_some(), &mut tessellator)?;
222        Ok(tessellator)
223    };
224    let Some(mut tessellator) = tessellate_readable(layer, selected, request.coords, build) else {
225        return context.layer_missing(&request.coords, &layer.name);
226    };
227    tessellator.apply_sort_keys();
228    let sort_keys = tessellator.sort_key_values();
229    context.layer_tessellation_finished(
230        &request.coords,
231        tessellator.buffer.into(),
232        tessellator.feature_indices,
233        (
234            tessellator.feature_colors,
235            sort_keys,
236            tessellator.feature_paint,
237        ),
238        layer_header(layer),
239        style.id.clone(),
240    )
241}

243/// Tessellates the layer with `build`. When a feature's geometry cannot be read, the features
244/// whose geometry cannot are left out and the rest are tessellated, so one malformed feature
245/// does not take the layer with it; `None` when the layer cannot be tessellated even so.
246fn tessellate_readable<Tessellator>(
247    layer: &tile::Layer,
248    selected: Option<&[usize]>,
249    coords: WorldTileCoords,
250    mut build: impl FnMut(Option<&[usize]>) -> geozero::error::Result<Tessellator>,
251) -> Option<Tessellator> {
252    let error = match build(selected) {
253        Ok(tessellator) => return Some(tessellator),
254        Err(error) => error,
255    };
256    let (readable, dropped) = retain_readable_indices(layer, selected);
257    if dropped == 0 {
258        tracing::error!(%coords, layer = %layer.name, ?error, "vector tessellation failed");
259        return None;
260    }
261    tracing::warn!(%coords, layer = %layer.name, dropped, ?error, "features whose geometry cannot be read are left out");
262    match build(Some(&readable)) {
263        Ok(tessellator) => Some(tessellator),
264        Err(error) => {
265            tracing::error!(%coords, layer = %layer.name, ?error, "vector tessellation failed");
266            None
267        }
268    }
269}

271fn symbol_layer<T: VectorTransferables, C: Context>(
272    layer: &tile::Layer,
273    selected: Option<&[usize]>,
274    style: &StyleLayer,
275    paint: &crate::style::layer::SymbolPaint,
276    request: &VectorTileRequest,
277    context: &mut ProcessVectorContext<T, C>,
278    atlas: std::sync::Arc<crate::sdf::assets::SymbolAtlas>,
279) -> Result<(), ProcessVectorError> {
280    let zoom = request.style_zoom();
281    let state_ctx = context.state_context_for_layer(&layer.name);
282    let build = |slice: Option<&[usize]>| {
283        let mut tessellator = TextTessellator::with_assets(paint.clone(), zoom, atlas.clone());
284        tessellator.coordinate_scale = extent_scale(layer);
285        tessellator.overscaling = request.overscaling();
286        tessellator.source_ids = source_ids_for_slice(layer, slice);
287        tessellator.feature_state_ctx = state_ctx.clone();
288        process_mvt_slice(layer, slice, false, &mut tessellator)?;
289        Ok(tessellator)
290    };
291    let Some(mut tessellator) = tessellate_readable(layer, selected, request.coords, build) else {
292        return context.layer_missing(&request.coords, &layer.name);
293    };
294    tessellator.finish();
295    let has_geometry = !tessellator.quad_buffer.indices.is_empty();
296    context.symbol_layer_tessellation_finished(
297        crate::vector::transferables::DefaultSymbolLayerTessellated {
298            coords: request.coords,
299            buffer: tessellator.quad_buffer.into(),
300            features: tessellator.features,
301            atlas: has_geometry.then_some(atlas),
302            layer_data: layer_header(layer),
303            style_layer_id: style.id.clone(),
304        },
305    )
306}
