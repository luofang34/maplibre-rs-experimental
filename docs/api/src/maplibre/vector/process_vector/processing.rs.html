13pub(crate) fn process_layer<T: VectorTransferables, C: Context>(
14    source_layer: Option<&tile::Layer>,
15    layer_name: &str,
16    predecoded_props: Option<&[Arc<FeatureProperties>]>,
17    style: &StyleLayer,
18    request: &VectorTileRequest,
19    context: &mut ProcessVectorContext<T, C>,
20    atlases: (
21        &std::sync::Arc<crate::sdf::assets::SymbolAtlas>,
22        &std::sync::Arc<crate::sdf::assets::SymbolAtlas>,
23    ),
24) -> Result<(), ProcessVectorError> {
25    let empty_layer;
26    let mut layer: Cow<'_, tile::Layer> = match source_layer {
27        Some(existing) => Cow::Borrowed(existing),
28        None => {
29            empty_layer = tile::Layer {
30                version: 2,
31                name: layer_name.to_owned(),
32                extent: Some(4096),
33                ..Default::default()
34            };
35            Cow::Borrowed(&empty_layer)
36        }
37    };
38    if crate::style::expression::layer_reads_geometry(style) && !layer.features.is_empty() {
39        super::geometry_tags::add_geometry_tags(layer.to_mut(), request.coords);
40    }
41    if crate::style::expression::layer_reads_identity(style) && !layer.features.is_empty() {
42        super::geometry_tags::add_identity_tags(layer.to_mut());
43    }
44    if matches!(style.paint, Some(LayerPaint::FillExtrusion(_)))
45        && layer.keys.iter().any(|key| key == "hide_3d")
46    {
47        unhide_unreplaced_building_bases(&mut layer);
48    }
49    let shared_props = matches!(layer, Cow::Borrowed(_))
50        .then_some(predecoded_props)
51        .flatten();
52    let selected = match &style.filter {
53        Some(raw_filter) => match Filter::parse(raw_filter) {
54            Ok(filter) => {
55                let state_ctx = context.state_context_for_layer(&layer.name);
56                Some(select_filtered_features(
57                    layer.as_ref(),
58                    &filter,
59                    request.style_zoom(),
60                    state_ctx.as_ref(),
61                    shared_props,
62                ))
63            }
64            Err(error) => {
65                // Rendering either all features or none silently would conceal an invalid style.
66                tracing::error!(layer = %style.id,%error,"unsupported filter; the layer renders nothing");
67                context.layer_missing(&request.coords, &layer.name)?;
68                return Ok(());
69            }
70        },
71        None => None,
72    };
73    match style.paint.as_ref() {
74        Some(
75            paint @ (LayerPaint::Line(_)
76            | LayerPaint::Fill(_)
77            | LayerPaint::FillExtrusion(_)
78            | LayerPaint::Circle(_)
79            | LayerPaint::Heatmap(_)),
80        ) => vector_layer(
81            layer.as_ref(),
82            selected.as_deref(),
83            shared_props,
84            style,
85            paint,
86            request,
87            context,
88        ),
89        Some(LayerPaint::Symbol(paint)) => symbol_layer(
90            layer.as_ref(),
91            selected.as_deref(),
92            shared_props,
93            style,
94            paint,
95            request,
96            context,
97            atlases,
98        ),
99        _ => {
100            tracing::warn!(layer = %style.id,"unhandled vector style layer type");
101            Ok(())
102        }
103    }
104}

106pub(crate) fn tessellator(
107    paint: &LayerPaint,
108    request: &VectorTileRequest,
109) -> ZeroTessellator<IndexDataType> {
110    let zoom = u8::from(request.coords.z);
111    let style_zoom = request.style_zoom();
112    let granularity = match paint {
113        LayerPaint::Fill(_) => granularity_for_zoom(128, 2, zoom),
114        LayerPaint::Line(_) => granularity_for_zoom(512, 0, zoom),
115        _ => 1,
116    };
117    let mut tessellator = match paint {
118        LayerPaint::Circle(circle) => ZeroTessellator::default().with_circles(CircleOptions {
119            // A circle lying on the map follows the globe's curve.
120            grid: request.projection.uses_globe_rendering(style_zoom)
121                && circle.circle_pitch_alignment == crate::style::circle::CirclePitchAlignment::Map,
122            ..CircleOptions::for_paint(circle, style_zoom)
123        }),
124        LayerPaint::Heatmap(heatmap) => ZeroTessellator::default().with_circles(CircleOptions {
125            grid: request.projection.uses_globe_rendering(style_zoom),
126            ..CircleOptions::for_heatmap(heatmap, style_zoom)
127        }),
128        LayerPaint::FillExtrusion(extrusion) => {
129            let tessellator = ZeroTessellator::default().with_extrusion(
130                ExtrusionOptions::for_paint(extrusion, request.overscaling()),
131            );
132            let min_grid = if extrusion.may_use_vertex_agl() {
133                32
134            } else {
135                2
136            };
137            if request.projection.uses_globe_rendering(style_zoom) {
138                // Roofs and the edges of walls follow the sphere as fills and lines do.
139                let last_tile = i64::from(crate::coords::ZOOM_BOUNDS[usize::from(zoom)]) - 1;
140                tessellator.with_globe_subdivision(
141                    granularity_for_zoom(128, min_grid, zoom),
142                    zoom == 0,
143                    request.coords.y == 0,
144                    i64::from(request.coords.y) == last_tile,
145                )
146            } else if extrusion.may_use_vertex_agl() {
147                tessellator.with_globe_subdivision(32, false, false, false)
148            } else {
149                tessellator
150            }
151        }
152        _ if request.projection.uses_globe_rendering(style_zoom) => {
153            let last_tile = i64::from(crate::coords::ZOOM_BOUNDS[usize::from(zoom)]) - 1;
154            ZeroTessellator::default().with_globe_subdivision(
155                granularity,
156                zoom == 0,
157                request.coords.y == 0,
158                i64::from(request.coords.y) == last_tile,
159            )
160        }
161        _ => ZeroTessellator::default(),
162    }
163    .with_feature_opacity(paint.opacity(), style_zoom);
164    match paint {
165        // An image repeated over a fill takes nothing from the fill's colour but its opacity.
166        LayerPaint::Fill(paint) if paint.fill_pattern.is_some() => {
167            tessellator.fallback_color = [1.0; 4];
168            tessellator.pattern_property =
169                crate::vector::pattern::per_feature_pattern(&LayerPaint::Fill(paint.clone()));
170        }
171        LayerPaint::Fill(paint) => {
172            tessellator.style_property = paint.fill_color.clone();
173            tessellator.outline_property = paint.outline_color();
174        }
175        LayerPaint::FillExtrusion(paint) if paint.fill_extrusion_pattern.is_some() => {
176            tessellator.fallback_color = [1.0; 4];
177            tessellator.pattern_property = crate::vector::pattern::per_feature_pattern(
178                &LayerPaint::FillExtrusion(paint.clone()),
179            );
180        }
181        LayerPaint::FillExtrusion(paint) => {
182            tessellator.style_property = paint.fill_extrusion_color.clone()
183        }
184        LayerPaint::Circle(paint) => tessellator.style_property = paint.circle_color.clone(),
185        LayerPaint::Line(paint) => {
186            tessellator.pattern_property =
187                crate::vector::pattern::per_feature_pattern(&LayerPaint::Line(paint.clone()));
188            tessellator.style_property = paint.line_color.clone();
189            tessellator.is_line_layer = true;
190            tessellator.line_gradient = paint.line_gradient.is_some();
191            tessellator.sharp_corner_offset =
192                crate::vector::tessellation::sharp_corner_offset(request.overscaling());
193            tessellator.split_corners = paint.line_gradient.is_some()
194                || paint.line_pattern.is_some()
195                || paint.line_dasharray.is_some();
196            tessellator.line_feature_style =
197                super::super::tessellation::LineFeatureStyle::for_paint(paint, style_zoom);
198            tessellator.line_elevation =
199                super::super::tessellation::LineElevationOptions::for_paint(paint, style_zoom);
200        }
201        _ => {}
202    }
203    tessellator
204}

206fn allow_prepack_for_style_inner(
207    coords: WorldTileCoords,
208    has_dem_source: bool,
209    style: &StyleLayer,
210    has_sort_keys: bool,
211    has_shared_geometry: bool,
212) -> bool {
213    let per_feature_pattern = style
214        .paint
215        .as_ref()
216        .and_then(crate::vector::pattern::per_feature_pattern)
217        .is_some();
218    let edge_y = (1_i32 << u8::from(coords.z).min(30)) - 1;
219    let is_polar_row = coords.y <= 0 || coords.y >= edge_y;
220    let structure_kind = crate::vector::structures::kind(style);
221    !(is_polar_row && style.type_ == "fill")
222        && !per_feature_pattern
223        && !has_sort_keys
224        && style.type_ != "circle"
225        && if has_dem_source {
226            structure_kind.is_none()
227        } else {
228            structure_kind
229                .is_none_or(|k| matches!(k, crate::vector::structures::StructureKind::Extrusion))
230        }
231        && !has_shared_geometry
232}

251fn vector_layer<T: VectorTransferables, C: Context>(
252    layer: &tile::Layer,
253    selected: Option<&[usize]>,
254    shared_props: Option<&[Arc<FeatureProperties>]>,
255    style: &StyleLayer,
256    paint: &LayerPaint,
257    request: &VectorTileRequest,
258    context: &mut ProcessVectorContext<T, C>,
259) -> Result<(), ProcessVectorError> {
260    let state_ctx = context.state_context_for_layer(&layer.name);
261    #[cfg(any(target_arch = "wasm32", not(test)))]
262    let uniform_paint =
263        crate::vector::upload_system::paint::uniform_color(style, request.style_zoom()).is_some()
264            && style.type_ != "circle"
265            && !crate::style::feature_state::layer_paint_reads_feature_state(style);
266    let build = |slice: Option<&[usize]>| {
267        let mut tessellator = tessellator(paint, request);
268        tessellator.set_layer_strokes(style);
269        tessellator.coordinate_scale = extent_scale(layer);
270        tessellator.source_ids = source_ids_for_slice(layer, slice);
271        tessellator.shared_properties = shared_props_for_slice(shared_props, slice);
272        tessellator.feature_state_ctx = state_ctx.clone();
273        tessellator.feature_paint_retention =
274            crate::vector::feature_paint::feature_paint_retention(style);
275        tessellator.sort_key = crate::vector::tessellation::SortKeys::by(
276            crate::vector::tessellation::sort_key_of(style),
277        );
278        #[cfg(any(target_arch = "wasm32", not(test)))]
279        if uniform_paint && tessellator.pattern_property.is_none() && !tessellator.line_gradient {
280            tessellator.omit_feature_colors = true;
281        }
282        if context.allow_prepack_for_style(
283            request.coords,
284            style,
285            tessellator.sort_key.is_keyed(),
286            false,
287        ) {
288            tessellator.maybe_enable_streaming_prepack(style.type_.as_str());
289        }
290        process_mvt_slice(layer, slice, shared_props.is_some(), &mut tessellator)?;
291        Ok(tessellator)
292    };
293    let Some(mut tessellator) = tessellate_readable(layer, selected, request.coords, build) else {
294        return context.layer_missing(&request.coords, &layer.name);
295    };
296    tessellator.apply_sort_keys();
297    let mut sort_keys = tessellator.sort_key_values();
298    #[cfg(any(target_arch = "wasm32", not(test)))]
299    if uniform_paint && tessellator.pattern_property.is_none() && !tessellator.line_gradient {
300        tessellator.feature_colors = Vec::new();
301    }
302    let mut feature_indices = std::mem::take(&mut tessellator.feature_indices);
303    let mut feature_colors = std::mem::take(&mut tessellator.feature_colors);
304    let mut feature_paint = std::mem::take(&mut tessellator.feature_paint);
305    feature_indices.shrink_to_fit();
306    feature_colors.shrink_to_fit();
307    sort_keys.shrink_to_fit();
308    feature_paint.shrink_to_fit();
309    let buffer = tessellator.into_over_aligned_buffer();
310    context.layer_tessellation_finished(
311        &request.coords,
312        buffer,
313        feature_indices,
314        (feature_colors, sort_keys, feature_paint),
315        layer_header(layer),
316        style,
317    )
318}

320/// Tessellates the layer with `build`. When a feature's geometry cannot be read, the features
321/// whose geometry cannot are left out and the rest are tessellated, so one malformed feature
322/// does not take the layer with it; `None` when the layer cannot be tessellated even so.
323fn tessellate_readable<Tessellator>(
324    layer: &tile::Layer,
325    selected: Option<&[usize]>,
326    coords: WorldTileCoords,
327    mut build: impl FnMut(Option<&[usize]>) -> geozero::error::Result<Tessellator>,
328) -> Option<Tessellator> {
329    let error = match build(selected) {
330        Ok(tessellator) => return Some(tessellator),
331        Err(error) => error,
332    };
333    let (readable, dropped) = retain_readable_indices(layer, selected);
334    if dropped == 0 {
335        tracing::error!(%coords, layer = %layer.name, ?error, "vector tessellation failed");
336        return None;
337    }
338    tracing::warn!(%coords, layer = %layer.name, dropped, ?error, "features whose geometry cannot be read are left out");
339    match build(Some(&readable)) {
340        Ok(tessellator) => Some(tessellator),
341        Err(error) => {
342            tracing::error!(%coords, layer = %layer.name, ?error, "vector tessellation failed");
343            None
344        }
345    }
346}

348#[allow(clippy::too_many_arguments)]
349fn symbol_layer<T: VectorTransferables, C: Context>(
350    layer: &tile::Layer,
351    selected: Option<&[usize]>,
352    shared_props: Option<&[Arc<FeatureProperties>]>,
353    style: &StyleLayer,
354    paint: &crate::style::layer::SymbolPaint,
355    request: &VectorTileRequest,
356    context: &mut ProcessVectorContext<T, C>,
357    (layout_atlas, gpu_atlas): (
358        &std::sync::Arc<crate::sdf::assets::SymbolAtlas>,
359        &std::sync::Arc<crate::sdf::assets::SymbolAtlas>,
360    ),
361) -> Result<(), ProcessVectorError> {
362    let zoom = request.style_zoom();
363    let state_ctx = context.state_context_for_layer(&layer.name);
364    let promote_prop = state_ctx
365        .as_ref()
366        .and_then(|ctx| ctx.promote_property.as_deref());
367    let stream_point = !context.index_geometry()
368        && !crate::sdf::tessellation::is_line_placed(paint)
369        && !paint.properties.contains_key("symbol-sort-key");
370    let build = |slice: Option<&[usize]>| {
371        let mut tessellator =
372            TextTessellator::with_assets(paint.clone(), zoom, layout_atlas.clone());
373        tessellator.coordinate_scale = extent_scale(layer);
374        tessellator.overscaling = request.overscaling();
375        tessellator.source_ids = source_ids_for_slice(layer, slice);
376        tessellator.source_indices =
377            crate::sdf::tessellation::source_feature_indices(layer.features.len(), slice);
378        tessellator.shared_properties = shared_props_for_slice(shared_props, slice);
379        tessellator.feature_state_ctx = state_ctx.clone();
380        if stream_point {
381            tessellator.enable_streaming_point_layout(
382                crate::style::expression::symbol_runtime_referenced_keys(style, promote_prop),
383            );
384        }
385        process_mvt_slice(layer, slice, shared_props.is_some(), &mut tessellator)?;
386        Ok(tessellator)
387    };
388    let Some(mut tessellator) = tessellate_readable(layer, selected, request.coords, build) else {
389        return context.layer_missing(&request.coords, &layer.name);
390    };
391    tessellator.finish();
392    if !context.index_geometry() && !tessellator.streamed_point_layout() {
393        crate::sdf::tessellation::prune_symbol_features(
394            &mut tessellator.features,
395            style,
396            promote_prop,
397        );
398    }
399    tessellator.quad_buffer.vertices.shrink_to_fit();
400    tessellator.quad_buffer.indices.shrink_to_fit();
401    tessellator.features.shrink_to_fit();
402    let has_geometry = !tessellator.quad_buffer.indices.is_empty();
403    context.symbol_layer_tessellation_finished(
404        crate::vector::transferables::DefaultSymbolLayerTessellated {
405            coords: request.coords,
406            buffer: tessellator.quad_buffer.into(),
407            features: tessellator.features,
408            atlas: has_geometry.then(|| gpu_atlas.clone()),
409            layer_data: layer_header(layer),
410            style_layer_id: style.id.clone(),
411        },
412    )
413}

419mod tests {

423    fn parse_layer(json: serde_json::Value) -> Option<StyleLayer> {
424        serde_json::from_value(json).ok()

426
427    #[test]
428    fn terrain_prepacks_draped_layers_but_keeps_structures_unpacked() {
429        let mid_tile = WorldTileCoords::from((5, 5, 4_u8.into()));
430        let fill = parse_layer(serde_json::json!({
431            "id": "water", "type": "fill", "paint": {"fill-color": "#38bdf8"}
432        }));
433        let line = parse_layer(serde_json::json!({
434            "id": "roads", "type": "line", "paint": {"line-color": "#64748b", "line-width": 2}
435        }));
436        let extrusion = parse_layer(serde_json::json!({
437            "id": "bld", "type": "fill-extrusion", "paint": {"fill-extrusion-color": "#cbd5e1"}
438        }));
439        let circle = parse_layer(serde_json::json!({
440            "id": "pts", "type": "circle", "paint": {"circle-radius": 4}
441        }));
442        let bridge = parse_layer(serde_json::json!({
443            "id": "bridge", "type": "line",
444            "metadata": {"maplibre-rs:terrain-structure": "bridge"},
445            "paint": {"line-color": "#64748b", "line-width": 2}
446        }));
447
448        assert!(
449            fill.is_some()
450                && line.is_some()
451                && extrusion.is_some()
452                && circle.is_some()
453                && bridge.is_some()
454        );
455        if let (Some(fill), Some(line), Some(extrusion), Some(circle), Some(bridge)) =
456            (fill, line, extrusion, circle, bridge)
457        {
458            assert!(allow_prepack_for_style_inner(
459                mid_tile, true, &fill, false, false
460            ));
461            assert!(allow_prepack_for_style_inner(
462                mid_tile, true, &line, false, false
463            ));
464            assert!(!allow_prepack_for_style_inner(
465                mid_tile, true, &extrusion, false, false
466            ));
467            assert!(allow_prepack_for_style_inner(
468                mid_tile, false, &extrusion, false, false
469            ));
470            assert!(!allow_prepack_for_style_inner(
471                mid_tile, true, &circle, false, false
472            ));
473            assert!(!allow_prepack_for_style_inner(
474                mid_tile, true, &bridge, false, false
475            ));
476        }
477    }
