4pub(super) fn process_layer<T: VectorTransferables, C: Context>(
5    mut layer: tile::Layer,
6    style: &StyleLayer,
7    request: &VectorTileRequest,
8    context: &mut ProcessVectorContext<T, C>,
9    atlas: std::sync::Arc<crate::sdf::assets::SymbolAtlas>,
10) -> Result<(), ProcessVectorError> {
11    if crate::geojson::reads_geometry(style) {
12        super::geometry_tags::add_geometry_tags(&mut layer, request.coords);
13    }
14    if crate::geojson::reads_identity(style) {
15        super::geometry_tags::add_identity_tags(&mut layer);
16    }
17    if matches!(style.paint, Some(LayerPaint::FillExtrusion(_))) {
18        unhide_unreplaced_building_bases(&mut layer);
19    }
20    if let Some(filter) = &style.filter {
21        match Filter::parse(filter) {
22            Ok(filter) => apply_filter_to_layer(&mut layer, &filter, request.style_zoom()),
23            Err(error) => {
24                // Rendering either all features or none silently would conceal an invalid style.
25                tracing::error!(layer = %style.id,%error,"unsupported filter; the layer renders nothing");
26                context.layer_missing(&request.coords, &layer.name)?;
27                return Ok(());
28            }
29        }
30    }
31    match style.paint.as_ref() {
32        Some(
33            paint @ (LayerPaint::Line(_)
34            | LayerPaint::Fill(_)
35            | LayerPaint::FillExtrusion(_)
36            | LayerPaint::Circle(_)
37            | LayerPaint::Heatmap(_)),
38        ) => vector_layer(layer, style, paint, request, context),
39        Some(LayerPaint::Symbol(paint)) => {
40            symbol_layer(layer, style, paint, request, context, atlas)
41        }
42        _ => {
43            tracing::warn!(layer = %style.id,"unhandled vector style layer type");
44            Ok(())
45        }
46    }
47}

49fn tessellator(paint: &LayerPaint, request: &VectorTileRequest) -> ZeroTessellator<IndexDataType> {
50    let zoom = u8::from(request.coords.z);
51    let style_zoom = request.style_zoom();
52    let granularity = match paint {
53        LayerPaint::Fill(_) => granularity_for_zoom(128, 2, zoom),
54        LayerPaint::Line(_) => granularity_for_zoom(512, 0, zoom),
55        _ => 1,
56    };
57    let mut tessellator = match paint {
58        LayerPaint::Circle(circle) => ZeroTessellator::default().with_circles(CircleOptions {
59            // A circle lying on the map follows the globe's curve.
60            grid: request.projection.uses_globe_rendering(style_zoom)
61                && circle.circle_pitch_alignment == crate::style::circle::CirclePitchAlignment::Map,
62            ..CircleOptions::for_paint(circle, style_zoom)
63        }),
64        LayerPaint::Heatmap(heatmap) => ZeroTessellator::default().with_circles(CircleOptions {
65            grid: request.projection.uses_globe_rendering(style_zoom),
66            ..CircleOptions::for_heatmap(heatmap, style_zoom)
67        }),
68        LayerPaint::FillExtrusion(extrusion) => {
69            let tessellator = ZeroTessellator::default().with_extrusion(
70                ExtrusionOptions::for_paint(extrusion, request.overscaling()),
71            );
72            let min_grid = if extrusion.may_use_vertex_agl() {
73                16
74            } else {
75                2
76            };
77            if request.projection.uses_globe_rendering(style_zoom) {
78                // Roofs and the edges of walls follow the sphere as fills and lines do.
79                let last_tile = i64::from(crate::coords::ZOOM_BOUNDS[usize::from(zoom)]) - 1;
80                tessellator.with_globe_subdivision(
81                    granularity_for_zoom(128, min_grid, zoom),
82                    zoom == 0,
83                    request.coords.y == 0,
84                    i64::from(request.coords.y) == last_tile,
85                )
86            } else if extrusion.may_use_vertex_agl() {
87                tessellator.with_globe_subdivision(16, false, false, false)
88            } else {
89                tessellator
90            }
91        }
92        _ if request.projection.uses_globe_rendering(style_zoom) => {
93            let last_tile = i64::from(crate::coords::ZOOM_BOUNDS[usize::from(zoom)]) - 1;
94            ZeroTessellator::default().with_globe_subdivision(
95                granularity,
96                zoom == 0,
97                request.coords.y == 0,
98                i64::from(request.coords.y) == last_tile,
99            )
100        }
101        _ => ZeroTessellator::default(),
102    }
103    .with_feature_opacity(paint.opacity(), style_zoom);
104    match paint {
105        // An image repeated over a fill takes nothing from the fill's colour but its opacity.
106        LayerPaint::Fill(paint) if paint.fill_pattern.is_some() => {
107            tessellator.fallback_color = [1.0; 4];
108            tessellator.pattern_property =
109                crate::vector::pattern::per_feature_pattern(&LayerPaint::Fill(paint.clone()));
110        }
111        LayerPaint::Fill(paint) => {
112            tessellator.style_property = paint.fill_color.clone();
113            tessellator.outline_property = paint.outline_color();
114        }
115        LayerPaint::FillExtrusion(paint) if paint.fill_extrusion_pattern.is_some() => {
116            tessellator.fallback_color = [1.0; 4];
117            tessellator.pattern_property = crate::vector::pattern::per_feature_pattern(
118                &LayerPaint::FillExtrusion(paint.clone()),
119            );
120        }
121        LayerPaint::FillExtrusion(paint) => {
122            tessellator.style_property = paint.fill_extrusion_color.clone()
123        }
124        LayerPaint::Circle(paint) => tessellator.style_property = paint.circle_color.clone(),
125        LayerPaint::Line(paint) => {
126            tessellator.pattern_property =
127                crate::vector::pattern::per_feature_pattern(&LayerPaint::Line(paint.clone()));
128            tessellator.style_property = paint.line_color.clone();
129            tessellator.is_line_layer = true;
130            tessellator.line_gradient = paint.line_gradient.is_some();
131            tessellator.sharp_corner_offset =
132                crate::vector::tessellation::sharp_corner_offset(request.overscaling());
133            tessellator.split_corners = paint.line_gradient.is_some()
134                || paint.line_pattern.is_some()
135                || paint.line_dasharray.is_some();
136            tessellator.line_feature_style =
137                super::super::tessellation::LineFeatureStyle::for_paint(paint, style_zoom);
138            tessellator.line_elevation =
139                super::super::tessellation::LineElevationOptions::for_paint(paint, style_zoom);
140        }
141        _ => {}
142    }
143    tessellator
144}

146fn vector_layer<T: VectorTransferables, C: Context>(
147    mut layer: tile::Layer,
148    style: &StyleLayer,
149    paint: &LayerPaint,
150    request: &VectorTileRequest,
151    context: &mut ProcessVectorContext<T, C>,
152) -> Result<(), ProcessVectorError> {
153    let build = |layer: &mut tile::Layer| {
154        let mut tessellator = tessellator(paint, request);
155        tessellator.set_layer_strokes(style);
156        tessellator.coordinate_scale = extent_scale(layer);
157        tessellator.sort_key = crate::vector::tessellation::SortKeys::by(
158            crate::vector::tessellation::sort_key_of(style),
159        );
160        layer.process(&mut tessellator)?;
161        Ok(tessellator)
162    };
163    let Some(mut tessellator) = tessellate_readable(&mut layer, request.coords, build) else {
164        return context.layer_missing(&request.coords, &layer.name);
165    };
166    tessellator.apply_sort_keys();
167    let sort_keys = tessellator.sort_key_values();
168    context.layer_tessellation_finished(
169        &request.coords,
170        tessellator.buffer.into(),
171        tessellator.feature_indices,
172        (tessellator.feature_colors, sort_keys),
173        layer,
174        style.id.clone(),
175    )
176}

178/// Tessellates the layer with `build`. When a feature's geometry cannot be read, the features
179/// whose geometry cannot are left out and the rest are tessellated, so one malformed feature
180/// does not take the layer with it; `None` when the layer cannot be tessellated even so.
181fn tessellate_readable<Tessellator>(
182    layer: &mut tile::Layer,
183    coords: WorldTileCoords,
184    mut build: impl FnMut(&mut tile::Layer) -> geozero::error::Result<Tessellator>,
185) -> Option<Tessellator> {
186    use geozero::{GeozeroGeometry, ProcessorSink};
187    let error = match build(layer) {
188        Ok(tessellator) => return Some(tessellator),
189        Err(error) => error,
190    };
191    let before = layer.features.len();
192    layer
193        .features
194        .retain(|feature| feature.process_geom(&mut ProcessorSink).is_ok());
195    let dropped = before - layer.features.len();
196    if dropped == 0 {
197        tracing::error!(%coords, layer = %layer.name, ?error, "vector tessellation failed");
198        return None;
199    }
200    tracing::warn!(%coords, layer = %layer.name, dropped, ?error, "features whose geometry cannot be read are left out");
201    match build(layer) {
202        Ok(tessellator) => Some(tessellator),
203        Err(error) => {
204            tracing::error!(%coords, layer = %layer.name, ?error, "vector tessellation failed");
205            None
206        }
207    }
208}

210fn symbol_layer<T: VectorTransferables, C: Context>(
211    mut layer: tile::Layer,
212    style: &StyleLayer,
213    paint: &crate::style::layer::SymbolPaint,
214    request: &VectorTileRequest,
215    context: &mut ProcessVectorContext<T, C>,
216    atlas: std::sync::Arc<crate::sdf::assets::SymbolAtlas>,
217) -> Result<(), ProcessVectorError> {
218    let zoom = request.style_zoom();
219    let build = |layer: &mut tile::Layer| {
220        let mut tessellator = TextTessellator::with_assets(paint.clone(), zoom, atlas.clone());
221        tessellator.coordinate_scale = extent_scale(layer);
222        tessellator.overscaling = request.overscaling();
223        tessellator.source_ids = layer.features.iter().map(|feature| feature.id).collect();
224        layer.process(&mut tessellator)?;
225        Ok(tessellator)
226    };
227    let Some(mut tessellator) = tessellate_readable(&mut layer, request.coords, build) else {
228        return context.layer_missing(&request.coords, &layer.name);
229    };
230    tessellator.finish();
231    let has_geometry = !tessellator.quad_buffer.indices.is_empty();
232    context.symbol_layer_tessellation_finished(
233        crate::vector::transferables::DefaultSymbolLayerTessellated {
234            coords: request.coords,
235            buffer: tessellator.quad_buffer.into(),
236            features: tessellator.features,
237            atlas: has_geometry.then_some(atlas),
238            layer_data: layer,
239            style_layer_id: style.id.clone(),
240        },
241    )
242}

244/// Clears `hide_3d` on building footprints whose interior `building:part` pieces all start
245/// above ground (`render_min_height > 0`), so stepped skyscrapers modeled with a `building`
246/// base podium and elevated `building:part` tiers do not float in mid-air.
247fn unhide_unreplaced_building_bases(layer: &mut tile::Layer) {
248    let Some(hide_3d_key) = layer
249        .keys
250        .iter()
251        .position(|key| key == "hide_3d")
252        .and_then(|idx| u32::try_from(idx).ok())
253    else {
254        return;
255    };
256    let Some(min_height_key) = layer
257        .keys
258        .iter()
259        .position(|key| key == "render_min_height")
260        .and_then(|idx| u32::try_from(idx).ok())
261    else {
262        return;
263    };
264    let height_key = layer
265        .keys
266        .iter()
267        .position(|key| key == "render_height")
268        .and_then(|idx| u32::try_from(idx).ok());
269
270    struct PartInfo {
271        bbox: [i32; 4],
272        centroids: Vec<[f32; 2]>,
273        min_height: f64,
274    }
275
276    let mut parts = Vec::new();
277    let mut has_hide_3d = false;
278    for feature in &layer.features {
279        if feature.r#type != Some(3) {
280            continue;
281        }
282        if feature_bool_tag(feature, hide_3d_key, &layer.values) == Some(true) {
283            has_hide_3d = true;
284            continue;
285        }
286        let rings = decode_mvt_rings(&feature.geometry);
287        let Some(bbox) = rings_bbox(&rings) else {
288            continue;
289        };
290        let centroids: Vec<[f32; 2]> = rings
291            .iter()
292            .filter_map(|ring| {
293                if ring.is_empty() {
294                    return None;
295                }
296                let n = ring.len() as f32;
297                let sum = ring.iter().fold([0.0_f32, 0.0_f32], |acc, pt| {
298                    [acc[0] + pt[0] as f32, acc[1] + pt[1] as f32]
299                });
300                Some([sum[0] / n, sum[1] / n])
301            })
302            .collect();
303        let min_height = feature_number_tag(feature, min_height_key, &layer.values).unwrap_or(0.0);
304        parts.push(PartInfo {
305            bbox,
306            centroids,
307            min_height,
308        });
309    }
310    if !has_hide_3d || parts.is_empty() {
311        return;
312    }
313
314    for feature_index in 0..layer.features.len() {
315        let feature = &layer.features[feature_index];
316        if feature.r#type != Some(3)
317            || feature_bool_tag(feature, hide_3d_key, &layer.values) != Some(true)
318            || feature_number_tag(feature, min_height_key, &layer.values).unwrap_or(0.0) > 0.0
319        {
320            continue;
321        }
322        let rings = decode_mvt_rings(&feature.geometry);
323        let Some(bbox) = rings_bbox(&rings) else {
324            continue;
325        };
326        let mut min_part_base: Option<f64> = None;
327        for part in &parts {
328            if part.bbox[2] < bbox[0]
329                || part.bbox[0] > bbox[2]
330                || part.bbox[3] < bbox[1]
331                || part.bbox[1] > bbox[3]
332            {
333                continue;
334            }
335            let inside = part.centroids.iter().any(|&point| {
336                rings
337                    .iter()
338                    .filter(|ring| point_in_mvt_ring(ring, point))
339                    .count()
340                    % 2
341                    == 1
342            });
343            if inside {
344                min_part_base =
345                    Some(min_part_base.map_or(part.min_height, |cur| cur.min(part.min_height)));
346            }
347        }
348        let Some(min_base) = min_part_base.filter(|&base| base > 0.0) else {
349            continue;
350        };
351        remove_feature_tag(&mut layer.features[feature_index].tags, hide_3d_key);
352        if let Some(height_key) = height_key {
353            if let Some(height) =
354                feature_number_tag(&layer.features[feature_index], height_key, &layer.values)
355            {
356                if height > min_base {
357                    let val_idx = u32::try_from(layer.values.len()).unwrap_or(u32::MAX);
358                    layer.values.push(tile::Value {
359                        double_value: Some(min_base),
360                        ..Default::default()
361                    });
362                    set_feature_tag(&mut layer.features[feature_index].tags, height_key, val_idx);
363                }
364            }
365        }
366    }
367}

369fn feature_bool_tag(feature: &tile::Feature, key: u32, values: &[tile::Value]) -> Option<bool> {
370    for &[k, v] in feature.tags.as_chunks::<2>().0 {
371        if k == key {
372            return values.get(v as usize)?.bool_value;
373        }
374    }
375    None
376}

378fn feature_number_tag(feature: &tile::Feature, key: u32, values: &[tile::Value]) -> Option<f64> {
379    for &[k, v] in feature.tags.as_chunks::<2>().0 {
380        if k == key {
381            let v = values.get(v as usize)?;
382            return v
383                .double_value
384                .or_else(|| v.float_value.map(f64::from))
385                .or_else(|| v.int_value.map(|n| n as f64))
386                .or_else(|| v.uint_value.map(|n| n as f64))
387                .or_else(|| v.sint_value.map(|n| n as f64));
388        }
389    }
390    None
391}

393fn remove_feature_tag(tags: &mut Vec<u32>, key: u32) {
394    let mut kept = Vec::with_capacity(tags.len());
395    for pair in tags.as_chunks::<2>().0 {
396        if pair[0] != key {
397            kept.extend_from_slice(pair);
398        }
399    }
400    *tags = kept;
401}
402
403fn set_feature_tag(tags: &mut [u32], key: u32, value_index: u32) {
404    for pair in tags.as_chunks_mut::<2>().0 {
405        if pair[0] == key {
406            pair[1] = value_index;
407            return;
408        }
409    }
410}

412fn decode_mvt_rings(geometry: &[u32]) -> Vec<Vec<[i32; 2]>> {
413    let mut rings: Vec<Vec<[i32; 2]>> = Vec::new();
414    let (mut cx, mut cy) = (0_i32, 0_i32);
415    let mut idx = 0;
416    while idx < geometry.len() {
417        let cmd_int = geometry[idx];
418        idx += 1;
419        let (cmd, count) = (cmd_int & 7, (cmd_int >> 3) as usize);
420        if cmd == 1 || cmd == 2 {
421            for _ in 0..count {
422                if idx + 1 >= geometry.len() {
423                    return rings;
424                }
425                let raw_dx = geometry[idx];
426                let raw_dy = geometry[idx + 1];
427                idx += 2;
428                let dx = ((raw_dx >> 1) as i32) ^ -((raw_dx & 1) as i32);
429                let dy = ((raw_dy >> 1) as i32) ^ -((raw_dy & 1) as i32);
430                cx = cx.saturating_add(dx);
431                cy = cy.saturating_add(dy);
432                if cmd == 1 {
433                    rings.push(vec![[cx, cy]]);
434                } else if let Some(ring) = rings.last_mut() {
435                    ring.push([cx, cy]);
436                }
437            }
438        }
439    }
440    rings
441}

443fn rings_bbox(rings: &[Vec<[i32; 2]>]) -> Option<[i32; 4]> {
444    let mut iter = rings.iter().flatten();
445    let &[x0, y0] = iter.next()?;
446    let mut bbox = [x0, y0, x0, y0];
447    for &[x, y] in iter {
448        bbox[0] = bbox[0].min(x);
449        bbox[1] = bbox[1].min(y);
450        bbox[2] = bbox[2].max(x);
451        bbox[3] = bbox[3].max(y);
452    }
453    Some(bbox)
454}

456fn point_in_mvt_ring(ring: &[[i32; 2]], [x, y]: [f32; 2]) -> bool {
457    let mut inside = false;
458    for (a, b) in ring.iter().zip(ring.iter().cycle().skip(1)) {
459        let (ax, ay) = (a[0] as f32, a[1] as f32);
460        let (bx, by) = (b[0] as f32, b[1] as f32);
461        if (ay > y) != (by > y) && x < (bx - ax) * (y - ay) / (by - ay) + ax {
462            inside = !inside;
463        }
464    }
465    inside
