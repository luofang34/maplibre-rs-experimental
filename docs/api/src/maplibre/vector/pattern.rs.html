13struct PatternBinding {
14    /// Identity of the image the binding was made for, so a replaced image is uploaded again.
15    fingerprint: u64,
16    bind_group: wgpu::BindGroup,
17    size: wgpu::Buffer,
18    /// The image's display size in layout pixels.
19    display: [f32; 2],
20    /// How far the fractional zoom stretches the pattern the buffer holds.
21    scale: f32,
22}

24/// The pipeline and the per-layer bindings of layers that fill with a repeating image.
25pub(crate) struct PatternResources {
26    /// The fill pipeline, once a vector plugin has built it; backgrounds need only the images.
27    pipeline: Option<wgpu::RenderPipeline>,
28    layout: wgpu::BindGroupLayout,
29    layers: HashMap<String, PatternBinding>,
30    /// Layers whose pattern varies by feature, and so bind an image per run of features.
31    per_feature: std::collections::HashSet<String>,
32    /// Every image of the style by its pattern key, for the layers in `per_feature`.
33    images: HashMap<u32, PatternBinding>,
34    sampler: wgpu::Sampler,
35}

37/// Bind group layout of one layer's pattern: the image size, the image and its sampler.
38pub(crate) fn layout_entries() -> Vec<wgpu::BindGroupLayoutEntry> {
39    vec![
40        wgpu::BindGroupLayoutEntry {
41            binding: 0,
42            visibility: wgpu::ShaderStages::VERTEX_FRAGMENT,
43            ty: wgpu::BindingType::Buffer {
44                ty: wgpu::BufferBindingType::Uniform,
45                has_dynamic_offset: false,
46                min_binding_size: None,
47            },
48            count: None,
49        },
50        wgpu::BindGroupLayoutEntry {
51            binding: 1,
52            visibility: wgpu::ShaderStages::FRAGMENT,
53            ty: wgpu::BindingType::Texture {
54                sample_type: wgpu::TextureSampleType::Float { filterable: true },
55                view_dimension: wgpu::TextureViewDimension::D2,
56                multisampled: false,
57            },
58            count: None,
59        },
60        wgpu::BindGroupLayoutEntry {
61            binding: 2,
62            visibility: wgpu::ShaderStages::FRAGMENT,
63            ty: wgpu::BindingType::Sampler(wgpu::SamplerBindingType::Filtering),
64            count: None,
65        },
66    ]
67}

69/// The name of the image a fill, extrusion or background layer repeats at a zoom, if it repeats one.
70pub(crate) fn pattern_name(paint: &LayerPaint, zoom: f64) -> Option<String> {
71    let value = match paint {
72        LayerPaint::Fill(fill) => fill.fill_pattern.as_ref()?,
73        LayerPaint::FillExtrusion(extrusion) => extrusion.fill_extrusion_pattern.as_ref()?,
74        LayerPaint::Background(background) => background.background_pattern.as_ref()?,
75        _ => return None,
76    };
77    StyleProperty::<PatternName>::parse(value)
78        .evaluate_at_zoom(zoom)
79        .map(|name| name.0)
80        .filter(|name| !name.is_empty())
81}

83/// The pattern property of a fill, extrusion or line layer when its image varies by feature.
84pub(crate) fn per_feature_pattern(paint: &LayerPaint) -> Option<StyleProperty<PatternName>> {
85    let value = match paint {
86        LayerPaint::Fill(fill) => fill.fill_pattern.as_ref()?,
87        LayerPaint::FillExtrusion(extrusion) => extrusion.fill_extrusion_pattern.as_ref()?,
88        LayerPaint::Line(line) => line.line_pattern.as_ref()?,
89        _ => return None,
90    };
91    let property = StyleProperty::<PatternName>::parse(value);
92    (!property.is_feature_constant()).then_some(property)
93}

95/// Whether a layer declares a pattern that names no image the style holds at `zoom`, so that GL
96/// JS draws nothing for it; a name that varies by feature counts as none here.
97pub(crate) fn names_missing_image(paint: Option<&LayerPaint>, style: &Style, zoom: f64) -> bool {
98    let value = match paint {
99        Some(LayerPaint::Fill(fill)) => fill.fill_pattern.as_ref(),
100        Some(LayerPaint::FillExtrusion(extrusion)) => extrusion.fill_extrusion_pattern.as_ref(),
101        Some(LayerPaint::Background(background)) => background.background_pattern.as_ref(),
102        _ => None,
103    };
104    value.is_some_and(|value| {
105        let property = StyleProperty::<PatternName>::parse(value);
106        // A feature's own name decides, where its geometry is drawn.
107        property.is_feature_constant()
108            && property
109                .evaluate_at_zoom(zoom)
110                .is_none_or(|name| !style.images.contains_key(&name.0))
111    })
112}

114/// Patterns are laid out in tile units of the whole zoom below the view's, as GL JS does, so
115/// between whole zooms they grow with the map.
116fn pattern_scale(zoom: f64) -> f32 {
117    2.0_f64.powf(zoom - zoom.floor()) as f32
118}

120/// The image's colours multiplied by its alpha, which filters without a dark fringe.
121pub(super) fn premultiplied(data: &[u8]) -> Vec<u8> {
122    let mut out = data.to_vec();
123    for texel in out.as_chunks_mut::<4>().0 {
124        let alpha = u16::from(texel[3]);
125        for channel in &mut texel[..3] {
126            *channel = ((u16::from(*channel) * alpha + 127) / 255) as u8;
127        }
128    }
129    out
130}

132fn size(display: [f32; 2], scale: f32) -> [f32; 4] {
133    [display[0] * scale, display[1] * scale, 0.0, 0.0]
134}

136fn fingerprint(name: &str, image: &StyleImage) -> u64 {
137    let mut hasher = std::hash::DefaultHasher::new();
138    name.hash(&mut hasher);
139    (image.width, image.height, image.sdf).hash(&mut hasher);
140    image.pixel_ratio.to_bits().hash(&mut hasher);
141    image.data.hash(&mut hasher);
142    hasher.finish()
143}

145impl PatternResources {
146    /// Creates the bindings, whose layout is [`layout_entries`], without a fill pipeline.
147    pub(crate) fn new(device: &wgpu::Device, layout: wgpu::BindGroupLayout) -> Self {
148        let sampler = device.create_sampler(&wgpu::SamplerDescriptor {
149            address_mode_u: wgpu::AddressMode::Repeat,
150            address_mode_v: wgpu::AddressMode::Repeat,
151            mag_filter: wgpu::FilterMode::Linear,
152            min_filter: wgpu::FilterMode::Linear,
153            ..Default::default()
154        });
155        Self {
156            pipeline: None,
157            layout,
158            layers: HashMap::new(),
159            per_feature: Default::default(),
160            images: HashMap::new(),
161            sampler,
162        }
163    }
164
165    /// Binds the image each pattern layer names at `zoom`, dropping layers that no longer
166    /// repeat an image the style holds.
167    pub(crate) fn update(
168        &mut self,
169        device: &wgpu::Device,
170        queue: &crate::render::upload_queue::UploadQueue,
171        style: &Style,
172        zoom: f64,
173    ) {
174        let scale = pattern_scale(zoom);
175        self.layers
176            .retain(|id, _| style.layers.iter().any(|layer| &layer.id == id));
177        self.per_feature = style
178            .layers
179            .iter()
180            .filter(|layer| layer.paint.as_ref().and_then(per_feature_pattern).is_some())
181            .map(|layer| layer.id.clone())
182            .collect();
183        self.update_images(device, queue, style, scale);
184        for layer in &style.layers {
185            let image = layer
186                .paint
187                .as_ref()
188                .and_then(|paint| pattern_name(paint, zoom))
189                .and_then(|name| style.images.get(&name).map(|image| (name, image)));
190            let Some((name, image)) = image else {
191                self.layers.remove(&layer.id);
192                continue;
193            };
194            let fingerprint = fingerprint(&name, image);
195            if let Some(binding) = self
196                .layers
197                .get_mut(&layer.id)
198                .filter(|binding| binding.fingerprint == fingerprint)
199            {
200                if binding.scale != scale {
201                    binding.scale = scale;
202                    queue.write_buffer(
203                        &binding.size,
204                        0,
205                        bytemuck::cast_slice(&size(binding.display, scale)),
206                    );
207                }
208                continue;
209            }
210            if let Some(binding) = self.bind(device, queue, image, (fingerprint, scale)) {
211                self.layers.insert(layer.id.clone(), binding);
212            }
213        }
214    }
215
216    /// Keeps a binding for every image of the style while a layer picks its image per feature.
217    fn update_images(
218        &mut self,
219        device: &wgpu::Device,
220        queue: &crate::render::upload_queue::UploadQueue,
221        style: &Style,
222        scale: f32,
223    ) {
224        if self.per_feature.is_empty() {
225            self.images.clear();
226            return;
227        }
228        let mut keys = std::collections::HashSet::new();
229        for (name, image) in &style.images {
230            let key = crate::style::pattern_key::pattern_key(name);
231            keys.insert(key);
232            let fingerprint = fingerprint(name, image);
233            if let Some(binding) = self
234                .images
235                .get_mut(&key)
236                .filter(|binding| binding.fingerprint == fingerprint)
237            {
238                if binding.scale != scale {
239                    binding.scale = scale;
240                    queue.write_buffer(
241                        &binding.size,
242                        0,
243                        bytemuck::cast_slice(&size(binding.display, scale)),
244                    );
245                }
246                continue;
247            }
248            match self.bind(device, queue, image, (fingerprint, scale)) {
249                Some(binding) => {
250                    self.images.insert(key, binding);
251                }
252                None => {
253                    self.images.remove(&key);
254                }
255            }
256        }
257        self.images.retain(|key, _| keys.contains(key));
258    }

260    /// Whether the layer picks its image per feature.
261    pub(crate) fn is_per_feature(&self, layer: &str) -> bool {
262        self.per_feature.contains(layer)
263    }

265    /// The image a pattern key names, for a layer that picks its image per feature.
266    pub(crate) fn image_binding(&self, key: u32) -> Option<&wgpu::BindGroup> {
267        self.images.get(&key).map(|binding| &binding.bind_group)
268    }
269
270    fn bind(
271        &self,
272        device: &wgpu::Device,
273        queue: &crate::render::upload_queue::UploadQueue,
274        image: &StyleImage,
275        (fingerprint, scale): (u64, f32),
276    ) -> Option<PatternBinding> {
277        if image.width == 0
278            || image.height == 0
279            || image.data.len() != image.width as usize * image.height as usize * 4
280        {
281            return None;
282        }
283        let texture = queue.create_texture_with_data(
284            device,
285            &wgpu::TextureDescriptor {
286                label: Some("fill pattern"),
287                size: wgpu::Extent3d {
288                    width: image.width,
289                    height: image.height,
290                    depth_or_array_layers: 1,
291                },
292                mip_level_count: 1,
293                sample_count: 1,
294                dimension: wgpu::TextureDimension::D2,
295                format: wgpu::TextureFormat::Rgba8Unorm,
296                usage: wgpu::TextureUsages::TEXTURE_BINDING,
297                view_formats: &[],
298            },
299            wgpu::util::TextureDataOrder::LayerMajor,
300            &premultiplied(&image.data),
301        );
302        let ratio = image.pixel_ratio.max(0.01);
303        let display = [image.width as f32 / ratio, image.height as f32 / ratio];
304        let size = queue.create_buffer_init(
305            device,
306            &wgpu::util::BufferInitDescriptor {
307                label: Some("fill pattern size"),
308                contents: bytemuck::cast_slice(&size(display, scale)),
309                usage: wgpu::BufferUsages::UNIFORM | wgpu::BufferUsages::COPY_DST,
310            },
311        );
312        let view = texture.create_view(&Default::default());
313        let bind_group = device.create_bind_group(&wgpu::BindGroupDescriptor {
314            label: Some("fill pattern"),
315            layout: &self.layout,
316            entries: &[
317                wgpu::BindGroupEntry {
318                    binding: 0,
319                    resource: size.as_entire_binding(),
320                },
321                wgpu::BindGroupEntry {
322                    binding: 1,
323                    resource: wgpu::BindingResource::TextureView(&view),
324                },
325                wgpu::BindGroupEntry {
326                    binding: 2,
327                    resource: wgpu::BindingResource::Sampler(&self.sampler),
328                },
329            ],
330        });
331        Some(PatternBinding {
332            fingerprint,
333            bind_group,
334            size,
335            display,
336            scale,
337        })
338    }
