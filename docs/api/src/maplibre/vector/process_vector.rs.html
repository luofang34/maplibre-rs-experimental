39/// Failure decoding vector tile bytes or returning processed results.
40#[derive(Error, Debug)]
41pub enum ProcessVectorError {
42    /// Sending of results failed
43    #[error("sending data back through context failed")]
44    SendError(#[source] SendError),
45    /// The source bytes are not a decodable vector tile.
46    #[error("decoding vector tile {coords} failed: {source}")]
47    Decoding {
48        /// Tile whose protocol buffer could not be decoded.
49        coords: WorldTileCoords,
50        /// Underlying protocol-buffer decoder failure.
51        #[source]
52        source: Box<dyn std::error::Error>,
53    },
54    /// A glyph range or sprite could not be fetched, so the tile's symbols would be incomplete.
55    #[error("loading symbol assets failed")]
56    SymbolAssets(#[source] SymbolAssetError),
57}

59/// Scale from a layer's declared coordinate extent to the 4096 grid the shaders expect, so a
60/// producer emitting 8192 keeps its precision instead of landing twice as far from the origin.
61pub fn extent_scale(layer: &tile::Layer) -> f64 {
62    match layer.extent {
63        Some(extent) if extent > 0 => EXTENT / f64::from(extent),
64        _ => 1.0,
65    }
66}

68/// A request for a tile at the given coordinates and in the given layers.
69#[derive(Clone)]
70pub struct VectorTileRequest {
71    /// Tile grid used to scale its geometry and index.
72    pub coords: WorldTileCoords,
73    /// Style entries to evaluate against the source layers.
74    pub layers: HashSet<StyleLayer>,
75    /// Projection controlling globe subdivision of tile geometry.
76    pub projection: ProjectionType,
77    /// The zoom the tile is drawn at when that is past the last zoom of its source, so its
78    /// geometry is magnified; zero when the tile is drawn at its own zoom.
79    pub overscaled_zoom: u8,
80}
81
82impl VectorTileRequest {
83    /// The zoom style expressions are evaluated at: the zoom the tile is magnified to, which is
84    /// the tile's own when it is not.
85    pub fn style_zoom(&self) -> f64 {
86        f64::from(self.overscaled_zoom.max(u8::from(self.coords.z)))
87    }
88
89    /// How many times the tile is magnified past its source's last zoom.
90    pub fn overscaling(&self) -> f64 {
91        let own = u8::from(self.coords.z);
92        2_f64.powi(i32::from(self.overscaled_zoom.saturating_sub(own)))
93    }
94}

96/// Reads an MVT feature's tags into typed filter values through the layer's key and value
97/// tables.
98pub(crate) fn feature_properties(
99    layer: &tile::Layer,
100    feature: &tile::Feature,
101) -> FeatureProperties {
102    let mut properties = FeatureProperties::new();
103    for pair in feature.tags.chunks(2) {
104        let [key_index, value_index] = pair else {
105            continue;
106        };
107        let (Some(key), Some(value)) = (
108            layer.keys.get(*key_index as usize),
109            layer.values.get(*value_index as usize),
110        ) else {
111            continue;
112        };
113        let value = if let Some(text) = &value.string_value {
114            Value::String(text.clone())
115        } else if let Some(number) = value.float_value {
116            Value::Number(f64::from(number))
117        } else if let Some(number) = value.double_value {
118            Value::Number(number)
119        } else if let Some(number) = value.int_value {
120            Value::Number(number as f64)
121        } else if let Some(number) = value.uint_value {
122            Value::Number(number as f64)
123        } else if let Some(number) = value.sint_value {
124            Value::Number(number as f64)
125        } else if let Some(flag) = value.bool_value {
126            Value::Bool(flag)
127        } else {
128            continue;
129        };
130        properties.insert(key.clone(), value);
131    }
132    properties
133}

135/// Keeps only the features of an MVT layer that pass the filter.
136fn apply_filter_to_layer(layer: &mut tile::Layer, filter: &Filter, zoom: f64) {
137    let keep: Vec<bool> = layer
138        .features
139        .iter()
140        .map(|feature| {
141            let properties = feature_properties(layer, feature);
142            filter.evaluate(&FeatureContext {
143                properties: &properties,
144                geometry_type: GeometryType::from_mvt(feature.r#type.unwrap_or_default()),
145                id: feature.id.map(|id| Value::Number(id as f64)),
146                zoom,
147            })
148        })
149        .collect();
150    let mut keep = keep.into_iter();
151    layer.features.retain(|_| keep.next().unwrap_or(false));
152}

154/// Decodes and processes a tile with the built-in fallback symbol atlas.
155/// Results stream through the context; decoding and delivery failures retain their causes.
156pub fn process_vector_tile<T: VectorTransferables, C: Context>(

166    )

172/// Decodes a tile, tolerating the zero padding some tile servers append: a zero tag is not a
173/// field, so a strict decoder rejects a tile that GL JS reads.
174pub(super) fn decode_tile(data: &[u8]) -> Result<geozero::mvt::Tile, Box<dyn std::error::Error>> {
175    // A payload that is not a protobuf tile with layers is tried as a MapLibre Tile.
176    #[cfg(feature = "mlt")]
177    match decode_mvt(data) {
178        Ok(tile) if !tile.layers.is_empty() => Ok(tile),
179        Ok(tile) => Ok(super::mlt::decode(data).unwrap_or(tile)),
180        Err(error) => super::mlt::decode(data).map_err(|_| error),
181    }
182    #[cfg(not(feature = "mlt"))]
183    decode_mvt(data)
184}
185
186fn decode_mvt(data: &[u8]) -> Result<geozero::mvt::Tile, Box<dyn std::error::Error>> {
187    geozero::mvt::Tile::decode(data).or_else(|error| {
188        let end = data
189            .iter()
190            .rposition(|byte| *byte != 0)
191            .map_or(0, |i| i + 1);
192        if end == data.len() {
193            return Err(error.into());
194        }
195        geozero::mvt::Tile::decode(&data[..end]).map_err(|_| error.into())
196    })
197}

199pub(crate) fn process_vector_tile_with_assets<T: VectorTransferables, C: Context>(
200    data: &[u8],
201    tile_request: VectorTileRequest,
202    context: &mut ProcessVectorContext<T, C>,
203    atlas: std::sync::Arc<crate::sdf::assets::SymbolAtlas>,
204) -> Result<(), ProcessVectorError> {
205    let mut tile = decode_tile(data).map_err(|source| ProcessVectorError::Decoding {
206        coords: tile_request.coords,
207        source,
208    })?;
209    for style in &tile_request.layers {
210        let (Some(_), Some(name)) = (&style.paint, &style.source_layer) else {
211            tracing::error!(layer = %style.id, "vector style layer misses a required attribute");
212            continue;
213        };
214        // Omitted source layers must clear successful replacements just like empty geometry.
215        let layer = tile
216            .layers
217            .iter()
218            .find(|layer| &layer.name == name)
219            .cloned()
220            .unwrap_or_else(|| tile::Layer {
221                version: 2,
222                name: name.clone(),
223                extent: Some(4096),
224                ..Default::default()
225            });
226        processing::process_layer(layer, style, &tile_request, context, atlas.clone())?;
227    }
228    let mut index = IndexProcessor::new();
229    for layer in &mut tile.layers {
230        index.set_coordinate_scale(extent_scale(layer));
231        index.begin_layer(
232            &layer.name,
233            layer.features.iter().map(|feature| feature.id).collect(),
234        );
235        // Query decoding cannot make successfully rendered geometry unavailable.
236        if let Err(error) = layer.process(&mut index) {
237            tracing::warn!(coords = %tile_request.coords, layer = %layer.name, ?error, "skipping query index for layer");
238        }
239    }
240    let source = tile_request
241        .layers
242        .iter()
243        .find_map(|layer| layer.source.clone());
244    context.layer_indexing_finished(&tile_request.coords, source, index.get_geometries())?;
245    context.tile_finished(&tile_request.coords)?;
246    Ok(())
247}

255/// Reply context and completion mode used while processing vector layers.
256pub struct ProcessVectorContext<T: VectorTransferables, C: Context> {
257    context: C,
258    completion: Completion,
259    phantom_t: PhantomData<T>,
260}
261
262impl<T: VectorTransferables, C: Context> ProcessVectorContext<T, C> {
263    /// Uses the supplied reply endpoint and completes the tile after this processing batch.
264    pub fn new(context: C) -> Self {
265        Self {
266            context,
267            completion: Completion::Finished,
268            phantom_t: Default::default(),
269        }
270    }
271}
272
273impl<T: VectorTransferables, C: Context> ProcessVectorContext<T, C> {
274    pub(crate) fn with_pending_symbols(mut self) -> Self {
275        self.completion = Completion::PendingSymbols;
276        self
277    }
278
279    pub(crate) fn without_completion(mut self) -> Self {
280        self.completion = Completion::Deferred;
281        self
282    }
283
284    /// Returns the owned reply endpoint after processing.
285    pub fn take_context(self) -> C {
286        self.context
287    }
288
289    fn tile_finished(&mut self, coords: &WorldTileCoords) -> Result<(), ProcessVectorError> {
290        let message = match self.completion {
291            Completion::Deferred => return Ok(()),
292            Completion::PendingSymbols => T::TileTessellated::build_partial(*coords),
293            Completion::Finished => T::TileTessellated::build_from(*coords),
294        };
295        self.context
296            .send_back(message)
297            .map_err(ProcessVectorError::SendError)
298    }
299
300    fn layer_missing(
301        &mut self,
302        coords: &WorldTileCoords,
303        layer_name: &str,
304    ) -> Result<(), ProcessVectorError> {
305        self.context
306            .send_back(T::LayerMissing::build_from(*coords, layer_name.to_owned()))
307            .map_err(ProcessVectorError::SendError)
308    }
309
310    fn layer_tessellation_finished(
311        &mut self,
312        coords: &WorldTileCoords,
313        buffer: OverAlignedVertexBuffer<ShaderVertex, IndexDataType>,
314        feature_indices: Vec<u32>,
315        (feature_colors, feature_sort_keys): (Vec<[f32; 4]>, Vec<f32>),
316        layer_data: tile::Layer,
317        style_layer_id: String,
318    ) -> Result<(), ProcessVectorError> {
319        self.context
320            .send_back(T::LayerTessellated::build_from(
321                *coords,
322                buffer,
323                feature_indices,
324                feature_colors,
325                feature_sort_keys,
326                layer_data,
327                style_layer_id,
328            ))
329            .map_err(ProcessVectorError::SendError)
330    }
331
332    fn symbol_layer_tessellation_finished(
333        &mut self,
334        layer: crate::vector::transferables::DefaultSymbolLayerTessellated,
335    ) -> Result<(), ProcessVectorError> {
336        self.context
337            .send_back(T::SymbolLayerTessellated::build_from(
338                layer.coords,
339                layer.buffer,
340                layer.features,
341                layer.atlas,
342                layer.layer_data,
343                layer.style_layer_id,
344            ))
345            .map_err(ProcessVectorError::SendError)
346    }
347
348    fn layer_indexing_finished(
349        &mut self,
350        coords: &WorldTileCoords,
351        source: Option<String>,
352        geometries: Vec<IndexedGeometry<f64>>,
353    ) -> Result<(), ProcessVectorError> {
354        self.context
355            .send_back(T::LayerIndexed::build_from(
356                *coords,
357                source,
358                TileIndex::Linear { list: geometries },
359            ))
360            .map_err(ProcessVectorError::SendError)
361    }
362}
