21pub fn resource_system(
22    MapContext {
23        world,
24        style,
25        view_state,
26        renderer:
27            Renderer {
28                adapter,
29                device,
30                queue,
31                resources: RenderResources { surface, .. },
32                settings,
33                ..
34            },
35        ..
36    }: &mut MapContext,
37) -> SystemResult {
38    let style_zoom = view_state.style_zoom().value();
39    let terrain_active = world
40        .resources
41        .get::<crate::terrain::TerrainFrame>()
42        .is_some_and(|frame| frame.active);
43    let mut prepared = world
44        .resources
45        .remove::<VectorPreparedState>()
46        .unwrap_or_default();
47    prepared.sync_style(style, terrain_active);
48    let update_dashes = prepared.should_update_dashes(style);
49    let update_patterns = prepared.should_update_patterns(style);
50    let needs_fill = prepared.needs_fill;
51    let needs_flat_lines = prepared.needs_flat_lines;
52    let needs_terrain_lines = prepared.needs_terrain_lines;
53    let needs_elevated_lines = prepared.needs_elevated_lines;
54    let needs_circle = prepared.needs_circle;
55    let needs_circle_feature_stroke = needs_circle && prepared.needs_circle_feature_stroke;
56    let needs_circle_dynamic_radius = needs_circle && prepared.needs_circle_dynamic_radius;
57    let uses_globe = style.uses_globe_at(style_zoom);
58    let terrain_needed = style.terrain.is_some() || terrain_active;
59    let needs_extrusion = prepared.needs_extrusion_at_zoom(style_zoom, uses_globe);
60    let needs_extrusion_opaque = needs_extrusion && prepared.needs_extrusion_opaque;
61    let needs_extrusion_pattern = needs_extrusion && prepared.needs_extrusion_pattern;
62    let needs_extrusion_translucent = needs_extrusion && prepared.needs_extrusion_translucent;
63    let (has_compact_extrusion, has_standard_extrusion) = if needs_extrusion {
64        detect_extrusion_encodings(world, &prepared, style_zoom, terrain_needed)
65    } else {
66        (false, false)
67    };
68    let extrusion_needs = extrusion_setup::ExtrusionPipelineNeeds {
69        globe: uses_globe,
70        standard_opaque: needs_extrusion_opaque && has_standard_extrusion,
71        compact_opaque: needs_extrusion_opaque && has_compact_extrusion,
72        pattern: needs_extrusion_opaque && needs_extrusion_pattern && has_standard_extrusion,
73        standard_translucent: needs_extrusion_translucent && has_standard_extrusion,
74        compact_translucent: needs_extrusion_translucent && has_compact_extrusion,
75        clear_mark: needs_extrusion_opaque && (has_compact_extrusion || has_standard_extrusion),
76    };
77    let has_pattern_paint = prepared.has_pattern_paint;
78    let needs_plain_flat_lines = needs_flat_lines && (!uses_globe || terrain_needed);
79    let needs_plain_globe_lines = needs_flat_lines && uses_globe;
80    let needs_full_flat_lines = needs_flat_lines && prepared.has_dash_or_pattern_line;
81    let needs_flat_fill_opaque =
82        needs_fill && prepared.needs_fill_opaque && (!uses_globe || terrain_needed);
83    let needs_flat_fill_blended =
84        needs_fill && prepared.needs_fill_blended && (!uses_globe || terrain_needed);
85    let needs_globe_fill_opaque = needs_fill && prepared.needs_fill_opaque && uses_globe;
86    let needs_globe_fill_blended = needs_fill && prepared.needs_fill_blended && uses_globe;
87    world.resources.insert(prepared);
88
89    let needs_any_lines = needs_flat_lines || needs_terrain_lines || needs_elevated_lines;
90    if needs_any_lines
91        && world
92            .resources
93            .get::<super::line_dash::LineDashResources>()
94            .is_none()
95    {
96        world
97            .resources
98            .insert(super::line_dash::LineDashResources::new(device, queue));
99    }
100    if update_dashes && needs_any_lines {
101        if let Some(dashes) = world
102            .resources
103            .get_mut::<super::line_dash::LineDashResources>()
104        {
105            dashes.update(device, queue, style, style_zoom);
106        }
107    }
108    let dashes_layout = world
109        .resources
110        .get::<super::line_dash::LineDashResources>()
111        .map(|dashes| dashes.layout.clone());
112    if needs_any_lines && dashes_layout.is_none() {
113        return Err(SystemError::Dependencies);
114    }
115    let pattern_res = world.resources.get::<super::pattern::PatternResources>();
116    let missing_pattern_resources = pattern_res.is_none();
117    let missing_pattern_pipeline = pattern_res.is_none_or(|patterns| patterns.pipeline().is_none());
118    let supports_base_vertex = world
119        .resources
120        .get::<crate::render::tracked_pass::DrawCapabilities>()
121        .map_or_else(
122            || {
123                crate::render::tracked_pass::DrawCapabilities::from_adapter_and_device(
124                    adapter, device,
125                )
126                .base_vertex
127            },
128            |caps| caps.base_vertex,
129        );
130    let Some((
131        buffer_pool,
132        vector_pipeline,
133        line_pipeline,
134        circle_pipeline,
135        extrusion_pipeline,
136        Initialized(projection_resources),
137    )) = world.resources.query_mut::<(
138        &mut Eventually<VectorBufferPool>,
139        &mut Eventually<VectorPipeline>,
140        &mut Eventually<LinePipeline>,
141        &mut Eventually<CirclePipeline>,
142        &mut Eventually<ExtrusionPipeline>,
143        &mut Eventually<ProjectionGpuResources>,
144    )>()
145    else {
146        return Err(SystemError::Dependencies);
147    };
148
149    buffer_pool.initialize(|| {
150        BufferPool::from_device_with_sizes(device, settings.buffer_pools)
151            .with_u16_chunks(supports_base_vertex)
152            .with_decoupled_feature_metadata()
153            .with_shared_layer_metadata()
154    });
155
156    let need_vector_pipeline = needs_fill
157        && !matches!(vector_pipeline, Initialized(vp) if
158            (!needs_flat_fill_opaque || vp.flat_opaque.is_some())
159            && (!needs_flat_fill_blended || vp.flat_blended.is_some())
160            && (!needs_globe_fill_opaque || vp.globe_opaque.is_some())
161            && (!needs_globe_fill_blended || vp.globe_blended.is_some()));
162    let need_line_pipeline = needs_any_lines
163        && !matches!(line_pipeline, Initialized(lp) if
164            (!needs_plain_flat_lines || lp.flat_plain.is_some())
165            && (!needs_plain_globe_lines || lp.globe_plain.is_some())
166            && (!needs_full_flat_lines || lp.flat.is_some())
167            && (!needs_terrain_lines || (lp.structure.is_some() && lp.terrain.is_some()))
168            && (!needs_elevated_lines || lp.elevated.is_some()));
169    let need_base_pipelines = need_vector_pipeline || need_line_pipeline;
170    let need_circle_pipeline = needs_circle
171        && !matches!(circle_pipeline, Initialized(cp) if
172            (!needs_circle_feature_stroke || cp.feature_stroke.is_some())
173            && (!needs_circle_dynamic_radius || cp.dynamic_radius.is_some()));
174    let need_extrusion_pipeline = needs_extrusion
175        && (has_compact_extrusion || has_standard_extrusion)
176        && !matches!(extrusion_pipeline, Initialized(ep) if
177        (!extrusion_needs.clear_mark || ep.clear_mark.is_some())
178            && (!(extrusion_needs.standard_opaque
179                || extrusion_needs.compact_opaque
180                || extrusion_needs.pattern
181                || extrusion_needs.standard_translucent
182                || extrusion_needs.compact_translucent)
183                || ep.for_globe(uses_globe).is_some_and(|mode| {
184                    (!extrusion_needs.standard_opaque
185                        || (mode.depth.is_some() && mode.color.is_some()))
186                        && (!extrusion_needs.compact_opaque
187                            || (mode.compact_depth.is_some() && mode.compact_color.is_some()))
188                        && (!extrusion_needs.pattern || mode.pattern.is_some())
189                        && (!extrusion_needs.standard_translucent
190                            || (mode.translucent_back.is_some()
191                                && mode.translucent_front.is_some()))
192                        && (!extrusion_needs.compact_translucent
193                            || (mode.compact_translucent_back.is_some()
194                                && mode.compact_translucent_front.is_some()))
195                })));
196    let needs_pattern = missing_pattern_pipeline
197        && has_pattern_paint
198        && style.layers.iter().any(|layer| {
199            layer.paint.as_ref().is_some_and(|paint| {
200                super::pattern::pattern_name(paint, style_zoom).is_some()
201                    || super::pattern::per_feature_pattern(paint).is_some()
202            })
203        });
204
205    let pattern_pipeline = if need_base_pipelines
206        || need_circle_pipeline
207        || need_extrusion_pipeline
208        || needs_pattern
209        || missing_pattern_resources
210    {
211        let setup = PipelineSetup {
212            device,
213            settings: *settings,
214            format: surface.surface_format(),
215            multisampling: surface.is_multisampling_supported(settings.msaa),
216            projection: projection_resources.bind_group_layout(),
217            pattern: crate::render::resource::create_bind_group_layout_cached(
218                device,
219                Some("pattern layout"),
220                &super::pattern::layout_entries(),
221            ),
222        };
223        if need_base_pipelines || need_circle_pipeline {
224            setup.initialize(
225                vector_pipeline,
226                line_pipeline,
227                circle_pipeline,
228                dashes_layout.as_ref(),
229                setup::VectorPipelineNeeds {
230                    flat_fill_opaque: needs_flat_fill_opaque,
231                    flat_fill_blended: needs_flat_fill_blended,
232                    globe_fill_opaque: needs_globe_fill_opaque,
233                    globe_fill_blended: needs_globe_fill_blended,
234                    plain_flat_lines: needs_plain_flat_lines,
235                    plain_globe_lines: needs_plain_globe_lines,
236                    flat_lines: needs_full_flat_lines,
237                    terrain_lines: needs_terrain_lines,
238                    elevated_lines: needs_elevated_lines,
239                    circle: needs_circle,
240                    circle_feature_stroke: needs_circle_feature_stroke,
241                    circle_dynamic_radius: needs_circle_dynamic_radius,
242                },
243            );
244        }
245        if need_extrusion_pipeline {
246            setup.initialize_extrusion(extrusion_pipeline, extrusion_needs);
247        }
248        let created_pattern = needs_pattern.then(|| setup.create_pattern());
249        let pattern_layout = setup.pattern;
250        if missing_pattern_resources {
251            world
252                .resources
253                .insert(super::pattern::PatternResources::new(
254                    device,
255                    pattern_layout,
256                ));
257        }
258        created_pattern
259    } else {
260        None
261    };
262    if let (Some(pipeline), Some(patterns)) = (
263        pattern_pipeline,
264        world
265            .resources
266            .get_mut::<super::pattern::PatternResources>(),
267    ) {
268        patterns.set_pipeline(pipeline);
269    }
270    if update_patterns {
271        if let Some(patterns) = world
272            .resources
273            .get_mut::<super::pattern::PatternResources>()
274        {
275            patterns.update(device, queue, style, style_zoom);
276        }
277    }
278    Ok(())
279}

281fn detect_extrusion_encodings(
282    world: &crate::tcs::world::World,
283    prepared: &VectorPreparedState,
284    style_zoom: f64,
285    terrain_needed: bool,
286) -> (bool, bool) {
287    let mut needs_compact = false;
288    let mut needs_standard = false;
289    if let Some(Eventually::Initialized(pool)) =
290        world.resources.get::<Eventually<VectorBufferPool>>()
291    {
292        for entry in pool.index().iter().flatten() {
293            if entry.style_layer.type_ == "fill-extrusion"
294                && entry.style_layer.is_visible_at(style_zoom)
295            {
296                if entry.is_compact_extrusion() {
297                    needs_compact = true;
298                } else {
299                    needs_standard = true;
300                }
301                if needs_compact && needs_standard {
302                    return (true, true);
303                }
304            }
305        }
306    }
307    if world
308        .tiles
309        .pyramid_of::<crate::vector::VectorLayerBucketComponent>()
310        .is_empty()
311    {
312        return (needs_compact, needs_standard);
313    }
314    let inspect_bucket = |bucket: &crate::vector::AvailableVectorLayerBucket,
315                          needs_compact: &mut bool,
316                          needs_standard: &mut bool| {
317        if !bucket.buffer.has_drawable_geometry()
318            || !prepared.is_visible_extrusion_layer(&bucket.style_layer_id, style_zoom)
319        {
320            return;
321        }
322        if let Some(pre) = bucket.buffer.prepacked_gpu() {
323            if pre.compact_extrusion {
324                *needs_compact = true;
325                if terrain_needed || !bucket.feature_colors.is_empty() {
326                    *needs_standard = true;
327                }
328            } else {
329                *needs_standard = true;
330            }
331        } else if !bucket.buffer.buffer.vertices.is_empty() {
332            if !terrain_needed
333                && !prepared.needs_extrusion_pattern
334                && bucket.feature_colors.is_empty()
335                && crate::render::shaders::can_encode_compact_extrusion(
336                    &bucket.buffer.buffer.vertices,
337                )
338            {
339                *needs_compact = true;
340            } else {
341                *needs_standard = true;
342            }
343        } else {
344            *needs_compact = true;
345            *needs_standard = true;
346        }
347    };
348    for tile in world.tiles.tiles.values() {
349        if let Some(buckets) = world
350            .tiles
351            .query::<&crate::vector::VectorLayerBucketComponent>(tile.coords)
352        {
353            for layer in &buckets.layers {
354                if let crate::vector::VectorLayerBucket::AvailableLayer(bucket) = layer {
355                    inspect_bucket(bucket, &mut needs_compact, &mut needs_standard);
356                    if needs_compact && needs_standard {
357                        return (true, true);
358                    }
359                }
360            }
361        }
362        if let Some(pending) = world
363            .tiles
364            .query::<&crate::vector::content::LayerReplacements>(tile.coords)
365        {
366            for bucket in &pending.vector {
367                inspect_bucket(bucket, &mut needs_compact, &mut needs_standard);
368                if needs_compact && needs_standard {
369                    return (true, true);
370                }
371            }
372        }
373    }
374    (needs_compact, needs_standard)
