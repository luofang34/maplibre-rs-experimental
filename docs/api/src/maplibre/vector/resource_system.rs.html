19/// The stencil bit no tile reference uses.
20const STENCIL_MARK: u32 = 0x80;
21
22pub fn resource_system(
23    MapContext {
24        world,
25        style,
26        view_state,
27        renderer:
28            Renderer {
29                device,
30                queue,
31                resources: RenderResources { surface, .. },
32                settings,
33                ..
34            },
35        ..
36    }: &mut MapContext,
37) -> SystemResult {
38    if world
39        .resources
40        .get::<super::line_dash::LineDashResources>()
41        .is_none()
42    {
43        world
44            .resources
45            .insert(super::line_dash::LineDashResources::new(device, queue));
46    }
47    if let Some(dashes) = world
48        .resources
49        .get_mut::<super::line_dash::LineDashResources>()
50    {
51        dashes.update(device, queue, style, view_state.style_zoom().value());
52    }
53    let Some(dashes_layout) = world
54        .resources
55        .get::<super::line_dash::LineDashResources>()
56        .map(|dashes| dashes.layout.clone())
57    else {
58        return Err(SystemError::Dependencies);
59    };
60    let missing_pattern_pipeline = world
61        .resources
62        .get::<super::pattern::PatternResources>()
63        .is_none_or(|patterns| patterns.pipeline().is_none());
64    let Some((
65        buffer_pool,
66        vector_pipeline,
67        line_pipeline,
68        circle_pipeline,
69        extrusion_pipeline,
70        Initialized(projection_resources),
71    )) = world.resources.query_mut::<(
72        &mut Eventually<VectorBufferPool>,
73        &mut Eventually<VectorPipeline>,
74        &mut Eventually<LinePipeline>,
75        &mut Eventually<CirclePipeline>,
76        &mut Eventually<ExtrusionPipeline>,
77        &mut Eventually<ProjectionGpuResources>,
78    )>()
79    else {
80        return Err(SystemError::Dependencies);
81    };
82
83    buffer_pool.initialize(|| BufferPool::from_device_with_sizes(device, settings.buffer_pools));
84
85    let setup = PipelineSetup {
86        device,
87        settings: *settings,
88        format: surface.surface_format(),
89        multisampling: surface.is_multisampling_supported(settings.msaa),
90        projection: projection_resources.bind_group_layout(),
91        pattern: device.create_bind_group_layout(&wgpu::BindGroupLayoutDescriptor {
92            label: Some("pattern layout"),
93            entries: &super::pattern::layout_entries(),
94        }),
95    };
96    setup.initialize(
97        vector_pipeline,
98        line_pipeline,
99        circle_pipeline,
100        &dashes_layout,
101    );
102    setup.initialize_extrusion(extrusion_pipeline);
103    let pattern_pipeline = missing_pattern_pipeline.then(|| setup.create_pattern());
104    let pattern_layout = setup.pattern.clone();
105    if world
106        .resources
107        .get::<super::pattern::PatternResources>()
108        .is_none()
109    {
110        world
111            .resources
112            .insert(super::pattern::PatternResources::new(
113                device,
114                pattern_layout,
115            ));
116    }
117    if let (Some(pipeline), Some(patterns)) = (
118        pattern_pipeline,
119        world
120            .resources
121            .get_mut::<super::pattern::PatternResources>(),
122    ) {
123        patterns.set_pipeline(pipeline);
124    }
125    if let Some(patterns) = world
126        .resources
127        .get_mut::<super::pattern::PatternResources>()
128    {
129        patterns.update(device, queue, style, view_state.style_zoom().value());
130    }
131    Ok(())
132}

134struct PipelineSetup<'a> {
135    device: &'a wgpu::Device,
136    settings: crate::render::settings::RendererSettings,
137    format: wgpu::TextureFormat,
138    multisampling: bool,
139    projection: &'a wgpu::BindGroupLayout,
140    /// The second bind group of every pipeline that draws an image.
141    pattern: wgpu::BindGroupLayout,
142}
143
144impl PipelineSetup<'_> {
145    fn create(
146        &self,
147        name: &'static str,
148        shader: &impl Shader,
149        layouts: &[&wgpu::BindGroupLayout],
150        depth: bool,
151    ) -> wgpu::RenderPipeline {
152        self.create_with_clipping(name, shader, layouts, depth, true)
153    }
154
155    /// `clipped` keeps the pipeline inside its tile's stencil mask.
156    fn create_with_clipping(
157        &self,
158        name: &'static str,
159        shader: &impl Shader,
160        layouts: &[&wgpu::BindGroupLayout],
161        depth: bool,
162        clipped: bool,
163    ) -> wgpu::RenderPipeline {
164        let pipeline = TilePipeline::new(
165            name.into(),
166            self.settings,
167            shader.describe_vertex(),
168            shader.describe_fragment(),
169            crate::render::resource::TilePipelineOptions {
170                depth_stencil_enabled: true,
171                update_stencil: false,
172                debug_stencil: !clipped,
173                wireframe: false,
174                multisampling: self.multisampling,
175                textured: false,
176            },
177        );
178        let pipeline = if depth {
179            pipeline.with_depth_write()
180        } else {
181            pipeline
182        };
183        let mut descriptor = pipeline.describe_render_pipeline();
184        if depth {
185            // Casing and deck are paint layers on the same road surface. Test against
186            // terrain, but let style order composite them without self-occluding edges.
187            if let Some(state) = &mut descriptor.depth_stencil {
188                state.depth_write_enabled = Some(false);
189            }
190        }
191        descriptor.initialize_with_prefix_layouts(self.device, layouts)
192    }
193
194    fn create_pattern(&self) -> wgpu::RenderPipeline {
195        self.create(
196            "fill_pattern_pipeline",
197            &shaders::FillPatternShader {
198                format: self.format,
199            },
200            &[self.projection, &self.pattern],
201            false,
202        )
203    }
204
205    fn create_extrusion(&self, pass: shaders::ExtrusionPass) -> wgpu::RenderPipeline {
206        use shaders::ExtrusionPass::{Clear, Color, Depth, PatternColor};
207
208        let shader = shaders::FillExtrusionShader {
209            format: self.format,
210            pass,
211        };
212        let mut descriptor = TilePipeline::new(
213            match pass {
214                Depth => "extrusion_depth_pipeline",
215                Color => "extrusion_color_pipeline",
216                PatternColor => "extrusion_pattern_pipeline",
217                Clear => "extrusion_clear_pipeline",
218            }
219            .into(),
220            self.settings,
221            shader.describe_vertex(),
222            shader.describe_fragment(),
223            crate::render::resource::TilePipelineOptions {
224                depth_stencil_enabled: true,
225                update_stencil: false,
226                debug_stencil: true,
227                wireframe: false,
228                multisampling: self.multisampling,
229                textured: false,
230            },
231        )
232        .with_depth_write()
233        .describe_render_pipeline();
234        if let Some(state) = &mut descriptor.depth_stencil {
235            state.depth_write_enabled = Some(pass == Depth);
236            state.depth_compare = Some(match pass {
237                Depth | Color | PatternColor => wgpu::CompareFunction::GreaterEqual,
238                Clear => wgpu::CompareFunction::Always,
239            });
240            // Tile references use the low seven bits; the top bit marks the pixels the colour
241            // pass has drawn, so a surface that two tiles both hold is blended once.
242            let marked = wgpu::StencilFaceState {
243                compare: wgpu::CompareFunction::Equal,
244                fail_op: wgpu::StencilOperation::Keep,
245                depth_fail_op: wgpu::StencilOperation::Keep,
246                pass_op: wgpu::StencilOperation::Invert,
247            };
248            let unmark = wgpu::StencilFaceState {
249                compare: wgpu::CompareFunction::Always,
250                fail_op: wgpu::StencilOperation::Keep,
251                depth_fail_op: wgpu::StencilOperation::Keep,
252                pass_op: wgpu::StencilOperation::Zero,
253            };
254            match pass {
255                Depth => {}
256                Color | PatternColor => state.stencil.front = marked,
257                Clear => state.stencil.front = unmark,
258            }
259            state.stencil.back = state.stencil.front;
260            if pass != Depth {
261                state.stencil.read_mask = STENCIL_MARK;
262                state.stencil.write_mask = STENCIL_MARK;
263            }
264        }
265        let layouts: &[&wgpu::BindGroupLayout] = if pass == PatternColor {
266            &[self.projection, &self.pattern]
267        } else {
268            &[self.projection]
269        };
270        descriptor.initialize_with_prefix_layouts(self.device, layouts)
271    }
272
273    fn initialize_extrusion(&self, extrusion: &mut Eventually<ExtrusionPipeline>) {
274        extrusion.initialize(|| ExtrusionPipeline {
275            depth: self.create_extrusion(shaders::ExtrusionPass::Depth),
276            color: self.create_extrusion(shaders::ExtrusionPass::Color),
277            clear: self.create_extrusion(shaders::ExtrusionPass::Clear),
278            pattern: self.create_extrusion(shaders::ExtrusionPass::PatternColor),
279        });
280    }
281
282    fn initialize(
283        &self,
284        vector: &mut Eventually<VectorPipeline>,
285        line: &mut Eventually<LinePipeline>,
286        circle: &mut Eventually<CirclePipeline>,
287        dashes: &wgpu::BindGroupLayout,
288    ) {
289        vector.initialize(|| {
290            VectorPipeline(self.create(
291                "vector_pipeline",
292                &shaders::FillShader {
293                    format: self.format,
294                },
295                &[self.projection],
296                false,
297            ))
298        });
299        line.initialize(|| {
300            let shader = shaders::LineShader {
301                format: self.format,
302            };
303            LinePipeline(
304                self.create("line_pipeline", &shader, &[self.projection, dashes], false),
305                self.create(
306                    "spatial_line_pipeline",
307                    &shader,
308                    &[self.projection, dashes],
309                    true,
310                ),
311            )
312        });
313        circle.initialize(|| {
314            // Circles cross tile boundaries, as in GL JS, where each point belongs to one tile.
315            CirclePipeline(self.create_with_clipping(
316                "circle_pipeline",
317                &shaders::CircleShader {
318                    format: self.format,
319                },
320                &[self.projection],
321                false,
322                false,
323            ))
324        });
325    }
