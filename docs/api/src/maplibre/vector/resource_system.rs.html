6use crate::{
7    context::MapContext,
8    render::{

12    },

14    vector::{
15        resource::BufferPool, CirclePipeline, ExtrusionPipeline, LinePipeline, VectorBufferPool,

20pub fn resource_system(
21    MapContext {
22        world,
23        style,
24        view_state,
25        renderer:
26            Renderer {
27                adapter,
28                device,
29                queue,
30                resources: RenderResources { surface, .. },
31                settings,
32                ..
33            },
34        ..
35    }: &mut MapContext,
36) -> SystemResult {
37    let style_zoom = view_state.style_zoom().value();
38    let terrain_active = world
39        .resources
40        .get::<crate::terrain::TerrainFrame>()
41        .is_some_and(|frame| frame.active);
42    let mut prepared = world
43        .resources
44        .remove::<VectorPreparedState>()
45        .unwrap_or_default();
46    prepared.sync_style(style, terrain_active);
47    let update_dashes = prepared.should_update_dashes(style);
48    let update_patterns = prepared.should_update_patterns(style);
49    let needs_fill = prepared.needs_fill;
50    let needs_flat_lines = prepared.needs_flat_lines;
51    let needs_terrain_lines = prepared.needs_terrain_lines;
52    let needs_elevated_lines = prepared.needs_elevated_lines;
53    let needs_circle = prepared.needs_circle;
54    let needs_circle_feature_stroke = needs_circle && prepared.needs_circle_feature_stroke;
55    let needs_circle_dynamic_radius = needs_circle && prepared.needs_circle_dynamic_radius;
56    let needs_extrusion =
57        prepared.needs_extrusion_at_zoom(style_zoom, style.uses_globe_at(style_zoom));
58    let needs_extrusion_pattern = needs_extrusion && prepared.needs_extrusion_pattern;
59    let needs_extrusion_translucent = needs_extrusion && prepared.needs_extrusion_translucent;
60    let has_pattern_paint = prepared.has_pattern_paint;
61    world.resources.insert(prepared);
62
63    let needs_any_lines = needs_flat_lines || needs_terrain_lines || needs_elevated_lines;
64    if needs_any_lines
65        && world
66            .resources
67            .get::<super::line_dash::LineDashResources>()
68            .is_none()
69    {
70        world
71            .resources
72            .insert(super::line_dash::LineDashResources::new(device, queue));
73    }
74    if update_dashes && needs_any_lines {
75        if let Some(dashes) = world
76            .resources
77            .get_mut::<super::line_dash::LineDashResources>()
78        {
79            dashes.update(device, queue, style, style_zoom);
80        }
81    }
82    let dashes_layout = world
83        .resources
84        .get::<super::line_dash::LineDashResources>()
85        .map(|dashes| dashes.layout.clone());
86    if needs_any_lines && dashes_layout.is_none() {
87        return Err(SystemError::Dependencies);
88    }
89    let pattern_res = world.resources.get::<super::pattern::PatternResources>();
90    let missing_pattern_resources = pattern_res.is_none();
91    let missing_pattern_pipeline = pattern_res.is_none_or(|patterns| patterns.pipeline().is_none());
92    let supports_base_vertex = world
93        .resources
94        .get::<crate::render::tracked_pass::DrawCapabilities>()
95        .map_or_else(
96            || {
97                crate::render::tracked_pass::DrawCapabilities::from_adapter_and_device(
98                    adapter, device,
99                )
100                .base_vertex
101            },
102            |caps| caps.base_vertex,
103        );
104    let Some((
105        buffer_pool,
106        vector_pipeline,
107        line_pipeline,
108        circle_pipeline,
109        extrusion_pipeline,
110        Initialized(projection_resources),
111    )) = world.resources.query_mut::<(
112        &mut Eventually<VectorBufferPool>,
113        &mut Eventually<VectorPipeline>,
114        &mut Eventually<LinePipeline>,
115        &mut Eventually<CirclePipeline>,
116        &mut Eventually<ExtrusionPipeline>,
117        &mut Eventually<ProjectionGpuResources>,
118    )>()
119    else {
120        return Err(SystemError::Dependencies);
121    };
122
123    buffer_pool.initialize(|| {
124        BufferPool::from_device_with_sizes(device, settings.buffer_pools)
125            .with_u16_chunks(supports_base_vertex)
126            .with_decoupled_feature_metadata()
127            .with_shared_layer_metadata()
128    });
129
130    let need_vector_pipeline = needs_fill && !matches!(vector_pipeline, Initialized(_));
131    let need_line_pipeline = needs_any_lines
132        && !matches!(line_pipeline, Initialized(lp) if
133            (!needs_flat_lines || lp.flat.is_some())
134            && (!needs_terrain_lines || (lp.structure.is_some() && lp.terrain.is_some()))
135            && (!needs_elevated_lines || lp.elevated.is_some()));
136    let need_base_pipelines = need_vector_pipeline || need_line_pipeline;
137    let need_circle_pipeline = needs_circle
138        && !matches!(circle_pipeline, Initialized(cp) if
139            (!needs_circle_feature_stroke || cp.feature_stroke.is_some())
140            && (!needs_circle_dynamic_radius || cp.dynamic_radius.is_some()));
141    let need_extrusion_pipeline = needs_extrusion
142        && !matches!(extrusion_pipeline, Initialized(ep) if
143            (!needs_extrusion_pattern || ep.pattern.is_some())
144            && (!needs_extrusion_translucent
145                || (ep.translucent_back.is_some() && ep.translucent_front.is_some())));
146    let needs_pattern = missing_pattern_pipeline
147        && has_pattern_paint
148        && style.layers.iter().any(|layer| {
149            layer.paint.as_ref().is_some_and(|paint| {
150                super::pattern::pattern_name(paint, style_zoom).is_some()
151                    || super::pattern::per_feature_pattern(paint).is_some()
152            })
153        });
154
155    let pattern_pipeline = if need_base_pipelines
156        || need_circle_pipeline
157        || need_extrusion_pipeline
158        || needs_pattern
159        || missing_pattern_resources
160    {
161        let setup = PipelineSetup {
162            device,
163            settings: *settings,
164            format: surface.surface_format(),
165            multisampling: surface.is_multisampling_supported(settings.msaa),
166            projection: projection_resources.bind_group_layout(),
167            pattern: crate::render::resource::create_bind_group_layout_cached(
168                device,
169                Some("pattern layout"),
170                &super::pattern::layout_entries(),
171            ),
172        };
173        if need_base_pipelines || need_circle_pipeline {
174            setup.initialize(
175                vector_pipeline,
176                line_pipeline,
177                circle_pipeline,
178                dashes_layout.as_ref(),
179                setup::VectorPipelineNeeds {
180                    fill: needs_fill,
181                    flat_lines: needs_flat_lines,
182                    terrain_lines: needs_terrain_lines,
183                    elevated_lines: needs_elevated_lines,
184                    circle: needs_circle,
185                    circle_feature_stroke: needs_circle_feature_stroke,
186                    circle_dynamic_radius: needs_circle_dynamic_radius,
187                },
188            );
189        }
190        if need_extrusion_pipeline {
191            setup.initialize_extrusion(
192                extrusion_pipeline,
193                needs_extrusion_pattern,
194                needs_extrusion_translucent,
195            );
196        }
197        let created_pattern = needs_pattern.then(|| setup.create_pattern());
198        let pattern_layout = setup.pattern;
199        if missing_pattern_resources {
200            world
201                .resources
202                .insert(super::pattern::PatternResources::new(
203                    device,
204                    pattern_layout,
205                ));
206        }
207        created_pattern
208    } else {
209        None
210    };
211    if let (Some(pipeline), Some(patterns)) = (
212        pattern_pipeline,
213        world
214            .resources
215            .get_mut::<super::pattern::PatternResources>(),
216    ) {
217        patterns.set_pipeline(pipeline);
218    }
219    if update_patterns {
220        if let Some(patterns) = world
221            .resources
222            .get_mut::<super::pattern::PatternResources>()
223        {
224            patterns.update(device, queue, style, style_zoom);
225        }
226    }
227    Ok(())
228}
