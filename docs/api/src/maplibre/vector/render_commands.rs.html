17#[inline]
18fn resolve_vector_entry<'w>(
19    world: &'w World,
20    item: &LayerItem,
21) -> Option<&'w crate::vector::resource::IndexEntry> {
22    let Initialized(pool) = world.resources.get::<Eventually<VectorBufferPool>>()? else {
23        return None;
24    };
25    pool.index()
26        .resolve_entry(item.tile.coords, item.style_layer, item.entry_slot)
27}

29fn is_uniform_entry(world: &World, item: &LayerItem) -> bool {
30    resolve_vector_entry(world, item)
31        .is_some_and(crate::vector::resource::IndexEntry::is_uniform_feature_metadata)
32}

34/// Binds the polygon-fill pipeline and the item's view or flat projection.
35pub struct SetVectorTilePipeline;
36impl RenderCommand<LayerItem> for SetVectorTilePipeline {
37    fn render<'w>(
38        world: &'w World,
39        item: &LayerItem,
40        pass: &mut crate::render::tracked_pass::TrackedRenderPass<'w>,
41    ) -> RenderCommandResult {
42        let Some((Initialized(pipeline), Initialized(projection_resources))) =
43            world.resources.query::<(
44                &Eventually<VectorPipeline>,
45                &Eventually<ProjectionGpuResources>,
46            )>()
47        else {
48            return RenderCommandResult::Failure;
49        };
50
51        let entry = resolve_vector_entry(world, item);
52        let globe = projection_resources.is_globe_active(item.projection_binding());
53        let opaque = entry.is_some_and(crate::vector::resource::IndexEntry::is_opaque_color);
54        let uniform =
55            entry.is_some_and(crate::vector::resource::IndexEntry::is_uniform_feature_metadata);
56        let Some(selected) = pipeline.select(globe, opaque, uniform) else {
57            return RenderCommandResult::Failure;
58        };
59
60        pass.set_pipeline(selected);
61        pass.set_bind_group(
62            0,
63            projection_resources.bind_group_for(item.projection_binding()),
64            &[],
65        );
66        RenderCommandResult::Success
67    }
68}

70/// Draws the tile's matching style-layer geometry with tile, layer and feature metadata.
71///
72/// With `RUNS`, a layer whose pattern varies by feature is drawn one run of features at a
73/// time, each with the image its pattern names bound at the second bind group.
74pub struct DrawVectorTile<const RUNS: bool = false>;
75impl<const RUNS: bool> RenderCommand<LayerItem> for DrawVectorTile<RUNS> {
76    fn render<'w>(
77        world: &'w World,
78        item: &LayerItem,
79        pass: &mut crate::render::tracked_pass::TrackedRenderPass<'w>,
80    ) -> RenderCommandResult {
81        let Some((Initialized(buffer_pool), Initialized(tile_view_pattern))) =
82            world.resources.query::<(
83                &Eventually<VectorBufferPool>,
84                &Eventually<WgpuTileViewPattern>,
85            )>()
86        else {
87            return RenderCommandResult::Failure;
88        };
89
90        let Some(entry) =
91            buffer_pool
92                .index()
93                .resolve_entry(item.tile.coords, item.style_layer, item.entry_slot)
94        else {
95            return RenderCommandResult::Failure;
96        };
97
98        let source_shape = &item.source_shape;
99
100        let reference = u32::from(source_shape.stencil_reference());
101
102        tracing::trace!(
103            "Drawing layer {:?} at {}",
104            entry.style_layer.source_layer,
105            entry.coords
106        );
107
108        let index_range = entry.indices_buffer_range();
109
110        if index_range.is_empty() {
111            tracing::error!("Tried to draw a vector tile without any vertices");
112            return RenderCommandResult::Failure;
113        }
114
115        pass.set_stencil_reference(reference);
116
117        let whole = world
118            .resources
119            .get::<crate::render::tracked_pass::DrawCapabilities>()
120            .is_some_and(|capabilities| capabilities.base_vertex);
121        let (first, base_vertex) = buffer_pool.bind_layer(pass, entry, (3, whole));
122        let instances = if whole {
123            let Some(tile_instance) = source_shape.instance_index() else {
124                return RenderCommandResult::Failure;
125            };
126            pass.set_vertex_buffer(1, tile_view_pattern.buffer().slice(..));
127            tile_instance..tile_instance + 1
128        } else {
129            let Some(tile_view_pattern_buffer) = source_shape.buffer_range() else {
130                return RenderCommandResult::Failure;
131            };
132            pass.set_vertex_buffer(
133                1,
134                tile_view_pattern.buffer().slice(tile_view_pattern_buffer),
135            );
136            0..1
137        };
138        pass.set_vertex_buffer(
139            2,
140            buffer_pool
141                .metadata_for(entry)
142                .slice(entry.layer_metadata_buffer_range()),
143        );
144        if let Some(run) = &item.run {
145            pass.draw_indexed(first + run.start..first + run.end, base_vertex, instances);
146            return RenderCommandResult::Success;
147        }
148        let runs = RUNS
149            .then(|| buffer_pool.pattern_runs_by_id(item.tile.coords, item.style_layer))
150            .flatten();
151        let Some(runs) = runs else {
152            if whole {
153                if let Some(chunks) = entry.u16_chunks() {
154                    for (v_off, range) in chunks {
155                        pass.draw_indexed(
156                            first + range.start..first + range.end,
157                            base_vertex + (*v_off as i32),
158                            instances.clone(),
159                        );
160                    }
161                    return RenderCommandResult::Success;
162                }
163            }
164            pass.draw_indexed(
165                first..first + entry.indices_range().len() as u32,
166                base_vertex,
167                instances,
168            );
169            return RenderCommandResult::Success;
170        };
171        let patterns = world.resources.get::<super::pattern::PatternResources>();
172        let dashes = world.resources.get::<super::line_dash::LineDashResources>();
173        let is_line = entry.style_layer.type_ == "line";
174        for (key, range) in runs {
175            let image = if is_line {
176                dashes.and_then(|dashes| dashes.image_binding(*key))
177            } else {
178                patterns.and_then(|patterns| patterns.image_binding(*key))
179            };
180            let Some(image) = image else {
181                continue;
182            };
183            pass.set_bind_group(1, image, &[]);
184            pass.draw_indexed(
185                first + range.start..first + range.end,
186                base_vertex,
187                instances.clone(),
188            );
189        }
190
191        RenderCommandResult::Success
192    }
193}

195/// Where a line lies this frame, which decides its pipeline and how it is drawn.
196#[derive(Clone, Copy, PartialEq, Eq)]
197enum LineMode {
198    /// On the map plane, or draped into the terrain textures.
199    Flat,
200    /// At heights of its own over the terrain.
201    Structure,
202    /// At `line-elevation` altitudes in 3D space without 2D ground stencil clipping.
203    Elevated,
204    /// On the terrain surface at its width in screen pixels.
205    Terrain,
206}
207
208fn line_mode(world: &World, item: &LayerItem) -> LineMode {
209    let kind = world
210        .resources
211        .get::<Eventually<VectorBufferPool>>()
212        .and_then(|pool| match pool {
213            Initialized(pool) => {
214                pool.index()
215                    .resolve_entry(item.tile.coords, item.style_layer, item.entry_slot)
216            }
217            _ => None,
218        })
219        .and_then(|entry| super::structures::kind(&entry.style_layer));
220    if matches!(kind, Some(super::structures::StructureKind::ElevatedLine)) {
221        return LineMode::Elevated;
222    }
223    let terrain = world
224        .resources
225        .get::<crate::terrain::TerrainFrame>()
226        .is_some_and(|frame| frame.active);
227    if !terrain {
228        return LineMode::Flat;
229    }
230    match kind {
231        None => LineMode::Flat,
232        Some(super::structures::StructureKind::Overlay) => LineMode::Terrain,
233        Some(_) => LineMode::Structure,
234    }
235}

237/// Binds the line pipeline and dash atlas; elevated structures and lines on the terrain surface
238/// use depth-aware drawing on terrain.
239pub struct SetLineTilePipeline;
240impl RenderCommand<LayerItem> for SetLineTilePipeline {
241    fn render<'w>(
242        world: &'w World,
243        item: &LayerItem,
244        pass: &mut crate::render::tracked_pass::TrackedRenderPass<'w>,
245    ) -> RenderCommandResult {
246        let Some((Initialized(pipeline), Initialized(projection_resources))) =
247            world.resources.query::<(
248                &Eventually<LinePipeline>,
249                &Eventually<ProjectionGpuResources>,
250            )>()
251        else {
252            return RenderCommandResult::Failure;
253        };
254
255        let mode = line_mode(world, item);
256        let dashes = world.resources.get::<super::line_dash::LineDashResources>();
257        let (chosen, needs_dash_group) = match mode {
258            LineMode::Flat => {
259                let is_plain = item.run.is_none()
260                    && dashes.is_some_and(|dashes| dashes.is_plain_solid(item.style_layer));
261                if is_plain {
262                    let plain = if projection_resources.is_globe_active(item.projection_binding()) {
263                        pipeline.globe_plain.as_ref()
264                    } else {
265                        pipeline.flat_plain.as_ref()
266                    };
267                    if let Some(plain) = plain {
268                        (Some(plain), false)
269                    } else {
270                        (pipeline.flat.as_ref(), true)
271                    }
272                } else {
273                    (pipeline.flat.as_ref(), true)
274                }
275            }
276            LineMode::Structure => (pipeline.structure.as_ref(), true),
277            LineMode::Elevated => (pipeline.elevated.as_ref(), true),
278            LineMode::Terrain => (pipeline.terrain.as_ref(), true),
279        };
280        let Some(chosen) = chosen else {
281            return RenderCommandResult::Failure;
282        };
283        pass.set_pipeline(chosen.select(is_uniform_entry(world, item)));
284        pass.set_bind_group(
285            0,
286            projection_resources.bind_group_for(item.projection_binding()),
287            &[],
288        );
289        if needs_dash_group {
290            let Some(dashes) = dashes else {
291                return RenderCommandResult::Failure;
292            };
293            pass.set_bind_group(1, dashes.binding(item.style_layer), &[]);
294        }
295        RenderCommandResult::Success
296    }
297}

329/// Binds the pattern pipeline and the pattern of the item's layer.
330pub struct SetPatternTilePipeline;
331impl RenderCommand<LayerItem> for SetPatternTilePipeline {
332    fn render<'w>(
333        world: &'w World,
334        item: &LayerItem,
335        pass: &mut crate::render::tracked_pass::TrackedRenderPass<'w>,
336    ) -> RenderCommandResult {
337        let (Some(patterns), Some(Initialized(projection_resources))) = (
338            world.resources.get::<super::pattern::PatternResources>(),
339            world.resources.get::<Eventually<ProjectionGpuResources>>(),
340        ) else {
341            return RenderCommandResult::Failure;
342        };
343        let binding = patterns.binding(item.style_layer);
344        if binding.is_none() && !patterns.is_per_feature(item.style_layer) {
345            return RenderCommandResult::Failure;
346        }
347        let Some(pipeline) = patterns.pipeline() else {
348            return RenderCommandResult::Failure;
349        };
350        pass.set_pipeline(pipeline.select(is_uniform_entry(world, item)));
351        pass.set_bind_group(
352            0,
353            projection_resources.bind_group_for(item.projection_binding()),
354            &[],
355        );
356        if let Some(binding) = binding {
357            pass.set_bind_group(1, binding, &[]);
358        }
359        RenderCommandResult::Success
360    }
361}

365/// Binds the image of the item's layer at the second bind group.

378            None if patterns.is_per_feature(item.style_layer) => {}
379            None => return RenderCommandResult::Failure,
380        }
381        RenderCommandResult::Success
382    }
383}

385/// Binds and draws a polygon-fill bucket when both commands succeed.
386pub type DrawVectorTiles = (SetVectorTilePipeline, DrawVectorTile);
387/// Draws a stroked-line bucket. A line on the terrain is drawn once over each terrain tile its
388/// tile overlaps, with that tile's surface bound, since its vertices take their height from it.
389pub struct DrawLineTile;
390impl RenderCommand<LayerItem> for DrawLineTile {
391    fn render<'w>(
392        world: &'w World,
393        item: &LayerItem,
394        pass: &mut crate::render::tracked_pass::TrackedRenderPass<'w>,
395    ) -> RenderCommandResult {
396        if line_mode(world, item) != LineMode::Terrain {
397            return DrawVectorTile::<true>::render(world, item, pass);
398        }
399        let Some(Initialized(terrain)) = world
400            .resources
401            .get::<Eventually<crate::terrain::resources::TerrainResources>>()
402        else {
403            return RenderCommandResult::Failure;
404        };
405        let tile = item.source_shape.coords();
406        for draw in terrain
407            .draws()
408            .iter()
409            .filter(|draw| overlaps(draw.coords, tile))
410        {
411            pass.set_bind_group(2, draw.bind_group.as_ref(), &[draw.uniform_offset]);
412            if let RenderCommandResult::Failure = DrawVectorTile::<true>::render(world, item, pass)
413            {
414                return RenderCommandResult::Failure;
415            }
416        }
417        RenderCommandResult::Success
418    }
419}

421/// Whether one tile contains the other.
422fn overlaps(a: WorldTileCoords, b: WorldTileCoords) -> bool {
423    let (coarse, fine) = if u8::from(a.z) <= u8::from(b.z) {
424        (a, b)
425    } else {
426        (b, a)
427    };
428    let levels = u8::from(fine.z) - u8::from(coarse.z);
429    fine.x >> levels == coarse.x && fine.y >> levels == coarse.y
430}
