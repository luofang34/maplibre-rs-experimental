1//! Host-thread remapping for worker-emitted `shared_geometry_key` values.
2//!
3//! Single-threaded Web Workers run in isolated WASM linear memories whose process-local
4//! `AtomicU64` counters each start at `1`. Remapping `(coords, attempt, raw_key)` on arrival on
5//! the render thread prevents distinct worker attempts from colliding in `VectorBufferPool` while
6//! preserving sharing across sibling layers within the same tile attempt.
7
8use rustc_hash::FxHashMap;
9
10use crate::coords::WorldTileCoords;
11
12/// Maps `(coords, attempt, raw_worker_key)` to a render-thread unique `u64` key.
13pub(crate) struct SharedGeometryKeyRemap {
14    next_key: u64,
15    active: FxHashMap<WorldTileCoords, (Option<u64>, FxHashMap<u64, u64>)>,
16}
17
18impl Default for SharedGeometryKeyRemap {
19    fn default() -> Self {
20        Self {
21            next_key: 1,
22            active: FxHashMap::default(),
23        }
24    }
25}
26
27impl SharedGeometryKeyRemap {
28    /// Returns the host-unique shared key for `(coords, attempt, raw_key)`.
29    pub(crate) fn remap(
30        &mut self,
31        coords: WorldTileCoords,
32        attempt: Option<u64>,
33        raw_key: u64,
34    ) -> u64 {
35        let next = &mut self.next_key;
36        let (stored_attempt, table) = self
37            .active
38            .entry(coords)
39            .or_insert_with(|| (attempt, FxHashMap::default()));
40        if *stored_attempt != attempt {
41            *stored_attempt = attempt;
42            table.clear();
43        }
44        *table.entry(raw_key).or_insert_with(|| {
45            let key = *next;
46            *next = next.wrapping_add(1).max(1);
47            key
48        })
49    }
50
51    /// Releases remap entries once the active attempt for `coords` completes.
52    pub(crate) fn finish(&mut self, coords: WorldTileCoords, attempt: Option<u64>) {
53        if self
54            .active
55            .get(&coords)
56            .is_some_and(|(stored, _)| *stored == attempt)
57        {
58            self.active.remove(&coords);
59        }
60    }
61
62    #[cfg(all(test, feature = "headless", feature = "thread-safe-futures"))]
63    pub(crate) fn active_entry_count(&self) -> usize {
64        self.active.values().map(|(_, table)| table.len()).sum()
65    }
66}
