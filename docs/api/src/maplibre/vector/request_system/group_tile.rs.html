1//! Decoded vector tile of one source layer group: fetched from a URL or cut from a GeoJSON
2//! index, cached in the shared asset cache, and brought into the tile's grid from the source's
3//! last tile above it when the source ends sooner than the tile's zoom.

36pub(super) async fn fetch_raw_group_tile<H: HttpClient>(
37    client: &SourceClient<H>,
38    coords: WorldTileCoords,
39    group: &SourceLayerGroup,
40    refetch: bool,
41) -> Result<RawGroupTile, SourceFetchError> {
42    let source_tile = group.source_tile(coords);
43    #[cfg(feature = "geojson")]
44    if let SourceType::GeoJson(geojson) = &group.source {
45        let index = client
46            .assets()
47            .geojson_index(client, &geojson.name, &geojson.source)
48            .await
49            .map_err(|failure| fetch_failure(failure, &geojson.name))?;
50        let data = index.tile(source_tile);
51        return Ok(RawGroupTile {
52            source_tile,
53            payload: RawPayload::Owned {
54                bytes: data,
55                label: format!("geojson tile {}", geojson.name),
56            },
57        });
58    }
59    #[cfg(not(feature = "geojson"))]
60    if let SourceType::GeoJson(geojson) = &group.source {
61        return Err(fetch_failure(
62            AssetFailure::Terminal(format!(
63                "geojson source {} requires the `geojson` Cargo feature",
64                geojson.name
65            )),
66            &geojson.name,
67        ));
68    }
69    let Some(url) = group.source.format(&source_tile) else {
70        let bytes = client.fetch(&source_tile, &group.source).await?;
71        return Ok(RawGroupTile {
72            source_tile,
73            payload: RawPayload::Owned {
74                bytes,
75                label: format!("vector tile {source_tile}"),
76            },
77        });
78    };
79    // On wasm32, non-overscaled tiles (`source_tile == coords`) are already deduplicated per
80    // viewport by `RequestSystem` and cached by the browser's HTTP cache outside WASM linear
81    // memory, so fetching owned `Vec<u8>` avoids retaining compressed MVT copies in `AssetCache`
82    // and moves zero-copy into `Arc<[u8]>` for `LazyMvtIndex`.
83    #[cfg(target_arch = "wasm32")]
84    if source_tile == coords {
85        let bytes = client.fetch(&source_tile, &group.source).await?;
86        return Ok(RawGroupTile {
87            source_tile,
88            payload: RawPayload::Owned { bytes, label: url },
89        });
90    }
91    let key = format!("tile:{url}");
92    if refetch {
93        client.assets().forget(&key);
94    }
95    let cached_bytes: Arc<Vec<u8>> = client
96        .assets()
97        .load(key, || async {
98            let bytes = client
99                .fetch(&source_tile, &group.source)
100                .await
101                .map_err(|error| {
102                    tracing::warn!(%url, error = %error.describe(), "vector tile unavailable");
103                    if error.is_not_found() {
104                        AssetFailure::NotFound
105                    } else if error.is_retryable() {
106                        AssetFailure::Retryable(error.describe())
107                    } else {
108                        AssetFailure::Terminal(error.describe())
109                    }
110                })?;
111            let len = bytes.len();
112            Ok((bytes, len))
113        })
114        .await
115        .map_err(|failure| fetch_failure(failure, &url))?;
116    Ok(RawGroupTile {
117        source_tile,
118        payload: RawPayload::Cached {
119            bytes: cached_bytes,
120            url,
121        },
122    })
123}

125pub(super) fn decode_group_tile<H: HttpClient>(
126    client: &SourceClient<H>,
127    raw: RawGroupTile,
128    coords: WorldTileCoords,
129    layers: &[crate::style::layer::StyleLayer],
130) -> Result<GroupTile, SourceFetchError> {
131    let RawGroupTile {
132        source_tile,
133        payload,
134    } = raw;
135    let keep_layer = |name: &str| {
136        layers
137            .iter()
138            .any(|sl| sl.source_layer.as_deref() == Some(name))
139    };
140    match payload {
141        RawPayload::Owned { bytes, label } => {
142            let decode_res = if source_tile == coords {
143                decode_tile_for_layers(&bytes, keep_layer)
144            } else {
145                decode_tile(&bytes)
146            };
147            let mut tile = decode_res.map_err(|error| {
148                fetch_failure(
149                    AssetFailure::Terminal(format!("{label} cannot be decoded: {error}")),
150                    &label,
151                )
152            })?;
153            let (mvt_bytes, mvt_compressed) = if source_tile != coords {
154                super::rescale::into_descendant(&mut tile, source_tile, coords);
155                let packed =
156                    crate::io::geometry_index::LazyMvtIndex::pack_vec(tile.encode_to_vec());
157                tile.layers.retain(|mvt| keep_layer(mvt.name.as_str()));
158                packed
159            } else {
160                crate::io::geometry_index::LazyMvtIndex::pack_vec(bytes)
161            };
162            Ok(GroupTile {
163                tile: Arc::new(tile),
164                mvt_bytes,
165                mvt_compressed,
166            })
167        }
168        RawPayload::Cached { bytes, url } => {
169            let decode_res = if source_tile == coords {
170                decode_tile_for_layers(&bytes, keep_layer)
171            } else {
172                decode_tile(&bytes)
173            };
174            let mut tile = decode_res.map_err(|error| {
175                client.assets().forget(&format!("tile:{url}"));
176                fetch_failure(
177                    AssetFailure::Terminal(format!("vector tile {url} cannot be decoded: {error}")),
178                    &url,
179                )
180            })?;
181            let (mvt_bytes, mvt_compressed) = if source_tile != coords {
182                super::rescale::into_descendant(&mut tile, source_tile, coords);
183                let packed =
184                    crate::io::geometry_index::LazyMvtIndex::pack_vec(tile.encode_to_vec());
185                tile.layers.retain(|mvt| keep_layer(mvt.name.as_str()));
186                packed
187            } else {
188                crate::io::geometry_index::LazyMvtIndex::pack_slice(bytes.as_slice())
189            };
190            Ok(GroupTile {
191                tile: Arc::new(tile),
192                mvt_bytes,
193                mvt_compressed,
194            })
195        }
196    }
197}

199/// The fetch error a failure to load `what` through the asset cache stands for.
200fn fetch_failure(failure: AssetFailure, what: &str) -> SourceFetchError {
201    match failure {
202        AssetFailure::NotFound => SourceFetchError::not_found(what),
203        AssetFailure::Retryable(reason) => {
204            SourceFetchError::temporary(std::io::Error::other(reason))
205        }
206        AssetFailure::Terminal(reason) => SourceFetchError(Box::new(std::io::Error::other(reason))),
207    }
208}
