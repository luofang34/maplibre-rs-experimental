27pub fn fetch_vector_apc<K: OffscreenKernel, T: VectorTransferables, C: Context + Clone + Send>(
28    input: Input,
29    context: C,
30    kernel: K,
31) -> AsyncProcedureFuture {
32    Box::pin(async move {
33        let Input {
34            coords,
35            style,
36            attempt,
37            overscaled_zoom,
38            pixel_ratio,
39            only_sources,
40        } = input;
41        let context = AttemptContext::new(context, attempt);
42        let client = kernel.source_client();
43        // A request for a change to some sources' data leaves the tile's other sources as the
44        // tile shows them.
45        let mut groups = source_layer_groups(&style, TileKind::Vector)
46            .into_iter()
47            .filter(|group| {
48                only_sources.as_ref().is_none_or(|only| {
49                    group
50                        .source_name
51                        .as_ref()
52                        .is_some_and(|name| only.contains(name))
53                })
54            })
55            .peekable();
56        let mut failed = false;
57        let mut retry = false;
58        let mut images = provided::TileImages::new(coords, Some(attempt), pixel_ratio);
59        while let Some(group) = groups.next() {
60            if group.has_nothing_at(coords) {
61                source_empty::<T, _>(coords, &group.layers, &context)
62                    .map_err(ProcedureError::Send)?;
63                continue;
64            }
65            let tile = match fetch_group_tile(&client, coords, &group).await {
66                Ok(tile) => tile,
67                // A source without a tile there has nothing to draw, as GL JS loads a vector
68                // tile that answered 404 empty: its layers are empty, and the tile's other
69                // sources are drawn as usual.
70                Err(error) if error.is_not_found() => {
71                    tracing::debug!(%coords, source = ?group.source_name, "no vector tile at the source; its layers are empty");
72                    source_empty::<T, _>(coords, &group.layers, &context)
73                        .map_err(ProcedureError::Send)?;
74                    continue;
75                }
76                Err(error) => {
77                    tracing::warn!(%coords, source = ?group.source_name, error = %error.describe(), "vector tile unavailable");
78                    failed = true;
79                    retry |= error.is_retryable();
80                    source_failed::<T, _>(coords, &group.layers, &context)
81                        .map_err(ProcedureError::Send)?;
82                    continue;
83                }
84            };
85            match process_source::<T, _, K::HttpClient>(
86                tile,
87                coords,
88                &style,
89                &client,
90                &group,
91                context.clone(),
92                SourcePass {
93                    last_source: groups.peek().is_none() && !failed,
94                    overscaled_zoom,
95                    pixel_ratio,
96                },
97            )
98            .await
99            {
100                Ok(symbols) => images.add(symbols),
101                Err(ProcessVectorError::SendError(source)) => {
102                    return Err(ProcedureError::Send(source))
103                }
104                Err(error @ ProcessVectorError::SymbolAssets(_)) => {
105                    // The base layers were delivered, so the tile stays complete and drawn without
106                    // labels; only the retry outcome asks for another attempt.
107                    tracing::warn!(%coords, source = ?group.source_name, error = %error, "symbol assets unavailable");
108                    retry = true;
109                }
110                Err(
111                    error @ (ProcessVectorError::Decoding { .. } | ProcessVectorError::CpuWork(_)),
112                ) => {
113                    tracing::warn!(%coords, source = ?group.source_name, %error, "invalid vector tile");
114                    failed = true;
115                    source_failed::<T, _>(coords, &group.layers, &context)
116                        .map_err(ProcedureError::Send)?;
117                }
118            }
119        }
120        images
121            .report_first(&context)
122            .map_err(ProcedureError::Send)?;
123        finish::<T, _>(&context, coords, Some(attempt), failed, retry)?;
124        // The tile is complete and drawn; its labels are laid out again as images arrive.
125        images
126            .lay_out_again::<T, _, K::HttpClient>(&client, &style, context)
127            .await
128            .map_err(ProcedureError::Send)
129    })
130}

132/// How one source of a tile is processed.
133struct SourcePass {
134    /// Whether no other source follows, so this one completes the tile.
135    last_source: bool,
136    /// The zoom the tile is magnified to, or zero.
137    overscaled_zoom: u8,
138    /// Device pixels per layout pixel, which provided images are made for.
139    pixel_ratio: f32,
140}
141
142/// Reports every layer of a source as missing from the tile, which clears what an earlier
143/// generation drew there.
144fn source_empty<T: VectorTransferables, C: Context>(
145    coords: WorldTileCoords,
146    layers: &[StyleLayer],
147    context: &C,
148) -> Result<(), SendError> {
149    for layer in layers {
150        if let Some(source_layer) = &layer.source_layer {
151            context.send_back(T::LayerMissing::build_from(coords, source_layer.clone()))?;
152        }
153    }
154    Ok(())
155}

157fn source_failed<T: VectorTransferables, C: Context>(
158    coords: WorldTileCoords,
159    layers: &[StyleLayer],
160    context: &C,
161) -> Result<(), SendError> {
162    source_empty::<T, C>(coords, layers, context)?;
163    context.send_back(T::TileTessellated::build_failed(coords, true))
164}

166async fn process_source<T: VectorTransferables, C: Context + Clone + Send, H: HttpClient>(
167    tile: Arc<geozero::mvt::Tile>,
168    coords: WorldTileCoords,
169    style: &Style,
170    client: &SourceClient<H>,
171    group: &SourceLayerGroup,
172    context: C,
173    pass: SourcePass,
174) -> Result<Option<provided::Symbols>, ProcessVectorError> {
175    let SourcePass {
176        last_source,
177        overscaled_zoom,
178        pixel_ratio,
179    } = pass;
180    let projection = style
181        .projection
182        .as_ref()
183        .map_or_else(Default::default, |specification| {
184            specification.projection_type.clone()
185        });
186    let (symbols, base): (HashSet<_>, HashSet<_>) = group
187        .layers
188        .iter()
189        .cloned()
190        .partition(|layer| layer.type_ == "symbol");
191    let base_indexed = !base.is_empty();
192    let request = VectorTileRequest {
193        coords,
194        layers: base,
195        projection: projection.clone(),
196        overscaled_zoom,
197    };
198    tessellate::<T, C>(
199        context.clone(),
200        (last_source, true),
201        (tile.clone(), request),
202        Arc::new(SymbolAtlas::default()),
203    )
204    .await?;
205    if symbols.is_empty() {
206        return Ok(None);
207    }
208    // The assets are asked for by the very layers the symbols are laid out with: this source's,
209    // as its tile is cut.
210    let assets = load_symbol_assets_awaiting_from_tile(
211        client,
212        SymbolAssetConfig {
213            pixel_ratio,
214            ..SymbolAssetConfig::of(style)
215        },
216        &symbols,
217        &tile,
218        f64::from(overscaled_zoom.max(u8::from(coords.z))),
219    )
220    .await
221    .map_err(ProcessVectorError::SymbolAssets)?;
222    let request = VectorTileRequest {
223        coords,
224        layers: symbols,
225        projection,
226        overscaled_zoom,
227    };
228    // Only a source still waiting for images keeps its tile and layers to lay out again.
229    let again = (!assets.awaiting.is_empty()).then(|| (tile.clone(), request.clone()));
230    tessellate::<T, C>(
231        context,
232        (last_source, !base_indexed),
233        (tile, request),
234        assets.atlas,
235    )
236    .await?;
237    Ok(Some(provided::Symbols {
238        again,
239        pixel_ratio,
240        provided: assets.provided,
241        awaiting: assets.awaiting,
242    }))
243}

245/// Tessellates the requested layers of one source's decoded tile off the fetching threads,
246/// replying with their geometry as it is made.
247async fn tessellate<T: VectorTransferables, C: Context + Send>(
248    context: C,
249    (last_source, index_geometry): (bool, bool),
250    (tile, request): (Arc<geozero::mvt::Tile>, VectorTileRequest),
251    atlas: Arc<SymbolAtlas>,
252) -> Result<(), ProcessVectorError> {
253    run_cpu_bound_replying(context, move |context| {
254        let mut processor = source_processor::<T, _>(context, last_source);
255        if !index_geometry {
256            processor = processor.without_indexing();
257        }
258        process_decoded_vector_tile_with_assets(&tile, request, &mut processor, atlas)
259    })
260    .await
261    .map_err(ProcessVectorError::CpuWork)?
262}
263
264fn source_processor<T: VectorTransferables, C: Context>(
265    context: C,
266    last_source: bool,
267) -> ProcessVectorContext<T, C> {
268    let processor = ProcessVectorContext::new(context).with_pending_symbols();
269    // Earlier sources cannot replace an ancestor before the other sources' base layers arrive.
270    if last_source {
271        processor
272    } else {
273        processor.without_completion()
274    }
275}

277fn finish<T: VectorTransferables, C: Context>(
278    context: &C,
279    coords: WorldTileCoords,
280    attempt: Option<u64>,
281    failed: bool,
282    retry: bool,
283) -> Result<(), ProcedureError> {
284    let completion = if failed {
285        T::TileTessellated::build_failed(coords, false)
286    } else {
287        T::TileTessellated::build_from(coords)
288    };
289    context
290        .send_back(completion)
291        .map_err(ProcedureError::Send)?;
292    context
293        .send_back(TileRequestOutcome {
294            coords,
295            kind: RequestKind::Vector,
296            attempt,
297            disposition: if retry {
298                RequestDisposition::Retry
299            } else {
300                RequestDisposition::Complete
301            },
302        })
303        .map_err(ProcedureError::Send)
304}
