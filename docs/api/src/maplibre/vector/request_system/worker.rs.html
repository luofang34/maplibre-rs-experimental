34pub fn fetch_vector_apc<K: OffscreenKernel, T: VectorTransferables, C: Context + Clone + Send>(
35    input: Input,
36    context: C,
37    kernel: K,
38) -> AsyncProcedureFuture {
39    Box::pin(async move {
40        let Input {
41            coords,
42            style,
43            attempt,
44            overscaled_zoom,
45            pixel_ratio,
46            only_sources,
47            refetch,
48        } = input;
49        let context = AttemptContext::new(context, attempt);
50        let client = kernel.source_client();
51        // A request for a change to some sources' data leaves the tile's other sources as the
52        // tile shows them.
53        let has_terrain = style.terrain.is_some();
54        let mut groups = source_layer_groups(&style, TileKind::Vector)
55            .into_iter()
56            .filter_map(|mut group| {
57                let included = only_sources.as_ref().is_none_or(|only| {
58                    group
59                        .source_name
60                        .as_ref()
61                        .is_some_and(|name| only.contains(name))
62                });
63                if !included {
64                    return None;
65                }
66                group
67                    .layers
68                    .retain(|layer| layer.needed_for_tile(has_terrain, coords, overscaled_zoom));
69                (!group.layers.is_empty()).then_some(group)
70            })
71            .peekable();
72        let mut failed = false;
73        let mut retry = false;
74        let mut images = provided::TileImages::new(coords, Some(attempt), pixel_ratio);
75        while let Some(group) = groups.next() {
76            if group.has_nothing_at(coords) {
77                source_empty::<T, _>(coords, &group.layers, &context)
78                    .map_err(ProcedureError::Send)?;
79                continue;
80            }
81            let has_symbols = group.layers.iter().any(|layer| layer.type_ == "symbol");
82            enum FetchStep {
83                Ready(super::group_tile::RawGroupTile),
84                NotFound,
85                Failed { retryable: bool },
86            }
87            let fetch_step = {
88                #[cfg(target_arch = "wasm32")]
89                let raw_res = if has_symbols {
90                    let (raw_res, ()) = futures::future::join(
91                        fetch_raw_group_tile(&client, coords, &group, refetch),
92                        crate::sdf::assets::warm_style_symbol_assets(&client, &style),
93                    )
94                    .await;
95                    raw_res
96                } else {
97                    fetch_raw_group_tile(&client, coords, &group, refetch).await
98                };
99                #[cfg(not(target_arch = "wasm32"))]
100                let raw_res = {
101                    let _ = has_symbols;
102                    fetch_raw_group_tile(&client, coords, &group, refetch).await
103                };
104                match raw_res {
105                    Ok(raw_tile) => FetchStep::Ready(raw_tile),
106                    Err(error) if error.is_not_found() => {
107                        tracing::debug!(%coords, source = ?group.source_name, "no vector tile at the source; its layers are empty");
108                        FetchStep::NotFound
109                    }
110                    Err(error) => {
111                        tracing::warn!(%coords, source = ?group.source_name, error = %error.describe(), "vector tile unavailable");
112                        FetchStep::Failed {
113                            retryable: error.is_retryable(),
114                        }
115                    }
116                }
117            };
118            let raw_tile = match fetch_step {
119                FetchStep::Ready(raw_tile) => raw_tile,
120                // A source without a tile there has nothing to draw, as GL JS loads a vector
121                // tile that answered 404 empty: its layers are empty, and the tile's other
122                // sources are drawn as usual.
123                FetchStep::NotFound => {
124                    source_empty::<T, _>(coords, &group.layers, &context)
125                        .map_err(ProcedureError::Send)?;
126                    continue;
127                }
128                FetchStep::Failed { retryable } => {
129                    failed = true;
130                    retry |= retryable;
131                    source_failed::<T, _>(coords, &group.layers, &context)
132                        .map_err(ProcedureError::Send)?;
133                    continue;
134                }
135            };
136            super::limiter::wait_for_shared_tile_slot().await;
137            let mut heavy_slot = Some(acquire_heavy_tile_slot().await);
138            let tile = match decode_group_tile(&client, raw_tile, coords, &group.layers) {
139                Ok(tile) => tile,
140                Err(error) => {
141                    tracing::warn!(%coords, source = ?group.source_name, error = %error.describe(), "vector tile unavailable");
142                    failed = true;
143                    retry |= error.is_retryable();
144                    source_failed::<T, _>(coords, &group.layers, &context)
145                        .map_err(ProcedureError::Send)?;
146                    continue;
147                }
148            };
149            match process_source::<T, _, K::HttpClient>(
150                tile,
151                coords,
152                &style,
153                &client,
154                &group,
155                context.clone(),
156                (
157                    &mut heavy_slot,
158                    SourcePass {
159                        last_source: groups.peek().is_none() && !failed,
160                        overscaled_zoom,
161                        pixel_ratio,
162                    },
163                ),
164            )
165            .await
166            {
167                Ok(symbols) => images.add(symbols),
168                Err(ProcessVectorError::SendError(source)) => {
169                    return Err(ProcedureError::Send(source))
170                }
171                Err(error @ ProcessVectorError::SymbolAssets(_)) => {
172                    // The base layers were delivered, so the tile stays complete and drawn without
173                    // labels; only the retry outcome asks for another attempt.
174                    tracing::warn!(%coords, source = ?group.source_name, error = %error, "symbol assets unavailable");
175                    retry = true;
176                }
177                Err(
178                    error @ (ProcessVectorError::Decoding { .. } | ProcessVectorError::CpuWork(_)),
179                ) => {
180                    tracing::warn!(%coords, source = ?group.source_name, %error, "invalid vector tile");
181                    failed = true;
182                    source_failed::<T, _>(coords, &group.layers, &context)
183                        .map_err(ProcedureError::Send)?;
184                }
185            }
186        }
187        images
188            .report_first(&context)
189            .map_err(ProcedureError::Send)?;
190        finish::<T, _>(&context, coords, Some(attempt), failed, retry)?;
191        // The tile is complete and drawn; its labels are laid out again as images arrive.
192        images
193            .lay_out_again::<T, _, K::HttpClient>(&client, &style, context)
194            .await
195            .map_err(ProcedureError::Send)
196    })
197}

199/// How one source of a tile is processed.
200struct SourcePass {
201    /// Whether no other source follows, so this one completes the tile.
202    last_source: bool,
203    /// The zoom the tile is magnified to, or zero.
204    overscaled_zoom: u8,
205    /// Device pixels per layout pixel, which provided images are made for.
206    pixel_ratio: f32,
207}
208
209/// Reports every layer of a source as missing from the tile, which clears what an earlier
210/// generation drew there.
211fn source_empty<T: VectorTransferables, C: Context>(
212    coords: WorldTileCoords,
213    layers: &[StyleLayer],
214    context: &C,
215) -> Result<(), SendError> {
216    for layer in layers {
217        if let Some(source_layer) = &layer.source_layer {
218            context.send_back(T::LayerMissing::build_from(coords, source_layer.clone()))?;
219        }
220    }
221    Ok(())
222}
223
224fn source_failed<T: VectorTransferables, C: Context>(
225    coords: WorldTileCoords,
226    layers: &[StyleLayer],
227    context: &C,
228) -> Result<(), SendError> {
229    source_empty::<T, C>(coords, layers, context)?;
230    context.send_back(T::TileTessellated::build_failed(coords, true))
231}

233async fn process_source<T: VectorTransferables, C: Context + Clone + Send, H: HttpClient>(
234    group_tile: GroupTile,
235    coords: WorldTileCoords,
236    style: &Style,
237    client: &SourceClient<H>,
238    group: &SourceLayerGroup,
239    context: C,
240    (heavy_slot, pass): (&mut Option<super::limiter::HeavyTileSlot>, SourcePass),
241) -> Result<Option<provided::Symbols>, ProcessVectorError> {
242    let GroupTile {
243        mut tile,
244        mvt_bytes,
245        mvt_compressed,
246    } = group_tile;
247    let SourcePass {
248        last_source,
249        overscaled_zoom,
250        pixel_ratio,
251    } = pass;
252    let projection = style
253        .projection
254        .as_ref()
255        .map_or_else(Default::default, |specification| {
256            specification.projection_type.clone()
257        });
258    let (symbols, base): (HashSet<_>, HashSet<_>) = group
259        .layers
260        .iter()
261        .cloned()
262        .partition(|layer| layer.type_ == "symbol");
263    let source_state = group.source_name.as_deref().map_or((None, None), |name| {
264        (
265            style.source_promote_id(name).cloned(),
266            style.feature_state.source(name).cloned(),
267        )
268    });
269    if T::LayerIndexed::requires_geometry_index() && (!base.is_empty() || !symbols.is_empty()) {
270        let lazy = LazyMvtIndex::from_packed_with_state(
271            mvt_bytes,
272            mvt_compressed,
273            source_state.0.clone(),
274            source_state.1.clone(),
275        );
276        context
277            .send_back(T::LayerIndexed::build_from(
278                coords,
279                group.source_name.clone(),
280                TileIndex::LazyMvt(lazy),
281            ))
282            .map_err(ProcessVectorError::SendError)?;
283    }
284    let request = VectorTileRequest {
285        coords,
286        layers: base,
287        projection: projection.clone(),
288        overscaled_zoom,
289    };
290    if symbols.is_empty() {
291        tessellate::<T, C>(
292            context,
293            (last_source, false),
294            (tile, request),
295            source_state,
296            Arc::new(SymbolAtlas::default()),
297        )
298        .await?;
299        return Ok(None);
300    }
301    // Poll symbol asset fetches before base-layer CPU tessellation so remote glyph/sprite
302    // requests fly on the network concurrently with road/building tessellation.
303    let asset_config = SymbolAssetConfig {
304        pixel_ratio,
305        ..SymbolAssetConfig::of(style)
306    };
307    let zoom = f64::from(overscaled_zoom.max(u8::from(coords.z)));
308    let (assets_res, base_res) = {
309        let assets_fut =
310            load_symbol_assets_awaiting_from_tile(client, asset_config, &symbols, &tile, zoom);
311        let base_fut = tessellate::<T, C>(
312            context.clone(),
313            (last_source, false),
314            (tile.clone(), request),
315            source_state.clone(),
316            Arc::new(SymbolAtlas::default()),
317        );
318        futures::future::join(assets_fut, base_fut).await
319    };
320    base_res?;
321    let assets = assets_res.map_err(ProcessVectorError::SymbolAssets)?;
322    let request = VectorTileRequest {
323        coords,
324        layers: symbols,
325        projection,
326        overscaled_zoom,
327    };
328    // Only a source still waiting for images keeps its tile and layers to lay out again.
329    let again = (!assets.awaiting.is_empty()).then(|| (tile.clone(), request.clone()));
330    if again.is_none() {
331        if let Some(inner) = Arc::get_mut(&mut tile) {
332            inner.layers.retain(|mvt| {
333                request
334                    .layers
335                    .iter()
336                    .any(|sl| sl.source_layer.as_deref() == Some(mvt.name.as_str()))
337            });
338        }
339        let _ = heavy_slot.take();
340    }
341    tessellate::<T, C>(
342        context,
343        (last_source, false),
344        (tile, request),
345        source_state,
346        assets.atlas,
347    )
348    .await?;
349    Ok(Some(provided::Symbols {
350        again,
351        pixel_ratio,
352        provided: assets.provided,
353        awaiting: assets.awaiting,
354    }))
355}

357/// Tessellates the requested layers of one source's decoded tile off the fetching threads,
358/// replying with their geometry as it is made.
359async fn tessellate<T: VectorTransferables, C: Context + Send>(
360    context: C,
361    (last_source, index_geometry): (bool, bool),
362    (tile, request): (Arc<geozero::mvt::Tile>, VectorTileRequest),
363    (promote_id, feature_state): (
364        Option<crate::style::source::PromoteId>,
365        Option<crate::style::feature_state::SourceFeatureState>,
366    ),
367    atlas: Arc<SymbolAtlas>,
368) -> Result<(), ProcessVectorError> {
369    run_cpu_bound_replying(context, move |context| {
370        let mut processor = source_processor::<T, _>(context, last_source)
371            .with_source_feature_state(promote_id, feature_state);
372        if !index_geometry {
373            processor = processor.without_indexing();
374        }
375        process_decoded_vector_tile_with_assets(&tile, request, &mut processor, atlas)
376    })
377    .await
378    .map_err(ProcessVectorError::CpuWork)?
379}
380
381fn source_processor<T: VectorTransferables, C: Context>(
382    context: C,
383    last_source: bool,
384) -> ProcessVectorContext<T, C> {
385    let processor = ProcessVectorContext::new(context).with_pending_symbols();
386    // Earlier sources cannot replace an ancestor before the other sources' base layers arrive.
387    if last_source {
388        processor
389    } else {
390        processor.without_completion()
391    }
392}

394fn finish<T: VectorTransferables, C: Context>(
395    context: &C,
396    coords: WorldTileCoords,
397    attempt: Option<u64>,
398    failed: bool,
399    retry: bool,
400) -> Result<(), ProcedureError> {
401    let completion = if failed {
402        T::TileTessellated::build_failed(coords, false)
403    } else {
404        T::TileTessellated::build_from(coords)
405    };
406    context
407        .send_back(completion)
408        .map_err(ProcedureError::Send)?;
409    let outcome = IntoMessage::into(TileRequestOutcome {
410        coords,
411        kind: RequestKind::Vector,
412        attempt,
413        disposition: if retry {
414            RequestDisposition::Retry
415        } else {
416            RequestDisposition::Complete
417        },
418    });
419    #[cfg(all(target_arch = "wasm32", target_feature = "atomics"))]
420    let outcome = outcome.with_shared_permit(super::limiter::issue_shared_tile_permit());
421    context.send_back(outcome).map_err(ProcedureError::Send)
