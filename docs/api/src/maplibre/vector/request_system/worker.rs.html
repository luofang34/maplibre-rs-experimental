39/// Fetches, decodes, tessellates, and shapes symbols for one vector tile on a worker.
40pub fn fetch_vector_apc<K: OffscreenKernel, T: VectorTransferables, C: Context + Clone + Send>(
41    input: Input,
42    context: C,
43    kernel: K,
44) -> AsyncProcedureFuture {
45    Box::pin(async move {
46        let Input {
47            coords,
48            style,
49            attempt,
50            overscaled_zoom,
51            pixel_ratio,
52            only_sources,
53            refetch,
54        } = input;
55        let context = AttemptContext::new(context, attempt);
56        let client = kernel.source_client();
57        // A request for a change to some sources' data leaves the tile's other sources as the
58        // tile shows them.
59        let has_terrain = style.terrain.is_some();
60        let mut group_list: Vec<_> = source_layer_groups(&style, TileKind::Vector)
61            .into_iter()
62            .filter_map(|mut group| {
63                let included = only_sources.as_ref().is_none_or(|only| {
64                    group
65                        .source_name
66                        .as_ref()
67                        .is_some_and(|name| only.contains(name))
68                });
69                if !included {
70                    return None;
71                }
72                group
73                    .layers
74                    .retain(|layer| layer.needed_for_tile(has_terrain, coords, overscaled_zoom));
75                (!group.layers.is_empty()).then_some(group)
76            })
77            .collect();
78        prioritize_remote_sources(&mut group_list);
79        let mut groups = group_list.into_iter().peekable();
80        let mut failed = false;
81        let mut retry = false;
82        let mut images = provided::TileImages::new(coords, Some(attempt), pixel_ratio);
83        #[cfg(all(target_arch = "wasm32", target_feature = "atomics"))]
84        let mut shared_permit: Option<super::limiter::SharedTilePermit> = None;
85        while let Some(group) = groups.next() {
86            if group.has_nothing_at(coords) {
87                source_empty::<T, _>(coords, &group.layers, &context)
88                    .map_err(ProcedureError::Send)?;
89                continue;
90            }
91            enum FetchStep {
92                Ready(super::group_tile::RawGroupTile),
93                NotFound,
94                Failed { retryable: bool },
95            }
96            let fetch_step = match fetch_raw_group_tile(&client, coords, &group, refetch).await {
97                Ok(raw_tile) => FetchStep::Ready(raw_tile),
98                Err(error) if error.is_not_found() => {
99                    tracing::debug!(%coords, source = ?group.source_name, "no vector tile at the source; its layers are empty");
100                    FetchStep::NotFound
101                }
102                Err(error) => {
103                    tracing::warn!(%coords, source = ?group.source_name, error = %error.describe(), "vector tile unavailable");
104                    FetchStep::Failed {
105                        retryable: error.is_retryable(),
106                    }
107                }
108            };
109            let raw_tile = match fetch_step {
110                FetchStep::Ready(raw_tile) => raw_tile,
111                // A source without a tile there has nothing to draw, as GL JS loads a vector
112                // tile that answered 404 empty: its layers are empty, and the tile's other
113                // sources are drawn as usual.
114                FetchStep::NotFound => {
115                    source_empty::<T, _>(coords, &group.layers, &context)
116                        .map_err(ProcedureError::Send)?;
117                    continue;
118                }
119                FetchStep::Failed { retryable } => {
120                    failed = true;
121                    retry |= retryable;
122                    source_failed::<T, _>(coords, &group.layers, &context)
123                        .map_err(ProcedureError::Send)?;
124                    continue;
125                }
126            };
127            #[cfg(all(target_arch = "wasm32", target_feature = "atomics"))]
128            if shared_permit.is_none() {
129                shared_permit = Some(super::limiter::acquire_shared_tile_permit().await);
130            }
131            let mut heavy_slot = Some(acquire_heavy_tile_slot().await);
132            let promote_id = group
133                .source_name
134                .as_deref()
135                .and_then(|name| style.source_promote_id(name));
136            let tile = match decode_group_tile(&client, raw_tile, coords, &group.layers, promote_id)
137            {
138                Ok(tile) => tile,
139                Err(error) => {
140                    tracing::warn!(%coords, source = ?group.source_name, error = %error.describe(), "vector tile unavailable");
141                    failed = true;
142                    retry |= error.is_retryable();
143                    source_failed::<T, _>(coords, &group.layers, &context)
144                        .map_err(ProcedureError::Send)?;
145                    continue;
146                }
147            };
148            match process_source::<T, _, K::HttpClient>(
149                tile,
150                coords,
151                &style,
152                &client,
153                &group,
154                context.clone(),
155                (
156                    &mut heavy_slot,
157                    SourcePass {
158                        last_source: groups.peek().is_none() && !failed,
159                        overscaled_zoom,
160                        pixel_ratio,
161                    },
162                ),
163            )
164            .await
165            {
166                Ok(symbols) => images.add(symbols),
167                Err(ProcessVectorError::SendError(source)) => {
168                    return Err(ProcedureError::Send(source))
169                }
170                Err(error @ ProcessVectorError::SymbolAssets(_)) => {
171                    // The base layers were delivered, so the tile stays complete and drawn without
172                    // labels; only the retry outcome asks for another attempt.
173                    tracing::warn!(%coords, source = ?group.source_name, error = %error, "symbol assets unavailable");
174                    retry = true;
175                }
176                Err(
177                    error @ (ProcessVectorError::Decoding { .. } | ProcessVectorError::CpuWork(_)),
178                ) => {
179                    tracing::warn!(%coords, source = ?group.source_name, %error, "invalid vector tile");
180                    failed = true;
181                    source_failed::<T, _>(coords, &group.layers, &context)
182                        .map_err(ProcedureError::Send)?;
183                }
184            }
185        }
186        images
187            .report_first(&context)
188            .map_err(ProcedureError::Send)?;
189        finish::<T, _>(
190            &context,
191            coords,
192            Some(attempt),
193            (failed, retry),
194            #[cfg(all(target_arch = "wasm32", target_feature = "atomics"))]
195            shared_permit,
196        )?;
197        // The tile is complete and drawn; its labels are laid out again as images arrive.
198        images
199            .lay_out_again::<T, _, K::HttpClient>(&client, &style, context)
200            .await
201            .map_err(ProcedureError::Send)
202    })
203}

205/// How one source of a tile is processed.
206struct SourcePass {
207    /// Whether no other source follows, so this one completes the tile.
208    last_source: bool,
209    /// The zoom the tile is magnified to, or zero.
210    overscaled_zoom: u8,
211    /// Device pixels per layout pixel, which provided images are made for.
212    pixel_ratio: f32,
213}

215/// Reports every layer of a source as missing from the tile, which clears what an earlier
216/// generation drew there.
217fn source_empty<T: VectorTransferables, C: Context>(
218    coords: WorldTileCoords,
219    layers: &[StyleLayer],
220    context: &C,
221) -> Result<(), SendError> {
222    for layer in layers {
223        if let Some(source_layer) = &layer.source_layer {
224            context.send_back(T::LayerMissing::build_from(coords, source_layer.clone()))?;
225        }
226    }
227    Ok(())
228}

230fn source_failed<T: VectorTransferables, C: Context>(
231    coords: WorldTileCoords,
232    layers: &[StyleLayer],
233    context: &C,
234) -> Result<(), SendError> {
235    source_empty::<T, C>(coords, layers, context)?;
236    context.send_back(T::TileTessellated::build_failed(coords, true))
237}

239async fn process_source<T: VectorTransferables, C: Context + Clone + Send, H: HttpClient>(
240    group_tile: GroupTile,
241    coords: WorldTileCoords,
242    style: &Style,
243    client: &SourceClient<H>,
244    group: &SourceLayerGroup,
245    context: C,
246    (heavy_slot, pass): (&mut Option<super::limiter::HeavyTileSlot>, SourcePass),
247) -> Result<Option<provided::Symbols>, ProcessVectorError> {
248    let GroupTile {
249        mut tile,
250        mvt_bytes,
251        mvt_compressed,
252        mvt_rescale,
253    } = group_tile;
254    let SourcePass {
255        last_source,
256        overscaled_zoom,
257        pixel_ratio,
258    } = pass;
259    let projection = style
260        .projection
261        .as_ref()
262        .map_or_else(Default::default, |specification| {
263            specification.projection_type.clone()
264        });
265    let (symbols, base): (HashSet<_>, HashSet<_>) = group
266        .layers
267        .iter()
268        .cloned()
269        .partition(|layer| layer.type_ == "symbol");
270    let source_state = group.source_name.as_deref().map_or((None, None), |name| {
271        (
272            style.source_promote_id(name).cloned(),
273            style.feature_state.source(name).cloned(),
274        )
275    });
276    if T::LayerIndexed::requires_geometry_index() && (!base.is_empty() || !symbols.is_empty()) {
277        let lazy = LazyMvtIndex::from_packed_with_state(
278            mvt_bytes,
279            mvt_compressed,
280            source_state.0.clone(),
281            source_state.1.clone(),
282        )
283        .with_rescale(mvt_rescale);
284        context
285            .send_back(T::LayerIndexed::build_from(
286                coords,
287                group.source_name.clone(),
288                TileIndex::LazyMvt(lazy),
289            ))
290            .map_err(ProcessVectorError::SendError)?;
291    }
292    let request = VectorTileRequest {
293        coords,
294        layers: base,
295        projection: projection.clone(),
296        overscaled_zoom,
297    };
298    let has_dem_source = crate::terrain::source::dem_source(style).is_some();
299    if symbols.is_empty() {
300        tessellate::<T, C>(
301            context,
302            (false, false, has_dem_source),
303            (tile, request),
304            source_state,
305            Arc::new(SymbolAtlas::default()),
306        )
307        .await?;
308        let _ = heavy_slot.take();
309        return Ok(None);
310    }
311    let zoom = f64::from(overscaled_zoom.max(u8::from(coords.z)));
312    let sym_requests = symbol_requests_from_tile(&symbols, &tile, zoom, Some(style));
313    let asset_config = SymbolAssetConfig {
314        pixel_ratio,
315        ..SymbolAssetConfig::of(style)
316    };
317    let mut assets_fut = std::pin::pin!(fetch_symbol_asset_inputs(
318        client,
319        asset_config,
320        sym_requests
321    ));
322    let mut pre_assets = None;
323    let _ = std::future::poll_fn(|cx| {
324        if let std::task::Poll::Ready(res) = std::future::Future::poll(assets_fut.as_mut(), cx) {
325            pre_assets = Some(res);
326        }
327        std::task::Poll::Ready(())
328    })
329    .await;
330    tessellate::<T, C>(
331        context.clone(),
332        (false, false, has_dem_source),
333        (tile.clone(), request),
334        source_state.clone(),
335        Arc::new(SymbolAtlas::default()),
336    )
337    .await?;
338
339    let sym_request = VectorTileRequest {
340        coords,
341        layers: symbols,
342        projection,
343        overscaled_zoom,
344    };
345    if let Some(crate::vector::process_vector::DecodedVectorTile::Mvt(inner)) =
346        Arc::get_mut(&mut tile)
347    {
348        inner.layers.retain(|mvt| {
349            sym_request
350                .layers
351                .iter()
352                .any(|sl| sl.source_layer.as_deref() == Some(mvt.name.as_str()))
353        });
354    }
355    if pre_assets.is_none() {
356        let _ = std::future::poll_fn(|cx| {
357            if let std::task::Poll::Ready(res) = std::future::Future::poll(assets_fut.as_mut(), cx)
358            {
359                pre_assets = Some(res);
360            }
361            std::task::Poll::Ready(())
362        })
363        .await;
364    }
365    let fetched_inputs = match pre_assets {
366        Some(res) => res,
367        None => {
368            if last_source {
369                context
370                    .send_back(T::TileTessellated::build_partial(coords))
371                    .map_err(ProcessVectorError::SendError)?;
372            }
373            // Release HeavyTileSlot strictly BEFORE awaiting remote glyph/sprite HTTP I/O.
374            let _ = heavy_slot.take();
375            let loaded = assets_fut.await;
376            *heavy_slot = Some(acquire_heavy_tile_slot().await);
377            loaded
378        }
379    }
380    .map_err(ProcessVectorError::SymbolAssets)?;
381    let assets = pack_symbol_assets(client, asset_config, fetched_inputs);
382    // Only a source still waiting for images keeps its tile and layers to lay out again.
383    let again = (!assets.awaiting.is_empty()).then(|| (tile.clone(), sym_request.clone()));
384    tessellate::<T, C>(
385        context,
386        (false, false, has_dem_source),
387        (tile, sym_request),
388        source_state,
389        assets.atlas,
390    )
391    .await?;
392    let _ = heavy_slot.take();
393    Ok(Some(provided::Symbols {
394        again,
395        pixel_ratio,
396        provided: assets.provided,
397        awaiting: assets.awaiting,
398    }))
399}

401/// Tessellates the requested layers of one source's decoded tile off the fetching threads,
402/// replying with their geometry as it is made.
403async fn tessellate<T: VectorTransferables, C: Context + Send>(
404    context: C,
405    (emit_partial, index_geometry, has_dem_source): (bool, bool, bool),
406    (tile, request): (
407        Arc<crate::vector::process_vector::DecodedVectorTile>,
408        VectorTileRequest,
409    ),
410    (promote_id, feature_state): (
411        Option<crate::style::source::PromoteId>,
412        Option<crate::style::feature_state::SourceFeatureState>,
413    ),
414    atlas: Arc<SymbolAtlas>,
415) -> Result<(), ProcessVectorError> {
416    run_cpu_bound_replying(context, move |context| {
417        let mut processor = source_processor::<T, _>(context, emit_partial)
418            .with_has_dem_source(has_dem_source)
419            .with_source_feature_state(promote_id, feature_state);
420        if !index_geometry {
421            processor = processor.without_indexing();
422        }
423        process_decoded_vector_tile_with_assets(&tile, request, &mut processor, atlas)
424    })
425    .await
426    .map_err(ProcessVectorError::CpuWork)?
427}

429fn source_processor<T: VectorTransferables, C: Context>(
430    context: C,
431    emit_partial: bool,
432) -> ProcessVectorContext<T, C> {
433    let processor = ProcessVectorContext::new(context).with_pending_symbols();
434    if emit_partial {
435        processor
436    } else {
437        processor.without_completion()
438    }
439}

441fn finish<T: VectorTransferables, C: Context>(
442    context: &C,
443    coords: WorldTileCoords,
444    attempt: Option<u64>,
445    (failed, retry): (bool, bool),
446    #[cfg(all(target_arch = "wasm32", target_feature = "atomics"))] permit: Option<
447        super::limiter::SharedTilePermit,
448    >,
449) -> Result<(), ProcedureError> {
450    let completion = if failed {
451        T::TileTessellated::build_failed(coords, false)
452    } else {
453        T::TileTessellated::build_from(coords)
454    };
455    context
456        .send_back(completion)
457        .map_err(ProcedureError::Send)?;
458    let outcome = IntoMessage::into(TileRequestOutcome {
459        coords,
460        kind: RequestKind::Vector,
461        attempt,
462        disposition: if retry {
463            RequestDisposition::Retry
464        } else {
465            RequestDisposition::Complete
466        },
467    });
468    #[cfg(all(target_arch = "wasm32", target_feature = "atomics"))]
469    let outcome = match permit {
470        Some(permit) => outcome.with_shared_permit(permit),
471        None => outcome,
472    };
473    context.send_back(outcome).map_err(ProcedureError::Send)
474}

476fn prioritize_remote_sources(groups: &mut [SourceLayerGroup]) {
477    groups.sort_by_key(|group| {
478        matches!(group.source, crate::io::source_type::SourceType::GeoJson(_))
479    });
