38/// Fetches, decodes, tessellates, and shapes symbols for one vector tile on a worker.
39pub fn fetch_vector_apc<K: OffscreenKernel, T: VectorTransferables, C: Context + Clone + Send>(
40    input: Input,
41    context: C,
42    kernel: K,
43) -> AsyncProcedureFuture {
44    Box::pin(async move {
45        let Input {
46            coords,
47            style,
48            attempt,
49            overscaled_zoom,
50            pixel_ratio,
51            only_sources,
52            refetch,
53        } = input;
54        let context = AttemptContext::new(context, attempt);
55        let client = kernel.source_client();
56        // A request for a change to some sources' data leaves the tile's other sources as the
57        // tile shows them.
58        let has_terrain = style.terrain.is_some();
59        let mut groups = source_layer_groups(&style, TileKind::Vector)
60            .into_iter()
61            .filter_map(|mut group| {
62                let included = only_sources.as_ref().is_none_or(|only| {
63                    group
64                        .source_name
65                        .as_ref()
66                        .is_some_and(|name| only.contains(name))
67                });
68                if !included {
69                    return None;
70                }
71                group
72                    .layers
73                    .retain(|layer| layer.needed_for_tile(has_terrain, coords, overscaled_zoom));
74                (!group.layers.is_empty()).then_some(group)
75            })
76            .peekable();
77        let mut failed = false;
78        let mut retry = false;
79        let mut images = provided::TileImages::new(coords, Some(attempt), pixel_ratio);
80        while let Some(group) = groups.next() {
81            if group.has_nothing_at(coords) {
82                source_empty::<T, _>(coords, &group.layers, &context)
83                    .map_err(ProcedureError::Send)?;
84                continue;
85            }
86            enum FetchStep {
87                Ready(super::group_tile::RawGroupTile),
88                NotFound,
89                Failed { retryable: bool },
90            }
91            let fetch_step = match fetch_raw_group_tile(&client, coords, &group, refetch).await {
92                Ok(raw_tile) => FetchStep::Ready(raw_tile),
93                Err(error) if error.is_not_found() => {
94                    tracing::debug!(%coords, source = ?group.source_name, "no vector tile at the source; its layers are empty");
95                    FetchStep::NotFound
96                }
97                Err(error) => {
98                    tracing::warn!(%coords, source = ?group.source_name, error = %error.describe(), "vector tile unavailable");
99                    FetchStep::Failed {
100                        retryable: error.is_retryable(),
101                    }
102                }
103            };
104            let raw_tile = match fetch_step {
105                FetchStep::Ready(raw_tile) => raw_tile,
106                // A source without a tile there has nothing to draw, as GL JS loads a vector
107                // tile that answered 404 empty: its layers are empty, and the tile's other
108                // sources are drawn as usual.
109                FetchStep::NotFound => {
110                    source_empty::<T, _>(coords, &group.layers, &context)
111                        .map_err(ProcedureError::Send)?;
112                    continue;
113                }
114                FetchStep::Failed { retryable } => {
115                    failed = true;
116                    retry |= retryable;
117                    source_failed::<T, _>(coords, &group.layers, &context)
118                        .map_err(ProcedureError::Send)?;
119                    continue;
120                }
121            };
122            super::limiter::wait_for_shared_tile_slot().await;
123            let mut heavy_slot = Some(acquire_heavy_tile_slot().await);
124            let promote_id = group
125                .source_name
126                .as_deref()
127                .and_then(|name| style.source_promote_id(name));
128            let tile = match decode_group_tile(&client, raw_tile, coords, &group.layers, promote_id)
129            {
130                Ok(tile) => tile,
131                Err(error) => {
132                    tracing::warn!(%coords, source = ?group.source_name, error = %error.describe(), "vector tile unavailable");
133                    failed = true;
134                    retry |= error.is_retryable();
135                    source_failed::<T, _>(coords, &group.layers, &context)
136                        .map_err(ProcedureError::Send)?;
137                    continue;
138                }
139            };
140            match process_source::<T, _, K::HttpClient>(
141                tile,
142                coords,
143                &style,
144                &client,
145                &group,
146                context.clone(),
147                (
148                    &mut heavy_slot,
149                    SourcePass {
150                        last_source: groups.peek().is_none() && !failed,
151                        overscaled_zoom,
152                        pixel_ratio,
153                    },
154                ),
155            )
156            .await
157            {
158                Ok(symbols) => images.add(symbols),
159                Err(ProcessVectorError::SendError(source)) => {
160                    return Err(ProcedureError::Send(source))
161                }
162                Err(error @ ProcessVectorError::SymbolAssets(_)) => {
163                    // The base layers were delivered, so the tile stays complete and drawn without
164                    // labels; only the retry outcome asks for another attempt.
165                    tracing::warn!(%coords, source = ?group.source_name, error = %error, "symbol assets unavailable");
166                    retry = true;
167                }
168                Err(
169                    error @ (ProcessVectorError::Decoding { .. } | ProcessVectorError::CpuWork(_)),
170                ) => {
171                    tracing::warn!(%coords, source = ?group.source_name, %error, "invalid vector tile");
172                    failed = true;
173                    source_failed::<T, _>(coords, &group.layers, &context)
174                        .map_err(ProcedureError::Send)?;
175                }
176            }
177        }
178        images
179            .report_first(&context)
180            .map_err(ProcedureError::Send)?;
181        finish::<T, _>(&context, coords, Some(attempt), failed, retry)?;
182        // The tile is complete and drawn; its labels are laid out again as images arrive.
183        images
184            .lay_out_again::<T, _, K::HttpClient>(&client, &style, context)
185            .await
186            .map_err(ProcedureError::Send)
187    })
188}

190/// How one source of a tile is processed.
191struct SourcePass {
192    /// Whether no other source follows, so this one completes the tile.
193    last_source: bool,
194    /// The zoom the tile is magnified to, or zero.
195    overscaled_zoom: u8,
196    /// Device pixels per layout pixel, which provided images are made for.
197    pixel_ratio: f32,
198}
199
200/// Reports every layer of a source as missing from the tile, which clears what an earlier
201/// generation drew there.
202fn source_empty<T: VectorTransferables, C: Context>(
203    coords: WorldTileCoords,
204    layers: &[StyleLayer],
205    context: &C,
206) -> Result<(), SendError> {
207    for layer in layers {
208        if let Some(source_layer) = &layer.source_layer {
209            context.send_back(T::LayerMissing::build_from(coords, source_layer.clone()))?;
210        }
211    }
212    Ok(())
213}

215fn source_failed<T: VectorTransferables, C: Context>(
216    coords: WorldTileCoords,
217    layers: &[StyleLayer],
218    context: &C,
219) -> Result<(), SendError> {
220    source_empty::<T, C>(coords, layers, context)?;
221    context.send_back(T::TileTessellated::build_failed(coords, true))
222}

224async fn process_source<T: VectorTransferables, C: Context + Clone + Send, H: HttpClient>(
225    group_tile: GroupTile,
226    coords: WorldTileCoords,
227    style: &Style,
228    client: &SourceClient<H>,
229    group: &SourceLayerGroup,
230    context: C,
231    (heavy_slot, pass): (&mut Option<super::limiter::HeavyTileSlot>, SourcePass),
232) -> Result<Option<provided::Symbols>, ProcessVectorError> {
233    let GroupTile {
234        mut tile,
235        mvt_bytes,
236        mvt_compressed,
237        mvt_rescale,
238    } = group_tile;
239    let SourcePass {
240        last_source,
241        overscaled_zoom,
242        pixel_ratio,
243    } = pass;
244    let projection = style
245        .projection
246        .as_ref()
247        .map_or_else(Default::default, |specification| {
248            specification.projection_type.clone()
249        });
250    let (symbols, base): (HashSet<_>, HashSet<_>) = group
251        .layers
252        .iter()
253        .cloned()
254        .partition(|layer| layer.type_ == "symbol");
255    let source_state = group.source_name.as_deref().map_or((None, None), |name| {
256        (
257            style.source_promote_id(name).cloned(),
258            style.feature_state.source(name).cloned(),
259        )
260    });
261    if T::LayerIndexed::requires_geometry_index() && (!base.is_empty() || !symbols.is_empty()) {
262        let lazy = LazyMvtIndex::from_packed_with_state(
263            mvt_bytes,
264            mvt_compressed,
265            source_state.0.clone(),
266            source_state.1.clone(),
267        )
268        .with_rescale(mvt_rescale);
269        context
270            .send_back(T::LayerIndexed::build_from(
271                coords,
272                group.source_name.clone(),
273                TileIndex::LazyMvt(lazy),
274            ))
275            .map_err(ProcessVectorError::SendError)?;
276    }
277    let request = VectorTileRequest {
278        coords,
279        layers: base,
280        projection: projection.clone(),
281        overscaled_zoom,
282    };
283    let has_dem_source = crate::terrain::source::dem_source(style).is_some();
284    if symbols.is_empty() {
285        tessellate::<T, C>(
286            context,
287            (false, false, has_dem_source),
288            (tile, request),
289            source_state,
290            Arc::new(SymbolAtlas::default()),
291        )
292        .await?;
293        let _ = heavy_slot.take();
294        return Ok(None);
295    }
296    let zoom = f64::from(overscaled_zoom.max(u8::from(coords.z)));
297    let sym_requests = symbol_requests_from_tile(&symbols, &tile, zoom, Some(style));
298    let asset_config = SymbolAssetConfig {
299        pixel_ratio,
300        ..SymbolAssetConfig::of(style)
301    };
302    let mut assets_fut = std::pin::pin!(load_symbol_assets_for_requests(
303        client,
304        asset_config,
305        sym_requests
306    ));
307    let mut pre_assets = None;
308    let _ = std::future::poll_fn(|cx| {
309        if let std::task::Poll::Ready(res) = std::future::Future::poll(assets_fut.as_mut(), cx) {
310            pre_assets = Some(res);
311        }
312        std::task::Poll::Ready(())
313    })
314    .await;
315    tessellate::<T, C>(
316        context.clone(),
317        (false, false, has_dem_source),
318        (tile.clone(), request),
319        source_state.clone(),
320        Arc::new(SymbolAtlas::default()),
321    )
322    .await?;
323
324    let sym_request = VectorTileRequest {
325        coords,
326        layers: symbols,
327        projection,
328        overscaled_zoom,
329    };
330    if let Some(crate::vector::process_vector::DecodedVectorTile::Mvt(inner)) =
331        Arc::get_mut(&mut tile)
332    {
333        inner.layers.retain(|mvt| {
334            sym_request
335                .layers
336                .iter()
337                .any(|sl| sl.source_layer.as_deref() == Some(mvt.name.as_str()))
338        });
339    }
340    if pre_assets.is_none() {
341        let _ = std::future::poll_fn(|cx| {
342            if let std::task::Poll::Ready(res) = std::future::Future::poll(assets_fut.as_mut(), cx)
343            {
344                pre_assets = Some(res);
345            }
346            std::task::Poll::Ready(())
347        })
348        .await;
349    }
350    let assets = match pre_assets {
351        Some(res) => res,
352        None => {
353            if last_source {
354                context
355                    .send_back(T::TileTessellated::build_partial(coords))
356                    .map_err(ProcessVectorError::SendError)?;
357            }
358            // Release HeavyTileSlot strictly BEFORE awaiting remote glyph/sprite HTTP I/O.
359            let _ = heavy_slot.take();
360            let loaded = assets_fut.await;
361            *heavy_slot = Some(acquire_heavy_tile_slot().await);
362            loaded
363        }
364    }
365    .map_err(ProcessVectorError::SymbolAssets)?;
366    // Only a source still waiting for images keeps its tile and layers to lay out again.
367    let again = (!assets.awaiting.is_empty()).then(|| (tile.clone(), sym_request.clone()));
368    tessellate::<T, C>(
369        context,
370        (false, false, has_dem_source),
371        (tile, sym_request),
372        source_state,
373        assets.atlas,
374    )
375    .await?;
376    let _ = heavy_slot.take();
377    Ok(Some(provided::Symbols {
378        again,
379        pixel_ratio,
380        provided: assets.provided,
381        awaiting: assets.awaiting,
382    }))
383}

385/// Tessellates the requested layers of one source's decoded tile off the fetching threads,
386/// replying with their geometry as it is made.
387async fn tessellate<T: VectorTransferables, C: Context + Send>(
388    context: C,
389    (emit_partial, index_geometry, has_dem_source): (bool, bool, bool),
390    (tile, request): (
391        Arc<crate::vector::process_vector::DecodedVectorTile>,
392        VectorTileRequest,
393    ),
394    (promote_id, feature_state): (
395        Option<crate::style::source::PromoteId>,
396        Option<crate::style::feature_state::SourceFeatureState>,
397    ),
398    atlas: Arc<SymbolAtlas>,
399) -> Result<(), ProcessVectorError> {
400    run_cpu_bound_replying(context, move |context| {
401        let mut processor = source_processor::<T, _>(context, emit_partial)
402            .with_has_dem_source(has_dem_source)
403            .with_source_feature_state(promote_id, feature_state);
404        if !index_geometry {
405            processor = processor.without_indexing();
406        }
407        process_decoded_vector_tile_with_assets(&tile, request, &mut processor, atlas)
408    })
409    .await
410    .map_err(ProcessVectorError::CpuWork)?
411}

413fn source_processor<T: VectorTransferables, C: Context>(
414    context: C,
415    emit_partial: bool,
416) -> ProcessVectorContext<T, C> {
417    let processor = ProcessVectorContext::new(context).with_pending_symbols();
418    if emit_partial {
419        processor
420    } else {
421        processor.without_completion()
422    }
423}

425fn finish<T: VectorTransferables, C: Context>(
426    context: &C,
427    coords: WorldTileCoords,
428    attempt: Option<u64>,
429    failed: bool,
430    retry: bool,
431) -> Result<(), ProcedureError> {
432    let completion = if failed {
433        T::TileTessellated::build_failed(coords, false)
434    } else {
435        T::TileTessellated::build_from(coords)
436    };
437    context
438        .send_back(completion)
439        .map_err(ProcedureError::Send)?;
440    let outcome = IntoMessage::into(TileRequestOutcome {
441        coords,
442        kind: RequestKind::Vector,
443        attempt,
444        disposition: if retry {
445            RequestDisposition::Retry
446        } else {
447            RequestDisposition::Complete
448        },
449    });
450    #[cfg(all(target_arch = "wasm32", target_feature = "atomics"))]
451    let outcome = outcome.with_shared_permit(super::limiter::issue_shared_tile_permit());
452    context.send_back(outcome).map_err(ProcedureError::Send)
453}
