22/// What one source's symbols need from providers.
23pub(super) struct Symbols {
24    /// The source's tile and symbol layers, kept while some images are still to come.
25    pub(super) again: Option<(Vec<u8>, VectorTileRequest)>,
26    /// Device pixels per layout pixel the images are made for.
27    pub(super) pixel_ratio: f32,
28    /// Every provided name the labels ask for.
29    pub(super) provided: Vec<String>,
30    /// The names without an answer when the labels were laid out.
31    pub(super) awaiting: Vec<String>,
32}
33
34/// The provided images of one tile's labels, across its sources.
35pub(super) struct TileImages {
36    coords: WorldTileCoords,
37    attempt: Option<u64>,
38    pixel_ratio: f32,
39    /// Every provided name the labels ask for.
40    provided: BTreeSet<String>,
41    /// The sources whose labels lack some.
42    waiting: Vec<Symbols>,
43}
44
45impl TileImages {
46    pub(super) fn new(coords: WorldTileCoords, attempt: Option<u64>, pixel_ratio: f32) -> Self {
47        Self {
48            coords,
49            attempt,
50            pixel_ratio,
51            provided: BTreeSet::new(),
52            waiting: Vec::new(),
53        }
54    }
55
56    /// Notes what one source's labels need.
57    pub(super) fn add(&mut self, symbols: Option<Symbols>) {
58        let Some(symbols) = symbols else {
59            return;
60        };
61        self.provided.extend(symbols.provided.iter().cloned());
62        if !symbols.awaiting.is_empty() {
63            self.waiting.push(symbols);
64        }
65    }
66
67    fn report(&self, state: ProvidedImagesState) -> ProvidedImagesReport {
68        ProvidedImagesReport {
69            coords: self.coords,
70            attempt: self.attempt,
71            names: self.provided.iter().cloned().collect(),
72            pixel_ratio: self.pixel_ratio,
73            state,
74        }
75    }
76
77    /// Tells the map which provided images the labels sent so far name, and whether some are
78    /// still to come.
79    pub(super) fn report_first<C: Context>(&self, context: &C) -> Result<(), SendError> {
80        if self.provided.is_empty() {
81            return Ok(());
82        }
83        context.send_back(self.report(if self.waiting.is_empty() {
84            ProvidedImagesState::Settled
85        } else {
86            ProvidedImagesState::Awaiting
87        }))
88    }
89
90    /// Waits for the images the labels lack, lays out again the labels of each source that
91    /// gained one, and reports whether some provider was unavailable for now.
92    pub(super) async fn lay_out_again<T: VectorTransferables, C: Context + Clone, H: HttpClient>(
93        self,
94        client: &SourceClient<H>,
95        style: &Style,
96        context: C,
97    ) -> Result<(), SendError> {
98        if self.waiting.is_empty() {
99            return Ok(());
100        }
101        let coords = self.coords;
102        let state = match lay_out_sources::<T, C, H>(
103            client,
104            style,
105            coords,
106            context.clone(),
107            &self.waiting,
108        )
109        .await
110        {
111            Ok(true) => ProvidedImagesState::Retry,
112            Ok(false) => ProvidedImagesState::Settled,
113            Err(ProcessVectorError::SendError(source)) => return Err(source),
114            Err(error) => {
115                tracing::warn!(%coords, %error, "labels with provided images not laid out again");
116                ProvidedImagesState::Retry
117            }
118        };
119        context.send_back(self.report(state))
120    }
121}

123/// Waits for the images `sources` lack and lays out again the labels of each source that
124/// gained one. Returns whether some provider was unavailable for now.
125async fn lay_out_sources<T: VectorTransferables, C: Context + Clone, H: HttpClient>(
126    client: &SourceClient<H>,
127    style: &Style,
128    coords: WorldTileCoords,
129    context: C,
130    sources: &[Symbols],
131) -> Result<bool, ProcessVectorError> {
132    let providers = client.image_providers();
133    let mut wanted: Vec<(&str, f32)> = sources
134        .iter()
135        .flat_map(|source| {
136            source
137                .awaiting
138                .iter()
139                .map(|name| (name.as_str(), source.pixel_ratio))
140        })
141        .collect();
142    wanted.sort_by(|a, b| a.0.cmp(b.0));
143    wanted.dedup_by(|a, b| a.0 == b.0);
144    // The names of a tile are asked for together; the registry runs each once however many
145    // tiles wait for it, and bounds how many run at all.
146    let answers = futures::future::join_all(
147        wanted
148            .iter()
149            .map(|(name, pixel_ratio)| providers.resolve(name, *pixel_ratio)),
150    )
151    .await;
152    let mut unavailable = false;
153    let mut gained = HashSet::new();
154    for ((name, _), answer) in wanted.iter().zip(answers) {
155        match answer.as_deref() {
156            Ok(Resolved::Image(_)) => {
157                gained.insert(*name);
158            }
159            Ok(Resolved::None) => {}
160            Err(reason) => {
161                tracing::debug!(%coords, %name, %reason, "provided image unavailable for now");
162                unavailable = true;
163            }
164        }
165    }
166    for source in sources {
167        let Some((data, request)) = &source.again else {
168            continue;
169        };
170        if !source
171            .awaiting
172            .iter()
173            .any(|name| gained.contains(name.as_str()))
174        {
175            continue;
176        }
177        let assets = load_symbol_assets_awaiting(
178            client,
179            SymbolAssetConfig {
180                pixel_ratio: source.pixel_ratio,
181                ..SymbolAssetConfig::of(style)
182            },
183            &request.layers,
184            data,
185            f64::from(request.overscaled_zoom.max(u8::from(coords.z))),
186        )
187        .await
188        .map_err(ProcessVectorError::SymbolAssets)?;
189        // An answer the cache dropped since it arrived is drawn as the fallback for now and
190        // asked for again when the tile is retried.
191        unavailable |= !assets.awaiting.is_empty();
192        // The tile was completed already; these labels replace the ones it was drawn with.
193        let mut processor = ProcessVectorContext::<T, C>::new(context.clone()).without_completion();
194        process_vector_tile_with_assets(data, request.clone(), &mut processor, assets.atlas)?;
195        providers.count_relaid();
196        tracing::debug!(%coords, images = gained.len(), "labels laid out again with provided images");
197    }
198    Ok(unavailable)
199}
