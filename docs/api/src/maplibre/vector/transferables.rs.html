34/// Tags used by the owned Rust vector message implementations.
35pub enum VectorMessageTag {
36    /// A tile processing batch finished, with or without pending symbol assets.
37    TileTessellated = 1,
38    /// A requested source layer is absent from the tile.
39    LayerMissing = 2,
40    /// Fill, line or circle geometry is ready for upload.
41    LayerTessellated = 3,
42    /// Symbol geometry and its placement metadata are ready.
43    SymbolLayerTessellated = 4,
44    /// Tile geometry is indexed for spatial queries.
45    LayerIndexed = 10,
46}
47
48impl MessageTag for VectorMessageTag {
49    fn dyn_clone(&self) -> Box<dyn MessageTag> {
50        Box::new(*self)
51    }
52}

54/// Tile completion marker; partial completion keeps pending symbol work in the request budget.
55pub trait TileTessellated: IntoMessage + Debug + Send {
56    /// Identifies this backend's completion payload for message dispatch.
57    fn message_tag() -> &'static dyn MessageTag;
58
59    /// Marks a processing batch complete with no outstanding symbol work.
60    fn build_from(coords: WorldTileCoords) -> Self
61    where
62        Self: Sized;
63
64    /// Reports unavailable source data, retaining parent coverage.
65    /// Keep `pending_symbols` true until the worker sends its final completion.
66    fn build_failed(coords: WorldTileCoords, pending_symbols: bool) -> Self
67    where
68        Self: Sized;
69    /// Whether at least one requested source could not be fetched or decoded.
70    fn failed(&self) -> bool;
71
72    /// Marks base geometry ready while symbol assets are still pending.
73    fn build_partial(coords: WorldTileCoords) -> Self
74    where
75        Self: Sized;
76    /// Whether the tile still needs its symbol-processing completion.
77    fn pending_symbols(&self) -> bool;
78    /// Tile-grid coordinates of the completed processing batch.
79    fn coords(&self) -> WorldTileCoords;
80}

82/// A requested source layer cannot produce a vector bucket for a tile.
83pub trait LayerMissing: IntoMessage + Debug + Send {
84    /// Identifies this backend's missing-layer payload for message dispatch.
85    fn message_tag() -> &'static dyn MessageTag;
86
87    /// Records a missing source-layer name at the supplied tile coordinates.
88    fn build_from(coords: WorldTileCoords, layer_name: String) -> Self
89    where
90        Self: Sized;
91
92    /// Tile-grid coordinates of the missing layer.
93    fn coords(&self) -> WorldTileCoords;
94
95    /// Source-layer name, rather than a style-layer ID.
96    fn layer_name(&self) -> &str;
97
98    /// Consumes the message into the tile's missing-layer record.
99    fn to_bucket(self) -> MissingVectorLayerBucket;
100}

102/// Tessellated vector geometry for one style layer, ready for renderer upload.
103pub trait LayerTessellated: IntoMessage + Debug + Send {
104    /// Identifies this backend's vector-geometry payload for message dispatch.
105    fn message_tag() -> &'static dyn MessageTag;
106
107    /// Takes ownership of padded geometry, per-feature vertex counts and RGBA colors.
108    /// Metadata follows geometry feature order; `layer_data` names the source layer, while
109    /// `style_layer_id` identifies the style entry whose paint produced this bucket.
110    fn build_from(
111        coords: WorldTileCoords,
112        buffer: OverAlignedVertexBuffer<ShaderVertex, IndexDataType>,
113        feature_indices: Vec<u32>,
114        feature_colors: Vec<[f32; 4]>,
115        feature_sort_keys: Vec<f32>,
116        layer_data: Layer,
117        style_layer_id: String,
118    ) -> Self
119    where
120        Self: Sized;
121
122    /// Builds a tessellated layer payload together with its per-entry feature attributes in one
123    /// step so single-threaded wire formats can pack everything into a single buffer.
124    #[allow(clippy::too_many_arguments)]
125    fn build_from_with_paint(
126        coords: WorldTileCoords,
127        buffer: OverAlignedVertexBuffer<ShaderVertex, IndexDataType>,
128        feature_indices: Vec<u32>,
129        feature_colors: Vec<[f32; 4]>,
130        feature_sort_keys: Vec<f32>,
131        layer_data: Layer,
132        style_layer_id: String,
133        entries: Vec<BucketFeatureEntry>,
134    ) -> Self
135    where
136        Self: Sized,
137    {
138        Self::build_from(
139            coords,
140            buffer,
141            feature_indices,
142            feature_colors,
143            feature_sort_keys,
144            layer_data,
145            style_layer_id,
146        )
147        .with_feature_paint(entries)
148    }
149
150    /// Builds a tessellated layer payload with optional worker pre-packing using `style_type`
151    /// (`"fill"`, `"line"`, `"fill-extrusion"`, etc.) when `allow_prepack` is enabled.
152    #[allow(clippy::too_many_arguments)]
153    fn build_from_with_paint_typed(
154        coords: WorldTileCoords,
155        buffer: OverAlignedVertexBuffer<ShaderVertex, IndexDataType>,
156        feature_indices: Vec<u32>,
157        feature_colors: Vec<[f32; 4]>,
158        feature_sort_keys: Vec<f32>,
159        layer_data: Layer,
160        style_layer_id: String,
161        entries: Vec<BucketFeatureEntry>,
162        _style_type: &str,
163        _allow_prepack: bool,
164    ) -> Self
165    where
166        Self: Sized,
167    {
168        Self::build_from_with_paint(
169            coords,
170            buffer,
171            feature_indices,
172            feature_colors,
173            feature_sort_keys,
174            layer_data,
175            style_layer_id,
176            entries,
177        )
178    }
179
180    /// Attaches per-entry feature attributes for in-place GPU paint and `feature-state` refresh.
181    fn with_feature_paint(self, _entries: Vec<BucketFeatureEntry>) -> Self
182    where
183        Self: Sized,
184    {
185        self
186    }
187
188    /// Extracts any attached per-entry feature attributes before converting into a bucket.
189    fn take_feature_paint(&mut self) -> Option<Vec<BucketFeatureEntry>> {
190        None
191    }
192
193    /// Marks the tile magnification zoom used when tessellating this layer.
194    fn with_overscaled_zoom(self, _overscaled_zoom: u8) -> Self
195    where
196        Self: Sized,
197    {
198        self
199    }
200
201    /// Tile magnification zoom recorded when this layer was tessellated (`0` if not overscaled).
202    fn overscaled_zoom(&self) -> u8 {
203        0
204    }
205
206    /// Tile-grid coordinates whose local space contains the geometry.
207    fn coords(&self) -> WorldTileCoords;
208
209    /// Whether the geometry contains no usable draw indices, excluding alignment padding.
210    fn is_empty(&self) -> bool;
211
212    /// Style entry to use when uploading and drawing the bucket.
213    fn style_layer_id(&self) -> &str;
214
215    /// Consumes the transfer representation into renderer-owned geometry and metadata.
216    fn to_bucket(self) -> AvailableVectorLayerBucket;
217}

219/// Text or icon geometry with placement metadata and an optional shared glyph/sprite atlas.
220pub trait SymbolLayerTessellated: IntoMessage + Debug + Send {
221    /// Identifies this backend's symbol payload for message dispatch.
222    fn message_tag() -> &'static dyn MessageTag;
223
224    /// Takes ownership of symbol geometry and feature ranges for one source/style layer pair.
225    /// Atlas ownership is shared so its texels remain available when the renderer uploads them.
226    fn build_from(
227        coords: WorldTileCoords,
228        buffer: OverAlignedVertexBuffer<ShaderSymbolVertex, IndexDataType>,
229        features: Vec<Feature>,
230        atlas: Option<std::sync::Arc<crate::sdf::assets::SymbolAtlas>>,
231        layer_data: Layer,
232        style_layer_id: String,
233    ) -> Self
234    where
235        Self: Sized;
236
237    /// Tile-grid coordinates whose local space contains the symbol anchors.
238    fn coords(&self) -> WorldTileCoords;
239
240    /// Whether there are no usable symbol draw indices, excluding alignment padding.
241    fn is_empty(&self) -> bool;
242
243    /// Consumes the message into the renderer's symbol layer data.
244    fn to_bucket(self) -> SymbolLayerData;
245}

247/// Spatial index produced from a tile's decoded vector geometry.
248pub trait LayerIndexed: IntoMessage + Debug + Send {
249    /// Identifies this backend's spatial-index payload for message dispatch.
250    fn message_tag() -> &'static dyn MessageTag;
251
252    /// Whether this backend transfers the spatial geometry index to the main thread.
253    /// Backends that ignore `_index` in [`Self::build_from`] can return `false` to skip
254    /// building the query index on the worker thread.
255    fn requires_geometry_index() -> bool {
256        true
257    }
258
259    /// Associates an owned spatial index with its tile-grid coordinates and the source it was
260    /// read from.
261    fn build_from(coords: WorldTileCoords, source: Option<String>, index: TileIndex) -> Self
262    where
263        Self: Sized;
264
265    /// Tile-grid coordinates associated with the index.
266    fn coords(&self) -> WorldTileCoords;
267
268    /// The style's name for the source the index was read from, when the payload carries it.
269    fn source(&self) -> Option<String>;
270
271    /// Consumes the transfer representation and returns its spatial index.
272    fn to_tile_index(self) -> TileIndex;
273}

275/// Selects the message representations used by a vector worker and its matching receiver.
276/// Implementations can retain Rust values or serialize them across a platform worker boundary.
277pub trait VectorTransferables: Copy + Clone + 'static {
278    /// Tile processing completion, including pending-symbol state.
279    type TileTessellated: TileTessellated;
280    /// A source layer absent from a requested tile.
281    type LayerMissing: LayerMissing;
282    /// Fill, line or circle geometry and paint metadata.
283    type LayerTessellated: LayerTessellated;
284    /// Text/icon geometry, atlas and placement metadata.
285    type SymbolLayerTessellated: SymbolLayerTessellated;
286    /// Tile geometry indexed for queries.
287    type LayerIndexed: LayerIndexed;
288}
