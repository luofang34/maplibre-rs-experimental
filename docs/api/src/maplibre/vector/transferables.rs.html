32/// Tags used by the owned Rust vector message implementations.
33pub enum VectorMessageTag {
34    /// A tile processing batch finished, with or without pending symbol assets.
35    TileTessellated = 1,
36    /// A requested source layer is absent from the tile.
37    LayerMissing = 2,
38    /// Fill, line or circle geometry is ready for upload.
39    LayerTessellated = 3,
40    /// Symbol geometry and its placement metadata are ready.
41    SymbolLayerTessellated = 4,
42    /// Tile geometry is indexed for spatial queries.
43    LayerIndexed = 10,
44}
45
46impl MessageTag for VectorMessageTag {
47    fn dyn_clone(&self) -> Box<dyn MessageTag> {
48        Box::new(*self)
49    }
50}

52/// Tile completion marker; partial completion keeps pending symbol work in the request budget.
53pub trait TileTessellated: IntoMessage + Debug + Send {
54    /// Identifies this backend's completion payload for message dispatch.
55    fn message_tag() -> &'static dyn MessageTag;
56
57    /// Marks a processing batch complete with no outstanding symbol work.
58    fn build_from(coords: WorldTileCoords) -> Self
59    where
60        Self: Sized;
61
62    /// Reports unavailable source data, retaining parent coverage.
63    /// Keep `pending_symbols` true until the worker sends its final completion.
64    fn build_failed(coords: WorldTileCoords, pending_symbols: bool) -> Self
65    where
66        Self: Sized;
67    /// Whether at least one requested source could not be fetched or decoded.
68    fn failed(&self) -> bool;
69
70    /// Marks base geometry ready while symbol assets are still pending.
71    fn build_partial(coords: WorldTileCoords) -> Self
72    where
73        Self: Sized;
74    /// Whether the tile still needs its symbol-processing completion.
75    fn pending_symbols(&self) -> bool;
76    /// Tile-grid coordinates of the completed processing batch.
77    fn coords(&self) -> WorldTileCoords;
78}

80/// A requested source layer cannot produce a vector bucket for a tile.
81pub trait LayerMissing: IntoMessage + Debug + Send {
82    /// Identifies this backend's missing-layer payload for message dispatch.
83    fn message_tag() -> &'static dyn MessageTag;
84
85    /// Records a missing source-layer name at the supplied tile coordinates.
86    fn build_from(coords: WorldTileCoords, layer_name: String) -> Self
87    where
88        Self: Sized;
89
90    /// Tile-grid coordinates of the missing layer.
91    fn coords(&self) -> WorldTileCoords;
92
93    /// Source-layer name, rather than a style-layer ID.
94    fn layer_name(&self) -> &str;
95
96    /// Consumes the message into the tile's missing-layer record.
97    fn to_bucket(self) -> MissingVectorLayerBucket;
98}

100/// Tessellated vector geometry for one style layer, ready for renderer upload.
101pub trait LayerTessellated: IntoMessage + Debug + Send {
102    /// Identifies this backend's vector-geometry payload for message dispatch.
103    fn message_tag() -> &'static dyn MessageTag;
104
105    /// Takes ownership of padded geometry, per-feature vertex counts and RGBA colors.
106    /// Metadata follows geometry feature order; `layer_data` names the source layer, while
107    /// `style_layer_id` identifies the style entry whose paint produced this bucket.
108    fn build_from(
109        coords: WorldTileCoords,
110        buffer: OverAlignedVertexBuffer<ShaderVertex, IndexDataType>,
111        feature_indices: Vec<u32>,
112        feature_colors: Vec<[f32; 4]>,
113        feature_sort_keys: Vec<f32>,
114        layer_data: Layer,
115        style_layer_id: String,
116    ) -> Self
117    where
118        Self: Sized;
119
120    /// Builds a tessellated layer payload together with its per-entry feature attributes in one
121    /// step so single-threaded wire formats can pack everything into a single buffer.
122    #[allow(clippy::too_many_arguments)]
123    fn build_from_with_paint(
124        coords: WorldTileCoords,
125        buffer: OverAlignedVertexBuffer<ShaderVertex, IndexDataType>,
126        feature_indices: Vec<u32>,
127        feature_colors: Vec<[f32; 4]>,
128        feature_sort_keys: Vec<f32>,
129        layer_data: Layer,
130        style_layer_id: String,
131        entries: Vec<BucketFeatureEntry>,
132    ) -> Self
133    where
134        Self: Sized,
135    {
136        Self::build_from(
137            coords,
138            buffer,
139            feature_indices,
140            feature_colors,
141            feature_sort_keys,
142            layer_data,
143            style_layer_id,
144        )
145        .with_feature_paint(entries)
146    }
147
148    /// Attaches per-entry feature attributes for in-place GPU paint and `feature-state` refresh.
149    fn with_feature_paint(self, _entries: Vec<BucketFeatureEntry>) -> Self
150    where
151        Self: Sized,
152    {
153        self
154    }
155
156    /// Extracts any attached per-entry feature attributes before converting into a bucket.
157    fn take_feature_paint(&mut self) -> Option<Vec<BucketFeatureEntry>> {
158        None
159    }
160
161    /// Tile-grid coordinates whose local space contains the geometry.
162    fn coords(&self) -> WorldTileCoords;
163
164    /// Whether the geometry contains no usable draw indices, excluding alignment padding.
165    fn is_empty(&self) -> bool;
166
167    /// Style entry to use when uploading and drawing the bucket.
168    fn style_layer_id(&self) -> &str;
169
170    /// Consumes the transfer representation into renderer-owned geometry and metadata.
171    fn to_bucket(self) -> AvailableVectorLayerBucket;
172}

174/// Text or icon geometry with placement metadata and an optional shared glyph/sprite atlas.
175pub trait SymbolLayerTessellated: IntoMessage + Debug + Send {
176    /// Identifies this backend's symbol payload for message dispatch.
177    fn message_tag() -> &'static dyn MessageTag;
178
179    /// Takes ownership of symbol geometry and feature ranges for one source/style layer pair.
180    /// Atlas ownership is shared so its texels remain available when the renderer uploads them.
181    fn build_from(
182        coords: WorldTileCoords,
183        buffer: OverAlignedVertexBuffer<ShaderSymbolVertex, IndexDataType>,
184        features: Vec<Feature>,
185        atlas: Option<std::sync::Arc<crate::sdf::assets::SymbolAtlas>>,
186        layer_data: Layer,
187        style_layer_id: String,
188    ) -> Self
189    where
190        Self: Sized;
191
192    /// Tile-grid coordinates whose local space contains the symbol anchors.
193    fn coords(&self) -> WorldTileCoords;
194
195    /// Whether there are no usable symbol draw indices, excluding alignment padding.
196    fn is_empty(&self) -> bool;
197
198    /// Consumes the message into the renderer's symbol layer data.
199    fn to_bucket(self) -> SymbolLayerData;
200}

202/// Spatial index produced from a tile's decoded vector geometry.
203pub trait LayerIndexed: IntoMessage + Debug + Send {
204    /// Identifies this backend's spatial-index payload for message dispatch.
205    fn message_tag() -> &'static dyn MessageTag;
206
207    /// Whether this backend transfers the spatial geometry index to the main thread.
208    /// Backends that ignore `_index` in [`Self::build_from`] can return `false` to skip
209    /// building the query index on the worker thread.
210    fn requires_geometry_index() -> bool {
211        true
212    }
213
214    /// Associates an owned spatial index with its tile-grid coordinates and the source it was
215    /// read from.
216    fn build_from(coords: WorldTileCoords, source: Option<String>, index: TileIndex) -> Self
217    where
218        Self: Sized;
219
220    /// Tile-grid coordinates associated with the index.
221    fn coords(&self) -> WorldTileCoords;
222
223    /// The style's name for the source the index was read from, when the payload carries it.
224    fn source(&self) -> Option<String>;
225
226    /// Consumes the transfer representation and returns its spatial index.
227    fn to_tile_index(self) -> TileIndex;
228}

230/// Selects the message representations used by a vector worker and its matching receiver.
231/// Implementations can retain Rust values or serialize them across a platform worker boundary.
232pub trait VectorTransferables: Copy + Clone + 'static {
233    /// Tile processing completion, including pending-symbol state.
234    type TileTessellated: TileTessellated;
235    /// A source layer absent from a requested tile.
236    type LayerMissing: LayerMissing;
237    /// Fill, line or circle geometry and paint metadata.
238    type LayerTessellated: LayerTessellated;
239    /// Text/icon geometry, atlas and placement metadata.
240    type SymbolLayerTessellated: SymbolLayerTessellated;
241    /// Tile geometry indexed for queries.
242    type LayerIndexed: LayerIndexed;
243}
