18pub(crate) struct LayerReplacements {
19    pub vector: Vec<AvailableVectorLayerBucket>,
20    pub symbols: Vec<SymbolLayerData>,
21    pub keep_vector: bool,
22    pub keep_symbols: bool,
23    pub revision: u64,
24    /// Non-symbol style layers the request in flight was asked to build; its result is the
25    /// tile's whole content for them, so those it leaves out are dropped when it finishes.
26    pub requested: Vec<String>,
27    /// Which of [`requested`](Self::requested) the request has delivered so far.
28    pub delivered: Vec<String>,
29}
30impl TileComponent for LayerReplacements {}

35pub(crate) fn ensure(tiles: &mut Tiles, coords: WorldTileCoords) {
36    if !tiles.exists(coords) {
37        return;
38    }
39    if tiles.query::<&LayerReplacements>(coords).is_none() {
40        if let Some(mut tile) = tiles.spawn_mut(coords) {
41            tile.insert(LayerReplacements::default());
42        }
43    }
44}

46pub(crate) fn begin(world: &mut World, coords: WorldTileCoords, requested: Vec<String>) {
47    let vector = world.tiles.query::<&VectorLayerBucketComponent>(coords);
48    let keep_vector = vector.is_some_and(|tile| tile.done && !tile.failed);
49    if vector.is_none() {
50        if let Some(mut tile) = world.tiles.spawn_mut(coords) {
51            tile.insert(VectorLayerBucketComponent::default());
52        }
53    }
54    let symbols = world.tiles.query::<&SymbolLayersDataComponent>(coords);
55    let keep_symbols = symbols.is_some_and(|tile| !tile.pending_assets && !tile.layers.is_empty());
56    if symbols.is_none() {
57        if let Some(mut tile) = world.tiles.spawn_mut(coords) {
58            tile.insert(SymbolLayersDataComponent::default());
59        }
60    }
61    ensure(&mut world.tiles, coords);
62    if let Some(replacements) = world.tiles.query_mut::<&mut LayerReplacements>(coords) {
63        replacements.keep_vector = keep_vector;
64        replacements.keep_symbols = keep_symbols;
65        replacements.requested = requested;
66        replacements.delivered.clear();
67    }
68}

70/// Drops the layers the finished request was asked to build but did not deliver: hidden or
71/// filtered out since the tile was last made, or left without features by new source data.
72/// Buckets are otherwise only replaced by layer id, so such a layer would keep showing the
73/// geometry of an earlier request for as long as the tile stays loaded.
74pub(crate) fn drop_undelivered(world: &mut World, coords: WorldTileCoords) {
75    let Some(pending) = world.tiles.query_mut::<&mut LayerReplacements>(coords) else {
76        return;
77    };
78    let requested = std::mem::take(&mut pending.requested);
79    let omitted: Vec<String> = requested
80        .into_iter()
81        .filter(|id| !pending.delivered.contains(id))
82        .collect();
83    if !omitted.is_empty() {
84        purge_vector_layers(world, coords, &omitted);
85    }
86}

88/// Whether `coords` already holds committed resident layers usable if a refresh is cancelled.
89pub(crate) fn has_committed_resident_layers(world: &World, coords: WorldTileCoords) -> bool {
90    world
91        .tiles
92        .query::<&VectorLayerBucketComponent>(coords)
93        .is_some_and(|v| v.done && !v.failed && !v.layers.is_empty())
94        && world
95            .tiles
96            .query::<&SymbolLayersDataComponent>(coords)
97            .is_none_or(|s| !s.pending_assets || !s.layers.is_empty())
98}

100/// Discards staged refresh data at `coords` after cancellation while preserving committed layers.
101pub(crate) fn discard_pending_refresh(world: &mut World, coords: WorldTileCoords) {
102    if let Some(pending) = world.tiles.query_mut::<&mut LayerReplacements>(coords) {
103        pending.vector.clear();
104        pending.symbols.clear();
105        pending.requested.clear();
106        pending.delivered.clear();
107        pending.keep_vector = false;
108        pending.keep_symbols = false;
109    }
110    if let Some(paint_store) = world
111        .tiles
112        .query_mut::<&mut TileFeaturePaintComponent>(coords)
113    {
114        paint_store.pending.clear();
115    }
116}

123pub(crate) fn accept_vector_with_paint(
124    world: &mut World,
125    bucket: AvailableVectorLayerBucket,
126    feature_paint: Option<Vec<BucketFeatureEntry>>,
127) {
128    let coords = bucket.coords;
129    if world
130        .tiles
131        .query::<&VectorLayerBucketComponent>(coords)
132        .is_none()
133    {
134        return;
135    }
136    let uploaded = matches!(world.resources.get::<Eventually<VectorBufferPool>>(),
137        Some(Eventually::Initialized(pool)) if pool.has_layer(coords, &bucket.style_layer_id));
138    let clock = world.resources.get_or_init_mut::<ContentClock>();
139    clock.0 = clock.0.wrapping_add(1);
140    let revision = clock.0;
141    ensure(&mut world.tiles, coords);
142    let layer_id = bucket.style_layer_id.clone();
143    if let Some((layers, pending)) = world
144        .tiles
145        .query_mut::<(&mut VectorLayerBucketComponent, &mut LayerReplacements)>(coords)
146    {
147        layers.uploaded_stamp = None;
148        pending.revision = revision;
149        if !pending.delivered.contains(&layer_id) {
150            pending.delivered.push(layer_id.clone());
151        }
152        pending
153            .vector
154            .retain(|layer| layer.style_layer_id != layer_id);
155        if uploaded {
156            pending.vector.push(bucket);
157        } else {
158            commit_vector(layers, bucket);
159        }
160    }
161    let feature_paint = feature_paint.filter(|entries| !entries.is_empty());
162    if feature_paint.is_some()
163        && world
164            .tiles
165            .query::<&TileFeaturePaintComponent>(coords)
166            .is_none()
167    {
168        if let Some(mut tile) = world.tiles.spawn_mut(coords) {
169            tile.insert(TileFeaturePaintComponent::default());
170        }
171    }
172    if let Some(paint_store) = world
173        .tiles
174        .query_mut::<&mut TileFeaturePaintComponent>(coords)
175    {
176        if uploaded {
177            if let Some(entries) = feature_paint {
178                paint_store.pending.insert(layer_id, entries);
179            } else {
180                paint_store.pending.remove(&layer_id);
181            }
182        } else {
183            paint_store.pending.remove(&layer_id);
184            if let Some(entries) = feature_paint {
185                paint_store.committed.insert(layer_id, entries);
186            } else {
187                paint_store.committed.remove(&layer_id);
188            }
189        }
190    }
191}

193pub(crate) fn commit_vector(
194    layers: &mut VectorLayerBucketComponent,
195    bucket: AvailableVectorLayerBucket,
196) {
197    layers.uploaded_stamp = None;
198    layers.layers.retain(|layer| match layer {
199        VectorLayerBucket::AvailableLayer(layer) => layer.style_layer_id != bucket.style_layer_id,
200        VectorLayerBucket::Missing(layer) => layer.source_layer != bucket.source_layer,
201    });
202    layers
203        .layers
204        .push(VectorLayerBucket::AvailableLayer(bucket));
205}

207pub(crate) fn commit_pending_vector(tiles: &mut Tiles, coords: WorldTileCoords, index: usize) {
208    let promoted = tiles
209        .query_mut::<(&mut VectorLayerBucketComponent, &mut LayerReplacements)>(coords)
210        .and_then(|(layers, pending)| {
211            (index < pending.vector.len()).then(|| {
212                let replaced = pending.vector.remove(index);
213                let id = replaced.style_layer_id.clone();
214                commit_vector(layers, replaced);
215                id
216            })
217        });
218    if let Some(id) = promoted {
219        if let Some(paint_store) = tiles.query_mut::<&mut TileFeaturePaintComponent>(coords) {
220            if let Some(entries) = paint_store.pending.remove(&id) {
221                paint_store.committed.insert(id, entries);
222            } else {
223                paint_store.committed.remove(&id);
224            }
225        }
226    }
227}

229pub(crate) fn release_committed_cpu_geometry(
230    tiles: &mut Tiles,
231    coords: WorldTileCoords,
232    style_layer_id: &str,
233    release_colors: bool,
234) {
235    let keep_feature_indices = tiles
236        .query::<&TileFeaturePaintComponent>(coords)
237        .is_some_and(|store| store.committed.contains_key(style_layer_id));
238    let Some(layers) = tiles.query_mut::<&mut VectorLayerBucketComponent>(coords) else {
239        return;
240    };
241    for layer in &mut layers.layers {
242        if let VectorLayerBucket::AvailableLayer(avail) = layer {
243            if avail.style_layer_id == style_layer_id && avail.buffer.usable_indices > 0 {
244                avail.buffer.release_cpu_geometry();
245                if release_colors {
246                    if !keep_feature_indices {
247                        avail.feature_indices = Vec::new();
248                    }
249                    avail.feature_colors = Vec::new();
250                    avail.feature_sort_keys = Vec::new();
251                }
252            }
253        }
254    }
255}

257pub(crate) fn accept_symbols(world: &mut World, bucket: SymbolLayerData) {
258    let coords = bucket.coords;
259    if world
260        .tiles
261        .query::<&SymbolLayersDataComponent>(coords)
262        .is_none()
263    {
264        return;
265    }
266    let uploaded = matches!(world.resources.get::<Eventually<SymbolBufferPool>>(),
267        Some(Eventually::Initialized(pool)) if pool.has_layer(coords, &bucket.style_layer_id));
268    ensure(&mut world.tiles, coords);
269    if let Some((layers, pending)) = world
270        .tiles
271        .query_mut::<(&mut SymbolLayersDataComponent, &mut LayerReplacements)>(coords)
272    {
273        layers.uploaded_stamp = None;
274        layers.atlas_zoom_q32 = None;
275        pending
276            .symbols
277            .retain(|layer| layer.style_layer_id != bucket.style_layer_id);
278        if uploaded {
279            pending.symbols.push(bucket);
280        } else {
281            commit_symbols(layers, bucket);
282        }
283    }
284}

295pub(crate) fn vector_layers(
296    tiles: &Tiles,
297    coords: WorldTileCoords,
298) -> Vec<&AvailableVectorLayerBucket> {
299    let pending = tiles.query::<&LayerReplacements>(coords);
300    let mut layers: Vec<_> = tiles
301        .query::<&VectorLayerBucketComponent>(coords)
302        .into_iter()
303        .flat_map(|layers| &layers.layers)
304        .filter_map(|layer| match layer {
305            VectorLayerBucket::AvailableLayer(layer) => Some(layer),
306            VectorLayerBucket::Missing(_) => None,
307        })
308        .collect();
309    if let Some(pending) = pending {
310        for layer in &pending.vector {
311            layers.retain(|current| current.style_layer_id != layer.style_layer_id);
312            layers.push(layer);
313        }
314    }
315    layers
316}

318/// Whether any committed vector bucket at `coords` released its CPU indices after upload and
319/// subsequently had its GPU allocation evicted from [`VectorBufferPool`].
320pub(crate) fn has_evicted_cpu_released_vector_layers(
321    coords: WorldTileCoords,
322    world: &World,
323) -> bool {
324    let Some(Eventually::Initialized(pool)) = world.resources.get::<Eventually<VectorBufferPool>>()
325    else {
326        return false;
327    };
328    if !pool.has_capacity() {
329        return false;
330    }
331    let Some(layers) = world.tiles.query::<&VectorLayerBucketComponent>(coords) else {
332        return false;
333    };
334    let pending = world.tiles.query::<&LayerReplacements>(coords);
335    if layers
336        .uploaded_stamp
337        .is_some_and(|(_, eviction_gen)| eviction_gen == pool.eviction_generation())
338        && pending.is_none_or(|p| p.vector.is_empty())
339    {
340        return false;
341    }
342    layers.layers.iter().any(|layer| match layer {
343        VectorLayerBucket::AvailableLayer(bucket) => {
344            bucket.buffer.usable_indices > 0
345                && !bucket.buffer.has_cpu_indices()
346                && !pool.has_layer(coords, &bucket.style_layer_id)
347                && !pending.is_some_and(|p| {
348                    p.vector
349                        .iter()
350                        .any(|item| item.style_layer_id == bucket.style_layer_id)
351                })
352        }
353        VectorLayerBucket::Missing(_) => false,
354    })
355}

357/// Whether `coords` holds CPU-released vector or symbol buckets whose GPU pool allocations were
358/// evicted while no replacement is pending, requiring a fresh worker pass to restore geometry.
359pub(crate) fn has_evicted_gpu_geometry(
360    world: &World,
361    style: &crate::style::Style,
362    (vectors_disabled, symbols_disabled): (bool, bool),
363    coords: WorldTileCoords,
364) -> bool {
365    let rev = style.revision();
366    let pending = world.tiles.query::<&LayerReplacements>(coords);
367    if !vectors_disabled {
368        if let Some(Eventually::Initialized(pool)) =
369            world.resources.get::<Eventually<VectorBufferPool>>()
370        {
371            if pool.has_capacity() {
372                if let Some(layers) = world.tiles.query::<&VectorLayerBucketComponent>(coords) {
373                    let vector_stamped = layers.uploaded_stamp
374                        == Some((rev, pool.eviction_generation()))
375                        && pending.is_none_or(|p| p.vector.is_empty());
376                    if !vector_stamped {
377                        let evicted_vector = layers.layers.iter().any(|layer| match layer {
378                            VectorLayerBucket::AvailableLayer(bucket) => {
379                                bucket.buffer.usable_indices > 0
380                                    && !bucket.buffer.has_cpu_indices()
381                                    && !pool.has_layer(coords, &bucket.style_layer_id)
382                                    && !pending.is_some_and(|p| {
383                                        p.vector.iter().any(|item| {
384                                            item.style_layer_id == bucket.style_layer_id
385                                        })
386                                    })
387                                    && style
388                                        .layers
389                                        .iter()
390                                        .any(|l| l.id == bucket.style_layer_id && !l.is_hidden())
391                            }
392                            VectorLayerBucket::Missing(_) => false,
393                        });
394                        if evicted_vector {
395                            return true;
396                        }
397                    }
398                }
399            }
400        }
401    }
402    if !symbols_disabled {
403        if let Some(Eventually::Initialized(pool)) =
404            world.resources.get::<Eventually<SymbolBufferPool>>()
405        {
406            if pool.has_capacity() {
407                if let Some(symbols) = world.tiles.query::<&SymbolLayersDataComponent>(coords) {
408                    let symbol_stamped = symbols.uploaded_stamp
409                        == Some((rev, pool.eviction_generation()))
410                        && pending.is_none_or(|p| p.symbols.is_empty());
411                    if !symbol_stamped {
412                        let evicted_symbol = symbols.layers.iter().any(|layer| {
413                            layer.buffer.usable_indices > 0
414                                && layer.buffer.buffer.vertices.is_empty()
415                                && !pool.has_layer(coords, &layer.style_layer_id)
416                                && !pending.is_some_and(|p| {
417                                    p.symbols
418                                        .iter()
419                                        .any(|item| item.style_layer_id == layer.style_layer_id)
420                                })
421                                && style
422                                    .layers
423                                    .iter()
424                                    .any(|l| l.id == layer.style_layer_id && !l.is_hidden())
425                        });
426                        if evicted_symbol {
427                            return true;
428                        }
429                    }
430                }
431            }
432        }
433    }
434    false
435}

437/// Drops everything the removed style layers left in loaded tiles: their buckets, pending
438/// replacements and GPU buffers. Buckets are otherwise only ever replaced by layer id, so a layer
439/// that is no longer in the style would keep its geometry until the tile is evicted.
440pub(crate) fn purge_layers(world: &mut World, removed: &[String]) {
441    let coords: Vec<WorldTileCoords> = world.tiles.tiles.values().map(|tile| tile.coords).collect();
442    for at in &coords {
443        purge_vector_layers(world, *at, removed);
444        if let Some(component) = world.tiles.query_mut::<&mut SymbolLayersDataComponent>(*at) {
445            component.uploaded_stamp = None;
446            component.atlas_zoom_q32 = None;
447            component
448                .layers
449                .retain(|layer| !removed.contains(&layer.style_layer_id));
450        }
451        if let Some(pending) = world.tiles.query_mut::<&mut LayerReplacements>(*at) {
452            pending
453                .symbols
454                .retain(|layer| !removed.contains(&layer.style_layer_id));
455        }
456    }
457    if let Some(Eventually::Initialized(pool)) =
458        world.resources.get_mut::<Eventually<SymbolBufferPool>>()
459    {
460        for at in &coords {
461            for id in removed {
462                pool.remove_layer(*at, id);
463            }
464        }
465    }
466}

468/// Drops the buckets, pending replacements and GPU buffers of the non-symbol `layers` in one tile.
469fn purge_vector_layers(world: &mut World, at: WorldTileCoords, layers: &[String]) {
470    if let Some(component) = world.tiles.query_mut::<&mut VectorLayerBucketComponent>(at) {
471        component.uploaded_stamp = None;
472        component.layers.retain(|layer| match layer {
473            VectorLayerBucket::AvailableLayer(bucket) => !layers.contains(&bucket.style_layer_id),
474            VectorLayerBucket::Missing(_) => true,
475        });
476    }
477    if let Some(pending) = world.tiles.query_mut::<&mut LayerReplacements>(at) {
478        pending
479            .vector
480            .retain(|layer| !layers.contains(&layer.style_layer_id));
481    }
482    if let Some(paint_store) = world.tiles.query_mut::<&mut TileFeaturePaintComponent>(at) {
483        for id in layers {
484            paint_store.committed.remove(id);
485            paint_store.pending.remove(id);
486        }
487    }
488    if let Some(Eventually::Initialized(pool)) =
489        world.resources.get_mut::<Eventually<VectorBufferPool>>()
490    {
491        for id in layers {
492            pool.remove_layer(at, id);
493        }
494    }
