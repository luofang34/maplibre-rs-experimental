24pub(super) struct ColumnarLayer<'a> {

31    pub(super) has_hide_3d: bool,

34#[derive(Clone, Copy, PartialEq, Eq, Hash)]
35enum PropAtom<'a> {
36    Bool(bool),
37    Bits(u64),
38    Str(&'a str),
39}
40
41impl<'a> PropAtom<'a> {
42    #[inline]
43    fn from_ref(val: PropValueRef<'a>) -> Self {
44        match val {
45            PropValueRef::Bool(b) => Self::Bool(b),
46            PropValueRef::Str(s) => Self::Str(s),
47            other => {
48                let Value::Number(n) = prop_ref_to_value(other) else {
49                    return Self::Bits(0);
50                };
51                Self::Bits(if n == 0.0 { 0 } else { n.to_bits() })
52            }
53        }
54    }
55
56    #[inline]
57    fn to_value(self) -> Value {
58        match self {
59            Self::Bool(b) => Value::Bool(b),
60            Self::Bits(bits) => Value::Number(f64::from_bits(bits)),
61            Self::Str(s) => Value::String(s.to_owned()),
62        }
63    }
64}

66pub(super) fn decode_columnar_layer<'a>(
67    layer: Layer01<'a>,
68    decoder: &mut Decoder,
69) -> Result<ColumnarLayer<'a>, MltError> {
70    let parsed = layer.decode_all(decoder)?;
71    let extent_u32 = parsed.extent().get();
72    let extent_scale = if extent_u32 > 0 {
73        EXTENT / f64::from(extent_u32)
74    } else {
75        1.0
76    };
77    let col_names: Vec<String> = parsed.iterate_prop_names().map(|n| n.to_string()).collect();
78    let has_hide_3d = col_names.iter().any(|name| name == "hide_3d");
79    let count = parsed.feature_count();
80    let mut ids = Vec::with_capacity(count);
81    let mut geom_types = Vec::with_capacity(count);
82    let mut base_props = Vec::with_capacity(count);
83    let empty = empty_feature_properties();
84
85    {
86        let mut memo: FxHashMap<Vec<(u16, PropAtom<'_>)>, Arc<FeatureProperties>> =
87            FxHashMap::default();
88        let mut key_scratch: Vec<(u16, PropAtom<'_>)> = Vec::with_capacity(col_names.len());
89        let mut iter = parsed.iter_features();
90        while let Some(feat_res) = iter.next() {
91            let feat = feat_res?;
92            ids.push(feat.id());
93            geom_types.push(geometry_type_of(feat.geometry()));
94
95            key_scratch.clear();
96            for (col_idx, opt_val) in feat.iter_all_properties().enumerate() {
97                if let Some(val) = opt_val {
98                    key_scratch.push((col_idx as u16, PropAtom::from_ref(val)));
99                }
100            }
101            if key_scratch.is_empty() {
102                base_props.push(Arc::clone(empty));
103                continue;
104            }
105            if let Some(existing) = memo.get(key_scratch.as_slice()) {
106                base_props.push(Arc::clone(existing));
107                continue;
108            }
109            let mut props = FeatureProperties::default();
110            for &(col_idx, atom) in &key_scratch {
111                if let Some(name) = col_names.get(usize::from(col_idx)) {
112                    props.insert(name.clone(), atom.to_value());
113                }
114            }
115            let arc = if props.is_empty() {
116                Arc::clone(empty)
117            } else {
118                Arc::new(props)
119            };
120            memo.insert(key_scratch.clone(), Arc::clone(&arc));
121            base_props.push(arc);
122        }
123    }
124
125    let header = geozero::mvt::tile::Layer {
126        version: 2,
127        name: parsed.name().to_owned(),
128        extent: Some(extent_u32),
129        ..Default::default()
130    };
131
132    Ok(ColumnarLayer {
133        parsed,
134        extent_scale,
135        header,
136        ids,
137        geom_types,
138        base_props,
139        has_hide_3d,
140    })
141}

143pub(super) fn props_for_style(
144    col: &ColumnarLayer<'_>,
145    coords: WorldTileCoords,
146    state_ctx: Option<&FeatureStateSyncContext>,
147    reads_id: bool,
148    reads_geom: bool,
149) -> Result<Vec<Arc<FeatureProperties>>, MltError> {
150    if state_ctx.is_none() && !reads_id && !reads_geom {
151        return Ok(col.base_props.clone());
152    }
153
154    let count = col.base_props.len();
155    let mut out = Vec::with_capacity(count);
156    let extent = f64::from(col.parsed.extent().get()).max(1.0);
157
158    if reads_geom {
159        let mut iter = col.parsed.iter_features();
160        let mut idx = 0;
161        while let Some(feat_res) = iter.next() {
162            let feat = feat_res?;
163            let mut map = col
164                .base_props
165                .get(idx)
166                .map_or_else(FeatureProperties::default, |arc| (**arc).clone());
167            if reads_id {
168                let gtype = col
169                    .geom_types
170                    .get(idx)
171                    .copied()
172                    .unwrap_or(GeometryType::Unknown);
173                map.insert(
174                    GEOMETRY_TYPE_PROPERTY.to_owned(),
175                    Value::String(gtype.name().to_owned()),
176                );
177                if let Some(id) = feat.id() {
178                    map.insert(ID_PROPERTY.to_owned(), Value::Number(id as f64));
179                }
180            }
181            let text = lonlat_i32(feat.geometry(), coords, extent)
182                .to_geojson()
183                .to_string();
184            map.insert(GEOMETRY_PROPERTY.to_owned(), Value::String(text));
185            if let Some(ctx) = state_ctx {
186                crate::style::feature_state::sync_properties_feature_state(
187                    &mut map,
188                    feat.id(),
189                    ctx.promote_property.as_deref(),
190                    ctx.source_state.as_deref(),
191                    ctx.source_layer.as_deref(),
192                );
193            }
194            out.push(Arc::new(map));
195            idx += 1;
196        }
197        return Ok(out);
198    }
199
200    let empty = empty_feature_properties();
201    for idx in 0..count {
202        let id = col.ids.get(idx).copied().flatten();
203        let mut map = col
204            .base_props
205            .get(idx)
206            .map_or_else(FeatureProperties::default, |arc| (**arc).clone());
207        if reads_id {
208            let gtype = col
209                .geom_types
210                .get(idx)
211                .copied()
212                .unwrap_or(GeometryType::Unknown);
213            map.insert(
214                GEOMETRY_TYPE_PROPERTY.to_owned(),
215                Value::String(gtype.name().to_owned()),
216            );
217            if let Some(id_val) = id {
218                map.insert(ID_PROPERTY.to_owned(), Value::Number(id_val as f64));
219            }
220        }
221        if let Some(ctx) = state_ctx {
222            crate::style::feature_state::sync_properties_feature_state(
223                &mut map,
224                id,
225                ctx.promote_property.as_deref(),
226                ctx.source_state.as_deref(),
227                ctx.source_layer.as_deref(),
228            );
229        }
230        if map.is_empty() {
231            out.push(Arc::clone(empty));
232        } else {
233            out.push(Arc::new(map));
234        }
235    }
236    Ok(out)
237}

239pub(super) fn stream_layer_slice(
240    col: &ColumnarLayer<'_>,
241    selected: Option<&[usize]>,
242    processor: &mut impl FeatureProcessor,
243) -> GeoResult<()> {
244    processor.dataset_begin(Some(col.parsed.name()))?;
245    let mut iter = col.parsed.iter_features();
246    match selected {
247        None => {
248            let mut out_idx = 0_u64;
249            while let Some(feat_res) = iter.next() {
250                let feat = feat_res
251                    .map_err(|err| geozero::error::GeozeroError::Geometry(err.to_string()))?;
252                processor.feature_begin(out_idx)?;
253                processor.geometry_begin()?;
254                stream_mlt_geometry(feat.geometry(), processor)?;
255                processor.geometry_end()?;
256                processor.feature_end(out_idx)?;
257                out_idx += 1;
258            }
259        }
260        Some(indices) => {
261            let mut sel_pos = 0_usize;
262            let mut feat_idx = 0_usize;
263            while sel_pos < indices.len() {
264                let Some(feat_res) = iter.next() else {
265                    break;
266                };
267                if indices[sel_pos] == feat_idx {
268                    let feat = feat_res
269                        .map_err(|err| geozero::error::GeozeroError::Geometry(err.to_string()))?;
270                    let out_idx = sel_pos as u64;
271                    processor.feature_begin(out_idx)?;
272                    processor.geometry_begin()?;
273                    stream_mlt_geometry(feat.geometry(), processor)?;
274                    processor.geometry_end()?;
275                    processor.feature_end(out_idx)?;
276                    sel_pos += 1;
277                }
278                feat_idx += 1;
279            }
280        }
281    }
282    processor.dataset_end()
283}

285pub(super) fn stream_mlt_geometry(
286    geom: &Geo<i32>,
287    processor: &mut impl GeomProcessor,
288) -> GeoResult<()> {
289    match geom {
290        Geo::Point(p) => {
291            processor.point_begin(0)?;
292            processor.xy(f64::from(p.x()), f64::from(p.y()), 0)?;
293            processor.point_end(0)
294        }
295        Geo::MultiPoint(mp) => match mp.0.as_slice() {
296            [] => Ok(()),
297            [single] => {
298                processor.point_begin(0)?;
299                processor.xy(f64::from(single.x()), f64::from(single.y()), 0)?;
300                processor.point_end(0)
301            }
302            many => {
303                processor.multipoint_begin(many.len(), 0)?;
304                for (i, p) in many.iter().enumerate() {
305                    processor.xy(f64::from(p.x()), f64::from(p.y()), i)?;
306                }
307                processor.multipoint_end(0)
308            }
309        },
310        Geo::Line(seg) => {
311            processor.linestring_begin(true, 2, 0)?;
312            processor.xy(f64::from(seg.start.x), f64::from(seg.start.y), 0)?;
313            processor.xy(f64::from(seg.end.x), f64::from(seg.end.y), 1)?;
314            processor.linestring_end(true, 0)
315        }
316        Geo::LineString(ls) => stream_linestring(ls, true, 0, processor),
317        Geo::MultiLineString(mls) => match mls.0.as_slice() {
318            [] => Ok(()),
319            [single] => stream_linestring(single, true, 0, processor),
320            many => {
321                processor.multilinestring_begin(many.len(), 0)?;
322                for (i, ls) in many.iter().enumerate() {
323                    stream_linestring(ls, false, i, processor)?;
324                }
325                processor.multilinestring_end(0)
326            }
327        },
328        Geo::Polygon(poly) => stream_polygon(poly, true, 0, processor),
329        Geo::MultiPolygon(mpoly) => match mpoly.0.as_slice() {
330            [] => Ok(()),
331            [single] => stream_polygon(single, true, 0, processor),
332            many => {
333                processor.multipolygon_begin(many.len(), 0)?;
334                for (i, poly) in many.iter().enumerate() {
335                    stream_polygon(poly, false, i, processor)?;
336                }
337                processor.multipolygon_end(0)
338            }
339        },
340        Geo::Rect(r) => stream_polygon(&r.to_polygon(), true, 0, processor),
341        Geo::Triangle(t) => stream_polygon(&t.to_polygon(), true, 0, processor),
342        Geo::GeometryCollection(gc) => {
343            for part in &gc.0 {
344                stream_mlt_geometry(part, processor)?;
345            }
346            Ok(())
347        }
348    }
349}

351fn stream_linestring(
352    ls: &geo_types::LineString<i32>,
353    tagged: bool,
354    idx: usize,
355    processor: &mut impl GeomProcessor,
356) -> GeoResult<()> {
357    if ls.0.is_empty() {
358        return Ok(());
359    }
360    processor.linestring_begin(tagged, ls.0.len(), idx)?;
361    for (i, c) in ls.0.iter().enumerate() {
362        processor.xy(f64::from(c.x), f64::from(c.y), i)?;
363    }
364    processor.linestring_end(tagged, idx)
365}
366
367fn stream_polygon(
368    poly: &geo_types::Polygon<i32>,
369    tagged: bool,
370    idx: usize,
371    processor: &mut impl GeomProcessor,
372) -> GeoResult<()> {
373    let ring_count = 1 + poly.interiors().len();
374    processor.polygon_begin(tagged, ring_count, idx)?;
375    stream_ring(poly.exterior(), 0, processor)?;
376    for (i, hole) in poly.interiors().iter().enumerate() {
377        stream_ring(hole, i + 1, processor)?;
378    }
379    processor.polygon_end(tagged, idx)
380}
381
382fn stream_ring(
383    ring: &geo_types::LineString<i32>,
384    ring_idx: usize,
385    processor: &mut impl GeomProcessor,
386) -> GeoResult<()> {
387    let coords = ring.0.as_slice();
388    if coords.len() < 3 {
389        return Ok(());
390    }
391    let already_closed = coords.first() == coords.last();
392    if already_closed && coords.len() < 4 {
393        return Ok(());
394    }
395    let unclosed_len = if already_closed {
396        coords.len() - 1
397    } else {
398        coords.len()
399    };
400    processor.linestring_begin(false, unclosed_len, ring_idx)?;
401    for (i, c) in coords.iter().enumerate() {
402        processor.xy(f64::from(c.x), f64::from(c.y), i)?;
403    }
404    if !already_closed {
405        let first = coords[0];
406        processor.xy(f64::from(first.x), f64::from(first.y), coords.len())?;
407    }
408    processor.linestring_end(false, ring_idx)
409}

411pub(super) fn geometry_type_of(geom: &Geo<i32>) -> GeometryType {
412    match geom {
413        Geo::Point(_) | Geo::MultiPoint(_) => GeometryType::Point,
414        Geo::Line(_) | Geo::LineString(_) | Geo::MultiLineString(_) => GeometryType::LineString,
415        Geo::Polygon(_) | Geo::MultiPolygon(_) | Geo::Rect(_) | Geo::Triangle(_) => {
416            GeometryType::Polygon
417        }
418        Geo::GeometryCollection(gc) => gc.0.first().map_or(GeometryType::Unknown, geometry_type_of),
419    }
420}

422#[inline]
423pub(super) fn prop_ref_to_value(val: PropValueRef<'_>) -> Value {
424    match val {
425        PropValueRef::Bool(b) => Value::Bool(b),
426        PropValueRef::I8(i) => Value::Number(f64::from(i)),
427        PropValueRef::U8(u) => Value::Number(f64::from(u)),
428        PropValueRef::I32(i) => Value::Number(f64::from(i)),
429        PropValueRef::U32(u) => Value::Number(f64::from(u)),
430        PropValueRef::I64(i) => Value::Number(i as f64),
431        PropValueRef::U64(u) => Value::Number(u as f64),
432        PropValueRef::F32(f) => Value::Number(f64::from(f)),
433        PropValueRef::F64(f) => Value::Number(f),
434        PropValueRef::Str(s) => Value::String(s.to_owned()),
435    }
436}

438fn lonlat_i32(geometry: &Geo<i32>, coords: WorldTileCoords, extent: f64) -> ExprGeometry {
439    let tiles = 2_f64.powi(i32::from(u8::from(coords.z)));
440    let convert = |c: &geo_types::Coord<i32>| {
441        let x = (f64::from(coords.x) + f64::from(c.x) / extent) / tiles;
442        let y = (f64::from(coords.y) + f64::from(c.y) / extent) / tiles;
443        let longitude = x * 360.0 - 180.0;
444        let latitude = (std::f64::consts::PI * (1.0 - 2.0 * y))
445            .sinh()
446            .atan()
447            .to_degrees();
448        [longitude, latitude]
449    };
450    let mut out = ExprGeometry::default();
451    collect_i32(geometry, &convert, &mut out);
452    out
453}

455fn collect_i32(
456    geometry: &Geo<i32>,
457    convert: &dyn Fn(&geo_types::Coord<i32>) -> [f64; 2],
458    out: &mut ExprGeometry,
459) {
460    let line = |ls: &geo_types::LineString<i32>| ls.0.iter().map(convert).collect::<Vec<_>>();
461    let polygon = |poly: &geo_types::Polygon<i32>| {
462        std::iter::once(poly.exterior())
463            .chain(poly.interiors())
464            .map(line)
465            .collect::<Vec<_>>()
466    };
467    match geometry {
468        Geo::Point(p) => out.points.push(convert(&p.0)),
469        Geo::MultiPoint(mp) => out.points.extend(mp.0.iter().map(|p| convert(&p.0))),
470        Geo::Line(seg) => out.lines.push(vec![convert(&seg.start), convert(&seg.end)]),
471        Geo::LineString(ls) => out.lines.push(line(ls)),
472        Geo::MultiLineString(mls) => out.lines.extend(mls.0.iter().map(line)),
473        Geo::Polygon(p) => out.polygons.push(polygon(p)),
474        Geo::MultiPolygon(mp) => out.polygons.extend(mp.0.iter().map(polygon)),
475        Geo::GeometryCollection(gc) => gc.0.iter().for_each(|part| collect_i32(part, convert, out)),
476        Geo::Rect(r) => out.polygons.push(polygon(&r.to_polygon())),
477        Geo::Triangle(t) => out.polygons.push(polygon(&t.to_polygon())),
478    }
