35pub(crate) fn process_mlt_tile_with_assets<T: VectorTransferables, C: Context>(
36    data: &[u8],
37    tile_request: VectorTileRequest,
38    context: &mut ProcessVectorContext<T, C>,
39    atlas: Arc<SymbolAtlas>,
40) -> Result<(), ProcessVectorError> {
41    let raw_layers =
42        Parser::default()
43            .parse_layers(data)
44            .map_err(|err| ProcessVectorError::Decoding {
45                coords: tile_request.coords,
46                source: Box::new(err),
47            })?;
48    let index_all = context.index_geometry() && T::LayerIndexed::requires_geometry_index();
49
50    let mut ordered_layers: Vec<&StyleLayer> = tile_request.layers.iter().collect();
51    ordered_layers.sort_by(|a, b| {
52        a.source_layer
53            .as_deref()
54            .unwrap_or("")
55            .cmp(b.source_layer.as_deref().unwrap_or(""))
56            .then_with(|| a.index.cmp(&b.index))
57    });
58
59    let mut decoder = Decoder::default();
60    let mut decoded_layers: FxHashMap<&str, ColumnarLayer<'_>> = FxHashMap::default();
61    let mut decode_order: Vec<&str> = Vec::new();
62    for layer in raw_layers {
63        let Layer::Tag01(layer01) = layer else {
64            continue;
65        };
66        let name = layer01.name();
67        let wanted = index_all
68            || ordered_layers
69                .iter()
70                .any(|sl| sl.source_layer.as_deref() == Some(name));
71        if !wanted {
72            continue;
73        }
74        let col = decode_columnar_layer(layer01, &mut decoder).map_err(|err| {
75            ProcessVectorError::Decoding {
76                coords: tile_request.coords,
77                source: Box::new(err),
78            }
79        })?;
80        decode_order.push(name);
81        decoded_layers.insert(name, col);
82    }
83
84    let gpu_atlas = atlas.gpu_upload_handle();
85    for style in &ordered_layers {
86        let Some(layer_name) = style.source_layer.as_deref() else {
87            tracing::error!(layer = %style.id, "vector style layer misses a required attribute");
88            continue;
89        };
90        if style.paint.is_none() {
91            tracing::error!(layer = %style.id, "vector style layer misses a required attribute");
92            continue;
93        }
94        let Some(col) = decoded_layers.get(layer_name) else {
95            context.layer_missing(&tile_request.coords, layer_name)?;
96            continue;
97        };
98
99        // Stepped extrusion podium bases with `hide_3d` reuse the exact polygon-part unhider.
100        if col.has_hide_3d && matches!(style.paint, Some(LayerPaint::FillExtrusion(_))) {
101            let mut mvt_decoder = Decoder::default();
102            if let Ok(tile_layer) = col.parsed.clone().into_tile(&mut mvt_decoder) {
103                let mvt_layer = super::convert_layer(&tile_layer);
104                process_layer_from_mvt(
105                    Some(&mvt_layer),
106                    layer_name,
107                    None,
108                    style,
109                    &tile_request,
110                    context,
111                    (&atlas, &gpu_atlas),
112                )?;
113                continue;
114            }
115        }
116
117        process_columnar_style_layer(
118            col,
119            layer_name,
120            style,
121            &tile_request,
122            context,
123            (&atlas, &gpu_atlas),
124        )?;
125    }
126
127    if context.index_geometry() {
128        let geometries = if index_all {
129            let mut index = IndexProcessor::new();
130            for name in decode_order {
131                let Some(col) = decoded_layers.get(name) else {
132                    continue;
133                };
134                index.set_coordinate_scale(col.extent_scale);
135                let state_ctx = context.state_context_for_layer(name);
136                let props =
137                    props_for_style(col, tile_request.coords, state_ctx.as_ref(), false, false)
138                        .unwrap_or_else(|_| col.base_props.clone());
139                index.set_feature_state_context(state_ctx);
140                index.begin_layer(name, col.ids.clone());
141                index.set_shared_properties(props);
142                if let Err(error) = stream_layer_slice(col, None, &mut index) {
143                    tracing::warn!(coords = %tile_request.coords, layer = %name, ?error, "skipping MLT query index for layer");
144                }
145            }
146            index.get_geometries()
147        } else {
148            Vec::new()
149        };
150        let source = tile_request
151            .layers
152            .iter()
153            .find_map(|layer| layer.source.clone());
154        context.layer_indexing_finished(&tile_request.coords, source, geometries)?;
155    }
156    context.tile_finished(&tile_request.coords)?;
157    Ok(())
158}

160fn process_columnar_style_layer<T: VectorTransferables, C: Context>(
161    col: &ColumnarLayer<'_>,
162    layer_name: &str,
163    style: &StyleLayer,
164    request: &VectorTileRequest,
165    context: &mut ProcessVectorContext<T, C>,
166    (layout_atlas, gpu_atlas): (&Arc<SymbolAtlas>, &Arc<SymbolAtlas>),
167) -> Result<(), ProcessVectorError> {
168    let state_ctx = context.state_context_for_layer(layer_name);
169    let reads_id = crate::style::expression::layer_reads_identity(style);
170    let reads_geom = crate::style::expression::layer_reads_geometry(style);
171    let props = props_for_style(
172        col,
173        request.coords,
174        state_ctx.as_ref(),
175        reads_id,
176        reads_geom,
177    )
178    .map_err(|err| ProcessVectorError::Decoding {
179        coords: request.coords,
180        source: Box::new(err),
181    })?;
182
183    let selected: Option<Vec<usize>> = match &style.filter {
184        Some(raw_filter) => match Filter::parse(raw_filter) {
185            Ok(filter) => {
186                let zoom = request.style_zoom();
187                let mut indices = Vec::new();
188                for (idx, prop) in props.iter().enumerate() {
189                    let gtype = col
190                        .geom_types
191                        .get(idx)
192                        .copied()
193                        .unwrap_or(GeometryType::Unknown);
194                    let id = col
195                        .ids
196                        .get(idx)
197                        .copied()
198                        .flatten()
199                        .map(|v| Value::Number(v as f64));
200                    if filter.evaluate(&FeatureContext {
201                        properties: prop,
202                        geometry_type: gtype,
203                        id,
204                        zoom,
205                    }) {
206                        indices.push(idx);
207                    }
208                }
209                Some(indices)
210            }
211            Err(error) => {
212                tracing::error!(layer = %style.id, %error, "unsupported filter; the layer renders nothing");
213                context.layer_missing(&request.coords, layer_name)?;
214                return Ok(());
215            }
216        },
217        None => None,
218    };
219
220    let sel_slice = selected.as_deref();
221    let (sliced_ids, sliced_props) = slice_ids_and_props(&col.ids, &props, sel_slice);
222
223    match style.paint.as_ref() {
224        Some(
225            paint @ (LayerPaint::Line(_)
226            | LayerPaint::Fill(_)
227            | LayerPaint::FillExtrusion(_)
228            | LayerPaint::Circle(_)
229            | LayerPaint::Heatmap(_)),
230        ) => {
231            #[cfg(target_arch = "wasm32")]
232            let uniform_paint =
233                crate::vector::upload_system::paint::uniform_color(style, request.style_zoom())
234                    .is_some()
235                    && style.type_ != "circle"
236                    && !crate::style::feature_state::layer_paint_reads_feature_state(style);
237
238            let mut tessellator = build_zero_tessellator(paint, request);
239            tessellator.set_layer_strokes(style);
240            tessellator.coordinate_scale = col.extent_scale;
241            tessellator.source_ids = sliced_ids;
242            tessellator.shared_properties = sliced_props;
243            tessellator.feature_state_ctx = state_ctx;
244            tessellator.feature_paint_retention =
245                crate::vector::feature_paint::feature_paint_retention(style);
246            tessellator.sort_key = crate::vector::tessellation::SortKeys::by(
247                crate::vector::tessellation::sort_key_of(style),
248            );
249            #[cfg(target_arch = "wasm32")]
250            if uniform_paint
251                && tessellator.pattern_property.is_none()
252                && tessellator.outline_property.is_none()
253                && !tessellator.line_gradient
254            {
255                tessellator.omit_feature_colors = true;
256            }
257            if let Err(error) = stream_layer_slice(col, sel_slice, &mut tessellator) {
258                tracing::error!(coords = %request.coords, layer = %layer_name, ?error, "MLT vector tessellation failed");
259                return context.layer_missing(&request.coords, layer_name);
260            }
261            tessellator.apply_sort_keys();
262            let mut sort_keys = tessellator.sort_key_values();
263            #[cfg(target_arch = "wasm32")]
264            if uniform_paint && tessellator.pattern_property.is_none() && !tessellator.line_gradient
265            {
266                tessellator.feature_colors = Vec::new();
267            }
268            tessellator.buffer.vertices.shrink_to_fit();
269            tessellator.buffer.indices.shrink_to_fit();
270            tessellator.feature_indices.shrink_to_fit();
271            tessellator.feature_colors.shrink_to_fit();
272            sort_keys.shrink_to_fit();
273            tessellator.feature_paint.shrink_to_fit();
274            context.layer_tessellation_finished(
275                &request.coords,
276                tessellator.buffer.into(),
277                tessellator.feature_indices,
278                (
279                    tessellator.feature_colors,
280                    sort_keys,
281                    tessellator.feature_paint,
282                ),
283                col.header.clone(),
284                style.id.clone(),
285            )
286        }
287        Some(LayerPaint::Symbol(paint)) => {
288            let mut tessellator = TextTessellator::with_assets(
289                paint.clone(),
290                request.style_zoom(),
291                layout_atlas.clone(),
292            );
293            tessellator.coordinate_scale = col.extent_scale;
294            tessellator.overscaling = request.overscaling();
295            tessellator.source_ids = sliced_ids;
296            tessellator.shared_properties = sliced_props;
297            tessellator.feature_state_ctx = state_ctx;
298            if let Err(error) = stream_layer_slice(col, sel_slice, &mut tessellator) {
299                tracing::error!(coords = %request.coords, layer = %layer_name, ?error, "MLT symbol tessellation failed");
300                return context.layer_missing(&request.coords, layer_name);
301            }
302            tessellator.finish();
303            tessellator.quad_buffer.vertices.shrink_to_fit();
304            tessellator.quad_buffer.indices.shrink_to_fit();
305            tessellator.features.shrink_to_fit();
306            let has_geometry = !tessellator.quad_buffer.indices.is_empty();
307            context.symbol_layer_tessellation_finished(DefaultSymbolLayerTessellated {
308                coords: request.coords,
309                buffer: tessellator.quad_buffer.into(),
310                features: tessellator.features,
311                atlas: has_geometry.then(|| gpu_atlas.clone()),
312                layer_data: col.header.clone(),
313                style_layer_id: style.id.clone(),
314            })
315        }
316        _ => Ok(()),
317    }
318}

320fn slice_ids_and_props(
321    ids: &[Option<u64>],
322    props: &[Arc<FeatureProperties>],
323    selected: Option<&[usize]>,
324) -> (Vec<Option<u64>>, Vec<Arc<FeatureProperties>>) {
325    match selected {
326        Some(indices) => (
327            indices
328                .iter()
329                .map(|&idx| ids.get(idx).copied().flatten())
330                .collect(),
331            indices
332                .iter()
333                .filter_map(|&idx| props.get(idx).cloned())
334                .collect(),
335        ),
336        None => (ids.to_vec(), props.to_vec()),
337    }
338}

340pub(crate) fn index_mlt_bytes(
341    data: &[u8],
342    promote_id: Option<&PromoteId>,
343    feature_state: Option<&Arc<crate::style::feature_state::SourceFeatureState>>,
344) -> Vec<IndexedGeometry<f64>> {
345    let Ok(raw_layers) = Parser::default().parse_layers(data) else {
346        return Vec::new();
347    };
348    let mut decoder = Decoder::default();
349    let mut index = IndexProcessor::new();
350    for layer in raw_layers {
351        let Layer::Tag01(layer01) = layer else {
352            continue;
353        };
354        let Ok(col) = decode_columnar_layer(layer01, &mut decoder) else {
355            continue;
356        };
357        let name = col.parsed.name();
358        let state_ctx =
359            (promote_id.is_some() || feature_state.is_some()).then(|| FeatureStateSyncContext {
360                promote_property: promote_id
361                    .and_then(|p| p.property_for(Some(name)))
362                    .map(str::to_owned),
363                source_state: feature_state.cloned(),
364                source_layer: Some(name.to_owned()),
365            });
366        let props = if let Some(ctx) = &state_ctx {
367            let mut synced = Vec::with_capacity(col.base_props.len());
368            for (idx, base) in col.base_props.iter().enumerate() {
369                let mut map = (**base).clone();
370                crate::style::feature_state::sync_properties_feature_state(
371                    &mut map,
372                    col.ids.get(idx).copied().flatten(),
373                    ctx.promote_property.as_deref(),
374                    ctx.source_state.as_deref(),
375                    ctx.source_layer.as_deref(),
376                );
377                synced.push(Arc::new(map));
378            }
379            synced
380        } else {
381            col.base_props.clone()
382        };
383        index.set_coordinate_scale(col.extent_scale);
384        index.set_feature_state_context(state_ctx);
385        index.begin_layer(name, col.ids.clone());
386        index.set_shared_properties(props);
387        let _ = stream_layer_slice(&col, None, &mut index);
388    }
389    index.get_geometries()
390}

392pub(crate) fn visit_mlt_features(
393    data: &[u8],
394    keep_layer: impl Fn(&str) -> bool,
395    mut visit: impl FnMut(&str, Option<u64>, GeometryType, &FeatureProperties),
396) {
397    let Ok(raw_layers) = Parser::default().parse_layers(data) else {
398        return;
399    };
400    let mut decoder = Decoder::default();
401    let empty = empty_feature_properties();
402    let mut scratch = FeatureProperties::new();
403    for layer in raw_layers {
404        let Layer::Tag01(layer01) = layer else {
405            continue;
406        };
407        let name = layer01.name();
408        if !keep_layer(name) {
409            continue;
410        }
411        let Ok(parsed) = layer01.decode_all(&mut decoder) else {
412            continue;
413        };
414        let col_names: Vec<String> = parsed.iterate_prop_names().map(|n| n.to_string()).collect();
415        let mut iter = parsed.iter_features();
416        while let Some(Ok(feat)) = iter.next() {
417            scratch.clear();
418            for (col_idx, opt_val) in feat.iter_all_properties().enumerate() {
419                if let (Some(col_name), Some(val)) = (col_names.get(col_idx), opt_val) {
420                    scratch.insert(col_name.clone(), prop_ref_to_value(val));
421                }
422            }
423            let props_ref: &FeatureProperties = if scratch.is_empty() { empty } else { &scratch };
424            visit(
425                name,
426                feat.id(),
427                geometry_type_of(feat.geometry()),
428                props_ref,
429            );
430        }
431    }
