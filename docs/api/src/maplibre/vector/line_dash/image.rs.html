3/// What fills a line instead of its colour, when something does.
4#[derive(Clone, Debug, Default, PartialEq)]
5pub(super) enum LineImage {
6    /// The line takes its colour from the layer.
7    #[default]
8    None,
9    /// Colours along the line, from `line-gradient`.
10    Gradient(Vec<[u8; 4]>),
11    /// An image repeated along the line, from `line-pattern`: its size in layout pixels and its
12    /// RGBA bytes.
13    Pattern {
14        width: u32,
15        height: u32,
16        display: [f32; 2],
17        data: Vec<u8>,
18    },
19}

21/// The uniform of a dash entry: the dash period, what fills the line, and the size of a pattern
22/// or, for a dash, whether its texture has the rows of round caps and how far the fractional
23/// zoom stretches it.
24pub(super) fn image_uniform(period: f32, image: &LineImage, round: bool, scale: f32) -> [f32; 4] {
25    let round = f32::from(round);
26    match image {
27        LineImage::None => [period, 0.0, round, scale],
28        // A stepped gradient's ramp is spread over the progress on a logarithmic scale.
29        LineImage::Gradient(texels)
30            if texels.len() == crate::style::line_gradient::STEP_RAMP_TEXELS =>
31        {
32            [period, 1.25, round, scale]
33        }
34        LineImage::Gradient(_) => [period, 1.0, round, scale],
35        LineImage::Pattern { display, .. } => [period, 2.0 + scale, display[0], display[1]],
36    }
37}

39/// The image a line layer draws instead of its colour at a zoom.
40pub(super) fn line_image(
41    paint: &crate::style::layer::LinePaint,
42    style: &Style,
43    zoom: f64,
44) -> LineImage {
45    if let Some(gradient) = &paint.line_gradient {
46        return LineImage::Gradient(crate::style::line_gradient::ramp(gradient));
47    }
48    let Some(pattern) = &paint.line_pattern else {
49        return LineImage::None;
50    };
51    let name = StyleProperty::<crate::style::layer::TextField>::parse(pattern)
52        .evaluate_at_zoom(zoom)
53        .map(|name| name.0);
54    name.and_then(|name| style.images.get(&name))
55        .and_then(pattern_image)
56        .unwrap_or(LineImage::None)
57}

59/// The pattern an image makes, when its pixels fit its size.
60pub(super) fn pattern_image(image: &crate::style::StyleImage) -> Option<LineImage> {
61    if image.width == 0
62        || image.height == 0
63        || image.data.len() != image.width as usize * image.height as usize * 4
64    {
65        return None;
66    }
67    let ratio = image.pixel_ratio.max(0.01);
68    Some(LineImage::Pattern {
69        width: image.width,
70        height: image.height,
71        display: [image.width as f32 / ratio, image.height as f32 / ratio],
72        data: super::super::pattern::premultiplied(&image.data),
73    })
74}

76/// The gradient ramp or pattern as a texture, or one white texel for a plain line.
77pub(super) fn ramp_texture(
78    device: &wgpu::Device,
79    queue: &crate::render::upload_queue::UploadQueue,
80    image: &LineImage,
81) -> (wgpu::TextureView, wgpu::Sampler) {
82    let white = [255_u8; 4];
83    let (width, height, bytes): (u32, u32, &[u8]) = match image {
84        LineImage::None => (1, 1, &white),
85        LineImage::Gradient(texels) => (texels.len() as u32, 1, bytemuck::cast_slice(texels)),
86        LineImage::Pattern {
87            width,
88            height,
89            data,
90            ..
91        } => (*width, *height, data),
92    };
93    let texture = device.create_texture(&wgpu::TextureDescriptor {
94        label: Some("line image"),
95        size: wgpu::Extent3d {
96            width,
97            height,
98            depth_or_array_layers: 1,
99        },
100        mip_level_count: 1,
101        sample_count: 1,
102        dimension: wgpu::TextureDimension::D2,
103        format: wgpu::TextureFormat::Rgba8Unorm,
104        usage: wgpu::TextureUsages::TEXTURE_BINDING | wgpu::TextureUsages::COPY_DST,
105        view_formats: &[],
106    });
107    queue.write_texture(
108        texture.as_image_copy(),
109        bytes,
110        wgpu::TexelCopyBufferLayout {
111            offset: 0,
112            bytes_per_row: Some(width * 4),
113            rows_per_image: Some(height),
114        },
115        texture.size(),
116    );
117    // The hard edges of a stepped gradient stay hard.
118    let filter = match image {
119        LineImage::Gradient(texels) if texels.len() > crate::style::line_gradient::RAMP_TEXELS => {
120            wgpu::FilterMode::Nearest
121        }
122        _ => wgpu::FilterMode::Linear,
123    };
124    let sampler = device.create_sampler(&wgpu::SamplerDescriptor {
125        address_mode_u: wgpu::AddressMode::ClampToEdge,
126        address_mode_v: wgpu::AddressMode::ClampToEdge,
127        mag_filter: filter,
128        min_filter: filter,
129        ..Default::default()
130    });
131    (texture.create_view(&Default::default()), sampler)
132}
