14const KIND_FEATURE: u8 = 0;
15const KIND_LINE: u8 = 1;
16const KIND_FILL_OUTLINE: u8 = 2;
17
18/// Serializes `entries` into a flat byte buffer, deduplicating shared `Arc<FeatureProperties>`
19/// allocations across entries.
20pub fn encode_feature_paint(entries: &[BucketFeatureEntry]) -> Vec<u8> {
21    let mut prop_blobs = Vec::new();
22    let mut prop_map: FxHashMap<usize, u32> = FxHashMap::default();
23    let mut prop_count = 0_u32;
24
25    let mut entry_bytes = Vec::with_capacity(entries.len() * 8);
26    for entry in entries {
27        let ptr = Arc::as_ptr(&entry.properties) as usize;
28        let group_idx = *prop_map.entry(ptr).or_insert_with(|| {
29            let idx = prop_count;
30            prop_count += 1;
31            encode_properties(&entry.properties, &mut prop_blobs);
32            idx
33        });
34        entry_bytes.extend_from_slice(&group_idx.to_le_bytes());
35        let kind_bits = match entry.kind {
36            BucketEntryKind::Feature => KIND_FEATURE,
37            BucketEntryKind::Line { .. } => KIND_LINE,
38            BucketEntryKind::FillOutline => KIND_FILL_OUTLINE,
39        };
40        let flags = (u8::from(entry.raw_id.is_some())) | (kind_bits << 1);
41        entry_bytes.push(flags);
42        if let Some(id) = entry.raw_id {
43            entry_bytes.extend_from_slice(&id.to_le_bytes());
44        }
45        if let BucketEntryKind::Line { length } = entry.kind {
46            entry_bytes.extend_from_slice(&length.to_le_bytes());
47        }
48    }
49
50    let mut out = Vec::with_capacity(8 + prop_blobs.len() + entry_bytes.len());
51    out.extend_from_slice(&prop_count.to_le_bytes());
52    out.extend_from_slice(&prop_blobs);
53    out.extend_from_slice(&(entries.len() as u32).to_le_bytes());
54    out.extend_from_slice(&entry_bytes);
55    out
56}

58/// Reconstructs [`BucketFeatureEntry`] records from bytes produced by [`encode_feature_paint`],
59/// preserving shared `Arc<FeatureProperties>` identity across entries.
60pub fn decode_feature_paint(mut buf: &[u8]) -> Option<Vec<BucketFeatureEntry>> {
61    let prop_count = read_u32(&mut buf)? as usize;
62    let mut groups: Vec<Arc<FeatureProperties>> = Vec::with_capacity(prop_count.min(4096));
63    for _ in 0..prop_count {
64        groups.push(Arc::new(decode_properties_from(&mut buf)?));
65    }
66    let entry_count = read_u32(&mut buf)? as usize;
67    let mut entries = Vec::with_capacity(entry_count.min(16_384));
68    for _ in 0..entry_count {
69        let group_idx = read_u32(&mut buf)? as usize;
70        let properties = Arc::clone(groups.get(group_idx)?);
71        let flags = read_u8(&mut buf)?;
72        let raw_id = if flags & 1 != 0 {
73            if buf.len() < 8 {
74                return None;
75            }
76            let (head, rest) = buf.split_at(8);
77            buf = rest;
78            Some(u64::from_le_bytes(head.try_into().ok()?))
79        } else {
80            None
81        };
82        let kind = match flags >> 1 {
83            KIND_FEATURE => BucketEntryKind::Feature,
84            KIND_LINE => {
85                if buf.len() < 4 {
86                    return None;
87                }
88                let (head, rest) = buf.split_at(4);
89                buf = rest;
90                BucketEntryKind::Line {
91                    length: f32::from_le_bytes(head.try_into().ok()?),
92                }
93            }
94            KIND_FILL_OUTLINE => BucketEntryKind::FillOutline,
95            _ => return None,
96        };
97        entries.push(BucketFeatureEntry {
98            raw_id,
99            properties,
100            kind,
101        });
102    }
103    Some(entries)
104}

106#[cfg(test)]
107mod tests {
108    use std::sync::Arc;
109
110    use super::{decode_feature_paint, encode_feature_paint};
111    use crate::{
112        style::expression::{FeatureProperties, Value},
113        vector::feature_paint::{BucketEntryKind, BucketFeatureEntry},
114    };
115
116    #[test]
117    fn wire_round_trips_entries_and_preserves_shared_arcs() {
118        let mut props_a = FeatureProperties::new();
119        props_a.insert("name".into(), Value::String("A".into()));
120        props_a.insert("\u{1}state:hover".into(), Value::Bool(true));
121        let shared = Arc::new(props_a);
122
123        let mut props_b = FeatureProperties::new();
124        props_b.insert("rank".into(), Value::Number(3.5));
125        let single = Arc::new(props_b);
126
127        let input = vec![
128            BucketFeatureEntry {
129                raw_id: Some(42),
130                properties: Arc::clone(&shared),
131                kind: BucketEntryKind::Feature,
132            },
133            BucketFeatureEntry {
134                raw_id: Some(42),
135                properties: Arc::clone(&shared),
136                kind: BucketEntryKind::FillOutline,
137            },
138            BucketFeatureEntry {
139                raw_id: None,
140                properties: single,
141                kind: BucketEntryKind::Line { length: 128.75 },
142            },
143        ];
144
145        let bytes = encode_feature_paint(&input);
146        let decoded = decode_feature_paint(&bytes).expect("valid wire payload");
147        assert_eq!(decoded.len(), 3);
148        assert!(Arc::ptr_eq(&decoded[0].properties, &decoded[1].properties));
149        assert!(!Arc::ptr_eq(&decoded[0].properties, &decoded[2].properties));
150        assert_eq!(decoded[0].raw_id, Some(42));
151        assert_eq!(decoded[0].kind, BucketEntryKind::Feature);
152        assert_eq!(decoded[1].kind, BucketEntryKind::FillOutline);
153        assert_eq!(decoded[2].raw_id, None);
154        assert_eq!(decoded[2].kind, BucketEntryKind::Line { length: 128.75 });
155        assert_eq!(decoded[0].properties, input[0].properties);
156        assert_eq!(decoded[2].properties, input[2].properties);
157    }
158}
