43impl<E: Environment, T: VectorTransferables> System for RequestSystem<E, T> {
44    fn name(&self) -> Cow<'static, str> {
45        "vector_request".into()
46    }
47
48    fn run(
49        &mut self,
50        MapContext {
51            style,
52            view_state,
53            world,
54            renderer,
55        }: &mut MapContext,
56    ) -> SystemResult {
57        tile_retry::stop_cancelled(world, self.kernel.apc());
58        let pixel_ratio = display_pixel_ratio(world, renderer, view_state);
59        let view_region = view_region_for_projection(
60            style,
61            view_state,
62            world,
63            view_state.zoom().zoom_level(DEFAULT_TILE_SIZE),
64            ViewStatePadding::Loose,
65        )?;
66
67        // Tile arrivals, eviction and a settling eye can change the covering without motion.
68        if let Some(view_region) = &view_region {
69            let max_zoom = source_max_zoom(style, TileKind::Vector);
70            let min_zoom = source_min_zoom(style, TileKind::Vector);
71            let mut requested = HashSet::new();
72            let mut budget = vector_request_budget(world, style.terrain.is_some());
73
74            let drapes = world
75                .resources
76                .get::<crate::terrain::request_system::DrapeRequests>()
77                .map(|requests| requests.0.clone())
78                .unwrap_or_default();
79            let prefetch = world
80                .resources
81                .get::<crate::terrain::request_system::DrapePrefetchRequests>()
82                .map(|requests| requests.0.clone())
83                .unwrap_or_default();
84            let overview = style
85                .terrain
86                .is_some()
87                .then_some(crate::coords::WorldTileCoords {
88                    x: 0,
89                    y: 0,
90                    z: crate::coords::ZoomLevel::new(0),
91                });
92            let draped = style.terrain.is_some();
93            let magnifies = depends_on_overscaling(style);
94            let level = u8::from(view_state.zoom().zoom_level(DEFAULT_TILE_SIZE));
95            for coords in overview
96                .into_iter()
97                .chain(drapes)
98                .chain(view_region.iter().filter(|_| !draped))
99                .chain(prefetch)
100            {
101                // Above the source maximum zoom the ancestor tile is fetched once and the
102                // view pattern scales it into every descendant in view.
103                if min_zoom.is_some_and(|min_zoom| u8::from(coords.z) < min_zoom) {
104                    continue;
105                }
106                let coords = clamp_to_max_zoom(coords, max_zoom);
107                if coords.build_quad_key().is_none()
108                    || outside_source_bounds(style, TileKind::Vector, coords)
109                    || !requested.insert(coords)
110                {
111                    continue;
112                }
113
114                // A source's last zoom is drawn magnified at the zooms past it, and what depends on
115                // that is laid out again when the magnification changes.
116                let overscaled_zoom = match max_zoom {
117                    Some(max_zoom) if magnifies && u8::from(coords.z) >= max_zoom => {
118                        level.max(u8::from(coords.z))
119                    }
120                    _ => 0,
121                };
122                let laid_out_for = (magnifies.then_some(overscaled_zoom), pixel_ratio);
123                if is_current(world, coords, laid_out_for) {
124                    continue;
125                }
126                // The rest wait for a later frame, once tiles in flight have landed.
127                if budget == 0 {
128                    continue;
129                }
130                budget -= 1;
131
132                self.request(coords, style, world, (overscaled_zoom, pixel_ratio))?;
133            }
134            tile_retry::want(world, RequestKind::Vector, &requested);
135            self.release_unwanted(world, &requested);
136        }
137        Ok(())
138    }
139}

141/// Whether the tile holds what a request would bring: laid out for the zoom it is magnified to,
142/// if that matters, and its provided images made for `pixel_ratio`, with no retry due.
143fn is_current(
144    world: &mut crate::tcs::world::World,
145    coords: crate::coords::WorldTileCoords,
146    (overscaled_zoom, pixel_ratio): (Option<u8>, f32),
147) -> bool {
148    let Some(tile) = world.tiles.query::<&VectorLayerBucketComponent>(coords) else {
149        return false;
150    };
151    overscaled_zoom.is_none_or(|zoom| tile.overscaled_zoom == zoom)
152        && !crate::sdf::provided::drawn_for_another_ratio(world, coords, pixel_ratio)
153        && !tile_retry::due(world, coords, RequestKind::Vector)
154}

156/// Device pixels per layout pixel of the display: as the host reported it, or else as the
157/// surface and viewport sizes give it.
158fn display_pixel_ratio(
159    world: &crate::tcs::world::World,
160    renderer: &crate::render::Renderer,
161    view_state: &crate::render::view_state::ViewState,
162) -> f32 {
163    world
164        .resources
165        .get::<crate::sdf::provided::DisplayPixelRatio>()
166        .map_or_else(
167            || {
168                pixel_ratio(
169                    renderer.state().surface().size().width(),
170                    view_state.width(),
171                )
172            },
173            |ratio| ratio.0,
174        )
175}
176
177/// Device pixels per layout pixel read from the surface and viewport, to the hundredth so that
178/// a ratio computed from rounded sizes does not ask for images again.
179fn pixel_ratio(physical_width: u32, logical_width: f64) -> f32 {
180    let ratio = f64::from(physical_width) / logical_width.max(1.0);
181    ((ratio * 100.0).round() / 100.0).clamp(0.25, 8.0) as f32
182}

184impl<E: Environment, T: VectorTransferables> RequestSystem<E, T> {
185    /// Stops the workers that wait for images for tiles no longer in view.
186    fn release_unwanted(
187        &self,
188        world: &mut crate::tcs::world::World,
189        wanted: &HashSet<crate::coords::WorldTileCoords>,
190    ) {
191        for attempt in crate::sdf::provided::release_unwanted(world, wanted) {
192            self.kernel.apc().cancel(attempt);
193        }
194    }
195
196    fn request(
197        &self,
198        coords: crate::coords::WorldTileCoords,
199        style: &crate::style::Style,
200        world: &mut crate::tcs::world::World,
201        (overscaled_zoom, pixel_ratio): (u8, f32),
202    ) -> SystemResult {
203        if world.tiles.spawn_mut(coords).is_none() {
204            return Err(SystemError::InvalidTile { coords });
205        }
206        let attempt = tile_retry::next_attempt(world);
207        self.kernel
208            .apc()
209            .call(
210                Input::TrackedTileRequest {
211                    coords,
212                    style: style.clone(),
213                    attempt,
214                    overscaled_zoom,
215                    pixel_ratio,
216                },
217                fetch_vector_apc::<E::OffscreenKernelEnvironment, T, _>,
218            )
219            .map_err(|source| SystemError::TileRequest {
220                kind: "vector",
221                coords,
222                source,
223            })?;
224        super::content::begin(world, coords);
225        if let Some(tile) = world
226            .tiles
227            .query_mut::<&mut VectorLayerBucketComponent>(coords)
228        {
229            tile.overscaled_zoom = overscaled_zoom;
230        }
231        tile_retry::started(world, coords, RequestKind::Vector, attempt);
232        if let Some(replaced) = crate::sdf::provided::restarted(world, coords, pixel_ratio) {
233            self.kernel.apc().cancel(replaced);
234        }
235        tracing::debug!(%coords, "vector tile request accepted");
236        Ok(())
237    }
238}

240/// Whether tiles are laid out for the zoom they are magnified to: layout and filter
241/// expressions are evaluated at that zoom, and labels along a line space themselves by it.
242pub fn depends_on_overscaling(style: &crate::style::Style) -> bool {
243    style.layers.iter().any(|layer| layer.paint.is_some())
244}
