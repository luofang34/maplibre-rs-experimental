4    pub(super) fn single_layer_limits(&self, strides: ElementStrides) -> [u64; 4] {
5        let max_v_elems = capped_elem_limit(
6            self.max_capacities.vertices,
7            self.max_page_bytes,
8            strides.vertex,
9        );
10        let max_v_elems = match self
11            .max_capacities
12            .feature_metadata
13            .checked_div(strides.feature_metadata)
14            .zip(self.max_page_bytes.checked_div(strides.feature_metadata))
15        {
16            Some((fm_cap, fm_page)) => max_v_elems.min(fm_cap).min(fm_page),
17            None => max_v_elems,
18        };
19        let max_i_elems = capped_elem_limit(
20            self.max_capacities.indices,
21            self.max_page_bytes,
22            strides.index,
23        );
24        let max_lm_elems = capped_elem_limit(
25            self.max_capacities.layer_metadata,
26            self.max_page_bytes,
27            strides.layer_metadata,
28        );
29        [
30            max_v_elems * strides.vertex,
31            max_i_elems * strides.index,
32            max_lm_elems * strides.layer_metadata,
33            max_v_elems * strides.feature_metadata,
34        ]
35    }

37    pub(super) fn validate_sizes(
38        &self,
39        sizes: [u64; 4],
40        strides: ElementStrides,
41    ) -> Result<(), AllocationError> {
42        let limits = self.single_layer_limits(strides);
43        let types = [
44            BackingBufferType::Vertices,
45            BackingBufferType::Indices,
46            BackingBufferType::Metadata,
47            BackingBufferType::FeatureMetadata,
48        ];
49        for ((typ, bytes), capacity) in types.into_iter().zip(sizes).zip(limits) {
50            if bytes > capacity {
51                return Err(AllocationError::Capacity {
52                    buffer: typ,
53                    requested: bytes,
54                    capacity,
55                });
56            }
57            if !bytes.is_multiple_of(wgpu::COPY_BUFFER_ALIGNMENT) {
58                return Err(AllocationError::Alignment { buffer: typ, bytes });
59            }
60        }
61        Ok(())
62    }

64    pub(super) fn try_grow_page_and_alloc(
65        &mut self,
66        sizes: [u64; 4],
67        strides: ElementStrides,
68    ) -> Option<PageAllocation> {
69        self.page_factory.as_ref()?;
70        // Any empty page reached here failed `try_allocate` in step 1 (too small for `sizes`), so
71        // return its capacity to the unallocated pool before shaping a new page.
72        self.reclaim_empty_pages(0);
73
74        let rem = self
75            .max_capacities
76            .saturating_sub(self.allocated_capacities());
77        let req_v_elems = elem_div(sizes[0], strides.vertex);
78        let req_i_elems = elem_div(sizes[1], strides.index);
79        let req_lm_elems = elem_div(sizes[2], strides.layer_metadata);
80
81        let rem_v_base = elem_div(rem.vertices, strides.vertex);
82        let rem_v_elems = match rem.feature_metadata.checked_div(strides.feature_metadata) {
83            Some(rem_fm) => rem_v_base.min(rem_fm),
84            None => rem_v_base,
85        };
86        let rem_i_elems = elem_div(rem.indices, strides.index);
87        let rem_lm_elems = elem_div(rem.layer_metadata, strides.layer_metadata);
88
89        if req_v_elems > rem_v_elems || req_i_elems > rem_i_elems || req_lm_elems > rem_lm_elems {
90            return None;
91        }
92
93        let max_v_base = page_elem_limit(self.max_page_bytes, strides.vertex);
94        let max_v_page = match self.max_page_bytes.checked_div(strides.feature_metadata) {
95            Some(fm_page) => max_v_base.min(fm_page),
96            None => max_v_base,
97        };
98        let max_i_page = page_elem_limit(self.max_page_bytes, strides.index);
99        let max_lm_page = page_elem_limit(self.max_page_bytes, strides.layer_metadata);
100
101        let total_v_elems = elem_div(self.max_capacities.vertices, strides.vertex).max(1);
102        let total_i_elems = elem_div(self.max_capacities.indices, strides.index);
103        let total_lm_elems = elem_div(self.max_capacities.layer_metadata, strides.layer_metadata);
104
105        let page_v_elems = DEFAULT_PAGE_VERTICES
106            .min(max_v_page)
107            .max(req_v_elems.next_power_of_two().min(max_v_page))
108            .min(rem_v_elems)
109            .max(req_v_elems);
110        let prop_i = (u128::from(total_i_elems) * u128::from(page_v_elems))
111            .div_ceil(u128::from(total_v_elems)) as u64;
112        let page_i_elems = prop_i
113            .min(max_i_page)
114            .max(req_i_elems.next_power_of_two().min(max_i_page))
115            .min(rem_i_elems)
116            .max(req_i_elems);
117        let prop_lm = (u128::from(total_lm_elems) * u128::from(page_v_elems))
118            .div_ceil(u128::from(total_v_elems)) as u64;
119        let page_lm_elems = prop_lm
120            .min(max_lm_page)
121            .max(req_lm_elems.next_power_of_two().min(max_lm_page))
122            .min(rem_lm_elems)
123            .max(req_lm_elems);
124
125        let v_bytes =
126            fitting_buffer_size(strides.vertex as usize, page_v_elems, self.max_page_bytes);
127        let i_bytes =
128            fitting_buffer_size(strides.index as usize, page_i_elems, self.max_page_bytes);
129        let lm_bytes = fitting_buffer_size(
130            strides.layer_metadata as usize,
131            page_lm_elems,
132            self.max_page_bytes,
133        );
134        let fm_bytes = fitting_buffer_size(
135            strides.feature_metadata as usize,
136            page_v_elems,
137            self.max_page_bytes,
138        );
139
140        let slot_idx = self
141            .pages
142            .iter()
143            .position(Option::is_none)
144            .unwrap_or(self.pages.len());
145        let page_id = u16::try_from(slot_idx).ok()?;
146
147        let factory = self.page_factory.as_ref()?;
148        let v_buf = (factory.0)(BackingBufferType::Vertices, v_bytes);
149        let i_buf = (factory.0)(BackingBufferType::Indices, i_bytes);
150        let lm_buf = (factory.0)(BackingBufferType::Metadata, lm_bytes);
151        let fm_buf = (factory.0)(BackingBufferType::FeatureMetadata, fm_bytes);
152
153        let mut page = BufferPage::new(
154            BackingBufferDescriptor::new(v_buf, v_bytes),
155            BackingBufferDescriptor::new(i_buf, i_bytes),
156            BackingBufferDescriptor::new(lm_buf, lm_bytes),
157            BackingBufferDescriptor::new(fm_buf, fm_bytes),
158            strides,
159        );
160        let alloc = page.try_allocate(page_id, sizes, strides)?;
161        if slot_idx == self.pages.len() {
162            self.pages.push(Some(page));
163        } else {
164            self.pages[slot_idx] = Some(page);
165        }
166        Some(alloc)
167    }

169    pub(super) fn saturation_error(
170        &self,
171        sizes: [u64; 4],
172        strides: ElementStrides,
173    ) -> AllocationError {
174        let limits = self.single_layer_limits(strides);
175        let types = [
176            BackingBufferType::Vertices,
177            BackingBufferType::Indices,
178            BackingBufferType::Metadata,
179            BackingBufferType::FeatureMetadata,
180        ];
181        for ((typ, requested), capacity) in types.into_iter().zip(sizes).zip(limits) {
182            let fits_somewhere = self.pages.iter().flatten().any(|page| match typ {
183                BackingBufferType::Vertices => page
184                    .vertex_alloc
185                    .can_allocate(requested, strides.vertex_align),
186                BackingBufferType::Indices => page
187                    .index_alloc
188                    .can_allocate(requested, strides.index_align),
189                BackingBufferType::Metadata => page
190                    .layer_alloc
191                    .can_allocate(requested, strides.layer_align),
192                BackingBufferType::FeatureMetadata => requested <= page.feature_metadata.inner_size,
193            });
194            if !fits_somewhere {
195                return AllocationError::Capacity {
196                    buffer: typ,
197                    requested,
198                    capacity,
199                };
200            }
201        }
202        AllocationError::Capacity {
203            buffer: BackingBufferType::Vertices,
204            requested: sizes[0],
205            capacity: limits[0],
206        }
207    }

210fn elem_div(bytes: u64, stride: u64) -> u64 {
211    bytes.checked_div(stride).unwrap_or(0)
212}
213
214fn page_elem_limit(max_page_bytes: u64, stride: u64) -> u64 {
215    max_page_bytes
216        .checked_div(stride)
217        .unwrap_or(0)
218        .min(u64::from(u32::MAX))
219}
220
221fn capped_elem_limit(max_pool_bytes: u64, max_page_bytes: u64, stride: u64) -> u64 {
222    elem_div(max_pool_bytes, stride).min(page_elem_limit(max_page_bytes, stride))
