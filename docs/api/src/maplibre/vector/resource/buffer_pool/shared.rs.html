1//! Reference-counted shared vertex slab allocations and 16-byte uniform feature metadata uploads.
2
3use super::{allocator::RangeAllocator, *};
4
5#[derive(Clone, Debug)]
6pub(super) struct SharedVertexAlloc {
7    pub vertices: Range<wgpu::BufferAddress>,
8    pub first_vertex: u32,
9    pub ref_count: u32,
10}
11
12impl<Q: Queue<B>, B, V: Pod, I: Pod, TM: Pod, FM: Pod> BufferPool<Q, B, V, I, TM, FM> {
13    /// Marks this pool to suballocate `feature_metadata` independently from `vertices`, allowing
14    /// uniform-paint layers to reserve a single 16-byte `VertexStepMode::Instance` record and
15    /// sibling layers on the same source layer to share a single vertex range.
16    pub fn with_decoupled_feature_metadata(mut self) -> Self {
17        self.decoupled_feature_metadata = true;
18        let strides = ElementStrides::of::<V, I, TM, FM>();
19        for page in self.pages.iter_mut().flatten() {
20            page.decoupled_feature_metadata = true;
21            if page.is_empty() {
22                let usable_vertex_bytes = page
23                    .vertices
24                    .inner_size
25                    .checked_div(strides.vertex)
26                    .unwrap_or(0)
27                    * strides.vertex;
28                page.vertex_alloc = RangeAllocator::new(usable_vertex_bytes);
29            }
30        }
31        self
32    }
33
34    /// Whether `feature_metadata` is suballocated independently rather than in vertex lockstep.
35    #[inline]
36    pub fn is_decoupled_feature_metadata(&self) -> bool {
37        self.decoupled_feature_metadata
38    }
39
40    pub(super) fn release_shared_vertex_ref(
41        &mut self,
42        coords: WorldTileCoords,
43        page: u16,
44        shared_key: Option<u64>,
45    ) -> bool {
46        let Some(key) = shared_key else {
47            return true;
48        };
49        let map_key = (coords, page, key);
50        match self.shared_vertices.get_mut(&map_key) {
51            Some(shared) if shared.ref_count > 1 => {
52                shared.ref_count -= 1;
53                false
54            }
55            Some(_) => {
56                self.shared_vertices.remove(&map_key);
57                true
58            }
59            None => true,
60        }
61    }
62
63    fn try_reuse_shared_page(
64        &mut self,
65        coords: WorldTileCoords,
66        shared_key: u64,
67        sizes_without_vertices: [u64; 4],
68        strides: ElementStrides,
69    ) -> Option<(PageAllocation, u32)> {
70        for idx in 0..self.pages.len() {
71            let page_id = idx as u16;
72            if !self
73                .shared_vertices
74                .contains_key(&(coords, page_id, shared_key))
75            {
76                continue;
77            }
78            let alloc = self.pages[idx]
79                .as_mut()
80                .and_then(|page| page.try_allocate(page_id, sizes_without_vertices, strides));
81            if let Some(mut alloc) = alloc {
82                if let Some(shared) = self.shared_vertices.get_mut(&(coords, page_id, shared_key)) {
83                    shared.ref_count += 1;
84                    alloc.vertices = shared.vertices.clone();
85                    return Some((alloc, shared.first_vertex));
86                }
87            }
88        }
89        None
90    }

92    /// Replaces `style_layer` at `coords` with uniform feature metadata, reusing an already-uploaded
93    /// shared vertex buffer when `geometry.shared_geometry_key()` matches a resident sibling on the
94    /// same page, or falling back to `fallback_vertices` when spilling onto a new page.
95    #[allow(clippy::too_many_arguments)]
96    pub fn replace_layer_geometry_uniform_shared_unless(
97        &mut self,
98        queue: &Q,
99        coords: WorldTileCoords,
100        style_layer: StyleLayer,
101        geometry: &OverAlignedVertexBuffer<V, I>,
102        fallback_vertices: Option<&[V]>,
103        layer_metadata: TM,
104        uniform_feature: FM,
105        records: usize,
106        mut protect: impl FnMut(WorldTileCoords) -> bool,
107    ) -> Result<(), AllocationError> {
108        if coords.build_quad_key().is_none() {
109            return Err(AllocationError::Coordinates { coords });
110        }
111        let strides = ElementStrides::of::<V, I, TM, FM>();
112        let source_vertices: &[V] = if !geometry.buffer.vertices.is_empty() {
113            &geometry.buffer.vertices
114        } else {
115            fallback_vertices.unwrap_or(&[])
116        };
117        let shared_key = geometry
118            .shared_geometry_key()
119            .filter(|_| self.decoupled_feature_metadata);
120        let vertices = source_vertices.len() as u64;
121        let records_u64 = records as u64;
122        if !self.decoupled_feature_metadata && records_u64 > vertices {
123            return Err(AllocationError::FeatureMetadata {
124                vertices,
125                records: records_u64,
126            });
127        }
128        let uniform_fm = self.decoupled_feature_metadata;
129        let fm_records = if uniform_fm {
130            usize::from(
131                strides.feature_metadata > 0 && (records > 0 || geometry.usable_indices > 0),
132            )
133        } else {
134            records
135        };
136        let index_bytes = geometry.buffer.indices.len() as u64 * strides.index;
137        let fm_bytes = (fm_records as u64) * strides.feature_metadata;
138
139        let reused = if let Some(key) = shared_key {
140            let reuse_sizes = [0, index_bytes, strides.layer_metadata, fm_bytes];
141            self.validate_sizes(reuse_sizes, strides)?;
142            self.try_reuse_shared_page(coords, key, reuse_sizes, strides)
143        } else {
144            None
145        };
146
147        let (alloc, first_vertex_u32, first_index_u32, uploaded_vertices) =
148            if let Some((alloc, fv)) = reused {
149                let first_index = alloc.indices.start.checked_div(strides.index).unwrap_or(0);
150                let fi = match super::upload::element_index(first_index) {
151                    Ok(fi) => fi,
152                    Err(err) => {
153                        let release_v =
154                            self.release_shared_vertex_ref(coords, alloc.page, shared_key);
155                        if let Some(page) = self
156                            .pages
157                            .get_mut(usize::from(alloc.page))
158                            .and_then(Option::as_mut)
159                        {
160                            page.free(
161                                alloc.vertices,
162                                alloc.indices,
163                                alloc.layer_metadata,
164                                alloc.feature_metadata,
165                                release_v,
166                            );
167                        }
168                        return Err(err);
169                    }
170                };
171                (alloc, fv, fi, false)
172            } else {
173                let sizes = [
174                    vertices * strides.vertex,
175                    index_bytes,
176                    strides.layer_metadata,
177                    fm_bytes,
178                ];
179                self.validate_sizes(sizes, strides)?;
180                let alloc =
181                    self.reserve_page_allocation(Some(queue), sizes, strides, &mut protect)?;
182                let first_vertex = alloc
183                    .vertices
184                    .start
185                    .checked_div(strides.vertex)
186                    .unwrap_or(0);
187                let first_index = alloc.indices.start.checked_div(strides.index).unwrap_or(0);
188                let (fv, fi) = match (
189                    super::upload::element_index(first_vertex),
190                    super::upload::element_index(first_index),
191                ) {
192                    (Ok(fv), Ok(fi)) => (fv, fi),
193                    (Err(err), _) | (_, Err(err)) => {
194                        if let Some(page) = self
195                            .pages
196                            .get_mut(usize::from(alloc.page))
197                            .and_then(Option::as_mut)
198                        {
199                            page.free(
200                                alloc.vertices,
201                                alloc.indices,
202                                alloc.layer_metadata,
203                                alloc.feature_metadata,
204                                true,
205                            );
206                        }
207                        return Err(err);
208                    }
209                };
210                if let Some(key) = shared_key {
211                    self.shared_vertices.insert(
212                        (coords, alloc.page, key),
213                        SharedVertexAlloc {
214                            vertices: alloc.vertices.clone(),
215                            first_vertex: fv,
216                            ref_count: 1,
217                        },
218                    );
219                }
220                (alloc, fv, fi, vertices * strides.vertex > 0)
221            };
222
223        let style_layer_id = style_layer.layer_id();
224        let allocation_id = self.revision.wrapping_add(1);
225        let entry = IndexEntry {
226            allocation_id,
227            paint_revision: 0,
228            page: alloc.page,
229            coords,
230            style_layer_id,
231            style_layer,
232            usable_indices: geometry.usable_indices,
233            first_vertex: first_vertex_u32,
234            first_index: first_index_u32,
235            buffer_vertices: alloc.vertices,
236            buffer_indices: alloc.indices,
237            buffer_layer_metadata: alloc.layer_metadata,
238            buffer_feature_metadata: alloc.feature_metadata,
239            uniform_feature_metadata: uniform_fm,
240            shared_geometry_key: shared_key,
241        };
242        let page = self.page_of(&entry);
243        if uploaded_vertices {
244            queue.write_buffer(
245                &page.vertices.inner,
246                entry.buffer_vertices.start,
247                bytemuck::cast_slice(source_vertices),
248            );
249        }
250        if index_bytes > 0 {
251            queue.write_buffer(
252                &page.indices.inner,
253                entry.buffer_indices.start,
254                bytemuck::cast_slice(&geometry.buffer.indices),
255            );
256        }
257        if strides.layer_metadata > 0 {
258            queue.write_buffer(
259                &page.layer_metadata.inner,
260                entry.buffer_layer_metadata.start,
261                bytemuck::bytes_of(&layer_metadata),
262            );
263        }
264        if fm_bytes > 0 {
265            Self::write_repeated_feature_metadata(
266                queue,
267                &page.feature_metadata.inner,
268                entry.buffer_feature_metadata.start,
269                uniform_feature,
270                fm_records,
271            );
272        }
273        self.index.push_back(entry);
274        self.revision = allocation_id;
275        let removed = self
276            .index
277            .remove_layer(coords, style_layer_id, Some(self.revision));
278        for old in &removed {
279            self.free_entry_ranges(old);
280        }
281        if !removed.is_empty() {
282            self.reclaim_empty_pages_with_queue(1, Some(queue));
283        }
284        Ok(())
285    }
286}
