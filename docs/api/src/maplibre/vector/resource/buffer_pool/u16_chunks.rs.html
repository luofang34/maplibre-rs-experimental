1//! Greedy `u16` sub-chunking for large (`> 65,536` vertices) index buffers when
2//! [`wgpu::DownlevelFlags::BASE_VERTEX`] is supported.
3
4use std::ops::Range;
5
6use crate::style::layer::StyleLayer;
7
8/// Maximum number of contiguous `base_vertex` chunks per layer before falling back to `u32`
9/// indices, preventing draw-call explosion on wildly non-local triangle streams.
10const MAX_U16_CHUNKS: usize = 64;
11
12/// Sub-ranges `(vertex_offset, local_index_range)` when > 65,536 vertices pack into `u16` indices.
13pub(super) type U16Chunks = Box<[(u32, Range<u32>)]>;
14
15/// Plan produced by [`plan_u16_chunks`]: `(is_u16, aligned_bytes, chunks)`.
16pub(super) type IndexPlan = (bool, u64, Option<U16Chunks>);
17
18/// Returns `true` when `style_layer` can safely emit multiple `draw_indexed` sub-ranges without
19/// colliding with `sort_runs` (`circle`) or per-feature `pattern_runs`.
20#[inline]
21pub(super) fn layer_allows_u16_chunks(style_layer: &StyleLayer) -> bool {
22    style_layer.type_ != "circle"
23        && style_layer
24            .paint
25            .as_ref()
26            .and_then(crate::vector::pattern::per_feature_pattern)
27            .is_none()
28}
29
30/// Attempts to partition `u32_slice[..usable]` triangles into contiguous `(base_vertex_offset,
31/// index_range)` chunks whose local index span fits in `u16`.
32pub(super) fn plan_u16_chunks(u32_slice: &[u32], usable: usize) -> Option<U16Chunks> {
33    if usable == 0 || usable > u32_slice.len() {
34        return None;
35    }
36    let (tris, rem) = u32_slice[..usable].as_chunks::<3>();
37    if !rem.is_empty() || tris.is_empty() {
38        return None;
39    }
40
41    let mut chunks: Vec<(u32, Range<u32>)> = Vec::with_capacity(4);
42    let mut cur_min = u32::MAX;
43    let mut cur_max = 0_u32;
44    let mut cur_start = 0_u32;
45
46    for (tri_idx, &[a, b, c]) in tris.iter().enumerate() {
47        let t_min = a.min(b).min(c);
48        let t_max = a.max(b).max(c);
49        if t_max - t_min > u32::from(u16::MAX) {
50            return None;
51        }
52        let next_min = cur_min.min(t_min);
53        let next_max = cur_max.max(t_max);
54        if next_max - next_min > u32::from(u16::MAX) {
55            if i32::try_from(cur_min).is_err() || chunks.len() + 1 >= MAX_U16_CHUNKS {
56                return None;
57            }
58            let end = (tri_idx as u32) * 3;
59            chunks.push((cur_min, cur_start..end));
60            cur_min = t_min;
61            cur_max = t_max;
62            cur_start = end;
63        } else {
64            cur_min = next_min;
65            cur_max = next_max;
66        }
67    }
68    if i32::try_from(cur_min).is_err() || chunks.len() >= MAX_U16_CHUNKS {
69        return None;
70    }
71    chunks.push((cur_min, cur_start..usable as u32));
72    Some(chunks.into_boxed_slice())
73}

75/// Writes `chunks` into `scratch`, subtracting each chunk's `v_off` so every index fits in `u16`.
76pub(super) fn write_chunked_u16_indices(
77    u32_slice: &[u32],
78    usable: usize,
79    chunks: &[(u32, Range<u32>)],
80    scratch: &mut Vec<u16>,
81) {
82    let padded = usable.next_multiple_of(2);
83    scratch.clear();
84    scratch.reserve(padded);
85    for (v_off, range) in chunks {
86        let start = range.start as usize;
87        let end = range.end as usize;
88        let off = *v_off;
89        scratch.extend(u32_slice[start..end].iter().map(|&idx| (idx - off) as u16));
90    }
91    if scratch.len() < padded {
92        scratch.push(0);
93    }
94}
