1//! Greedy `u16` sub-chunking for large (`> 65,536` vertices) index buffers when
2//! [`wgpu::DownlevelFlags::BASE_VERTEX`] is supported.
3
4use std::ops::Range;
5
6use crate::style::layer::StyleLayer;
7
8/// Maximum number of contiguous `base_vertex` chunks per layer before falling back to `u32`
9/// indices, preventing draw-call explosion on wildly non-local triangle streams.
10const MAX_U16_CHUNKS: usize = 64;
11
12/// Sub-ranges `(vertex_offset, local_index_range)` when > 65,536 vertices pack into `u16` indices.
13pub(super) type U16Chunks = Box<[(u32, Range<u32>)]>;
14
15/// Plan produced by [`plan_u16_chunks`]: `(is_u16, aligned_bytes, chunks)`.
16pub(super) type IndexPlan = (bool, u64, Option<U16Chunks>);
17
18/// Returns `true` when `style_layer` can safely emit multiple `draw_indexed` sub-ranges without
19/// colliding with `sort_runs` (`circle`), 3-pass `fill-extrusion` (`DEPTH`/`COLOR`/`CLEAR`),
20/// or per-feature `pattern_runs`.
21#[inline]
22pub(super) fn layer_allows_u16_chunks(style_layer: &StyleLayer) -> bool {
23    style_layer.type_ != "circle"
24        && style_layer.type_ != "fill-extrusion"
25        && style_layer
26            .paint
27            .as_ref()
28            .and_then(crate::vector::pattern::per_feature_pattern)
29            .is_none()
30}
31
32/// Attempts to partition `u32_slice[..usable]` triangles into contiguous `(base_vertex_offset,
33/// index_range)` chunks whose local index span fits in `u16`.
34pub(super) fn plan_u16_chunks(u32_slice: &[u32], usable: usize) -> Option<U16Chunks> {
35    if usable == 0 || usable > u32_slice.len() {
36        return None;
37    }
38    let (tris, rem) = u32_slice[..usable].as_chunks::<3>();
39    if !rem.is_empty() || tris.is_empty() {
40        return None;
41    }
42
43    let mut chunks: Vec<(u32, Range<u32>)> = Vec::with_capacity(4);
44    let mut cur_min = u32::MAX;
45    let mut cur_max = 0_u32;
46    let mut cur_start = 0_u32;
47
48    for (tri_idx, &[a, b, c]) in tris.iter().enumerate() {
49        let t_min = a.min(b).min(c);
50        let t_max = a.max(b).max(c);
51        if t_max - t_min > u32::from(u16::MAX) {
52            return None;
53        }
54        let next_min = cur_min.min(t_min);
55        let next_max = cur_max.max(t_max);
56        if next_max - next_min > u32::from(u16::MAX) {
57            if i32::try_from(cur_min).is_err() || chunks.len() + 1 >= MAX_U16_CHUNKS {
58                return None;
59            }
60            let end = (tri_idx as u32) * 3;
61            chunks.push((cur_min, cur_start..end));
62            cur_min = t_min;
63            cur_max = t_max;
64            cur_start = end;
65        } else {
66            cur_min = next_min;
67            cur_max = next_max;
68        }
69    }
70    if i32::try_from(cur_min).is_err() || chunks.len() >= MAX_U16_CHUNKS {
71        return None;
72    }
73    chunks.push((cur_min, cur_start..usable as u32));
74    Some(chunks.into_boxed_slice())
75}

77/// Writes `chunks` into `scratch`, subtracting each chunk's `v_off` so every index fits in `u16`.
78pub(super) fn write_chunked_u16_indices(
79    u32_slice: &[u32],
80    usable: usize,
81    chunks: &[(u32, Range<u32>)],
82    scratch: &mut Vec<u16>,
83) {
84    let padded = usable.next_multiple_of(2);
85    scratch.clear();
86    scratch.reserve(padded);
87    for (v_off, range) in chunks {
88        let start = range.start as usize;
89        let end = range.end as usize;
90        let off = *v_off;
91        scratch.extend(u32_slice[start..end].iter().map(|&idx| (idx - off) as u16));
92    }
93    if scratch.len() < padded {
94        scratch.push(0);
95    }
96}
