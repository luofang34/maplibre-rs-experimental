6    /// Creates an on-demand paged GPU pool with [`BufferPoolSizes::default`] maximum capacity.
7    pub fn from_device(device: &wgpu::Device) -> Self {
8        Self::from_device_with_sizes(device, BufferPoolSizes::default())

11    /// Creates an on-demand paged GPU pool whose total element capacities across slabs are bounded
12    /// by `sizes`, with each buffer clipped to the device's `max_buffer_size`.
13    ///
14    /// Zero GPU buffers are allocated until the first non-empty geometry upload arrives, and
15    /// unused buffer classes (`EmptyLayerMetadata`, standalone `feature_metadata`, `u32` fallback
16    /// indices) remain at `0` bytes until genuinely required.
17    pub fn from_device_with_sizes(device: &wgpu::Device, sizes: BufferPoolSizes) -> Self {
18        let max_page_bytes = device.limits().max_buffer_size;
19        let device = device.clone();
20        let mut pool = Self::with_page_factory(sizes, max_page_bytes, move |typ, logical_bytes| {
21            let physical = logical_bytes.max(16).next_multiple_of(16);
22            let (label, usage) = match typ {
23                BackingBufferType::Vertices => (
24                    "vertex slab",
25                    wgpu::BufferUsages::VERTEX | wgpu::BufferUsages::COPY_DST,
26                ),
27                BackingBufferType::Indices => (
28                    "index slab",
29                    wgpu::BufferUsages::INDEX | wgpu::BufferUsages::COPY_DST,
30                ),
31                BackingBufferType::Metadata => (
32                    "layer metadata slab",
33                    wgpu::BufferUsages::VERTEX | wgpu::BufferUsages::COPY_DST,
34                ),
35                BackingBufferType::FeatureMetadata => (
36                    "feature metadata slab",
37                    wgpu::BufferUsages::VERTEX | wgpu::BufferUsages::COPY_DST,
38                ),
39            };
40            device.create_buffer(&wgpu::BufferDescriptor {
41                label: Some(label),
42                size: physical,
43                usage,
44                mapped_at_creation: false,
45            })
46        });
47        pool.slab_destroyer = Some(page::SlabDestroyer(|buf| buf.destroy()));
48        pool.enable_u16_indices = size_of::<I>() == size_of::<u32>();
49        pool.enable_u16_chunks = false;
50        pool.zero_initialized_slabs = true;
51        pool
52    }

54    /// Enables or disables `u16` base-vertex index chunks (when `DownlevelFlags::BASE_VERTEX`
55    /// and `u16` indices are both supported).
56    pub fn with_u16_chunks(mut self, enabled: bool) -> Self {
57        self.enable_u16_chunks = enabled && self.enable_u16_indices;
58        self

61    /// Total GPU buffer bytes currently physically allocated on the device across all live slabs.
62    pub fn device_buffer_bytes(&self) -> wgpu::BufferAddress {
63        let v_bytes: u64 = self
64            .vertex_slabs
65            .iter()
66            .flatten()
67            .map(|slab| {
68                slab.vertices.inner.size() + slab.paired_fm.as_ref().map_or(0, |fm| fm.inner.size())
69            })
70            .sum();
71        let other_bytes: u64 = self
72            .index_u16_slabs
73            .iter()
74            .flatten()
75            .chain(self.index_native_slabs.iter().flatten())
76            .chain(self.layer_slabs.iter().flatten())
77            .chain(self.feature_slabs.iter().flatten())
78            .map(|slab| slab.buffer.inner.size())
79            .sum();
80        v_bytes + other_bytes
81    }

83    /// Binds a layer's indices (`u16` or `u32`), vertices, and feature metadata (at `feature_slot`)
84    /// from their decoupled slabs and returns `(first_index, base_vertex)`.
85    pub fn bind_layer<'w>(
86        &self,
87        pass: &mut crate::render::tracked_pass::TrackedRenderPass<'w>,
88        entry: &IndexEntry,
89        (feature_slot, whole): (u32, bool),
90    ) -> (u32, i32) {
91        let format = if entry.index_u16 {
92            wgpu::IndexFormat::Uint16
93        } else {
94            crate::render::INDEX_FORMAT
95        };
96        let idx_buf = self.indices_for(entry);
97        let v_slab = self.vertex_slab_of(entry.vertex_page);
98
99        if entry.is_uniform_feature_metadata() {
100            let fm_range = entry.feature_metadata_buffer_range();
101            if !fm_range.is_empty() {
102                pass.set_vertex_buffer(
103                    feature_slot,
104                    self.feature_metadata_for(entry).slice(fm_range),
105                );
106            }
107            if whole {
108                pass.set_index_buffer(idx_buf.slice(..), format);
109                pass.set_vertex_buffer(0, v_slab.vertices.inner.slice(..));
110                (entry.whole_buffer_indices().start, entry.base_vertex())
111            } else {
112                pass.set_index_buffer(idx_buf.slice(entry.indices_buffer_range()), format);
113                pass.set_vertex_buffer(
114                    0,
115                    v_slab.vertices.inner.slice(entry.vertices_buffer_range()),
116                );
117                (0, 0)
118            }
119        } else if whole && entry.has_paired_fm() && !entry.is_dual_feature_metadata() {
120            pass.set_index_buffer(idx_buf.slice(..), format);
121            pass.set_vertex_buffer(0, v_slab.vertices.inner.slice(..));
122            pass.set_vertex_buffer(feature_slot, self.feature_metadata_for(entry).slice(..));
123            (entry.whole_buffer_indices().start, entry.base_vertex())
124        } else if whole {
125            pass.set_index_buffer(idx_buf.slice(..), format);
126            pass.set_vertex_buffer(
127                0,
128                v_slab.vertices.inner.slice(entry.vertices_buffer_range()),
129            );
130            pass.set_vertex_buffer(
131                feature_slot,
132                self.feature_metadata_for(entry)
133                    .slice(entry.buffer_feature_metadata.start..),
134            );
135            (entry.whole_buffer_indices().start, 0)
136        } else {
137            pass.set_index_buffer(idx_buf.slice(entry.indices_buffer_range()), format);
138            pass.set_vertex_buffer(
139                0,
140                v_slab.vertices.inner.slice(entry.vertices_buffer_range()),
141            );
142            pass.set_vertex_buffer(
143                feature_slot,
144                self.feature_metadata_for(entry)
145                    .slice(entry.buffer_feature_metadata.start..),
146            );
147            (0, 0)
148        }
149    }
