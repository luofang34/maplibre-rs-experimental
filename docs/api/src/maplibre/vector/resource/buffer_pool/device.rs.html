11    /// Creates an on-demand paged GPU pool whose total element capacities across pages are bounded
12    /// by `sizes`, with each page clipped to the device's `max_buffer_size`.
13    ///
14    /// Zero GPU buffers are allocated until the first non-empty geometry upload arrives, avoiding
15    /// upfront hundreds-of-megabytes allocations and keeping [`BufferPoolSizes::EMPTY`] at `0` B.
16    pub fn from_device_with_sizes(device: &wgpu::Device, sizes: BufferPoolSizes) -> Self {
17        let max_page_bytes = device.limits().max_buffer_size;
18        let device = device.clone();
19        let mut pool = Self::with_page_factory(sizes, max_page_bytes, move |typ, logical_bytes| {
20            let physical = logical_bytes.max(16).next_multiple_of(16);
21            let (label, usage) = match typ {
22                BackingBufferType::Vertices => (
23                    "vertex buffer page",
24                    wgpu::BufferUsages::VERTEX | wgpu::BufferUsages::COPY_DST,
25                ),
26                BackingBufferType::Indices => (
27                    "indices buffer page",
28                    wgpu::BufferUsages::INDEX | wgpu::BufferUsages::COPY_DST,
29                ),
30                BackingBufferType::Metadata => (
31                    "layer metadata buffer page",
32                    wgpu::BufferUsages::VERTEX | wgpu::BufferUsages::COPY_DST,
33                ),
34                BackingBufferType::FeatureMetadata => (
35                    "feature metadata buffer page",
36                    wgpu::BufferUsages::VERTEX | wgpu::BufferUsages::COPY_DST,
37                ),
38            };
39            device.create_buffer(&wgpu::BufferDescriptor {
40                label: Some(label),
41                size: physical,
42                usage,
43                mapped_at_creation: false,
44            })
45        });
46        pool.page_destroyer = Some(page::PageDestroyer(|page| {
47            page.vertices.inner.destroy();
48            page.indices.inner.destroy();
49            page.layer_metadata.inner.destroy();
50            page.feature_metadata.inner.destroy();
51        }));
52        pool
53    }

55    /// Total GPU buffer bytes currently physically allocated on the device across all live pages.
56    pub fn device_buffer_bytes(&self) -> wgpu::BufferAddress {
57        self.pages
58            .iter()
59            .flatten()
60            .map(|page| {
61                page.vertices.inner.size()
62                    + page.indices.inner.size()
63                    + page.layer_metadata.inner.size()
64                    + page.feature_metadata.inner.size()
65            })
66            .sum()
67    }

69    /// Binds a layer's indices, vertices and feature metadata (at `feature_slot`) from its page
70    /// and returns the first index and base vertex to draw its indices with. With `whole` the
71    /// page's buffers are bound whole, so consecutive layers on the same page keep them bound;
72    /// without, as on WebGL2, which has no base vertex, the layer's own ranges are bound and
73    /// drawn from zero.
74    pub fn bind_layer<'w>(
75        &self,
76        pass: &mut crate::render::tracked_pass::TrackedRenderPass<'w>,
77        entry: &IndexEntry,
78        (feature_slot, whole): (u32, bool),
79    ) -> (u32, i32) {
80        let format = crate::render::INDEX_FORMAT;
81        let page = self.page_of(entry);
82        if whole {
83            pass.set_index_buffer(page.indices.inner.slice(..), format);
84            pass.set_vertex_buffer(0, page.vertices.inner.slice(..));
85            pass.set_vertex_buffer(feature_slot, page.feature_metadata.inner.slice(..));
86            (entry.whole_buffer_indices().start, entry.base_vertex())
87        } else {
88            pass.set_index_buffer(
89                page.indices.inner.slice(entry.indices_buffer_range()),
90                format,
91            );
92            pass.set_vertex_buffer(0, page.vertices.inner.slice(entry.vertices_buffer_range()));
93            pass.set_vertex_buffer(
94                feature_slot,
95                page.feature_metadata
96                    .inner
97                    .slice(entry.feature_metadata_buffer_range()),
98            );
99            (0, 0)
100        }
101    }
