9    /// A pool whose buffers hold `sizes` elements, each clipped to the device's largest buffer.
10    /// Too small a pool has tiles unloaded and reloaded from frame to frame.
11    pub fn from_device_with_sizes(device: &wgpu::Device, sizes: BufferPoolSizes) -> Self {
12        let largest = device.limits().max_buffer_size;
13        let fitting = |element: usize, count: wgpu::BufferAddress| {
14            fitting_buffer_size(element, count, largest)
15        };
16        let vertex_buffer_desc = wgpu::BufferDescriptor {
17            label: Some("vertex buffer"),
18            size: fitting(size_of::<V>(), sizes.vertices),
19            usage: wgpu::BufferUsages::VERTEX | wgpu::BufferUsages::COPY_DST,
20            mapped_at_creation: false,
21        };
22
23        let indices_buffer_desc = wgpu::BufferDescriptor {
24            label: Some("indices buffer"),
25            size: fitting(size_of::<I>(), sizes.indices),
26            usage: wgpu::BufferUsages::INDEX | wgpu::BufferUsages::COPY_DST,
27            mapped_at_creation: false,
28        };
29
30        let feature_metadata_desc = wgpu::BufferDescriptor {
31            label: Some("feature metadata buffer"),
32            size: fitting(size_of::<FM>(), sizes.feature_metadata),
33            usage: wgpu::BufferUsages::VERTEX | wgpu::BufferUsages::COPY_DST,
34            mapped_at_creation: false,
35        };
36
37        let layer_metadata_desc = wgpu::BufferDescriptor {
38            label: Some("layer metadata buffer"),
39            size: fitting(size_of::<TM>(), sizes.layer_metadata),
40            usage: wgpu::BufferUsages::VERTEX | wgpu::BufferUsages::COPY_DST,
41            mapped_at_creation: false,
42        };
43
44        BufferPool::new(
45            BackingBufferDescriptor::new(
46                device.create_buffer(&vertex_buffer_desc),
47                vertex_buffer_desc.size,
48            ),
49            BackingBufferDescriptor::new(
50                device.create_buffer(&indices_buffer_desc),
51                indices_buffer_desc.size,
52            ),
53            BackingBufferDescriptor::new(
54                device.create_buffer(&layer_metadata_desc),
55                layer_metadata_desc.size,
56            ),
57            BackingBufferDescriptor::new(
58                device.create_buffer(&feature_metadata_desc),
59                feature_metadata_desc.size,
60            ),
61        )
62    }
63}

68    /// Binds a layer's indices, vertices and feature metadata (at `feature_slot`) and returns
69    /// the first index and base vertex to draw its indices with. With `whole` the pool's
70    /// buffers are bound whole, so consecutive layers keep them bound; without, as on WebGL2,
71    /// which has no base vertex, the layer's own ranges are bound and drawn from zero.
72    pub fn bind_layer<'w>(
73        &self,
74        pass: &mut crate::render::tracked_pass::TrackedRenderPass<'w>,
75        entry: &IndexEntry,
76        (feature_slot, whole): (u32, bool),
77    ) -> (u32, i32) {
78        let format = crate::render::INDEX_FORMAT;
79        if whole {
80            pass.set_index_buffer(self.indices.inner.slice(..), format);
81            pass.set_vertex_buffer(0, self.vertices.inner.slice(..));
82            pass.set_vertex_buffer(feature_slot, self.feature_metadata.inner.slice(..));
83            (entry.whole_buffer_indices().start, entry.base_vertex())
84        } else {
85            pass.set_index_buffer(
86                self.indices.inner.slice(entry.indices_buffer_range()),
87                format,
88            );
89            pass.set_vertex_buffer(0, self.vertices.inner.slice(entry.vertices_buffer_range()));
90            pass.set_vertex_buffer(
91                feature_slot,
92                self.feature_metadata
93                    .inner
94                    .slice(entry.feature_metadata_buffer_range()),
95            );
96            (0, 0)
97        }
98    }
