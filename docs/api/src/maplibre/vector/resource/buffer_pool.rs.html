25/// The pattern key and index range of each run of features that share one image.
26pub type PatternRuns = Vec<(u32, Range<u32>)>;

28/// The sort key and index range of each run of features that share a key.
29pub type SortRuns = Vec<(f32, Range<u32>)>;

31/// This is inspired by the memory pool in Vulkan documented
32/// [here](https://gpuopen-librariesandsdks.github.io/VulkanMemoryAllocator/html/custom_memory_pools.html).
33#[derive(Debug)]
34pub struct BufferPool<Q, B, V, I, TM, FM> {
35    vertices: BackingBuffer<B>,
36    indices: BackingBuffer<B>,
37    layer_metadata: BackingBuffer<B>,
38    feature_metadata: BackingBuffer<B>,
39
40    index: RingIndex,
41    /// For layers whose pattern varies by feature: the pattern key and index range of each run
42    /// of features sharing one image, relative to the layer's first index.
43    pattern_runs: std::collections::HashMap<(WorldTileCoords, String), PatternRuns>,
44    /// For layers whose features draw in the order of a sort key across tiles: the key and index
45    /// range of each run of features sharing one, relative to the layer's first index.
46    sort_runs: std::collections::HashMap<(WorldTileCoords, String), SortRuns>,
47    /// Advances whenever geometry is allocated or evicted, so cached renders can refresh.
48    revision: u64,
49    phantom_v: PhantomData<V>,
50    phantom_i: PhantomData<I>,
51    phantom_q: PhantomData<Q>,
52    phantom_m: PhantomData<TM>,
53    phantom_fm: PhantomData<FM>,
54}

64#[derive(Debug)]
65struct BackingBuffer<B> {
66    /// The internal structure which is used for storage
67    inner: B,
68    /// The size of the `inner` buffer
69    inner_size: wgpu::BufferAddress,
70    typ: BackingBufferType,
71}

73impl<B> BackingBuffer<B> {
74    fn new(inner: B, inner_size: wgpu::BufferAddress, typ: BackingBufferType) -> Self {
75        Self {
76            inner,
77            inner_size,
78            typ,
79        }
80    }
81}

83/// Bytes for `count` elements of `element` bytes, cut down to whole elements of the largest
84/// buffer the device creates; the pool is sized for desktop GPUs and a simulator allows less.
85fn fitting_buffer_size(
86    element: usize,
87    count: wgpu::BufferAddress,
88    largest: wgpu::BufferAddress,
89) -> wgpu::BufferAddress {
90    let element = element as wgpu::BufferAddress;
91    (element * count).min(largest) / element * element
92}

134    pub fn clear(&mut self) {
135        self.index.clear();
136        self.pattern_runs.clear();
137        self.sort_runs.clear();
138        self.revision = self.revision.wrapping_add(1);
139    }

141    /// Releases GPU geometry when the tile data and its glyph atlas leave the CPU cache.
142    pub fn remove_tile(&mut self, coords: WorldTileCoords) {
143        self.pattern_runs.retain(|(tile, _), _| *tile != coords);
144        self.sort_runs.retain(|(tile, _), _| *tile != coords);
145        if self.index.remove_tile(coords) {
146            self.revision = self.revision.wrapping_add(1);
147        }
148    }
149
150    /// Removes one style layer without evicting other sources at the same coordinates.
151    pub fn remove_layer(&mut self, coords: WorldTileCoords, id: &str) {
152        self.pattern_runs.remove(&(coords, id.to_owned()));
153        self.sort_runs.remove(&(coords, id.to_owned()));
154        if self.index.remove_layer(coords, id, None) {
155            self.revision = self.revision.wrapping_add(1);
156        }
157    }

197        let gap = self.index.find_largest_gap(

199            match typ {

227    /// The VertexBuffers can contain padding elements. Not everything from a VertexBuffers is usable.
228    /// The function returns the `bytes` and `aligned_bytes`. See [`OverAlignedVertexBuffer`].
229    fn align(
230        stride: wgpu::BufferAddress,
231        elements: wgpu::BufferAddress,
232        usable_elements: wgpu::BufferAddress,
233    ) -> (wgpu::BufferAddress, wgpu::BufferAddress) {
234        let bytes = elements * stride;
235
236        let usable_bytes = (usable_elements * stride) as wgpu::BufferAddress;
237
238        let align = wgpu::COPY_BUFFER_ALIGNMENT;
239        let padding = (align - usable_bytes % align) % align;
240
241        let aligned_bytes = usable_bytes + padding;
242
243        (bytes, aligned_bytes)
244    }

260    #[tracing::instrument(skip_all)]
261    pub fn update_layer_metadata(&self, queue: &Q, entry: &IndexEntry, layer_metadata: TM) {
262        let layer_metadata_stride = size_of::<TM>() as wgpu::BufferAddress; // TODO: deduplicate
263        let (layer_metadata_bytes, aligned_layer_metadata_bytes) =
264            Self::align(layer_metadata_stride, 1, 1);
265
266        let allocated = entry.buffer_layer_metadata.end - entry.buffer_layer_metadata.start;
267        if allocated != layer_metadata_bytes {
268            tracing::error!(
269                coords = %entry.coords,
270                layer = %entry.style_layer.id,
271                allocated,
272                offered = layer_metadata_bytes,
273                "layer metadata update skipped: size differs from the allocation"
274            );
275            return;
276        }
277
278        queue.write_buffer(
279            &self.layer_metadata.inner,
280            entry.buffer_layer_metadata.start,
281            &bytemuck::cast_slice(&[layer_metadata])[0..aligned_layer_metadata_bytes as usize],
282        );
283    }
284
285    #[tracing::instrument(skip_all)]
286    pub fn update_feature_metadata(&self, queue: &Q, entry: &IndexEntry, feature_metadata: &[FM]) {
287        let feature_metadata_stride = size_of::<FM>() as wgpu::BufferAddress; // TODO: deduplicate
288
289        let (feature_metadata_bytes, aligned_feature_metadata_bytes) = Self::align(
290            feature_metadata_stride,
291            feature_metadata.len() as wgpu::BufferAddress,
292            feature_metadata.len() as wgpu::BufferAddress,
293        );
294
295        let allocated = entry.buffer_feature_metadata.end - entry.buffer_feature_metadata.start;
296        if allocated != feature_metadata_bytes
297            || feature_metadata_bytes != aligned_feature_metadata_bytes
298        {
299            // Writing past the allocation would corrupt a neighbouring layer's metadata.
300            tracing::error!(
301                coords = %entry.coords,
302                layer = %entry.style_layer.id,
303                allocated,
304                offered = feature_metadata_bytes,
305                aligned = aligned_feature_metadata_bytes,
306                "feature metadata update skipped: size differs from the allocation"
307            );
308            return;
309        }
310
311        queue.write_buffer(
312            &self.feature_metadata.inner,
313            entry.buffer_feature_metadata.start,
314            &bytemuck::cast_slice(feature_metadata)[0..aligned_feature_metadata_bytes as usize],
315        );
316    }

318    /// Rewrites a layer's indices in place, as when its labels are drawn in a new order; the
319    /// count must be the one it was uploaded with.
320    pub fn update_indices(&self, queue: &Q, entry: &IndexEntry, indices: &[I]) {
321        let bytes = std::mem::size_of_val(indices) as wgpu::BufferAddress;
322        let allocated = entry.buffer_indices.end - entry.buffer_indices.start;
323        if bytes > allocated || !bytes.is_multiple_of(wgpu::COPY_BUFFER_ALIGNMENT) {
324            tracing::error!(
325                coords = %entry.coords,
326                layer = %entry.style_layer.id,
327                allocated,
328                offered = bytes,
329                "index update skipped: it does not fit the allocation"
330            );
331            return;
332        }
333        queue.write_buffer(
334            &self.indices.inner,
335            entry.buffer_indices.start,
336            bytemuck::cast_slice(indices),
337        );
338    }
