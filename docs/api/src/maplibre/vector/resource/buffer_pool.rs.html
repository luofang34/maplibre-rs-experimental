1//! Paged, coalescing GPU geometry buffer pool.
2//!
3//! Geometry layers suballocate vertices, indices, per-layer metadata, and per-vertex feature
4//! metadata from fixed-size GPU slab pages allocated on demand up to [`BufferPoolSizes`]. Freed
5//! ranges coalesce immediately in `O(log F)` within their page, and completely empty burst pages
6//! are returned to the GPU automatically.

33/// The pattern key and index range of each run of features that share one image.
34pub type PatternRuns = Vec<(u32, Range<u32>)>;

36/// The sort key and index range of each run of features that share a key.

39/// Paged, address-coalescing pool of GPU geometry buffers inspired by Vulkan Memory Allocator
40/// custom pools (`VmaPool`).
41#[derive(Debug)]
42pub struct BufferPool<Q, B, V, I, TM, FM> {
43    pages: Vec<Option<BufferPage<B>>>,
44    max_capacities: ByteCapacities,
45    max_page_bytes: wgpu::BufferAddress,
46    page_factory: Option<PageFactory<B>>,
47
48    index: RingIndex,
49    /// For layers whose pattern varies by feature: the pattern key and index range of each run
50    /// of features sharing one image, relative to the layer's first index.
51    pattern_runs: std::collections::HashMap<(WorldTileCoords, String), PatternRuns>,
52    /// For layers whose features draw in the order of a sort key across tiles: the key and index
53    /// range of each run of features sharing one, relative to the layer's first index.
54    sort_runs: std::collections::HashMap<(WorldTileCoords, String), SortRuns>,
55    /// Advances whenever geometry is allocated or evicted, so cached renders can refresh.
56    revision: u64,
57    phantom_v: PhantomData<V>,
58    phantom_i: PhantomData<I>,
59    phantom_q: PhantomData<Q>,
60    phantom_m: PhantomData<TM>,
61    phantom_fm: PhantomData<FM>,
62}

64/// Identifies one of the four typed sub-buffers in a [`BufferPool`].
65#[derive(Clone, Copy, Debug, PartialEq, Eq)]
66pub enum BackingBufferType {
67    /// Per-vertex positions/normals/attributes.
68    Vertices,
69    /// Indexed triangle list indices.
70    Indices,
71    /// Per-layer paint/index metadata.
72    Metadata,
73    /// Per-vertex feature paint metadata (indexed in lockstep with [`Self::Vertices`]).
74    FeatureMetadata,
75}

77#[derive(Debug)]
78struct BackingBuffer<B> {
79    /// The internal structure which is used for storage
80    inner: B,
81    /// The logical usable byte size of `inner`
82    inner_size: wgpu::BufferAddress,
83}

85impl<B> BackingBuffer<B> {
86    fn new(inner: B, inner_size: wgpu::BufferAddress) -> Self {
87        Self { inner, inner_size }
88    }

91/// Bytes for `count` elements of `element` bytes, cut down to whole elements of the largest
92/// buffer the device creates; the pool is sized for desktop GPUs and a simulator allows less.
93fn fitting_buffer_size(
94    element: usize,
95    count: wgpu::BufferAddress,
96    largest: wgpu::BufferAddress,
97) -> wgpu::BufferAddress {
98    let element = element as wgpu::BufferAddress;
99    if element == 0 {
100        return 0;
101    }
102    element.saturating_mul(count).min(largest) / element * element
103}

105impl<Q: Queue<B>, B, V: Pod, I: Pod, TM: Pod, FM: Pod> BufferPool<Q, B, V, I, TM, FM> {
106    /// Creates a fixed single-page pool backed by pre-created buffer descriptors.
107    pub fn new(
108        vertices: BackingBufferDescriptor<B>,
109        indices: BackingBufferDescriptor<B>,
110        layer_metadata: BackingBufferDescriptor<B>,
111        feature_metadata: BackingBufferDescriptor<B>,
112    ) -> Self {
113        let strides = ElementStrides::of::<V, I, TM, FM>();
114        let page = BufferPage::new(vertices, indices, layer_metadata, feature_metadata, strides);
115        let max_capacities = page.capacities();
116        Self {
117            pages: vec![Some(page)],
118            max_capacities,
119            max_page_bytes: u64::MAX,
120            page_factory: None,
121            index: RingIndex::new(),
122            pattern_runs: Default::default(),
123            sort_runs: Default::default(),
124            revision: 0,
125            phantom_v: Default::default(),
126            phantom_i: Default::default(),
127            phantom_q: Default::default(),
128            phantom_m: Default::default(),
129            phantom_fm: Default::default(),
130        }
131    }
132
133    /// Creates an on-demand paged pool that allocates pages lazily through `factory` up to
134    /// `sizes` total elements, with each individual buffer capped at `max_page_bytes`.
135    pub fn with_page_factory(
136        sizes: BufferPoolSizes,
137        max_page_bytes: wgpu::BufferAddress,
138        factory: impl Fn(BackingBufferType, wgpu::BufferAddress) -> B + 'static,
139    ) -> Self {
140        let max_capacities = ByteCapacities {
141            vertices: sizes.vertices.saturating_mul(size_of::<V>() as u64),
142            indices: sizes.indices.saturating_mul(size_of::<I>() as u64),
143            layer_metadata: sizes.layer_metadata.saturating_mul(size_of::<TM>() as u64),
144            feature_metadata: sizes
145                .feature_metadata
146                .saturating_mul(size_of::<FM>() as u64),
147        };
148        Self {
149            pages: Vec::new(),
150            max_capacities,
151            max_page_bytes,
152            page_factory: (!max_capacities.is_empty()).then(|| PageFactory(Box::new(factory))),
153            index: RingIndex::new(),
154            pattern_runs: Default::default(),
155            sort_runs: Default::default(),
156            revision: 0,
157            phantom_v: Default::default(),
158            phantom_i: Default::default(),
159            phantom_q: Default::default(),
160            phantom_m: Default::default(),
161            phantom_fm: Default::default(),
162        }
163    }

165    /// Clears all resident geometry allocations and reclaims excess empty pages.
166    pub fn clear(&mut self) {
167        self.index.clear();
168        self.pattern_runs.clear();
169        self.sort_runs.clear();
170        for page in self.pages.iter_mut().flatten() {
171            page.clear();
172        }
173        self.reclaim_empty_pages(1);
174        self.revision = self.revision.wrapping_add(1);
175    }
176
177    /// Drops empty pages beyond `keep_spare` when the pool owns a page factory.
178    ///
179    /// Passing `1` retains one warm page to avoid allocation churn across pan/reload boundaries;
180    /// passing `0` under memory pressure releases all unused GPU pages immediately.
181    pub fn reclaim_empty_pages(&mut self, keep_spare: usize) {
182        if self.page_factory.is_none() {
183            return;
184        }
185        let mut spare_kept = 0usize;
186        for slot in &mut self.pages {
187            let is_empty = slot.as_ref().is_some_and(BufferPage::is_empty);
188            if is_empty {
189                if spare_kept < keep_spare {
190                    spare_kept += 1;
191                } else {
192                    *slot = None;
193                }
194            }
195        }
196        while matches!(self.pages.last(), Some(None)) {
197            self.pages.pop();
198        }
199    }
200
201    /// Number of currently allocated backing pages.
202    pub fn page_count(&self) -> usize {
203        self.pages.iter().filter(|slot| slot.is_some()).count()
204    }

206    /// Whether the pool holds non-zero vertex and index backing capacity.
207    pub fn has_capacity(&self) -> bool {
208        self.max_capacities.vertices > 0 && self.max_capacities.indices > 0
209    }

211    /// Releases GPU geometry when the tile data and its glyph atlas leave the CPU cache.
212    pub fn remove_tile(&mut self, coords: WorldTileCoords) {
213        self.pattern_runs.retain(|(tile, _), _| *tile != coords);
214        self.sort_runs.retain(|(tile, _), _| *tile != coords);
215        let removed = self.index.remove_tile(coords);
216        if !removed.is_empty() {
217            for entry in &removed {
218                self.free_entry_ranges(entry);
219            }
220            self.reclaim_empty_pages(1);
221            self.revision = self.revision.wrapping_add(1);
222        }
223    }
224
225    /// Removes one style layer without evicting other sources at the same coordinates.
226    pub fn remove_layer(&mut self, coords: WorldTileCoords, id: &str) {
227        self.pattern_runs.remove(&(coords, id.to_owned()));
228        self.sort_runs.remove(&(coords, id.to_owned()));
229        let removed = self.index.remove_layer(coords, id, None);
230        if !removed.is_empty() {
231            for entry in &removed {
232                self.free_entry_ranges(entry);
233            }
234            self.reclaim_empty_pages(1);
235            self.revision = self.revision.wrapping_add(1);
236        }
237    }

239    /// Records, or with `None` forgets, the pattern runs of a layer's geometry.
240    pub fn set_pattern_runs(
241        &mut self,
242        coords: WorldTileCoords,
243        id: &str,
244        runs: Option<PatternRuns>,
245    ) {
246        match runs {
247            Some(runs) => self.pattern_runs.insert((coords, id.to_owned()), runs),
248            None => self.pattern_runs.remove(&(coords, id.to_owned())),

250    }

255            Some(runs) => self.sort_runs.insert((coords, id.to_owned()), runs),

275    #[cfg(test)]
276    fn available_space(&self, typ: BackingBufferType) -> wgpu::BufferAddress {
277        let in_pages = self
278            .pages
279            .iter()
280            .flatten()
281            .map(|page| match typ {
282                BackingBufferType::Vertices => page.vertex_alloc.largest_free_block(),
283                BackingBufferType::Indices => page.index_alloc.largest_free_block(),
284                BackingBufferType::Metadata => page.layer_alloc.largest_free_block(),
285                BackingBufferType::FeatureMetadata => {
286                    let v_stride = size_of::<V>() as u64;
287                    let fm_stride = size_of::<FM>() as u64;
288                    page.vertex_alloc
289                        .largest_free_block()
290                        .checked_div(v_stride)
291                        .unwrap_or(0)
292                        * fm_stride
293                }
294            })
295            .max()
296            .unwrap_or(0);
297        if self.page_factory.is_some() {
298            let unallocated = self
299                .max_capacities
300                .saturating_sub(self.allocated_capacities())
301                .get(typ);
302            in_pages.max(unallocated.min(self.max_page_bytes))
303        } else {
304            in_pages
305        }
306    }
