1//! Paged, coalescing GPU geometry buffer pool with decoupled buffer-class slabs.
2//!
3//! Geometry layers suballocate compact GPU vertices (`V::Gpu`), `u16`/`u32` indices, shared
4//! per-layer metadata, and optional per-vertex or standalone feature metadata from independent
5//! on-demand GPU slabs up to [`BufferPoolSizes`]. Freed ranges coalesce immediately in `O(log F)`
6//! within their slab, and completely empty burst slabs return to the GPU automatically.

79/// Identifies one of the four typed sub-buffers in a [`BufferPool`].

84    /// Indexed triangle list indices.

104/// Bytes for `count` elements of `element` bytes, cut down to whole elements of the largest
105/// buffer the device creates; the pool is sized for desktop GPUs and a simulator allows less.
106fn fitting_buffer_size(
107    element: usize,
108    count: wgpu::BufferAddress,
109    largest: wgpu::BufferAddress,
110) -> wgpu::BufferAddress {
111    let element = element as wgpu::BufferAddress;
112    if element == 0 {
113        return 0;
114    }
115    element.saturating_mul(count).min(largest) / element * element
116}

118impl<Q: Queue<B>, B, V: PoolVertex, I: Pod, TM: Pod, FM: Pod> BufferPool<Q, B, V, I, TM, FM> {
119    /// Creates a fixed single-slab pool backed by pre-created buffer descriptors.
120    pub fn new(
121        vertices: BackingBufferDescriptor<B>,
122        indices: BackingBufferDescriptor<B>,
123        layer_metadata: BackingBufferDescriptor<B>,
124        feature_metadata: BackingBufferDescriptor<B>,
125    ) -> Self {
126        let strides = ElementStrides::of::<V::Gpu, I, TM, FM>();
127        let fm_elems = feature_metadata
128            .inner_size
129            .checked_div(strides.feature_metadata)
130            .unwrap_or(0);
131        let (paired_fm, feature_slabs) = if strides.feature_metadata > 0 {
132            (Some(feature_metadata), Vec::new())
133        } else {
134            (
135                None,
136                vec![Some(BufferSlab::new(
137                    feature_metadata,
138                    strides.feature_metadata,
139                ))],
140            )
141        };
142        let v_slab = VertexSlab::new(vertices, paired_fm, strides);
143        let i_slab = BufferSlab::new(indices, strides.index);
144        let lm_slab = BufferSlab::new(layer_metadata, strides.layer_metadata);
145        let max_elements = BufferPoolSizes {
146            vertices: v_slab.element_capacity,
147            indices: i_slab.element_capacity,
148            layer_metadata: lm_slab.element_capacity,
149            feature_metadata: fm_elems,
150        };
151        Self {
152            vertex_slabs: vec![Some(v_slab)],
153            index_u16_slabs: Vec::new(),
154            index_native_slabs: vec![Some(i_slab)],
155            layer_slabs: vec![Some(lm_slab)],
156            feature_slabs,
157            max_elements,
158            max_page_bytes: u64::MAX,
159            page_factory: None,
160            slab_destroyer: None,
161            decoupled_feature_metadata: false,
162            shared_layer_metadata: false,
163            enable_u16_indices: false,
164            enable_u16_chunks: false,
165            zero_initialized_slabs: false,
166            index: RingIndex::new(),
167            shared_vertices: Default::default(),
168            shared_layer_meta: Default::default(),
169            pattern_runs: Default::default(),
170            sort_runs: Default::default(),
171            vertex_scratch: Vec::new(),
172            u16_index_scratch: Vec::new(),
173            revision: 0,
174            eviction_generation: 0,
175            phantom_i: Default::default(),
176            phantom_q: Default::default(),
177            phantom_m: Default::default(),
178            phantom_fm: Default::default(),
179        }
180    }

182    /// Creates an on-demand paged pool that allocates decoupled slabs lazily through `factory` up
183    /// to `sizes` total elements, with each individual buffer capped at `max_page_bytes`.
184    pub fn with_page_factory(
185        sizes: BufferPoolSizes,
186        max_page_bytes: wgpu::BufferAddress,
187        factory: impl Fn(BackingBufferType, wgpu::BufferAddress) -> B + 'static,
188    ) -> Self {
189        let has_any = !sizes.is_empty() && size_of::<V::Gpu>() > 0 && size_of::<I>() > 0;
190        Self {
191            vertex_slabs: Vec::new(),
192            index_u16_slabs: Vec::new(),
193            index_native_slabs: Vec::new(),
194            layer_slabs: Vec::new(),
195            feature_slabs: Vec::new(),
196            max_elements: sizes,
197            max_page_bytes,
198            page_factory: has_any.then(|| PageFactory(Box::new(factory))),
199            slab_destroyer: None,
200            decoupled_feature_metadata: sizes.feature_metadata < sizes.vertices,
201            shared_layer_metadata: false,
202            enable_u16_indices: false,
203            enable_u16_chunks: false,
204            zero_initialized_slabs: false,
205            index: RingIndex::new(),
206            shared_vertices: Default::default(),
207            shared_layer_meta: Default::default(),
208            pattern_runs: Default::default(),
209            sort_runs: Default::default(),

253    fn remove_tile_inner(
254        &mut self,
255        coords: WorldTileCoords,
256        queue: Option<&Q>,
257        bump_eviction: bool,
258    ) {
259        self.pattern_runs.retain(|(tile, _), _| *tile != coords);
260        self.sort_runs.retain(|(tile, _), _| *tile != coords);
261        let removed = self.index.remove_tile(coords);
262        if !removed.is_empty() {
263            for entry in &removed {
264                self.free_entry_ranges(entry);
265            }
266            let keep_spare = usize::from(!cfg!(target_arch = "wasm32"));
267            self.reclaim_empty_pages_with_queue(keep_spare, queue);
268            self.revision = self.revision.wrapping_add(1);
269            if bump_eviction {
270                self.eviction_generation = self.eviction_generation.wrapping_add(1);
271            }
272        }
273    }

275    /// Removes one style layer without evicting other sources at the same coordinates.
276    pub fn remove_layer(&mut self, coords: WorldTileCoords, id: &str) {
277        let layer_id = crate::style::StyleLayerId::intern(id);
278        self.pattern_runs.remove(&(coords, layer_id));
279        self.sort_runs.remove(&(coords, layer_id));
280        let removed = self.index.remove_layer(coords, layer_id, None);
281        if !removed.is_empty() {
282            for entry in &removed {
283                self.free_entry_ranges(entry);
284            }
285            let keep_spare = usize::from(!cfg!(target_arch = "wasm32"));
286            self.reclaim_empty_pages(keep_spare);
287            self.revision = self.revision.wrapping_add(1);
288            self.eviction_generation = self.eviction_generation.wrapping_add(1);
289        }
290    }

292    /// Records, or with `None` forgets, the pattern runs of a layer's geometry.
293    pub fn set_pattern_runs(
294        &mut self,
295        coords: WorldTileCoords,
296        id: &str,
297        runs: Option<PatternRuns>,
298    ) {
299        let layer_id = crate::style::StyleLayerId::intern(id);
300        match runs {
301            Some(runs) => {
302                if self.pattern_runs.get(&(coords, layer_id)) == Some(&runs) {
303                    return;
304                }
305                self.pattern_runs.insert((coords, layer_id), runs);
306            }
307            None => {
308                self.pattern_runs.remove(&(coords, layer_id));
309            }
310        };
311    }

412impl<Q, B, V: PoolVertex, I, TM, FM> Drop for BufferPool<Q, B, V, I, TM, FM> {
413    fn drop(&mut self) {
414        if let Some(destroyer) = &self.slab_destroyer {
415            for slab in self.vertex_slabs.iter().flatten() {
416                (destroyer.0)(&slab.vertices.inner);
417                if let Some(fm) = &slab.paired_fm {
418                    (destroyer.0)(&fm.inner);
419                }
420            }
421            for slab in self
422                .index_u16_slabs
423                .iter()
424                .flatten()
425                .chain(self.index_native_slabs.iter().flatten())
426                .chain(self.layer_slabs.iter().flatten())
427                .chain(self.feature_slabs.iter().flatten())
428            {
429                (destroyer.0)(&slab.buffer.inner);
430            }
431        }
432    }
433}
