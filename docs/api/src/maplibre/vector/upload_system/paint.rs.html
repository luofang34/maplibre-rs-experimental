14pub(super) fn refresh_layer_paint(
15    buffer_pool: &VectorBufferPool,
16    queue: &crate::render::upload_queue::UploadQueue,
17    source_tiles: &[crate::coords::WorldTileCoords],
18    frame: VectorPaintFrame,
19    previous: Option<VectorPaintFrame>,
20) {
21    let VectorPaintFrame { zoom, .. } = frame;
22    let zoom_unchanged = previous.is_some_and(|prev| prev.zoom == zoom);
23    let mut meta_memo: FxHashMap<(u32, u8), (ShaderLayerMetadata, bool)> = FxHashMap::default();
24    let mut color_memo: FxHashMap<u32, Option<[f32; 4]>> = FxHashMap::default();
25
26    for coords in source_tiles {
27        let tile_z = u8::from(coords.z);
28        for entry in buffer_pool
29            .index()
30            .get_layers(*coords)
31            .into_iter()
32            .flatten()
33        {
34            if zoom_unchanged && !paint_depends_on_bearing(entry.style_layer.paint.as_ref()) {
35                continue;
36            }
37            let (metadata, changed) = *meta_memo
38                .entry((entry.style_layer.index, tile_z))
39                .or_insert_with(|| {
40                    let metadata = metadata_for_layer(&entry.style_layer, *coords, frame);
41                    let changed = previous.is_none_or(|prev| {
42                        bytemuck::bytes_of(&metadata_for_layer(&entry.style_layer, *coords, prev))
43                            != bytemuck::bytes_of(&metadata)
44                    });
45                    (metadata, changed)
46                });
47            if changed {
48                buffer_pool.update_layer_metadata(queue, entry, metadata);
49            }
50            // Uniform fill colours already ride in `ShaderLayerMetadata::stroke_color`.
51            if zoom_unchanged || entry.style_layer.type_ == "fill" {
52                continue;
53            }
54            let changed_color = *color_memo
55                .entry(entry.style_layer.index)
56                .or_insert_with(|| {
57                    let color = uniform_color(&entry.style_layer, f64::from(zoom))?;
58                    let unchanged = previous
59                        .and_then(|prev| uniform_color(&entry.style_layer, f64::from(prev.zoom)))
60                        .is_some_and(|prev_color| same_quantized_rgba(prev_color, color));
61                    (!unchanged).then_some(color)
62                });
63            if let Some(color) = changed_color {
64                buffer_pool.fill_feature_metadata(
65                    queue,
66                    entry,
67                    FillShaderFeatureMetadata { color },
68                );
69            }
70        }
71    }
72}

74/// White with the opacity of a fill that repeats an image, unless the opacity varies by feature.
75fn pattern_color(opacity: Option<&StyleProperty<f32>>, zoom: f64) -> Option<[f32; 4]> {
76    if opacity.is_some_and(|opacity| !opacity.is_feature_constant()) {
77        return None;
78    }
79    let opacity = opacity
80        .and_then(|opacity| opacity.evaluate_at_zoom(zoom))
81        .unwrap_or(1.0)
82        .clamp(0.0, 1.0);
83    Some([1.0, 1.0, 1.0, opacity])
84}

86pub(crate) fn uniform_color(layer: &StyleLayer, zoom: f64) -> Option<[f32; 4]> {
87    let (color, opacity) = match layer.paint.as_ref()? {
88        // A pattern takes only the opacity of the fill.
89        LayerPaint::Fill(paint) if paint.fill_pattern.is_some() => {
90            // A pattern that varies by feature rides in the feature colours.
91            if super::super::pattern::per_feature_pattern(layer.paint.as_ref()?).is_some() {
92                return None;
93            }
94            return pattern_color(paint.fill_opacity.as_ref(), zoom);
95        }
96        // The outline is a feature of its own, with its own colour.
97        LayerPaint::Fill(paint) if paint.fill_outline_color.is_some() => return None,
98        LayerPaint::Fill(paint) => (&paint.fill_color, &paint.fill_opacity),
99        // A gradient's colour comes from its ramp, so the feature colours carry the length.
100        LayerPaint::Line(paint) if paint.line_gradient.is_some() => return None,
101        // A pattern that varies by feature rides in the feature colours.
102        LayerPaint::Line(paint) if paint.line_pattern.is_some() => {
103            if super::super::pattern::per_feature_pattern(layer.paint.as_ref()?).is_some() {
104                return None;
105            }
106            (&paint.line_color, &paint.line_opacity)
107        }
108        LayerPaint::Line(paint) => (&paint.line_color, &paint.line_opacity),
109        LayerPaint::FillExtrusion(paint) if paint.fill_extrusion_pattern.is_some() => {
110            if super::super::pattern::per_feature_pattern(layer.paint.as_ref()?).is_some() {
111                return None;
112            }
113            return if paint.fill_extrusion_translucent == Some(true) {
114                pattern_color(paint.fill_extrusion_opacity.as_ref(), zoom)
115            } else {
116                Some([1.0, 1.0, 1.0, 1.0])
117            };
118        }
119        LayerPaint::FillExtrusion(paint) => {
120            let opacity = if paint.fill_extrusion_translucent == Some(true) {
121                &paint.fill_extrusion_opacity
122            } else {
123                &None
124            };
125            (&paint.fill_extrusion_color, opacity)
126        }
127        _ => return None,
128    };
129    if color.as_ref().is_some_and(|v| !v.is_feature_constant())
130        || opacity.as_ref().is_some_and(|v| !v.is_feature_constant())
131    {
132        return None;
133    }
134    let color = color
135        .as_ref()
136        .and_then(|v| v.evaluate_at_zoom(zoom))
137        .unwrap_or_default();
138    let opacity = opacity
139        .as_ref()
140        .and_then(|v| v.evaluate_at_zoom(zoom))
141        .unwrap_or(1.0)
142        .clamp(0.0, 1.0);
143    Some([color.r, color.g, color.b, color.a * opacity])
144}

149#[derive(Default)]
150struct PaintZoom(Option<f64>);

152pub(super) fn stabilize_zoom(world: &mut crate::tcs::world::World, requested: f64) -> f64 {
153    let state = world.resources.get_or_init_mut::<PaintZoom>();
154    let zoom = next_zoom(state.0, requested);
155    state.0 = Some(zoom);
156    zoom
157}
158
159fn next_zoom(previous: Option<f64>, requested: f64) -> f64 {
160    previous
161        .filter(|previous| (requested - previous).abs() < 0.125)
162        .unwrap_or_else(|| super::drape_paint_zoom(requested))
163}
164
165pub(crate) fn current_zoom(world: &crate::tcs::world::World, fallback: f64) -> f64 {
166    world
167        .resources
168        .get::<PaintZoom>()
169        .and_then(|state| state.0)
170        .unwrap_or_else(|| super::drape_paint_zoom(fallback))
171}
