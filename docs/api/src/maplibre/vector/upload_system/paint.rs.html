4/// White with the opacity of a fill that repeats an image, unless the opacity varies by feature.
5fn pattern_color(opacity: Option<&StyleProperty<f32>>, zoom: f64) -> Option<[f32; 4]> {
6    if opacity.is_some_and(|opacity| !opacity.is_feature_constant()) {
7        return None;
8    }
9    let opacity = opacity
10        .and_then(|opacity| opacity.evaluate_at_zoom(zoom))
11        .unwrap_or(1.0)
12        .clamp(0.0, 1.0);
13    Some([1.0, 1.0, 1.0, opacity])
14}

16pub(super) fn uniform_color(layer: &StyleLayer, zoom: f64) -> Option<[f32; 4]> {
17    let (color, opacity) = match layer.paint.as_ref()? {
18        // A pattern takes only the opacity of the fill.
19        LayerPaint::Fill(paint) if paint.fill_pattern.is_some() => {
20            // A pattern that varies by feature rides in the feature colours.
21            if super::super::pattern::per_feature_pattern(layer.paint.as_ref()?).is_some() {
22                return None;
23            }
24            return pattern_color(paint.fill_opacity.as_ref(), zoom);
25        }
26        // The outline is a feature of its own, with its own colour.
27        LayerPaint::Fill(paint) if paint.fill_outline_color.is_some() => return None,
28        LayerPaint::Fill(paint) => (&paint.fill_color, &paint.fill_opacity),
29        // A gradient's colour comes from its ramp, so the feature colours carry the length.
30        LayerPaint::Line(paint) if paint.line_gradient.is_some() => return None,
31        // A pattern that varies by feature rides in the feature colours.
32        LayerPaint::Line(paint) if paint.line_pattern.is_some() => {
33            if super::super::pattern::per_feature_pattern(layer.paint.as_ref()?).is_some() {
34                return None;
35            }
36            (&paint.line_color, &paint.line_opacity)
37        }
38        LayerPaint::Line(paint) => (&paint.line_color, &paint.line_opacity),
39        _ => return None,
40    };
41    if color.as_ref().is_some_and(|v| !v.is_feature_constant())
42        || opacity.as_ref().is_some_and(|v| !v.is_feature_constant())
43    {
44        return None;
45    }
46    let color = color
47        .as_ref()
48        .and_then(|v| v.evaluate_at_zoom(zoom))
49        .unwrap_or_default();
50    let opacity = opacity
51        .as_ref()
52        .and_then(|v| v.evaluate_at_zoom(zoom))
53        .unwrap_or(1.0)
54        .clamp(0.0, 1.0);
55    Some([
56        color.r as f32,
57        color.g as f32,
58        color.b as f32,
59        color.a as f32 * opacity,
60    ])
61}

66#[derive(Default)]
67struct PaintZoom(Option<f64>);
68
69pub(super) fn stabilize_zoom(world: &mut crate::tcs::world::World, requested: f64) -> f64 {
70    let state = world.resources.get_or_init_mut::<PaintZoom>();
71    let zoom = next_zoom(state.0, requested);
72    state.0 = Some(zoom);
73    zoom
74}
75
76fn next_zoom(previous: Option<f64>, requested: f64) -> f64 {
77    previous
78        .filter(|previous| (requested - previous).abs() < 0.125)
79        .unwrap_or_else(|| super::drape_paint_zoom(requested))
80}
81
82pub(crate) fn current_zoom(world: &crate::tcs::world::World, fallback: f64) -> f64 {
83    world
84        .resources
85        .get::<PaintZoom>()
86        .and_then(|state| state.0)
87        .unwrap_or_else(|| super::drape_paint_zoom(fallback))
88}
