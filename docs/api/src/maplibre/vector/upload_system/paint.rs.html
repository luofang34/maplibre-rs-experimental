17pub(super) fn refresh_layer_paint(
18    buffer_pool: &VectorBufferPool,
19    queue: &crate::render::upload_queue::UploadQueue,
20    source_tiles: &[crate::coords::WorldTileCoords],
21    frame: VectorPaintFrame,
22    previous: Option<VectorPaintFrame>,
23) {
24    let VectorPaintFrame { zoom, .. } = frame;
25    let zoom_changed = previous.is_none_or(|prev| prev.zoom != zoom);
26    let bearing_changed = previous.is_none_or(|prev| prev.bearing != frame.bearing);
27    let light_changed = previous.is_none_or(|prev| prev.light != frame.light);
28    let mut meta_memo: FxHashMap<(u32, u8), (ShaderLayerMetadata, bool)> = FxHashMap::default();
29    let mut updated_layer_slots: FxHashSet<(u16, u64)> = FxHashSet::default();
30    let mut color_memo: FxHashMap<u32, Option<[f32; 4]>> = FxHashMap::default();
31
32    for coords in source_tiles {
33        let tile_z = u8::from(coords.z);
34        for entry in buffer_pool
35            .index()
36            .get_layers(*coords)
37            .into_iter()
38            .flatten()
39        {
40            let paint = entry.style_layer.paint.as_ref();
41            let meta_dirty = (zoom_changed && paint_metadata_depends_on_zoom(paint))
42                || (bearing_changed && paint_depends_on_bearing(paint))
43                || (light_changed && matches!(paint, Some(LayerPaint::FillExtrusion(_))));
44            if meta_dirty {
45                let (metadata, changed) = *meta_memo
46                    .entry((entry.style_layer.index, tile_z))
47                    .or_insert_with(|| {
48                        let metadata = metadata_for_layer(&entry.style_layer, *coords, frame);
49                        let changed = previous.is_none_or(|prev| {
50                            bytemuck::bytes_of(&metadata_for_layer(
51                                &entry.style_layer,
52                                *coords,
53                                prev,
54                            )) != bytemuck::bytes_of(&metadata)
55                        });
56                        (metadata, changed)
57                    });
58                if changed
59                    && updated_layer_slots
60                        .insert((entry.page(), entry.layer_metadata_buffer_range().start))
61                {
62                    buffer_pool.update_layer_metadata(queue, entry, metadata);
63                }
64            }
65            // Uniform fill and line colours already ride in `ShaderLayerMetadata::stroke_color`.
66            if !zoom_changed
67                || uses_layer_metadata_uniform_color(&entry.style_layer)
68                || !uniform_color_depends_on_zoom(&entry.style_layer)
69            {
70                continue;
71            }
72            let changed_color = *color_memo
73                .entry(entry.style_layer.index)
74                .or_insert_with(|| {
75                    let color = uniform_color(&entry.style_layer, f64::from(zoom))?;
76                    let unchanged = previous
77                        .and_then(|prev| uniform_color(&entry.style_layer, f64::from(prev.zoom)))
78                        .is_some_and(|prev_color| same_quantized_rgba(prev_color, color));
79                    (!unchanged).then_some(color)
80                });
81            if let Some(color) = changed_color {
82                buffer_pool.fill_feature_metadata(
83                    queue,
84                    entry,
85                    FillShaderFeatureMetadata::from_rgba(color),
86                );
87            }
88        }
89    }
90}

92/// White with the opacity of a fill that repeats an image, unless the opacity varies by feature.
93fn pattern_color(opacity: Option<&StyleProperty<f32>>, zoom: f64) -> Option<[f32; 4]> {
94    if opacity.is_some_and(|opacity| !opacity.is_feature_constant()) {
95        return None;
96    }
97    let opacity = opacity
98        .and_then(|opacity| opacity.evaluate_at_zoom(zoom))
99        .unwrap_or(1.0)
100        .clamp(0.0, 1.0);
101    Some([1.0, 1.0, 1.0, opacity])
102}

104pub(crate) fn uniform_color(layer: &StyleLayer, zoom: f64) -> Option<[f32; 4]> {
105    let Some(layer_paint) = layer.paint.as_ref() else {
106        return uses_layer_metadata_uniform_color(layer).then_some([0.0, 0.0, 0.0, 1.0]);
107    };
108    let (color, opacity) = match layer_paint {
109        // A pattern takes only the opacity of the fill.
110        LayerPaint::Fill(paint) if paint.fill_pattern.is_some() => {
111            // A pattern that varies by feature rides in the feature colours.
112            if super::super::pattern::per_feature_pattern(layer_paint).is_some() {
113                return None;
114            }
115            return pattern_color(paint.fill_opacity.as_ref(), zoom);
116        }
117        // The outline is a feature of its own, with its own colour.
118        LayerPaint::Fill(paint) if paint.fill_outline_color.is_some() => return None,
119        LayerPaint::Fill(paint) => (&paint.fill_color, &paint.fill_opacity),
120        // A gradient's colour comes from its ramp, so the feature colours carry the length.
121        LayerPaint::Line(paint) if paint.line_gradient.is_some() => return None,
122        // A pattern that varies by feature rides in the feature colours.
123        LayerPaint::Line(paint) if paint.line_pattern.is_some() => {
124            if super::super::pattern::per_feature_pattern(layer_paint).is_some() {
125                return None;
126            }
127            (&paint.line_color, &paint.line_opacity)
128        }
129        LayerPaint::Line(paint) => (&paint.line_color, &paint.line_opacity),
130        LayerPaint::FillExtrusion(paint) if paint.fill_extrusion_pattern.is_some() => {
131            if super::super::pattern::per_feature_pattern(layer_paint).is_some() {
132                return None;
133            }
134            return if paint.fill_extrusion_translucent == Some(true) {
135                pattern_color(paint.fill_extrusion_opacity.as_ref(), zoom)
136            } else {
137                Some([1.0, 1.0, 1.0, 1.0])
138            };
139        }
140        LayerPaint::FillExtrusion(paint) => {
141            let opacity = if paint.fill_extrusion_translucent == Some(true) {
142                &paint.fill_extrusion_opacity
143            } else {
144                &None
145            };
146            (&paint.fill_extrusion_color, opacity)
147        }
148        LayerPaint::Circle(paint) => {
149            if paint.stroke_color_per_feature().is_some() {
150                return None;
151            }
152            (&paint.circle_color, &paint.circle_opacity)
153        }
154        LayerPaint::Heatmap(_) => return Some([0.0, 0.0, 0.0, 1.0]),
155        _ => return None,
156    };
157    if color.as_ref().is_some_and(|v| !v.is_feature_constant())
158        || opacity.as_ref().is_some_and(|v| !v.is_feature_constant())
159    {
160        return None;
161    }
162    let color = color
163        .as_ref()
164        .and_then(|v| v.evaluate_at_zoom(zoom))
165        .unwrap_or_default();
166    let opacity = opacity
167        .as_ref()
168        .and_then(|v| v.evaluate_at_zoom(zoom))
169        .unwrap_or(1.0)
170        .clamp(0.0, 1.0);
171    Some([color.r, color.g, color.b, color.a * opacity])
172}

177#[derive(Default)]
178struct PaintZoom(Option<f64>);

180pub(super) fn stabilize_zoom(world: &mut crate::tcs::world::World, requested: f64) -> f64 {
181    let state = world.resources.get_or_init_mut::<PaintZoom>();
182    let zoom = next_zoom(state.0, requested);
183    state.0 = Some(zoom);
184    zoom
185}
186
187fn next_zoom(previous: Option<f64>, requested: f64) -> f64 {
188    previous
189        .filter(|previous| (requested - previous).abs() < 0.125)
190        .unwrap_or_else(|| super::drape_paint_zoom(requested))
191}
192
193pub(crate) fn current_zoom(world: &crate::tcs::world::World, fallback: f64) -> f64 {
194    world
195        .resources
196        .get::<PaintZoom>()
197        .and_then(|state| state.0)
198        .unwrap_or_else(|| super::drape_paint_zoom(fallback))
199}
