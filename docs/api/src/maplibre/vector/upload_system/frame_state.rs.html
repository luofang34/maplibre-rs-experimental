1//! Per-frame quantized paint state for vector uploads and drape shading.
2
3use super::paint;
4use crate::{render::view_state::ViewState, style::Style, tcs::world::World};
5
6/// Rounds `zoom` to eighth-zoom steps for drape paint evaluation.
7pub(crate) fn drape_paint_zoom(zoom: f64) -> f64 {
8    (zoom * 8.0).round() / 8.0
9}
10
11#[derive(Clone, Copy, PartialEq)]
12pub(in crate::vector::upload_system) struct VectorPaintFrame {
13    pub(in crate::vector::upload_system) zoom: f32,
14    pub(in crate::vector::upload_system) bearing: f32,
15    /// The root light as extruded polygons are shaded with it.
16    pub(in crate::vector::upload_system) light: crate::style::light::ExtrusionLight,
17}
18
19/// Computes the current frame's quantized paint state alongside the previous frame's state.
20pub(in crate::vector::upload_system) fn frame_paint(
21    world: &mut World,
22    style: &Style,
23    view_state: &ViewState,
24) -> (VectorPaintFrame, Option<VectorPaintFrame>) {
25    let zoom = if style.terrain.is_some() {
26        paint::stabilize_zoom(world, view_state.style_zoom().value()) as f32
27    } else {
28        (view_state.style_zoom().level() * 256.0).round() / 256.0
29    };
30    let bearing = ((view_state.camera().get_bearing().0.to_degrees() * 4.0).round() * 0.25)
31        .to_radians() as f32;
32    let light = style
33        .light
34        .as_ref()
35        .map_or_else(
36            || {
37                crate::style::light::LightSpecification::default()
38                    .extrusion_light(view_state.style_zoom().value(), f64::from(bearing))
39            },
40            |light| light.extrusion_light(view_state.style_zoom().value(), f64::from(bearing)),
41        )
42        .unwrap_or_default();
43    let paint_frame = VectorPaintFrame {
44        zoom,
45        bearing,
46        light,
47    };
48    let previous_paint = world.resources.get::<VectorPaintFrame>().copied();
49    world.resources.insert(paint_frame);
50    (paint_frame, previous_paint)
51}
