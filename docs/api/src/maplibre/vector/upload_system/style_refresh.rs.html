43/// Queued in-place style and `feature-state` mutations awaiting the next GPU upload pass.
44#[derive(Default, Debug)]
45pub(crate) struct DirtyVectorStyle {
46    /// Sync `StyleLayer` metadata (`index`, zoom range, visibility, uniforms) on GPU pools.
47    pub sync_layers: bool,
48    /// Style layer IDs whose per-feature colors/opacity/outline may have changed.
49    pub paint_layers: BTreeSet<String>,
50    /// Style source names whose runtime `feature-state` changed.
51    pub feature_state_sources: BTreeSet<String>,
52}
53
54impl DirtyVectorStyle {
55    /// Queues synchronization of layer order, zoom bounds, visibility, and paint for `layers`.
56    pub fn mark_style_edit(world: &mut World, paint_layers: impl IntoIterator<Item = String>) {
57        let dirty = world.resources.get_or_init_mut::<Self>();
58        dirty.sync_layers = true;
59        dirty.paint_layers.extend(paint_layers);
60    }
61
62    /// Queues an in-place `feature-state` update for `source`.
63    pub fn mark_feature_state(world: &mut World, source: impl Into<String>) {
64        let dirty = world.resources.get_or_init_mut::<Self>();
65        dirty.feature_state_sources.insert(source.into());
66    }
67}

69pub(super) fn apply_dirty_style(
70    world: &mut World,
71    style: &Style,
72    queue: &UploadQueue,
73    frame: VectorPaintFrame,
74) {
75    let Some(dirty) = world.resources.remove::<DirtyVectorStyle>() else {
76        return;
77    };
78    let mut needs_worker_refresh = false;
79    let mut recolor_layers = dirty.paint_layers;
80
81    if !dirty.feature_state_sources.is_empty() {
82        let mut memo = FeatureStateArcMemo::default();
83        for source in &dirty.feature_state_sources {
84            world
85                .tiles
86                .geometry_index
87                .sync_source_feature_state_with_memo(
88                    source,
89                    style.source_promote_id(source),
90                    style.feature_state.source(source),
91                    &mut memo,
92                );
93        }
94        for layer in &style.layers {
95            let Some(source) = layer.source.as_deref() else {
96                continue;
97            };
98            if !dirty.feature_state_sources.contains(source) {
99                continue;
100            };
101            if layer_geometry_reads_feature_state(layer) {
102                needs_worker_refresh = true;
103            }
104            if layer.type_ != "symbol" && layer_paint_reads_feature_state(layer) {
105                recolor_layers.insert(layer.id.clone());
106            }
107        }
108        feature_state::sync_tile_feature_states(
109            world,
110            style,
111            &dirty.feature_state_sources,
112            &mut memo,
113            &mut needs_worker_refresh,
114        );
115    }
116
117    if dirty.sync_layers {
118        sync_symbol_pool_layers(world, style, queue);
119    }
120
121    if dirty.sync_layers || !recolor_layers.is_empty() {
122        refresh_vector_pool(
123            world,
124            style,
125            queue,
126            frame,
127            dirty.sync_layers,
128            &recolor_layers,
129            &mut needs_worker_refresh,
130        );
131    }
132
133    if needs_worker_refresh {
134        tile_retry::refresh(world, RequestKind::Vector);
135    }
136}

138fn sync_symbol_pool_layers(world: &mut World, style: &Style, queue: &UploadQueue) {
139    let Some(Initialized(symbol_pool)) = world.resources.get_mut::<Eventually<SymbolBufferPool>>()
140    else {
141        return;
142    };
143    for style_layer in &style.layers {
144        if style_layer.type_ != "symbol" {
145            continue;
146        }
147        let touched = symbol_pool.sync_style_layer(style_layer);
148        let metadata = ShaderLayerMetadata::new(style_layer.index as f32, 0.0, [0.0; 2]);
149        for coords in touched {
150            if let Some(entry) = symbol_pool
151                .index()
152                .get_layers(coords)
153                .and_then(|layers| layers.iter().find(|e| e.style_layer.id == style_layer.id))
154            {
155                symbol_pool.update_layer_metadata(queue, entry, metadata);
156            }
157        }
158    }
159}

161fn refresh_vector_pool(
162    world: &mut World,
163    style: &Style,
164    queue: &UploadQueue,
165    frame: VectorPaintFrame,
166    sync_all_layers: bool,
167    recolor_layers: &BTreeSet<String>,
168    needs_worker_refresh: &mut bool,
169) {
170    let mut touched_by_layer: HashMap<String, HashSet<WorldTileCoords>> = HashMap::new();
171    let mut paint_changed_by_layer: HashMap<String, HashSet<WorldTileCoords>> = HashMap::new();
172    let mut uploaded_by_layer: HashMap<String, HashSet<WorldTileCoords>> = HashMap::new();
173    {
174        let Some(Initialized(vector_pool)) =
175            world.resources.get_mut::<Eventually<VectorBufferPool>>()
176        else {
177            return;
178        };
179        if sync_all_layers {
180            for style_layer in &style.layers {
181                let paint_coords: HashSet<WorldTileCoords> = vector_pool
182                    .index()
183                    .iter()
184                    .flatten()
185                    .filter(|e| {
186                        e.style_layer.id == style_layer.id && !e.style_layer.same_paint(style_layer)
187                    })
188                    .map(|e| e.coords)
189                    .collect();
190                if !paint_coords.is_empty() {
191                    paint_changed_by_layer.insert(style_layer.id.clone(), paint_coords);
192                }
193                let coords: HashSet<WorldTileCoords> = vector_pool
194                    .sync_style_layer(style_layer)
195                    .into_iter()
196                    .collect();
197                if !coords.is_empty() {
198                    touched_by_layer.insert(style_layer.id.clone(), coords);
199                }
200            }
201        }
202        for layer_id in recolor_layers {
203            let coords: HashSet<WorldTileCoords> = vector_pool
204                .index()
205                .iter()
206                .flatten()
207                .filter(|e| &e.style_layer.id == layer_id)
208                .map(|e| e.coords)
209                .collect();
210            if !coords.is_empty() {
211                uploaded_by_layer.insert(layer_id.clone(), coords);
212            }
213        }
214    }
215
216    struct GpuPaintUpdate {
217        coords: WorldTileCoords,
218        layer_id: String,
219        feature_meta: Vec<FillShaderFeatureMetadata>,
220        runs: Option<Vec<(u32, std::ops::Range<u32>)>>,
221        bump_revision: bool,
222    }
223
224    let all_tile_coords: Vec<WorldTileCoords> =
225        world.tiles.tiles.values().map(|tile| tile.coords).collect();
226    let mut gpu_updates: Vec<GpuPaintUpdate> = Vec::new();
227
228    for style_layer in &style.layers {
229        if !recolor_layers.contains(&style_layer.id) {
230            continue;
231        }
232        let uniform = paint::uniform_color(style_layer, f64::from(frame.zoom));
233        let fallback_color: [f32; 4] = style_layer
234            .paint
235            .as_ref()
236            .and_then(LayerPaint::get_color)
237            .map(Into::into)
238            .unwrap_or([0.0, 0.0, 0.0, 1.0]);
239        let mut candidate_coords: BTreeSet<WorldTileCoords> =
240            all_tile_coords.iter().copied().collect();
241        if let Some(uploaded) = uploaded_by_layer.get(&style_layer.id) {
242            candidate_coords.extend(uploaded.iter().copied());
243        }
244
245        for coords in candidate_coords {
246            let is_uploaded = uploaded_by_layer
247                .get(&style_layer.id)
248                .is_some_and(|set| set.contains(&coords));
249            let is_layer_touched = touched_by_layer
250                .get(&style_layer.id)
251                .is_some_and(|set| set.contains(&coords));
252            let is_paint_changed = paint_changed_by_layer
253                .get(&style_layer.id)
254                .is_some_and(|set| set.contains(&coords));
255            let (committed_entries, pending_entries) = world
256                .tiles
257                .query::<&TileFeaturePaintComponent>(coords)
258                .map(|store| {
259                    (
260                        store.committed.get(&style_layer.id).cloned(),
261                        store.pending.get(&style_layer.id).cloned(),
262                    )
263                })
264                .unwrap_or((None, None));
265            let overscaled_zoom = world
266                .tiles
267                .query::<&VectorLayerBucketComponent>(coords)
268                .map_or(0, |b| b.overscaled_zoom);
269            let style_zoom = f64::from(overscaled_zoom.max(u8::from(coords.z)));
270
271            if let Some(pending) = world.tiles.query_mut::<&mut LayerReplacements>(coords) {
272                if let Some(bucket) = pending
273                    .vector
274                    .iter_mut()
275                    .find(|b| b.style_layer_id == style_layer.id)
276                {
277                    let _ = recolor_cpu_bucket(
278                        style_layer,
279                        bucket,
280                        pending_entries.as_deref(),
281                        uniform,
282                        style_zoom,
283                        needs_worker_refresh,
284                    );
285                }
286            }
287
288            let mut handled = false;
289            if let Some(buckets) = world
290                .tiles
291                .query_mut::<&mut VectorLayerBucketComponent>(coords)
292            {
293                if let Some(bucket) = buckets.layers.iter_mut().find_map(|b| match b {
294                    VectorLayerBucket::AvailableLayer(avail)
295                        if avail.style_layer_id == style_layer.id =>
296                    {
297                        Some(avail)
298                    }
299                    _ => None,
300                }) {
301                    handled = true;
302                    let (ok, colors_changed) = recolor_cpu_bucket(
303                        style_layer,
304                        bucket,
305                        committed_entries.as_deref(),
306                        uniform,
307                        style_zoom,
308                        needs_worker_refresh,
309                    );
310                    if !ok || !is_uploaded || (!colors_changed && !is_paint_changed) {
311                        continue;
312                    }
313                    let metadata = if let Some(color) = uniform {
314                        feature_metadata(&bucket.feature_indices, &[], color)
315                    } else {
316                        feature_metadata(
317                            &bucket.feature_indices,
318                            &bucket.feature_colors,
319                            fallback_color,
320                        )
321                    };
322                    let runs = style_layer
323                        .paint
324                        .as_ref()
325                        .and_then(per_feature_pattern)
326                        .map(|_| {
327                            pattern_runs(
328                                &bucket.feature_indices,
329                                &bucket.feature_colors,
330                                &bucket.buffer.buffer.indices,
331                            )
332                        });
333                    gpu_updates.push(GpuPaintUpdate {
334                        coords,
335                        layer_id: style_layer.id.clone(),
336                        feature_meta: metadata,
337                        runs,
338                        bump_revision: !is_layer_touched,
339                    });
340                }
341            }
342            if !handled && is_uploaded && is_paint_changed {
343                if let Some(color) = uniform {
344                    if let Some(Initialized(pool)) =
345                        world.resources.get::<Eventually<VectorBufferPool>>()
346                    {
347                        if let Some(entry) = pool
348                            .index()
349                            .get_layers(coords)
350                            .and_then(|l| l.iter().find(|e| e.style_layer.id == style_layer.id))
351                        {
352                            let count = (entry.feature_metadata_buffer_range().end
353                                - entry.feature_metadata_buffer_range().start)
354                                as usize
355                                / size_of::<FillShaderFeatureMetadata>();
356                            gpu_updates.push(GpuPaintUpdate {
357                                coords,
358                                layer_id: style_layer.id.clone(),
359                                feature_meta: vec![FillShaderFeatureMetadata { color }; count],
360                                runs: None,
361                                bump_revision: !is_layer_touched,
362                            });
363                        }
364                    }
365                }
366            }
367        }
368    }
369
370    let Some(Initialized(vector_pool)) = world.resources.get_mut::<Eventually<VectorBufferPool>>()
371    else {
372        return;
373    };
374    for style_layer in &style.layers {
375        let Some(coords_set) = touched_by_layer.get(&style_layer.id) else {
376            continue;
377        };
378        for &coords in coords_set {
379            let layer_meta = metadata_for_layer(style_layer, coords, frame);
380            if let Some(entry) = vector_pool
381                .index()
382                .get_layers(coords)
383                .and_then(|layers| layers.iter().find(|e| e.style_layer.id == style_layer.id))
384            {
385                vector_pool.update_layer_metadata(queue, entry, layer_meta);
386            }
387        }
388    }
389    for GpuPaintUpdate {
390        coords,
391        layer_id,
392        feature_meta,
393        runs,
394        bump_revision,
395    } in gpu_updates
396    {
397        if let Some(entry) = vector_pool
398            .index()
399            .get_layers(coords)
400            .and_then(|layers| layers.iter().find(|e| e.style_layer.id == layer_id))
401        {
402            vector_pool.update_feature_metadata(queue, entry, &feature_meta);
403        }
404        vector_pool.set_pattern_runs(coords, &layer_id, runs);
405        if bump_revision {
406            vector_pool.bump_layer_paint_revision(coords, &layer_id);
407        }
408    }
409}

411fn recolor_cpu_bucket(
412    style_layer: &StyleLayer,
413    bucket: &mut AvailableVectorLayerBucket,
414    entries: Option<&[BucketFeatureEntry]>,
415    uniform: Option<[f32; 4]>,
416    style_zoom: f64,
417    needs_worker_refresh: &mut bool,
418) -> (bool, bool) {
419    if let Some(entries) = entries {
420        let new_colors = evaluate_bucket_colors(style_layer, entries, style_zoom);
421        let changed = new_colors != bucket.feature_colors;
422        bucket.feature_colors = new_colors;
423        (true, changed)
424    } else if uniform.is_some() {
425        (true, false)
426    } else if layer_is_feature_constant(style_layer) {
427        let single = [BucketFeatureEntry {
428            raw_id: None,
429            properties: Arc::default(),
430            kind: BucketEntryKind::Feature,
431        }];
432        let const_color = evaluate_bucket_colors(style_layer, &single, style_zoom)[0];
433        let new_colors = vec![const_color; bucket.feature_indices.len()];
434        let changed = new_colors != bucket.feature_colors;
435        bucket.feature_colors = new_colors;
436        (true, changed)
437    } else {
438        if !bucket.buffer.buffer.indices.is_empty() {
439            *needs_worker_refresh = true;
440        }
441        (false, false)
442    }
443}

445fn layer_is_feature_constant(layer: &StyleLayer) -> bool {
446    match layer.paint.as_ref() {
447        Some(LayerPaint::Circle(c)) => {
448            c.circle_color
449                .as_ref()
450                .is_none_or(|p| p.is_feature_constant())
451                && c.circle_opacity
452                    .as_ref()
453                    .is_none_or(|p| p.is_feature_constant())
454                && c.stroke_color_per_feature().is_none()
455        }
456        Some(LayerPaint::FillExtrusion(e)) => {
457            e.fill_extrusion_color
458                .as_ref()
459                .is_none_or(|p| p.is_feature_constant())
460                && e.fill_extrusion_opacity
461                    .as_ref()
462                    .is_none_or(|p| p.is_feature_constant())
463        }
464        _ => false,
465    }
