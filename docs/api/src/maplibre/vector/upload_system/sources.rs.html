1//! Geometry uploads follow the actual sources selected for each terrain texture.
2use std::collections::HashSet;
3
4use crate::{
5    coords::WorldTileCoords,
6    io::tile_sources::{clamp_to_max_zoom, source_max_zoom, TileKind},
7    style::Style,
8    tcs::world::World,
9    terrain::{drape_targets::select_targets, request_system::DrapeRequests},
10};
11
12pub(super) fn terrain_sources(world: &World, style: &Style) -> Vec<WorldTileCoords> {
13    let max_zoom = source_max_zoom(style, TileKind::Vector);
14    let targets = std::iter::once(WorldTileCoords::default()).chain(
15        world
16            .resources
17            .get::<DrapeRequests>()
18            .into_iter()
19            .flat_map(|requests| requests.0.iter().copied()),
20    );
21    let selected = select_targets(targets, world, &[]);
22    let mut seen = HashSet::new();
23    let mut sources = Vec::new();
24    for (target, shapes) in selected {
25        // A CPU-ready ancestor or complete child cover must reach the GPU even when
26        // the requested tile exceeds maxzoom or has not arrived yet.
27        for source in shapes
28            .into_iter()
29            .filter(|shape| shape.raster_source.is_none())
30            .map(|shape| shape.coords)
31            .chain(std::iter::once(clamp_to_max_zoom(target, max_zoom)))
32        {
33            if seen.insert(source) {
34                sources.push(source);
35            }
36        }
37    }
38    sources
39}
