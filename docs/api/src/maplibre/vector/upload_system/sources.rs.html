17const CHILDREN_SEARCH_DEPTH: usize = 4;
18
19pub(super) fn frame_sources_and_painted(
20    world: &World,
21    style: &Style,
22    view_state: &ViewState,
23) -> Result<(Vec<WorldTileCoords>, Vec<WorldTileCoords>), SystemError> {
24    let view_region = view_region_for_projection(
25        style,
26        view_state,
27        world,
28        view_state.zoom().zoom_level(DEFAULT_TILE_SIZE),
29        ViewStatePadding::Tight,
30    )
31    .map_err(|error| {
32        tracing::error!(%error, "unable to select tiles for vector upload");
33        SystemError::Setup
34    })?;
35    let Some(view_tile_sources) = world.resources.get::<ViewTileSources>() else {
36        return Err(SystemError::Dependencies);
37    };
38    let cpu_vector = view_tile_sources.of_kind(TileKind::Vector);
39    let gpu_vector = (
40        view_tile_sources.of_kind(TileKind::Vector),
41        crate::vector::UploadedVectorGeometry::for_world(world),
42    );
43    let source_tiles = if style.terrain.is_some() {
44        terrain_sources(world, style)
45    } else {
46        let mut sources = Vec::new();
47        let mut seen = HashSet::new();
48        let mut targets = Vec::new();
49        if let Some(region) = &view_region {
50            for copy in region.copies() {
51                let Some((coords, _)) = copy.wrapped() else {
52                    continue;
53                };
54                targets.push(coords);
55                for ready in loaded_cover(&cpu_vector, coords, world)
56                    .or_else(|| {
57                        cpu_vector.get_available_children(coords, world, CHILDREN_SEARCH_DEPTH)
58                    })
59                    .into_iter()
60                    .flatten()
61                {
62                    if seen.insert(ready) {
63                        sources.push(ready);
64                    }
65                }
66            }
67        }
68        for coords in targets {
69            if seen.insert(coords) {
70                sources.push(coords);
71            }
72        }
73        sources
74    };
75    let mut painted_tiles = source_tiles.clone();
76    let mut seen_painted: HashSet<_> = painted_tiles.iter().copied().collect();
77    if let Some(region) = &view_region {
78        for copy in region.copies() {
79            let Some((coords, _)) = copy.wrapped() else {
80                continue;
81            };
82            for stand_in in loaded_cover(&gpu_vector, coords, world)
83                .or_else(|| gpu_vector.get_available_children(coords, world, CHILDREN_SEARCH_DEPTH))
84                .into_iter()
85                .flatten()
86            {
87                if seen_painted.insert(stand_in) {
88                    painted_tiles.push(stand_in);
89                }
90            }
91        }
92    }
93    Ok((source_tiles, painted_tiles))
94}

96pub(super) fn terrain_sources(world: &World, style: &Style) -> Vec<WorldTileCoords> {
97    let max_zoom = source_max_zoom(style, TileKind::Vector);
98    let targets = std::iter::once(WorldTileCoords::default())
99        .chain(
100            world
101                .resources
102                .get::<DrapeRequests>()
103                .into_iter()
104                .flat_map(|requests| requests.0.iter().copied()),
105        )
106        .chain(
107            world
108                .resources
109                .get::<crate::render::eye_covering::SlotResidencyFloor>()
110                .into_iter()
111                .flat_map(|floor| floor.retained_drapes.iter().copied()),
112        );
113    let selected = select_targets(targets, world, &[]);
114    let mut seen = HashSet::new();
115    let mut sources = Vec::new();
116    for (target, shapes) in selected {
117        // A CPU-ready ancestor or complete child cover must reach the GPU even when
118        // the requested tile exceeds maxzoom or has not arrived yet.
119        for source in shapes
120            .into_iter()
121            .filter(|shape| shape.raster_source.is_none())
122            .map(|shape| shape.coords)
123            .chain(std::iter::once(clamp_to_max_zoom(target, max_zoom)))
124        {
125            if seen.insert(source) {
126                sources.push(source);
127            }
128        }
129    }
130    sources
131}
