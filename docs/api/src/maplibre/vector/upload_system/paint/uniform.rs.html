1//! Evaluates uniform layer colors for geometry buckets with feature-constant paint.
2
3use crate::{
4    style::layer::{LayerPaint, StyleLayer, StyleProperty},
5    vector::upload_system::metadata::uses_layer_metadata_uniform_color,
6};
7
8/// White with the opacity of a fill that repeats an image, unless the opacity varies by feature.
9fn pattern_color(opacity: Option<&StyleProperty<f32>>, zoom: f64) -> Option<[f32; 4]> {
10    if opacity.is_some_and(|opacity| !opacity.is_feature_constant()) {
11        return None;
12    }
13    let opacity = opacity
14        .and_then(|opacity| opacity.evaluate_at_zoom(zoom))
15        .unwrap_or(1.0)
16        .clamp(0.0, 1.0);
17    Some([1.0, 1.0, 1.0, opacity])
18}
19
20pub(crate) fn uniform_color(layer: &StyleLayer, zoom: f64) -> Option<[f32; 4]> {
21    let Some(layer_paint) = layer.paint.as_ref() else {
22        return uses_layer_metadata_uniform_color(layer).then_some([0.0, 0.0, 0.0, 1.0]);
23    };
24    let (color, opacity) = match layer_paint {
25        // A pattern takes only the opacity of the fill.
26        LayerPaint::Fill(paint) if paint.fill_pattern.is_some() => {
27            // A pattern that varies by feature rides in the feature colours.
28            if crate::vector::pattern::per_feature_pattern(layer_paint).is_some() {
29                return None;
30            }
31            return pattern_color(paint.fill_opacity.as_ref(), zoom);
32        }
33        // The outline is a feature of its own, with its own colour.
34        LayerPaint::Fill(paint) if paint.fill_outline_color.is_some() => return None,
35        LayerPaint::Fill(paint) => (&paint.fill_color, &paint.fill_opacity),
36        // A gradient's colour comes from its ramp, so the feature colours carry the length.
37        LayerPaint::Line(paint) if paint.line_gradient.is_some() => return None,
38        // A pattern that varies by feature rides in the feature colours.
39        LayerPaint::Line(paint) if paint.line_pattern.is_some() => {
40            if crate::vector::pattern::per_feature_pattern(layer_paint).is_some() {
41                return None;
42            }
43            (&paint.line_color, &paint.line_opacity)
44        }
45        LayerPaint::Line(paint) => (&paint.line_color, &paint.line_opacity),
46        LayerPaint::FillExtrusion(paint) if paint.fill_extrusion_pattern.is_some() => {
47            if crate::vector::pattern::per_feature_pattern(layer_paint).is_some() {
48                return None;
49            }
50            return if paint.fill_extrusion_translucent == Some(true) {
51                pattern_color(paint.fill_extrusion_opacity.as_ref(), zoom)
52            } else {
53                Some([1.0, 1.0, 1.0, 1.0])
54            };
55        }
56        LayerPaint::FillExtrusion(paint) => {
57            let opacity = if paint.fill_extrusion_translucent == Some(true) {
58                &paint.fill_extrusion_opacity
59            } else {
60                &None
61            };
62            (&paint.fill_extrusion_color, opacity)
63        }
64        LayerPaint::Circle(paint) => {
65            if paint.stroke_color_per_feature().is_some() {
66                return None;
67            }
68            (&paint.circle_color, &paint.circle_opacity)
69        }
70        LayerPaint::Heatmap(_) => return Some([0.0, 0.0, 0.0, 1.0]),
71        _ => return None,
72    };
73    if color.as_ref().is_some_and(|v| !v.is_feature_constant())
74        || opacity.as_ref().is_some_and(|v| !v.is_feature_constant())
75    {
76        return None;
77    }
78    let color = color
79        .as_ref()
80        .and_then(|v| v.evaluate_at_zoom(zoom))
81        .unwrap_or_default();
82    let opacity = opacity
83        .as_ref()
84        .and_then(|v| v.evaluate_at_zoom(zoom))
85        .unwrap_or(1.0)
86        .clamp(0.0, 1.0);
87    Some([color.r, color.g, color.b, color.a * opacity])
88}
