14pub(super) fn can_release_cpu_geometry(
15    style_layer: &StyleLayer,
16    bucket: &AvailableVectorLayerBucket,
17    has_dem_source: bool,
18) -> bool {
19    let z = u8::from(bucket.coords.z);
20    let edge_y = (1_i32 << z.min(30)) - 1;
21    let is_polar_row = bucket.coords.y <= 0 || bucket.coords.y >= edge_y;
22    let allow_row = z > 0 && !(has_dem_source && is_polar_row && style_layer.type_ == "fill");
23    allow_row
24        && (!has_dem_source || crate::vector::structures::kind(style_layer).is_none())
25        && style_layer
26            .paint
27            .as_ref()
28            .and_then(crate::vector::pattern::per_feature_pattern)
29            .is_none()
30        && !(style_layer.type_ == "circle" && !bucket.feature_sort_keys.is_empty())
31}

33/// Finds raw sibling CPU vertices within the same upload lane (`pending.vector` when
34/// `in_pending_lane` is true, or committed `layers.layers` otherwise) so a replacement attempt
35/// never borrows vertices from an older committed generation.
36pub(super) fn find_shared_fallback_vertices<'a>(
37    bucket: &AvailableVectorLayerBucket,
38    layers: &'a VectorLayerBucketComponent,
39    pending: &'a LayerReplacements,
40    in_pending_lane: bool,
41) -> Option<&'a [crate::render::ShaderVertex]> {
42    let shared_key = bucket
43        .buffer
44        .shared_geometry_key()
45        .filter(|_| !bucket.buffer.has_cpu_vertices())?;
46    if in_pending_lane {
47        pending.vector.iter().find(|sib| {
48            sib.buffer.shared_geometry_key() == Some(shared_key)
49                && !sib.buffer.buffer.vertices.is_empty()
50        })
51    } else {
52        layers.layers.iter().find_map(|sib| match sib {
53            VectorLayerBucket::AvailableLayer(sib)
54                if sib.buffer.shared_geometry_key() == Some(shared_key)
55                    && !sib.buffer.buffer.vertices.is_empty() =>
56            {
57                Some(sib)
58            }
59            _ => None,
60        })
61    }
62    .map(|sib| sib.buffer.buffer.vertices.as_slice())
63}

65/// Returns `true` when `bucket` needs shared vertices that are absent from CPU memory, lane
66/// fallback siblings, and `buffer_pool`'s resident shared vertex slabs.
67pub(super) fn missing_shared_vertices(
68    buffer_pool: &VectorBufferPool,
69    coords: WorldTileCoords,
70    style_layer: &StyleLayer,
71    bucket: &AvailableVectorLayerBucket,
72    fallback_vertices: Option<&[crate::render::ShaderVertex]>,
73) -> bool {
74    bucket.buffer.usable_indices > 0
75        && !bucket.buffer.has_cpu_vertices()
76        && fallback_vertices.is_none()
77        && bucket
78            .buffer
79            .shared_geometry_key()
80            .is_some_and(|key| !buffer_pool.has_shared_vertex_slot(coords, key, &style_layer.type_))
81}

83pub(super) fn flush_safe_cpu_releases(
84    tiles: &mut Tiles,
85    buffer_pool: &VectorBufferPool,
86    coords: WorldTileCoords,
87    releases: &[(String, Option<u64>, bool)],
88) {
89    for (layer_id, shared_key, release_colors) in releases {
90        let sibling_pending = shared_key.is_some_and(|key| {
91            tiles
92                .query::<(&VectorLayerBucketComponent, &LayerReplacements)>(coords)
93                .is_some_and(|(layers, pending)| {
94                    pending.vector.iter().any(|sib| {
95                        sib.buffer.shared_geometry_key() == Some(key)
96                            && sib.buffer.has_cpu_indices()
97                    }) || layers.layers.iter().any(|sib| match sib {
98                        VectorLayerBucket::AvailableLayer(sib) => {
99                            sib.buffer.shared_geometry_key() == Some(key)
100                                && sib.buffer.has_cpu_indices()
101                                && !buffer_pool.has_layer(coords, &sib.style_layer_id)
102                        }
103                        _ => false,
104                    })
105                })
106        });
107        if !sibling_pending {
108            release_committed_cpu_geometry(tiles, coords, layer_id, *release_colors);
109        }
110    }
