21/// Returns `true` when `layer` is a non-symbol vector layer whose GPU feature paint depends on
22/// per-feature properties or runtime `feature-state`, requiring retained [`BucketFeatureEntry`]s.
23pub(crate) fn layer_retains_feature_paint(layer: &StyleLayer) -> bool {
24    if layer_paint_reads_feature_state(layer) {
25        return true;
26    }
27    let Some(paint) = layer.paint.as_ref() else {
28        return false;
29    };
30    layer_paint_varies_by_feature(paint)
31}

33/// Returns `true` when any GPU-evaluated paint property in `paint` varies by feature.
34pub(crate) fn layer_paint_varies_by_feature(paint: &LayerPaint) -> bool {
35    if paint
36        .opacity()
37        .as_ref()
38        .is_some_and(|prop| !prop.is_feature_constant())
39        || per_feature_pattern(paint).is_some()
40    {
41        return true;
42    }
43    match paint {
44        LayerPaint::Fill(fill) => {
45            fill.fill_color
46                .as_ref()
47                .is_some_and(|prop| !prop.is_feature_constant())
48                || fill
49                    .outline_color()
50                    .as_ref()
51                    .is_some_and(|prop| !prop.is_feature_constant())
52        }
53        LayerPaint::Line(line) => line
54            .line_color
55            .as_ref()
56            .is_some_and(|prop| !prop.is_feature_constant()),
57        LayerPaint::Circle(circle) => {
58            circle
59                .circle_color
60                .as_ref()
61                .is_some_and(|prop| !prop.is_feature_constant())
62                || circle.stroke_color_per_feature().is_some()
63        }
64        LayerPaint::FillExtrusion(ext) => ext
65            .fill_extrusion_color
66            .as_ref()
67            .is_some_and(|prop| !prop.is_feature_constant()),
68        _ => false,
69    }
70}

72/// Distinguishes how a bucket feature entry computes its RGBA metadata vector.
73#[derive(Clone, Copy, Debug, PartialEq)]
74pub enum BucketEntryKind {
75    /// Standard polygon fill, extrusion building part, circle quad, or heatmap point.
76    Feature,
77    /// Stroked path carrying its tessellated tile-space length for `line-gradient`.
78    Line {
79        /// Maximum contour path length in tile units recorded during stroke tessellation.
80        length: f32,
81    },
82    /// One-pixel `fill-outline-color` stroke emitted immediately after a polygon's fill.
83    FillOutline,
84}

86/// Source feature identity and attributes for one entry of `AvailableVectorLayerBucket::feature_indices`.
87#[derive(Clone, Debug)]
88pub struct BucketFeatureEntry {
89    /// Raw MVT or numeric GeoJSON feature ID, if present.
90    pub raw_id: Option<u64>,
91    /// Shared feature properties (including internal `\u{1}state_id` and `\u{1}state:*` keys).
92    pub properties: Arc<FeatureProperties>,
93    /// Entry kind controlling whether `fill-outline-color` or `line-gradient` length applies.
94    pub kind: BucketEntryKind,
95}

109/// Re-evaluates per-entry RGBA feature metadata for `style_layer` at `style_zoom`, matching the
110/// exact encoding produced by [`crate::vector::tessellation::ZeroTessellator`].
111pub(crate) fn evaluate_bucket_colors(
112    style_layer: &StyleLayer,
113    entries: &[BucketFeatureEntry],
114    style_zoom: f64,
115) -> Vec<[f32; 4]> {
116    let Some(paint) = style_layer.paint.as_ref() else {
117        return vec![[0.0, 0.0, 0.0, 1.0]; entries.len()];
118    };
119    let opacity = paint.opacity();
120    let mut fallback_color = [0.0, 0.0, 0.0, 1.0];
121    let mut style_property = None;
122    let mut outline_property = None;
123    let mut pattern_property = None;
124    let mut line_gradient = false;
125    let mut circle_stroke_color = None;
126
127    match paint {
128        LayerPaint::Fill(fill) if fill.fill_pattern.is_some() => {
129            fallback_color = [1.0; 4];
130            pattern_property = per_feature_pattern(paint);
131        }
132        LayerPaint::Fill(fill) => {
133            style_property = fill.fill_color.as_ref();
134            outline_property = fill.outline_color();
135        }
136        LayerPaint::FillExtrusion(ext) if ext.fill_extrusion_pattern.is_some() => {
137            fallback_color = [1.0; 4];
138            pattern_property = per_feature_pattern(paint);
139        }
140        LayerPaint::FillExtrusion(ext) => {
141            style_property = ext.fill_extrusion_color.as_ref();
142        }
143        LayerPaint::Circle(circle) => {
144            style_property = circle.circle_color.as_ref();
145            circle_stroke_color = circle.stroke_color_per_feature();
146        }
147        LayerPaint::Line(line) => {
148            pattern_property = per_feature_pattern(paint);
149            style_property = line.line_color.as_ref();
150            line_gradient = line.line_gradient.is_some();
151        }
152        LayerPaint::Background(bg) => {
153            style_property = bg.background_color.as_ref();
154        }
155        _ => {}
156    }
157
158    let mut out = Vec::with_capacity(entries.len());
159    for entry in entries {
160        let props = entry.properties.as_ref();
161        let feat_opacity = opacity.as_ref().map_or(1.0, |prop| {
162            prop.evaluate_for(props, style_zoom)
163                .unwrap_or(1.0)
164                .clamp(0.0, 1.0)
165        });
166        match entry.kind {
167            BucketEntryKind::FillOutline => {
168                let color = outline_property
169                    .as_ref()
170                    .and_then(|prop| prop.evaluate_for(props, style_zoom))
171                    .map_or([0.0, 0.0, 0.0, 0.0], |c| {
172                        [c.r, c.g, c.b, c.a * feat_opacity]
173                    });
174                out.push(color);
175            }
176            BucketEntryKind::Feature | BucketEntryKind::Line { .. } => {
177                let mut color = style_property
178                    .and_then(|prop| prop.evaluate_for(props, style_zoom))
179                    .map_or(fallback_color, |c| [c.r, c.g, c.b, c.a]);
180                if line_gradient {
181                    let length = match entry.kind {
182                        BucketEntryKind::Line { length } => length,
183                        _ => 0.0,
184                    };
185                    color = [length, 0.0, 0.0, 1.0];
186                }
187                color[3] *= feat_opacity;
188                if let Some(prop) = &pattern_property {
189                    color[0] = pattern_key::pattern_value(
190                        prop.evaluate_for(props, style_zoom)
191                            .as_ref()
192                            .map(|name| name.0.as_str()),
193                    );
194                }
195                if let Some(stroke_prop) = circle_stroke_color {
196                    let stroke = stroke_prop
197                        .evaluate_for(props, style_zoom)
198                        .map_or([0.0, 0.0, 0.0, 1.0], |c| c.to_array());
199                    color = circle_pack_colors(color, stroke);
200                }
201                out.push(color);
202            }
203        }
204    }
205    out
