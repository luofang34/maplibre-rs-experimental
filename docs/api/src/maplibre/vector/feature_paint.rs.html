21/// Returns `true` when `layer` is a non-symbol vector layer whose GPU feature paint depends on
22/// per-feature properties or runtime `feature-state`, requiring retained [`BucketFeatureEntry`]s.
23pub(crate) fn layer_retains_feature_paint(layer: &StyleLayer) -> bool {
24    if layer_paint_reads_feature_state(layer) {
25        return true;
26    }
27    let Some(paint) = layer.paint.as_ref() else {
28        return false;
29    };
30    layer_paint_varies_by_feature(paint)
31}

33/// How much of each feature a layer's retained [`BucketFeatureEntry`]s keep.
34#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
35pub(crate) enum FeaturePaintRetention {
36    /// Every slot of a bucket shares one colour, so nothing is kept.
37    #[default]
38    None,
39    /// Slots differ only by [`BucketEntryKind`] (a fill beside its outline, a gradient line's
40    /// length), so entries keep the kind and share one empty property map.
41    SlotKinds,
42    /// Paint reads feature properties or `feature-state`, so entries keep both.
43    Full,
44}
45
46/// The retention `layer`'s buckets need for their paint to be refreshed in place.
47pub(crate) fn feature_paint_retention(layer: &StyleLayer) -> FeaturePaintRetention {
48    if layer_retains_feature_paint(layer) {
49        return FeaturePaintRetention::Full;
50    }
51    match layer.paint.as_ref() {
52        Some(LayerPaint::Fill(fill)) if fill.outline_color().is_some() => {
53            FeaturePaintRetention::SlotKinds
54        }
55        Some(LayerPaint::Line(line)) if line.line_gradient.is_some() => {
56            FeaturePaintRetention::SlotKinds
57        }
58        _ => FeaturePaintRetention::None,
59    }
60}

62/// Returns `true` when any GPU-evaluated paint property in `paint` varies by feature.
63pub(crate) fn layer_paint_varies_by_feature(paint: &LayerPaint) -> bool {
64    if paint
65        .opacity()
66        .as_ref()
67        .is_some_and(|prop| !prop.is_feature_constant())
68        || per_feature_pattern(paint).is_some()
69    {
70        return true;
71    }
72    match paint {
73        LayerPaint::Fill(fill) => {
74            fill.fill_color
75                .as_ref()
76                .is_some_and(|prop| !prop.is_feature_constant())
77                || fill
78                    .outline_color()
79                    .as_ref()
80                    .is_some_and(|prop| !prop.is_feature_constant())
81        }
82        LayerPaint::Line(line) => line
83            .line_color
84            .as_ref()
85            .is_some_and(|prop| !prop.is_feature_constant()),
86        LayerPaint::Circle(circle) => {
87            circle
88                .circle_color
89                .as_ref()
90                .is_some_and(|prop| !prop.is_feature_constant())
91                || circle.stroke_color_per_feature().is_some()
92        }
93        LayerPaint::FillExtrusion(ext) => ext
94            .fill_extrusion_color
95            .as_ref()
96            .is_some_and(|prop| !prop.is_feature_constant()),
97        _ => false,
98    }
99}

101/// Distinguishes how a bucket feature entry computes its RGBA metadata vector.
102#[derive(Clone, Copy, Debug, PartialEq)]
103pub enum BucketEntryKind {
104    /// Standard polygon fill, extrusion building part, circle quad, or heatmap point.
105    Feature,
106    /// Stroked path carrying its tessellated tile-space length for `line-gradient`.
107    Line {
108        /// Maximum contour path length in tile units recorded during stroke tessellation.
109        length: f32,
110    },
111    /// One-pixel `fill-outline-color` stroke emitted immediately after a polygon's fill.
112    FillOutline,
113}

115/// Source feature identity and attributes for one entry of `AvailableVectorLayerBucket::feature_indices`.
116#[derive(Clone, Debug)]
117pub struct BucketFeatureEntry {
118    /// Raw MVT or numeric GeoJSON feature ID, if present.
119    pub raw_id: Option<u64>,
120    /// Shared feature properties (including internal `\u{1}state_id` and `\u{1}state:*` keys).
121    pub properties: Arc<FeatureProperties>,
122    /// Entry kind controlling whether `fill-outline-color` or `line-gradient` length applies.
123    pub kind: BucketEntryKind,
124}

138/// Re-evaluates per-entry RGBA feature metadata for `style_layer` at `style_zoom`, matching the
139/// exact encoding produced by [`crate::vector::tessellation::ZeroTessellator`].
140pub(crate) fn evaluate_bucket_colors(
141    style_layer: &StyleLayer,
142    entries: &[BucketFeatureEntry],
143    style_zoom: f64,
144) -> Vec<[f32; 4]> {
145    let Some(paint) = style_layer.paint.as_ref() else {
146        return vec![[0.0, 0.0, 0.0, 1.0]; entries.len()];
147    };
148    let opacity = paint.opacity();
149    let mut fallback_color = [0.0, 0.0, 0.0, 1.0];
150    let mut style_property = None;
151    let mut outline_property = None;
152    let mut pattern_property = None;
153    let mut line_gradient = false;
154    let mut circle_stroke_color = None;
155
156    match paint {
157        LayerPaint::Fill(fill) if fill.fill_pattern.is_some() => {
158            fallback_color = [1.0; 4];
159            pattern_property = per_feature_pattern(paint);
160        }
161        LayerPaint::Fill(fill) => {
162            style_property = fill.fill_color.as_ref();
163            outline_property = fill.outline_color();
164        }
165        LayerPaint::FillExtrusion(ext) if ext.fill_extrusion_pattern.is_some() => {
166            fallback_color = [1.0; 4];
167            pattern_property = per_feature_pattern(paint);
168        }
169        LayerPaint::FillExtrusion(ext) => {
170            style_property = ext.fill_extrusion_color.as_ref();
171        }
172        LayerPaint::Circle(circle) => {
173            style_property = circle.circle_color.as_ref();
174            circle_stroke_color = circle.stroke_color_per_feature();
175        }
176        LayerPaint::Line(line) => {
177            pattern_property = per_feature_pattern(paint);
178            style_property = line.line_color.as_ref();
179            line_gradient = line.line_gradient.is_some();
180        }
181        LayerPaint::Background(bg) => {
182            style_property = bg.background_color.as_ref();
183        }
184        _ => {}
185    }
186
187    let mut out = Vec::with_capacity(entries.len());
188    for entry in entries {
189        let props = entry.properties.as_ref();
190        let feat_opacity = opacity.as_ref().map_or(1.0, |prop| {
191            prop.evaluate_for(props, style_zoom)
192                .unwrap_or(1.0)
193                .clamp(0.0, 1.0)
194        });
195        match entry.kind {
196            BucketEntryKind::FillOutline => {
197                let color = outline_property
198                    .as_ref()
199                    .and_then(|prop| prop.evaluate_for(props, style_zoom))
200                    .map_or([0.0, 0.0, 0.0, 0.0], |c| {
201                        [c.r, c.g, c.b, c.a * feat_opacity]
202                    });
203                out.push(color);
204            }
205            BucketEntryKind::Feature | BucketEntryKind::Line { .. } => {
206                let mut color = style_property
207                    .and_then(|prop| prop.evaluate_for(props, style_zoom))
208                    .map_or(fallback_color, |c| [c.r, c.g, c.b, c.a]);
209                if line_gradient {
210                    let length = match entry.kind {
211                        BucketEntryKind::Line { length } => length,
212                        _ => 0.0,
213                    };
214                    color = [length, 0.0, 0.0, 1.0];
215                }
216                color[3] *= feat_opacity;
217                if let Some(prop) = &pattern_property {
218                    color[0] = pattern_key::pattern_value(
219                        prop.evaluate_for(props, style_zoom)
220                            .as_ref()
221                            .map(|name| name.0.as_str()),
222                    );
223                }
224                if let Some(stroke_prop) = circle_stroke_color {
225                    let stroke = stroke_prop
226                        .evaluate_for(props, style_zoom)
227                        .map_or([0.0, 0.0, 0.0, 1.0], |c| c.to_array());
228                    color = circle_pack_colors(color, stroke);
229                }
230                out.push(color);
231            }
232        }
233    }
234    out
