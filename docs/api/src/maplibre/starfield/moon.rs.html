1//! The moon's disc, lit on the side the sun direction shows it, and its halo.
2//!
3//! The phase is geometry only: each pixel of the disc stands for a point on a sphere, lit where
4//! its normal faces the sun direction, so the lit share and the bright limb's orientation both
5//! follow from the sun and moon directions. The surface scatters as the moon's regolith does,
6//! so a full moon is as bright at its limb as at its centre, and saturates towards white well
7//! above any star. A halo around it falls with the square of the angle, as bright as the
8//! moon's magnitude at its phase. The dark side keeps a faint earthshine. With no sun
9//! direction set the whole disc is drawn lit, a fallback rather than a phase. The disc lies at
10//! the far plane like the sun's, and does not fade with the sky: the moon shows by day.
11

30/// The halo reaches this many disc radii from the moon's centre.
31const HALO_REACH: f32 = 16.0;
32/// A full moon's visual magnitude.
33const FULL_MOON_MAGNITUDE: f64 = -12.73;
34
35/// The moon's visual magnitude at phase angle `degrees`, the sun-moon-observer angle, as
36/// Allen's Astrophysical Quantities fits it: a full moon at zero, fainter towards new.
37pub fn magnitude_at_phase_angle(degrees: f64) -> f64 {
38    FULL_MOON_MAGNITUDE + 0.026 * degrees.abs() + 4e-9 * degrees.powi(4)
39}
40
41/// The moon's brightness relative to a full moon, 10^(-0.4 Δm), from the phase angle the sun
42/// and moon directions give, both in the same axes. With no sun the moon is drawn full.
43pub fn relative_brightness(sun: Option<Vector3<f64>>, moon: Vector3<f64>) -> f64 {
44    let Some(sun) = sun else {
45        return 1.0;
46    };
47    // The sun is far enough that the phase angle is the angle from the moon's direction to the
48    // sun's opposite.
49    let phase_angle = (-sun.normalize().dot(moon.normalize()))
50        .clamp(-1.0, 1.0)
51        .acos()
52        .to_degrees();
53    10_f64.powf(-0.4 * (magnitude_at_phase_angle(phase_angle) - FULL_MOON_MAGNITUDE))
54}

56/// The moon's brightness relative to a full moon at its phase, as [`relative_brightness`] gives
57/// it, times the share of sunlight the Earth's shadow lets reach the moon's centre.
58pub fn eclipsed_brightness(
59    sun: Option<Vector3<f64>>,
60    moon: Disc,
61    shadow: Option<EarthShadow>,
62) -> f64 {
63    let sunlit = shadow.map_or(1.0, |shadow| shadow.sunlit_share(moon.direction()));
64    relative_brightness(sun, moon.direction().globe_world()) * sunlit
65}

67/// What the moon's shaders take for the frame.
68#[repr(C)]
69#[derive(Clone, Copy, Debug, Pod, Zeroable)]
70struct MoonFrame {
71    /// Direction to the moon's centre in globe world axes, then the tangent of its radius.
72    direction_radius: [f32; 4],
73    /// Direction to the sun in globe world axes, then 1 when a sun lights the phase.
74    sun: [f32; 4],
75    /// The globe transition the moon fades in with, its brightness relative to a full moon,
76    /// and the halo's reach in disc radii.
77    transition_flux_reach: [f32; 4],
78    /// The Earth's shadow's axis in globe world axes, then its umbra's radius in radians.
79    shadow_umbra: [f32; 4],
80    /// The penumbra's radius in radians, the umbra's Danjon value, then 1 while a shadow is set.
81    penumbra_danjon: [f32; 4],
82}

84/// The moon's frame values on the GPU, and whether a moon is drawn this frame.
85#[derive(Default)]
86pub struct MoonGpu {
87    frame: Option<(wgpu::Buffer, wgpu::BindGroup)>,
88    drawn: bool,
89}
90
91impl MoonGpu {
92    /// The frame's bind group, when a moon is drawn this frame.
93    pub fn drawable(&self) -> Option<&wgpu::BindGroup> {
94        let (_, frame) = self.frame.as_ref()?;
95        self.drawn.then_some(frame)
96    }
97}

99/// Uploads the moon's frame values, when the host has set a moon and the map is a globe.
100pub(super) fn queue_system(
101    MapContext {
102        world,
103        style,
104        view_state,
105        renderer,
106        ..
107    }: &mut MapContext,
108) -> SystemResult {
109    let transition = style.projection.as_ref().map_or(0.0, |specification| {
110        specification
111            .projection_type
112            .globe_transition(view_state.zoom().value())
113    });
114    let Some(Initialized(MoonPipeline(pipeline))) =
115        world.resources.get::<Eventually<MoonPipeline>>()
116    else {
117        return Ok(());
118    };
119    let frame_layout = pipeline.get_bind_group_layout(1);
120    let sun_globe = sun_direction(world).map(|sun| sun.globe_world());
121    let sun = sun_globe.map_or([0.0; 4], |sun| {
122        [sun.x as f32, sun.y as f32, sun.z as f32, 1.0]
123    });
124    let discs = discs(world);
125    let (shadow_umbra, penumbra_danjon) =
126        discs.earth_shadow.map_or(([0.0; 4], [0.0; 4]), |shadow| {
127            let axis = shadow.axis().globe_world();
128            (
129                [
130                    axis.x as f32,
131                    axis.y as f32,
132                    axis.z as f32,
133                    shadow.umbra_radius_degrees().to_radians() as f32,
134                ],
135                [
136                    shadow.penumbra_radius_degrees().to_radians() as f32,
137                    shadow.danjon(),
138                    1.0,
139                    0.0,
140                ],
141            )
142        });
143    let frame = discs.moon.map(|moon| {
144        let direction = moon.direction().globe_world();
145        MoonFrame {
146            direction_radius: [
147                direction.x as f32,
148                direction.y as f32,
149                direction.z as f32,
150                moon.radius().degrees().to_radians().tan() as f32,
151            ],
152            sun,
153            transition_flux_reach: [
154                transition,
155                eclipsed_brightness(sun_globe, moon, discs.earth_shadow) as f32,
156                HALO_REACH,
157                0.0,
158            ],
159            shadow_umbra,
160            penumbra_danjon,
161        }
162    });
163    let gpu = world.resources.get_or_init_mut::<MoonGpu>();
164    gpu.drawn = frame.is_some() && transition > 0.0;
165    let Some(frame) = frame else {
166        return Ok(());
167    };
168    let (buffer, _) = gpu.frame.get_or_insert_with(|| {
169        let buffer = renderer.device.create_buffer(&wgpu::BufferDescriptor {
170            label: Some("moon frame"),
171            size: std::mem::size_of::<MoonFrame>() as u64,
172            usage: wgpu::BufferUsages::UNIFORM | wgpu::BufferUsages::COPY_DST,
173            mapped_at_creation: false,
174        });
175        let bind_group = renderer
176            .device
177            .create_bind_group(&wgpu::BindGroupDescriptor {
178                label: Some("moon frame"),
179                layout: &frame_layout,
180                entries: &[wgpu::BindGroupEntry {
181                    binding: 0,
182                    resource: buffer.as_entire_binding(),
183                }],
184            });
185        (buffer, bind_group)
186    });
187    renderer
188        .queue
189        .write_buffer(buffer, 0, bytemuck::bytes_of(&frame));
190    Ok(())
191}

193/// Draws the moon, if one is set; the translucent pass calls it with the sun, before its items.
194pub fn draw<'w>(pass: &mut TrackedRenderPass<'w>, world: &'w World) {
195    let Some((Initialized(MoonPipeline(pipeline)), Initialized(projection), gpu)) =
196        world.resources.query::<(
197            &Eventually<MoonPipeline>,
198            &Eventually<ProjectionGpuResources>,
199            &MoonGpu,
200        )>()
201    else {
202        return;
203    };
204    let Some(frame) = gpu.drawable() else {
205        return;
206    };
207    pass.set_pipeline(pipeline);
208    pass.set_bind_group(0, projection.bind_group(), &[]);
209    pass.set_bind_group(1, frame, &[]);
210    pass.draw(0..6, 0..1);
211}
