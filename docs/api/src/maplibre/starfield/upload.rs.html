17/// The magnitude drawn at full brightness with the reference radius.
18const REFERENCE_MAGNITUDE: f32 = 1.0;
19/// Radius in logical pixels of a star of the reference magnitude is this plus one.
20const RADIUS_SCALE: f32 = 1.5;
21const MIN_RADIUS: f32 = 1.0;
22const MAX_RADIUS: f32 = 4.0;
23/// The brightness a star saturates at: short of white, so the moon outshines every star.
24pub const STAR_PEAK_BRIGHTNESS: f32 = 0.75;
25
26/// One star as the shader draws it.
27#[repr(C)]
28#[derive(Clone, Copy, Debug, Pod, Zeroable, PartialEq)]
29pub struct StarInstance {
30    /// Direction in globe world axes, then the radius in logical pixels.
31    pub direction_radius: [f32; 4],
32    /// sRGB colour, then the brightness in 0..=1.
33    pub color_brightness: [f32; 4],
34}
35
36/// How a star is drawn: its flux relative to the reference magnitude, 10^(-0.4 Δm), sets both
37/// its brightness, which saturates at the reference at [`STAR_PEAK_BRIGHTNESS`], and its size,
38/// which keeps growing for the brightest stars until it is capped.
39pub fn star_instance(star: &Star) -> StarInstance {
40    let direction = star.direction().globe_world();
41    let flux = 10_f32.powf(-0.4 * (star.magnitude() - REFERENCE_MAGNITUDE));
42    let amplitude = flux.sqrt();
43    let radius = (MIN_RADIUS + RADIUS_SCALE * amplitude).clamp(MIN_RADIUS, MAX_RADIUS);
44    let [red, green, blue] = star.color();
45    StarInstance {
46        direction_radius: [
47            direction.x as f32,
48            direction.y as f32,
49            direction.z as f32,
50            radius,
51        ],
52        color_brightness: [red, green, blue, amplitude.min(1.0) * STAR_PEAK_BRIGHTNESS],
53    }
54}

56/// What the star shader needs of the frame: where the view looks and the sky it looks through.
57#[repr(C)]
58#[derive(Clone, Copy, Debug, Pod, Zeroable)]
59struct StarfieldFrame {
60    /// The viewport in physical pixels, physical pixels per logical one, and the globe transition.
61    viewport_ratio_transition: [f32; 4],
62    /// Columns of the rotation from globe world axes into view axes.
63    view_rotation: [[f32; 4]; 3],
64    /// The atmosphere's inputs, as [`AtmosphereLayerMetadata::sky_inputs`] gives them.
65    sky: [[f32; 4]; 3],
66    /// Direction to the moon in globe world axes, then its brightness relative to a full moon;
67    /// all zero without a moon.
68    moon: [f32; 4],
69}
70
71/// The frame's view rotation and sky, with no sky where the atmosphere is not drawn or the
72/// globe camera cannot be built.
73fn view_and_sky(
74    style: &crate::style::Style,
75    view_state: &crate::render::view_state::ViewState,
76    world: &crate::tcs::world::World,
77) -> ([[f32; 4]; 3], [[f32; 4]; 3]) {
78    let disabled = AtmosphereLayerMetadata::disabled().sky_inputs();
79    let Ok(camera) = crate::render::projection::globe_camera_for_view(view_state) else {
80        return (
81            [
82                [1.0, 0.0, 0.0, 0.0],
83                [0.0, 1.0, 0.0, 0.0],
84                [0.0, 0.0, 1.0, 0.0],
85            ],
86            disabled,
87        );
88    };
89    let view = camera.view();
90    let column = |axis: cgmath::Vector4<f64>| {
91        let axis = axis.truncate().normalize();
92        [axis.x as f32, axis.y as f32, axis.z as f32, 0.0]
93    };
94    let rotation = [column(view.x), column(view.y), column(view.z)];
95    let blend = crate::background::queue_system::atmosphere_blend(style, view_state, world);
96    if blend <= 0.0 {
97        return (rotation, disabled);
98    }
99    let light = style.light.clone().unwrap_or_default();
100    let sun = crate::celestial::sun_direction(world);
101    let sky =
102        AtmosphereLayerMetadata::from_view(&camera, &light, sun, view_state.zoom().value(), blend)
103            .map_or(disabled, |metadata| metadata.sky_inputs());
104    (rotation, sky)
105}

107/// The stars on the GPU and the frame's viewport, pixel ratio and globe transition.
108#[derive(Default)]
109pub struct StarfieldGpu {
110    instances: Option<wgpu::Buffer>,
111    count: u32,
112    generation: Option<u64>,
113    frame: Option<(wgpu::Buffer, wgpu::BindGroup)>,
114    visible: bool,
115}
116
117impl StarfieldGpu {
118    /// The instance buffer and frame bind group, when there are stars to draw this frame.
119    pub fn drawable(&self) -> Option<(&wgpu::Buffer, &wgpu::BindGroup)> {
120        let instances = self.instances.as_ref()?;
121        let (_, frame) = self.frame.as_ref()?;
122        (self.visible && self.count > 0).then_some((instances, frame))
123    }
124
125    /// Number of stars uploaded.
126    pub fn count(&self) -> u32 {
127        self.count
128    }
129}

131/// Uploads the stars when they changed and the frame's values; stars show only on a globe.
132pub(super) fn queue_system(
133    MapContext {
134        world,
135        style,
136        view_state,
137        renderer,
138        ..
139    }: &mut MapContext,
140) -> SystemResult {
141    let transition = style.projection.as_ref().map_or(0.0, |specification| {
142        specification
143            .projection_type
144            .globe_transition(view_state.zoom().value())
145    });
146    let Some(Initialized(StarfieldPipeline(pipeline))) =
147        world.resources.get::<Eventually<StarfieldPipeline>>()
148    else {
149        return Ok(());
150    };
151    let frame_layout = pipeline.get_bind_group_layout(1);
152    let starfield = world.resources.get::<Starfield>().cloned();
153    let pixel_ratio = world
154        .resources
155        .get::<crate::sdf::provided::DisplayPixelRatio>()
156        .map_or(1.0, |ratio| ratio.0);
157    let size = renderer.state().surface().size();
158    let (view_rotation, sky) = view_and_sky(style, view_state, world);
159    let discs = crate::celestial::discs(world);
160    let moon = discs.moon.map_or([0.0; 4], |moon| {
161        let direction = moon.direction().globe_world();
162        let sun = crate::celestial::sun_direction(world).map(|sun| sun.globe_world());
163        let brightness = super::moon::eclipsed_brightness(sun, moon, discs.earth_shadow);
164        [
165            direction.x as f32,
166            direction.y as f32,
167            direction.z as f32,
168            brightness as f32,
169        ]
170    });
171    let gpu = world.resources.get_or_init_mut::<StarfieldGpu>();
172    gpu.visible = transition > 0.0;
173    if let Some(starfield) = starfield.filter(|field| Some(field.generation()) != gpu.generation) {
174        let instances: Vec<StarInstance> = starfield.stars().iter().map(star_instance).collect();
175        gpu.count = u32::try_from(instances.len()).unwrap_or(u32::MAX);
176        gpu.instances = (!instances.is_empty()).then(|| {
177            renderer.queue.create_buffer_init(
178                &renderer.device,
179                &wgpu::util::BufferInitDescriptor {
180                    label: Some("starfield instances"),
181                    contents: bytemuck::cast_slice(&instances),
182                    usage: wgpu::BufferUsages::VERTEX,
183                },
184            )
185        });
186        gpu.generation = Some(starfield.generation());
187    }
188    let (buffer, _) = gpu.frame.get_or_insert_with(|| {
189        let buffer = renderer.device.create_buffer(&wgpu::BufferDescriptor {
190            label: Some("starfield frame"),
191            size: std::mem::size_of::<StarfieldFrame>() as u64,
192            usage: wgpu::BufferUsages::UNIFORM | wgpu::BufferUsages::COPY_DST,
193            mapped_at_creation: false,
194        });
195        let bind_group = renderer
196            .device
197            .create_bind_group(&wgpu::BindGroupDescriptor {
198                label: Some("starfield frame"),
199                layout: &frame_layout,
200                entries: &[wgpu::BindGroupEntry {
201                    binding: 0,
202                    resource: buffer.as_entire_binding(),
203                }],
204            });
205        (buffer, bind_group)
206    });
207    let frame = StarfieldFrame {
208        viewport_ratio_transition: [
209            size.width() as f32,
210            size.height() as f32,
211            pixel_ratio,
212            transition,
213        ],
214        view_rotation,
215        sky,
216        moon,
217    };
218    renderer
219        .queue
220        .write_buffer(buffer, 0, bytemuck::bytes_of(&frame));
221    Ok(())
222}
