1//! A copy of the frame's depth after the main pass, which the sun's glare reads to learn how

12};

14/// The draw graph's node copying the depth, between the main and translucent passes.
15pub const SUN_DEPTH_COPY: &str = "sun_depth_copy";
16
17/// The copied depth and the bind group through which the sun's shader reads it.
18pub struct SunDepth {
19    texture: wgpu::Texture,
20    view: wgpu::TextureView,
21    samples: u32,
22    /// Bind group 2 of the sun's pipeline.
23    pub binding: wgpu::BindGroup,
24    copy: DepthCopyPipeline,
25}
26
27impl SunDepth {
28    /// The bind group layout of the copied depth, as the sun's vertex shader reads it.
29    pub fn layout_entries() -> [wgpu::BindGroupLayoutEntry; 1] {
30        [wgpu::BindGroupLayoutEntry {
31            binding: 0,
32            visibility: wgpu::ShaderStages::VERTEX,
33            ty: wgpu::BindingType::Texture {
34                // An unfiltered float binding preserves raw depth reads on GLSL backends.
35                sample_type: wgpu::TextureSampleType::Float { filterable: false },
36                view_dimension: wgpu::TextureViewDimension::D2,
37                multisampled: false,
38            },
39            count: None,
40        }]
41    }
42
43    /// A copy target of `size` for a depth texture of `samples` samples per pixel.
44    pub fn new(
45        device: &wgpu::Device,
46        size: wgpu::Extent3d,
47        samples: u32,
48        layout: &wgpu::BindGroupLayout,
49    ) -> Self {
50        let texture = device.create_texture(&wgpu::TextureDescriptor {
51            label: Some("sun depth"),
52            size,
53            mip_level_count: 1,
54            sample_count: 1,
55            dimension: wgpu::TextureDimension::D2,
56            format: DepthCopyPipeline::TARGET_FORMAT,
57            usage: wgpu::TextureUsages::RENDER_ATTACHMENT | wgpu::TextureUsages::TEXTURE_BINDING,
58            view_formats: &[],
59        });
60        let view = texture.create_view(&Default::default());
61        let binding = device.create_bind_group(&wgpu::BindGroupDescriptor {
62            label: Some("sun depth"),
63            layout,
64            entries: &[wgpu::BindGroupEntry {
65                binding: 0,
66                resource: wgpu::BindingResource::TextureView(&view),
67            }],
68        });
69        Self {
70            texture,
71            view,
72            samples,
73            binding,
74            copy: DepthCopyPipeline::new(device, samples),
75        }
76    }
77}
78
79impl HasChanged for SunDepth {
80    type Criteria = (u32, u32, u32);
81
82    fn has_changed(&self, criteria: &Self::Criteria) -> bool {
83        (self.texture.width(), self.texture.height(), self.samples) != *criteria
84    }
85}

87/// Copies the depth the main pass left, while a sun is drawn.
88pub struct SunDepthCopyNode;
89
90impl Node for SunDepthCopyNode {
91    fn run(
92        &self,
93        _graph: &mut RenderGraphContext,
94        render_context: &mut RenderContext,
95        state: &RenderResources,
96        world: &World,
97    ) -> Result<(), NodeRunError> {
98        let drawn = world
99            .resources
100            .get::<super::SunGpu>()
101            .is_some_and(|gpu| gpu.drawable().is_some());
102        let (Some(Initialized(depth)), Initialized(source), true) = (
103            world.resources.get::<Eventually<SunDepth>>(),
104            &state.depth_texture,
105            drawn,
106        ) else {
107            return Ok(());
108        };
109        depth.copy.copy(
110            render_context.device,
111            &mut render_context.command_encoder,
112            &source.texture,
113            &depth.view,
114        );
115        Ok(())
116    }
117}
