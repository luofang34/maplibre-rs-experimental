10/// The jumps made before the renderer is ready, as one: a later jump's fields replace an
11/// earlier one's, and what neither names is kept.
12pub(super) fn merged(earlier: Option<CameraOptions>, later: CameraOptions) -> CameraOptions {
13    let earlier = earlier.unwrap_or_default();
14    CameraOptions {
15        center: later.center.or(earlier.center),
16        zoom: later.zoom.or(earlier.zoom),
17        bearing: later.bearing.or(earlier.bearing),
18        pitch: later.pitch.or(earlier.pitch),
19        roll: later.roll.or(earlier.roll),
20        elevation: later.elevation.or(earlier.elevation),
21    }
22}

24/// The view a map starts rendering with: the one retained across a reset, or the style's
25/// initial camera, moved by the jump made while the renderer was not ready.
26pub(super) fn starting_view(
27    style: &Style,
28    retained: Option<&ViewState>,
29    window_size: PhysicalSize,
30    max_pitch: cgmath::Deg<f64>,
31    jump: Option<CameraOptions>,
32) -> Result<ViewState, NavigationError> {
33    let mut view = match retained {
34        Some(view) => {
35            let mut view = view.clone();
36            view.clear_external_view();
37            view.set_max_pitch(max_pitch);
38            view
39        }
40        None => initial_view_state(style, window_size, max_pitch),
41    };
42    if let Some(jump) = jump {
43        view.jump_to(jump)?;
44    }
45    Ok(view)
46}
47
48fn initial_view_state(
49    style: &Style,
50    window_size: PhysicalSize,
51    max_pitch: cgmath::Deg<f64>,
52) -> ViewState {
53    let center = style.center.unwrap_or_default();
54    let zoom = style.zoom.map(Zoom::new).unwrap_or_default();
55    let mut view = ViewState::new(
56        window_size,
57        WorldCoords::from_lat_lon(LatLon::new(center[1], center[0]), zoom),
58        zoom,
59        cgmath::Deg(style.pitch.unwrap_or_default()),
60        cgmath::Rad(0.6435011087932844),
61    );
62    view.set_max_pitch(max_pitch);
63    view.camera_mut()
64        .set_pitch(cgmath::Deg(style.pitch.unwrap_or_default()));
65    view.camera_mut()
66        .set_bearing(cgmath::Deg(style.bearing.unwrap_or_default()));
67    view.camera_mut()
68        .set_roll(cgmath::Deg(style.roll.unwrap_or_default()));
69    view.take_style_center(style);
70    view
71}
