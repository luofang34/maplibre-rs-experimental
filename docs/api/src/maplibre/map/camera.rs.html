17/// The jumps made before the renderer is ready, as one: a later jump's fields replace an
18/// earlier one's, and what neither names is kept.
19pub(super) fn merged(earlier: Option<CameraOptions>, later: CameraOptions) -> CameraOptions {
20    let earlier = earlier.unwrap_or_default();
21    CameraOptions {
22        center: later.center.or(earlier.center),
23        zoom: later.zoom.or(earlier.zoom),
24        bearing: later.bearing.or(earlier.bearing),
25        pitch: later.pitch.or(earlier.pitch),
26        roll: later.roll.or(earlier.roll),
27        elevation: later.elevation.or(earlier.elevation),
28    }
29}

31/// The view a map starts rendering with: the one retained across a reset, or the style's
32/// initial camera, moved by the jump made while the renderer was not ready.
33pub(super) fn starting_view(
34    style: &Style,
35    retained: Option<&ViewState>,
36    window_size: PhysicalSize,
37    max_pitch: cgmath::Deg<f64>,
38    camera_bounds: CameraBounds,
39    jump: Option<CameraOptions>,
40) -> Result<ViewState, NavigationError> {
41    let mut view = match retained {
42        Some(view) => {
43            let mut view = view.clone();
44            view.clear_external_view();
45            view.set_max_pitch(max_pitch);
46            view.set_camera_bounds(camera_bounds);
47            view
48        }
49        None => {
50            let mut view = initial_view_state(style, window_size, max_pitch);
51            view.set_camera_bounds(camera_bounds);
52            view
53        }
54    };
55    if let Some(jump) = jump {
56        view.jump_to(jump)?;
57    }
58    Ok(view)
59}

61/// Returns `style.center` as [`LatLon`] with latitude clamped to [`MERCATOR_LATITUDE_LIMIT`].
62pub(crate) fn clamped_style_center(style: &Style) -> LatLon {
63    let [lon, lat] = style.center.unwrap_or_default();
64    LatLon::new(
65        lat.clamp(-MERCATOR_LATITUDE_LIMIT, MERCATOR_LATITUDE_LIMIT),
66        lon,
67    )
68}

70fn initial_view_state(
71    style: &Style,
72    window_size: PhysicalSize,
73    max_pitch: cgmath::Deg<f64>,
74) -> ViewState {
75    let zoom = style.zoom.map(Zoom::new).unwrap_or_default();
76    let mut view = ViewState::new(
77        window_size,
78        WorldCoords::from_lat_lon(clamped_style_center(style), zoom),
79        zoom,
80        cgmath::Deg(style.pitch.unwrap_or_default()),
81        cgmath::Rad(0.6435011087932844),
82    );
83    view.set_max_pitch(max_pitch);
84    view.camera_mut()
85        .set_pitch(cgmath::Deg(style.pitch.unwrap_or_default()));
86    view.camera_mut()
87        .set_bearing(cgmath::Deg(style.bearing.unwrap_or_default()));
88    view.camera_mut()
89        .set_roll(cgmath::Deg(style.roll.unwrap_or_default()));
90    view.take_style_center(style);
91    view
92}

94impl<E: Environment> Map<E>
95where

98    /// Moves the camera to `options` along `curve`, as
99    /// [`crate::context::MapContext::fly_to`]. Frames advance a move, so it needs an
100    /// initialized renderer.
101    pub fn fly_to(
102        &mut self,
103        options: CameraOptions,
104        curve: FlightCurve,
105    ) -> Result<FlightId, MapError> {
106        Ok(self.context_mut()?.fly_to(options, curve))
107    }
108
109    /// How the move `id` stands, as [`crate::context::MapContext::flight_outcome`]. A map
110    /// whose renderer is not ready keeps no moves, so every move reads as
111    /// [`FlightOutcome::Superseded`].
112    pub fn flight_outcome(&self, id: FlightId) -> FlightOutcome {
113        self.context().map_or(FlightOutcome::Superseded, |context| {
114            context.flight_outcome(id)
115        })
116    }
117
118    /// Stops the move under way where the camera is, as
119    /// [`crate::context::MapContext::stop_flight`].
120    pub fn stop_flight(&mut self) {
121        if let Ok(context) = self.context_mut() {
122            context.stop_flight();
123        }
124    }
125
126    /// Returns the active zoom and geographic center bounds enforced on the camera.
127    pub fn camera_bounds(&self) -> CameraBounds {
128        match &self.map_context {
129            CurrentMapContext::Ready(context) => context.camera_bounds(),
130            CurrentMapContext::Pending(_) => self.camera_bounds,
131        }
132    }
133
134    /// Replaces the zoom and geographic center bounds, before or after renderer initialization.
135    pub fn set_camera_bounds(&mut self, bounds: CameraBounds) {
136        self.camera_bounds = bounds;
137        if let CurrentMapContext::Ready(context) = &mut self.map_context {
138            context.set_camera_bounds(bounds);
139        }
140    }
141
142    /// Sets or clears the minimum continuous zoom level enforced on the camera.
143    pub fn set_min_zoom(&mut self, min_zoom: Option<f64>) {
144        self.camera_bounds.min_zoom = min_zoom.filter(|v| v.is_finite());
145        if let CurrentMapContext::Ready(context) = &mut self.map_context {
146            context.set_min_zoom(min_zoom);
147        }
148    }
149
150    /// Sets or clears the maximum continuous zoom level enforced on the camera.
151    pub fn set_max_zoom(&mut self, max_zoom: Option<f64>) {
152        self.camera_bounds.max_zoom = max_zoom.filter(|v| v.is_finite());
153        if let CurrentMapContext::Ready(context) = &mut self.map_context {
154            context.set_max_zoom(max_zoom);
155        }
156    }
157
158    /// Sets or clears the geographic bounding box restricting the camera's center.
159    pub fn set_max_bounds(&mut self, max_bounds: Option<LatLonBounds>) {
160        self.camera_bounds.max_bounds = max_bounds;
161        if let CurrentMapContext::Ready(context) = &mut self.map_context {
162            context.set_max_bounds(max_bounds);
163        }
164    }
165
166    /// Computes the center and maximum zoom that frame `bounds` within `padding`, before or after
167    /// renderer initialization.
168    pub fn camera_for_bounds(
169        &self,
170        bounds: LatLonBounds,
171        padding: EdgePadding,
172        bearing: Option<cgmath::Deg<f64>>,
173        pitch: Option<cgmath::Deg<f64>>,
174    ) -> Result<CameraFit, MapError> {
175        match &self.map_context {
176            CurrentMapContext::Ready(context) => {
177                Ok(context.camera_for_bounds(bounds, padding, bearing, pitch))
178            }
179            CurrentMapContext::Pending(pending) => {
180                let window_size = self.window.size();
181                let mut temp_view = starting_view(
182                    &pending.style,
183                    pending.view_state.as_ref(),
184                    window_size,
185                    self.max_pitch,
186                    self.camera_bounds,
187                    pending.jump,
188                )?;
189                temp_view.resize(window_size.to_logical(self.window.scale_factor()));
190                Ok(temp_view.camera_for_bounds(bounds, padding, bearing, pitch))
191            }
192        }
193    }
194
195    /// Moves the camera immediately to frame `bounds` within `padding`, before or after renderer
196    /// initialization.
197    pub fn fit_bounds(
198        &mut self,
199        bounds: LatLonBounds,
200        padding: EdgePadding,
201    ) -> Result<CameraFit, MapError> {
202        let window_size = self.window.size();
203        let scale_factor = self.window.scale_factor();
204        let max_pitch = self.max_pitch;
205        let camera_bounds = self.camera_bounds;
206        match &mut self.map_context {
207            CurrentMapContext::Ready(context) => Ok(context.fit_bounds(bounds, padding)?),
208            CurrentMapContext::Pending(pending) => {
209                let mut temp_view = starting_view(
210                    &pending.style,
211                    pending.view_state.as_ref(),
212                    window_size,
213                    max_pitch,
214                    camera_bounds,
215                    pending.jump,
216                )?;
217                temp_view.resize(window_size.to_logical(scale_factor));
218                let fit = temp_view.camera_for_bounds(bounds, padding, None, None);
219                pending.jump = Some(merged(pending.jump, fit.into()));
220                Ok(fit)
221            }
222        }
223    }
