31/// Configuration for spawning a [`HeadlessRuntimeThread`] around a live [`HeadlessMap`].
32#[derive(Clone)]
33pub struct HeadlessRuntimeConfig {
34    /// Physical width of the offscreen render target in pixels (must be non-zero).
35    pub width: u32,
36    /// Physical height of the offscreen render target in pixels (must be non-zero).
37    pub height: u32,
38    /// Physical-to-logical display pixel ratio (`<= 0.0` defaults to `1.0`).
39    pub pixel_ratio: f64,
40    /// Initial map style document.
41    pub style: Style,
42    /// Optional filesystem directory for HTTP tile caching.
43    pub cache_dir: Option<String>,
44    /// Optional maximum camera pitch in degrees (`None` preserves the default limit).
45    pub max_pitch_deg: Option<f64>,
46    /// Offscreen renderer settings (texture format, MSAA, buffer pool sizes).
47    pub renderer_settings: RendererSettings,
48    /// Number of Tokio worker threads used for asynchronous tile fetching and decoding.
49    pub worker_threads: usize,
50}

52impl std::fmt::Debug for HeadlessRuntimeConfig {

81/// Why a [`HeadlessRuntimeThread`] could not start.
82#[derive(Debug, Error)]
83pub enum HeadlessRuntimeError {
84    /// Spawning the dedicated map OS thread failed.
85    #[error("spawning headless runtime thread failed: {0}")]
86    ThreadSpawn(std::io::Error),
87    /// Creating the Tokio worker runtime failed.
88    #[error("creating headless Tokio runtime failed: {0}")]
89    Runtime(std::io::Error),
90    /// Offscreen GPU renderer creation failed.
91    #[error("headless renderer initialization failed: {0}")]
92    Renderer(String),
93    /// Initializing [`HeadlessMap`] plugins failed.
94    #[error("headless map initialization failed: {0}")]
95    Map(String),
96    /// Custom map initialization failed.
97    #[error("headless map initialization failed: {0}")]
98    Custom(String),
99    /// The worker thread exited before reporting initialization status.
100    #[error("headless runtime thread exited during startup")]
101    Aborted,
102}

107    }
108}

116#[derive(Default)]
117struct WorkerState {
118    woken: bool,
119    shutdown: bool,
120}
121
122struct CondvarWake {
123    pair: Arc<(Mutex<WorkerState>, Condvar)>,
124}
125
126impl SyncWakeHost for CondvarWake {
127    fn wake(&self) {
128        let (mutex, condvar) = &*self.pair;
129        let mut state = match mutex.lock() {
130            Ok(guard) => guard,
131            Err(poisoned) => poisoned.into_inner(),
132        };
133        state.woken = true;
134        drop(state);
135        condvar.notify_one();
136    }
137}

139/// Owns a [`HeadlessMap`] and its multi-threaded Tokio runtime on a dedicated OS thread,
140/// exposing a `Clone + Send + Sync` [`SyncHeadlessHandle`] usable from any thread (including
141/// JNI threads and C FFI callers).
142///
143/// Dropping or calling [`Self::shutdown`] wakes the worker thread, closes pending replies with
144/// [`super::MapClosed`], drops the map inside its Tokio runtime context, and joins the thread.
145pub struct HeadlessRuntimeThread {
146    handle: SyncHeadlessHandle,
147    signal: Arc<(Mutex<WorkerState>, Condvar)>,
148    join_handle: Option<JoinHandle<()>>,
149}
150
151impl HeadlessRuntimeThread {
152    /// Spawns a dedicated thread that initializes a [`HeadlessMap`] with standard rendering
153    /// plugins according to `config`.
154    pub fn spawn(config: HeadlessRuntimeConfig) -> Result<Self, HeadlessRuntimeError> {
155        let worker_threads = config.worker_threads.max(1);
156        Self::spawn_with_threads(worker_threads, move |runtime| {
157            let mut style = config.style;
158            let (kernel, renderer) = runtime.block_on(create_headless_renderer_with_settings(
159                config.width,
160                config.height,
161                config.cache_dir,
162                config.renderer_settings,
163            ))?;
164            runtime.block_on(resolve_tile_json_sources(
165                &mut style,
166                kernel.source_client(),
167            ));
168            let plugins = standard_headless_plugins();
169            let mut map = HeadlessMap::new(style, renderer, kernel, plugins)?;
170            if config.pixel_ratio.is_finite() && config.pixel_ratio > 0.0 {
171                map.set_pixel_ratio(config.pixel_ratio);
172            }
173            if let Some(max_pitch) = config.max_pitch_deg.filter(|deg| deg.is_finite()) {
174                map.set_max_pitch(Deg(max_pitch.clamp(0.0, 89.5)));
175            }
176            Ok(map)
177        })
178    }
179
180    /// Spawns a dedicated thread running a custom `build` closure inside a fresh Tokio runtime.
181    pub fn spawn_with(
182        build: impl FnOnce(&tokio::runtime::Runtime) -> Result<HeadlessMap, HeadlessRuntimeError>
183            + Send
184            + 'static,
185    ) -> Result<Self, HeadlessRuntimeError> {
186        Self::spawn_with_threads(2, build)
187    }
188
189    fn spawn_with_threads(
190        worker_threads: usize,
191        build: impl FnOnce(&tokio::runtime::Runtime) -> Result<HeadlessMap, HeadlessRuntimeError>
192            + Send
193            + 'static,
194    ) -> Result<Self, HeadlessRuntimeError> {
195        let signal = Arc::new((Mutex::new(WorkerState::default()), Condvar::new()));
196        let worker_signal = Arc::clone(&signal);
197        let (startup_tx, startup_rx) =
198            mpsc::sync_channel::<Result<SyncHeadlessHandle, HeadlessRuntimeError>>(1);
199
200        let join_handle = thread::Builder::new()
201            .name("maplibre-headless".to_owned())
202            .spawn(move || {
203                let runtime = match tokio::runtime::Builder::new_multi_thread()
204                    .worker_threads(worker_threads.max(1))
205                    .thread_name("maplibre-tile-worker")
206                    .enable_all()
207                    .build()
208                {
209                    Ok(rt) => rt,
210                    Err(err) => {
211                        let _ = startup_tx.send(Err(HeadlessRuntimeError::Runtime(err)));
212                        return;
213                    }
214                };
215                let _enter = runtime.enter();
216                let mut map = match build(&runtime) {
217                    Ok(map) => map,
218                    Err(err) => {
219                        let _ = startup_tx.send(Err(err));
220                        return;
221                    }
222                };
223
224                let commands = SyncHeadlessCommands::new();
225                let handle = commands.handle(CondvarWake {
226                    pair: Arc::clone(&worker_signal),
227                });
228                if startup_tx.send(Ok(handle)).is_err() {
229                    commands.close();
230                    drop(map);
231                    return;
232                }
233
234                run_worker_loop(&mut map, &commands, &worker_signal);
235                commands.close();
236                drop(map);
237            })
238            .map_err(HeadlessRuntimeError::ThreadSpawn)?;
239
240        match startup_rx.recv() {
241            Ok(Ok(handle)) => Ok(Self {
242                handle,
243                signal,
244                join_handle: Some(join_handle),
245            }),
246            Ok(Err(err)) => {
247                let _ = join_handle.join();
248                Err(err)
249            }
250            Err(_) => {
251                let _ = join_handle.join();
252                Err(HeadlessRuntimeError::Aborted)
253            }
254        }
255    }
256
257    /// Returns a `Clone + Send + Sync` handle to enqueue commands and watchers on this map.
258    pub fn handle(&self) -> SyncHeadlessHandle {
259        self.handle.clone()
260    }
261
262    /// Shuts down the dedicated map thread and waits for it to exit cleanly.
263    pub fn shutdown(mut self) {
264        self.stop_and_join();
265    }
266
267    fn stop_and_join(&mut self) {
268        {
269            let (mutex, condvar) = &*self.signal;
270            let mut state = match mutex.lock() {
271                Ok(guard) => guard,
272                Err(poisoned) => poisoned.into_inner(),
273            };
274            state.shutdown = true;
275            state.woken = true;
276            drop(state);
277            condvar.notify_all();
278        }
279        if let Some(join_handle) = self.join_handle.take() {
280            if join_handle.thread().id() != thread::current().id() {
281                let _ = join_handle.join();
282            }
283        }
284    }
285}

287impl Drop for HeadlessRuntimeThread {
288    fn drop(&mut self) {
289        self.stop_and_join();
290    }
291}

293fn run_worker_loop(
294    map: &mut HeadlessMap,
295    commands: &SyncHeadlessCommands,
296    signal: &(Mutex<WorkerState>, Condvar),
297) {
298    const WATCHER_POLL_INTERVAL: Duration = Duration::from_millis(8);
299    const FRAME_STEP: Duration = Duration::from_millis(16);
300
301    loop {
302        {
303            let (mutex, _) = signal;
304            let mut state = match mutex.lock() {
305                Ok(guard) => guard,
306                Err(poisoned) => poisoned.into_inner(),
307            };
308            if state.shutdown {
309                return;
310            }
311            state.woken = false;
312        }
313
314        commands.apply(map);
315
316        if commands.has_watchers() && (map.has_ready_work() || map.needs_redraw()) {
317            map.frame_input_mut().advance(FRAME_STEP);
318            let _ = map.run_frame();
319            commands.observe(map);
320        }
321
322        if commands.has_pending_commands() {
323            continue;
324        }
325
326        let (mutex, condvar) = signal;
327        let state = match mutex.lock() {
328            Ok(guard) => guard,
329            Err(poisoned) => poisoned.into_inner(),
330        };
331        if state.shutdown {
332            return;
333        }
334        if state.woken {
335            continue;
336        }
337        if commands.has_watchers() {
338            if map.has_ready_work() {
339                continue;
340            }
341            drop(condvar.wait_timeout(state, WATCHER_POLL_INTERVAL));
342        } else {
343            drop(condvar.wait(state));
344        }
345    }
346}
