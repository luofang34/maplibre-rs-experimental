34/// Ordered layers, named data sources and initial view settings from a style document.
35/// [`Self::validate`] checks layer rendering capabilities, not all style-spec constraints.
36#[derive(Serialize, Deserialize, Debug, Clone)]
37pub struct Style {
38    /// Style specification version, normally 8; deserialization does not validate the version.
39    pub version: u16,
40    /// Human-readable document name with no rendering effect.
41    #[serde(default, skip_serializing_if = "Option::is_none")]
42    pub name: Option<String>,
43    /// Application-defined JSON properties retained without changing rendering.
44    #[serde(default)]
45    pub metadata: HashMap<String, serde_json::Value>,
46    /// Data sources keyed by the identifiers referenced by layers and terrain.
47    #[serde(default)]
48    pub sources: HashMap<String, Source>,
49    /// Painter order from bottom to top; deserialization assigns zero-based layer indices.
50    #[serde(deserialize_with = "layer_order::deserialize_layers")]
51    pub layers: Vec<StyleLayer>,
52    /// URL template for font glyph ranges.
53    #[serde(default, skip_serializing_if = "Option::is_none")]
54    pub glyphs: Option<String>,
55    /// Sprite URL or named sprite sources.
56    #[serde(default, skip_serializing_if = "Option::is_none")]
57    pub sprite: Option<serde_json::Value>,
58    /// Initial `[longitude, latitude]` in degrees, before host camera overrides.
59    #[serde(default, skip_serializing_if = "Option::is_none")]
60    pub center: Option<[f64; 2]>,
61    /// Initial altitude in metres above sea level of the point the camera looks at, before
62    /// host overrides; terrain that holds the center to the ground takes its place.
63    #[serde(
64        default,
65        rename = "centerAltitude",
66        skip_serializing_if = "Option::is_none"
67    )]
68    pub center_altitude: Option<f64>,
69    /// Initial continuous camera zoom, before host overrides.
70    #[serde(default, skip_serializing_if = "Option::is_none")]
71    pub zoom: Option<f64>,
72    /// Initial clockwise rotation from north in degrees.
73    #[serde(default, skip_serializing_if = "Option::is_none")]
74    pub bearing: Option<f64>,
75    /// Initial camera tilt from the map normal in degrees, clamped by the camera pitch limit.
76    #[serde(default, skip_serializing_if = "Option::is_none")]
77    pub pitch: Option<f64>,
78    /// Initial roll in degrees, counterclockwise about the camera's line of sight.
79    #[serde(default, skip_serializing_if = "Option::is_none")]
80    pub roll: Option<f64>,
81    /// Map projection and its parameters; omission uses the renderer's default projection.
82    #[serde(default, skip_serializing_if = "Option::is_none")]
83    pub projection: Option<ProjectionSpecification>,
84    /// Lighting parameters used by style-driven shading.
85    #[serde(default, skip_serializing_if = "Option::is_none")]
86    pub light: Option<light::LightSpecification>,
87    /// Sky and atmosphere appearance; omission leaves the sky layer disabled.
88    #[serde(default, skip_serializing_if = "Option::is_none")]
89    pub sky: Option<sky::SkySpecification>,
90    /// Elevation source and vertical exaggeration; omission renders without terrain.
91    #[serde(default, skip_serializing_if = "Option::is_none")]
92    pub terrain: Option<terrain::TerrainSpecification>,
93    /// Defaults for the values `global-state` expressions read.
94    #[serde(default, skip_serializing_if = "HashMap::is_empty")]
95    pub state: HashMap<String, state::StateDeclaration>,
96    /// Values set at runtime; they override the defaults and travel with the style to workers.
97    #[serde(
98        default,
99        rename = "globalState",
100        skip_serializing_if = "HashMap::is_empty"
101    )]
102    pub global_state: HashMap<String, serde_json::Value>,
103    /// Layers as declared, kept while the layers in use have global state substituted. Public
104    /// only so a style can still be built with struct update syntax.
105    #[doc(hidden)]
106    #[serde(skip)]
107    pub state_templates: HashMap<String, state::StateTemplate>,
108    /// The light as declared, kept while the light in use has global state substituted.
109    #[doc(hidden)]
110    #[serde(skip)]
111    pub light_template: Option<light::LightSpecification>,
112    /// Images a host added by name, which `icon-image` can use next to the sprite's own.
113    #[serde(default, skip_serializing_if = "HashMap::is_empty")]
114    pub images: HashMap<String, StyleImage>,
115}

117/// An RGBA image a host added to the style, as GL JS `addImage` takes it.
118#[derive(Serialize, Deserialize, Debug, Clone, PartialEq)]
119pub struct StyleImage {
120    /// Width in pixels.
121    pub width: u32,
122    /// Height in pixels.
123    pub height: u32,
124    /// Straight-alpha RGBA bytes, `width * height * 4` of them.
125    pub data: Vec<u8>,
126    /// Image pixels per layout pixel; 1 unless the image is drawn smaller than it is.
127    #[serde(default = "one")]
128    pub pixel_ratio: f32,
129    /// Whether the alpha channel is a signed distance field that `icon-color` tints.
130    #[serde(default)]
131    pub sdf: bool,
132}
