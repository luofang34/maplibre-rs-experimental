10/// Owns the window factory, executor, worker transport and source loader selected by [`Environment`].
11/// Maps share the kernel with their plugins for the lifetime of those services.
12pub struct Kernel<E: Environment> {
13    map_window_config: E::MapWindowConfig,
14    apc: E::AsyncProcedureCall,
15    scheduler: E::Scheduler,
16    source_client: SourceClient<E::HttpClient>,
17}
18
19impl<E: Environment> Kernel<E> {
20    /// Window factory used when a map is created.
21    pub fn map_window_config(&self) -> &E::MapWindowConfig {
22        &self.map_window_config
23    }
24
25    /// Worker request and reply transport for this map.
26    pub fn apc(&self) -> &E::AsyncProcedureCall {
27        &self.apc
28    }
29
30    /// Executor for asynchronous tasks submitted by map systems.
31    pub fn scheduler(&self) -> &E::Scheduler {
32        &self.scheduler
33    }
34
35    /// Source loader used by the map's calling thread.
36    pub fn source_client(&self) -> &SourceClient<E::HttpClient> {
37        &self.source_client
38    }
39}
40
41/// A required host service was not supplied to [`KernelBuilder`].
42#[derive(Debug, Clone, Copy, PartialEq, Eq, thiserror::Error)]
43pub enum KernelBuildError {
44    /// No window configuration was supplied.
45    #[error("kernel requires a map window configuration")]
46    MissingWindowConfig,
47    /// No worker request and reply transport was supplied.
48    #[error("kernel requires an asynchronous procedure call transport")]
49    MissingAsyncProcedureCall,
50    /// No asynchronous executor was supplied.
51    #[error("kernel requires a scheduler")]
52    MissingScheduler,
53    /// No HTTP implementation was supplied.
54    #[error("kernel requires an HTTP client")]
55    MissingHttpClient,
56}
57
58/// Collects the four required host services before constructing a [`Kernel`].
59pub struct KernelBuilder<E: Environment> {
60    map_window_config: Option<E::MapWindowConfig>,
61    apc: Option<E::AsyncProcedureCall>,
62    scheduler: Option<E::Scheduler>,
63    http_client: Option<E::HttpClient>,
64}
65
66impl<E: Environment> Default for KernelBuilder<E> {
67    fn default() -> Self {
68        Self::new()
69    }
70}
71
72impl<E: Environment> KernelBuilder<E> {
73    /// Starts with no services; all four `with_*` methods are required before [`Self::build`].
74    pub fn new() -> Self {
75        Self {
76            scheduler: None,
77            apc: None,
78            http_client: None,
79            map_window_config: None,
80        }
81    }
82
83    /// Sets the window factory, replacing any configuration already supplied.
84    pub fn with_map_window_config(mut self, map_window_config: E::MapWindowConfig) -> Self {
85        self.map_window_config = Some(map_window_config);
86        self
87    }
88
89    /// Sets the executor, replacing any scheduler already supplied.
90    pub fn with_scheduler(mut self, scheduler: E::Scheduler) -> Self {
91        self.scheduler = Some(scheduler);
92        self
93    }
94
95    /// Sets the worker transport, replacing any transport already supplied.
96    pub fn with_apc(mut self, apc: E::AsyncProcedureCall) -> Self {
97        self.apc = Some(apc);
98        self
99    }
100
101    /// Sets the source HTTP implementation, replacing any client already supplied.
102    pub fn with_http_client(mut self, http_client: E::HttpClient) -> Self {
103        self.http_client = Some(http_client);
104        self
105    }
106
107    /// Consumes the services without creating a window or submitting work.
108    /// Returns the first missing dependency in window, transport, scheduler, HTTP order.
109    pub fn build(self) -> Result<Kernel<E>, KernelBuildError> {
110        Ok(Kernel {
111            map_window_config: self
112                .map_window_config
113                .ok_or(KernelBuildError::MissingWindowConfig)?,
114            apc: self
115                .apc
116                .ok_or(KernelBuildError::MissingAsyncProcedureCall)?,
117            scheduler: self.scheduler.ok_or(KernelBuildError::MissingScheduler)?,
118            source_client: SourceClient::new(HttpSourceClient::new(
119                self.http_client
120                    .ok_or(KernelBuildError::MissingHttpClient)?,
121            )),
122        })
123    }
124}
