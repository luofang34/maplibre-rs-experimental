46/// Whether terrain tiles hang skirts below their edges, as GL JS `terrainSkirtLength`: skirts
47/// hide the cracks between neighbouring tiles of different levels, and show through a
48/// translucent drape.
49#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
50pub enum TerrainSkirts {
51    /// Skirts drop by a fifth of a tile's circumference at the camera's zoom.
52    #[default]
53    Auto,
54    /// No skirts.
55    None,
56}

76/// Elevation data of one `raster-dem` tile as it moves through the pipeline.
77///
78/// Stored at the DEM tile's own coordinates, which sit one zoom level below the tiles they drape.
79#[derive(Debug)]
80pub enum DemTileComponent {
81    /// The tile has been requested and is being fetched or decoded.
82    Pending,
83    /// The tile is available for rendering and elevation queries.
84    Loaded(LoadedDem),
85    /// The tile could not be fetched or decoded; ancestors stand in for it.
86    Missing,
87}

119/// Requests, stores, drapes and draws the DEM tiles of a style that declares `terrain`.
120pub struct TerrainPlugin<T>(PhantomData<T>);
121
122impl<T: DemTransferables> Default for TerrainPlugin<T> {
123    fn default() -> Self {
124        Self(PhantomData)
125    }
126}
127
128impl<E: Environment, T: DemTransferables> Plugin<E> for TerrainPlugin<T> {
129    fn build(
130        &self,
131        schedule: &mut Schedule,
132        kernel: Rc<Kernel<E>>,
133        world: &mut World,
134        graph: &mut RenderGraph,
135    ) {
136        world
137            .resources
138            .insert(Eventually::<TerrainResources>::Uninitialized);
139        world.resources.init::<DrapePhase>();
140        world.resources.init::<TerrainFrame>();
141        world.resources.init::<TerrainCoverageIndex>();
142
143        schedule.add_system_to_stage(
144            RenderStageLabel::Extract,
145            SystemContainer::new(RequestSystem::<E, T>::new(&kernel)),
146        );
147        schedule.add_system_to_stage(
148            RenderStageLabel::Extract,
149            SystemContainer::new(PopulateWorldSystem::<E, T>::new(&kernel)),
150        );
151        // Prepare rather than Extract: headless rendering drops the Extract stage, and the
152        // elevation must be known before the Queue stage builds the view pattern. The index
153        // runs first so the center elevation samples this frame's tiles.
154        schedule.add_system_to_stage(RenderStageLabel::Prepare, coverage::coverage_system);
155        schedule.add_system_to_stage(
156            RenderStageLabel::Prepare,
157            elevation::center_elevation_system,
158        );
159        schedule.add_system_to_stage(RenderStageLabel::Prepare, resource_system::resource_system);
160        schedule.add_system_to_stage(RenderStageLabel::Queue, upload_system::upload_system);
161        schedule.add_system_to_stage(RenderStageLabel::Queue, queue_system::queue_system);
162
163        let Some(draw_graph) = graph.get_sub_graph_mut(draw_graph::NAME) else {
164            tracing::error!("draw graph is missing; terrain will not be drawn");
165            return;
166        };
167        draw_graph.add_node(DRAPE_PASS, DrapePassNode);
168        let input = draw_graph.input_node().map(|node| node.id);
169        let edges = [
170            input.map(|input| draw_graph.add_node_edge(NodeLabel::Id(input), DRAPE_PASS)),
171            Some(draw_graph.add_node_edge(DRAPE_PASS, draw_graph::node::MAIN_PASS)),
172        ];
173        for edge in edges.into_iter().flatten() {
174            if let Err(error) = edge {
175                tracing::error!(
176                    ?error,
177                    "unable to order the drape pass before the main pass"
178                );
179            }
180        }
181    }
182}
