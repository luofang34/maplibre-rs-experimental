48/// Whether terrain tiles hang skirts below their edges, as GL JS `terrainSkirtLength`: skirts
49/// hide the cracks between neighbouring tiles of different levels, and show through a
50/// translucent drape.
51#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
52pub enum TerrainSkirts {
53    /// Skirts drop by a fifth of a tile's circumference at the camera's zoom.
54    #[default]
55    Auto,
56    /// No skirts.
57    None,
58}

90/// Elevation data of one `raster-dem` tile as it moves through the pipeline.
91///
92/// Stored at the DEM tile's own coordinates, which sit one zoom level below the tiles they drape.
93#[derive(Clone, Debug)]
94pub enum DemTileComponent {
95    /// The tile has been requested and is being fetched or decoded.
96    Pending,
97    /// The tile is available for rendering and elevation queries.
98    Loaded(LoadedDem),
99    /// The tile could not be fetched or decoded; ancestors stand in for it.
100    Missing,
101}

133/// Requests, stores, drapes and draws the DEM tiles of a style that declares `terrain`.
134pub struct TerrainPlugin<T>(PhantomData<T>);
135
136impl<T: DemTransferables> Default for TerrainPlugin<T> {
137    fn default() -> Self {
138        Self(PhantomData)
139    }
140}
141
142impl<E: Environment, T: DemTransferables> Plugin<E> for TerrainPlugin<T> {
143    fn build(
144        &self,
145        schedule: &mut Schedule,
146        kernel: Rc<Kernel<E>>,
147        world: &mut World,
148        renderer: &mut crate::render::Renderer,
149    ) -> Result<(), crate::plugin::PluginError> {
150        world
151            .resources
152            .insert(Eventually::<TerrainResources>::Uninitialized);
153        world.resources.init::<DrapePhase>();
154        world.resources.init::<TerrainFrame>();
155        world.resources.init::<TerrainCoverageIndex>();
156
157        schedule.add_system_to_stage(
158            RenderStageLabel::Request,
159            SystemContainer::new(RequestSystem::<E, T>::new(&kernel)),
160        )?;
161        schedule.add_system_to_stage(
162            RenderStageLabel::Request,
163            SystemContainer::new(PopulateWorldSystem::<E, T>::new(&kernel)),
164        )?;
165        // Prepare rather than Request: headless rendering drops the Request stage, and the
166        // elevation must be known before the Queue stage builds the view pattern. The index
167        // runs first so the center elevation samples this frame's tiles.
168        schedule.add_system_to_stage(RenderStageLabel::Prepare, coverage::coverage_system)?;
169        schedule.add_system_to_stage(
170            RenderStageLabel::Prepare,
171            elevation::center_elevation_system,
172        )?;
173        schedule.add_system_to_stage(
174            RenderStageLabel::Prepare,
175            crate::render::resolved_view::resolve_frame_view_system,
176        )?;
177        schedule
178            .add_system_to_stage(RenderStageLabel::Prepare, resource_system::resource_system)?;
179        schedule.add_system_to_stage(RenderStageLabel::Queue, upload_system::upload_system)?;
180        schedule.add_system_to_stage(RenderStageLabel::Queue, queue_system::queue_system)?;
181        // Before the Render stage, so the last measurement is read before this frame's drape
182        // passes start the next one.
183        schedule.add_system_to_stage(
184            RenderStageLabel::PhaseSort,
185            drape_timing::drape_timer_system,
186        )?;
187        world
188            .resources
189            .get_or_init_mut::<crate::render::BeforeSubmitHooks>()
190            .register(drape_timing::resolve_before_submit);
191        world
192            .resources
193            .get_or_init_mut::<crate::render::AfterSubmitCallbacks>()
194            .register(drape_timing::map_after_submit);
195
196        let draw_graph = &mut renderer.render_graph;
197        let edges = [
198            draw_graph.add_node(DRAPE_PASS, DrapePassNode),
199            draw_graph.add_node_edge(DRAPE_PASS, draw_graph::node::MAIN_PASS),
200        ];
201        for edge in edges {
202            edge?;
203        }
204        Ok(())
205    }
206}
