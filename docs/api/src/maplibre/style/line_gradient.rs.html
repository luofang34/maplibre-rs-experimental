1//! The colour ramp a `line-gradient` is drawn through.
2
3use csscolorparser::Color;
4
5use crate::style::{
6    expression::{Color as ExpressionColor, EvaluationContext, Value},
7    layer::StyleProperty,
8};
9
10/// Texels in the ramp: the colour at each of this many positions from the start to the end of
11/// a line.
12pub const RAMP_TEXELS: usize = 256;
13
14/// Texels in the ramp of a gradient made of steps, sampled without blending: wide enough that a
15/// step as narrow as a thousandth of a line stays where it is, as GL JS widens it.
16pub const STEP_RAMP_TEXELS: usize = 8192;
17
18/// How far a stepped gradient's ramp reaches into the small progress values: texel `i` of `n`
19/// holds the colour at `(2^(u * STEP_RAMP_SPAN_BITS) - 1) / 2^STEP_RAMP_SPAN_BITS` for
20/// `u = i / (n - 1)`, so a step at a millionth of a very long line stays as sharp as one at a
21/// tenth of a short one.
22pub const STEP_RAMP_SPAN_BITS: f64 = 20.0;
23
24/// The progress a position `u` in `0..=1` of a stepped gradient's ramp stands for.
25pub fn stepped_progress(u: f64) -> f64 {
26    ((u * STEP_RAMP_SPAN_BITS).exp2() - 1.0) / STEP_RAMP_SPAN_BITS.exp2()
27}
28
29/// Whether the gradient is a `step` expression, whose edges are hard.
30pub fn is_stepped(gradient: &StyleProperty<Color>) -> bool {
31    matches!(
32        gradient.expression(),
33        Some(crate::style::expression::Expression::Step { .. })
34    )
35}
36
37/// The colour of each of [`RAMP_TEXELS`] positions along the line, or [`STEP_RAMP_TEXELS`] for
38/// steps, straight-alpha 8-bit RGBA. A position whose colour cannot be evaluated is transparent.
39pub fn ramp(gradient: &StyleProperty<Color>) -> Vec<[u8; 4]> {
40    let texels = if is_stepped(gradient) {
41        STEP_RAMP_TEXELS
42    } else {
43        RAMP_TEXELS
44    };
45    (0..texels)
46        .map(|texel| {
47            let position = texel as f64 / (texels - 1) as f64;
48            let progress = if texels == STEP_RAMP_TEXELS {
49                stepped_progress(position)
50            } else {
51                position
52            };
53            let color = match gradient {
54                StyleProperty::Constant(color) => Some(ExpressionColor::from(color.clone())),
55                StyleProperty::Expression(property) => {
56                    let context = EvaluationContext {
57                        line_progress: progress,
58                        ..EvaluationContext::default()
59                    };
60                    match property.expression().evaluate(&context) {
61                        Ok(Value::Color(color)) => Some(color),
62                        _ => None,
63                    }
64                }
65                StyleProperty::Unsupported(_) => None,
66            };
67            let [r, g, b, a] = color.map_or([0.0; 4], |color| color.straight());
68            [r, g, b, a].map(|channel| (channel * 255.0).round() as u8)
69        })
70        .collect()
71}
