20fn parse_scheme(
21    map: &Map<String, Value>,
22) -> Result<Option<TileAddressingScheme>, serde_json::Error> {
23    match json_opt_str(map, "scheme")? {
24        None => Ok(None),
25        Some("xyz") => Ok(Some(TileAddressingScheme::XYZ)),
26        Some("tms") => Ok(Some(TileAddressingScheme::TMS)),
27        Some(_) => Err(field_err("scheme", "expected \"xyz\" or \"tms\"")),
28    }
29}

31fn parse_encoding(map: &Map<String, Value>) -> Result<Option<DemEncoding>, serde_json::Error> {
32    match json_opt_str(map, "encoding")? {
33        None => Ok(None),
34        Some("mapbox") => Ok(Some(DemEncoding::Mapbox)),
35        Some("terrarium") => Ok(Some(DemEncoding::Terrarium)),
36        Some("custom") => Ok(Some(DemEncoding::Custom)),
37        Some(_) => Err(field_err("encoding", "unknown DEM encoding")),
38    }
39}

41fn parse_vector_source(map: &Map<String, Value>) -> Result<VectorSource, serde_json::Error> {
42    let promote_id = map
43        .get("promoteId")
44        .filter(|v| !v.is_null())
45        .map(PromoteId::from_json_value)
46        .transpose()?;
47    Ok(VectorSource {
48        attribution: json_opt_string(map, "attribution")?,
49        bounds: json_opt_bounds4(map, "bounds")?,
50        maxzoom: json_opt_u8(map, "maxzoom")?,
51        minzoom: json_opt_u8(map, "minzoom")?,
52        promote_id,
53        scheme: parse_scheme(map)?,
54        tiles: json_opt_string_vec(map, "tiles")?,
55        tile_size: json_opt_u32(map, "tileSize")?,
56        url: json_opt_string(map, "url")?,
57        volatile: json_bool_or(map, "volatile", false)?,
58    })
59}

61fn parse_raster_dem_source(map: &Map<String, Value>) -> Result<RasterDemSource, serde_json::Error> {
62    Ok(RasterDemSource {
63        attribution: json_opt_string(map, "attribution")?,
64        bounds: json_opt_bounds4(map, "bounds")?,
65        maxzoom: json_opt_u8(map, "maxzoom")?,
66        minzoom: json_opt_u8(map, "minzoom")?,
67        tiles: json_opt_string_vec(map, "tiles")?,
68        url: json_opt_string(map, "url")?,
69        tile_size: json_opt_u32(map, "tileSize")?,
70        encoding: parse_encoding(map)?,
71        red_factor: json_f64_or(map, "redFactor", 1.0)?,
72        green_factor: json_f64_or(map, "greenFactor", 1.0)?,
73        blue_factor: json_f64_or(map, "blueFactor", 1.0)?,
74        base_shift: json_f64_or(map, "baseShift", 0.0)?,
75        volatile: json_bool_or(map, "volatile", false)?,
76    })
77}

79fn parse_geojson_source(map: &Map<String, Value>) -> Result<GeoJsonSource, serde_json::Error> {
80    let Some(data_val) = map.get("data") else {
81        return Err(missing_field_err("data"));
82    };
83    let promote_id = map
84        .get("promoteId")
85        .filter(|v| !v.is_null())
86        .map(PromoteId::from_json_value)
87        .transpose()?;
88    Ok(GeoJsonSource {
89        data: GeoJsonData::from_json_value(data_val),
90        maxzoom: json_opt_u8(map, "maxzoom")?,
91        buffer: json_opt_u32(map, "buffer")?,
92        tolerance: json_opt_f64(map, "tolerance")?,
93        line_metrics: json_bool_or(map, "lineMetrics", false)?,
94        minzoom: json_opt_u8(map, "minzoom")?,
95        promote_id,
96        generate_id: json_bool_or(map, "generateId", false)?,
97        cluster: json_bool_or(map, "cluster", false)?,
98        cluster_radius: json_opt_f64(map, "clusterRadius")?,
99        cluster_max_zoom: json_opt_u8(map, "clusterMaxZoom")?,
100        cluster_min_points: json_opt_usize(map, "clusterMinPoints")?,
101        cluster_properties: map
102            .get("clusterProperties")
103            .filter(|v| !v.is_null())
104            .cloned(),
105        filter: map.get("filter").filter(|v| !v.is_null()).cloned(),
106        volatile: json_bool_or(map, "volatile", false)?,
107        generation: json_opt_u64(map, "generation")?.unwrap_or_else(fresh_generation),
108    })
109}

111impl Source {
112    /// Parses a [`Source`] directly from a JSON [`Value`].
113    pub(crate) fn from_json_value(value: &Value) -> Result<Self, serde_json::Error> {
114        let Some(map) = value.as_object() else {
115            return Err(de::Error::custom("source must be a JSON object"));
116        };
117        let Some(kind) = map.get("type").and_then(Value::as_str) else {
118            return Err(de::Error::missing_field("type"));
119        };
120        match kind {
121            "vector" => Ok(Self::Vector(parse_vector_source(map)?)),
122            "raster" => Ok(Self::Raster(parse_vector_source(map)?)),
123            "raster-dem" => Ok(Self::RasterDem(parse_raster_dem_source(map)?)),
124            "geojson" => Ok(Self::GeoJson(parse_geojson_source(map)?)),
125            "image" => Ok(Self::Image(ImageSource {
126                url: json_req_string(map, "url")?,
127                coordinates: json_req_coords4(map, "coordinates")?,
128                generation: json_opt_u64(map, "generation")?.unwrap_or_else(fresh_generation),
129            })),
130            "video" => Ok(Self::Video(VideoSource {
131                urls: json_string_vec_or_empty(map, "urls")?,
132                coordinates: json_req_coords4(map, "coordinates")?,
133                frame: None,
134                generation: fresh_generation(),
135            })),
136            "canvas" => Ok(Self::Canvas(CanvasSource {
137                canvas: json_opt_string(map, "canvas")?,
138                coordinates: json_req_coords4(map, "coordinates")?,
139                animate: json_bool_or(map, "animate", true)?,
140                frame: None,
141                generation: fresh_generation(),
142            })),
143            other => Err(de::Error::unknown_variant(
144                other,
145                &[
146                    "vector",
147                    "raster",
148                    "raster-dem",
149                    "geojson",
150                    "image",
151                    "video",
152                    "canvas",
153                ],
154            )),
155        }
156    }
157}

159impl<'de> Deserialize<'de> for Source {
160    fn deserialize<D: Deserializer<'de>>(deserializer: D) -> Result<Self, D::Error> {
161        let value = Value::deserialize(deserializer)?;
162        Self::from_json_value(&value).map_err(de::Error::custom)
163    }
164}

166impl Serialize for Source {
167    fn serialize<S: Serializer>(&self, serializer: S) -> Result<S::Ok, S::Error> {
168        match self {
169            Self::Vector(source) => serialize_tagged_object("vector", source, serializer),
170            Self::Raster(source) => serialize_tagged_object("raster", source, serializer),
171            Self::RasterDem(source) => serialize_tagged_object("raster-dem", source, serializer),
172            Self::GeoJson(source) => serialize_tagged_object("geojson", source, serializer),
173            Self::Image(source) => serialize_tagged_object("image", source, serializer),
174            Self::Video(source) => serialize_tagged_object("video", source, serializer),
175            Self::Canvas(source) => serialize_tagged_object("canvas", source, serializer),
176        }
177    }
