19/// GeoJSON data — either an inline JSON value or a URL pointing to a GeoJSON file.
20#[derive(Debug, Clone, PartialEq)]
21pub enum GeoJsonData {
22    /// Address of a GeoJSON document.
23    Url(String),
24    /// Embedded feature, feature collection or geometry JSON, shared by every request.
25    Inline(Arc<serde_json::Value>),
26}
27
28impl GeoJsonData {
29    /// Converts a parsed JSON [`serde_json::Value`] directly into [`GeoJsonData`].
30    pub fn from_json_value(value: &serde_json::Value) -> Self {
31        match value {
32            serde_json::Value::String(url) => Self::Url(url.clone()),
33            other => Self::Inline(Arc::new(other.clone())),
34        }
35    }
36}
37
38impl Serialize for GeoJsonData {
39    fn serialize<S: serde::Serializer>(&self, serializer: S) -> Result<S::Ok, S::Error> {
40        match self {
41            Self::Url(url) => serializer.serialize_str(url),
42            Self::Inline(value) => value.serialize(serializer),
43        }
44    }
45}
46
47impl<'de> Deserialize<'de> for GeoJsonData {
48    fn deserialize<D: serde::Deserializer<'de>>(deserializer: D) -> Result<Self, D::Error> {
49        let value = serde_json::Value::deserialize(deserializer)?;
50        Ok(Self::from_json_value(&value))
51    }
52}

54/// Feature property that replaces the feature id, as the style specification's `promoteId`.
55#[derive(Debug, Clone, PartialEq, Eq)]
56pub enum PromoteId {
57    /// One property for every layer of the source.
58    Property(String),
59    /// A property for each source layer; GeoJSON has the single layer `_geojson`.
60    PerLayer(HashMap<String, String>),
61}
62
63impl PromoteId {
64    /// Parses `promoteId` from a parsed JSON [`serde_json::Value`].
65    pub(crate) fn from_json_value(value: &serde_json::Value) -> Result<Self, serde_json::Error> {
66        match value {
67            serde_json::Value::String(prop) => Ok(Self::Property(prop.clone())),
68            serde_json::Value::Object(map) => {
69                let mut out = HashMap::with_capacity_and_hasher(map.len(), Default::default());
70                for (k, v) in map {
71                    let Some(s) = v.as_str() else {
72                        return Err(serde::de::Error::custom(
73                            "promoteId layer map values must be strings",
74                        ));
75                    };
76                    out.insert(k.clone(), s.to_owned());
77                }
78                Ok(Self::PerLayer(out))
79            }
80            _ => Err(serde::de::Error::custom(
81                "promoteId must be a string or an object",
82            )),
83        }
84    }
85}
86
87impl Serialize for PromoteId {
88    fn serialize<S: serde::Serializer>(&self, serializer: S) -> Result<S::Ok, S::Error> {
89        match self {
90            Self::Property(prop) => serializer.serialize_str(prop),
91            Self::PerLayer(map) => map.serialize(serializer),
92        }
93    }
94}

96impl<'de> Deserialize<'de> for PromoteId {
97    fn deserialize<D: serde::Deserializer<'de>>(deserializer: D) -> Result<Self, D::Error> {
98        let value = serde_json::Value::deserialize(deserializer)?;
99        Self::from_json_value(&value).map_err(serde::de::Error::custom)
100    }
101}

103impl PromoteId {
104    /// Returns the property promoted as the feature ID for `source_layer` (`None` or `_geojson`
105    /// selects the GeoJSON layer entry when `promoteId` is per-layer).
106    pub fn property_for(&self, source_layer: Option<&str>) -> Option<&str> {
107        match self {
108            Self::Property(name) => Some(name.as_str()),
109            Self::PerLayer(map) => {
110                let layer = match source_layer {
111                    None | Some("") => GEOJSON_LAYER,
112                    Some(name) => name,
113                };
114                map.get(layer)
115                    .or_else(|| {
116                        (layer != GEOJSON_LAYER)
117                            .then(|| map.get(GEOJSON_LAYER))
118                            .flatten()
119                    })
120                    .map(String::as_str)
121            }
122        }
123    }
124}

129/// GeoJSON source declaration; workers fetch URL data, index it once per generation and tile it.
130#[derive(Serialize, Deserialize, Debug, Clone)]
131pub struct GeoJsonSource {
132    /// Embedded geometry or a document URL.
133    pub data: GeoJsonData,
134    /// Upper zoom the source is tiled at; deeper views overzoom it. GL JS defaults to 18.
135    #[serde(skip_serializing_if = "Option::is_none")]
136    pub maxzoom: Option<u8>,
137    /// Pixels of a 512-pixel tile that the geometry of a tile extends past its edge. GL JS
138    /// defaults to 128.
139    #[serde(default, skip_serializing_if = "Option::is_none")]
140    pub buffer: Option<u32>,
141    /// Douglas-Peucker simplification tolerance in pixels on a 512-pixel tile (`0.375` when
142    /// absent; `0.0` disables simplification).
143    #[serde(default, skip_serializing_if = "Option::is_none")]
144    pub tolerance: Option<f64>,
145    /// Computes cumulative distance metrics along `LineString` features so `line-gradient`
146    /// (`["line-progress"]`) interpolates smoothly across tile boundaries.
147    #[serde(rename = "lineMetrics", default, skip_serializing_if = "is_false")]
148    pub line_metrics: bool,
149    /// Lower source zoom; nothing is tiled below it.
150    #[serde(skip_serializing_if = "Option::is_none")]
151    pub minzoom: Option<u8>,
152    /// Property whose value becomes the feature id.
153    #[serde(rename = "promoteId", default, skip_serializing_if = "Option::is_none")]
154    pub promote_id: Option<PromoteId>,
155    /// Numbers features without an id by their position in the document.
156    #[serde(rename = "generateId", default, skip_serializing_if = "is_false")]
157    pub generate_id: bool,
158    /// Groups nearby points into clusters that carry `cluster`, `point_count` and
159    /// `point_count_abbreviated`.
160    #[serde(default, skip_serializing_if = "is_false")]
161    pub cluster: bool,
162    /// Pixel radius, on a 512-pixel tile, within which points cluster; 50 when absent.
163    #[serde(
164        rename = "clusterRadius",
165        default,
166        skip_serializing_if = "Option::is_none"
167    )]
168    pub cluster_radius: Option<f64>,
169    /// Highest zoom that clusters; one below `maxzoom` when absent.
170    #[serde(
171        rename = "clusterMaxZoom",
172        default,
173        skip_serializing_if = "Option::is_none"
174    )]
175    pub cluster_max_zoom: Option<u8>,
176    /// Fewest points that make a cluster; 2 when absent.
177    #[serde(
178        rename = "clusterMinPoints",
179        default,
180        skip_serializing_if = "Option::is_none"
181    )]
182    pub cluster_min_points: Option<usize>,
183    /// Properties aggregated over the points of each cluster: a name for each
184    /// `[operator, map expression]`.
185    #[serde(
186        rename = "clusterProperties",
187        default,
188        skip_serializing_if = "Option::is_none"
189    )]
190    pub cluster_properties: Option<serde_json::Value>,
191    /// Keeps only the features this filter accepts, before any clustering.
192    #[serde(default, skip_serializing_if = "Option::is_none")]
193    pub filter: Option<serde_json::Value>,
194    /// Bypasses persistent HTTP caching so live feeds are fetched afresh across sessions.
195    #[serde(default, skip_serializing_if = "is_false")]
196    pub volatile: bool,
197    /// Identifies this version of the data. It is unique per parsed style and per change, so
198    /// workers that share one cache never mix up two documents of the same source name, and a
199    /// changed value makes them load and index the data again. It is serialized, so the tile
200    /// requests a worker receives as messages keep the value and share one cached index.
201    #[serde(default = "fresh_generation")]
202    pub generation: u64,
203}

205fn is_false(value: &bool) -> bool {
206    !value

209/// A generation no other source version has, without any global counter.
210pub fn fresh_generation() -> u64 {
211    static NEXT_GENERATION: AtomicU64 = AtomicU64::new(1);
212    NEXT_GENERATION.fetch_add(1, Ordering::Relaxed)
213}

271/// How elevation in meters is packed into 8-bit RGB channels of a `raster-dem` tile.
272#[derive(Serialize, Deserialize, Debug, Clone, Copy, PartialEq, Eq, Default)]
273pub enum DemEncoding {
274    /// Mapbox Terrain-RGB: `(R * 256 * 256 + G * 256 + B) / 10 - 10000`.
275    #[default]
276    #[serde(rename = "mapbox")]
277    Mapbox,
278    /// Terrarium: `R * 256 + G + B / 256 - 32768`.
279    #[serde(rename = "terrarium")]
280    Terrarium,
281    /// Channel factors and base shift taken from the source definition.
282    #[serde(rename = "custom")]
283    Custom,
284}

359impl RasterDemSource {
360    /// Nominal tile size in pixels, defaulting as the style specification does.
361    pub fn tile_size(&self) -> u32 {
362        self.tile_size.unwrap_or(DEM_DEFAULT_TILE_SIZE)
363    }
364
365    /// Elevation packing of the tile pixels, defaulting as the style specification does.
366    pub fn encoding(&self) -> DemEncoding {
367        self.encoding.unwrap_or_default()
368    }
369
370    /// Returns `[red, green, blue, base_shift]` such that
371    /// `elevation = R * red + G * green + B * blue - base_shift` for 0..=255 channel values.
372    pub fn unpack_vector(&self) -> [f64; 4] {
373        match self.encoding() {
374            DemEncoding::Mapbox => [6553.6, 25.6, 0.1, 10000.0],
375            DemEncoding::Terrarium => [256.0, 1.0, 1.0 / 256.0, 32768.0],
376            DemEncoding::Custom => [
377                self.red_factor,
378                self.green_factor,
379                self.blue_factor,
380                self.base_shift,
381            ],
382        }
383    }
384}

386/// Data source selected by the JSON `type` field.
387#[derive(Debug, Clone)]
388pub enum Source {
389    /// Vector tile geometry and feature properties.
390    Vector(VectorSource),
391    /// Raster image tiles, using the same URL and TileJSON fields as vector sources.
392    Raster(VectorSource),
393    /// Packed elevation images consumed by terrain, hillshade and relief rendering.
394    RasterDem(RasterDemSource),
395    /// GeoJSON features declared inline or by document URL.
396    GeoJson(GeoJsonSource),
397    /// One image stretched over four geographic corners.
398    Image(ImageSource),
399    /// Host-decoded video frames stretched over four geographic corners.
400    Video(VideoSource),
401    /// Host-drawn 2D/WebGL canvas frames stretched over four geographic corners.
402    Canvas(CanvasSource),
403}

407impl Source {
408    /// Whether this source opts out of persistent HTTP caching (`"volatile": true`).
409    pub fn is_volatile(&self) -> bool {
410        match self {
411            Self::Vector(source) | Self::Raster(source) => source.volatile,
412            Self::RasterDem(source) => source.volatile,
413            Self::GeoJson(source) => source.volatile,
414            Self::Image(_) | Self::Video(_) | Self::Canvas(_) => false,
415        }
416    }
417}
