8/// Error returned when parsing an unrecognized style-spec layer, parameter, or property name.
9#[derive(Debug, Clone, PartialEq, Eq)]
10pub struct UnknownSpecIdentifier(pub String);
11
12impl fmt::Display for UnknownSpecIdentifier {
13    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
14        write!(f, "unknown style-spec identifier `{}`", self.0)
15    }
16}
17
18impl std::error::Error for UnknownSpecIdentifier {}

20/// Whether a style-spec layer property belongs to `paint` or `layout`.
21#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
22pub enum SpecPropertyKind {
23    /// A layer `paint` property.
24    Paint,
25    /// A layer `layout` property.
26    Layout,
27}
28
29impl SpecPropertyKind {
30    /// Canonical table scope (`"paint"` or `"layout"`).
31    pub const fn as_str(self) -> &'static str {
32        match self {
33            Self::Paint => "paint",
34            Self::Layout => "layout",
35        }
36    }
37}
38
39impl fmt::Display for SpecPropertyKind {
40    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
41        f.write_str(self.as_str())
42    }
43}

60/// Style-spec value type of a `paint` or `layout` property.
61#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
62pub enum SpecValueType {
63    /// Floating-point scalar (`number`).
64    Number,
65    /// CSS / RGBA colour (`color`).
66    Color,
67    /// UTF-8 string (`string`).
68    String,
69    /// Boolean flag (`boolean`).
70    Boolean,
71    /// String restricted to [`SpecPropertyMeta::enum_values`] (`enum`).
72    Enum,
73    /// Homogeneous array with item type and optional fixed length (`array`).
74    Array {
75        /// Element type of the array.
76        item: SpecArrayItemType,
77        /// Fixed array length when specified by the style spec.
78        length: Option<usize>,
79    },
80    /// Rich formatted text (`formatted`).
81    Formatted,
82    /// Sprite or host image name (`resolvedImage`).
83    ResolvedImage,
84    /// Directional `[top, right, bottom, left]` box padding (`padding`).
85    Padding,
86    /// Variable-length numeric list (`numberArray`).
87    NumberArray,
88    /// Variable-length colour list (`colorArray`).
89    ColorArray,
90    /// Alternating anchor and `[x, y]` offset pairs (`variableAnchorOffsetCollection`).
91    VariableAnchorOffsetCollection,
92    /// Map projection name or `[from, to, transition]` blend (`projectionDefinition`).
93    ProjectionDefinition,
94}

96/// Input parameter permitted inside expressions for a style-spec property.

117impl SpecExpressionParameter {

125            Self::HeatmapDensity => "heatmap-density",
126            Self::SkyRadialProgress => "sky-radial-progress",

130    }
131}

134    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
135        f.write_str(self.as_str())
136    }

139impl FromStr for SpecExpressionParameter {
140    type Err = UnknownSpecIdentifier;
141
142    fn from_str(s: &str) -> Result<Self, Self::Err> {
143        match s {
144            "zoom" => Ok(Self::Zoom),
145            "feature" => Ok(Self::Feature),
146            "feature-state" => Ok(Self::FeatureState),
147            "line-progress" => Ok(Self::LineProgress),
148            "heatmap-density" => Ok(Self::HeatmapDensity),
149            "sky-radial-progress" => Ok(Self::SkyRadialProgress),
150            "measure-light" => Ok(Self::MeasureLight),
151            "elevation" => Ok(Self::Elevation),
152            _ => Err(UnknownSpecIdentifier(s.to_owned())),
153        }
154    }
155}

168/// Complete build-time metadata for one style-spec `paint` or `layout` property.
169#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
170pub struct SpecPropertyMeta {
171    /// Canonical kebab-case property name (e.g. `"fill-color"`).
172    pub name: &'static str,
173    /// Layer type that owns this property.
174    pub layer_type: SpecLayerType,
175    /// Whether this is a `paint` or `layout` property.
176    pub kind: SpecPropertyKind,
177    /// Value type accepted by this property.
178    pub value_type: SpecValueType,
179    /// Expression capabilities and allowed input parameters.
180    pub expression: SpecPropertyExpression,
181    /// Allowed string literals when `value_type` is `Enum` or `Array { item: Enum, .. }`.
182    pub enum_values: &'static [&'static str],
183    /// Whether `<property>-transition` is supported on `paint`.
184    pub supports_transition: bool,
185    /// Whether `{token}` string templates are lowered for this property.
186    pub supports_tokens: bool,
187    /// Serialized JSON representation of the specification default, when defined.
188    pub default_json: Option<&'static str>,
189    /// Specification documentation text.
190    pub doc: &'static str,
191}
192
193impl SpecPropertyMeta {
194    /// Parses [`Self::default_json`] into a `serde_json::Value`, when the spec defines a default.
195    pub fn default_value(&self) -> Option<serde_json::Value> {
196        self.default_json
197            .and_then(|raw| serde_json::from_str(raw).ok())
198    }
199
200    /// Converts [`Self::value_type`] and [`Self::enum_values`] into an expression [`PropertyKind`].
201    pub fn property_kind(&self) -> PropertyKind {
202        let enum_strings = || {
203            self.enum_values
204                .iter()
205                .map(|value| (*value).to_owned())
206                .collect()
207        };
208        match self.value_type {
209            SpecValueType::Number => PropertyKind::Number,
210            SpecValueType::Color => PropertyKind::Color,
211            SpecValueType::String => PropertyKind::String,
212            SpecValueType::Boolean => PropertyKind::Boolean,
213            SpecValueType::Enum => PropertyKind::Enum(enum_strings()),
214            SpecValueType::Formatted => PropertyKind::Formatted,
215            SpecValueType::ResolvedImage => PropertyKind::ResolvedImage,
216            SpecValueType::Padding => PropertyKind::Padding,
217            SpecValueType::NumberArray => PropertyKind::NumberArray,
218            SpecValueType::ColorArray => PropertyKind::ColorArray,
219            SpecValueType::VariableAnchorOffsetCollection => {
220                PropertyKind::VariableAnchorOffsetCollection
221            }
222            SpecValueType::ProjectionDefinition => PropertyKind::ProjectionDefinition,
223            SpecValueType::Array { item, length } => PropertyKind::Array {
224                item: Box::new(match item {
225                    SpecArrayItemType::Number => PropertyKind::Number,
226                    SpecArrayItemType::String => PropertyKind::String,
227                    SpecArrayItemType::Boolean => PropertyKind::Boolean,
228                    SpecArrayItemType::Enum => PropertyKind::Enum(enum_strings()),
229                    SpecArrayItemType::Value => PropertyKind::Value,
230                }),
231                length,
232            },
233        }
234    }
235
236    /// Builds the [`LegacyPropertySpec`] used to parse constants, `{token}` templates,
237    /// legacy functions, and expressions for this property.
238    pub fn legacy_property_spec(&self) -> LegacyPropertySpec {
239        LegacyPropertySpec {
240            kind: self.property_kind(),
241            interpolated: self.expression.supports_interpolated,
242            default: self.default_value(),
243            tokens: self.supports_tokens,
244        }
245    }
246}

257/// Looks up the static descriptor for a `paint` property on `layer_type`.
258pub fn lookup_paint_property(layer_type: &str, name: &str) -> Option<&'static SpecPropertyMeta> {
259    let layer = SpecLayerType::from_str(layer_type).ok()?;
260    match layer {
261        SpecLayerType::Fill => FillPaintProperty::from_str(name)
262            .ok()
263            .map(FillPaintProperty::metadata),
264        SpecLayerType::Line => LinePaintProperty::from_str(name)
265            .ok()
266            .map(LinePaintProperty::metadata),
267        SpecLayerType::Symbol => SymbolPaintProperty::from_str(name)
268            .ok()
269            .map(SymbolPaintProperty::metadata),
270        SpecLayerType::Circle => CirclePaintProperty::from_str(name)
271            .ok()
272            .map(CirclePaintProperty::metadata),
273        SpecLayerType::Heatmap => HeatmapPaintProperty::from_str(name)
274            .ok()
275            .map(HeatmapPaintProperty::metadata),
276        SpecLayerType::FillExtrusion => FillExtrusionPaintProperty::from_str(name)
277            .ok()
278            .map(FillExtrusionPaintProperty::metadata),
279        SpecLayerType::Raster => RasterPaintProperty::from_str(name)
280            .ok()
281            .map(RasterPaintProperty::metadata),
282        SpecLayerType::Hillshade => HillshadePaintProperty::from_str(name)
283            .ok()
284            .map(HillshadePaintProperty::metadata),
285        SpecLayerType::Background => BackgroundPaintProperty::from_str(name)
286            .ok()
287            .map(BackgroundPaintProperty::metadata),
288        SpecLayerType::Sky => SkyPaintProperty::from_str(name)
289            .ok()
290            .map(SkyPaintProperty::metadata),
291        SpecLayerType::ColorRelief => ColorReliefPaintProperty::from_str(name)
292            .ok()
293            .map(ColorReliefPaintProperty::metadata),
294        SpecLayerType::Custom => None,
295    }
296}
