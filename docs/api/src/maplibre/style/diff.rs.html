1//! Diffing whole-style replacement (`setStyle` / [`Style::diff_and_replace`]).
2//!
3//! Comparing an incoming [`Style`] against the running style preserves source `generation` IDs
4//! and resident tiles for unchanged sources, updates GPU paint and layer order in-place when
5//! layout/filtering did not change, and initiates configured paint transitions on retained layers.
6

23/// Minimal update plan produced when replacing one [`Style`] with another via [`Style::diff_and_replace`].
24#[derive(Debug, Clone, Default, PartialEq, Eq)]
25pub struct StyleReplacementDiff {
26    /// Source names newly introduced in the replacement style.
27    pub added_sources: Vec<String>,
28    /// Source names removed from the previous style.
29    pub removed_sources: Vec<String>,
30    /// Retained source names whose definition changed.
31    pub updated_sources: Vec<String>,
32    /// Layer IDs newly introduced in the replacement style.
33    pub added_layers: Vec<String>,
34    /// Layer IDs removed from the previous style.
35    pub removed_layers: Vec<String>,
36    /// Retained layer IDs whose paint, layout, filter, visibility or zoom bounds changed.
37    pub updated_layers: Vec<String>,
38    /// Retained layer IDs whose painter index changed.
39    pub reordered_layers: Vec<String>,
40    /// Whether vector tiles must be fetched and tessellated again.
41    pub redraw_tiles: bool,
42    /// Whether tiled raster or DEM sources were added or reconfigured.
43    pub redraw_raster_tiles: bool,
44    /// Image/video/canvas sources whose corners or image URL changed.
45    pub reloaded_image_sources: Vec<String>,
46    /// Raster, DEM, image, video or canvas sources whose resident GPU textures must be dropped.
47    pub removed_raster_sources: Vec<String>,
48    /// Whether the root `terrain` specification changed.
49    pub terrain_changed: bool,
50    /// Whether the root `sky` specification changed.
51    pub sky_changed: bool,
52    /// Whether the root `light` specification changed.
53    pub light_changed: bool,
54    /// Whether the root `projection` specification changed.
55    pub projection_changed: bool,
56    /// Whether `sprite` or host `images` changed.
57    pub sprite_changed: bool,
58    /// Whether the `glyphs` font URL template changed.
59    pub glyphs_changed: bool,
60    /// Whether the root `transition` specification changed.
61    pub transition_changed: bool,
62    /// Whether resolved `global-state` values changed.
63    pub global_state_changed: bool,
64}

66impl StyleReplacementDiff {
67    /// Deduplicated list of added, updated and reordered layer IDs for in-place GPU style refresh.
68    pub fn affected_layers(&self) -> Vec<String> {
69        let mut out = Vec::new();
70        for id in self
71            .added_layers
72            .iter()
73            .chain(&self.updated_layers)
74            .chain(&self.reordered_layers)
75        {
76            push_unique(&mut out, id);
77        }
78        out
79    }
80}

82fn geojson_definition_eq(a: &GeoJsonSource, b: &GeoJsonSource) -> bool {
83    a.data == b.data
84        && a.maxzoom == b.maxzoom
85        && a.minzoom == b.minzoom
86        && a.buffer == b.buffer
87        && a.tolerance == b.tolerance
88        && a.line_metrics == b.line_metrics
89        && a.promote_id == b.promote_id
90        && a.generate_id == b.generate_id
91        && a.cluster == b.cluster
92        && a.cluster_radius == b.cluster_radius
93        && a.cluster_max_zoom == b.cluster_max_zoom
94        && a.cluster_min_points == b.cluster_min_points
95        && a.cluster_properties == b.cluster_properties
96        && a.filter == b.filter
97        && a.volatile == b.volatile
98}
99
100fn image_definition_eq(a: &ImageSource, b: &ImageSource) -> bool {
101    a.url == b.url && a.coordinates == b.coordinates
102}
103
104fn video_definition_eq(a: &VideoSource, b: &VideoSource) -> bool {
105    a.urls == b.urls && a.coordinates == b.coordinates
106}
107
108fn canvas_definition_eq(a: &CanvasSource, b: &CanvasSource) -> bool {
109    a.canvas == b.canvas && a.animate == b.animate && a.coordinates == b.coordinates
110}
111
112fn is_raster_like(source: &Source) -> bool {
113    matches!(
114        source,
115        Source::Raster(_)
116            | Source::RasterDem(_)
117            | Source::Image(_)
118            | Source::Video(_)
119            | Source::Canvas(_)
120    )
121}
122
123fn renumber_layers(layers: &mut [StyleLayer]) {
124    for (position, layer) in layers.iter_mut().enumerate() {
125        layer.index = u32::try_from(position).unwrap_or(u32::MAX);
126    }
127}

129impl Style {
130    /// Diffs `next` against `self`, preserving unchanged source generations and resident tile
131    /// state while computing minimal layer, source and root-property updates, then replaces `self`.
132    pub fn diff_and_replace(&mut self, mut next: Style) -> StyleReplacementDiff {
133        renumber_layers(&mut next.layers);
134        next.resolve_global_state();
135        next.active_transitions =
136            super::transitions::ActiveTransitions::with_clock(self.active_transitions.clock_ms());
137
138        let mut diff = StyleReplacementDiff {
139            terrain_changed: self.terrain != next.terrain,
140            sky_changed: serde_json::to_value(&self.sky).ok()
141                != serde_json::to_value(&next.sky).ok(),
142            light_changed: serde_json::to_value(&self.light).ok()
143                != serde_json::to_value(&next.light).ok(),
144            projection_changed: self.projection != next.projection,
145            sprite_changed: self.sprite != next.sprite || self.images != next.images,
146            glyphs_changed: self.glyphs != next.glyphs
147                || self.font_faces != next.font_faces
148                || self.local_ideograph_font_family != next.local_ideograph_font_family,
149            transition_changed: self.transition != next.transition,
150            global_state_changed: self.global_state_values() != next.global_state_values(),
151            ..Default::default()
152        };
153
154        let mut changed_vector_sources = HashSet::new();
155
156        for (name, old_source) in &self.sources {
157            if !next.sources.contains_key(name) {
158                diff.removed_sources.push(name.clone());
159                if is_raster_like(old_source) {
160                    diff.removed_raster_sources.push(name.clone());
161                }
162            }
163        }
164
165        for (name, new_source) in &mut next.sources {
166            let Some(old_source) = self.sources.get(name) else {
167                match new_source {
168                    Source::GeoJson(g) => g.generation = fresh_generation(),
169                    Source::Image(i) => i.generation = fresh_generation(),
170                    Source::Video(v) => v.generation = fresh_generation(),
171                    Source::Canvas(c) => c.generation = fresh_generation(),
172                    Source::Raster(_) | Source::RasterDem(_) => {
173                        diff.redraw_raster_tiles = true;
174                    }
175                    Source::Vector(_) => {}
176                }
177                diff.added_sources.push(name.clone());
178                continue;
179            };
180
181            match (old_source, new_source) {
182                (Source::Vector(a), Source::Vector(b)) => {
183                    if a != b {
184                        diff.updated_sources.push(name.clone());
185                        changed_vector_sources.insert(name.clone());
186                    }
187                }
188                (Source::Raster(a), Source::Raster(b)) => {
189                    if a != b {
190                        diff.updated_sources.push(name.clone());
191                        push_unique(&mut diff.removed_raster_sources, name);
192                        diff.redraw_raster_tiles = true;
193                    }
194                }
195                (Source::RasterDem(a), Source::RasterDem(b)) => {
196                    if a != b {
197                        diff.updated_sources.push(name.clone());
198                        push_unique(&mut diff.removed_raster_sources, name);
199                        diff.redraw_raster_tiles = true;
200                    }
201                }
202                (Source::GeoJson(a), Source::GeoJson(b)) => {
203                    if geojson_definition_eq(a, b) {
204                        b.generation = a.generation;
205                    } else {
206                        b.generation = fresh_generation();
207                        diff.updated_sources.push(name.clone());
208                    }
209                }
210                (Source::Image(a), Source::Image(b)) => {
211                    if image_definition_eq(a, b) {
212                        b.generation = a.generation;
213                    } else {
214                        b.generation = fresh_generation();
215                        diff.updated_sources.push(name.clone());
216                        push_unique(&mut diff.reloaded_image_sources, name);
217                    }
218                }
219                (Source::Video(a), Source::Video(b)) => {
220                    if video_definition_eq(a, b) {
221                        if b.frame.is_none() {
222                            b.frame = a.frame.clone();
223                        }
224                        b.generation = a.generation;
225                    } else {
226                        if b.frame.is_none() && a.urls == b.urls {
227                            b.frame = a.frame.clone();
228                        }
229                        b.generation = fresh_generation();
230                        diff.updated_sources.push(name.clone());
231                        push_unique(&mut diff.reloaded_image_sources, name);
232                    }
233                }
234                (Source::Canvas(a), Source::Canvas(b)) => {
235                    if canvas_definition_eq(a, b) {
236                        if b.frame.is_none() {
237                            b.frame = a.frame.clone();
238                        }
239                        b.generation = a.generation;
240                    } else {
241                        if b.frame.is_none() && a.canvas == b.canvas {
242                            b.frame = a.frame.clone();
243                        }
244                        b.generation = fresh_generation();
245                        diff.updated_sources.push(name.clone());
246                        push_unique(&mut diff.reloaded_image_sources, name);
247                    }
248                }
249                (old_other, new_other) => {
250                    diff.updated_sources.push(name.clone());
251                    if is_raster_like(old_other) {
252                        push_unique(&mut diff.removed_raster_sources, name);
253                    }
254                    match new_other {
255                        Source::GeoJson(g) => g.generation = fresh_generation(),
256                        Source::Image(i) => {
257                            i.generation = fresh_generation();
258                            push_unique(&mut diff.reloaded_image_sources, name);
259                        }
260                        Source::Video(v) => {
261                            v.generation = fresh_generation();
262                            push_unique(&mut diff.reloaded_image_sources, name);
263                        }
264                        Source::Canvas(c) => {
265                            c.generation = fresh_generation();
266                            push_unique(&mut diff.reloaded_image_sources, name);
267                        }
268                        Source::Raster(_) | Source::RasterDem(_) => {
269                            diff.redraw_raster_tiles = true;
270                        }
271                        Source::Vector(_) => {
272                            changed_vector_sources.insert(name.clone());
273                        }
274                    }
275                }
276            }
277        }
278
279        diff.added_sources.sort();
280        diff.removed_sources.sort();
281        diff.updated_sources.sort();
282        diff.removed_raster_sources.sort();
283        diff.reloaded_image_sources.sort();
284
285        // Preserve runtime feature-state for sources that remain unchanged unless `next` supplied its own.
286        if next.feature_state.is_empty() && !self.feature_state.is_empty() {
287            let mut preserved = self.feature_state.clone();
288            for removed in &diff.removed_sources {
289                let target = super::FeatureStateTarget {
290                    source: removed.clone(),
291                    source_layer: None,
292                };
293                preserved.remove(&target, None, None);
294            }
295            next.feature_state = preserved;
296        }
297
298        let old_layers: HashMap<&str, &StyleLayer> = self
299            .layers
300            .iter()
301            .map(|layer| (layer.id.as_str(), layer))
302            .collect();
303        let next_layer_ids: HashSet<&str> =
304            next.layers.iter().map(|layer| layer.id.as_str()).collect();
305
306        for old_layer in &self.layers {
307            if !next_layer_ids.contains(old_layer.id.as_str()) {
308                diff.removed_layers.push(old_layer.id.clone());
309            }
310        }
311
312        let mut pending_transitions: Vec<ActiveTransition> = Vec::new();
313        for new_layer in &next.layers {
314            let Some(old_layer) = old_layers.get(new_layer.id.as_str()).copied() else {
315                diff.added_layers.push(new_layer.id.clone());
316                if from_vector_tiles(new_layer) && !new_layer.is_hidden() {
317                    diff.redraw_tiles = true;
318                }
319                continue;
320            };
321
322            if old_layer.index != new_layer.index {
323                diff.reordered_layers.push(new_layer.id.clone());
324            }
325
326            let source_changed = new_layer
327                .source
328                .as_deref()
329                .is_some_and(|src| changed_vector_sources.contains(src));
330            if source_changed || needs_tile_redraw(old_layer, new_layer) {
331                diff.redraw_tiles = true;
332                diff.updated_layers.push(new_layer.id.clone());
333                continue;
334            }
335
336            let paint_changed = !old_layer.same_paint(new_layer);
337            let meta_or_bounds_changed = old_layer.visibility != new_layer.visibility
338                || old_layer.minzoom != new_layer.minzoom
339                || old_layer.maxzoom != new_layer.maxzoom
340                || old_layer.metadata != new_layer.metadata;
341
342            if paint_changed || meta_or_bounds_changed {
343                diff.updated_layers.push(new_layer.id.clone());
344            }
345
346            if paint_changed && old_layer.type_ == new_layer.type_ {
347                collect_layer_transitions(
348                    old_layer,
349                    new_layer,
350                    next.transition,
351                    &mut pending_transitions,
352                );
353            }
354        }
355
356        if diff.projection_changed && next.layers.iter().any(from_vector_tiles) {
357            diff.redraw_tiles = true;
358        }
359        if (diff.glyphs_changed || diff.sprite_changed)
360            && next
361                .layers
362                .iter()
363                .any(|layer| !layer.is_hidden() && layer.type_ == "symbol")
364        {
365            diff.redraw_tiles = true;
366        }
367
368        // Start any active paint transitions collected for retained layers at progress 0.
369        for entry in pending_transitions {
370            if let Some(initial) = interpolate_paint_value(&entry.from, &entry.to, 0.0) {
371                let _ = next.set_paint_property_immediate(&entry.layer, &entry.property, initial);
372                push_unique(&mut diff.updated_layers, &entry.layer);
373                let _ = next.apply_paint_with_transition(&entry.layer, &entry.property, entry.to);
374            }
375        }
376
377        *self = next;
378        diff
379    }
380}

382fn collect_layer_transitions(
383    old_layer: &StyleLayer,
384    new_layer: &StyleLayer,
385    root_transition: Option<TransitionSpec>,
386    out: &mut Vec<ActiveTransition>,
387) {
388    let (Ok(old_maps), Ok(new_maps)) = (old_layer.property_maps(), new_layer.property_maps())
389    else {
390        return;
391    };
392    for (prop, to_val) in &new_maps.paint {
393        if !is_transitionable_property(prop) {
394            continue;
395        }
396        let timing = new_layer.transitions.get(prop).copied().or(root_transition);
397        if !timing.is_some_and(TransitionSpec::is_active) {
398            continue;
399        }
400        let Some(from_val) = old_maps
401            .paint
402            .get(prop)
403            .cloned()
404            .or_else(|| default_paint_json(old_layer, prop))
405        else {
406            continue;
407        };
408        if interpolate_paint_value(&from_val, to_val, 0.5).is_some()
409            && interpolate_paint_value(&from_val, to_val, 0.0)
410                != interpolate_paint_value(&from_val, to_val, 1.0)
411        {
412            if let Some(spec) = timing {
413                out.push(ActiveTransition {
414                    layer: new_layer.id.clone(),
415                    property: prop.clone(),
416                    from: from_val,
417                    to: to_val.clone(),
418                    timing: spec,
419                    started_ms: 0.0,
420                });
421            }
422        }
423    }
