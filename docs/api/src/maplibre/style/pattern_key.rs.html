1//! The key a feature's pattern name travels under from tessellation to the draw that binds
2//! its image.
3
4/// Number of key bits, which a float's mantissa holds exactly.
5const KEY_BITS: u32 = 23;
6/// The exponent of one, so a key reads back as a normal float.
7const ONE: u32 = 0x3F80_0000;
8
9/// A stable key for a pattern name; a name the style does not hold has none.
10pub fn pattern_key(name: &str) -> u32 {
11    let mut hash: u32 = 0x811C_9DC5;
12    for byte in name.bytes() {
13        hash = (hash ^ u32::from(byte)).wrapping_mul(0x0100_0193);
14    }
15    // Zero is kept for "no pattern".
16    ((hash >> 9) & ((1 << KEY_BITS) - 1)).max(1)
17}
18
19/// The float that carries the key of `name`, or zero for a feature without a pattern.
20pub fn pattern_value(name: Option<&str>) -> f32 {
21    name.filter(|name| !name.is_empty())
22        .map_or(0.0, |name| f32::from_bits(ONE | pattern_key(name)))
23}
24
25/// The key a float carries, if it carries one.
26pub fn key_of_value(value: f32) -> Option<u32> {
27    let bits = value.to_bits();
28    (bits & !((1 << KEY_BITS) - 1) == ONE).then_some(bits & ((1 << KEY_BITS) - 1))
29}

31#[cfg(test)]
32mod tests {
33    use super::*;

35    #[test]
36    fn a_name_survives_the_trip_through_a_float() {
37        let value = pattern_value(Some("generic_icon"));
38        assert_eq!(key_of_value(value), Some(pattern_key("generic_icon")));
39        assert_eq!(key_of_value(pattern_value(None)), None);
40        assert_ne!(pattern_key("a"), pattern_key("b"));
41    }
