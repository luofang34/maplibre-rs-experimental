44fn default_layers() -> Vec<StyleLayer> {
45    [
46        layer(
47            "background",
48            "background",
49            None,
50            LayerPaint::Background(BackgroundPaint {

52                background_opacity: None,
53                background_pattern: None,
54            }),
55        ),
56        fill("park", [200, 250, 204]),
57        fill("landuse", [224, 223, 223]),
58        fill("landcover", [174, 223, 163]),
59        line("transportation", [255, 255, 255]),
60        fill("building", [217, 208, 201]),
61        fill("water", [170, 211, 223]),
62        fill("waterway", [170, 211, 223]),
63        line("boundary", [0, 0, 0]),
64        layer(
65            "raster",
66            "raster",
67            None,
68            LayerPaint::Raster(RasterPaint::default()),
69        ),
70        symbol("text", "place"),
71        symbol("transportation_name", "transportation_name-disabled"),
72    ]
73    .into_iter()
74    .enumerate()

76        layer.index = index as u32;

82fn layer(id: &str, kind: &str, source_layer: Option<&str>, paint: LayerPaint) -> StyleLayer {
83    StyleLayer {
84        id: id.into(),
85        type_: kind.into(),
86        source_layer: source_layer.map(str::to_owned),
87        paint: Some(paint),
88        ..Default::default()
89    }
90}
91
92fn color([red, green, blue]: [u8; 3]) -> StyleProperty<Color> {
93    StyleProperty::Constant(Color::from_rgba8(red, green, blue, 255))
94}
95
96fn fill(id: &str, rgb: [u8; 3]) -> StyleLayer {
97    layer(
98        id,
99        "fill",
100        Some(id),
101        LayerPaint::Fill(FillPaint {
102            fill_color: Some(color(rgb)),
103            ..Default::default()
104        }),
105    )
106}
107
108fn line(id: &str, rgb: [u8; 3]) -> StyleLayer {
109    layer(
110        id,
111        "line",
112        Some(id),
113        LayerPaint::Line(LinePaint {
114            line_color: Some(color(rgb)),
115            ..Default::default()
116        }),
117    )
118}
119
120fn symbol(id: &str, source_layer: &str) -> StyleLayer {
121    layer(
122        id,
123        "symbol",
124        Some(source_layer),
125        LayerPaint::Symbol(SymbolPaint {
126            text_field: Some(StyleProperty::parse(&serde_json::json!("{name}"))),
127            ..Default::default()
128        }),
129    )
