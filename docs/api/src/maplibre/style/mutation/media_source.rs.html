1//! Pushing RGBA frames and updating corner coordinates on `video` and `canvas` sources.

11impl Style {
12    /// Pushes a straight-alpha RGBA8 frame (`width * height * 4` bytes) and optional updated
13    /// `[longitude, latitude]` corner coordinates (`[top_left, top_right, bottom_right, bottom_left]`)
14    /// onto a `"video"` source.
15    pub fn set_video_source_frame(
16        &mut self,
17        source_name: &str,
18        width: u32,
19        height: u32,
20        rgba: &[u8],
21        coordinates: Option<[[f64; 2]; 4]>,
22    ) -> Result<StyleChange, StyleMutationError> {
23        let frame = build_media_frame(source_name, width, height, rgba)?;
24        let video = video_source_mut(&mut self.sources, source_name)?;
25        video.frame = Some(frame);
26        if let Some(coords) = coordinates {
27            video.coordinates = coords;
28        }
29        video.generation = fresh_generation();
30        Ok(reloaded_media(self, source_name))
31    }
32
33    /// Updates the four `[longitude, latitude]` corners of a `"video"` source without replacing
34    /// its current frame, matching GL JS `VideoSource.setCoordinates`.
35    pub fn set_video_coordinates(

37        source_name: &str,

81fn build_media_frame(
82    source_name: &str,
83    width: u32,
84    height: u32,
85    rgba: &[u8],
86) -> Result<MediaFrame, StyleMutationError> {
87    let expected = (width as usize)
88        .checked_mul(height as usize)
89        .and_then(|pixels| pixels.checked_mul(4));
90    if width == 0 || height == 0 || expected != Some(rgba.len()) {
91        return Err(StyleMutationError::InvalidMediaFrame {
92            source_name: source_name.to_owned(),
93            width,
94            height,
95            byte_len: rgba.len(),
96        });
97    }
98    Ok(MediaFrame {
99        width,
100        height,
101        rgba: Arc::from(rgba),
102    })
103}

105fn video_source_mut<'a>(
106    sources: &'a mut std::collections::HashMap<String, Source>,
107    name: &str,
108) -> Result<&'a mut VideoSource, StyleMutationError> {
109    match sources.get_mut(name) {
110        Some(Source::Video(video)) => Ok(video),
111        Some(_) => Err(StyleMutationError::NotAVideoSource {
112            source_name: name.to_owned(),
113        }),
114        None => Err(StyleMutationError::UnknownSource {
115            source_name: name.to_owned(),
116        }),
117    }
118}
119
120fn canvas_source_mut<'a>(
121    sources: &'a mut std::collections::HashMap<String, Source>,
122    name: &str,
123) -> Result<&'a mut CanvasSource, StyleMutationError> {
124    match sources.get_mut(name) {
125        Some(Source::Canvas(canvas)) => Ok(canvas),
126        Some(_) => Err(StyleMutationError::NotACanvasSource {
127            source_name: name.to_owned(),
128        }),
129        None => Err(StyleMutationError::UnknownSource {
130            source_name: name.to_owned(),
131        }),
132    }
133}

135fn reloaded_media(style: &Style, source_name: &str) -> StyleChange {
136    let layers = style
137        .layers
138        .iter()
139        .filter(|layer| layer.source.as_deref() == Some(source_name))
140        .map(|layer| layer.id.clone())
141        .collect();
142    StyleChange {
143        layers,
144        reloaded_image_sources: vec![source_name.to_owned()],
145        ..StyleChange::default()
146    }
