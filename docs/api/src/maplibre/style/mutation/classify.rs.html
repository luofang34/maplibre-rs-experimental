12pub(super) fn push_unique(ids: &mut Vec<String>, id: &str) {
13    if !ids.iter().any(|existing| existing == id) {
14        ids.push(id.to_owned());
15    }

18/// Sets `document[section][name]`, or removes it for `null`.
19pub(super) fn set_member(document: &mut Value, section: &str, name: &str, value: Value) {
20    let Some(map) = document.as_object_mut() else {
21        return;
22    };
23    if value.is_null() {
24        if let Some(Value::Object(members)) = map.get_mut(section) {
25            members.remove(name);
26        }
27        return;
28    }
29    let members = map.entry(section).or_insert_with(|| json!({}));
30    if let Some(members) = members.as_object_mut() {
31        members.insert(name.to_owned(), value);
32    }
33}

35fn same_prop<T: Serialize>(a: &T, b: &T) -> bool {
36    serde_json::to_value(a).ok() == serde_json::to_value(b).ok()
37}

39fn is_data_driven<T: crate::style::property::PropertyValue>(
40    prop: Option<&StyleProperty<T>>,
41) -> bool {
42    prop.is_some_and(|p| !p.is_feature_constant())
43}

45/// Returns `true` only when `before -> after` changes geometry, filtering, or layout baked by
46/// workers rather than layer order, visibility hiding, zoom bounds, or GPU paint buffers.
47pub(crate) fn needs_tile_redraw(before: &StyleLayer, after: &StyleLayer) -> bool {
48    if !super::from_vector_tiles(after) {
49        return false;
50    }
51    // Hidden layers are omitted from worker tile requests, so unhiding needs tessellation.
52    if before.is_hidden() && !after.is_hidden() {
53        return true;
54    }
55    // Hiding a visible layer updates GPU pool visibility in-place.
56    if after.is_hidden() {
57        return false;
58    }
59    if before.type_ != after.type_
60        || before.source != after.source
61        || before.source_layer != after.source_layer
62        || before.filter != after.filter
63        || before.metadata != after.metadata
64    {
65        return true;
66    }
67    let Ok(mut before_maps) = before.property_maps() else {
68        return true;
69    };
70    let Ok(mut after_maps) = after.property_maps() else {
71        return true;
72    };
73    before_maps.layout.remove("visibility");
74    after_maps.layout.remove("visibility");
75    if before_maps.layout != after_maps.layout {
76        return true;
77    }
78    needs_paint_retessellation(before.paint.as_ref(), after.paint.as_ref())
79}

81fn needs_paint_retessellation(before: Option<&LayerPaint>, after: Option<&LayerPaint>) -> bool {
82    match (before, after) {
83        (None, None) => false,
84        (Some(bp @ LayerPaint::Fill(a)), Some(ap @ LayerPaint::Fill(b))) => {
85            a.outline_color().is_some() != b.outline_color().is_some()
86                || ((per_feature_pattern(bp).is_some() || per_feature_pattern(ap).is_some())
87                    && a.fill_pattern != b.fill_pattern)
88        }
89        (Some(bp @ LayerPaint::Line(a)), Some(ap @ LayerPaint::Line(b))) => {
90            let per_feature_stroke = is_data_driven(a.line_width.as_ref())
91                || is_data_driven(b.line_width.as_ref())
92                || is_data_driven(a.line_offset.as_ref())
93                || is_data_driven(b.line_offset.as_ref())
94                || is_data_driven(a.line_gap_width.as_ref())
95                || is_data_driven(b.line_gap_width.as_ref())
96                || is_data_driven(a.line_blur.as_ref())
97                || is_data_driven(b.line_blur.as_ref());
98            (per_feature_stroke
99                && (!same_prop(&a.line_width, &b.line_width)
100                    || !same_prop(&a.line_offset, &b.line_offset)
101                    || !same_prop(&a.line_gap_width, &b.line_gap_width)
102                    || !same_prop(&a.line_blur, &b.line_blur)))
103                || a.line_gradient.is_some() != b.line_gradient.is_some()
104                || a.line_dasharray.is_some() != b.line_dasharray.is_some()
105                || a.line_pattern.is_some() != b.line_pattern.is_some()
106                || ((per_feature_pattern(bp).is_some() || per_feature_pattern(ap).is_some())
107                    && a.line_pattern != b.line_pattern)
108                || !same_prop(&a.line_elevation, &b.line_elevation)
109                || !same_prop(&a.line_elevation_datum, &b.line_elevation_datum)
110        }
111        (Some(LayerPaint::Circle(a)), Some(LayerPaint::Circle(b))) => {
112            !same_prop(&a.circle_radius, &b.circle_radius)
113                || !same_prop(&a.circle_stroke_width, &b.circle_stroke_width)
114                || a.circle_pitch_alignment != b.circle_pitch_alignment
115                || ((is_data_driven(a.circle_blur.as_ref())
116                    || is_data_driven(b.circle_blur.as_ref()))
117                    && !same_prop(&a.circle_blur, &b.circle_blur))
118                || ((is_data_driven(a.circle_stroke_opacity.as_ref())
119                    || is_data_driven(b.circle_stroke_opacity.as_ref()))
120                    && !same_prop(&a.circle_stroke_opacity, &b.circle_stroke_opacity))
121        }
122        (Some(LayerPaint::FillExtrusion(a)), Some(LayerPaint::FillExtrusion(b))) => {
123            !same_prop(&a.fill_extrusion_height, &b.fill_extrusion_height)
124                || !same_prop(&a.fill_extrusion_base, &b.fill_extrusion_base)
125                || !same_prop(&a.fill_extrusion_base_datum, &b.fill_extrusion_base_datum)
126                || !same_prop(
127                    &a.fill_extrusion_height_datum,
128                    &b.fill_extrusion_height_datum,
129                )
130                || !same_prop(&a.fill_extrusion_part, &b.fill_extrusion_part)
131                || a.fill_extrusion_translucent != b.fill_extrusion_translucent
132                || a.fill_extrusion_pattern != b.fill_extrusion_pattern
133        }
134        (Some(LayerPaint::Heatmap(a)), Some(LayerPaint::Heatmap(b))) => {
135            !same_prop(&a.heatmap_weight, &b.heatmap_weight)
136                || ((a.radius_per_feature() || b.radius_per_feature())
137                    && !same_prop(&a.heatmap_radius, &b.heatmap_radius))
138        }
139        (Some(LayerPaint::Symbol(_)), Some(LayerPaint::Symbol(_))) => false,
140        _ => true,
141    }
142}
