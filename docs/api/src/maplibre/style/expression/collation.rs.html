2//! on `wasm32`, plus the script check behind `is-supported-script`.

14/// How two strings are compared, as `Intl.Collator` sensitivities do.
15#[derive(Clone, Debug, PartialEq, Eq)]
16pub struct Collation {
17    /// Whether letters that differ only in case are different.
18    pub case_sensitive: bool,
19    /// Whether letters that differ only in their accents are different.
20    pub diacritic_sensitive: bool,
21    /// The locale the strings are compared in, if the style names one.
22    pub locale: Option<String>,
23}
24
25#[cfg(not(target_arch = "wasm32"))]
26impl Collation {
27    /// The canonical BCP-47 locale tag the collation resolves to; unset or invalid tags fall back
28    /// to `"en"`.
29    pub fn resolved_locale(&self) -> String {
30        self.parsed_locale().to_string()
31    }
32
33    /// Orders two strings using ICU4X compiled collation data and ECMA-402 sensitivity rules.
34    pub fn compare(&self, a: &str, b: &str) -> Ordering {
35        let locale = self.parsed_locale();
36        let options = self.collator_options();
37        match CollatorBorrowed::try_new((&locale).into(), options)
38            .or_else(|_| CollatorBorrowed::try_new(Default::default(), options))
39        {
40            Ok(collator) => collator.compare(a, b),
41            Err(_) => a.cmp(b),
42        }
43    }
44
45    fn parsed_locale(&self) -> Locale {
46        self.locale
47            .as_deref()
48            .and_then(|tag| Locale::try_from_str(tag.trim()).ok())
49            .filter(|locale| !locale.id.language.is_unknown())
50            .unwrap_or(locale!("en"))
51    }
52
53    fn collator_options(&self) -> CollatorOptions {
54        let mut options = CollatorOptions::default();
55        match (self.case_sensitive, self.diacritic_sensitive) {
56            (false, false) => {
57                options.strength = Some(Strength::Primary);
58                options.case_level = Some(CaseLevel::Off);
59            }
60            (false, true) => {
61                options.strength = Some(Strength::Secondary);
62                options.case_level = Some(CaseLevel::Off);
63            }
64            (true, false) => {
65                options.strength = Some(Strength::Primary);
66                options.case_level = Some(CaseLevel::On);
67            }
68            (true, true) => {
69                options.strength = Some(Strength::Tertiary);
70                options.case_level = Some(CaseLevel::Off);
71            }
72        }
73        options
74    }
75}

77#[cfg(target_arch = "wasm32")]
78fn canonicalize_bcp47(raw: &str) -> Option<String> {
79    use wasm_bindgen::{JsCast, JsValue};
80    let intl = js_sys::Reflect::get(&js_sys::global(), &JsValue::from_str("Intl")).ok()?;
81    let func = js_sys::Reflect::get(&intl, &JsValue::from_str("getCanonicalLocales"))
82        .ok()?
83        .dyn_into::<js_sys::Function>()
84        .ok()?;
85    let arr = func
86        .call1(&intl, &JsValue::from_str(raw))
87        .ok()?
88        .dyn_into::<js_sys::Array>()
89        .ok()?;
90    let first = arr.get(0).as_string()?;
91    (!first.starts_with("und")).then_some(first)
92}

94#[cfg(target_arch = "wasm32")]
95impl Collation {
96    /// The canonical BCP-47 locale tag the collation resolves to; unset or invalid tags fall back
97    /// to `"en"`.
98    pub fn resolved_locale(&self) -> String {
99        self.locale
100            .as_deref()
101            .map(str::trim)
102            .filter(|s| !s.is_empty())
103            .and_then(canonicalize_bcp47)
104            .unwrap_or_else(|| "en".to_owned())
105    }
106
107    /// Orders two strings using the browser's built-in ECMA-402 `Intl.Collator`.
108    pub fn compare(&self, a: &str, b: &str) -> Ordering {
109        use wasm_bindgen::{JsCast, JsValue};
110        let loc = self.resolved_locale();
111        let sensitivity = match (self.case_sensitive, self.diacritic_sensitive) {
112            (false, false) => "base",
113            (false, true) => "accent",
114            (true, false) => "case",
115            (true, true) => "variant",
116        };
117        let opts = js_sys::Object::new();
118        let _ = js_sys::Reflect::set(
119            &opts,
120            &JsValue::from_str("sensitivity"),
121            &JsValue::from_str(sensitivity),
122        );
123        let locales = js_sys::Array::of1(&JsValue::from_str(&loc));
124        let cmp_res = js_sys::Reflect::get(&js_sys::global(), &JsValue::from_str("Intl"))
125            .and_then(|intl| js_sys::Reflect::get(&intl, &JsValue::from_str("Collator")))
126            .and_then(|ctor| ctor.dyn_into::<js_sys::Function>())
127            .and_then(|ctor| {
128                js_sys::Reflect::construct(&ctor, &js_sys::Array::of2(&locales, &opts))
129            })
130            .and_then(|collator| js_sys::Reflect::get(&collator, &JsValue::from_str("compare")))
131            .and_then(|cmp| cmp.dyn_into::<js_sys::Function>())
132            .and_then(|cmp| {
133                cmp.call2(&JsValue::NULL, &JsValue::from_str(a), &JsValue::from_str(b))
134            });
135        match cmp_res.ok().and_then(|v| v.as_f64()) {
136            Some(diff) if diff < 0.0 => Ordering::Less,
137            Some(diff) if diff > 0.0 => Ordering::Greater,
138            Some(_) => Ordering::Equal,
139            None => a.cmp(b),
140        }
141    }
142}

144/// Scripts lacking shaping support (empty: complex Indic, Tibetan, Myanmar, Khmer, and RTL
145/// scripts are shaped via `harfrust` / browser shaping and `arabic_shaping`).
146const UNSUPPORTED_SCRIPTS: &[(u32, u32)] = &[];
147
148/// Whether every character of `text` is in a script the renderer can lay out.
149pub fn is_supported_script(text: &str) -> bool {
150    text.chars().all(|character| {
151        let code = u32::from(character);
152        !UNSUPPORTED_SCRIPTS
153            .iter()
154            .any(|(first, last)| (*first..=*last).contains(&code))
155    })
156}
