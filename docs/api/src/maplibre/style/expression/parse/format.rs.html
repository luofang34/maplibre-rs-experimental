1//! `["format", ...]`: text in sections that differ in size, colour and font, any of which may
2//! be an image.
3//!
4//! The control characters below are the form a [`crate::style::property::TextField`] with
5//! sections is written in, so a layer serialized and parsed again keeps its sections.
6
7use serde_json::Value as Json;
8
9use super::{Parser, Result};
10use crate::style::expression::{
11    ast::{Expression, FormatSection},
12    value::Type,
13};
14
15/// Starts the written form of a formatted text.
16pub const FORMATTED_START: char = '\u{1}';
17/// Starts each section.
18pub const SECTION: char = '\u{2}';
19/// Separates the fields of a section: font scale, colour, font stack, text, image name.
20pub const FIELD: char = '\u{3}';
21/// The character that stands for an image in the text of a section.
22pub const IMAGE_PLACEHOLDER: char = '\u{E000}';
23
24const VERTICAL_ALIGNMENTS: [&str; 3] = ["bottom", "center", "top"];
25
26impl Parser {
27    /// Parses the arguments of a `format` expression as GL JS does: each content, typed as any
28    /// value, may be followed by an object of options for its section.
29    pub(super) fn parse_format(&mut self, args: &[Json]) -> Result<Expression> {
30        match args.first() {
31            None => return Err(self.error("Expected at least one argument.")),
32            Some(Json::Object(_)) => {
33                return Err(self.error("First argument must be an image or text section."))
34            }
35            Some(_) => {}
36        }
37        let mut sections: Vec<FormatSection> = Vec::new();
38        let mut options_may_follow = false;
39        for (offset, arg) in args.iter().enumerate() {
40            let index = offset + 1;
41            match (arg, sections.last_mut()) {
42                (Json::Object(options), Some(_)) if options_may_follow => {
43                    options_may_follow = false;
44                    let styled = self.parse_section_options(options, index)?;
45                    if let Some(section) = sections.last_mut() {
46                        section.scale = styled.scale;
47                        section.font = styled.font;
48                        section.color = styled.color;
49                        section.vertical_align = styled.vertical_align;
50                    }
51                }
52                _ => {
53                    let content = self.parse_at(arg, index, Some(&Type::Value))?;
54                    let kind = content.output_type();
55                    if !matches!(
56                        kind,
57                        Type::String | Type::Value | Type::Null | Type::ResolvedImage
58                    ) {
59                        return Err(self.error(
60                            "Formatted text type must be 'string', 'value', 'image' or 'null'.",
61                        ));
62                    }
63                    options_may_follow = true;
64                    sections.push(FormatSection {
65                        content,
66                        scale: None,
67                        font: None,
68                        color: None,
69                        vertical_align: None,
70                    });
71                }
72            }
73        }
74        Ok(Expression::Format(sections))
75    }
76
77    /// The options object of a section: what each styles, parsed into an otherwise empty
78    /// section.
79    fn parse_section_options(
80        &mut self,
81        options: &serde_json::Map<String, Json>,
82        index: usize,
83    ) -> Result<FormatSection> {
84        let option = |parser: &mut Self, name: &str, expected: Type| {
85            options
86                .get(name)
87                .map(|value| parser.parse_at(value, index, Some(&expected)))
88                .transpose()
89        };
90        if let Some(Json::String(align)) = options.get("vertical-align") {
91            if !VERTICAL_ALIGNMENTS.contains(&align.as_str()) {
92                return Err(self.error(format!(
93                    "'vertical-align' must be one of: 'bottom', 'center', 'top' but found \
94                     '{align}' instead."
95                )));
96            }
97        }
98        Ok(FormatSection {
99            content: Expression::Literal(crate::style::expression::Value::Null),
100            scale: option(self, "font-scale", Type::Number)?,
101            font: option(self, "text-font", Type::array(Type::String, None))?,
102            color: option(self, "text-color", Type::Color)?,
103            vertical_align: option(self, "vertical-align", Type::String)?,
104        })
105    }
