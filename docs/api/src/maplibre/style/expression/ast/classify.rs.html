1//! Static output-type and constantness classification for [`Expression`] trees.
2
3use super::{Coercion, Expression, FeatureProperty, Global};
4use crate::style::expression::value::Type;
5
6impl Expression {
7    /// The type the expression evaluates to, as far as it is known before evaluation.
8    pub fn output_type(&self) -> Type {
9        match self {
10            Self::Literal(value) => value.type_of(),
11            Self::Folded { output, .. } => output.clone(),
12            Self::Error(_) => Type::Error,
13            Self::Global(_) => Type::Number,
14            Self::Feature(FeatureProperty::Id) | Self::Accumulated => Type::Value,
15            Self::Feature(FeatureProperty::GeometryType) => Type::String,
16            Self::Feature(FeatureProperty::Properties) => Type::Object,
17            Self::Get { .. } | Self::GlobalState(_) => Type::Value,
18            Self::Has { .. }
19            | Self::Compare { .. }
20            | Self::All(_)
21            | Self::Any(_)
22            | Self::Not(_)
23            | Self::IsSupportedScript(_)
24            | Self::Within(_)
25            | Self::In { .. } => Type::Boolean,
26            Self::Distance(_) => Type::Number,
27            Self::Collator { .. } => Type::Collator,
28            Self::Var { bound, .. } => bound.output_type(),
29            Self::Let { body, .. } => body.output_type(),
30            Self::Case { output, .. }
31            | Self::Match { output, .. }
32            | Self::Coalesce { output, .. }
33            | Self::Slice { output, .. }
34            | Self::At { output, .. }
35            | Self::Interpolate { output, .. }
36            | Self::Step { output, .. } => output.clone(),
37            Self::IndexOf { .. }
38            | Self::Length(_)
39            | Self::Arithmetic { .. }
40            | Self::Math { .. }
41            | Self::MinMax { .. } => Type::Number,
42            Self::TypeOf(_)
43            | Self::Concat(_)
44            | Self::Join { .. }
45            | Self::StringCase { .. }
46            | Self::ResolvedLocale(_)
47            | Self::NumberFormat { .. } => Type::String,
48            Self::Split { .. } => Type::array(Type::String, None),
49            Self::Image(_) => Type::ResolvedImage,
50            Self::Format(_) => Type::Formatted,
51            Self::Assert { required, .. } => required.clone(),
52            Self::Coerce { coercion, .. } => match coercion {
53                Coercion::Number => Type::Number,
54                Coercion::String => Type::String,
55                Coercion::Boolean => Type::Boolean,
56                Coercion::Color => Type::Color,
57                Coercion::Formatted => Type::Formatted,
58                Coercion::ResolvedImage => Type::ResolvedImage,
59                Coercion::Padding => Type::Padding,
60                Coercion::NumberArray => Type::NumberArray,
61                Coercion::ColorArray => Type::ColorArray,
62                Coercion::ProjectionDefinition => Type::ProjectionDefinition,
63                Coercion::VariableAnchorOffsetCollection => Type::VariableAnchorOffsetCollection,
64            },
65            Self::ToRgba(_) => Type::array(Type::Number, Some(4)),
66            Self::Rgba(_) => Type::Color,
67        }
68    }
69
70    /// Whether the result is the same for every feature.
71    pub fn is_feature_constant(&self) -> bool {
72        match self {
73            Self::Feature(_) | Self::Within(_) | Self::Distance(_) | Self::Accumulated => false,
74            Self::Get { object: None, .. } | Self::Has { object: None, .. } => false,
75            _ => self.children_all(Self::is_feature_constant),
76        }
77    }
78
79    /// Whether the result is the same at every zoom.
80    pub fn is_zoom_constant(&self) -> bool {
81        match self {
82            Self::Global(Global::Zoom) => false,
83            _ => self.children_all(Self::is_zoom_constant),
84        }
85    }
86
87    /// Whether the result depends on neither the feature nor the map.
88    pub fn is_constant(&self) -> bool {
89        match self {
90            // Whether an image exists is known only where it is drawn.
91            Self::Global(_) | Self::GlobalState(_) | Self::Image(_) => false,
92            _ => self.is_feature_constant() && self.children_all(Self::is_constant),
93        }
94    }
95
96    fn children_all(&self, predicate: fn(&Expression) -> bool) -> bool {
97        let mut all = true;
98        self.for_each_child(&mut |child| {
99            if all && !predicate(child) {
100                all = false;
101            }
102        });
103        all
104    }
105}
