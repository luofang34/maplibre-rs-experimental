1//! The direct children of an expression node, which classification walks.
2
3use super::Expression;
4
5impl Expression {
6    /// Calls `visit` on every direct child.
7    pub fn for_each_child<'a>(&'a self, visit: &mut dyn FnMut(&'a Expression)) {
8        match self {
9            Self::Literal(_)
10            | Self::Folded { .. }
11            | Self::Global(_)
12            | Self::Feature(_)
13            | Self::Within(_)
14            | Self::Distance(_)
15            | Self::GlobalState(_) => {}
16            Self::NumberFormat { input, options } => {
17                visit(input);
18                options.iter().for_each(|(_, option)| visit(option));
19            }
20            Self::Get { key, object } | Self::Has { key, object } => {
21                visit(key);
22                if let Some(object) = object {
23                    visit(object);
24                }
25            }
26            Self::Var { bound, .. } => visit(bound),
27            Self::Let { bindings, body } => {
28                for (_, bound) in bindings {
29                    visit(bound);
30                }
31                visit(body);
32            }
33            Self::Case {
34                branches, fallback, ..
35            } => {
36                for (condition, output) in branches {
37                    visit(condition);
38                    visit(output);
39                }
40                visit(fallback);
41            }
42            Self::Match {
43                input,
44                cases,
45                fallback,
46                ..
47            } => {
48                visit(input);
49                for (_, output) in cases {
50                    visit(output);
51                }
52                visit(fallback);
53            }
54            Self::Coalesce { operands, .. }
55            | Self::All(operands)
56            | Self::Any(operands)
57            | Self::Arithmetic { operands, .. }
58            | Self::MinMax { operands, .. }
59            | Self::Assert { operands, .. }
60            | Self::Coerce { operands, .. }
61            | Self::Rgba(operands)
62            | Self::Concat(operands) => operands.iter().for_each(visit),
63            Self::Compare {
64                left,
65                right,
66                collator,
67                ..
68            } => {
69                visit(left);
70                visit(right);
71                if let Some(collator) = collator {
72                    visit(collator);
73                }
74            }
75            Self::Collator {
76                case_sensitive,
77                diacritic_sensitive,
78                locale,
79            } => {
80                visit(case_sensitive);
81                visit(diacritic_sensitive);
82                if let Some(locale) = locale {
83                    visit(locale);
84                }
85            }
86            Self::Not(operand)
87            | Self::ResolvedLocale(operand)
88            | Self::IsSupportedScript(operand)
89            | Self::Length(operand)
90            | Self::Math { operand, .. }
91            | Self::TypeOf(operand)
92            | Self::Image(operand)
93            | Self::ToRgba(operand)
94            | Self::StringCase { operand, .. } => visit(operand),
95            Self::In { needle, haystack } => {
96                visit(needle);
97                visit(haystack);
98            }
99            Self::IndexOf {
100                needle,
101                haystack,
102                from,
103            } => {
104                visit(needle);
105                visit(haystack);
106                if let Some(from) = from {
107                    visit(from);
108                }
109            }
110            Self::Slice {
111                input, from, to, ..
112            } => {
113                visit(input);
114                visit(from);
115                if let Some(to) = to {
116                    visit(to);
117                }
118            }
119            Self::Interpolate { input, stops, .. } | Self::Step { input, stops, .. } => {
120                visit(input);
121                for (_, output) in stops {
122                    visit(output);
123                }
124            }
125            Self::Format(sections) => sections.iter().for_each(|section| section.for_each(visit)),
126        }
127    }
128}
