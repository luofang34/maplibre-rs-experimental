12/// The kind of value a style property holds, which decides how legacy syntax is lowered.
13#[derive(Clone, Debug, PartialEq)]
14pub enum PropertyKind {
15    /// Any value; what a property without a specification accepts.
16    Value,
17    /// A number.
18    Number,
19    /// A string.
20    String,
21    /// A boolean.
22    Boolean,
23    /// A colour.
24    Color,
25    /// One of a fixed set of strings.
26    Enum(Vec<String>),
27    /// Text in sections, any of which may be an image: what `text-field` holds.
28    Formatted,
29    /// An image name, with whether the map holds the image.
30    ResolvedImage,
31    /// An array of one item kind, with a fixed length when given.
32    Array {
33        /// Kind of every item.
34        item: Box<PropertyKind>,
35        /// Number of items.
36        length: Option<usize>,
37    },
38}
39
40impl PropertyKind {
41    /// The expression type a property of this kind must produce.
42    pub fn expected_type(&self) -> Type {
43        match self {
44            Self::Value => Type::Value,
45            Self::Number => Type::Number,
46            Self::String | Self::Enum(_) => Type::String,
47            Self::Boolean => Type::Boolean,
48            Self::Color => Type::Color,
49            Self::Formatted => Type::Formatted,
50            Self::ResolvedImage => Type::ResolvedImage,
51            Self::Array { item, length } => Type::array(item.expected_type(), *length),
52        }
53    }
54
55    fn assertion_name(&self) -> &'static str {
56        match self {
57            // The root annotation turns whatever the function reads into the property's type.
58            Self::Value | Self::Formatted | Self::ResolvedImage => "coalesce",
59            Self::Number => "number",
60            Self::String | Self::Enum(_) => "string",
61            Self::Boolean => "boolean",
62            Self::Color => "to-color",
63            Self::Array { .. } => "array",
64        }
65    }
66}

68/// What the specification says about a property, as far as legacy syntax needs it.
69#[derive(Clone, Debug, PartialEq)]
70pub struct LegacyPropertySpec {
71    /// Kind of value the property holds.
72    pub kind: PropertyKind,
73    /// Whether the property interpolates between stops; otherwise stops are steps.
74    pub interpolated: bool,
75    /// The specification's default, used when a function names none.
76    pub default: Option<Json>,
77    /// Whether string values may hold `{token}` references to feature properties.
78    pub tokens: bool,
79}
80
81impl LegacyPropertySpec {
82    /// A spec for an interpolated property of `kind` with no default.
83    pub fn interpolated(kind: PropertyKind) -> Self {
84        Self {
85            kind,
86            interpolated: true,
87            default: None,
88            tokens: false,
89        }
90    }
91
92    /// A spec for a stepped property of `kind` with no default.
93    pub fn stepped(kind: PropertyKind) -> Self {
94        Self {
95            kind,
96            interpolated: false,
97            default: None,
98            tokens: false,
99        }
100    }
101
102    /// The expression type the property must produce.
103    pub fn expected_type(&self) -> Type {
104        self.kind.expected_type()
105    }
106}

108fn literal(value: &Json) -> Json {
109    match value {
110        Json::Array(_) | Json::Object(_) => json!(["literal", value]),
111        other => other.clone(),
112    }
113}

115/// Lowers a legacy function object into an expression, as GL JS `convertFunction` does.
116pub fn convert_function(parameters: &Map<String, Json>, spec: &LegacyPropertySpec) -> Json {
117    let Some(stops) = parameters.get("stops").and_then(Json::as_array) else {
118        return convert_identity_function(parameters, spec);
119    };
120    let zoom_and_feature = stops
121        .first()
122        .and_then(|stop| stop.get(0))
123        .is_some_and(Json::is_object);
124    let feature_dependent = zoom_and_feature || parameters.contains_key("property");
125    let zoom_dependent = zoom_and_feature || !feature_dependent;
126    let stops: Vec<(Json, Json)> = stops
127        .iter()
128        .map(|stop| {
129            let input = stop.get(0).cloned().unwrap_or(Json::Null);
130            let output = stop.get(1).cloned().unwrap_or(Json::Null);
131            let output = match &output {
132                Json::String(text) if !feature_dependent && spec.tokens => {
133                    convert_token_string(text)
134                }
135                other => literal(other),
136            };
137            (input, output)
138        })
139        .collect();
140    if zoom_and_feature {
141        convert_zoom_and_property_function(parameters, spec, &stops)
142    } else if zoom_dependent {
143        convert_zoom_function(parameters, spec, &stops)
144    } else {
145        convert_property_function(parameters, spec, &stops)
146    }
147}

149fn convert_identity_function(parameters: &Map<String, Json>, spec: &LegacyPropertySpec) -> Json {
150    let get = json!([
151        "get",
152        parameters.get("property").cloned().unwrap_or(Json::Null)
153    ]);
154    match parameters.get("default") {
155        None => match spec.kind {
156            PropertyKind::String => json!(["string", get]),
157            _ => get,
158        },
159        Some(default) => match &spec.kind {
160            PropertyKind::Enum(values) => json!(["match", get, values, get, default]),
161            PropertyKind::Array { item, length } => json!([
162                "array",
163                item.assertion_name(),
164                length.map_or(Json::Null, |length| json!(length)),
165                get,
166                literal(default)
167            ]),
168            kind => json!([kind.assertion_name(), get, literal(default)]),
169        },
170    }
171}

173fn interpolate_operator(parameters: &Map<String, Json>) -> &'static str {
174    match parameters.get("colorSpace").and_then(Json::as_str) {
175        Some("hcl") => "interpolate-hcl",
176        Some("lab") => "interpolate-lab",
177        _ => "interpolate",
178    }
179}

181fn function_type<'a>(parameters: &'a Map<String, Json>, spec: &LegacyPropertySpec) -> &'a str {
182    match parameters.get("type").and_then(Json::as_str) {
183        Some(kind) => kind,
184        None if spec.interpolated => "exponential",
185        None => "interval",
186    }
187}

189fn convert_zoom_and_property_function(
190    parameters: &Map<String, Json>,
191    spec: &LegacyPropertySpec,
192    stops: &[(Json, Json)],
193) -> Json {
194    let mut zoom_stops: Vec<(f64, Vec<(Json, Json)>)> = Vec::new();
195    for (input, output) in stops {
196        let zoom = input.get("zoom").and_then(Json::as_f64).unwrap_or(0.0);
197        let value = input.get("value").cloned().unwrap_or(Json::Null);
198        match zoom_stops.iter_mut().find(|(stop, _)| *stop == zoom) {
199            Some((_, group)) => group.push((value, output.clone())),
200            None => zoom_stops.push((zoom, vec![(value, output.clone())])),
201        }
202    }
203    let mut feature_parameters = Map::new();
204    for key in ["type", "property", "default"] {
205        if let Some(value) = parameters.get(key) {
206            feature_parameters.insert(key.to_string(), value.clone());
207        }
208    }
209    let exponential = function_type(&Map::new(), spec) == "exponential";
210    let mut expression = if exponential {
211        json!([interpolate_operator(parameters), ["linear"], ["zoom"]])
212    } else {
213        json!(["step", ["zoom"]])
214    };
215    for (zoom, group) in &zoom_stops {
216        let output = convert_property_function(&feature_parameters, spec, group);
217        append_stop_pair(&mut expression, json!(zoom), output, !exponential);
218    }
219    if !exponential {
220        fixup_degenerate_step_curve(&mut expression);
221    }
222    expression
223}

225fn fallback(parameters: &Map<String, Json>, spec: &LegacyPropertySpec) -> Json {
226    parameters
227        .get("default")
228        .or(spec.default.as_ref())
229        .map_or(Json::Null, literal)
230}

232fn convert_property_function(
233    parameters: &Map<String, Json>,
234    spec: &LegacyPropertySpec,
235    stops: &[(Json, Json)],
236) -> Json {
237    let kind = function_type(parameters, spec);
238    let get = json!([
239        "get",
240        parameters.get("property").cloned().unwrap_or(Json::Null)
241    ]);
242    match kind {
243        "categorical" if stops.first().is_some_and(|(input, _)| input.is_boolean()) => {
244            let mut expression = json!(["case"]);
245            for (input, output) in stops {
246                push(&mut expression, json!(["==", get, input]));
247                push(&mut expression, output.clone());
248            }
249            push(&mut expression, fallback(parameters, spec));
250            expression
251        }
252        "categorical" => {
253            let mut expression = json!(["match", get]);
254            for (input, output) in stops {
255                append_stop_pair(&mut expression, input.clone(), output.clone(), false);
256            }
257            push(&mut expression, fallback(parameters, spec));
258            expression
259        }
260        "interval" => {
261            let mut expression = json!(["step", ["number", get]]);
262            for (input, output) in stops {
263                append_stop_pair(&mut expression, input.clone(), output.clone(), true);
264            }
265            fixup_degenerate_step_curve(&mut expression);
266            guard_default(parameters, &get, expression)
267        }
268        _ => {
269            let base = parameters.get("base").and_then(Json::as_f64).unwrap_or(1.0);
270            let curve = if base == 1.0 {
271                json!(["linear"])
272            } else {
273                json!(["exponential", base])
274            };
275            let mut expression = json!([interpolate_operator(parameters), curve, ["number", get]]);
276            for (input, output) in stops {
277                append_stop_pair(&mut expression, input.clone(), output.clone(), false);
278            }
279            guard_default(parameters, &get, expression)
280        }
281    }
282}

284/// Wraps a numeric ramp so a feature without a numeric value takes the default.
285fn guard_default(parameters: &Map<String, Json>, get: &Json, expression: Json) -> Json {
286    match parameters.get("default") {
287        None => expression,
288        Some(default) => json!([
289            "case",
290            ["==", ["typeof", get], "number"],
291            expression,
292            literal(default)
293        ]),
294    }
295}

297fn convert_zoom_function(
298    parameters: &Map<String, Json>,
299    spec: &LegacyPropertySpec,
300    stops: &[(Json, Json)],
301) -> Json {
302    let kind = function_type(parameters, spec);
303    let is_step = kind == "interval";
304    let mut expression = if is_step {
305        json!(["step", ["zoom"]])
306    } else {
307        let base = parameters.get("base").and_then(Json::as_f64).unwrap_or(1.0);
308        let curve = if base == 1.0 {
309            json!(["linear"])
310        } else {
311            json!(["exponential", base])
312        };
313        json!([interpolate_operator(parameters), curve, ["zoom"]])
314    };
315    for (input, output) in stops {
316        append_stop_pair(&mut expression, input.clone(), output.clone(), is_step);
317    }
318    fixup_degenerate_step_curve(&mut expression);
319    expression
320}

322pub(super) fn push(expression: &mut Json, value: Json) {
323    if let Json::Array(items) = expression {
324        items.push(value);
325    }
326}

328fn fixup_degenerate_step_curve(expression: &mut Json) {
329    if let Json::Array(items) = expression {
330        if items.first().and_then(Json::as_str) == Some("step") && items.len() == 3 {
331            let output = items[2].clone();
332            items.push(json!(0));
333            items.push(output);
334        }
335    }
336}
337
338fn append_stop_pair(curve: &mut Json, input: Json, output: Json, is_step: bool) {
339    let Json::Array(items) = curve else {
340        return;
341    };
342    if items.len() > 3 && items[items.len() - 2] == input {
343        return;
344    }
345    if !(is_step && items.len() == 2) {
346        items.push(input);
347    }
348    items.push(output);
349}

351/// Turns a `{token}` string into a `concat` of literals and property reads.
352pub fn convert_token_string(text: &str) -> Json {
353    let mut parts: Vec<Json> = Vec::new();
354    let mut rest = text;
355    while let Some(open) = rest.find('{') {
356        let Some(close) = rest[open..].find('}') else {
357            break;
358        };
359        let name = &rest[open + 1..open + close];
360        if name.is_empty() || name.contains('{') {
361            break;
362        }
363        if open > 0 {
364            parts.push(Json::String(rest[..open].to_string()));
365        }
366        parts.push(json!(["get", name]));
367        rest = &rest[open + close + 1..];
368    }
369    if parts.is_empty() {
370        return Json::String(text.to_string());
371    }
372    if !rest.is_empty() {
373        parts.push(Json::String(rest.to_string()));
374    } else if parts.len() == 1 {
375        return json!(["to-string", parts[0]]);
376    }
377    let mut expression = json!(["concat"]);
378    for part in parts {
379        push(&mut expression, part);
380    }
381    expression
