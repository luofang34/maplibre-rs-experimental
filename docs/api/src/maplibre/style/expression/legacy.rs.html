12/// The kind of value a style property holds, which decides how legacy syntax is lowered.
13#[derive(Clone, Debug, PartialEq)]
14pub enum PropertyKind {
15    /// Any value; what a property without a specification accepts.
16    Value,
17    /// A number.
18    Number,
19    /// A string.
20    String,
21    /// A boolean.
22    Boolean,
23    /// A colour.
24    Color,
25    /// One of a fixed set of strings.
26    Enum(Vec<String>),
27    /// Text in sections, any of which may be an image: what `text-field` holds.
28    Formatted,
29    /// An image name, with whether the map holds the image.
30    ResolvedImage,
31    /// Directional `[top, right, bottom, left]` box padding (`padding`).
32    Padding,
33    /// Variable-length numeric list (`numberArray`).
34    NumberArray,
35    /// Variable-length colour list (`colorArray`).
36    ColorArray,
37    /// Map projection name or blend (`projectionDefinition`).
38    ProjectionDefinition,
39    /// Alternating anchor and `[x, y]` offset pairs (`variableAnchorOffsetCollection`).
40    VariableAnchorOffsetCollection,
41    /// An array of one item kind, with a fixed length when given.
42    Array {
43        /// Kind of every item.
44        item: Box<PropertyKind>,
45        /// Number of items.
46        length: Option<usize>,
47    },
48}
49
50impl PropertyKind {
51    /// The expression type a property of this kind must produce.
52    pub fn expected_type(&self) -> Type {
53        match self {
54            Self::Value => Type::Value,
55            Self::Number => Type::Number,
56            Self::String | Self::Enum(_) => Type::String,
57            Self::Boolean => Type::Boolean,
58            Self::Color => Type::Color,
59            Self::Formatted => Type::Formatted,
60            Self::ResolvedImage => Type::ResolvedImage,
61            Self::Padding => Type::Padding,
62            Self::NumberArray => Type::NumberArray,
63            Self::ColorArray => Type::ColorArray,
64            Self::ProjectionDefinition => Type::ProjectionDefinition,
65            Self::VariableAnchorOffsetCollection => Type::VariableAnchorOffsetCollection,
66            Self::Array { item, length } => Type::array(item.expected_type(), *length),
67        }
68    }
69
70    fn assertion_name(&self) -> &'static str {
71        match self {
72            // The root annotation turns whatever the function reads into the property's type.
73            Self::Value
74            | Self::Formatted
75            | Self::ResolvedImage
76            | Self::Padding
77            | Self::NumberArray
78            | Self::ColorArray
79            | Self::ProjectionDefinition
80            | Self::VariableAnchorOffsetCollection => "coalesce",
81            Self::Number => "number",
82            Self::String | Self::Enum(_) => "string",
83            Self::Boolean => "boolean",
84            Self::Color => "to-color",
85            Self::Array { .. } => "array",
86        }
87    }
88}

90/// What the specification says about a property, as far as legacy syntax needs it.
91#[derive(Clone, Debug, PartialEq)]
92pub struct LegacyPropertySpec {
93    /// Kind of value the property holds.
94    pub kind: PropertyKind,
95    /// Whether the property interpolates between stops; otherwise stops are steps.
96    pub interpolated: bool,
97    /// The specification's default, used when a function names none.
98    pub default: Option<Json>,
99    /// Whether string values may hold `{token}` references to feature properties.
100    pub tokens: bool,
101}
102
103impl LegacyPropertySpec {
104    /// A spec for an interpolated property of `kind` with no default.
105    pub fn interpolated(kind: PropertyKind) -> Self {
106        Self {
107            kind,
108            interpolated: true,
109            default: None,
110            tokens: false,
111        }
112    }
113
114    /// A spec for a stepped property of `kind` with no default.
115    pub fn stepped(kind: PropertyKind) -> Self {
116        Self {
117            kind,
118            interpolated: false,
119            default: None,
120            tokens: false,
121        }
122    }
123
124    /// The expression type the property must produce.
125    pub fn expected_type(&self) -> Type {
126        self.kind.expected_type()
127    }
128}

130fn literal(value: &Json) -> Json {
131    match value {
132        Json::Array(_) | Json::Object(_) => json!(["literal", value]),
133        other => other.clone(),
134    }
135}

137/// Lowers a legacy function object into an expression, as GL JS `convertFunction` does.
138pub fn convert_function(parameters: &Map<String, Json>, spec: &LegacyPropertySpec) -> Json {
139    let Some(stops) = parameters.get("stops").and_then(Json::as_array) else {
140        return convert_identity_function(parameters, spec);
141    };
142    let zoom_and_feature = stops
143        .first()
144        .and_then(|stop| stop.get(0))
145        .is_some_and(Json::is_object);
146    let feature_dependent = zoom_and_feature || parameters.contains_key("property");
147    let zoom_dependent = zoom_and_feature || !feature_dependent;
148    let stops: Vec<(Json, Json)> = stops
149        .iter()
150        .map(|stop| {
151            let input = stop.get(0).cloned().unwrap_or(Json::Null);
152            let output = stop.get(1).cloned().unwrap_or(Json::Null);
153            let output = match &output {
154                Json::String(text) if !feature_dependent && spec.tokens => {
155                    convert_token_string(text)
156                }
157                other => literal(other),
158            };
159            (input, output)
160        })
161        .collect();
162    if zoom_and_feature {
163        convert_zoom_and_property_function(parameters, spec, &stops)
164    } else if zoom_dependent {
165        convert_zoom_function(parameters, spec, &stops)
166    } else {
167        convert_property_function(parameters, spec, &stops)
168    }
169}

171fn convert_identity_function(parameters: &Map<String, Json>, spec: &LegacyPropertySpec) -> Json {
172    let get = json!([
173        "get",
174        parameters.get("property").cloned().unwrap_or(Json::Null)
175    ]);
176    match parameters.get("default") {
177        None => match spec.kind {
178            PropertyKind::String => json!(["string", get]),
179            _ => get,
180        },
181        Some(default) => match &spec.kind {
182            PropertyKind::Enum(values) => json!(["match", get, values, get, default]),
183            PropertyKind::Array { item, length } => json!([
184                "array",
185                item.assertion_name(),
186                length.map_or(Json::Null, |length| json!(length)),
187                get,
188                literal(default)
189            ]),
190            kind => json!([kind.assertion_name(), get, literal(default)]),
191        },
192    }
193}

195fn interpolate_operator(parameters: &Map<String, Json>) -> &'static str {
196    match parameters.get("colorSpace").and_then(Json::as_str) {
197        Some("hcl") => "interpolate-hcl",
198        Some("lab") => "interpolate-lab",
199        _ => "interpolate",
200    }
201}

203fn function_type<'a>(parameters: &'a Map<String, Json>, spec: &LegacyPropertySpec) -> &'a str {
204    match parameters.get("type").and_then(Json::as_str) {
205        Some(kind) => kind,
206        None if spec.interpolated => "exponential",
207        None => "interval",
208    }
209}

211fn convert_zoom_and_property_function(
212    parameters: &Map<String, Json>,
213    spec: &LegacyPropertySpec,
214    stops: &[(Json, Json)],
215) -> Json {
216    let mut zoom_stops: Vec<(f64, Vec<(Json, Json)>)> = Vec::new();
217    for (input, output) in stops {
218        let zoom = input.get("zoom").and_then(Json::as_f64).unwrap_or(0.0);
219        let value = input.get("value").cloned().unwrap_or(Json::Null);
220        match zoom_stops.iter_mut().find(|(stop, _)| *stop == zoom) {
221            Some((_, group)) => group.push((value, output.clone())),
222            None => zoom_stops.push((zoom, vec![(value, output.clone())])),
223        }
224    }
225    let mut feature_parameters = Map::new();
226    for key in ["type", "property", "default"] {
227        if let Some(value) = parameters.get(key) {
228            feature_parameters.insert(key.to_string(), value.clone());
229        }
230    }
231    let exponential = function_type(&Map::new(), spec) == "exponential";
232    let mut expression = if exponential {
233        json!([interpolate_operator(parameters), ["linear"], ["zoom"]])
234    } else {
235        json!(["step", ["zoom"]])
236    };
237    for (zoom, group) in &zoom_stops {
238        let output = convert_property_function(&feature_parameters, spec, group);
239        append_stop_pair(&mut expression, json!(zoom), output, !exponential);
240    }
241    if !exponential {
242        fixup_degenerate_step_curve(&mut expression);
243    }
244    expression
245}

247fn fallback(parameters: &Map<String, Json>, spec: &LegacyPropertySpec) -> Json {
248    parameters
249        .get("default")
250        .or(spec.default.as_ref())
251        .map_or(Json::Null, literal)
252}

254fn convert_property_function(
255    parameters: &Map<String, Json>,
256    spec: &LegacyPropertySpec,
257    stops: &[(Json, Json)],
258) -> Json {
259    let kind = function_type(parameters, spec);
260    let get = json!([
261        "get",
262        parameters.get("property").cloned().unwrap_or(Json::Null)
263    ]);
264    match kind {
265        "categorical" if stops.first().is_some_and(|(input, _)| input.is_boolean()) => {
266            let mut expression = json!(["case"]);
267            for (input, output) in stops {
268                push(&mut expression, json!(["==", get, input]));
269                push(&mut expression, output.clone());
270            }
271            push(&mut expression, fallback(parameters, spec));
272            expression
273        }
274        "categorical" => {
275            let mut expression = json!(["match", get]);
276            for (input, output) in stops {
277                append_stop_pair(&mut expression, input.clone(), output.clone(), false);
278            }
279            push(&mut expression, fallback(parameters, spec));
280            expression
281        }
282        "interval" => {
283            let mut expression = json!(["step", ["number", get]]);
284            for (input, output) in stops {
285                append_stop_pair(&mut expression, input.clone(), output.clone(), true);
286            }
287            fixup_degenerate_step_curve(&mut expression);
288            guard_default(parameters, &get, expression)
289        }
290        _ => {
291            let base = parameters.get("base").and_then(Json::as_f64).unwrap_or(1.0);
292            let curve = if base == 1.0 {
293                json!(["linear"])
294            } else {
295                json!(["exponential", base])
296            };
297            let mut expression = json!([interpolate_operator(parameters), curve, ["number", get]]);
298            for (input, output) in stops {
299                append_stop_pair(&mut expression, input.clone(), output.clone(), false);
300            }
301            guard_default(parameters, &get, expression)
302        }
303    }
304}

306/// Wraps a numeric ramp so a feature without a numeric value takes the default.
307fn guard_default(parameters: &Map<String, Json>, get: &Json, expression: Json) -> Json {
308    match parameters.get("default") {
309        None => expression,
310        Some(default) => json!([
311            "case",
312            ["==", ["typeof", get], "number"],
313            expression,
314            literal(default)
315        ]),
316    }
317}

319fn convert_zoom_function(
320    parameters: &Map<String, Json>,
321    spec: &LegacyPropertySpec,
322    stops: &[(Json, Json)],
323) -> Json {
324    let kind = function_type(parameters, spec);
325    let is_step = kind == "interval";
326    let mut expression = if is_step {
327        json!(["step", ["zoom"]])
328    } else {
329        let base = parameters.get("base").and_then(Json::as_f64).unwrap_or(1.0);
330        let curve = if base == 1.0 {
331            json!(["linear"])
332        } else {
333            json!(["exponential", base])
334        };
335        json!([interpolate_operator(parameters), curve, ["zoom"]])
336    };
337    for (input, output) in stops {
338        append_stop_pair(&mut expression, input.clone(), output.clone(), is_step);
339    }
340    fixup_degenerate_step_curve(&mut expression);
341    expression
342}

344pub(super) fn push(expression: &mut Json, value: Json) {
345    if let Json::Array(items) = expression {
346        items.push(value);
347    }
348}

350fn fixup_degenerate_step_curve(expression: &mut Json) {
351    if let Json::Array(items) = expression {
352        if items.first().and_then(Json::as_str) == Some("step") && items.len() == 3 {
353            let output = items[2].clone();
354            items.push(json!(0));
355            items.push(output);
356        }
357    }
358}
359
360fn append_stop_pair(curve: &mut Json, input: Json, output: Json, is_step: bool) {
361    let Json::Array(items) = curve else {
362        return;
363    };
364    if items.len() > 3 && items[items.len() - 2] == input {
365        return;
366    }
367    if !(is_step && items.len() == 2) {
368        items.push(input);
369    }
370    items.push(output);
371}

373/// Turns a `{token}` string into a `concat` of literals and property reads.
374pub fn convert_token_string(text: &str) -> Json {
375    let mut parts: Vec<Json> = Vec::new();
376    let mut rest = text;
377    while let Some(open) = rest.find('{') {
378        let Some(close) = rest[open..].find('}') else {
379            break;
380        };
381        let name = &rest[open + 1..open + close];
382        if name.is_empty() || name.contains('{') {
383            break;
384        }
385        if open > 0 {
386            parts.push(Json::String(rest[..open].to_string()));
387        }
388        parts.push(json!(["get", name]));
389        rest = &rest[open + close + 1..];
390    }
391    if parts.is_empty() {
392        return Json::String(text.to_string());
393    }
394    if !rest.is_empty() {
395        parts.push(Json::String(rest.to_string()));
396    } else if parts.len() == 1 {
397        return json!(["to-string", parts[0]]);
398    }
399    let mut expression = json!(["concat"]);
400    for part in parts {
401        push(&mut expression, part);
402    }
403    expression
