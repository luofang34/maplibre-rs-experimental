36/// Comparison operators.
37#[derive(Clone, Copy, Debug, PartialEq, Eq)]
38pub enum Comparison {
39    /// `==`
40    Equal,
41    /// `!=`
42    NotEqual,
43    /// `<`
44    Less,
45    /// `<=`
46    LessEqual,
47    /// `>`
48    Greater,
49    /// `>=`
50    GreaterEqual,
51}
52
53impl Comparison {
54    /// The operator as the style writes it.
55    pub fn symbol(self) -> &'static str {
56        match self {
57            Self::Equal => "==",
58            Self::NotEqual => "!=",
59            Self::Less => "<",
60            Self::LessEqual => "<=",
61            Self::Greater => ">",
62            Self::GreaterEqual => ">=",
63        }
64    }
65
66    /// Whether the operator orders its operands rather than testing equality.
67    pub fn is_ordering(self) -> bool {
68        !matches!(self, Self::Equal | Self::NotEqual)
69    }
70}

148/// A parsed expression.
149///
150/// Every node knows its output type, so a parsed expression can be classified and checked
151/// without evaluating it.
152#[derive(Clone, Debug, PartialEq)]
153pub enum Expression {
154    /// A constant.
155    Literal(Value),
156    /// A constant computed while parsing, keeping the type the source expression declared.
157    Folded {
158        /// The computed value.
159        value: Value,
160        /// Type of the source expression.
161        output: Type,
162    },
163    /// A property of the map.
164    Global(Global),
165    /// A property of the feature.
166    Feature(FeatureProperty),
167    /// `global-state`: a map-level value, `null` unless the evaluation context carries one.
168    GlobalState(String),
169    /// `get`: a property of the feature, or of `object` when given.
170    Get {
171        /// Name of the property.
172        key: Box<Expression>,
173        /// Object to read from instead of the feature.
174        object: Option<Box<Expression>>,
175    },
176    /// `has`: whether the feature, or `object`, has the property.
177    Has {
178        /// Name of the property.
179        key: Box<Expression>,
180        /// Object to look in instead of the feature.
181        object: Option<Box<Expression>>,
182    },
183    /// `var`: a variable bound by an enclosing `let`, resolved when parsed.
184    Var {
185        /// Name of the variable.
186        name: String,
187        /// The expression bound to it.
188        bound: Box<Expression>,
189    },
190    /// `let`: bindings for the variables `body` may use.
191    Let {
192        /// Bound expressions, kept so classification sees them.
193        bindings: Vec<(String, Expression)>,
194        /// The result.
195        body: Box<Expression>,
196    },
197    /// `case`: the output of the first branch whose condition holds.
198    Case {
199        /// Condition and output pairs.
200        branches: Vec<(Expression, Expression)>,
201        /// Output when no condition holds.
202        fallback: Box<Expression>,
203        /// Type of every output.
204        output: Type,
205    },
206    /// `match`: the output whose labels contain the input.
207    Match {
208        /// The value to look up.
209        input: Box<Expression>,
210        /// Type the labels share; an input of another type takes the fallback.
211        input_type: Type,
212        /// Labels and output pairs.
213        cases: Vec<(Vec<Value>, Expression)>,
214        /// Output when no label matches.
215        fallback: Box<Expression>,
216        /// Type of every output.
217        output: Type,
218    },
219    /// `coalesce`: the first output that is not `null`.
220    Coalesce {
221        /// Candidates in order.
222        operands: Vec<Expression>,
223        /// Type of the outputs.
224        output: Type,
225    },
226    /// A comparison of two values.
227    Compare {
228        /// The operator.
229        operator: Comparison,
230        /// Left operand.
231        left: Box<Expression>,
232        /// Right operand.
233        right: Box<Expression>,
234        /// Whether an operand's type is only known at run time.
235        untyped: bool,
236        /// How strings compare, when the expression names a collator.
237        collator: Option<Box<Expression>>,
238    },
239    /// `collator`: a rule for comparing strings.
240    Collator {
241        /// Whether letters differing only in case differ.
242        case_sensitive: Box<Expression>,
243        /// Whether letters differing only in their accents differ.
244        diacritic_sensitive: Box<Expression>,
245        /// The locale, if given.
246        locale: Option<Box<Expression>>,
247    },
248    /// `within`: whether the feature's geometry lies inside the area.
249    Within(super::geometry::Geometry),
250    /// `distance`: metres from the feature's geometry to the geometry.
251    Distance(super::geometry::Geometry),
252    /// `number-format`: a number as text, in the style of a locale.
253    NumberFormat {
254        /// The number.
255        input: Box<Expression>,
256        /// Options: `locale`, `currency`, `min-fraction-digits`, `max-fraction-digits`.
257        options: Vec<(String, Expression)>,
258    },
259    /// `resolved-locale`: the locale a collator compares in.
260    ResolvedLocale(Box<Expression>),
261    /// `is-supported-script`: whether a string is in scripts the renderer can lay out.
262    IsSupportedScript(Box<Expression>),
263    /// `all`
264    All(Vec<Expression>),
265    /// `any`
266    Any(Vec<Expression>),
267    /// `!`
268    Not(Box<Expression>),
269    /// `in`: whether a string or array contains the needle.
270    In {
271        /// What to look for.
272        needle: Box<Expression>,
273        /// Where to look.
274        haystack: Box<Expression>,
275    },
276    /// `index-of`: where a string or array holds the needle, or minus one.
277    IndexOf {
278        /// What to look for.
279        needle: Box<Expression>,
280        /// Where to look.
281        haystack: Box<Expression>,
282        /// Index to start from.
283        from: Option<Box<Expression>>,
284    },
285    /// `slice` of a string or array.
286    Slice {
287        /// The string or array.
288        input: Box<Expression>,
289        /// First index kept.
290        from: Box<Expression>,
291        /// First index dropped.
292        to: Option<Box<Expression>>,
293        /// Type of the input, which is also the output's.
294        output: Type,
295    },
296    /// `length` of a string or array.
297    Length(Box<Expression>),
298    /// Arithmetic over numbers.
299    Arithmetic {
300        /// The operator.
301        operator: Arithmetic,
302        /// Operands, all numbers.
303        operands: Vec<Expression>,
304    },
305    /// A one-operand mathematical function.
306    Math {
307        /// The function.
308        function: MathFunction,
309        /// Its operand.
310        operand: Box<Expression>,
311    },
312    /// `min` or `max` over numbers.
313    MinMax {
314        /// `true` for `max`.
315        max: bool,
316        /// Operands, all numbers.
317        operands: Vec<Expression>,
318    },
319    /// `typeof`
320    TypeOf(Box<Expression>),
321    /// `number`, `string`, `boolean`, `object` and `array`: the first operand of the type.
322    Assert {
323        /// Required type.
324        required: Type,
325        /// Candidates in order.
326        operands: Vec<Expression>,
327    },
328    /// An image by name, with whether the map holds it.
329    Image(Box<Expression>),
330    /// `format`: text in sections, each of which may be an image.
331    Format(Vec<FormatSection>),
332    /// `to-number`, `to-string`, `to-boolean` and `to-color`.
333    Coerce {
334        /// Conversion.
335        coercion: Coercion,
336        /// Candidates in order.
337        operands: Vec<Expression>,
338    },
339    /// `to-rgba`
340    ToRgba(Box<Expression>),
341    /// `rgb` and `rgba`.
342    Rgba(Vec<Expression>),
343    /// `interpolate` and its colour-space variants.
344    Interpolate {
345        /// The curve.
346        interpolation: Interpolation,
347        /// Colour space for colour outputs.
348        space: ColorSpace,
349        /// The number to look up.
350        input: Box<Expression>,
351        /// Ascending stop inputs with their outputs.
352        stops: Vec<(f64, Expression)>,
353        /// Type of every output.
354        output: Type,
355    },
356    /// `step`: the output of the last stop at or below the input.
357    Step {
358        /// The number to look up.
359        input: Box<Expression>,
360        /// Ascending stop inputs with their outputs; the first is minus infinity.
361        stops: Vec<(f64, Expression)>,
362        /// Type of every output.
363        output: Type,
364    },
365    /// `concat`
366    Concat(Vec<Expression>),
367    /// `upcase` and `downcase`.
368    StringCase {
369        /// Which case.
370        function: StringFunction,
371        /// The string.
372        operand: Box<Expression>,
373    },
374}

376impl Expression {
377    /// The type the expression evaluates to, as far as it is known before evaluation.
378    pub fn output_type(&self) -> Type {
379        match self {
380            Self::Literal(value) => value.type_of(),
381            Self::Folded { output, .. } => output.clone(),
382            Self::Global(_) => Type::Number,
383            Self::Feature(FeatureProperty::Id) => Type::Value,
384            Self::Feature(FeatureProperty::GeometryType) => Type::String,
385            Self::Feature(FeatureProperty::Properties) => Type::Object,
386            Self::Get { .. } | Self::GlobalState(_) => Type::Value,
387            Self::Has { .. }
388            | Self::Compare { .. }
389            | Self::All(_)
390            | Self::Any(_)
391            | Self::Not(_)
392            | Self::IsSupportedScript(_)
393            | Self::Within(_)
394            | Self::In { .. } => Type::Boolean,
395            Self::Distance(_) => Type::Number,
396            Self::Collator { .. } => Type::Collator,
397            Self::Var { bound, .. } => bound.output_type(),
398            Self::Let { body, .. } => body.output_type(),
399            Self::Case { output, .. }
400            | Self::Match { output, .. }
401            | Self::Coalesce { output, .. }
402            | Self::Slice { output, .. }
403            | Self::Interpolate { output, .. }
404            | Self::Step { output, .. } => output.clone(),
405            Self::IndexOf { .. }
406            | Self::Length(_)
407            | Self::Arithmetic { .. }
408            | Self::Math { .. }
409            | Self::MinMax { .. } => Type::Number,
410            Self::TypeOf(_)
411            | Self::Concat(_)
412            | Self::StringCase { .. }
413            | Self::ResolvedLocale(_)
414            | Self::NumberFormat { .. } => Type::String,
415            Self::Image(_) => Type::ResolvedImage,
416            Self::Format(_) => Type::Formatted,
417            Self::Assert { required, .. } => required.clone(),
418            Self::Coerce { coercion, .. } => match coercion {
419                Coercion::Number => Type::Number,
420                Coercion::String => Type::String,
421                Coercion::Boolean => Type::Boolean,
422                Coercion::Color => Type::Color,
423                Coercion::Formatted => Type::Formatted,
424                Coercion::ResolvedImage => Type::ResolvedImage,
425            },
426            Self::ToRgba(_) => Type::array(Type::Number, Some(4)),
427            Self::Rgba(_) => Type::Color,
428        }
429    }
430
431    /// Whether the result is the same for every feature.
432    pub fn is_feature_constant(&self) -> bool {
433        match self {
434            Self::Feature(_) | Self::Within(_) | Self::Distance(_) => false,
435            Self::Get { object: None, .. } | Self::Has { object: None, .. } => false,
436            _ => self.children_all(Self::is_feature_constant),
437        }
438    }
439
440    /// Whether the result is the same at every zoom.
441    pub fn is_zoom_constant(&self) -> bool {
442        match self {
443            Self::Global(Global::Zoom) => false,
444            _ => self.children_all(Self::is_zoom_constant),
445        }
446    }
447
448    /// Whether the result depends on neither the feature nor the map.
449    pub fn is_constant(&self) -> bool {
450        match self {
451            // Whether an image exists is known only where it is drawn.
452            Self::Global(_) | Self::GlobalState(_) | Self::Image(_) => false,
453            _ => self.is_feature_constant() && self.children_all(Self::is_constant),
454        }
455    }
456
457    fn children_all(&self, predicate: fn(&Expression) -> bool) -> bool {
458        let mut all = true;
459        self.for_each_child(&mut |child| {
460            if all && !predicate(child) {
461                all = false;
462            }
463        });
464        all
465    }
