11/// A colour with straight (not premultiplied) components in `0..=1`.
12///
13/// GL JS stores colours premultiplied; [`Color::premultiplied`] gives that form, which is also
14/// how the conformance suite writes its expected colours.
15#[derive(Clone, Copy, Debug, PartialEq)]
16pub struct Color {
17    /// Red, straight.
18    pub r: f64,
19    /// Green, straight.
20    pub g: f64,
21    /// Blue, straight.
22    pub b: f64,
23    /// Alpha.
24    pub a: f64,
25}
26
27impl Color {
28    /// A colour from straight components.
29    pub const fn new(r: f64, g: f64, b: f64, a: f64) -> Self {
30        Self { r, g, b, a }
31    }
32
33    /// Parses any CSS colour string, as GL JS `Color.parse` does.
34    pub fn parse(text: &str) -> Option<Self> {
35        let color = csscolorparser::parse(text).ok()?;
36        Some(Self::new(color.r, color.g, color.b, color.a))
37    }
38
39    /// Components with red, green and blue multiplied by alpha, as GL JS stores them.
40    pub fn premultiplied(&self) -> [f64; 4] {
41        [self.r * self.a, self.g * self.a, self.b * self.a, self.a]
42    }
43
44    /// Straight components, red, green and blue in `0..=1`.
45    pub fn straight(&self) -> [f64; 4] {
46        [self.r, self.g, self.b, self.a]
47    }
48
49    /// The `rgba(r,g,b,a)` form GL JS prints, with channels in `0..=255`.
50    pub fn css(&self) -> String {
51        let channel = |value: f64| (value * 255.0).round();
52        format!(
53            "rgba({},{},{},{})",
54            channel(self.r),
55            channel(self.g),
56            channel(self.b),
57            js_number(self.a)
58        )
59    }
60}
61
62impl From<Color> for csscolorparser::Color {
63    fn from(color: Color) -> Self {
64        csscolorparser::Color::new(color.r, color.g, color.b, color.a)
65    }
66}

69    fn from(color: csscolorparser::Color) -> Self {

71    }

74/// A value at run time.
75#[derive(Clone, Debug, PartialEq)]
76pub enum Value {
77    /// `null`, also what a missing property evaluates to.
78    Null,
79    /// A boolean.
80    Bool(bool),
81    /// A number.
82    Number(f64),
83    /// A string.
84    String(String),
85    /// A colour.
86    Color(Color),
87    /// An array of values.
88    Array(Vec<Value>),
89    /// An object with string keys.
90    Object(BTreeMap<String, Value>),
91    /// How strings compare, made by `collator`.
92    Collator(Collation),
93    /// An image an `image` expression names.
94    Image(ResolvedImage),
95    /// Text in sections, made by `format`.
96    Formatted(Formatted),
97}
98
99impl Value {
100    /// The value a JSON document holds; nothing is coerced, so a colour stays a string.
101    pub fn from_json(json: &serde_json::Value) -> Self {
102        match json {
103            serde_json::Value::Null => Self::Null,
104            serde_json::Value::Bool(flag) => Self::Bool(*flag),
105            serde_json::Value::Number(number) => Self::Number(number.as_f64().unwrap_or(f64::NAN)),
106            serde_json::Value::String(text) => Self::String(text.clone()),
107            serde_json::Value::Array(items) => {
108                Self::Array(items.iter().map(Self::from_json).collect())
109            }
110            serde_json::Value::Object(members) => Self::Object(
111                members
112                    .iter()
113                    .map(|(key, value)| (key.clone(), Self::from_json(value)))
114                    .collect(),
115            ),
116        }
117    }
118
119    /// The JSON form; a colour becomes its premultiplied `[r, g, b, a]`.
120    pub fn to_json(&self) -> serde_json::Value {
121        match self {
122            Self::Null => serde_json::Value::Null,
123            Self::Bool(flag) => serde_json::Value::Bool(*flag),
124            Self::Number(number) => {
125                if number.fract() == 0.0 && number.abs() < 9007199254740992.0 {
126                    serde_json::Value::Number((*number as i64).into())
127                } else {
128                    serde_json::Number::from_f64(*number)
129                        .map_or(serde_json::Value::Null, serde_json::Value::Number)
130                }
131            }
132            Self::String(text) => serde_json::Value::String(text.clone()),
133            Self::Color(color) => serde_json::Value::Array(
134                color
135                    .premultiplied()
136                    .iter()
137                    .map(|component| Self::Number(*component).to_json())
138                    .collect(),
139            ),
140            Self::Array(items) => {
141                serde_json::Value::Array(items.iter().map(Self::to_json).collect())
142            }
143            Self::Object(members) => serde_json::Value::Object(
144                members
145                    .iter()
146                    .map(|(key, value)| (key.clone(), value.to_json()))
147                    .collect(),
148            ),
149            Self::Collator(_) => serde_json::Value::Null,
150            Self::Image(image) => image.to_json(),
151            Self::Formatted(formatted) => formatted.to_json(),
152        }
153    }
154
155    /// The type GL JS `typeof` reports; an array's item type and length are inferred.
156    pub fn type_of(&self) -> Type {
157        match self {
158            Self::Null => Type::Null,
159            Self::Bool(_) => Type::Boolean,
160            Self::Number(_) => Type::Number,
161            Self::String(_) => Type::String,
162            Self::Color(_) => Type::Color,
163            Self::Object(_) => Type::Object,
164            Self::Collator(_) => Type::Collator,
165            Self::Image(_) => Type::ResolvedImage,
166            Self::Formatted(_) => Type::Formatted,
167            Self::Array(items) => {
168                let mut item_type: Option<Type> = None;
169                for item in items {
170                    let candidate = item.type_of();
171                    match &item_type {
172                        None => item_type = Some(candidate),
173                        Some(current) if *current == candidate => {}
174                        Some(_) => {
175                            item_type = Some(Type::Value);
176                            break;
177                        }
178                    }
179                }
180                Type::array(item_type.unwrap_or(Type::Value), Some(items.len()))
181            }
182        }
183    }
184
185    /// The number, if the value is one.
186    pub fn as_number(&self) -> Option<f64> {
187        match self {
188            Self::Number(number) => Some(*number),
189            _ => None,
190        }
191    }
192
193    /// The string, if the value is one.
194    pub fn as_str(&self) -> Option<&str> {
195        match self {
196            Self::String(text) => Some(text),
197            _ => None,
198        }
199    }
200
201    /// The boolean, if the value is one.
202    pub fn as_bool(&self) -> Option<bool> {
203        match self {
204            Self::Bool(flag) => Some(*flag),
205            _ => None,
206        }
207    }
208
209    /// Whether the value is `null`.
210    pub fn is_null(&self) -> bool {
211        matches!(self, Self::Null)
212    }
213
214    /// JavaScript truthiness, what `to-boolean` and `case` conditions use.
215    pub fn truthy(&self) -> bool {
216        match self {
217            Self::Null => false,
218            Self::Bool(flag) => *flag,
219            Self::Number(number) => *number != 0.0 && !number.is_nan(),
220            Self::String(text) => !text.is_empty(),
221            Self::Color(_)
222            | Self::Array(_)
223            | Self::Object(_)
224            | Self::Collator(_)
225            | Self::Image(_)
226            | Self::Formatted(_) => true,
227        }
228    }
229
230    /// The string GL JS `to-string` and `concat` produce.
231    pub fn to_display_string(&self) -> String {
232        match self {
233            Self::Null => String::new(),
234            Self::Bool(flag) => flag.to_string(),
235            Self::Number(number) => js_number(*number),
236            Self::String(text) => text.clone(),
237            Self::Color(color) => color.css(),
238            Self::Array(_) | Self::Object(_) => self.to_json().to_string(),
239            Self::Collator(_) => String::new(),
240            Self::Image(image) => image.name.clone(),
241            Self::Formatted(formatted) => formatted.text(),
242        }
243    }
244}

276/// Formats a number the way JavaScript's `String(number)` does.
277pub fn js_number(number: f64) -> String {
278    if number.is_nan() {
279        return "NaN".to_string();
280    }
281    if number.is_infinite() {
282        return if number > 0.0 {
283            "Infinity"
284        } else {
285            "-Infinity"
286        }
287        .to_string();
288    }
289    if number == 0.0 {
290        return "0".to_string();
291    }
292    let magnitude = number.abs();
293    if (1e-6..1e21).contains(&magnitude) {
294        return format!("{number}");
295    }
296    let scientific = format!("{number:e}");
297    match scientific.split_once('e') {
298        Some((mantissa, exponent)) if !exponent.starts_with('-') => {
299            format!("{mantissa}e+{exponent}")
300        }
301        _ => scientific,
302    }
303}

305/// The static type of an expression or the dynamic type of a value.
306#[derive(Clone, Debug, PartialEq, Eq)]
307pub enum Type {
308    /// `null`.
309    Null,
310    /// A number.
311    Number,
312    /// A string.
313    String,
314    /// A boolean.
315    Boolean,
316    /// A colour.
317    Color,
318    /// An object.
319    Object,
320    /// A string comparison rule.
321    Collator,
322    /// An image name and whether the map holds it.
323    ResolvedImage,
324    /// Text in sections.
325    Formatted,
326    /// Any value; the type of a property read or an untyped branch.
327    Value,
328    /// An array, with an item type and, when known, a length.
329    Array {
330        /// Type of every item.
331        item: Box<Type>,
332        /// Number of items, when the type fixes it.
333        length: Option<usize>,
334    },
335}
336
337impl Type {
338    /// An array type.
339    pub fn array(item: Type, length: Option<usize>) -> Self {
340        Self::Array {
341            item: Box::new(item),
342            length,
343        }
344    }
345
346    /// Whether a value of this type is acceptable where `expected` is required, as GL JS
347    /// `checkSubtype` decides.
348    pub fn is_subtype_of(&self, expected: &Type) -> bool {
349        match (expected, self) {
350            (
351                Type::Array {
352                    item: expected_item,
353                    length: expected_length,
354                },
355                Type::Array { item, length },
356            ) => {
357                let items_fit = (*length == Some(0) && **item == Type::Value)
358                    || item.is_subtype_of(expected_item);
359                items_fit && expected_length.is_none_or(|expected| Some(expected) == *length)
360            }
361            (Type::Array { .. }, _) => false,
362            (Type::Value, _) => true,
363            (expected, actual) => expected == actual,
364        }
365    }
366
367    /// The name GL JS prints, such as `array<number, 3>`.
368    pub fn name(&self) -> String {
369        match self {
370            Self::Null => "null".to_string(),
371            Self::Number => "number".to_string(),
372            Self::String => "string".to_string(),
373            Self::Boolean => "boolean".to_string(),
374            Self::Color => "color".to_string(),
375            Self::Object => "object".to_string(),
376            Self::Collator => "collator".to_string(),
377            Self::ResolvedImage => "resolvedImage".to_string(),
378            Self::Formatted => "formatted".to_string(),
379            Self::Value => "value".to_string(),
380            Self::Array { item, length } => match (item.as_ref(), length) {
381                (_, Some(length)) => format!("array<{}, {length}>", item.name()),
382                (Type::Value, None) => "array".to_string(),
383                (item, None) => format!("array<{}>", item.name()),
384            },
385        }
386    }
387}
388
389impl fmt::Display for Type {
390    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
391        formatter.write_str(&self.name())
392    }
