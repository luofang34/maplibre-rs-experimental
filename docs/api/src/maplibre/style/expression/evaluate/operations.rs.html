65/// Position of `needle` in a string or array from `from`, counted in characters for strings,
66/// or minus one.
67pub(super) fn index_of(needle: &Value, haystack: &Value, from: i64) -> Result<i64> {
68    match haystack {
69        Value::String(text) => {
70            // JavaScript spells a null needle out and starts a string search no earlier than
71            // its beginning.
72            let needle = match needle {
73                Value::Null => "null".to_string(),
74                other => other.to_display_string(),
75            };
76            let characters: Vec<char> = text.chars().collect();
77            let start = clamp_index(from.max(0), characters.len());
78            let needle_chars: Vec<char> = needle.chars().collect();
79            if needle_chars.is_empty() {
80                return Ok(start as i64);
81            }
82            Ok(characters[start..]
83                .windows(needle_chars.len())
84                .position(|window| window == needle_chars.as_slice())
85                .map_or(-1, |position| (position + start) as i64))
86        }
87        Value::Array(items) => {
88            let start = clamp_index(from, items.len());
89            Ok(items[start..]
90                .iter()
91                .position(|item| item == needle)
92                .map_or(-1, |position| (position + start) as i64))
93        }
94        other => Err(EvaluationError::NotIndexable {
95            found: other.type_of(),
96        }),
97    }
98}

100/// JavaScript index semantics: negative counts from the end, and the result stays in range.
101pub(super) fn clamp_index(index: i64, length: usize) -> usize {
102    if index < 0 {
103        (length as i64 + index).max(0) as usize
104    } else {
105        (index as usize).min(length)
106    }
107}

109pub(super) fn slice(input: Value, from: i64, to: Option<i64>) -> Result<Value> {
110    match input {
111        Value::String(text) => {
112            let characters: Vec<char> = text.chars().collect();
113            let start = clamp_index(from, characters.len());
114            let end = to.map_or(characters.len(), |to| clamp_index(to, characters.len()));
115            Ok(Value::String(
116                characters[start..end.max(start)].iter().collect(),
117            ))
118        }
119        Value::Array(items) => {
120            let start = clamp_index(from, items.len());
121            let end = to.map_or(items.len(), |to| clamp_index(to, items.len()));
122            Ok(Value::Array(items[start..end.max(start)].to_vec()))
123        }
124        other => Err(EvaluationError::NotIndexable {
125            found: other.type_of(),
126        }),
127    }
128}

130pub(super) fn compare(
131    operator: Comparison,
132    left: Value,
133    right: Value,
134    untyped: bool,
135    collation: Option<&Collation>,
136) -> Result<Value> {
137    if operator.is_ordering() && untyped {
138        let (left_type, right_type) = (left.type_of(), right.type_of());
139        if left_type != right_type || !matches!(left_type, Type::String | Type::Number) {
140            return Err(EvaluationError::Incomparable {
141                operator: operator.symbol(),
142                left: left_type,
143                right: right_type,
144            });
145        }
146    }
147    let ordering = match (&left, &right) {
148        (Value::Number(a), Value::Number(b)) => a.partial_cmp(b),
149        (Value::String(a), Value::String(b)) => Some(match collation {
150            Some(collation) => collation.compare(a, b),
151            None => a.cmp(b),
152        }),
153        _ => None,
154    };
155    let equal = match (&left, &right, collation) {
156        (Value::String(a), Value::String(b), Some(collation)) => collation.compare(a, b).is_eq(),
157        _ => left == right,
158    };
159    Ok(Value::Bool(match operator {
160        Comparison::Equal => equal,
161        Comparison::NotEqual => !equal,
162        Comparison::Less => ordering.is_some_and(|ordering| ordering.is_lt()),
163        Comparison::LessEqual => ordering.is_some_and(|ordering| ordering.is_le()),
164        Comparison::Greater => ordering.is_some_and(|ordering| ordering.is_gt()),
165        Comparison::GreaterEqual => ordering.is_some_and(|ordering| ordering.is_ge()),
166    }))
167}

169pub(super) fn arithmetic(operator: Arithmetic, numbers: &[f64]) -> f64 {
170    match operator {
171        Arithmetic::Add => numbers.iter().sum(),
172        Arithmetic::Multiply => numbers.iter().product(),
173        Arithmetic::Subtract => match numbers {
174            [only] => -only,
175            [first, second, ..] => first - second,
176            [] => 0.0,
177        },
178        Arithmetic::Divide => numbers[0] / numbers[1],
179        Arithmetic::Remainder => numbers[0] % numbers[1],
180        Arithmetic::Power => numbers[0].powf(numbers[1]),
181    }
182}

184pub(super) fn math(function: MathFunction, number: f64) -> f64 {
185    match function {
186        MathFunction::Sqrt => number.sqrt(),
187        MathFunction::Ln => number.ln(),
188        MathFunction::Log10 => number.log10(),
189        MathFunction::Log2 => number.log2(),
190        MathFunction::Sin => number.sin(),
191        MathFunction::Cos => number.cos(),
192        MathFunction::Tan => number.tan(),
193        MathFunction::Asin => number.asin(),
194        MathFunction::Acos => number.acos(),
195        MathFunction::Atan => number.atan(),
196        MathFunction::Abs => number.abs(),
197        // JavaScript rounds halves towards positive infinity; mirroring negatives keeps
198        // -2.5 at -3 as GL JS does.
199        MathFunction::Round => {
200            if number < 0.0 {
201                -js_round(-number)
202            } else {
203                js_round(number)
204            }
205        }
206        MathFunction::Floor => number.floor(),
207        MathFunction::Ceil => number.ceil(),
208    }
209}

211pub(super) fn js_round(number: f64) -> f64 {
212    (number + 0.5).floor()

215pub(super) fn coerce(
216    coercion: Coercion,
217    operands: &[Expression],
218    context: &EvaluationContext,
219) -> Result<Value> {
220    match coercion {
221        Coercion::Boolean => Ok(Value::Bool(operands[0].evaluate(context)?.truthy())),
222        Coercion::Formatted => Ok(super::images::to_formatted(operands[0].evaluate(context)?)),
223        Coercion::ResolvedImage => Ok(super::images::to_image(
224            operands[0].evaluate(context)?,
225            context,
226        )),
227        Coercion::String => Ok(Value::String(
228            operands[0].evaluate(context)?.to_display_string(),
229        )),
230        Coercion::Number => {
231            let mut last = Value::Null;
232            for operand in operands {
233                last = operand.evaluate(context)?;
234                if last.is_null() {
235                    return Ok(Value::Number(0.0));
236                }
237                let number = match &last {
238                    Value::Bool(flag) => Some(f64::from(u8::from(*flag))),
239                    Value::Number(number) => Some(*number),
240                    Value::String(text) => {
241                        let trimmed = text.trim();
242                        if trimmed.is_empty() {
243                            Some(0.0)
244                        } else {
245                            trimmed.parse::<f64>().ok()
246                        }
247                    }
248                    _ => None,
249                };
250                if let Some(number) = number {
251                    return Ok(Value::Number(number));
252                }
253            }
254            Err(EvaluationError::NotANumber {
255                value: last.to_json().to_string(),
256            })
257        }
258        Coercion::Color => {
259            let mut error = None;
260            let mut last = Value::Null;
261            for operand in operands {
262                last = operand.evaluate(context)?;
263                error = None;
264                match &last {
265                    Value::Color(color) => return Ok(Value::Color(*color)),
266                    Value::String(text) => {
267                        if let Some(color) = Color::parse(text) {
268                            return Ok(Value::Color(color));
269                        }
270                    }
271                    Value::Array(items) => {
272                        let numbers: Vec<f64> = items.iter().filter_map(Value::as_number).collect();
273                        if numbers.len() != items.len() || !(3..=4).contains(&items.len()) {
274                            error = Some(EvaluationError::InvalidRgba {
275                                value: last.to_json().to_string(),
276                                reason: "expected an array containing either three or four numeric values"
277                                    .to_string(),
278                            });
279                        } else {
280                            match rgba(numbers[0], numbers[1], numbers[2], numbers.get(3).copied())
281                            {
282                                Ok(color) => return Ok(Value::Color(color)),
283                                Err(rgba_error) => error = Some(rgba_error),
284                            }
285                        }
286                    }
287                    _ => {}
288                }
289            }
290            Err(error.unwrap_or_else(|| EvaluationError::InvalidColor {
291                value: match &last {
292                    Value::String(text) => text.clone(),
293                    other => other.to_json().to_string(),
294                },
295            }))
296        }
297    }
298}

300/// A colour from `0..=255` channels and an optional alpha, validated as GL JS does.
301pub(super) fn rgba(r: f64, g: f64, b: f64, a: Option<f64>) -> Result<Color> {
302    let value = || {
303        Value::Array(
304            [Some(r), Some(g), Some(b), a]
305                .into_iter()
306                .flatten()
307                .map(Value::Number)
308                .collect(),
309        )
310        .to_json()
311        .to_string()
312    };
313    if ![r, g, b]
314        .iter()
315        .all(|channel| (0.0..=255.0).contains(channel))
316    {
317        return Err(EvaluationError::InvalidRgba {
318            value: value(),
319            reason: "'r', 'g', and 'b' must be between 0 and 255".to_string(),
320        });
321    }
322    if let Some(alpha) = a {
323        if !(0.0..=1.0).contains(&alpha) {
324            return Err(EvaluationError::InvalidRgba {
325                value: value(),
326                reason: "'a' must be between 0 and 1".to_string(),
327            });
328        }
329    }
330    Ok(Color::new(
331        r / 255.0,
332        g / 255.0,
333        b / 255.0,
334        a.unwrap_or(1.0),
335    ))
336}

338/// Index of the last stop whose input is at or below `input`; the caller keeps `input`
339/// strictly inside the stop range.
340pub(super) fn stop_at_or_below(stops: &[(f64, Expression)], input: f64) -> usize {
341    let index = match stops
342        .binary_search_by(|(stop, _)| stop.partial_cmp(&input).unwrap_or(std::cmp::Ordering::Less))
343    {
344        Ok(index) => index,
345        Err(index) => index.saturating_sub(1),
346    };
347    index.min(stops.len().saturating_sub(2))
348}

350pub(super) fn interpolate_values(
351    from: Value,
352    to: Value,
353    t: f64,
354    space: ColorSpace,
355) -> Result<Value> {
356    match (from, to) {
357        (Value::Number(from), Value::Number(to)) => {
358            Ok(Value::Number(interpolate_number(from, to, t)))
359        }
360        (Value::Color(from), Value::Color(to)) => {
361            Ok(Value::Color(Color::interpolate(from, to, t, space)))
362        }
363        (Value::Array(from), Value::Array(to)) => Ok(Value::Array(
364            from.iter()
365                .zip(&to)
366                .map(|(from, to)| match (from, to) {
367                    (Value::Number(from), Value::Number(to)) => {
368                        Ok(Value::Number(interpolate_number(*from, *to, t)))
369                    }
370                    (other, _) => Err(EvaluationError::Expected {
371                        expected: Type::Number,
372                        found: other.type_of(),
373                    }),
374                })
375                .collect::<Result<Vec<Value>>>()?,
376        )),
377        (from, _) => Err(EvaluationError::Expected {
378            expected: Type::Number,
379            found: from.type_of(),
380        }),
381    }
