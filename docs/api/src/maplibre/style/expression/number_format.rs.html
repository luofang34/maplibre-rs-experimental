1//! `number-format`: a number written the way an English locale writes it.
2
3/// Formats `number` with digit grouping, between `min` and `max` fraction digits, and a
4/// currency symbol when `currency` names one.
5pub(super) fn format(
6    number: f64,
7    currency: Option<&str>,
8    min: Option<usize>,
9    max: Option<usize>,
10) -> String {
11    if !number.is_finite() {
12        return if number.is_nan() {
13            "NaN"
14        } else if number > 0.0 {
15            "∞"
16        } else {
17            "-∞"
18        }
19        .to_owned();
20    }
21    let currency_digits = currency.map(|_| 2);
22    let min_digits = min.or(currency_digits).unwrap_or(0);
23    let max_digits = max.or(currency_digits).unwrap_or(3).max(min_digits);
24    let fixed = format!("{:.*}", max_digits, number.abs());
25    let (whole, fraction) = fixed.split_once('.').unwrap_or((&fixed, ""));
26    let mut fraction = fraction.to_owned();
27    while fraction.len() > min_digits && fraction.ends_with('0') {
28        fraction.pop();
29    }
30    let mut grouped = String::new();
31    for (index, digit) in whole.chars().enumerate() {
32        if index > 0 && (whole.len() - index) % 3 == 0 {
33            grouped.push(',');
34        }
35        grouped.push(digit);
36    }
37    let mut text = grouped;
38    if !fraction.is_empty() {
39        text.push('.');
40        text.push_str(&fraction);
41    }
42    let negative = number < 0.0 && text.chars().any(|c| c.is_ascii_digit() && c != '0');
43    let sign = if negative { "-" } else { "" };
44    match currency {
45        Some(code) => format!("{sign}{}{text}", symbol(code)),
46        None => format!("{sign}{text}"),
47    }
48}
49
50fn symbol(code: &str) -> String {
51    match code {
52        "USD" => "$".to_owned(),
53        "EUR" => "€".to_owned(),
54        "GBP" => "£".to_owned(),
55        "JPY" => "¥".to_owned(),
56        other => format!("{other}\u{a0}"),
57    }
58}

60#[cfg(test)]
61mod tests {
62    use super::format;
63
64    #[test]
65    fn groups_digits_and_trims_trailing_zeros() {
66        assert_eq!(format(1234567.891, None, None, None), "1,234,567.891");
67        assert_eq!(format(0.5, None, None, None), "0.5");
68        assert_eq!(format(2.0, None, Some(2), None), "2.00");
69        assert_eq!(format(110.5744, None, None, Some(3)), "110.574");
70    }
71
72    #[test]
73    fn a_currency_has_two_fraction_digits_and_a_symbol() {
74        assert_eq!(format(1234.5, Some("USD"), None, None), "$1,234.50");
75        assert_eq!(format(-3.0, Some("EUR"), None, None), "-€3.00");
76    }
