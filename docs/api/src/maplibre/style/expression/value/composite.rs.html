10/// Valid symbol anchor names accepted by `variableAnchorOffsetCollection`.
11const VALID_ANCHORS: &[&str] = &[
12    "center",
13    "left",
14    "right",
15    "top",
16    "bottom",
17    "top-left",
18    "top-right",
19    "bottom-left",
20    "bottom-right",
21];
22
23fn is_valid_anchor(anchor: &str) -> bool {
24    VALID_ANCHORS.contains(&anchor)
25}

27/// Directional `[top, right, bottom, left]` box padding (`padding`).
28#[derive(Clone, Copy, Debug, PartialEq, Serialize, Deserialize)]
29pub struct Padding(pub [f64; 4]);
30
31impl Padding {
32    /// Parses a runtime value into `[top, right, bottom, left]` CSS-shorthand padding.
33    pub fn from_value(value: &Value) -> Option<Self> {
34        match value {
35            Value::Padding(padding) => Some(*padding),
36            Value::Number(n) if n.is_finite() => Some(Self([*n, *n, *n, *n])),
37            Value::Array(items) if (1..=4).contains(&items.len()) => {
38                let mut nums = [0.0; 4];
39                for (idx, item) in items.iter().enumerate() {
40                    let n = item.as_number().filter(|v| v.is_finite())?;
41                    nums[idx] = n;
42                }
43                Some(Self(match items.len() {
44                    1 => [nums[0], nums[0], nums[0], nums[0]],
45                    2 => [nums[0], nums[1], nums[0], nums[1]],
46                    3 => [nums[0], nums[1], nums[2], nums[1]],
47                    _ => nums,
48                }))
49            }
50            _ => None,
51        }
52    }
53
54    /// Componentwise linear interpolation between two padding values.
55    pub fn interpolate(self, to: Self, t: f64) -> Self {
56        Self([
57            interpolate_number(self.0[0], to.0[0], t),
58            interpolate_number(self.0[1], to.0[1], t),
59            interpolate_number(self.0[2], to.0[2], t),
60            interpolate_number(self.0[3], to.0[3], t),
61        ])
62    }
63
64    /// Serializes to the GL JS `{ "values": [top, right, bottom, left] }` representation.
65    pub fn to_json(self) -> Json {
66        json!({
67            "values": self.0.map(|v| Value::Number(v).to_json())
68        })
69    }
70}

72/// Variable-length numeric array (`numberArray`).
73#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
74pub struct NumberArray(pub Vec<f64>);
75
76impl NumberArray {
77    /// Parses a single number or homogeneous number array into a [`NumberArray`].
78    pub fn from_value(value: &Value) -> Option<Self> {
79        match value {
80            Value::NumberArray(arr) => Some(arr.clone()),
81            Value::Number(n) if n.is_finite() => Some(Self(vec![*n])),
82            Value::Array(items) => items
83                .iter()
84                .map(|item| item.as_number().filter(|v| v.is_finite()))
85                .collect::<Option<Vec<f64>>>()
86                .map(Self),
87            _ => None,
88        }
89    }
90
91    /// Componentwise interpolation when both arrays have matching length.
92    pub fn interpolate(&self, to: &Self, t: f64) -> Option<Self> {
93        if self.0.len() != to.0.len() {
94            return None;
95        }
96        Some(Self(
97            self.0
98                .iter()
99                .zip(&to.0)
100                .map(|(a, b)| interpolate_number(*a, *b, t))
101                .collect(),
102        ))
103    }
104
105    /// Serializes to `{ "values": [...] }`.
106    pub fn to_json(&self) -> Json {
107        json!({
108            "values": self.0.iter().map(|v| Value::Number(*v).to_json()).collect::<Vec<_>>()
109        })
110    }
111}

113/// Variable-length colour array (`colorArray`).
114#[derive(Clone, Debug, PartialEq)]
115pub struct ColorArray(pub Vec<Color>);
116
117impl ColorArray {
118    /// Parses a colour, CSS colour string, or array of colours/strings into a [`ColorArray`].
119    pub fn from_value(value: &Value) -> Option<Self> {
120        match value {
121            Value::ColorArray(arr) => Some(arr.clone()),
122            Value::Color(color) => Some(Self(vec![*color])),
123            Value::String(text) => Color::parse(text).map(|color| Self(vec![color])),
124            Value::Array(items) => items
125                .iter()
126                .map(|item| match item {
127                    Value::Color(color) => Some(*color),
128                    Value::String(text) => Color::parse(text),
129                    _ => None,
130                })
131                .collect::<Option<Vec<Color>>>()
132                .map(Self),
133            _ => None,
134        }
135    }
136
137    /// Componentwise colour interpolation in `space` when both arrays have matching length.
138    pub fn interpolate(&self, to: &Self, t: f64, space: ColorSpace) -> Option<Self> {
139        if self.0.len() != to.0.len() {
140            return None;
141        }
142        Some(Self(
143            self.0
144                .iter()
145                .zip(&to.0)
146                .map(|(a, b)| Color::interpolate(*a, *b, t, space))
147                .collect(),
148        ))
149    }
150
151    /// Serializes to `{ "values": [{ "r", "g", "b", "a" }, ...] }` with premultiplied channels.
152    pub fn to_json(&self) -> Json {
153        json!({
154            "values": self.0.iter().map(|color| {
155                let [r, g, b, a] = color.premultiplied();
156                json!({
157                    "r": Value::Number(r).to_json(),
158                    "g": Value::Number(g).to_json(),
159                    "b": Value::Number(b).to_json(),
160                    "a": Value::Number(a).to_json(),
161                })
162            }).collect::<Vec<_>>()
163        })
164    }
165}

167/// Map projection definition (`projectionDefinition`): a named projection or transition blend.
168#[derive(Clone, Debug, PartialEq)]
169pub enum ProjectionDefinition {
170    /// Named projection such as `"mercator"` or `"vertical-perspective"`.
171    Named(String),
172    /// Explicit `[from, to, transition]` array literal.
173    Tuple {
174        /// Starting projection name.
175        from: String,
176        /// Target projection name.
177        to: String,
178        /// Transition factor in `0..=1`.
179        transition: f64,
180    },
181    /// Interpolated `{ from, to, transition }` object produced between `interpolate` stops.
182    Interpolated {
183        /// Starting projection name.
184        from: String,
185        /// Target projection name.
186        to: String,
187        /// Transition factor in `0..=1`.
188        transition: f64,
189    },
190}
191
192impl ProjectionDefinition {
193    /// Parses a projection name string, `[from, to, t]` tuple, or `{ from, to, transition }`.
194    pub fn from_value(value: &Value) -> Option<Self> {
195        match value {
196            Value::ProjectionDefinition(proj) => Some(proj.clone()),
197            Value::String(name) if !name.is_empty() => Some(Self::Named(name.clone())),
198            Value::Array(items) if items.len() == 3 => {
199                let from = items[0].as_str().filter(|s| !s.is_empty())?;
200                let to = items[1].as_str().filter(|s| !s.is_empty())?;
201                let transition = items[2].as_number().filter(|n| n.is_finite())?;
202                Some(Self::Tuple {
203                    from: from.to_owned(),
204                    to: to.to_owned(),
205                    transition,
206                })
207            }
208            Value::Object(members) => {
209                let from = members.get("from")?.as_str().filter(|s| !s.is_empty())?;
210                let to = members.get("to")?.as_str().filter(|s| !s.is_empty())?;
211                let transition = members
212                    .get("transition")?
213                    .as_number()
214                    .filter(|n| n.is_finite())?;
215                Some(Self::Interpolated {
216                    from: from.to_owned(),
217                    to: to.to_owned(),
218                    transition,
219                })
220            }
221            _ => None,
222        }
223    }
224
225    fn target_name(&self) -> &str {
226        match self {
227            Self::Named(name)
228            | Self::Tuple { to: name, .. }
229            | Self::Interpolated { to: name, .. } => name,
230        }
231    }
232
233    /// Interpolates between two projection stops at blend factor `t`.
234    pub fn interpolate(&self, to: &Self, t: f64) -> Self {
235        Self::Interpolated {
236            from: self.target_name().to_owned(),
237            to: to.target_name().to_owned(),
238            transition: t,
239        }
240    }
241
242    /// Serializes to the GL JS representation expected by the expression conformance suite.
243    pub fn to_json(&self) -> Json {
244        match self {
245            Self::Named(name) => Json::String(name.clone()),
246            Self::Tuple {
247                from,
248                to,
249                transition,
250            } => json!([from, to, Value::Number(*transition).to_json()]),
251            Self::Interpolated {
252                from,
253                to,
254                transition,
255            } => json!({
256                "from": from,
257                "to": to,
258                "transition": Value::Number(*transition).to_json(),
259            }),
260        }
261    }
262}

264/// Alternating anchor name and `[x, y]` em-offset pairs (`variableAnchorOffsetCollection`).
265#[derive(Clone, Debug, PartialEq)]
266pub struct VariableAnchorOffsetCollection(pub Vec<(String, [f64; 2])>);
267
268impl VariableAnchorOffsetCollection {
269    /// Parses a non-empty alternating `[anchor, [x, y], ...]` array.
270    pub fn from_value(value: &Value) -> Option<Self> {
271        match value {
272            Value::VariableAnchorOffsetCollection(coll) => Some(coll.clone()),
273            Value::Array(items) if !items.is_empty() && items.len().is_multiple_of(2) => {
274                let mut pairs = Vec::with_capacity(items.len() / 2);
275                for [anchor_val, offset_val] in items.as_chunks::<2>().0 {
276                    let anchor = anchor_val.as_str().filter(|a| is_valid_anchor(a))?;
277                    let Value::Array(xy) = offset_val else {
278                        return None;
279                    };
280                    if xy.len() != 2 {
281                        return None;
282                    }
283                    let x = xy[0].as_number().filter(|n| n.is_finite())?;
284                    let y = xy[1].as_number().filter(|n| n.is_finite())?;
285                    pairs.push((anchor.to_owned(), [x, y]));
286                }
287                Some(Self(pairs))
288            }
289            _ => None,
290        }
291    }
292
293    /// Componentwise offset interpolation when both collections list identical anchors in order.
294    pub fn interpolate(&self, to: &Self, t: f64) -> Option<Self> {
295        if self.0.len() != to.0.len() {
296            return None;
297        }
298        let mut pairs = Vec::with_capacity(self.0.len());
299        for ((anchor_a, [x1, y1]), (anchor_b, [x2, y2])) in self.0.iter().zip(&to.0) {
300            if anchor_a != anchor_b {
301                return None;
302            }
303            pairs.push((
304                anchor_a.clone(),
305                [
306                    interpolate_number(*x1, *x2, t),
307                    interpolate_number(*y1, *y2, t),
308                ],
309            ));
310        }
311        Some(Self(pairs))
312    }
313
314    /// Serializes to `{ "values": [anchor, [x, y], ...] }`.
315    pub fn to_json(&self) -> Json {
316        let mut values = Vec::with_capacity(self.0.len() * 2);
317        for (anchor, [x, y]) in &self.0 {
318            values.push(Json::String(anchor.clone()));
319            values.push(json!([
320                Value::Number(*x).to_json(),
321                Value::Number(*y).to_json(),
322            ]));
323        }
324        json!({ "values": values })
325    }
