1//! Static expression types (`Type`) and GL JS subtype / `typeof` naming rules.
2
3use std::fmt;
4
5/// The static type of an expression or the dynamic type of a value.
6#[derive(Clone, Debug, PartialEq, Eq)]
7pub enum Type {
8    /// `null`.
9    Null,
10    /// A number.
11    Number,
12    /// A string.
13    String,
14    /// A boolean.
15    Boolean,
16    /// A colour.
17    Color,
18    /// An object.
19    Object,
20    /// A string comparison rule.
21    Collator,
22    /// An image name and whether the map holds it.
23    ResolvedImage,
24    /// Text in sections.
25    Formatted,
26    /// Directional `[top, right, bottom, left]` box padding (`padding`).
27    Padding,
28    /// Variable-length numeric list (`numberArray`).
29    NumberArray,
30    /// Variable-length colour list (`colorArray`).
31    ColorArray,
32    /// Map projection name or blend (`projectionDefinition`).
33    ProjectionDefinition,
34    /// Alternating anchor and `[x, y]` offset pairs (`variableAnchorOffsetCollection`).
35    VariableAnchorOffsetCollection,
36    /// Bottom type produced by `["error", ...]` expressions; subtypes every type.
37    Error,
38    /// Any value; the type of a property read or an untyped branch.
39    Value,
40    /// An array, with an item type and, when known, a length.
41    Array {
42        /// Type of every item.
43        item: Box<Type>,
44        /// Number of items, when the type fixes it.
45        length: Option<usize>,
46    },
47}
48
49impl Type {
50    /// An array type.
51    pub fn array(item: Type, length: Option<usize>) -> Self {
52        Self::Array {
53            item: Box::new(item),
54            length,
55        }
56    }
57
58    /// Whether a value of this type is acceptable where `expected` is required, as GL JS
59    /// `checkSubtype` decides.
60    pub fn is_subtype_of(&self, expected: &Type) -> bool {
61        match (expected, self) {
62            (_, Type::Error) => true,
63            (
64                Type::Array {
65                    item: expected_item,
66                    length: expected_length,
67                },
68                Type::Array { item, length },
69            ) => {
70                let items_fit = (*length == Some(0) && **item == Type::Value)
71                    || item.is_subtype_of(expected_item);
72                items_fit && expected_length.is_none_or(|expected| Some(expected) == *length)
73            }
74            (Type::Array { .. }, _) => false,
75            (Type::Value, _) => true,
76            (expected, actual) => expected == actual,
77        }
78    }
79
80    /// The name GL JS prints, such as `array<number, 3>` or `padding`.
81    pub fn name(&self) -> String {
82        match self {
83            Self::Null => "null".to_string(),
84            Self::Number => "number".to_string(),
85            Self::String => "string".to_string(),
86            Self::Boolean => "boolean".to_string(),
87            Self::Color => "color".to_string(),
88            Self::Object => "object".to_string(),
89            Self::Collator => "collator".to_string(),
90            Self::ResolvedImage => "resolvedImage".to_string(),
91            Self::Formatted => "formatted".to_string(),
92            Self::Padding => "padding".to_string(),
93            Self::NumberArray => "numberArray".to_string(),
94            Self::ColorArray => "colorArray".to_string(),
95            Self::ProjectionDefinition => "projectionDefinition".to_string(),
96            Self::VariableAnchorOffsetCollection => "variableAnchorOffsetCollection".to_string(),
97            Self::Error => "error".to_string(),
98            Self::Value => "value".to_string(),
99            Self::Array { item, length } => match (item.as_ref(), length) {
100                (_, Some(length)) => format!("array<{}, {length}>", item.name()),
101                (Type::Value, None) => "array".to_string(),
102                (item, None) => format!("array<{}>", item.name()),
103            },
104        }
105    }
106}
107
108impl fmt::Display for Type {
109    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
110        formatter.write_str(&self.name())
111    }
