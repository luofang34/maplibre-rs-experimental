1//! Feature geometry for `within` and `distance`.
2//!
3//! A feature's geometry travels in a property of its own, as GeoJSON text in longitude and
4//! latitude, so every place that evaluates an expression for a feature can offer it unchanged.
5
6use serde_json::Value as Json;
7
8/// The property that carries a feature's GeoJSON geometry.
9pub const GEOMETRY_PROPERTY: &str = "\u{1}geometry";
10
11/// Longitude-latitude geometry, flattened to its points, lines and polygons.
12#[derive(Clone, Debug, Default, PartialEq)]
13pub struct Geometry {
14    /// Point and multipoint coordinates.
15    pub points: Vec<[f64; 2]>,
16    /// Line strings and multi line strings.
17    pub lines: Vec<Vec<[f64; 2]>>,
18    /// Polygons as rings, the first the outer one.
19    pub polygons: Vec<Vec<Vec<[f64; 2]>>>,
20}
21
22impl Geometry {
23    /// Reads GeoJSON, following features, collections and geometry collections.
24    pub fn from_geojson(json: &Json) -> Option<Self> {
25        let mut geometry = Self::default();
26        geometry.add(json).then_some(geometry)
27    }
28
29    fn add(&mut self, json: &Json) -> bool {
30        let coordinates = json.get("coordinates");
31        match json.get("type").and_then(Json::as_str) {
32            Some("Feature") => json.get("geometry").is_some_and(|inner| self.add(inner)),
33            Some("FeatureCollection") => json
34                .get("features")
35                .and_then(Json::as_array)
36                .is_some_and(|items| self.add_all(items)),
37            Some("GeometryCollection") => json
38                .get("geometries")
39                .and_then(Json::as_array)
40                .is_some_and(|items| self.add_all(items)),
41            Some("Point") => coordinates
42                .and_then(position)
43                .map(|p| self.points.push(p))
44                .is_some(),
45            Some("MultiPoint") => coordinates
46                .and_then(positions)
47                .map(|points| self.points.extend(points))
48                .is_some(),
49            Some("LineString") => coordinates
50                .and_then(positions)
51                .map(|line| self.lines.push(line))
52                .is_some(),
53            Some("MultiLineString") => coordinates
54                .and_then(Json::as_array)
55                .and_then(|lines| lines.iter().map(positions).collect::<Option<Vec<_>>>())
56                .map(|lines| self.lines.extend(lines))
57                .is_some(),
58            Some("Polygon") => coordinates
59                .and_then(rings)
60                .map(|polygon| self.polygons.push(polygon))
61                .is_some(),
62            Some("MultiPolygon") => coordinates
63                .and_then(Json::as_array)
64                .and_then(|polygons| polygons.iter().map(rings).collect::<Option<Vec<_>>>())
65                .map(|polygons| self.polygons.extend(polygons))
66                .is_some(),
67            _ => false,
68        }
69    }
70
71    /// The GeoJSON of the geometry, a collection when it has more than one kind of part.
72    pub fn to_geojson(&self) -> Json {
73        let mut parts = Vec::new();
74        if !self.points.is_empty() {
75            parts.push(serde_json::json!({"type": "MultiPoint", "coordinates": self.points}));
76        }
77        if !self.lines.is_empty() {
78            parts.push(serde_json::json!({"type": "MultiLineString", "coordinates": self.lines}));
79        }
80        if !self.polygons.is_empty() {
81            parts.push(serde_json::json!({"type": "MultiPolygon", "coordinates": self.polygons}));
82        }
83        serde_json::json!({"type": "GeometryCollection", "geometries": parts})
84    }
85
86    /// Adds every item, whether or not an earlier one was read.
87    fn add_all(&mut self, items: &[Json]) -> bool {
88        let mut any = false;
89        for item in items {
90            any |= self.add(item);
91        }
92        any
93    }

95    /// Whether the geometry, a feature's, lies inside `area`: every point inside a polygon, and
96    /// every line inside one without touching its boundary. A polygon feature is never within.
97    pub fn is_within(&self, area: &Geometry) -> bool {
98        if !self.polygons.is_empty() || (self.points.is_empty() && self.lines.is_empty()) {
99            return false;
100        }
101        let inside = |point: &[f64; 2]| {
102            area.polygons
103                .iter()
104                .any(|polygon| polygon_contains(polygon, *point) == Containment::Inside)
105        };
106        let line_inside = |line: &Vec<[f64; 2]>| {
107            line.iter().all(inside)
108                && !area.polygons.iter().flatten().any(|ring| {
109                    line.windows(2)
110                        .any(|pair| crosses_ring(ring, pair[0], pair[1]))
111                })
112        };
113        self.points.iter().all(inside) && self.lines.iter().all(line_inside)
114    }

116    /// The shortest distance in metres from this geometry to `other`, or `None` when either is
117    /// empty.
118    pub fn distance_to(&self, other: &Geometry) -> Option<f64> {
119        let origin = self.first_position().or_else(|| other.first_position())?;
120        let scale = LocalMetres::around(origin);
121        let ours = scale.project(self);
122        let theirs = scale.project(other);
123        if ours.is_empty() || theirs.is_empty() {
124            return None;
125        }
126        // Unwrapped lines and rings may reach past the antimeridian on either side of the origin,
127        // so the other geometry is also measured a turn of the globe east and west.
128        let turn = 360.0 * scale.per_degree[0];
129        let best = [0.0, -turn, turn]
130            .into_iter()
131            .map(|shift| ours.planar_distance(&theirs.shifted(shift)))
132            .fold(f64::INFINITY, f64::min);
133        Some(best)
134    }
135
136    /// The shortest distance between two projected geometries on the plane.
137    fn planar_distance(&self, theirs: &Geometry) -> f64 {
138        let touching = self.polygons.iter().any(|polygon| {
139            theirs
140                .all_points()
141                .any(|point| polygon_contains(polygon, point) != Containment::Outside)
142        }) || theirs.polygons.iter().any(|polygon| {
143            self.all_points()
144                .any(|point| polygon_contains(polygon, point) != Containment::Outside)
145        });
146        if touching {
147            return 0.0;
148        }
149        let mut best = f64::INFINITY;
150        for (a, b) in self.segments() {
151            for (c, d) in theirs.segments() {
152                best = best.min(segment_distance(a, b, c, d));
153            }
154        }
155        best
156    }
157
158    fn shifted(&self, x: f64) -> Geometry {
159        let shift = |points: &Vec<[f64; 2]>| points.iter().map(|p| [p[0] + x, p[1]]).collect();
160        Geometry {
161            points: shift(&self.points),
162            lines: self.lines.iter().map(shift).collect(),
163            polygons: self
164                .polygons
165                .iter()
166                .map(|polygon| polygon.iter().map(shift).collect())
167                .collect(),
168        }
169    }
170
171    fn is_empty(&self) -> bool {
172        self.points.is_empty() && self.lines.is_empty() && self.polygons.is_empty()
173    }
174
175    fn first_position(&self) -> Option<[f64; 2]> {
176        self.all_points().next()
177    }
178
179    fn all_points(&self) -> impl Iterator<Item = [f64; 2]> + '_ {
180        self.points
181            .iter()
182            .copied()
183            .chain(self.lines.iter().flatten().copied())
184            .chain(self.polygons.iter().flatten().flatten().copied())
185    }
186
187    /// Every piece as a segment; a point is a zero-length one.
188    fn segments(&self) -> Vec<([f64; 2], [f64; 2])> {
189        let mut segments: Vec<_> = self.points.iter().map(|p| (*p, *p)).collect();
190        for line in self.lines.iter().chain(self.polygons.iter().flatten()) {
191            segments.extend(line.windows(2).map(|pair| (pair[0], pair[1])));
192            if let [only] = line.as_slice() {
193                segments.push((*only, *only));
194            }
195        }
196        segments
197    }
198}

200/// Longitude-latitude to local metres around the first point of the feature, as the
201/// `CheapRuler` of MapLibre's style specification measures them: the WGS84 ellipsoid's curvature
202/// at that latitude, and longitudes taken the short way round the antimeridian.
203struct LocalMetres {
204    origin: [f64; 2],
205    per_degree: [f64; 2],
206}
207
208impl LocalMetres {
209    fn around(origin: [f64; 2]) -> Self {
210        const EQUATORIAL_RADIUS_KM: f64 = 6378.137;
211        const FLATTENING: f64 = 1.0 / 298.257_223_563;
212        const ECCENTRICITY_SQUARED: f64 = FLATTENING * (2.0 - FLATTENING);
213        let metres_per_degree_of_arc = std::f64::consts::PI / 180.0 * EQUATORIAL_RADIUS_KM * 1000.0;
214        let cos_latitude = origin[1].to_radians().cos();
215        let w2 = 1.0 / (1.0 - ECCENTRICITY_SQUARED * (1.0 - cos_latitude * cos_latitude));
216        let w = w2.sqrt();
217        Self {
218            origin,
219            per_degree: [
220                metres_per_degree_of_arc * w * cos_latitude,
221                metres_per_degree_of_arc * w * w2 * (1.0 - ECCENTRICITY_SQUARED),
222            ],
223        }
224    }
225
226    /// Each longitude is taken the short way round from the point before it, the origin for the
227    /// first, so a line or ring that crosses the antimeridian stays in one piece.
228    fn path(&self, path: &[[f64; 2]]) -> Vec<[f64; 2]> {
229        let mut previous = self.origin;
230        let mut longitude = self.origin[0];
231        path.iter()
232            .map(|p| {
233                longitude += wrap_longitude(p[0] - previous[0]);
234                previous = *p;
235                [
236                    (longitude - self.origin[0]) * self.per_degree[0],
237                    (p[1] - self.origin[1]) * self.per_degree[1],
238                ]
239            })
240            .collect()
241    }
242
243    fn project(&self, geometry: &Geometry) -> Geometry {
244        Geometry {
245            points: geometry
246                .points
247                .iter()
248                .flat_map(|p| self.path(std::slice::from_ref(p)))
249                .collect(),
250            lines: geometry.lines.iter().map(|line| self.path(line)).collect(),
251            polygons: geometry
252                .polygons
253                .iter()
254                .map(|polygon| polygon.iter().map(|ring| self.path(ring)).collect())
255                .collect(),
256        }
257    }
258}

260/// A longitude difference taken the short way round, in `[-180, 180]`.
261fn wrap_longitude(degrees: f64) -> f64 {
262    if (-180.0..=180.0).contains(&degrees) {
263        return degrees;
264    }
265    let wrapped = (degrees + 180.0).rem_euclid(360.0) - 180.0;
266    if wrapped == -180.0 && degrees > 0.0 {
267        180.0
268    } else {
269        wrapped
270    }
271}

273fn position(json: &Json) -> Option<[f64; 2]> {
274    let items = json.as_array()?;
275    Some([items.first()?.as_f64()?, items.get(1)?.as_f64()?])
276}

278fn positions(json: &Json) -> Option<Vec<[f64; 2]>> {
279    json.as_array()?.iter().map(position).collect()
280}

282fn rings(json: &Json) -> Option<Vec<Vec<[f64; 2]>>> {
283    json.as_array()?.iter().map(positions).collect()
284}

286#[derive(Clone, Copy, PartialEq, Eq)]
287enum Containment {
288    Inside,
289    Boundary,
290    Outside,
291}
292
293fn polygon_contains(polygon: &[Vec<[f64; 2]>], point: [f64; 2]) -> Containment {
294    let mut inside = false;
295    for ring in polygon {
296        for pair in ring.windows(2) {
297            let (a, b) = (pair[0], pair[1]);
298            if on_segment(a, b, point) {
299                return Containment::Boundary;
300            }
301            if (a[1] > point[1]) != (b[1] > point[1])
302                && point[0] < (b[0] - a[0]) * (point[1] - a[1]) / (b[1] - a[1]) + a[0]
303            {
304                inside = !inside;
305            }
306        }
307    }
308    if inside {
309        Containment::Inside
310    } else {
311        Containment::Outside
312    }
313}

315fn cross(o: [f64; 2], a: [f64; 2], b: [f64; 2]) -> f64 {
316    (a[0] - o[0]) * (b[1] - o[1]) - (a[1] - o[1]) * (b[0] - o[0])
317}
318
319fn on_segment(a: [f64; 2], b: [f64; 2], p: [f64; 2]) -> bool {
320    cross(a, b, p) == 0.0
321        && p[0] >= a[0].min(b[0])
322        && p[0] <= a[0].max(b[0])
323        && p[1] >= a[1].min(b[1])
324        && p[1] <= a[1].max(b[1])
325}
326
327fn segments_intersect(a: [f64; 2], b: [f64; 2], c: [f64; 2], d: [f64; 2]) -> bool {
328    let (d1, d2) = (cross(a, b, c), cross(a, b, d));
329    let (d3, d4) = (cross(c, d, a), cross(c, d, b));
330    ((d1 > 0.0) != (d2 > 0.0) && (d3 > 0.0) != (d4 > 0.0) && d1 * d2 < 0.0 && d3 * d4 < 0.0)
331        || on_segment(a, b, c)
332        || on_segment(a, b, d)
333        || on_segment(c, d, a)
334        || on_segment(c, d, b)
335}
336
337fn crosses_ring(ring: &[[f64; 2]], a: [f64; 2], b: [f64; 2]) -> bool {
338    ring.windows(2)
339        .any(|edge| segments_intersect(a, b, edge[0], edge[1]))
340}
341
342fn point_segment_distance(p: [f64; 2], a: [f64; 2], b: [f64; 2]) -> f64 {
343    let (dx, dy) = (b[0] - a[0], b[1] - a[1]);
344    let length_squared = dx * dx + dy * dy;
345    let t = if length_squared == 0.0 {
346        0.0
347    } else {
348        (((p[0] - a[0]) * dx + (p[1] - a[1]) * dy) / length_squared).clamp(0.0, 1.0)
349    };
350    (p[0] - (a[0] + t * dx)).hypot(p[1] - (a[1] + t * dy))
351}
352
353fn segment_distance(a: [f64; 2], b: [f64; 2], c: [f64; 2], d: [f64; 2]) -> f64 {
354    if segments_intersect(a, b, c, d) {
355        return 0.0;
356    }
357    point_segment_distance(a, c, d)
358        .min(point_segment_distance(b, c, d))
359        .min(point_segment_distance(c, a, b))
360        .min(point_segment_distance(d, a, b))
361}
