14enum Property {

16    Number(super::Evaluation),

18    Image,
19    Boolean,

30fn property(name: &str) -> Option<Property> {
31    Some(match name {
32        "text-color" | "text-halo-color" | "icon-color" | "icon-halo-color" => Property::Color,
33        "text-opacity" | "text-halo-width" | "text-halo-blur" | "icon-opacity"
34        | "icon-halo-width" | "icon-halo-blur" | "text-size" | "icon-size" => {
35            Property::Number(Feature)
36        }
37        "text-padding" | "icon-padding" | "symbol-spacing" | "text-max-angle" => {
38            Property::Number(Zoom)
39        }
40        "text-max-width"
41        | "text-line-height"
42        | "text-letter-spacing"
43        | "symbol-sort-key"
44        | "text-rotate"
45        | "icon-rotate"
46        | "text-radial-offset"
47        | "symbol-height-offset"
48        | "text-height-offset"
49        | "icon-height-offset" => Property::Number(Feature),
50        "text-field" | "text-anchor" | "icon-anchor" | "text-justify" => Property::Text,
51        "icon-image" => Property::Image,
52        "text-allow-overlap"
53        | "icon-allow-overlap"
54        | "text-ignore-placement"
55        | "icon-ignore-placement"
56        | "text-optional"
57        | "icon-optional" => Property::BooleanExpression,
58        "text-overlap" | "icon-overlap" => Property::Enum(&["never", "always", "cooperative"]),
59        "text-keep-upright" | "icon-keep-upright" => Property::Boolean,
60        "text-offset" | "icon-offset" => Property::Pair(Feature),
61        "text-font" => Property::Font,
62        "text-transform" => Property::Enum(&["none", "uppercase", "lowercase"]),
63        "symbol-placement" => Property::Enum(&["point", "line", "line-center"]),
64        "text-pitch-alignment" | "icon-pitch-alignment" | "icon-rotation-alignment" => {
65            Property::Enum(&["auto", "map", "viewport"])
66        }
67        "text-rotation-alignment" => Property::Enum(&["auto", "map", "viewport", "viewport-glyph"]),
68        "symbol-height-anchor" => Property::Enum(&["ground", "absolute"]),
69        "text-translate" | "icon-translate" => Property::Pair(Zoom),
70        "text-translate-anchor" | "icon-translate-anchor" => Property::Enum(&["map", "viewport"]),
71        "text-variable-anchor" => Property::Anchors,
72        "text-variable-anchor-offset" => Property::AnchorOffsets,
73        "text-writing-mode" => Property::WritingModes,
74        "symbol-z-order" => Property::Enum(&["auto", "viewport-y", "source"]),
75        "icon-text-fit" => Property::Enum(&["none", "width", "height", "both"]),
76        "icon-text-fit-padding" => Property::Padding,
77        "text-height-anchor" | "icon-height-anchor" => {
78            Property::Enum(&["ground", "sea", "absolute"])
79        }
80        _ => return None,
81    })
82}

84pub(crate) fn supports(name: &str) -> bool {
85    property(name).is_some()
86}

88pub(crate) fn is_layout(name: &str) -> bool {
89    supports(name)
90        && !matches!(
91            name,
92            "text-color"
93                | "text-halo-color"
94                | "icon-color"
95                | "icon-halo-color"
96                | "text-opacity"
97                | "text-halo-width"
98                | "text-halo-blur"
99                | "icon-opacity"
100                | "icon-halo-width"
101                | "icon-halo-blur"
102                | "text-translate"
103                | "icon-translate"
104                | "text-translate-anchor"
105                | "icon-translate-anchor"
106                | "text-height-offset"
107                | "text-height-anchor"
108                | "icon-height-offset"
109                | "icon-height-anchor"
110        )
111}

113impl LayerValidation<'_> {
114    pub(super) fn symbol(&mut self, paint: &SymbolPaint) {
115        self.property("layout.text-field", paint.text_field.as_ref(), Feature);
116        self.property("layout.text-size", paint.text_size.as_ref(), Feature);
117        for (name, value) in &paint.properties {
118            let scope = if is_layout(name) { "layout" } else { "paint" };
119            let path = format!("{scope}.{name}");
120            match property(name) {
121                Some(Property::Color) => self.property(&path, Some(&StyleProperty::<csscolorparser::Color>::parse(value)), Feature),
122                Some(Property::Pair(evaluation)) => self.property(&path, Some(&StyleProperty::<crate::style::translation::Pair>::parse(value)), evaluation),
123                Some(Property::Number(evaluation)) => self.property(&path, Some(&StyleProperty::<f32>::parse(value)), evaluation),
124                Some(Property::Text) => self.property(&path, Some(&StyleProperty::<TextField>::parse(value)), Feature),
125                Some(Property::Image) => self.property(&path, Some(&StyleProperty::<crate::style::layer::ImageName>::parse(value)), Feature),
126                Some(Property::BooleanExpression) => self.property(&path, Some(&StyleProperty::<bool>::parse(value)), Feature),
127                Some(kind) if valid_literal(&kind, value) => {}
128                Some(_) => self.unsupported(&path, "this property requires a supported literal value; expressions are not evaluated"),
129                None => self.unsupported(&path, "property is not implemented"),
130            }
131        }
132    }
133}

135fn valid_literal(property: &Property, value: &Value) -> bool {
136    match property {
137        Property::Boolean => value.is_boolean(),
138        Property::AnchorOffsets => value.as_array().is_some_and(|items| {
139            items.len() % 2 == 0
140                && items.as_chunks::<2>().0.iter().all(|pair| {
141                    pair[0].as_str().is_some()
142                        && pair[1].as_array().is_some_and(|offset| {
143                            offset.len() == 2 && offset.iter().all(Value::is_number)
144                        })
145                })
146        }),
147        Property::Anchors => value.as_array().is_some_and(|items| {
148            items.iter().all(|item| {
149                item.as_str().is_some_and(|anchor| {
150                    matches!(
151                        anchor,
152                        "center"
153                            | "left"
154                            | "right"
155                            | "top"
156                            | "bottom"
157                            | "top-left"
158                            | "top-right"
159                            | "bottom-left"
160                            | "bottom-right"
161                    )
162                })
163            })
164        }),
165        Property::WritingModes => value.as_array().is_some_and(|items| {
166            items
167                .iter()
168                .all(|item| matches!(item.as_str(), Some("horizontal" | "vertical")))
169        }),
170        Property::Padding => value
171            .as_array()
172            .is_some_and(|items| items.len() == 4 && items.iter().all(Value::is_number)),
173        Property::Font => value.as_array().is_some_and(|items| {
174            !items.is_empty()
175                && !crate::style::expression::is_expression(items)
176                && items.iter().all(Value::is_string)
177        }),
178        Property::Enum(allowed) => value.as_str().is_some_and(|value| allowed.contains(&value)),
179        _ => false,
180    }
