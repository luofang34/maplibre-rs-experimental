1//! Preserves the style document's painter order across asynchronous tile processing, and resolves
2//! layers that take their definition from another layer with `ref`.
3use serde::{Deserialize, Deserializer};
4use serde_json::{Map, Value};
5
6use super::layer::StyleLayer;
7
8pub(super) fn deserialize_layers<'de, D: Deserializer<'de>>(
9    deserializer: D,
10) -> Result<Vec<StyleLayer>, D::Error> {
11    let documents = dereferenced(Vec::<Value>::deserialize(deserializer)?);
12    let mut layers = documents
13        .into_iter()
14        .map(|document| StyleLayer::deserialize(document).map_err(serde::de::Error::custom))
15        .collect::<Result<Vec<_>, _>>()?;
16    for (index, layer) in layers.iter_mut().enumerate() {
17        layer.index = u32::try_from(index).map_err(serde::de::Error::custom)?;
18    }
19    Ok(layers)
20}
21
22/// The properties a layer with `ref` takes from the layer it refers to.
23const REF_PROPERTIES: [&str; 7] = [
24    "type",
25    "source",
26    "source-layer",
27    "minzoom",
28    "maxzoom",
29    "filter",
30    "layout",
31];
32
33/// `layers` with each `ref` replaced by the properties of the layer it names.
34fn dereferenced(layers: Vec<Value>) -> Vec<Value> {
35    let parents: Vec<(String, Map<String, Value>)> = layers
36        .iter()
37        .filter_map(|layer| {
38            let layer = layer.as_object()?;
39            Some((layer.get("id")?.as_str()?.to_owned(), layer.clone()))
40        })
41        .collect();
42    layers
43        .into_iter()
44        .map(|layer| {
45            let Some(object) = layer.as_object() else {
46                return layer;
47            };
48            let Some(reference) = object.get("ref").and_then(Value::as_str) else {
49                return layer;
50            };
51            let mut result = object.clone();
52            result.remove("ref");
53            if let Some((_, parent)) = parents.iter().find(|(id, _)| id == reference) {
54                for key in REF_PROPERTIES {
55                    if let Some(value) = parent.get(key) {
56                        result.insert(key.to_owned(), value.clone());
57                    }
58                }
59            }
60            Value::Object(result)
61        })
62        .collect()
63}
