9/// Two numbers, as the specification's `[x, y]` properties hold them.
10#[derive(Clone, Copy, Debug, PartialEq, Serialize)]
11pub struct Pair(pub [f64; 2]);
12
13impl PropertyValue for Pair {
14    fn spec() -> LegacyPropertySpec {
15        LegacyPropertySpec::interpolated(PropertyKind::Array {
16            item: Box::new(PropertyKind::Number),
17            length: Some(2),
18        })
19    }
20
21    fn from_value(value: &Value) -> Option<Self> {
22        match value {
23            Value::Array(items) => match items.as_slice() {
24                [x, y] => Some(Self([x.as_number()?, y.as_number()?])),
25                _ => None,
26            },
27            _ => None,
28        }
29    }
30
31    fn from_literal(json: &serde_json::Value) -> Option<Self> {
32        Self::from_json(json)
33    }
34}
35
36impl Pair {
37    /// A pair written out as a JSON array of two numbers.
38    pub fn from_json(json: &serde_json::Value) -> Option<Self> {
39        match json.as_array()?.as_slice() {
40            [x, y] => Some(Self([x.as_f64()?, y.as_f64()?])),
41            _ => None,
42        }
43    }
44}

46/// A translation in screen pixels: the same at every zoom, or a function of zoom.
47#[derive(Debug, Clone, Serialize, Deserialize)]
48#[serde(transparent)]
49pub struct Translation(StyleProperty<Pair>);
50
51impl Translation {
52    /// The `[x, y]` translation at `zoom`; none is no translation.
53    pub fn at_zoom(&self, zoom: f64) -> [f32; 2] {
54        self.0
55            .evaluate_at_zoom(zoom)
56            .map_or([0.0; 2], |Pair([x, y])| [x as f32, y as f32])
57    }
58}
59
60impl From<[f32; 2]> for Translation {
61    fn from([x, y]: [f32; 2]) -> Self {
62        Self(StyleProperty::Constant(Pair([f64::from(x), f64::from(y)])))
63    }
64}

66/// A layer's `*-translate` at `zoom` in screen pixels along the map's axes: a translation
67/// anchored to the viewport turns with the view's `bearing`, in radians.
68pub fn layer_translate_pixels(
69    paint: Option<&super::layer::LayerPaint>,
70    zoom: f64,
71    bearing: f64,
72) -> [f64; 2] {
73    use super::layer::{LayerPaint, TranslateAnchor};
74    let (translate, anchor) = match paint {
75        Some(LayerPaint::Fill(paint)) => (
76            translation_at(paint.fill_translate.as_ref(), zoom),
77            paint.fill_translate_anchor,
78        ),
79        Some(LayerPaint::Line(paint)) => (
80            translation_at(paint.line_translate.as_ref(), zoom),
81            paint.line_translate_anchor,
82        ),
83        Some(LayerPaint::Circle(paint)) => (
84            translation_at(paint.circle_translate.as_ref(), zoom),
85            paint.circle_translate_anchor,
86        ),
87        Some(LayerPaint::FillExtrusion(paint)) => (
88            translation_at(paint.fill_extrusion_translate.as_ref(), zoom),
89            paint.fill_extrusion_translate_anchor,
90        ),
91        _ => return [0.0; 2],
92    };
93    let [x, y] = translate.map(f64::from);
94    if anchor == TranslateAnchor::Viewport {
95        let (sin, cos) = bearing.sin_cos();
96        [x * cos - y * sin, x * sin + y * cos]
97    } else {
98        [x, y]
99    }
100}

102/// The translation a layer asks for at `zoom`; none is no translation.
103pub fn translation_at(translation: Option<&Translation>, zoom: f64) -> [f32; 2] {
104    translation.map_or([0.0; 2], |translation| translation.at_zoom(zoom))
