1//! Arabic, Persian, Urdu, and extended Arabic contextual shaping into Unicode presentation forms.
2//!
3//! When an OpenType font face is loaded, [`crate::font`] runs full `GSUB`/`GPOS` shaping via
4//! `harfrust`. When only legacy `.pbf` SDF glyph ranges are available, this module maps Unicode
5//! `U+0600..U+06FF` letters into their isolated, final, initial, and medial presentation forms
6//! (`U+FB50..U+FDFF` and `U+FE80..U+FEFC`), handles `U+0640` Tatweel and `ZWJ`/`ZWNJ`, and forms
7//! Lam-Alef ligatures across intervening `tashkeel` diacritics.
8
9/// Contextual joining class defined by Unicode Arabic Cursive Joining (Section 9.2).
10#[derive(Clone, Copy, PartialEq, Eq)]
11enum Joining {
12    /// Joins to both neighbours (isolated, final, initial, medial).
13    Dual,
14    /// Joins only to a preceding dual-joining letter (isolated, final).
15    Right,
16    /// Breaks cursive joining on both sides.
17    None,
18}
19
20/// Explicit `[isolated, final, initial, medial]` glyph codepoints and joining class for `c`.
21fn letter_forms(c: char) -> Option<([u32; 4], Joining)> {
22    let cp = u32::from(c);
23    let dual = |iso: u32| ([iso, iso + 1, iso + 2, iso + 3], Joining::Dual);
24    let right = |iso: u32| ([iso, iso + 1, iso, iso + 1], Joining::Right);
25    let self_dual = |u: u32| ([u, u, u, u], Joining::Dual);
26    let self_right = |u: u32| ([u, u, u, u], Joining::Right);
27
28    let entry = match cp {
29        // Core Arabic (U+0621..U+064A) -> Presentation Forms-B (U+FE80..U+FEF4)
30        0x0621 => ([0xFE80; 4], Joining::None),
31        0x0622 => right(0xFE81),
32        0x0623 => right(0xFE83),
33        0x0624 => right(0xFE85),
34        0x0625 => right(0xFE87),
35        0x0626 => dual(0xFE89),
36        0x0627 => right(0xFE8D),
37        0x0628 => dual(0xFE8F),
38        0x0629 => right(0xFE93),
39        0x062A..=0x062E => dual(0xFE95 + (cp - 0x062A) * 4),
40        0x062F => right(0xFEA9),
41        0x0630 => right(0xFEAB),
42        0x0631 => right(0xFEAD),
43        0x0632 => right(0xFEAF),
44        0x0633..=0x063A => dual(0xFEB1 + (cp - 0x0633) * 4),
45        // Tatweel / Kashida (U+0640) & ZWJ (U+200D) cause adjacent letters to join without changing codepoint.
46        0x0640 | 0x200D => self_dual(cp),
47        0x0641..=0x0646 => dual(0xFED1 + (cp - 0x0641) * 4),
48        0x0647 => dual(0xFEE9),
49        0x0648 => right(0xFEED),
50        0x0649 => right(0xFEEF),
51        0x064A => dual(0xFEF1),
52        // Persian, Urdu, Kurdish, Sindhi, Uyghur, Pashto -> Presentation Forms-A (U+FB50..U+FBFF)
53        0x0671 => right(0xFB50), // Alef Wasla (ٱ)
54        0x0679 => dual(0xFB66),  // Tteh (ٹ)
55        0x067A => dual(0xFB5E),  // Tteheh (ٺ)
56        0x067B => dual(0xFB52),  // Beeh (ٻ)
57        0x067E => dual(0xFB56),  // Peh (پ)
58        0x067F => dual(0xFB62),  // Teheh (ٿ)
59        0x0680 => dual(0xFB5A),  // Beheh (ڀ)
60        0x0683 => dual(0xFB76),  // Nyeh (ڃ)
61        0x0684 => dual(0xFB72),  // Dyeh (ڄ)
62        0x0686 => dual(0xFB7A),  // Tcheh (چ)
63        0x0687 => dual(0xFB7E),  // Tcheheh (ڇ)
64        0x0688 => right(0xFB88), // Ddal (ڈ)
65        0x068C => right(0xFB84), // Dahal (ڌ)
66        0x068D => right(0xFB82), // Ddahal (ڍ)
67        0x068E => right(0xFB86), // Dul (ڎ)
68        0x0691 => right(0xFB8C), // Rreh (ڑ)
69        0x0698 => right(0xFB8A), // Jeh (ژ)
70        0x06A4 => dual(0xFB6A),  // Veh (ڤ)
71        0x06A6 => dual(0xFB6E),  // Peheh (ڦ)
72        0x06A9 => dual(0xFB8E),  // Keheh (ک)
73        0x06AD => dual(0xFBD3),  // Ng (ڭ)
74        0x06AF => dual(0xFB92),  // Gaf (گ)
75        0x06B1 => dual(0xFB9A),  // Ngoeh (ڱ)
76        0x06B3 => dual(0xFB96),  // Gueh (ڳ)
77        // Urdu Noon Ghunna (ں): dotless in isolated/final, dotted Noon in initial/medial.
78        0x06BA => ([0xFB9E, 0xFB9F, 0xFEE7, 0xFEE8], Joining::Dual),
79        0x06BB => dual(0xFBA0),  // Rnoon (ڻ)
80        0x06BE => dual(0xFBAA),  // Heh Doachashmee (ھ)
81        0x06C0 => right(0xFBA4), // Heh with Yeh above (ۀ)
82        0x06C1 => dual(0xFBA6),  // Heh Goal (ہ)
83        0x06C5 => right(0xFBE0),
84        0x06C6 => right(0xFBD9),
85        0x06C7 => right(0xFBD7),
86        0x06C8 => right(0xFBDB),
87        0x06C9 => right(0xFBE2),
88        0x06CB => right(0xFBDE),
89        0x06CC => dual(0xFBFC), // Farsi Yeh (ی)
90        0x06D0 => dual(0xFBE4),
91        0x06D2 => right(0xFBAE), // Yeh Barree (ے)
92        0x06D3 => right(0xFBB0), // Yeh Barree with Hamza above (ۓ)
93        // Remaining U+0600..U+06FF Arabic letters participate in contextual joining for neighbours.
94        0x0672..=0x0677 | 0x0689..=0x0699 | 0x06C2..=0x06CD | 0x06D5 | 0x06EE..=0x06EF => {
95            self_right(cp)
96        }
97        0x063B..=0x063F
98        | 0x066E..=0x066F
99        | 0x0678..=0x0687
100        | 0x069A..=0x06BF
101        | 0x06CE..=0x06D1
102        | 0x06FA..=0x06FC
103        | 0x06FF => self_dual(cp),
104        _ => return None,
105    };
106    Some(entry)
107}

109/// Non-spacing harakat, tashkeel, and Quranic marks that do not interrupt cursive joining.
110fn is_transparent(c: char) -> bool {
111    matches!(
112        u32::from(c),
113        0x0610..=0x061A
114            | 0x064B..=0x065F
115            | 0x0670
116            | 0x06D6..=0x06DC
117            | 0x06DF..=0x06E4
118            | 0x06E7..=0x06E8
119            | 0x06EA..=0x06ED
120    )
121}

123/// Isolated and final Lam-Alef ligature codepoints for an Alef character.
124fn lam_alef(alef: char) -> Option<(char, char)> {
125    let (isolated, joined) = match u32::from(alef) {
126        0x0622 => (0xFEF5, 0xFEF6),
127        0x0623 => (0xFEF7, 0xFEF8),
128        0x0625 => (0xFEF9, 0xFEFA),
129        0x0627 => (0xFEFB, 0xFEFC),
130        _ => return None,
131    };
132    Some((char::from_u32(isolated)?, char::from_u32(joined)?))
133}

135/// Whether `text` contains any Arabic/Persian/Urdu letter that undergoes contextual shaping.
136pub fn needs_shaping(text: &str) -> bool {
137    text.chars()
138        .any(|c| !matches!(u32::from(c), 0x0640 | 0x200D) && letter_forms(c).is_some())
139}

141/// Returns `text` in logical order with each Arabic/Persian/Urdu letter mapped to its contextual
142/// presentation form and Lam + (optional marks) + Alef combined into a Lam-Alef ligature.
143pub fn shape(text: &str) -> String {
144    let chars: Vec<char> = text.chars().collect();
145    let joins_before = |index: usize| -> bool {
146        chars[..index]
147            .iter()
148            .rev()
149            .find(|c| !is_transparent(**c))
150            .and_then(|c| letter_forms(*c))
151            .is_some_and(|(_, joining)| joining == Joining::Dual)
152    };
153    let joins_after = |index: usize| -> bool {
154        chars[index + 1..]
155            .iter()
156            .find(|c| !is_transparent(**c))
157            .and_then(|c| letter_forms(*c))
158            .is_some_and(|(_, joining)| joining != Joining::None)
159    };
160
161    let mut shaped = String::with_capacity(text.len());
162    let mut index = 0;
163    while index < chars.len() {
164        let c = chars[index];
165        if c == '\u{200D}' {
166            index += 1;
167            continue;
168        }
169        let Some((table, joining)) = letter_forms(c) else {
170            shaped.push(c);
171            index += 1;
172            continue;
173        };
174        let before = joins_before(index);
175
176        // Lam + optional transparent tashkeel + Alef -> Lam-Alef ligature + preserved tashkeel.
177        if u32::from(c) == 0x0644 {
178            let mut look = index + 1;
179            while look < chars.len() && is_transparent(chars[look]) {
180                look += 1;
181            }
182            if let Some((alone, joined)) = chars.get(look).and_then(|next| lam_alef(*next)) {
183                shaped.push(if before { joined } else { alone });
184                for &mark in &chars[index + 1..look] {
185                    shaped.push(mark);
186                }
187                index = look + 1;
188                continue;
189            }
190        }
191
192        let slot = match (joining, before, joins_after(index)) {
193            (Joining::None, ..) | (_, false, false) => 0,
194            (_, true, false) => 1,
195            (Joining::Dual, false, true) => 2,
196            (Joining::Dual, true, true) => 3,
197            (Joining::Right, false, true) => 0,
198            (Joining::Right, true, true) => 1,
199        };
200        shaped.push(char::from_u32(table[slot]).unwrap_or(c));
201        index += 1;
202    }
203    shaped
204}
