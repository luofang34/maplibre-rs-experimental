1//! Style expressions: the typed language of data-driven and zoom-driven style properties.
2//!
3//! An expression is parsed once into an [`Expression`] tree, classified as feature or zoom
4//! dependent, and evaluated per feature at bucket build or per frame. Legacy function objects
5//! and legacy filters are lowered into the same tree, so one evaluator serves every property.
6//! Semantics follow the GL JS style specification; the conformance suite vendored under the
7//! crate's `tests` directory pins them.

37impl Expression {
38    /// Parses an expression with no expectation about its type.
39    pub fn parse(json: &serde_json::Value) -> Result<Self, ParseError> {
40        Parser::new().parse(json, None)
41    }
42
43    /// Parses an expression that must produce `expected`; a bare string for a colour property
44    /// becomes the colour, and a property read is checked at run time.
45    pub fn parse_for(json: &serde_json::Value, expected: &Type) -> Result<Self, ParseError> {
46        Parser::new().parse(json, Some(expected))
47    }
48
49    /// Parses the value of a style property: an expression, a legacy function object, or a
50    /// constant.
51    pub fn parse_property(
52        json: &serde_json::Value,
53        spec: &LegacyPropertySpec,
54    ) -> Result<Self, ParseError> {
55        let expected = spec.expected_type();
56        // GL JS coerces rather than asserts at the root of string properties, so a number
57        // read from a feature becomes its text instead of an error.
58        let annotation = if spec.kind == PropertyKind::String {
59            Annotation::Coerce
60        } else {
61            Annotation::Assert
62        };
63        let expression = match json {
64            serde_json::Value::Object(function) => Parser::for_legacy_function().parse_with(
65                &convert_function(function, spec),
66                Some(&expected),
67                annotation,
68            )?,
69            // An array that does not start with an operator is a literal array value, such as
70            // a light position or a font stack.
71            serde_json::Value::Array(items) if !is_expression(items) => Parser::new().parse_with(
72                &serde_json::json!(["literal", json]),
73                Some(&expected),
74                annotation,
75            )?,
76            other => Parser::new().parse_with(other, Some(&expected), annotation)?,
77        };
78        check_zoom_curve(&expression)?;
79        Ok(expression)
80    }
81
82    /// Parses a layer filter, legacy or expression syntax, into a boolean expression.
83    pub fn parse_filter(json: &serde_json::Value) -> Result<Self, ParseError> {
84        Self::parse_for(&convert_filter(json), &Type::Boolean)
85    }
86}

88/// The zoom curve a property expression may hold: at most one `step` or `interpolate` over
89/// `["zoom"]`, at the root or under `let` and `coalesce`, as GL JS `findZoomCurve` allows.
90fn check_zoom_curve(expression: &Expression) -> Result<(), ParseError> {
91    match find_zoom_curve(expression)? {
92        None if !expression.is_zoom_constant() => Err(ParseError {
93            key: String::new(),
94            message: "\"zoom\" expression may only be used as input to a top-level \"step\" or \"interpolate\" expression.".to_string(),
95        }),
96        _ => Ok(()),
97    }
98}
99
100fn find_zoom_curve(expression: &Expression) -> Result<Option<&Expression>, ParseError> {
101    let result = match expression {
102        Expression::Let { body, .. } => find_zoom_curve(body)?,
103        Expression::Coalesce { operands, .. } => {
104            let mut found = None;
105            for operand in operands {
106                found = find_zoom_curve(operand)?;
107                if found.is_some() {
108                    break;
109                }
110            }
111            found
112        }
113        Expression::Step { input, .. } | Expression::Interpolate { input, .. }
114            if matches!(**input, Expression::Global(Global::Zoom)) =>
115        {
116            Some(expression)
117        }
118        _ => None,
119    };
120    let mut children = Vec::new();
121    expression.for_each_child(&mut |child| children.push(child));
122    for child in children {
123        match (result, find_zoom_curve(child)?) {
124            (None, Some(_)) => {
125                return Err(ParseError {
126                    key: String::new(),
127                    message: "\"zoom\" expression may only be used as input to a top-level \"step\" or \"interpolate\" expression.".to_string(),
128                })
129            }
130            (Some(found), Some(child_found)) if !std::ptr::eq(found, child_found) => {
131                return Err(ParseError {
132                    key: String::new(),
133                    message: "Only one zoom-based \"step\" or \"interpolate\" subexpression may be used in an expression.".to_string(),
134                })
135            }
136            (None, None) | (Some(_), _) => {}
137        }
138    }
139    Ok(result)
140}
