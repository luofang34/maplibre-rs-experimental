11/// JSON properties grouped by their paint or layout scope.

20impl StyleLayer {
21    /// Parses a [`StyleLayer`] directly from a [`serde_json::Value`] without per-field serde derives.
22    pub fn from_json_value(value: &Value) -> Result<Self, serde_json::Error> {
23        let Some(map) = value.as_object() else {
24            return Err(de::Error::custom("layer must be a JSON object"));
25        };
26        let id = json_req_string(map, "id")?;
27        let type_ = json_req_string(map, "type")?;
28        let filter = map.get("filter").filter(|v| !v.is_null()).cloned();
29        let maxzoom = json_opt_f64(map, "maxzoom")?;
30        let minzoom = json_opt_f64(map, "minzoom")?;
31        let metadata = json_opt_value_map(map, "metadata")?;
32        let source = json_opt_string(map, "source")?;
33        let source_layer = json_opt_string(map, "source-layer")?;
34        let layout_map = match map.get("layout") {
35            None | Some(Value::Null) => None,
36            Some(Value::Object(obj)) => Some(obj),
37            Some(_) => return Err(field_err("layout", "expected object")),
38        };
39        let paint_val = map.get("paint").filter(|v| !v.is_null());
40        let rounded_from_layout = if type_ == "fill-extrusion" {
41            layout_map.and_then(|l| l.get("fill-extrusion-rounded-corner-distance"))
42        } else {
43            None
44        };
45        let input_flags = Some(compute_input_flags(
46            &type_,
47            filter.as_ref(),
48            paint_val,
49            layout_map,
50        ));
51        #[cfg(any(not(target_arch = "wasm32"), feature = "webgpu", feature = "web-webgl"))]
52        let transitions = crate::style::transitions::extract_layer_transitions_ref(
53            &id,
54            paint_val.and_then(Value::as_object),
55        )
56        .map_err(de::Error::custom)?;
57        #[cfg(not(any(not(target_arch = "wasm32"), feature = "webgpu", feature = "web-webgl")))]
58        let transitions = Default::default();
59        let mut unrecognized = LayerProperties::default();
60        let mut paint = super::paint_from_value::parse_layer_paint_from_value(
61            &type_,
62            paint_val,
63            rounded_from_layout,
64            &mut unrecognized.paint,
65        )
66        .map_err(|error| de::Error::custom(format!("layer `{id}` paint: {error}")))?;
67        let visibility = match layout_map
68            .and_then(|l| l.get("visibility"))
69            .filter(|v| !v.is_null())
70        {
71            Some(v) => super::paint_from_value::parse_visibility(v).map_err(|error| {
72                de::Error::custom(format!("layer `{id}` layout.visibility: {error}"))
73            })?,
74            None => LayerVisibility::Visible,
75        };
76        let skip_rounded = type_ == "fill-extrusion";
77        if let Some(LayerPaint::Symbol(symbol)) = &mut paint {
78            merge_symbol_layout_ref(symbol, layout_map, &mut unrecognized);
79        } else if let Some(layout) = layout_map {
80            for (k, v) in layout {
81                if k == "visibility"
82                    || (skip_rounded && k == "fill-extrusion-rounded-corner-distance")
83                {
84                    continue;
85                }
86                unrecognized.layout.insert(k.clone(), v.clone());
87            }
88        }
89        let _ = crate::style::StyleLayerId::intern(&id);
90        if let Some(source_layer) = &source_layer {
91            let _ = crate::style::SourceLayerId::intern(source_layer);
92        }
93        Ok(Self {
94            index: 0,
95            id,
96            type_,
97            filter,
98            maxzoom,
99            minzoom,
100            metadata,
101            paint,
102            source,
103            source_layer,
104            visibility,
105            transitions,
106            unrecognized,
107            input_flags,
108        })
109    }
110}

113    fn deserialize<D: serde::Deserializer<'de>>(deserializer: D) -> Result<Self, D::Error> {
114        let value = Value::deserialize(deserializer)?;
115        Self::from_json_value(&value).map_err(de::Error::custom)
116    }

119fn json_reads_geometry(value: &Value) -> bool {
120    match value {
121        Value::Array(items) => {
122            matches!(
123                items.first().and_then(Value::as_str),
124                Some("within" | "distance")
125            ) || items.iter().any(json_reads_geometry)
126        }
127        Value::Object(map) => map.values().any(json_reads_geometry),
128        _ => false,
129    }
130}

132fn is_gpu_feature_state_paint_key(layer_type: &str, key: &str) -> bool {
133    match layer_type {
134        "fill" => matches!(key, "fill-color" | "fill-opacity" | "fill-outline-color"),
135        "line" => matches!(key, "line-color" | "line-opacity"),
136        "circle" => matches!(
137            key,
138            "circle-color" | "circle-opacity" | "circle-stroke-color"
139        ),
140        "fill-extrusion" => matches!(key, "fill-extrusion-color" | "fill-extrusion-opacity"),
141        "symbol" => matches!(
142            key,
143            "text-color"
144                | "text-halo-color"
145                | "text-halo-width"
146                | "text-halo-blur"
147                | "text-opacity"
148                | "icon-color"
149                | "icon-halo-color"
150                | "icon-halo-width"
151                | "icon-halo-blur"
152                | "icon-opacity"
153        ),
154        _ => false,
155    }
156}

158fn compute_input_flags(
159    layer_type: &str,
160    filter: Option<&Value>,
161    paint: Option<&Value>,
162    layout: Option<&Map<String, Value>>,
163) -> u8 {
164    use super::{
165        FLAG_GEOM_READS_FEATURE_STATE, FLAG_READS_FEATURE_STATE, FLAG_READS_GEOMETRY,
166        FLAG_READS_IDENTITY,
167    };
168    use crate::style::{expression::reads_identity, feature_state::json_reads_feature_state};
169
170    let mut flags = 0_u8;
171    let sym_layout_reads = |pred: fn(&Value) -> bool| {
172        layer_type == "symbol"
173            && layout.is_some_and(|map| {
174                map.iter().any(|(k, v)| {
175                    k != "visibility" && crate::style::validation::symbol::supports(k) && pred(v)
176                })
177            })
178    };
179    if paint.is_some_and(reads_identity) || sym_layout_reads(reads_identity) {
180        flags |= FLAG_READS_IDENTITY;
181    }
182    if filter.is_some_and(json_reads_geometry)
183        || paint.is_some_and(json_reads_geometry)
184        || layout.is_some_and(|map| map.values().any(json_reads_geometry))
185    {
186        flags |= FLAG_READS_GEOMETRY;
187    }
188    if paint.is_some_and(json_reads_feature_state) || sym_layout_reads(json_reads_feature_state) {
189        flags |= FLAG_READS_FEATURE_STATE;
190    }
191    let geom_fs = filter.is_some_and(json_reads_feature_state)
192        || layout.is_some_and(|map| {
193            map.iter()
194                .any(|(k, v)| k != "visibility" && json_reads_feature_state(v))
195        })
196        || paint.and_then(Value::as_object).is_some_and(|map| {
197            map.iter().any(|(k, v)| {
198                !is_gpu_feature_state_paint_key(layer_type, k) && json_reads_feature_state(v)
199            })
200        });
201    if geom_fs {
202        flags |= FLAG_GEOM_READS_FEATURE_STATE;
203    }
204    flags
205}

207fn merge_symbol_layout_ref(
208    symbol: &mut SymbolPaint,
209    layout: Option<&Map<String, Value>>,
210    extra: &mut LayerProperties,
211) {
212    symbol.properties.retain(|name, value| {
213        if crate::style::validation::symbol::supports(name) {
214            true
215        } else {
216            extra.paint.insert(name.clone(), value.clone());
217            false
218        }
219    });
220    let Some(layout) = layout else {
221        return;
222    };
223    for (name, value) in layout {
224        match name.as_str() {
225            "visibility" => {}
226            "text-field" => symbol.text_field = Some(StyleProperty::parse(value)),
227            "text-size" => symbol.text_size = Some(StyleProperty::parse(value)),
228            _ if crate::style::validation::symbol::supports(name) => {
229                symbol.properties.insert(name.clone(), value.clone());
230            }
231            _ => {
232                extra.layout.insert(name.clone(), value.clone());
233            }
234        }
235    }
236}

238impl Serialize for StyleLayer {
239    fn serialize<S: serde::Serializer>(&self, serializer: S) -> Result<S::Ok, S::Error> {
240        let mut map = serializer.serialize_map(None)?;
241        map.serialize_entry("id", &self.id)?;
242        map.serialize_entry("type", &self.type_)?;
243        if let Some(value) = &self.filter {
244            map.serialize_entry("filter", value)?;
245        }
246        if let Some(value) = &self.maxzoom {
247            map.serialize_entry("maxzoom", value)?;
248        }
249        if let Some(value) = &self.minzoom {
250            map.serialize_entry("minzoom", value)?;
251        }
252        if let Some(value) = &self.metadata {
253            map.serialize_entry("metadata", value)?;
254        }
255        if let Some(value) = &self.source {
256            map.serialize_entry("source", value)?;
257        }
258        if let Some(value) = &self.source_layer {
259            map.serialize_entry("source-layer", value)?;
260        }
261        let properties = self.property_maps().map_err(serde::ser::Error::custom)?;
262        let LayerProperties { paint, layout } = properties;
263        if !paint.is_empty() || self.paint.is_some() {
264            map.serialize_entry("paint", &paint)?;
265        }
266        if !layout.is_empty() {
267            map.serialize_entry("layout", &layout)?;
268        }
269        map.end()
270    }
271}

273impl StyleLayer {
274    /// What GL JS `groupByLayout` compares: layers equal in type, source, source layer, zoom
275    /// range, filter and every layout property share one symbol bucket and one placement.
276    pub(crate) fn layout_group_key(&self) -> Option<String> {
277        use std::hash::{Hash, Hasher};
278
279        fn feed_json(hasher: &mut rustc_hash::FxHasher, value: &Value) {
280            match value {
281                Value::Null => 0_u8.hash(hasher),
282                Value::Bool(b) => {
283                    1_u8.hash(hasher);
284                    b.hash(hasher);
285                }
286                Value::Number(n) => {
287                    2_u8.hash(hasher);
288                    if let Some(u) = n.as_u64() {
289                        (0_u8, u).hash(hasher);
290                    } else if let Some(i) = n.as_i64() {
291                        (1_u8, i).hash(hasher);
292                    } else if let Some(f) = n.as_f64() {
293                        (2_u8, f.to_bits()).hash(hasher);
294                    }
295                }
296                Value::String(s) => {
297                    3_u8.hash(hasher);
298                    s.hash(hasher);
299                }
300                Value::Array(items) => {
301                    4_u8.hash(hasher);
302                    items.len().hash(hasher);
303                    for item in items {
304                        feed_json(hasher, item);
305                    }
306                }
307                Value::Object(entries) => {
308                    5_u8.hash(hasher);
309                    entries.len().hash(hasher);
310                    for (k, v) in entries {
311                        k.hash(hasher);
312                        feed_json(hasher, v);
313                    }
314                }
315            }
316        }
317
318        fn feed_style_prop<T, F: Fn(&T, &mut rustc_hash::FxHasher)>(
319            hasher: &mut rustc_hash::FxHasher,
320            prop: &StyleProperty<T>,
321            hash_const: F,
322        ) {
323            match prop {
324                StyleProperty::Constant(v) => {
325                    0_u8.hash(hasher);
326                    hash_const(v, hasher);
327                }
328                StyleProperty::Expression(expr) => {
329                    1_u8.hash(hasher);
330                    feed_json(hasher, expr.source());
331                }
332                StyleProperty::Unsupported(raw) => {
333                    2_u8.hash(hasher);
334                    feed_json(hasher, &raw.source);
335                }
336            }
337        }
338
339        let mut h0 = rustc_hash::FxHasher::default();
340        let mut h1 = rustc_hash::FxHasher::default();
341        0x243f_6a88_85a3_08d3_u64.hash(&mut h0);
342        0x1319_8a2e_0370_7344_u64.hash(&mut h1);
343
344        for hasher in [&mut h0, &mut h1] {
345            self.type_.hash(hasher);
346            self.source.hash(hasher);
347            self.source_layer.hash(hasher);
348            self.minzoom.map(f64::to_bits).hash(hasher);
349            self.maxzoom.map(f64::to_bits).hash(hasher);
350            if let Some(filter) = &self.filter {
351                1_u8.hash(hasher);
352                feed_json(hasher, filter);
353            } else {
354                0_u8.hash(hasher);
355            }
356            for (k, v) in &self.unrecognized.layout {
357                if matches!(k.as_str(), "text-field" | "text-size" | "visibility") {
358                    continue;
359                }
360                if let Some(LayerPaint::Symbol(sym)) = &self.paint {
361                    if sym.properties.contains_key(k) {
362                        continue;
363                    }
364                }
365                k.hash(hasher);
366                feed_json(hasher, v);
367            }
368            if let Some(LayerPaint::Symbol(symbol)) = &self.paint {
369                if let Some(field) = &symbol.text_field {
370                    "text-field".hash(hasher);
371                    feed_style_prop(hasher, field, |ft, h| ft.hash_layout_key(h));
372                }
373                if let Some(size) = &symbol.text_size {
374                    "text-size".hash(hasher);
375                    feed_style_prop(hasher, size, |s, h| s.to_bits().hash(h));
376                }
377                for (name, value) in &symbol.properties {
378                    if crate::style::validation::symbol::is_layout(name) {
379                        name.hash(hasher);
380                        feed_json(hasher, value);
381                    }
382                }
383            }
384            if self.is_hidden() {
385                "visibility".hash(hasher);
386                "none".hash(hasher);
387            }
388        }
389        Some(format!("{:016x}{:016x}", h0.finish(), h1.finish()))
390    }

392    pub(crate) fn property_maps(&self) -> Result<LayerProperties, serde_json::Error> {
393        let mut paint = self.unrecognized.paint.clone();
394        let mut layout = self.unrecognized.layout.clone();
395        if let Some(model) = &self.paint {
396            let tagged = serde_json::to_value(model)?;
397            if let Some(properties) = tagged.get("paint").and_then(Value::as_object) {
398                for (name, value) in properties {
399                    let destination = if matches!(model, LayerPaint::Symbol(_))
400                        && crate::style::validation::symbol::is_layout(name)
401                    {
402                        &mut layout
403                    } else {
404                        &mut paint
405                    };
406                    destination.insert(name.clone(), value.clone());
407                }
408            }
409        }
410        for (property, spec) in &self.transitions {
411            paint.insert(
412                format!("{property}-transition"),
413                serde_json::to_value(spec)?,
414            );
415        }
416        if self.is_hidden() {
417            layout.insert("visibility".into(), Value::String("none".into()));
418        }
419        Ok(LayerProperties { paint, layout })
420    }
