6/// Symbol layout enumerations whose value may follow the zoom but not the feature, per the
7/// style specification's `"expression": {"parameters": ["zoom"]}`.
8pub const ZOOM_ONLY_ENUMS: &[&str] = &[
9    "symbol-placement",
10    "text-rotation-alignment",
11    "icon-rotation-alignment",
12    "text-pitch-alignment",
13    "icon-pitch-alignment",
14    "text-overlap",
15    "icon-overlap",
16    "icon-text-fit",
17    "symbol-z-order",
18];

20/// Symbol paint enumerations that may follow the zoom but not the feature. Paint follows the
21/// view's zoom, so unlike [`ZOOM_ONLY_ENUMS`] a tile's layout does not settle them.
22pub const ZOOM_ONLY_PAINT_ENUMS: &[&str] = &["text-translate-anchor", "icon-translate-anchor"];

30impl SymbolPaint {
31    /// Returns the compiled property `name` if defined, parsing and caching it on demand.
32    pub fn compiled_property<T>(&self, name: &str) -> Option<CompiledProperty<T>>
33    where
34        T: PropertyValue + Send + Sync + 'static,
35    {
36        let source = self.properties.get(name)?;
37        self.compiled.get(name, source)
38    }
39
40    /// Evaluates a numeric symbol property using pre-compiled expression.
41    pub fn number(
42        &self,
43        name: &str,
44        properties: &FeatureProperties,
45        zoom: f64,
46        fallback: f32,
47    ) -> f32 {
48        self.compiled_property::<f32>(name)
49            .and_then(|prop| prop.evaluate_for(properties, zoom))
50            .filter(|value| value.is_finite())
51            .unwrap_or(fallback)
52    }
53
54    /// Evaluates a string or token template property using pre-compiled expression.
55    pub fn text(&self, name: &str, properties: &FeatureProperties, zoom: f64) -> Option<String> {
56        if name == "text-field" {
57            return self.field(properties, zoom, None).map(|field| field.0);
58        }
59        self.compiled_property::<TextField>(name)
60            .and_then(|prop| prop.evaluate_for(properties, zoom))
61            .map(|text| text.0)
62    }
63
64    /// The text field, evaluated against `images` when given: what an `image` in it resolves
65    /// to, and so what a `coalesce` around it chooses, depends on the images the map holds.
66    fn field(
67        &self,
68        properties: &FeatureProperties,
69        zoom: f64,
70        images: Option<&dyn crate::style::expression::ImageSet>,
71    ) -> Option<TextField> {
72        let property = self.text_field.as_ref()?;
73        let context = crate::style::expression::EvaluationContext {
74            available_images: images,
75            ..crate::style::expression::EvaluationContext::for_feature(zoom, properties)
76        };
77        property.evaluate(&context).map(|text| text.0)
78    }
79
80    /// Evaluates the image property `name`, such as `icon-image`, against the images the map
81    /// holds: what an `image` expression in it resolves to, and so what a `coalesce` around it
82    /// chooses, depends on them.
83    pub fn text_among_images(
84        &self,
85        name: &str,
86        properties: &FeatureProperties,
87        zoom: f64,
88        images: &dyn crate::style::expression::ImageSet,
89    ) -> Option<String> {
90        let property = self.compiled_property::<crate::style::layer::ImageName>(name)?;
91        let context = crate::style::expression::EvaluationContext {
92            available_images: Some(images),
93            ..crate::style::expression::EvaluationContext::for_feature(zoom, properties)
94        };
95        match property {
96            CompiledProperty::Constant(name) => Some(name.0),
97            CompiledProperty::CameraFunction(curve) => {
98                let val = curve.expression().evaluate(&context).ok()?;
99                crate::style::layer::ImageName::from_value(&val).map(|name| name.0)
100            }
101            CompiledProperty::DataDriven(expr) | CompiledProperty::Composite(expr) => {
102                let val = expr.evaluate(&context).ok()?;
103                crate::style::layer::ImageName::from_value(&val).map(|name| name.0)
104            }
105        }
106    }

108    /// Whether shared symbol height is evaluated during placement rather than tile layout.
109    pub fn uses_shared_height(&self) -> bool {
110        self.properties.contains_key("symbol-height-offset")
111    }
112
113    /// The enumeration `name` at `zoom`: the literal, or the value of a zoom expression such as
114    /// `["step", ["zoom"], "point", 11, "line"]`, which the style specification allows for the
115    /// properties in [`ZOOM_ONLY_ENUMS`] and [`ZOOM_ONLY_PAINT_ENUMS`].
116    pub fn layout_enum(&self, name: &str, zoom: f64) -> Option<String> {
117        let value = self.properties.get(name)?;
118        if let Some(literal) = value.as_str() {
119            return Some(literal.to_owned());
120        }
121        if !ZOOM_ONLY_ENUMS.contains(&name) && !ZOOM_ONLY_PAINT_ENUMS.contains(&name) {
122            return None;
123        }
124        self.text(name, &FeatureProperties::new(), zoom)
125    }
126
127    /// Replaces the zoom expressions of [`ZOOM_ONLY_ENUMS`] with their value at `zoom`, as GL JS
128    /// evaluates the layout of a tile's bucket at the tile's zoom, so that what reads them as
129    /// literals lays out the tile as the style asks.
130    pub fn resolve_zoom_enums(&mut self, zoom: f64) {
131        for name in ZOOM_ONLY_ENUMS {
132            if self
133                .properties
134                .get(*name)
135                .is_some_and(|value| !value.is_string())
136            {
137                match self.layout_enum(name, zoom) {
138                    Some(value) => {
139                        self.properties
140                            .insert((*name).to_owned(), serde_json::Value::String(value));
141                    }
142                    None => {
143                        self.properties.remove(*name);
144                    }
145                }
146            }
147        }
148    }

150    /// Evaluates the shared point-symbol height, accepting component offsets as a fallback.
151    pub fn height_offset(&self, component: &str, properties: &FeatureProperties, zoom: f64) -> f32 {
152        let name = if self.properties.contains_key("symbol-height-offset") {
153            "symbol-height-offset".to_string()
154        } else {
155            format!("{component}-height-offset")
156        };
157        self.number(&name, properties, zoom, 0.0)
158    }

160    /// Whether a symbol's height includes the terrain elevation below its anchor.
161    pub fn height_follows_ground(&self, component: &str) -> bool {
162        self.properties
163            .get("symbol-height-anchor")
164            .or_else(|| self.properties.get(&format!("{component}-height-anchor")))
165            .and_then(|value| value.as_str())
166            .unwrap_or("ground")
167            == "ground"
168    }

170    /// Font stack requested by the style.
171    pub fn font_stack(&self) -> String {
172        self.properties
173            .get("text-font")
174            .and_then(|value| value.as_array())
175            .map(|fonts| {
176                fonts
177                    .iter()
178                    .filter_map(|font| font.as_str())
179                    .collect::<Vec<_>>()
180                    .join(",")
181            })
182            .filter(|font| !font.is_empty())
183            .unwrap_or_else(|| "Open Sans Regular".to_string())
184    }

191    /// [`Self::label`] with the images the map holds, which decide what an `image` in the
192    /// field resolves to.
193    pub fn label_among(
194        &self,
195        properties: &FeatureProperties,
196        zoom: f64,
197        images: Option<&dyn crate::style::expression::ImageSet>,
198    ) -> Option<String> {
199        let (text, _) = self.shaped_field(properties, zoom, images)?;

201        (!text.is_empty()).then(|| text.to_string())
202    }

204    /// The text field with `text-transform` applied and the Arabic of each section shaped on
205    /// its own, as GL JS shapes each run of a formatted label; the sections, empty for plain
206    /// text, keep the lengths their shaped runs have.
207    fn shaped_field(
208        &self,
209        properties: &FeatureProperties,
210        zoom: f64,
211        images: Option<&dyn crate::style::expression::ImageSet>,
212    ) -> Option<(String, Vec<crate::style::property::TextSection>)> {
213        let TextField(text, sections) = self.field(properties, zoom, images)?;
214        // The case may vary by feature, as the style specification allows.
215        let text = match self.text("text-transform", properties, zoom).as_deref() {
216            Some("uppercase") => text.to_uppercase(),
217            Some("lowercase") => text.to_lowercase(),
218            _ => text,
219        };
220        let shape = |text: String| {
221            if crate::style::arabic_shaping::needs_shaping(&text) {
222                crate::style::arabic_shaping::shape(&text)
223            } else {
224                text
225            }
226        };
227        let total: usize = sections.iter().map(|section| section.length).sum();
228        // A case change that alters the length of the text leaves no way to tell which
229        // characters belong to which section.
230        if sections.is_empty() || total != text.chars().count() {
231            return Some((shape(text), Vec::new()));
232        }
233        let mut chars = text.chars();
234        let mut shaped = String::with_capacity(text.len());
235        let mut shaped_sections = Vec::with_capacity(sections.len());
236        for section in sections {
237            let run = shape(chars.by_ref().take(section.length).collect());
238            shaped_sections.push(crate::style::property::TextSection {
239                length: run.chars().count(),
240                ..section
241            });
242            shaped.push_str(&run);
243        }
244        Some((shaped, shaped_sections))
245    }

248impl SymbolPaint {
249    /// The sections of the label's text, if `text-field` is a `format` expression: the runs of
250    /// `label`'s characters with their own size, colour or font; empty for one plain section.
251    pub fn label_sections(
252        &self,
253        properties: &FeatureProperties,
254        zoom: f64,
255    ) -> Vec<crate::style::property::TextSection> {
256        self.label_sections_among(properties, zoom, None)
257    }
258
259    /// [`Self::label_sections`] with the images the map holds.
260    pub fn label_sections_among(
261        &self,
262        properties: &FeatureProperties,
263        zoom: f64,
264        images: Option<&dyn crate::style::expression::ImageSet>,
265    ) -> Vec<crate::style::property::TextSection> {
266        let Some((text, sections)) = self.shaped_field(properties, zoom, images) else {
267            return Vec::new();
268        };
269        if sections.is_empty() || text.trim().is_empty() {
270            return Vec::new();
271        }
272        let leading = text.chars().take_while(|c| c.is_whitespace()).count();
273        let mut skip = leading;
274        let mut remaining = text.trim().chars().count();
275        let mut trimmed = Vec::new();
276        for mut section in sections {
277            let dropped = skip.min(section.length);
278            skip -= dropped;
279            section.length -= dropped;
280            section.length = section.length.min(remaining);
281            remaining -= section.length;
282            if section.length > 0 {
283                trimmed.push(section);
284            }
285        }
286        trimmed
287    }
