1//! Which layers GL JS draws in its opaque pass, ahead of every translucent layer.
2//!
3//! An opaque background or fill below the first fill-extrusion layer writes depth in that pass.
4//! Layers that test depth are hidden behind it exactly as painter order hides them; a heatmap
5//! ignores depth and so shows over the opaque layers above it.
6
7use super::{
8    layer::{LayerPaint, StyleLayer},
9    Style,
10};
11
12impl Style {
13    /// The highest painter index of an opaque-pass layer drawn after `layer`, when `layer`
14    /// draws over such layers regardless of its place in the order.
15    pub fn opaque_layer_above(&self, layer: &StyleLayer, zoom: f64) -> Option<u32> {
16        if self.terrain.is_some() {
17            return None;
18        }
19        let cutoff = self
20            .layers
21            .iter()
22            .find(|candidate| candidate.type_ == "fill-extrusion")
23            .map_or(u32::MAX, |candidate| candidate.index);
24        if layer.index >= cutoff {
25            return None;
26        }
27        self.layers
28            .iter()
29            .filter(|candidate| {
30                candidate.index > layer.index
31                    && candidate.index < cutoff
32                    && candidate.is_visible_at(zoom)
33                    && candidate.is_opaque(zoom)
34            })
35            .map(|candidate| candidate.index)
36            .max()
37    }
38}
39
40impl StyleLayer {
41    /// Whether the layer paints every pixel it covers with one opaque colour.
42    fn is_opaque(&self, zoom: f64) -> bool {
43        let opaque = |color: Option<f64>, opacity: Option<f64>| {
44            color == Some(1.0) && opacity.unwrap_or(1.0) == 1.0
45        };
46        match self.paint.as_ref() {
47            Some(LayerPaint::Background(paint)) => {
48                paint.background_pattern.is_none()
49                    && opaque(
50                        paint
51                            .background_color
52                            .as_ref()
53                            .and_then(|color| color.evaluate_at_zoom(zoom))
54                            .map(|color| color.a),
55                        paint
56                            .background_opacity
57                            .as_ref()
58                            .and_then(|opacity| opacity.evaluate_at_zoom(zoom))
59                            .map(f64::from),
60                    )
61            }
62            Some(LayerPaint::Fill(paint)) => {
63                paint.fill_pattern.is_none()
64                    && paint.fill_color.as_ref().is_some_and(|color| {
65                        color.is_feature_constant()
66                            && opaque(
67                                color.evaluate_at_zoom(zoom).map(|color| color.a),
68                                paint
69                                    .fill_opacity
70                                    .as_ref()
71                                    .filter(|opacity| opacity.is_feature_constant())
72                                    .and_then(|opacity| opacity.evaluate_at_zoom(zoom))
73                                    .map(f64::from)
74                                    .or(Some(1.0)),
75                            )
76                    })
77            }
78            _ => false,
79        }
80    }
81}
