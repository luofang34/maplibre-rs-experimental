1//! Changing a loaded style: layers, their properties and their sources.
2//!
3//! Every change is checked before it is applied and leaves the style untouched when it fails.
4//! Edits are made to a layer as declared, so a layer that reads global state keeps doing so. The
5//! returned [`StyleChange`] says which layers changed and whether tiles must be fetched again;
6//! per-tile buffers bake in draw order, colors and filters, so only layers drawn from vector
7//! tiles need that, while background and raster layers are read from the style every frame.
8//! Until the fetched tiles arrive, a moved layer can be drawn in its old order relative to layers
9//! that read the live order.
10

22/// Whether a layer's content lives in per-tile buffers built from vector tiles.
23pub(crate) fn from_vector_tiles(layer: &StyleLayer) -> bool {
24    layer.type_ != "background"
25        && layer.type_ != "custom"
26        && !crate::io::tile_sources::RASTER_LAYER_TYPES.contains(&layer.type_.as_str())
27}

45    fn symbol_layers_changed(&self) -> StyleChange {
46        let layers = self
47            .layers
48            .iter()
49            .filter(|layer| layer.type_ == "symbol")
50            .map(|layer| layer.id.clone())
51            .collect();
52        self.finish(StyleChange {
53            layers,
54            ..StyleChange::default()
55        })
56    }

58    /// Adds a layer given as style JSON above the layer `before`, or on top when `None`.
59    pub fn add_layer(
60        &mut self,
61        layer: Value,
62        before: Option<&str>,
63    ) -> Result<StyleChange, StyleMutationError> {
64        let id = layer
65            .get("id")
66            .and_then(Value::as_str)
67            .unwrap_or_default()
68            .to_owned();
69        let parsed = StyleLayer::from_json_value(&layer).map_err(|source| {
70            StyleMutationError::InvalidLayer {
71                layer: id.clone(),
72                source,
73            }
74        })?;
75        if self.layers.iter().any(|existing| existing.id == parsed.id) {
76            return Err(StyleMutationError::DuplicateLayer { layer: id });
77        }
78        if let Some(source_name) = &parsed.source {
79            if !self.sources.contains_key(source_name) {
80                return Err(StyleMutationError::MissingSource {
81                    layer: id,
82                    source_name: source_name.clone(),
83                });
84            }
85        }
86        let position = self.position_before(before)?;
87        self.ensure_supported(&parsed)?;
88        self.ensure_shape(&parsed)?;
89        let needs_tessellation = from_vector_tiles(&parsed) && !parsed.is_hidden();
90        self.layers.insert(position, parsed);
91        let mut change = self.renumber();
92        push_unique(&mut change.layers, &id);
93        self.resolve_global_state();
94        self.bump_revision();
95        change.redraw_tiles = needs_tessellation;
96        Ok(change)
97    }

99    /// Removes a layer.
100    pub fn remove_layer(&mut self, id: &str) -> Result<StyleChange, StyleMutationError> {
101        let position = self.position_of(id)?;
102        let _removed = self.layers.remove(position);
103        self.active_transitions.remove_layer(id);
104        let mut change = self.renumber();
105        push_unique(&mut change.layers, id);
106        self.resolve_global_state();
107        self.bump_revision();
108        change.removed_layers.push(id.to_owned());
109        Ok(change)
110    }

112    /// Moves a layer above the layer `before`, or to the top when `None`.
113    pub fn move_layer(
114        &mut self,
115        id: &str,
116        before: Option<&str>,
117    ) -> Result<StyleChange, StyleMutationError> {
118        let position = self.position_of(id)?;
119        if before == Some(id) {
120            return Ok(StyleChange::default());
121        }
122        let layer = self.layers.remove(position);
123        let target = match self.position_before(before) {
124            Ok(target) => target,
125            Err(error) => {
126                self.layers.insert(position, layer);
127                return Err(error);
128            }
129        };
130        self.layers.insert(target, layer);
131        let change = self.renumber();
132        self.bump_revision();
133        Ok(change)
134    }

168    /// Sets a layer's filter; `None` removes it.
169    pub fn set_filter(
170        &mut self,
171        layer: &str,
172        filter: Option<Value>,
173    ) -> Result<StyleChange, StyleMutationError> {
174        self.edit_layer(layer, |document| match (document.as_object_mut(), filter) {
175            (Some(map), Some(filter)) => {
176                map.insert("filter".into(), filter);
177            }
178            (Some(map), None) => {
179                map.remove("filter");
180            }
181            _ => {}
182        })
183    }

185    /// Sets the zooms between which a layer is drawn; `None` removes a bound.
186    pub fn set_layer_zoom_range(
187        &mut self,
188        layer: &str,
189        minzoom: Option<f64>,
190        maxzoom: Option<f64>,
191    ) -> Result<StyleChange, StyleMutationError> {
192        self.edit_layer(layer, |document| {
193            if let Some(map) = document.as_object_mut() {
194                for (key, bound) in [("minzoom", minzoom), ("maxzoom", maxzoom)] {
195                    match bound {
196                        Some(bound) => map.insert(key.into(), json!(bound)),
197                        None => map.remove(key),
198                    };
199                }
200            }
201        })
202    }

204    /// Adds a source. A GeoJSON source gets a generation of its own, so a source removed and
205    /// added again under one name never shows the data it had before.
206    pub fn add_source(
207        &mut self,
208        name: &str,
209        mut source: Source,
210    ) -> Result<StyleChange, StyleMutationError> {
211        if self.sources.contains_key(name) {
212            return Err(StyleMutationError::DuplicateSource {
213                source_name: name.to_owned(),
214            });
215        }
216        let url_only = match &source {
217            Source::Vector(vector) | Source::Raster(vector) => {
218                vector.tiles.is_none() && vector.url.is_some()
219            }
220            Source::RasterDem(dem) => dem.tiles.is_none() && dem.url.is_some(),
221            Source::GeoJson(_) | Source::Image(_) | Source::Video(_) | Source::Canvas(_) => false,
222        };
223        if url_only {
224            return Err(StyleMutationError::TileJsonNotSupported {
225                source_name: name.to_owned(),
226            });
227        }
228        match &mut source {
229            Source::GeoJson(geojson) => geojson.generation = fresh_generation(),
230            Source::Image(image) => image.generation = fresh_generation(),
231            Source::Video(video) => video.generation = fresh_generation(),
232            Source::Canvas(canvas) => canvas.generation = fresh_generation(),
233            Source::Vector(_) | Source::Raster(_) | Source::RasterDem(_) => {}
234        }
235        self.sources.insert(name.to_owned(), source);
236        self.bump_revision();
237        Ok(StyleChange::default())
238    }

240    /// Removes a source no layer draws from.
241    pub fn remove_source(&mut self, name: &str) -> Result<StyleChange, StyleMutationError> {
242        if !self.sources.contains_key(name) {
243            return Err(StyleMutationError::UnknownSource {
244                source_name: name.to_owned(),
245            });
246        }
247        if let Some(layer) = self
248            .layers
249            .iter()
250            .find(|layer| layer.source.as_deref() == Some(name))
251        {
252            return Err(StyleMutationError::SourceInUse {
253                source_name: name.to_owned(),
254                layer: layer.id.clone(),
255            });
256        }
257        let removed = self.sources.remove(name);
258        self.bump_revision();
259        Ok(StyleChange {
260            removed_raster_sources: matches!(
261                removed,
262                Some(
263                    Source::Raster(_)
264                        | Source::RasterDem(_)
265                        | Source::Image(_)
266                        | Source::Video(_)
267                        | Source::Canvas(_)
268                )
269            )
270            .then(|| name.to_owned())
271            .into_iter()
272            .collect(),
273            ..StyleChange::default()
274        })
275    }

277    fn position_of(&self, id: &str) -> Result<usize, StyleMutationError> {
278        self.layers
279            .iter()
280            .position(|layer| layer.id == id)
281            .ok_or_else(|| StyleMutationError::UnknownLayer {
282                layer: id.to_owned(),
283            })
284    }

286    /// Where a layer added above `before` goes: the end when `None`.
287    fn position_before(&self, before: Option<&str>) -> Result<usize, StyleMutationError> {
288        match before {
289            Some(id) => self.position_of(id),
290            None => Ok(self.layers.len()),
291        }
292    }

294    fn ensure_supported(&self, layer: &StyleLayer) -> Result<(), StyleMutationError> {
295        let findings = self.validate_layer(layer);
296        if findings.is_empty() {
297            Ok(())
298        } else {
299            Err(StyleMutationError::Unsupported {
300                layer: layer.id.clone(),
301                findings,
302            })
303        }
304    }

306    /// Gives every layer the index of its position; returns the layers whose index changed.
307    fn renumber(&mut self) -> StyleChange {
308        let mut change = StyleChange::default();
309        for (position, layer) in self.layers.iter_mut().enumerate() {
310            let index = u32::try_from(position).unwrap_or(u32::MAX);
311            if layer.index != index {
312                layer.index = index;
313                change.layers.push(layer.id.clone());
314            }
315        }
316        change
317    }
318
319    fn finish(&self, mut change: StyleChange) -> StyleChange {
320        change.redraw_tiles |= change.layers.iter().any(|id| {
321            self.layers
322                .iter()
323                .find(|layer| &layer.id == id)
324                .is_some_and(from_vector_tiles)
325        });
326        change
327    }

329    fn edit_layer(
330        &mut self,
331        id: &str,
332        edit: impl FnOnce(&mut Value),
333    ) -> Result<StyleChange, StyleMutationError> {
334        let position = self.position_of(id)?;
335        let mut document =
336            serde_json::to_value(self.declared_layer_at(position)).map_err(|source| {
337                StyleMutationError::InvalidLayer {
338                    layer: id.to_owned(),
339                    source,
340                }
341            })?;
342        edit(&mut document);
343        let mut edited = StyleLayer::from_json_value(&document).map_err(|source| {
344            StyleMutationError::InvalidLayer {
345                layer: id.to_owned(),
346                source,
347            }
348        })?;
349        edited.index = self.layers[position].index;
350        self.ensure_supported(&edited)?;
351        // Only the zoom range is checked: an edit does not change the layer's source, so a layer
352        // loaded in a legacy shape stays editable.
353        ensure_zoom_range(&edited)?;
354        let before_layer = self.layers[position].clone();
355        let before = serde_json::to_value(&before_layer).ok();
356        self.layers[position] = edited;
357        self.state_templates.remove(id);
358        self.resolve_global_state();
359        let after = serde_json::to_value(&self.layers[position]).ok();
360        Ok(if before == after {
361            StyleChange::default()
362        } else {
363            self.bump_revision();
364            StyleChange {
365                layers: vec![id.to_owned()],
366                redraw_tiles: classify::needs_tile_redraw(&before_layer, &self.layers[position]),
367                ..Default::default()
368            }
369        })
370    }
