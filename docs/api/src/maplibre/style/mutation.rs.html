1//! Changing a loaded style: layers, their properties and their sources.
2//!
3//! Every change is checked before it is applied and leaves the style untouched when it fails.
4//! Edits are made to a layer as declared, so a layer that reads global state keeps doing so. The
5//! returned [`StyleChange`] says which layers changed and whether tiles must be fetched again;
6//! per-tile buffers bake in draw order, colors and filters, so only layers drawn from vector
7//! tiles need that, while background and raster layers are read from the style every frame.
8//! Until the fetched tiles arrive, a moved layer can be drawn in its old order relative to layers
9//! that read the live order.
10

22/// Whether a layer's content lives in per-tile buffers built from vector tiles.
23pub(crate) fn from_vector_tiles(layer: &StyleLayer) -> bool {
24    layer.type_ != "background"
25        && layer.type_ != "custom"
26        && !crate::io::tile_sources::RASTER_LAYER_TYPES.contains(&layer.type_.as_str())
27}

43    fn symbol_layers_changed(&self) -> StyleChange {
44        let layers = self
45            .layers
46            .iter()
47            .filter(|layer| layer.type_ == "symbol")
48            .map(|layer| layer.id.clone())
49            .collect();
50        self.finish(StyleChange {
51            layers,
52            ..StyleChange::default()
53        })
54    }

56    /// Adds a layer given as style JSON above the layer `before`, or on top when `None`.
57    pub fn add_layer(
58        &mut self,
59        layer: Value,
60        before: Option<&str>,
61    ) -> Result<StyleChange, StyleMutationError> {
62        let id = layer
63            .get("id")
64            .and_then(Value::as_str)
65            .unwrap_or_default()
66            .to_owned();
67        let parsed: StyleLayer =
68            serde_json::from_value(layer).map_err(|source| StyleMutationError::InvalidLayer {
69                layer: id.clone(),
70                source,
71            })?;
72        if self.layers.iter().any(|existing| existing.id == parsed.id) {
73            return Err(StyleMutationError::DuplicateLayer { layer: id });
74        }
75        if let Some(source_name) = &parsed.source {
76            if !self.sources.contains_key(source_name) {
77                return Err(StyleMutationError::MissingSource {
78                    layer: id,
79                    source_name: source_name.clone(),
80                });
81            }
82        }
83        let position = self.position_before(before)?;
84        self.ensure_supported(&parsed)?;
85        self.ensure_shape(&parsed)?;
86        let needs_tessellation = from_vector_tiles(&parsed) && !parsed.is_hidden();
87        self.layers.insert(position, parsed);
88        let mut change = self.renumber();
89        push_unique(&mut change.layers, &id);
90        self.resolve_global_state();
91        change.redraw_tiles = needs_tessellation;
92        Ok(change)
93    }

95    /// Removes a layer.
96    pub fn remove_layer(&mut self, id: &str) -> Result<StyleChange, StyleMutationError> {
97        let position = self.position_of(id)?;
98        let _removed = self.layers.remove(position);
99        self.active_transitions.remove_layer(id);
100        let mut change = self.renumber();
101        push_unique(&mut change.layers, id);
102        self.resolve_global_state();
103        change.removed_layers.push(id.to_owned());
104        Ok(change)
105    }

107    /// Moves a layer above the layer `before`, or to the top when `None`.
108    pub fn move_layer(
109        &mut self,
110        id: &str,
111        before: Option<&str>,
112    ) -> Result<StyleChange, StyleMutationError> {
113        let position = self.position_of(id)?;
114        if before == Some(id) {
115            return Ok(StyleChange::default());
116        }
117        let layer = self.layers.remove(position);
118        let target = match self.position_before(before) {
119            Ok(target) => target,
120            Err(error) => {
121                self.layers.insert(position, layer);
122                return Err(error);
123            }
124        };
125        self.layers.insert(target, layer);
126        Ok(self.renumber())
127    }

161    /// Sets a layer's filter; `None` removes it.
162    pub fn set_filter(
163        &mut self,
164        layer: &str,
165        filter: Option<Value>,
166    ) -> Result<StyleChange, StyleMutationError> {
167        self.edit_layer(layer, |document| match (document.as_object_mut(), filter) {
168            (Some(map), Some(filter)) => {
169                map.insert("filter".into(), filter);
170            }
171            (Some(map), None) => {
172                map.remove("filter");
173            }
174            _ => {}
175        })
176    }

178    /// Sets the zooms between which a layer is drawn; `None` removes a bound.
179    pub fn set_layer_zoom_range(
180        &mut self,
181        layer: &str,
182        minzoom: Option<f64>,
183        maxzoom: Option<f64>,
184    ) -> Result<StyleChange, StyleMutationError> {
185        self.edit_layer(layer, |document| {
186            if let Some(map) = document.as_object_mut() {
187                for (key, bound) in [("minzoom", minzoom), ("maxzoom", maxzoom)] {
188                    match bound {
189                        Some(bound) => map.insert(key.into(), json!(bound)),
190                        None => map.remove(key),
191                    };
192                }
193            }
194        })
195    }

197    /// Adds a source. A GeoJSON source gets a generation of its own, so a source removed and
198    /// added again under one name never shows the data it had before.
199    pub fn add_source(
200        &mut self,
201        name: &str,
202        mut source: Source,
203    ) -> Result<StyleChange, StyleMutationError> {
204        if self.sources.contains_key(name) {
205            return Err(StyleMutationError::DuplicateSource {
206                source_name: name.to_owned(),
207            });
208        }
209        let url_only = match &source {
210            Source::Vector(vector) | Source::Raster(vector) => {
211                vector.tiles.is_none() && vector.url.is_some()
212            }
213            Source::RasterDem(dem) => dem.tiles.is_none() && dem.url.is_some(),
214            Source::GeoJson(_) | Source::Image(_) | Source::Video(_) | Source::Canvas(_) => false,
215        };
216        if url_only {
217            return Err(StyleMutationError::TileJsonNotSupported {
218                source_name: name.to_owned(),
219            });
220        }
221        match &mut source {
222            Source::GeoJson(geojson) => geojson.generation = fresh_generation(),
223            Source::Image(image) => image.generation = fresh_generation(),
224            Source::Video(video) => video.generation = fresh_generation(),
225            Source::Canvas(canvas) => canvas.generation = fresh_generation(),
226            Source::Vector(_) | Source::Raster(_) | Source::RasterDem(_) => {}
227        }
228        self.sources.insert(name.to_owned(), source);
229        Ok(StyleChange::default())
230    }

232    /// Removes a source no layer draws from.
233    pub fn remove_source(&mut self, name: &str) -> Result<StyleChange, StyleMutationError> {
234        if !self.sources.contains_key(name) {
235            return Err(StyleMutationError::UnknownSource {
236                source_name: name.to_owned(),
237            });
238        }
239        if let Some(layer) = self
240            .layers
241            .iter()
242            .find(|layer| layer.source.as_deref() == Some(name))
243        {
244            return Err(StyleMutationError::SourceInUse {
245                source_name: name.to_owned(),
246                layer: layer.id.clone(),
247            });
248        }
249        let removed = self.sources.remove(name);
250        Ok(StyleChange {
251            removed_raster_sources: matches!(
252                removed,
253                Some(
254                    Source::Raster(_)
255                        | Source::RasterDem(_)
256                        | Source::Image(_)
257                        | Source::Video(_)
258                        | Source::Canvas(_)
259                )
260            )
261            .then(|| name.to_owned())
262            .into_iter()
263            .collect(),
264            ..StyleChange::default()
265        })
266    }

268    fn position_of(&self, id: &str) -> Result<usize, StyleMutationError> {
269        self.layers
270            .iter()
271            .position(|layer| layer.id == id)
272            .ok_or_else(|| StyleMutationError::UnknownLayer {
273                layer: id.to_owned(),
274            })
275    }

277    /// Where a layer added above `before` goes: the end when `None`.
278    fn position_before(&self, before: Option<&str>) -> Result<usize, StyleMutationError> {
279        match before {
280            Some(id) => self.position_of(id),
281            None => Ok(self.layers.len()),
282        }
283    }

285    fn ensure_supported(&self, layer: &StyleLayer) -> Result<(), StyleMutationError> {
286        let findings = self.validate_layer(layer);
287        if findings.is_empty() {
288            Ok(())
289        } else {
290            Err(StyleMutationError::Unsupported {
291                layer: layer.id.clone(),
292                findings,
293            })
294        }
295    }

297    /// Gives every layer the index of its position; returns the layers whose index changed.
298    fn renumber(&mut self) -> StyleChange {
299        let mut change = StyleChange::default();
300        for (position, layer) in self.layers.iter_mut().enumerate() {
301            let index = u32::try_from(position).unwrap_or(u32::MAX);
302            if layer.index != index {
303                layer.index = index;
304                change.layers.push(layer.id.clone());
305            }
306        }
307        change
308    }
309
310    fn finish(&self, mut change: StyleChange) -> StyleChange {
311        change.redraw_tiles |= change.layers.iter().any(|id| {
312            self.layers
313                .iter()
314                .find(|layer| &layer.id == id)
315                .is_some_and(from_vector_tiles)
316        });
317        change
318    }

320    fn edit_layer(
321        &mut self,
322        id: &str,
323        edit: impl FnOnce(&mut Value),
324    ) -> Result<StyleChange, StyleMutationError> {
325        let position = self.position_of(id)?;
326        let mut document =
327            serde_json::to_value(self.declared_layer_at(position)).map_err(|source| {
328                StyleMutationError::InvalidLayer {
329                    layer: id.to_owned(),
330                    source,
331                }
332            })?;
333        edit(&mut document);
334        let mut edited: StyleLayer = serde_json::from_value(document).map_err(|source| {
335            StyleMutationError::InvalidLayer {
336                layer: id.to_owned(),
337                source,
338            }
339        })?;
340        edited.index = self.layers[position].index;
341        self.ensure_supported(&edited)?;
342        // Only the zoom range is checked: an edit does not change the layer's source, so a layer
343        // loaded in a legacy shape stays editable.
344        ensure_zoom_range(&edited)?;
345        let before_layer = self.layers[position].clone();
346        let before = serde_json::to_value(&before_layer).ok();
347        self.layers[position] = edited;
348        self.state_templates.remove(id);
349        self.resolve_global_state();
350        let after = serde_json::to_value(&self.layers[position]).ok();
351        Ok(if before == after {
352            StyleChange::default()
353        } else {
354            StyleChange {
355                layers: vec![id.to_owned()],
356                redraw_tiles: classify::needs_tile_redraw(&before_layer, &self.layers[position]),
357                ..Default::default()
358            }
359        })
360    }
