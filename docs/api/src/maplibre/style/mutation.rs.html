1//! Changing a loaded style: layers, their properties and their sources.
2//!
3//! Every change is checked before it is applied and leaves the style untouched when it fails.
4//! Edits are made to a layer as declared, so a layer that reads global state keeps doing so. The
5//! returned [`StyleChange`] says which layers changed and whether tiles must be fetched again;
6//! per-tile buffers bake in draw order, colors and filters, so only layers drawn from vector
7//! tiles need that, while background and raster layers are read from the style every frame.
8//! Until the fetched tiles arrive, a moved layer can be drawn in its old order relative to layers
9//! that read the live order.
10

25/// Why a change was refused.
26#[derive(Debug, Error)]
27pub enum StyleMutationError {
28    /// A layer with this id exists already.
29    #[error("style already has a layer `{layer}`")]
30    DuplicateLayer {
31        /// The clashing id.
32        layer: String,
33    },
34    /// The style has no layer with this id.
35    #[error("style has no layer `{layer}`")]
36    UnknownLayer {
37        /// The requested id.
38        layer: String,
39    },
40    /// A source with this name exists already.
41    #[error("style already has a source `{source_name}`")]
42    DuplicateSource {
43        /// The clashing name.
44        source_name: String,
45    },
46    /// The style has no source with this name.
47    #[error("style has no source `{source_name}`")]
48    UnknownSource {
49        /// The requested name.
50        source_name: String,
51    },
52    /// A layer names a source the style does not have.
53    #[error("layer `{layer}` uses source `{source_name}`, which the style does not have")]
54    MissingSource {
55        /// The layer.
56        layer: String,
57        /// The absent source.
58        source_name: String,
59    },
60    /// A layer still draws from the source.
61    #[error("source `{source_name}` is used by layer `{layer}`")]
62    SourceInUse {
63        /// The source.
64        source_name: String,
65        /// A layer using it.
66        layer: String,
67    },
68    /// A layer that draws from a source names none.
69    #[error("layer `{layer}` needs a source")]
70    SourceRequired {
71        /// The layer.
72        layer: String,
73    },
74    /// A layer on a vector source names no source layer.
75    #[error("layer `{layer}` needs a source-layer of vector source `{source_name}`")]
76    SourceLayerRequired {
77        /// The layer.
78        layer: String,
79        /// The vector source.
80        source_name: String,
81    },
82    /// The layer's type cannot draw from the kind of source it names.
83    #[error("layer `{layer}` of type `{kind}` cannot draw from source `{source_name}`")]
84    WrongSourceKind {
85        /// The layer.
86        layer: String,
87        /// The layer's type.
88        kind: String,
89        /// The source it names.
90        source_name: String,
91    },
92    /// The lower zoom bound is above the upper one.
93    #[error("layer `{layer}` has minzoom {minzoom} above maxzoom {maxzoom}")]
94    InvalidZoomRange {
95        /// The layer.
96        layer: String,
97        /// The lower bound.
98        minzoom: f64,
99        /// The upper bound.
100        maxzoom: f64,
101    },
102    /// A vector, raster or DEM source given only as a TileJSON url, which is not fetched here.
103    #[error("source `{source_name}` gives only a TileJSON url; list its tiles instead")]
104    TileJsonNotSupported {
105        /// The source.
106        source_name: String,
107    },
108    /// The layer, or the layer with the change applied, cannot be read.
109    #[error("layer `{layer}` is invalid")]
110    InvalidLayer {
111        /// The layer.
112        layer: String,
113        /// Why it cannot be read.
114        #[source]
115        source: serde_json::Error,
116    },
117    /// The layer uses something this renderer does not support.
118    #[error("layer `{layer}` is not supported: {}", findings.first().map(ToString::to_string).unwrap_or_default())]
119    Unsupported {
120        /// The layer.
121        layer: String,
122        /// Everything that is not supported.
123        findings: Vec<StyleValidationError>,
124    },
125    /// An image update names a source that is not an image source.
126    #[error("source `{source_name}` is not an image source")]
127    NotAnImageSource {
128        /// The source.
129        source_name: String,
130    },
131}

133/// What a change did.
134#[derive(Debug, Clone, Default, PartialEq, Eq)]
135pub struct StyleChange {
136    /// Layers that were added, removed, edited or moved.
137    pub layers: Vec<String>,
138    /// Whether loaded vector tiles must be fetched again to show it.
139    pub redraw_tiles: bool,
140    /// Layers that are gone, whose leftovers in loaded tiles must be dropped.
141    pub removed_layers: Vec<String>,
142    /// Image sources whose picture or corners changed, whose tiles must be made again.
143    pub reloaded_image_sources: Vec<String>,
144    /// Raster and image sources that are gone, whose tiles and textures must be dropped.
145    pub removed_raster_sources: Vec<String>,
146}

148/// Whether a layer's content lives in per-tile buffers built from vector tiles.
149pub(crate) fn from_vector_tiles(layer: &StyleLayer) -> bool {
150    layer.type_ != "background"
151        && !crate::io::tile_sources::RASTER_LAYER_TYPES.contains(&layer.type_.as_str())
152}

155    /// Adds or replaces an image that symbol layers can draw by name. Every symbol layer is
156    /// reported as changed, since any of them may name it.
157    pub fn add_image(&mut self, name: &str, image: super::StyleImage) -> StyleChange {
158        self.images.insert(name.to_owned(), image);
159        self.symbol_layers_changed()
160    }

162    /// Removes an added image; layers that named it lose the icon.
163    pub fn remove_image(&mut self, name: &str) -> StyleChange {
164        self.images.remove(name);
165        self.symbol_layers_changed()
166    }

168    fn symbol_layers_changed(&self) -> StyleChange {
169        let layers = self
170            .layers
171            .iter()
172            .filter(|layer| layer.type_ == "symbol")
173            .map(|layer| layer.id.clone())
174            .collect();
175        self.finish(StyleChange {
176            layers,
177            ..StyleChange::default()
178        })
179    }

181    /// Adds a layer given as style JSON above the layer `before`, or on top when `None`.
182    pub fn add_layer(
183        &mut self,
184        layer: Value,
185        before: Option<&str>,
186    ) -> Result<StyleChange, StyleMutationError> {
187        let id = layer
188            .get("id")
189            .and_then(Value::as_str)
190            .unwrap_or_default()
191            .to_owned();
192        let parsed: StyleLayer =
193            serde_json::from_value(layer).map_err(|source| StyleMutationError::InvalidLayer {
194                layer: id.clone(),
195                source,
196            })?;
197        if self.layers.iter().any(|existing| existing.id == parsed.id) {
198            return Err(StyleMutationError::DuplicateLayer { layer: id });
199        }
200        if let Some(source_name) = &parsed.source {
201            if !self.sources.contains_key(source_name) {
202                return Err(StyleMutationError::MissingSource {
203                    layer: id,
204                    source_name: source_name.clone(),
205                });
206            }
207        }
208        let position = self.position_before(before)?;
209        self.ensure_supported(&parsed)?;
210        self.ensure_shape(&parsed)?;
211        self.layers.insert(position, parsed);
212        let mut change = self.renumber();
213        push_unique(&mut change.layers, &id);
214        self.resolve_global_state();
215        Ok(self.finish(change))
216    }
217
218    /// Removes a layer.
219    pub fn remove_layer(&mut self, id: &str) -> Result<StyleChange, StyleMutationError> {
220        let position = self.position_of(id)?;
221        let removed = self.layers.remove(position);
222        let mut change = self.renumber();
223        push_unique(&mut change.layers, id);
224        self.resolve_global_state();
225        change.redraw_tiles |= from_vector_tiles(&removed);
226        change.removed_layers.push(id.to_owned());
227        Ok(change)
228    }

230    /// Moves a layer above the layer `before`, or to the top when `None`.
231    pub fn move_layer(
232        &mut self,
233        id: &str,
234        before: Option<&str>,
235    ) -> Result<StyleChange, StyleMutationError> {
236        let position = self.position_of(id)?;
237        if before == Some(id) {
238            return Ok(StyleChange::default());
239        }
240        let layer = self.layers.remove(position);
241        let target = match self.position_before(before) {
242            Ok(target) => target,
243            Err(error) => {
244                self.layers.insert(position, layer);
245                return Err(error);
246            }
247        };
248        self.layers.insert(target, layer);
249        let change = self.renumber();
250        Ok(self.finish(change))
251    }

275    /// Sets a layer's filter; `None` removes it.
276    pub fn set_filter(
277        &mut self,
278        layer: &str,
279        filter: Option<Value>,
280    ) -> Result<StyleChange, StyleMutationError> {
281        self.edit_layer(layer, |document| match (document.as_object_mut(), filter) {
282            (Some(map), Some(filter)) => {
283                map.insert("filter".into(), filter);
284            }
285            (Some(map), None) => {
286                map.remove("filter");
287            }
288            _ => {}
289        })
290    }

292    /// Sets the zooms between which a layer is drawn; `None` removes a bound.
293    pub fn set_layer_zoom_range(
294        &mut self,
295        layer: &str,
296        minzoom: Option<f64>,
297        maxzoom: Option<f64>,
298    ) -> Result<StyleChange, StyleMutationError> {
299        self.edit_layer(layer, |document| {
300            if let Some(map) = document.as_object_mut() {
301                for (key, bound) in [("minzoom", minzoom), ("maxzoom", maxzoom)] {
302                    match bound {
303                        Some(bound) => map.insert(key.into(), json!(bound)),
304                        None => map.remove(key),
305                    };
306                }
307            }
308        })
309    }

311    /// Adds a source. A GeoJSON source gets a generation of its own, so a source removed and
312    /// added again under one name never shows the data it had before.
313    pub fn add_source(
314        &mut self,
315        name: &str,
316        mut source: Source,
317    ) -> Result<StyleChange, StyleMutationError> {
318        if self.sources.contains_key(name) {
319            return Err(StyleMutationError::DuplicateSource {
320                source_name: name.to_owned(),
321            });
322        }
323        let url_only = match &source {
324            Source::Vector(vector) | Source::Raster(vector) => {
325                vector.tiles.is_none() && vector.url.is_some()
326            }
327            Source::RasterDem(dem) => dem.tiles.is_none() && dem.url.is_some(),
328            Source::GeoJson(_) | Source::Image(_) => false,
329        };
330        if url_only {
331            return Err(StyleMutationError::TileJsonNotSupported {
332                source_name: name.to_owned(),
333            });
334        }
335        if let Source::GeoJson(geojson) = &mut source {
336            geojson.generation = fresh_generation();
337        }
338        self.sources.insert(name.to_owned(), source);
339        Ok(StyleChange::default())
340    }

342    /// Removes a source no layer draws from.
343    pub fn remove_source(&mut self, name: &str) -> Result<StyleChange, StyleMutationError> {
344        if !self.sources.contains_key(name) {
345            return Err(StyleMutationError::UnknownSource {
346                source_name: name.to_owned(),
347            });
348        }
349        if let Some(layer) = self
350            .layers
351            .iter()
352            .find(|layer| layer.source.as_deref() == Some(name))
353        {
354            return Err(StyleMutationError::SourceInUse {
355                source_name: name.to_owned(),
356                layer: layer.id.clone(),
357            });
358        }
359        let removed = self.sources.remove(name);
360        Ok(StyleChange {
361            removed_raster_sources: matches!(
362                removed,
363                Some(Source::Raster(_) | Source::RasterDem(_) | Source::Image(_))
364            )
365            .then(|| name.to_owned())
366            .into_iter()
367            .collect(),
368            ..StyleChange::default()
369        })
370    }

372    fn position_of(&self, id: &str) -> Result<usize, StyleMutationError> {
373        self.layers
374            .iter()
375            .position(|layer| layer.id == id)
376            .ok_or_else(|| StyleMutationError::UnknownLayer {
377                layer: id.to_owned(),
378            })
379    }

381    /// Where a layer added above `before` goes: the end when `None`.
382    fn position_before(&self, before: Option<&str>) -> Result<usize, StyleMutationError> {
383        match before {
384            Some(id) => self.position_of(id),
385            None => Ok(self.layers.len()),
386        }
387    }

389    /// Checks what GL JS checks of a layer: that its source exists, is of a kind its type can draw
390    /// from and, for vector sources, is read through a source layer, and that the zoom range is
391    /// ordered.
392    fn ensure_shape(&self, layer: &StyleLayer) -> Result<(), StyleMutationError> {
393        let id = layer.id.clone();
394        ensure_zoom_range(layer)?;
395        if layer.type_ == "background" {
396            return Ok(());
397        }
398        let Some(source_name) = &layer.source else {
399            return Err(StyleMutationError::SourceRequired { layer: id });
400        };
401        let Some(source) = self.sources.get(source_name) else {
402            return Err(StyleMutationError::MissingSource {
403                layer: id,
404                source_name: source_name.clone(),
405            });
406        };
407        let fits = match layer.type_.as_str() {
408            "raster" => matches!(source, Source::Raster(_) | Source::RasterDem(_)),
409            "hillshade" | "color-relief" => matches!(source, Source::RasterDem(_)),
410            _ => matches!(source, Source::Vector(_) | Source::GeoJson(_)),
411        };
412        if !fits {
413            return Err(StyleMutationError::WrongSourceKind {
414                layer: id,
415                kind: layer.type_.clone(),
416                source_name: source_name.clone(),
417            });
418        }
419        if matches!(source, Source::Vector(_)) && layer.source_layer.is_none() {
420            return Err(StyleMutationError::SourceLayerRequired {
421                layer: id,
422                source_name: source_name.clone(),
423            });
424        }
425        Ok(())
426    }

428    fn ensure_supported(&self, layer: &StyleLayer) -> Result<(), StyleMutationError> {
429        let findings = self.validate_layer(layer);
430        if findings.is_empty() {
431            Ok(())
432        } else {
433            Err(StyleMutationError::Unsupported {
434                layer: layer.id.clone(),
435                findings,
436            })
437        }
438    }

440    /// Gives every layer the index of its position; returns the layers whose index changed.
441    fn renumber(&mut self) -> StyleChange {
442        let mut change = StyleChange::default();
443        for (position, layer) in self.layers.iter_mut().enumerate() {
444            let index = u32::try_from(position).unwrap_or(u32::MAX);
445            if layer.index != index {
446                layer.index = index;
447                change.layers.push(layer.id.clone());
448                change.redraw_tiles |= from_vector_tiles(layer);
449            }
450        }
451        change
452    }
453
454    fn finish(&self, mut change: StyleChange) -> StyleChange {
455        change.redraw_tiles |= change.layers.iter().any(|id| {
456            self.layers
457                .iter()
458                .find(|layer| &layer.id == id)
459                .is_some_and(from_vector_tiles)
460        });
461        change
462    }

464    fn edit_layer(
465        &mut self,
466        id: &str,
467        edit: impl FnOnce(&mut Value),
468    ) -> Result<StyleChange, StyleMutationError> {
469        let position = self.position_of(id)?;
470        let mut document =
471            serde_json::to_value(self.declared_layer_at(position)).map_err(|source| {
472                StyleMutationError::InvalidLayer {
473                    layer: id.to_owned(),
474                    source,
475                }
476            })?;
477        edit(&mut document);
478        let mut edited: StyleLayer = serde_json::from_value(document).map_err(|source| {
479            StyleMutationError::InvalidLayer {
480                layer: id.to_owned(),
481                source,
482            }
483        })?;
484        edited.index = self.layers[position].index;
485        self.ensure_supported(&edited)?;
486        // Only the zoom range is checked: an edit does not change the layer's source, so a layer
487        // loaded in a legacy shape stays editable.
488        ensure_zoom_range(&edited)?;
489        let before = serde_json::to_value(&self.layers[position]).ok();
490        self.layers[position] = edited;
491        self.state_templates.remove(id);
492        self.resolve_global_state();
493        let after = serde_json::to_value(&self.layers[position]).ok();
494        Ok(if before == after {
495            StyleChange::default()
496        } else {
497            self.finish(StyleChange {
498                layers: vec![id.to_owned()],
499                ..Default::default()
500            })
501        })
502    }
503}
504
505fn ensure_zoom_range(layer: &StyleLayer) -> Result<(), StyleMutationError> {
506    match (layer.minzoom, layer.maxzoom) {
507        (Some(minzoom), Some(maxzoom)) if minzoom > maxzoom => {
508            Err(StyleMutationError::InvalidZoomRange {
509                layer: layer.id.clone(),
510                minzoom,
511                maxzoom,
512            })
513        }
514        _ => Ok(()),
515    }
516}
517
518fn push_unique(ids: &mut Vec<String>, id: &str) {
519    if !ids.iter().any(|existing| existing == id) {
520        ids.push(id.to_owned());
521    }
522}

524/// Sets `document[section][name]`, or removes it for `null`.
525fn set_member(document: &mut Value, section: &str, name: &str, value: Value) {
526    let Some(map) = document.as_object_mut() else {
527        return;
528    };
529    if value.is_null() {
530        if let Some(Value::Object(members)) = map.get_mut(section) {
531            members.remove(name);
532        }
533        return;
534    }
535    let members = map.entry(section).or_insert_with(|| json!({}));
536    if let Some(members) = members.as_object_mut() {
537        members.insert(name.to_owned(), value);
538    }
539}

541impl MapContext {
542    /// Applies a style change and fetches loaded vector tiles again when it needs that; the old
543    /// content stays on screen until the new tiles arrive.
544    pub fn mutate_style(
545        &mut self,
546        apply: impl FnOnce(&mut Style) -> Result<StyleChange, StyleMutationError>,
547    ) -> Result<StyleChange, StyleMutationError> {
548        let change = apply(&mut self.style)?;
549        crate::render::frame_signals::mark_dirty(&mut self.world);
550        if !change.removed_layers.is_empty() {
551            crate::vector::content::purge_layers(&mut self.world, &change.removed_layers);
552        }
553        if change.redraw_tiles {
554            tile_retry::refresh(&mut self.world, RequestKind::Vector);
555        }
556        for name in &change.removed_raster_sources {
557            crate::raster::forget_source(&mut self.world, name);
558        }
559        for name in &change.reloaded_image_sources {
560            crate::raster::image_source::reload(&mut self.world, &self.style, name);
561        }
562        Ok(change)
563    }
564}
