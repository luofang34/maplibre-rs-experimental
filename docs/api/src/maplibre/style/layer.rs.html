29/// Typed properties selected by a style layer's rendering type.
30#[derive(Serialize, Deserialize, Debug, Clone)]
31#[serde(tag = "type", content = "paint")]
32pub enum LayerPaint {
33    /// Background color beneath source-backed layers.
34    #[serde(rename = "background")]
35    Background(BackgroundPaint),
36    /// Stroked paths with width, opacity, translation and dash properties.
37    #[serde(rename = "line")]
38    Line(LinePaint),
39    /// Filled polygon color, opacity and translation.
40    #[serde(rename = "fill")]
41    Fill(FillPaint),
42    /// Polygons raised to a height, with lit walls and roof.
43    #[serde(rename = "fill-extrusion")]
44    FillExtrusion(FillExtrusionPaint),
45    /// Image sampling and adjustment properties retained from the style.
46    #[serde(rename = "raster")]
47    Raster(RasterPaint),
48    /// Elevation-dependent illumination and shadow colors.
49    #[serde(rename = "hillshade")]
50    Hillshade(HillshadePaint),
51    /// Color ramp evaluated against decoded elevation.
52    #[serde(rename = "color-relief")]
53    ColorRelief(ColorReliefPaint),
54    /// Text and icon appearance and layout properties.
55    #[serde(rename = "symbol")]
56    Symbol(SymbolPaint),
57    /// Point radius, fill, stroke and alignment properties.
58    #[serde(rename = "circle")]
59    Circle(CirclePaint),
60    /// Point density radius, weight, intensity, opacity and colour ramp.
61    #[serde(rename = "heatmap")]
62    Heatmap(HeatmapPaint),
63}
64
65impl LayerPaint {
66    /// The opacity property multiplied into the layer's colour, when the layer type has one.
67    pub fn opacity(&self) -> Option<StyleProperty<f32>> {
68        match self {
69            LayerPaint::Fill(paint) => paint.fill_opacity.clone(),
70            LayerPaint::Line(paint) => paint.line_opacity.clone(),
71            LayerPaint::Circle(paint) => paint.circle_opacity.clone(),
72            LayerPaint::ColorRelief(paint) => paint.color_relief_opacity.clone(),
73            LayerPaint::Background(_)
74            | LayerPaint::FillExtrusion(_)
75            | LayerPaint::Raster(_)
76            | LayerPaint::Hillshade(_)
77            | LayerPaint::Symbol(_)
78            | LayerPaint::Heatmap(_) => None,
79        }
80    }
81
82    /// Returns a constant base color in encoded sRGB with straight alpha.
83    /// Expressions, absent colors and paint families without one base color return `None`.
84    /// Layer opacity is not multiplied into this value.
85    pub fn get_color(&self) -> Option<Alpha<EncodedSrgb<f32>>> {
86        let property = match self {
87            Self::Background(paint) => &paint.background_color,
88            Self::Line(paint) => &paint.line_color,
89            Self::Fill(paint) => &paint.fill_color,
90            Self::Circle(paint) => &paint.circle_color,
91            Self::FillExtrusion(paint) => &paint.fill_extrusion_color,
92            Self::Raster(_)
93            | Self::Hillshade(_)
94            | Self::ColorRelief(_)
95            | Self::Symbol(_)
96            | Self::Heatmap(_) => return None,
97        };
98        match property.as_ref()? {
99            StyleProperty::Constant(color) => Some(color.clone().into()),
100            _ => None,
101        }
102    }
103}

105/// Whether a layer is drawn at all, the `layout.visibility` property.
106#[derive(Serialize, Deserialize, Debug, Clone, Copy, Default, Eq, PartialEq)]
107pub enum LayerVisibility {
108    /// The layer is drawn.
109    #[default]
110    #[serde(rename = "visible")]
111    Visible,
112    /// The layer is neither laid out nor drawn.
113    #[serde(rename = "none")]
114    None,
115}

117/// One entry in the style's painter order. Equality and hashing use only the layer ID.
118/// Zoom bounds control display visibility independently of source tile zoom limits.
119#[derive(Debug, Clone)]
120pub struct StyleLayer {
121    /// Zero-based painter order assigned when the containing style is deserialized.
122    pub index: u32,
123    /// Document-local identity used for equality, hashing and matching processed geometry.
124    pub id: String,
125    /// Rendering type from JSON, used to select paint parsing and the corresponding pipeline.
126    pub type_: String,
127    /// Feature-selection expression retained as JSON for filter validation and evaluation.
128    pub filter: Option<serde_json::Value>,
129    /// Exclusive upper display zoom; fractional values are preserved. `None` sets no upper bound.
130    pub maxzoom: Option<f64>,
131    /// Inclusive lower display zoom; fractional values are preserved. `None` sets no lower bound.
132    pub minzoom: Option<f64>,
133    /// Application JSON properties retained across worker serialization.
134    /// Reserved `maplibre-rs:terrain-structure` keys opt into terrain-relative road profiles.
135    pub metadata: Option<HashMap<String, serde_json::Value>>,
136    /// Typed rendering properties; absence lets the relevant rendering path choose defaults.
137    pub paint: Option<LayerPaint>,
138    /// Key of a source in the containing style, or `None` for source-independent or host-supplied data.
139    pub source: Option<String>,
140    /// Layer name inside a vector tile, distinct from the style layer ID.
141    pub source_layer: Option<String>,
142    /// Whether the layer is drawn at all.
143    pub visibility: LayerVisibility,
144    /// Properties without a typed representation, retained for diagnostics and round trips.
145    pub unrecognized: LayerProperties,
146}
147
148impl StyleLayer {
149    /// Whether the layer is switched off by its `layout.visibility`, at every zoom.
150    pub fn is_hidden(&self) -> bool {
151        self.visibility == LayerVisibility::None
152    }
153
154    /// Returns whether the layer is drawn at a continuous zoom: it is not hidden and the zoom
155    /// lies in its range, using the style-spec rule that `minzoom` is inclusive and `maxzoom`
156    /// exclusive.
157    pub fn is_visible_at(&self, zoom: f64) -> bool {
158        !self.is_hidden()
159            && self.minzoom.is_none_or(|minzoom| zoom >= minzoom)
160            && self.maxzoom.is_none_or(|maxzoom| zoom < maxzoom)
161    }
162}
