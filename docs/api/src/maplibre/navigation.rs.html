1//! Navigation controls shared by every map host: 2D/3D mode, Auto-Tilt, North-Lock, the RTS
2//! pointer profile, HUD button interaction, and smooth camera settling.
3//!
4//! A host feeds pointer positions in logical pixels, maps its buttons and gestures onto the
5//! [`NavigationState`] actions, and calls [`NavigationState::update_frame`] once per frame, so
6//! the winit and web hosts and embedders that draw their own controls behave the same.
7

27/// Pointer interaction semantics for mouse/trackpad input.
28#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
29pub enum PointerProfile {
30    /// Standard map controls: Left-drag pans (`Shift`+Left-drag box-selects), Right-drag orbits.
31    #[default]
32    Map,
33    /// RTS controls (RA2 / C&C Generals style): Left-drag box-selects, Right-drag pans,
34    /// Middle-drag or `Alt`+drag orbits, and screen edges scroll the viewport.
35    Rts,
36}

38/// Snapshot of the initial camera pose used by the `Home` reset action and Auto-Tilt calibration.
39#[derive(Clone, Copy, Debug)]
40pub struct HomeCamera {
41    /// Normalized Mercator coordinates in `0..1`.
42    pub normalized_center: Point2<f64>,
43    /// Initial zoom level.
44    pub zoom: f64,
45    /// Initial camera bearing.
46    pub bearing: Rad<f64>,
47    /// Initial camera pitch.
48    pub pitch: Rad<f64>,
49    /// Whether the style uses 3D terrain or a non-Mercator globe projection.
50    pub has_3d_or_globe: bool,
51}

53/// Shared state across input handlers for 2D/3D mode, Auto-Tilt, North-Lock, RTS mode,
54/// HUD button interactions, and smooth settle animations.
55pub struct NavigationState {
56    initialized: bool,
57    home: Option<HomeCamera>,
58    /// Whether 3D pitch is allowed (`true` = 3D, `false` = locked 2D top-down).
59    pub three_dimensional: bool,
60    /// Whether zoom gestures automatically adjust camera pitch along the zoom-tilt curve.
61    pub auto_tilt: bool,
62    /// Whether camera bearing is locked to North (`0°`).
63    pub north_locked: bool,
64    /// Active mouse pointer profile (`Map` vs `Rts`).
65    pub pointer_profile: PointerProfile,
66    /// Whether the floating top-right navigation HUD is visible.
67    pub show_hud: bool,
68    /// Latest cursor position in logical window pixels.
69    pub cursor_position: Option<Vector2<f64>>,
70    /// Currently hovered HUD button.
71    pub hovered_button: Option<HudButton>,
72    /// Currently pressed HUD button.
73    pub pressed_button: Option<HudButton>,
74    /// Initial press position when interacting with the HUD compass, to distinguish click vs drag.
75    compass_press_start: Option<Vector2<f64>>,
76    /// Previous cursor position while dragging the HUD compass.
77    compass_drag_last: Option<Vector2<f64>>,
78    /// Whether the current HUD compass press turned into a drag gesture.
79    compass_dragged: bool,
80    /// Active rubber-band selection box (`start_px`, `current_px`) in logical window pixels.
81    pub selection_drag: Option<(Vector2<f64>, Vector2<f64>)>,
82    /// Smoothly animated target pitch in degrees.
83    target_pitch_deg: Option<f64>,
84    /// Smoothly animated target bearing in radians.
85    target_bearing_rad: Option<f64>,
86    /// Whether a smooth fly-to-home animation is running.
87    fly_home: bool,
88    /// Smooth zoom delta requested by HUD buttons or double-click, consumed by `ZoomHandler`.
89    pending_smooth_zoom: f64,
90    /// Last left-click / tap time and logical pixel position for double-click detection.
91    last_primary_click: Option<(Instant, Vector2<f64>)>,
92    /// Pending trackpad rotation gesture delta in radians (`WindowEvent::RotationGesture`).
93    pending_trackpad_rotation: Rad<f64>,
94    /// Whether `Shift` modifier is currently held.
95    pub shift_held: bool,
96    /// Whether `Alt` / `Option` modifier is currently held.
97    pub alt_held: bool,
98    /// Whether `Control` / `Super` modifier is currently held.
99    pub ctrl_held: bool,
100    /// Whether screen-edge scrolling is explicitly enabled even in windowed Map mode.
101    pub edge_scroll_override: bool,
102    /// Maximum screen-edge scrolling speed in logical pixels per second.
103    pub edge_scroll_speed_px_per_sec: f64,
104}
