128impl HasTile for VectorTilesDone {
129    fn has_tile(&self, coords: WorldTileCoords, world: &World) -> bool {
130        world
131            .tiles
132            .query::<&VectorLayerBucketComponent>(coords)
133            .is_some_and(|buckets| buckets.done && !buckets.failed)
134    }
135
136    fn nearest_ancestor_candidate(
137        &self,
138        coords: WorldTileCoords,
139        world: &World,
140    ) -> Option<WorldTileCoords> {
141        world
142            .tiles
143            .pyramid_of::<VectorLayerBucketComponent>()
144            .nearest_ancestor(coords)
145            .map(|(ancestor, _)| ancestor)
146    }
147
148    fn has_descendant_candidates(
149        &self,
150        coords: WorldTileCoords,
151        max_depth: usize,
152        world: &World,
153    ) -> bool {
154        world
155            .tiles
156            .pyramid_of::<VectorLayerBucketComponent>()
157            .has_descendants(coords, max_depth)
158    }
159}

161/// Whether a tile's geometry has reached the buffer pool, which the upload system fills in
162/// one pass per tile, or the tile has nothing to upload: no vector data at all, or none with
163/// geometry. A drape drawn from a finished tile before its upload would miss every layer.
164pub fn geometry_uploaded(coords: WorldTileCoords, world: &World) -> bool {
165    let Some(buckets) = world.tiles.query::<&VectorLayerBucketComponent>(coords) else {
166        return true;
167    };
168    if !buckets.done {
169        return false;
170    }
171    let has_geometry = buckets.layers.iter().any(|layer| match layer {
172        VectorLayerBucket::AvailableLayer(bucket) => !bucket.buffer.buffer.indices.is_empty(),
173        VectorLayerBucket::Missing(_) => false,
174    });
175    if !has_geometry {
176        return true;
177    }
178    match world.resources.get::<Eventually<VectorBufferPool>>() {
179        Some(Eventually::Initialized(pool)) => {
180            if !pool.has_capacity() {
181                return true;
182            }
183            buckets.layers.iter().all(|layer| match layer {
184                VectorLayerBucket::AvailableLayer(bucket) => {
185                    bucket.buffer.buffer.indices.is_empty()
186                        || pool.has_layer(coords, &bucket.style_layer_id)
187                }
188                VectorLayerBucket::Missing(_) => true,
189            })
190        }
191        _ => false,
192    }
193}

195/// Checks that a tile's vector geometry has reached the active GPU buffer pool.
196pub(crate) struct UploadedVectorGeometry {
197    check_gpu: bool,
198    pub(crate) max_zoom: Option<u8>,
199}
200
201impl UploadedVectorGeometry {
202    pub(crate) fn for_world(world: &World) -> Self {
203        let (check_gpu, pool_max) = match world.resources.get::<Eventually<VectorBufferPool>>() {
204            Some(Eventually::Initialized(pool)) if pool.has_capacity() => {
205                (true, pool.index().max_zoom())
206            }
207            _ => (false, None),
208        };
209        let tiles_max = world.tiles.pyramid().max_zoom();
210        let max_zoom = if check_gpu {
211            Some(pool_max.unwrap_or(0).max(tiles_max.unwrap_or(0)))
212        } else {
213            None
214        };
215        Self {
216            check_gpu,
217            max_zoom,
218        }
219    }
220}

222impl HasTile for UploadedVectorGeometry {
223    fn has_tile(&self, coords: WorldTileCoords, world: &World) -> bool {
224        !self.check_gpu || geometry_uploaded(coords, world)
225    }
226}

228impl<E: Environment, T: VectorTransferables> Plugin<E> for VectorPlugin<T> {
229    fn build(
230        &self,
231        schedule: &mut Schedule,
232        kernel: Rc<Kernel<E>>,
233        world: &mut World,
234        _renderer: &mut crate::render::Renderer,
235    ) -> Result<(), crate::plugin::PluginError> {
236        let resources = &mut world.resources;
237
238        resources.insert(Eventually::<VectorBufferPool>::Uninitialized);
239        resources.insert(Eventually::<VectorPipeline>::Uninitialized);
240        resources.insert(Eventually::<LinePipeline>::Uninitialized);
241        resources.insert(Eventually::<CirclePipeline>::Uninitialized);
242        resources.insert(Eventually::<ExtrusionPipeline>::Uninitialized);
243
244        resources
245            .get_or_init_mut::<ViewTileSources>()
246            .add::<VectorTilesDone>(TileKind::Vector);
247
248        schedule.add_system_to_stage(
249            RenderStageLabel::Request,
250            SystemContainer::new(RequestSystem::<E, T>::new(&kernel)),
251        )?;
252        schedule.add_system_to_stage(
253            RenderStageLabel::Request,
254            SystemContainer::new(PopulateWorldSystem::<E, T>::new(&kernel)),
255        )?;
256
257        schedule.add_system_to_stage(RenderStageLabel::Prepare, resource_system)?;
258        schedule.add_system_to_stage(RenderStageLabel::Queue, queue_system)?;
259        Ok(())
260    }
261}

263/// Tessellated geometry and paint for one style layer of a source tile, before GPU upload.
264pub struct AvailableVectorLayerBucket {
265    /// Tile-grid coordinates owning the geometry.
266    pub coords: WorldTileCoords,
267    /// Layer name inside the vector source tile, distinct from the style layer ID.
268    pub source_layer: String,
269    /// Style layer whose filter and paint produced this bucket.
270    pub style_layer_id: String,
271    /// Tile-space geometry with an unpadded draw-index count.
272    pub buffer: OverAlignedVertexBuffer<ShaderVertex, IndexDataType>,
273    /// Number of vertices contributed by each feature, in tessellation order.
274    pub feature_indices: Vec<u32>,
275    /// Encoded-sRGB colors with straight alpha, indexed in the same feature order.
276    /// Missing entries use the layer's fallback color during upload.
277    pub feature_colors: Vec<[f32; 4]>,
278    /// The sort key of each feature, in the same order; empty without a sort key.
279    pub feature_sort_keys: Vec<f32>,
280}

290/// One worker result; an available bucket may still have empty geometry.
291pub enum VectorLayerBucket {
292    /// Tessellation and paint data awaiting upload.
293    AvailableLayer(AvailableVectorLayerBucket),
294    /// No usable geometry was returned for the source layer.
295    Missing(MissingVectorLayerBucket),

298/// Accumulated tile buckets and completion state used to select loading fallbacks.
299#[derive(Default)]
300pub struct VectorLayerBucketComponent {
301    /// Base processing has finished; symbol work may still hold the loading slot.
302    pub done: bool,
303    /// Source data was unavailable, so this tile must not replace a usable ancestor.
304    pub failed: bool,
305    /// Worker results, which may arrive before `done` marks base processing complete.
306    pub layers: Vec<VectorLayerBucket>,
307    /// The zoom the tile was last requested to be magnified to, zero when it was not.
308    pub overscaled_zoom: u8,
309}
