17/// Platform services selected together at compile time.
18/// The worker transport must support the same offscreen kernel used to fetch source data.
19pub trait Environment: 'static {
20    /// Factory for the map's host window.
21    type MapWindowConfig: MapWindowConfig;
22
23    /// Request/reply transport that runs procedures with the offscreen kernel.
24    type AsyncProcedureCall: AsyncProcedureCall<Self::OffscreenKernelEnvironment>;
25
26    /// Executor available to systems running on the map thread.
27    type Scheduler: Scheduler;
28
29    /// HTTP implementation used by this environment's source loader.
30    type HttpClient: HttpClient;
31
32    /// Services that a worker can create from serialized configuration.
33    type OffscreenKernelEnvironment: OffscreenKernel;
34}
35
36/// Configuration passed across worker boundaries to construct source services.
37#[derive(Serialize, Deserialize, Clone, Debug, Default)]
38pub struct OffscreenKernelConfig {
39    /// Filesystem cache directory for hosts that support persistent HTTP caching.
40    /// `None` disables that cache; browser-only kernels may ignore the setting.
41    pub cache_directory: Option<String>,
42    /// Decoded glyphs and sprites shared by every call made from clones of this configuration.
43    /// It is not serialized: a worker that receives the configuration over a channel starts
44    /// with its own empty cache.
45    #[serde(skip)]
46    pub asset_cache: AssetCache,
47    /// The providers of images labels name and no sprite supplies, shared like the asset
48    /// cache. A worker that receives its configuration over a channel registers its own.
49    #[serde(skip)]
50    pub image_providers: ImageProviders,
51    /// The loader the map thread uses, so in-process workers fetch through the same one.
52    /// Like the asset cache it is not serialized: a worker in another process or a browser
53    /// worker builds its platform's own loader.
54    #[serde(skip)]
55    pub loader: Option<crate::io::resource_loader::SharedLoader>,
56}
57
58/// Worker-side services, independent of window and GPU ownership.
59pub trait OffscreenKernel: Send + Sync + 'static {
60    /// HTTP implementation used by this environment's source loader.
61    type HttpClient: HttpClient;
62    /// Constructs worker services from the configuration supplied at worker startup.
63    fn create(config: OffscreenKernelConfig) -> Self;
64
65    /// Creates a source loader using this worker's HTTP and cache configuration.
66    fn source_client(&self) -> SourceClient<Self::HttpClient>;
67}
