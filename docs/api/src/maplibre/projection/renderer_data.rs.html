8/// Matrices used while selecting or transitioning between map projections.
9#[derive(Clone, Copy, Debug, PartialEq)]
10pub struct ProjectionMatrices {
11    /// Projects Mercator tile coordinates to clip space.
12    pub mercator: Matrix4<f32>,
13    /// Projects the unit globe to clip space.
14    pub globe: Matrix4<f32>,

17/// Inputs controlling how projection data is prepared for one draw.
18#[derive(Clone, Copy, Debug, Default, Eq, PartialEq)]
19pub struct ProjectionDataParams {
20    /// Canonical tile rendered by this draw, or `None` for world-space draws.
21    pub tile: Option<TileCoords>,
22    /// Enables globe projection or its transition for this draw.
23    pub apply_globe_matrix: bool,
24}

26/// Projection data consumed by renderer shaders.
27#[derive(Clone, Copy, Debug, PartialEq)]
28pub struct RendererProjectionData {
29    /// Active main projection matrix.
30    pub main_matrix: Matrix4<f32>,
31    /// Mercator tile offset and per-tile-coordinate scale as `[x, y, scale_x, scale_y]`.
32    pub tile_mercator_coords: Vector4<f32>,
33    /// Unit-sphere horizon plane; the visible half-space has non-negative distance.
34    pub clipping_plane: Vector4<f32>,
35    /// Interpolation from Mercator zero to globe one.
36    pub projection_transition: f32,
37    /// Mercator projection matrix used during globe transitions.
38    pub fallback_matrix: Matrix4<f32>,
39    /// Whether zoom-zero line fragments must be clipped at the antimeridian.
40    pub clip_antimeridian: bool,
41}

43/// Camera geometry needed to derive the globe horizon plane.
44#[derive(Clone, Copy, Debug)]
45pub struct GlobeViewGeometry {
46    /// Geographic location at the center of the viewport.
47    pub center: LatLon,
48    /// Clockwise map bearing in degrees.
49    pub bearing_degrees: f64,
50    /// Camera pitch in degrees.
51    pub pitch_degrees: f64,
52    /// Camera distance to the map center in pixels.
53    pub camera_to_center_distance: f64,
54    /// Globe radius in pixels at the map center latitude.
55    pub globe_radius_pixels: f64,

58/// Failure while preparing renderer projection data.
59#[derive(Clone, Copy, Debug, Error, PartialEq)]
60pub enum ProjectionDataError {
61    /// Globe radius must be finite and greater than zero.
62    #[error("globe radius must be finite and positive, got {radius}")]
63    InvalidGlobeRadius {
64        /// Invalid radius in pixels.
65        radius: f64,
66    },
67    /// Camera-to-center distance must be finite and non-negative.
68    #[error("camera-to-center distance must be finite and non-negative, got {distance}")]
69    InvalidCameraDistance {
70        /// Invalid distance in pixels.
71        distance: f64,
72    },
73    /// An angle required for globe orientation must be finite.
74    #[error("{name} must be finite, got {degrees}")]
75    InvalidAngle {
76        /// Name of the invalid view angle.
77        name: &'static str,
78        /// Invalid value in degrees.
79        degrees: f64,
80    },
81}

83/// Combines projection matrices and per-draw parameters into renderer data.
84pub fn compose_projection_data(
85    matrices: ProjectionMatrices,
86    clipping_plane: Vector4<f32>,
87    globe_transition: f32,
88    params: ProjectionDataParams,
89) -> RendererProjectionData {
90    let transition = if globe_transition.is_finite() {
91        globe_transition.clamp(0.0, 1.0)
92    } else {
93        0.0
94    };
95    let use_globe_rendering = transition > 0.0;
96    RendererProjectionData {
97        main_matrix: if use_globe_rendering {
98            matrices.globe
99        } else {
100            matrices.mercator
101        },
102        tile_mercator_coords: tile_mercator_coordinates(params.tile),
103        clipping_plane,
104        projection_transition: if params.apply_globe_matrix {
105            transition
106        } else {
107            0.0
108        },
109        fallback_matrix: matrices.mercator,
110        clip_antimeridian: params.tile.is_some_and(|tile| u8::from(tile.z) == 0),
111    }
112}

114/// Returns Mercator offset and scale for tile-local coordinates in range `0..EXTENT`.
115pub fn tile_mercator_coordinates(tile: Option<TileCoords>) -> Vector4<f32> {
116    let Some(tile) = tile else {
117        let scale = (1.0 / EXTENT) as f32;
118        return Vector4::new(0.0, 0.0, scale, scale);
119    };
120    let tile_count = 2_f64.powi(i32::from(u8::from(tile.z)));
121    let extent_scale = 1.0 / (tile_count * EXTENT);
122    Vector4::new(
123        (f64::from(tile.x) / tile_count) as f32,
124        (f64::from(tile.y) / tile_count) as f32,
125        extent_scale as f32,
126        extent_scale as f32,
127    )
128}

130/// Computes the unit-sphere horizon plane used to reject occluded globe geometry.
131pub fn compute_globe_clipping_plane(
132    geometry: GlobeViewGeometry,
133) -> Result<Vector4<f64>, ProjectionDataError> {
134    validate_view_geometry(geometry)?;
135    let pitch = geometry.pitch_degrees.to_radians();
136    let camera_to_surface = geometry.camera_to_center_distance / geometry.globe_radius_pixels;
137    let camera_horizontal = pitch.sin() * camera_to_surface;
138    let camera_vertical = pitch.cos() * camera_to_surface + 1.0;
139    let camera_to_globe_center = camera_horizontal.hypot(camera_vertical);
140    let tangent_plane_distance = 1.0 / camera_to_globe_center;
141
142    let camera_direction = Vector3::new(-camera_horizontal, camera_vertical, 0.0).normalize();
143    let plane_normal = rotate_globe_plane_normal(
144        Vector3::new(0.0, camera_direction.x, camera_direction.y),
145        geometry.center,
146        geometry.bearing_degrees,
147    )
148    .normalize();
149    Ok(plane_normal.extend(-tangent_plane_distance))
150}

152fn validate_view_geometry(geometry: GlobeViewGeometry) -> Result<(), ProjectionDataError> {
153    if !geometry.globe_radius_pixels.is_finite() || geometry.globe_radius_pixels <= 0.0 {
154        return Err(ProjectionDataError::InvalidGlobeRadius {
155            radius: geometry.globe_radius_pixels,
156        });
157    }
158    if !geometry.camera_to_center_distance.is_finite() || geometry.camera_to_center_distance < 0.0 {
159        return Err(ProjectionDataError::InvalidCameraDistance {
160            distance: geometry.camera_to_center_distance,
161        });
162    }
163    for (name, degrees) in [
164        ("center latitude", geometry.center.latitude),
165        ("center longitude", geometry.center.longitude),
166        ("bearing", geometry.bearing_degrees),
167        ("pitch", geometry.pitch_degrees),
168    ] {
169        if !degrees.is_finite() {
170            return Err(ProjectionDataError::InvalidAngle { name, degrees });
171        }
172    }
173    Ok(())
174}

176fn rotate_globe_plane_normal(
177    normal: Vector3<f64>,
178    center: LatLon,
179    bearing_degrees: f64,
180) -> Vector3<f64> {
181    let bearing = bearing_degrees.to_radians();
182    let latitude = center.latitude.to_radians();
183    let longitude = center.longitude.to_radians();
184    let after_bearing = Vector3::new(
185        normal.x * bearing.cos() + normal.y * bearing.sin(),
186        -normal.x * bearing.sin() + normal.y * bearing.cos(),
187        normal.z,
188    );
189    let after_latitude = Vector3::new(
190        after_bearing.x,
191        after_bearing.y * latitude.cos() + after_bearing.z * latitude.sin(),
192        -after_bearing.y * latitude.sin() + after_bearing.z * latitude.cos(),
193    );
194    Vector3::new(
195        after_latitude.x * longitude.cos() + after_latitude.z * longitude.sin(),
196        after_latitude.y,
197        -after_latitude.x * longitude.sin() + after_latitude.z * longitude.cos(),
198    )
199}
