1//! Serde wire format for projection expressions.
2
3use serde::{de::Error as DeserializeError, Deserialize, Deserializer, Serialize, Serializer};
4use serde_json::Value;
5
6use super::{
7    InterpolationCurve, NamedProjection, ProjectionExpression, ProjectionExpressionError,
8    ProjectionExpressionKind, ProjectionStop,

10use crate::projection::ProjectionType;

12#[derive(Clone, Debug)]
13pub(crate) struct ProjectionWire(ProjectionType);
14
15impl From<ProjectionWire> for ProjectionType {
16    fn from(value: ProjectionWire) -> Self {
17        value.0
18    }
19}
20
21impl From<ProjectionType> for ProjectionWire {
22    fn from(value: ProjectionType) -> Self {
23        Self(value)
24    }
25}
26
27impl<'de> Deserialize<'de> for ProjectionWire {
28    fn deserialize<D>(deserializer: D) -> Result<Self, D::Error>
29    where
30        D: Deserializer<'de>,
31    {
32        let value = Value::deserialize(deserializer)?;
33        parse_projection(&value).map(Self).map_err(D::Error::custom)
34    }
35}
36
37impl Serialize for ProjectionWire {
38    fn serialize<S>(&self, serializer: S) -> Result<S::Ok, S::Error>
39    where
40        S: Serializer,
41    {
42        projection_value(&self.0).serialize(serializer)
43    }
44}

46fn parse_projection(value: &Value) -> Result<ProjectionType, ProjectionExpressionError> {
47    if let Some(name) = value.as_str() {
48        return parse_projection_type(name);
49    }
50    let values = value
51        .as_array()
52        .ok_or_else(|| unsupported("projection must be a name or expression array"))?;
53    if values.len() == 3 && values.first().and_then(Value::as_str) != Some("step") {
54        return parse_transition(values);
55    }
56    match values.first().and_then(Value::as_str) {
57        Some("interpolate") => parse_interpolate(values),
58        Some("step") => parse_step(values),
59        _ => Err(unsupported("unknown projection expression operator")),
60    }
61}
62
63fn parse_projection_type(name: &str) -> Result<ProjectionType, ProjectionExpressionError> {
64    match name {
65        "mercator" => Ok(ProjectionType::Mercator),
66        "globe" => Ok(ProjectionType::Globe),
67        "vertical-perspective" => Ok(ProjectionType::VerticalPerspective),
68        _ => Err(unsupported("unknown projection name")),
69    }
70}
71
72fn parse_named(value: &Value) -> Result<NamedProjection, ProjectionExpressionError> {
73    match value.as_str() {
74        Some("mercator") => Ok(NamedProjection::Mercator),
75        Some("vertical-perspective") => Ok(NamedProjection::VerticalPerspective),
76        _ => Err(unsupported("expression outputs must be named projections")),
77    }
78}
79
80fn parse_transition(values: &[Value]) -> Result<ProjectionType, ProjectionExpressionError> {
81    let from = values
82        .first()
83        .map(parse_named)
84        .transpose()?
85        .ok_or_else(|| unsupported("missing transition source"))?;
86    let to = values
87        .get(1)
88        .map(parse_named)
89        .transpose()?
90        .ok_or_else(|| unsupported("missing transition target"))?;
91    let transition = values
92        .get(2)
93        .and_then(Value::as_f64)
94        .ok_or_else(|| invalid_number("transition must be numeric"))?;
95    ProjectionType::transition(from, to, transition)
96}
97
98fn parse_interpolate(values: &[Value]) -> Result<ProjectionType, ProjectionExpressionError> {
99    if values.len() < 7 || values.len().is_multiple_of(2) || !is_zoom_input(values.get(2)) {
100        return Err(unsupported("invalid interpolate expression shape"));
101    }
102    let curve = parse_curve(
103        values
104            .get(1)
105            .ok_or_else(|| unsupported("missing interpolation curve"))?,
106    )?;
107    ProjectionType::interpolate(curve, parse_stops(&values[3..])?)
108}
109
110fn parse_step(values: &[Value]) -> Result<ProjectionType, ProjectionExpressionError> {
111    if values.len() < 5 || values.len().is_multiple_of(2) || !is_zoom_input(values.get(1)) {
112        return Err(unsupported("invalid step expression shape"));
113    }
114    let default = parse_named(&values[2])?;
115    ProjectionType::step(default, parse_stops(&values[3..])?)
116}
117
118fn parse_stops(values: &[Value]) -> Result<Vec<ProjectionStop>, ProjectionExpressionError> {
119    values
120        .as_chunks::<2>()
121        .0
122        .iter()
123        .map(|pair| {
124            let zoom = pair[0]
125                .as_f64()
126                .ok_or_else(|| invalid_number("zoom stop must be numeric"))?;
127            Ok(ProjectionStop::new(zoom, parse_named(&pair[1])?))
128        })
129        .collect()
130}
131
132fn parse_curve(value: &Value) -> Result<InterpolationCurve, ProjectionExpressionError> {
133    let parts = value
134        .as_array()
135        .ok_or_else(|| unsupported("interpolation curve must be an array"))?;
136    match parts.first().and_then(Value::as_str) {
137        Some("linear") if parts.len() == 1 => Ok(InterpolationCurve::Linear),
138        Some("exponential") if parts.len() == 2 => parts[1]
139            .as_f64()
140            .map(InterpolationCurve::Exponential)
141            .ok_or_else(|| invalid_number("exponential base must be numeric")),
142        Some("cubic-bezier") if parts.len() == 5 => Ok(InterpolationCurve::CubicBezier(
143            curve_number(parts, 1)?,
144            curve_number(parts, 2)?,
145            curve_number(parts, 3)?,
146            curve_number(parts, 4)?,
147        )),
148        _ => Err(unsupported("unsupported interpolation curve")),
149    }
150}
151
152fn curve_number(parts: &[Value], index: usize) -> Result<f64, ProjectionExpressionError> {
153    parts
154        .get(index)
155        .and_then(Value::as_f64)
156        .ok_or_else(|| invalid_number("cubic-bezier controls must be numeric"))
157}
158
159fn is_zoom_input(value: Option<&Value>) -> bool {
160    value.and_then(Value::as_array).is_some_and(|parts| {
161        parts.len() == 1 && parts.first().and_then(Value::as_str) == Some("zoom")
162    })
163}
164
165fn invalid_number(reason: &str) -> ProjectionExpressionError {
166    ProjectionExpressionError::InvalidNumber {
167        reason: reason.to_string(),
168    }
169}
170
171fn unsupported(reason: &str) -> ProjectionExpressionError {
172    ProjectionExpressionError::Unsupported {
173        reason: reason.to_string(),
174    }
175}

177fn projection_value(projection: &ProjectionType) -> Value {
178    match projection {
179        ProjectionType::Mercator => Value::String("mercator".to_string()),
180        ProjectionType::Globe => Value::String("globe".to_string()),
181        ProjectionType::VerticalPerspective => Value::String("vertical-perspective".to_string()),
182        ProjectionType::Expression(expression) => expression_value(expression),
183    }
184}
185
186fn expression_value(expression: &ProjectionExpression) -> Value {
187    match &expression.kind {
188        ProjectionExpressionKind::Transition {
189            from,
190            to,
191            transition,
192        } => Value::Array(vec![
193            named_value(*from),
194            named_value(*to),
195            Value::from(*transition),
196        ]),
197        ProjectionExpressionKind::Interpolate { curve, stops } => {
198            let mut values = vec![
199                Value::String("interpolate".to_string()),
200                curve_value(*curve),
201                zoom_value(),
202            ];
203            append_stops(&mut values, stops);
204            Value::Array(values)
205        }
206        ProjectionExpressionKind::Step { default, stops } => {
207            let mut values = vec![
208                Value::String("step".to_string()),
209                zoom_value(),
210                named_value(*default),
211            ];
212            append_stops(&mut values, stops);
213            Value::Array(values)
214        }
215    }
216}
217
218fn append_stops(values: &mut Vec<Value>, stops: &[ProjectionStop]) {
219    for stop in stops {
220        values.push(Value::from(stop.zoom));
221        values.push(named_value(stop.projection));
222    }
223}
224
225fn named_value(projection: NamedProjection) -> Value {
226    Value::String(
227        match projection {
228            NamedProjection::Mercator => "mercator",
229            NamedProjection::VerticalPerspective => "vertical-perspective",
230        }
231        .to_string(),
232    )
233}
234
235fn zoom_value() -> Value {
236    Value::Array(vec![Value::String("zoom".to_string())])
237}
238
239fn curve_value(curve: InterpolationCurve) -> Value {
240    let values = match curve {
241        InterpolationCurve::Linear => vec![Value::String("linear".to_string())],
242        InterpolationCurve::Exponential(base) => {
243            vec![Value::String("exponential".to_string()), Value::from(base)]
244        }
245        InterpolationCurve::CubicBezier(x1, y1, x2, y2) => vec![
246            Value::String("cubic-bezier".to_string()),
247            Value::from(x1),
248            Value::from(y1),
249            Value::from(x2),
250            Value::from(y2),
251        ],
252    };
253    Value::Array(values)
254}
