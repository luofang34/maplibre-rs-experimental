1//! Rays from viewport pixels and where they meet the globe.
2
3use cgmath::{InnerSpace, Point2, Vector3, Vector4};
4
5use super::{GlobeCameraState, MIN_DIRECTION_LENGTH_SQUARED};
6use crate::{
7    coords::LatLon,
8    projection::globe::{
9        closest_point_on_sphere, horizon_plane_to_circle, ray_sphere_intersection,
10        unit_sphere_to_lat_lon,
11    },
12};
13
14const HORIZON_FALLBACK_RAY_LENGTH: f64 = 2.0;
15
16impl GlobeCameraState {
17    /// The viewport pixel of a point in normalized device coordinates.
18    pub fn ndc_to_pixel(&self, ndc: Point2<f64>) -> Point2<f64> {
19        Point2::new(
20            (ndc.x * 0.5 + 0.5) * self.options.width,
21            (-ndc.y * 0.5 + 0.5) * self.options.height,
22        )
23    }
24
25    /// Returns a normalized world-space ray from the camera through a viewport pixel.
26    pub fn ray_direction_from_pixel(&self, pixel: Point2<f64>) -> Option<Vector3<f64>> {
27        let clip = Vector4::new(
28            pixel.x / self.options.width * 2.0 - 1.0,
29            -(pixel.y / self.options.height * 2.0 - 1.0),
30            1.0,
31            1.0,
32        );
33        let world = self.inverse_view_projection * clip;
34        if world.w.abs() <= f64::EPSILON {
35            return None;
36        }
37        let point = world.truncate() / world.w;
38        let direction = point - self.camera_position;
39        (direction.magnitude2() > MIN_DIRECTION_LENGTH_SQUARED).then(|| direction.normalize())
40    }
41
42    /// Converts a viewport pixel to a surface location, clamping misses to the visible horizon.
43    pub fn screen_point_to_location(&self, pixel: Point2<f64>) -> Option<LatLon> {
44        let direction = self.ray_direction_from_pixel(pixel)?;
45        if let Some(intersection) = ray_sphere_intersection(self.camera_position, direction, 1.0) {
46            let point = self.camera_position + direction * intersection.t_min;
47            if point.magnitude2() > MIN_DIRECTION_LENGTH_SQUARED {
48                return Some(unit_sphere_to_lat_lon(point.normalize()));
49            }
50        }
51        self.closest_horizon_location(direction)
52    }
53
54    /// Where the ray through a viewport pixel first meets the sphere `elevation_meters` above
55    /// the mean radius, as a point raised that high would sit under the pixel; `None` where
56    /// the ray passes the sphere.
57    pub fn screen_point_to_location_at(
58        &self,
59        pixel: Point2<f64>,
60        elevation_meters: f64,
61    ) -> Option<LatLon> {
62        let direction = self.ray_direction_from_pixel(pixel)?;
63        let radius = self.options.body.unit_radius_at(elevation_meters);
64        let hit = ray_sphere_intersection(self.camera_position, direction, radius)?;
65        let t = if hit.t_min > 0.0 {
66            hit.t_min
67        } else {
68            hit.t_max
69        };
70        (t > 0.0)
71            .then(|| unit_sphere_to_lat_lon((self.camera_position + direction * t).normalize()))
72    }
73
74    /// Returns whether the ray through a viewport pixel intersects the unit globe.
75    pub fn is_point_on_map_surface(&self, pixel: Point2<f64>) -> bool {
76        self.ray_direction_from_pixel(pixel)
77            .is_some_and(|direction| {
78                ray_sphere_intersection(self.camera_position, direction, 1.0).is_some()
79            })
80    }
81
82    fn closest_horizon_location(&self, direction: Vector3<f64>) -> Option<LatLon> {
83        let normal = self.clipping_plane.truncate();
84        let denominator = normal.dot(direction);
85        let origin_distance = normal.dot(self.camera_position) + self.clipping_plane.w;
86        let distance = if denominator.abs() > f64::EPSILON {
87            -origin_distance / denominator
88        } else {
89            -1.0
90        };
91        let plane_point = if distance.is_finite() && distance > 0.0 {
92            self.camera_position + direction * distance
93        } else {
94            let distant = self.camera_position + direction * HORIZON_FALLBACK_RAY_LENGTH;
95            let plane_distance = normal.dot(distant) + self.clipping_plane.w;
96            distant - normal * plane_distance
97        };
98        let horizon = horizon_plane_to_circle(self.clipping_plane);
99        closest_point_on_sphere(horizon.center, horizon.radius, plane_point)
100            .or_else(|| (horizon.radius == 0.0).then_some(horizon.center))
101            .map(unit_sphere_to_lat_lon)
102    }
103}
