1//! A globe camera driven by an eye the host tracks, rather than by center, zoom and angles.
2//!
3//! The eye arrives in unit-sphere coordinates with its own frustum. The matrices use the same
4//! pixel units as a camera built from map angles, so the tile covering, symbol placement and
5//! depth read them the way they read any globe camera.
6
7use cgmath::{InnerSpace, Matrix, Matrix3, Matrix4, SquareMatrix, Vector3, Vector4};
8
9use super::{validate_options, GlobeCameraError, GlobeCameraOptions, GlobeCameraState};
10use crate::render::camera::EyeFrustum;
11
12/// An eye placed in unit-sphere coordinates.
13#[derive(Clone, Copy, Debug)]
14pub struct ExternalGlobeEye {
15    /// Where the eye is; a length of one sits on the surface.
16    pub position: Vector3<f64>,
17    /// Axes of the eye's space in unit-sphere coordinates, columns x right, y up and z
18    /// backwards, as OpenGL orders them.
19    pub axes: Matrix3<f64>,
20    /// The eye's frustum with clip distances in screen pixels.
21    pub frustum: EyeFrustum,
22    /// Distance from the eye to the map center in screen pixels, which the zoom-dependent
23    /// readers of the camera take as its distance to the surface.
24    pub camera_to_center_distance: f64,
25}
26
27impl ExternalGlobeEye {
28    /// The eye a unit-sphere-to-eye view matrix places; `None` when the matrix is singular.
29    pub fn from_view(
30        view: Matrix4<f64>,
31        frustum: EyeFrustum,
32        camera_to_center_distance: f64,
33    ) -> Option<Self> {
34        let inverse = view.invert()?;
35        let position = inverse * Vector4::new(0.0, 0.0, 0.0, 1.0);
36        if position.w.abs() <= f64::EPSILON {
37            return None;
38        }
39        let axis =
40            |direction: Vector3<f64>| (inverse * direction.extend(0.0)).truncate().normalize();
41        Some(Self {
42            position: position.truncate() / position.w,
43            axes: Matrix3::from_cols(
44                axis(Vector3::unit_x()),
45                axis(Vector3::unit_y()),
46                axis(Vector3::unit_z()),
47            ),
48            frustum,
49            camera_to_center_distance,
50        })
51    }
52}
53
54impl GlobeCameraState {
55    /// Builds the camera an external eye sees the globe with.
56    ///
57    /// The options give the map center, zoom and angles the eye's pose derives to, which
58    /// scale the globe to pixels and orient the readers that sort or cull by map angles; the
59    /// eye itself sets the matrices.
60    pub fn from_external_eye(
61        options: GlobeCameraOptions,
62        eye: ExternalGlobeEye,
63    ) -> Result<Self, GlobeCameraError> {
64        validate_options(options)?;
65        // The eye is where the host placed it; it orbits nothing, so no elevation raises the
66        // point the camera reports as its target.
67        let options = GlobeCameraOptions {
68            target_elevation_meters: 0.0,
69            ..options
70        };
71        let clipping_plane = super::horizon_plane(eye.position, 1.0)?;
72        let radius = super::radius_pixels(options);
73        // The eye's axes are expressed in sphere space, so their transpose takes sphere
74        // space into the eye's; the sphere is scaled to pixels first.
75        let view = Matrix4::from(eye.axes.transpose())
76            * Matrix4::from_translation(-eye.position * radius)
77            * Matrix4::from_scale(radius);
78        let far_z = if eye.frustum.far.is_finite() {
79            eye.frustum.far
80        } else {
81            eye.camera_to_center_distance + radius * 2.0
82        };
83        let frustum = EyeFrustum {
84            far: far_z,
85            ..eye.frustum
86        };
87        let projection = frustum.projection();
88        let inverse_projection = projection
89            .invert()
90            .ok_or(GlobeCameraError::NonInvertibleViewProjection)?;
91        let view_projection = projection * view;
92        let inverse_view_projection = view_projection
93            .invert()
94            .ok_or(GlobeCameraError::NonInvertibleViewProjection)?;
95        Ok(Self {
96            options,
97            projection,
98            inverse_projection,
99            view,
100            view_projection,
101            inverse_view_projection,
102            camera_position: eye.position,
103            clipping_plane,
104            globe_radius_pixels: radius,
105            camera_to_center_distance: eye.camera_to_center_distance,
106            near_z: frustum.near,
107            far_z,
108            external_eye: true,
109        })
110    }
111}
