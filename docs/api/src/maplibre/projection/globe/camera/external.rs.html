1//! A globe camera driven by an eye the host tracks, rather than by center, zoom and angles.
2//!
3//! The eye arrives in unit-sphere coordinates with its own frustum. The matrices use the same
4//! pixel units as a camera built from map angles, so the tile covering, symbol placement and
5//! depth read them the way they read any globe camera.
6
7use cgmath::{InnerSpace, Matrix, Matrix3, Matrix4, SquareMatrix, Vector3, Vector4};
8
9use super::{validate_options, GlobeCameraError, GlobeCameraOptions, GlobeCameraState};
10use crate::render::camera::EyeFrustum;
11
12/// An eye placed in unit-sphere coordinates.
13#[derive(Clone, Copy, Debug)]
14pub struct ExternalGlobeEye {
15    /// Where the eye is; a length of one sits on the surface.
16    pub position: Vector3<f64>,
17    /// Axes of the eye's space in unit-sphere coordinates, columns x right, y up and z
18    /// backwards, as OpenGL orders them.
19    pub axes: Matrix3<f64>,
20    /// The eye's frustum with clip distances in screen pixels.
21    pub frustum: EyeFrustum,
22    /// Distance from the eye to the map center in screen pixels, which the zoom-dependent
23    /// readers of the camera take as its distance to the surface.
24    pub camera_to_center_distance: f64,
25}
26
27impl ExternalGlobeEye {
28    /// The eye a unit-sphere-to-eye view matrix places; `None` when the matrix is singular.
29    pub fn from_view(
30        view: Matrix4<f64>,
31        frustum: EyeFrustum,
32        camera_to_center_distance: f64,
33    ) -> Option<Self> {
34        let inverse = view.invert()?;
35        let position = inverse * Vector4::new(0.0, 0.0, 0.0, 1.0);
36        if position.w.abs() <= f64::EPSILON {
37            return None;
38        }
39        let axis =
40            |direction: Vector3<f64>| (inverse * direction.extend(0.0)).truncate().normalize();
41        Some(Self {
42            position: position.truncate() / position.w,
43            axes: Matrix3::from_cols(
44                axis(Vector3::unit_x()),
45                axis(Vector3::unit_y()),
46                axis(Vector3::unit_z()),
47            ),
48            frustum,
49            camera_to_center_distance,
50        })
51    }
52}

54impl GlobeCameraState {
55    /// Builds the camera an external eye sees the globe with.
56    ///
57    /// The options give the map center, zoom and angles the eye's pose derives to, which
58    /// scale the globe to pixels and orient the readers that sort or cull by map angles; the
59    /// eye itself sets the matrices.
60    pub fn from_external_eye(
61        options: GlobeCameraOptions,
62        eye: ExternalGlobeEye,
63    ) -> Result<Self, GlobeCameraError> {
64        validate_options(options)?;
65        // The eye is where the host placed it; it orbits nothing, so no elevation raises the
66        // point the camera reports as its target.
67        let options = GlobeCameraOptions {
68            target_elevation_meters: 0.0,
69            ..options
70        };
71        let lowest_radius = options
72            .body
73            .unit_radius_at(options.body.lowest_ground_meters)
74            .min(1.0);
75        let distance = eye.position.magnitude();
76        let clearance = 0.25 / options.body.radius_meters;
77        let horizon_radius = if distance > lowest_radius {
78            (distance - clearance).clamp(lowest_radius, 1.0)
79        } else {
80            lowest_radius
81        };
82        let clipping_plane = super::horizon_plane(eye.position, horizon_radius)?;
83        let radius = super::radius_pixels(options);
84        // The eye's axes are expressed in sphere space, so their transpose takes sphere
85        // space into the eye's; the sphere is scaled to pixels first.
86        let view = Matrix4::from(eye.axes.transpose())
87            * Matrix4::from_translation(-eye.position * radius)
88            * Matrix4::from_scale(radius);
89        let far_z = if eye.frustum.far.is_finite() {
90            eye.frustum.far
91        } else {
92            eye.camera_to_center_distance + radius * 2.0
93        };
94        let frustum = EyeFrustum {
95            far: far_z,
96            ..eye.frustum
97        };
98        let projection = frustum.projection();
99        let inverse_projection = projection
100            .invert()
101            .ok_or(GlobeCameraError::NonInvertibleViewProjection)?;
102        let view_projection = projection * view;
103        let inverse_view_projection = view_projection
104            .invert()
105            .ok_or(GlobeCameraError::NonInvertibleViewProjection)?;
106        Ok(Self {
107            options,
108            projection,
109            inverse_projection,
110            view,
111            view_projection,
112            inverse_view_projection,
113            camera_position: eye.position,
114            clipping_plane,
115            globe_radius_pixels: radius,
116            camera_to_center_distance: eye.camera_to_center_distance,
117            near_z: frustum.near,
118            far_z,
119            external_eye: true,
120        })
121    }
122}

124const GAZE_REACH_HEIGHTS: f64 = 2.0;
125
126pub(super) fn focus_unit_sphere(
127    eye: Vector3<f64>,
128    view: &Matrix4<f64>,
129    external_eye: bool,
130) -> Vector3<f64> {
131    let forward_raw = -Vector3::new(view.x.z, view.y.z, view.z.z);
132    let len2 = forward_raw.magnitude2();
133    if len2 <= super::MIN_DIRECTION_LENGTH_SQUARED || eye.magnitude2() <= 1.0 {
134        return if eye.magnitude2() > super::MIN_DIRECTION_LENGTH_SQUARED {
135            eye.normalize()
136        } else {
137            Vector3::unit_z()
138        };
139    }
140    let forward = forward_raw / len2.sqrt();
141    if external_eye {
142        return external_focus_unit_sphere(eye, forward);
143    }
144    let b = eye.dot(forward);
145    let disc = b * b - (eye.magnitude2() - 1.0);
146    let t = if b < 0.0 && disc >= 0.0 {
147        -b - disc.sqrt()
148    } else {
149        (-b).max(0.0)
150    };
151    (eye + forward * t).normalize()
152}

154fn external_focus_unit_sphere(origin: Vector3<f64>, direction: Vector3<f64>) -> Vector3<f64> {
155    let distance = origin.magnitude();
156    if distance <= 1.0 {
157        return if distance > super::MIN_DIRECTION_LENGTH_SQUARED.sqrt() {
158            origin / distance
159        } else {
160            Vector3::unit_z()
161        };
162    }
163    let beneath = origin / distance;
164    let along = origin.dot(direction);
165    let discriminant = along * along - (distance * distance - 1.0);
166    let reach = (GAZE_REACH_HEIGHTS * (distance - 1.0) / distance).atan();
167    if discriminant >= 0.0 {
168        let t = -along - discriminant.sqrt();
169        if t >= 0.0 {
170            let center = (origin + direction * t).normalize();
171            let cosine = beneath.dot(center).clamp(-1.0, 1.0);
172            if cosine.acos() <= reach {
173                return center;
174            }
175            let tangent = center - beneath * cosine;
176            if tangent.magnitude2() <= super::MIN_DIRECTION_LENGTH_SQUARED {
177                return beneath;
178            }
179            return beneath * reach.cos() + tangent.normalize() * reach.sin();
180        }
181    }
182    let up_dot = beneath.dot(direction);
183    let tangent = direction - beneath * up_dot;
184    let tangent_len = tangent.magnitude();
185    if tangent_len <= super::MIN_DIRECTION_LENGTH_SQUARED.sqrt() {
186        return beneath;
187    }
188    let limb_angle = (1.0 / distance).clamp(-1.0, 1.0).acos();
189    let base_angle = reach.min(limb_angle);
190    let angle = if up_dot > 0.0 {
191        base_angle * tangent_len.clamp(0.0, 1.0)
192    } else {
193        base_angle
194    };
195    beneath * angle.cos() + (tangent / tangent_len) * angle.sin()
196}
