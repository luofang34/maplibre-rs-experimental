9const MAX_MERCATOR_HORIZON_DEGREES: f64 = 89.25;
10/// Latitude beyond which Mercator has no coordinates.
11const MAX_MERCATOR_LATITUDE_DEGREES: f64 = 85.051_129;
12const MAX_ZOOM_LEVELS_ON_SCREEN: f64 = 9.314;
13const TILE_COUNT_MAX_MIN_RATIO: f64 = 3.0;
14const INTEGRATION_POINTS: usize = 10;
15/// The vertical field of a map camera, `2 * atan(0.75)`, as in GL JS.
16const NORMAL_FIELD_OF_VIEW_DEGREES: f64 = 36.869_897_645_844_02;

19pub(crate) struct LodContext {
20    camera: Point2<f64>,
21    eye_focal_pixels: Option<f64>,
22    distance_z: f64,
23    distance_to_center_3d: f64,
24    requested_zoom: f64,
25    field_of_view_degrees: f64,
26    /// Whether a tile's distance is the straight distance to its rectangle, as on the Mercator
27    /// plane, instead of the farther of the two axis distances GL JS uses on the globe.
28    planar: bool,
29}

31impl LodContext {
32    pub(crate) fn new(camera: &GlobeCameraState, requested_zoom: f64) -> Self {
33        if camera.is_external_eye() {
34            // The center, pitch and distance an eye derives to describe where it looks, not
35            // where it is: a level gaze derives to a pitch near ninety degrees, which would
36            // put the camera a few percent of its height above the ground and have every
37            // tile counted as seen at grazing incidence, levels coarser than the eye sees.
38            let position = camera.camera_position();
39            let beneath = unit_sphere_to_lat_lon(position);
40            let height_radii = (position.magnitude() - 1.0).max(0.0);
41            return Self::from_positions(
42                mercator_point(beneath),
43                mercator_center(camera),
44                height_radii * radius_world_units(beneath.latitude),
45                camera.field_of_view_degrees(),
46                requested_zoom,
47            );
48        }
49        Self::from_view(
50            mercator_center(camera),
51            camera.camera_to_center_distance() / camera.world_size(),
52            camera.pitch_degrees(),
53            camera.bearing_degrees(),
54            camera.field_of_view_degrees(),
55            requested_zoom,
56        )
57    }
58
59    /// Builds the context from the map center in Mercator `0..1` units, the camera distance to
60    /// it in the same units, and the camera angles in GL JS's conventions.
61    pub(crate) fn from_view(
62        center: Point2<f64>,
63        distance: f64,
64        pitch_degrees: f64,
65        bearing_degrees: f64,
66        field_of_view_degrees: f64,
67        requested_zoom: f64,
68    ) -> Self {
69        let pitch = pitch_degrees.to_radians();
70        let bearing = bearing_degrees.to_radians();
71        let horizontal = pitch.sin();
72        let direction_x = horizontal * bearing.sin();
73        let direction_y = -horizontal * bearing.cos();
74        let camera_point = Point2::new(
75            center.x - distance * direction_x,
76            center.y - distance * direction_y,
77        );
78        Self::from_positions(
79            camera_point,
80            center,
81            distance * pitch.cos(),
82            field_of_view_degrees,
83            requested_zoom,
84        )
85    }
86
87    /// Builds the context from the camera's and the center's positions in Mercator `0..1`
88    /// units and the camera's height above the center in the same units.
89    pub(crate) fn from_positions(
90        camera: Point2<f64>,
91        center: Point2<f64>,
92        distance_z: f64,
93        field_of_view_degrees: f64,
94        requested_zoom: f64,
95    ) -> Self {
96        let distance_to_center_2d = (center.x - camera.x).hypot(center.y - camera.y);
97        Self {
98            camera,
99            eye_focal_pixels: None,
100            distance_z,
101            distance_to_center_3d: distance_to_center_2d.hypot(distance_z),
102            requested_zoom,
103            field_of_view_degrees,
104            planar: false,
105        }
106    }

112    }

140    pub(crate) fn stable_zoom_for_tile(
141        &self,
142        tile: TileCoords,
143        rounding: ZoomRounding,
144        history: Option<&crate::projection::lod_history::LodHistory>,
145    ) -> ZoomLevel {
146        let desired = rounding
147            .apply(self.desired_zoom(tile) + history.map_or(0.0, |h| h.bias(tile)))
148            .clamp(0.0, (MAX_ZOOM - 1) as f64);
149        ZoomLevel::new(desired as u8)
150    }
151
152    /// The fractional level `tile` should be drawn at, before rounding.
153    fn desired_zoom(&self, tile: TileCoords) -> f64 {
154        let distance_2d = if self.planar {
155            distance_to_tile_planar(self.camera, tile)
156        } else {
157            distance_to_tile_2d(self.camera, tile)
158        };
159        if let Some(focal) = self.eye_focal_pixels {
160            let distance = distance_2d.hypot(self.distance_z).max(f64::EPSILON);
161            // Bound the longest projected texel axis. A grazing-angle area estimate
162            // would erase cross-slope detail on terrain and blur roads toward the horizon.
163            (focal / (crate::coords::TILE_SIZE * distance)).log2()
164        } else {
165            calculate_tile_zoom(
166                self.requested_zoom,
167                distance_2d,
168                self.distance_z,
169                self.distance_to_center_3d,
170                self.field_of_view_degrees,
171            )
172        }
173    }
174}

176/// The straight distance from a point to a tile's rectangle in Mercator `0..1` units.
177fn distance_to_tile_planar(point: Point2<f64>, tile: TileCoords) -> f64 {
178    let scale = 2_f64.powi(i32::from(u8::from(tile.z)));
179    let size = 1.0 / scale;
180    let axis = |point: f64, corner: f64| {
181        let delta = point - corner;
182        if delta < 0.0 {
183            -delta
184        } else {
185            (delta - size).max(0.0)
186        }
187    };
188    axis(point.x, f64::from(tile.x) / scale).hypot(axis(point.y, f64::from(tile.y) / scale))
189}

195/// A location in Mercator `0..1` units.
196fn mercator_point(location: LatLon) -> Point2<f64> {
197    let x = location.longitude / 360.0 + 0.5;
198    let latitude = location
199        .latitude
200        .clamp(
201            -MAX_MERCATOR_LATITUDE_DEGREES,
202            MAX_MERCATOR_LATITUDE_DEGREES,
203        )
204        .to_radians();
205    let y = (1.0 - latitude.tan().asinh() / std::f64::consts::PI) * 0.5;
206    Point2::new(x, y)
207}

209/// The globe's radius in Mercator `0..1` units at a latitude, where a world unit spans the
210/// circumference scaled by the Mercator stretch there.
211fn radius_world_units(latitude_degrees: f64) -> f64 {
212    let latitude = latitude_degrees
213        .clamp(
214            -MAX_MERCATOR_LATITUDE_DEGREES,
215            MAX_MERCATOR_LATITUDE_DEGREES,
216        )
217        .to_radians();
218    1.0 / (2.0 * std::f64::consts::PI * latitude.cos())
219}

221fn calculate_tile_zoom(
222    requested_center_zoom: f64,
223    distance_to_tile_2d: f64,
224    distance_to_tile_z: f64,
225    distance_to_center_3d: f64,
226    field_of_view_degrees: f64,
227) -> f64 {
228    // GL JS's factor grows without bound as the field narrows, so a sensor's view of a few
229    // degrees, tilted, would load tiles levels coarser than its screen resolves. The factor
230    // holds back the tile count of a view spanning much of the ground toward the horizon; a
231    // field narrower than the normal one sees less ground and needs no more restraint.
232    let pitch_behavior =
233        pitch_tile_loading_behavior(field_of_view_degrees.max(NORMAL_FIELD_OF_VIEW_DEGREES));
234    let center_pitch = (distance_to_tile_z / distance_to_center_3d).acos();
235    let half_fov = (field_of_view_degrees * 0.5).to_radians();
236    let tile_count_pitch_zero = 2.0 * integral_cos_power(pitch_behavior - 1.0, 0.0, half_fov);
237    let highest_pitch = (center_pitch + half_fov).min(MAX_MERCATOR_HORIZON_DEGREES.to_radians());
238    let lowest_pitch = (center_pitch - half_fov).min(highest_pitch);
239    let tile_count = integral_cos_power(pitch_behavior - 1.0, lowest_pitch, highest_pitch);
240    let tile_pitch = (distance_to_tile_2d / distance_to_tile_z).atan();
241    let distance_to_tile_3d = distance_to_tile_2d.hypot(distance_to_tile_z);
242    let distance_scale = distance_to_center_3d / distance_to_tile_3d / 0.5_f64.max(half_fov.cos());
243    requested_center_zoom + distance_scale.log2() + pitch_behavior * tile_pitch.cos().log2() * 0.5
244        - (tile_count / tile_count_pitch_zero / TILE_COUNT_MAX_MIN_RATIO)
245            .max(1.0)
246            .log2()
247            * 0.5
248}
249
250fn pitch_tile_loading_behavior(field_of_view_degrees: f64) -> f64 {
251    let numerator = (MAX_MERCATOR_HORIZON_DEGREES - field_of_view_degrees)
252        .to_radians()
253        .cos();
254    let denominator = MAX_MERCATOR_HORIZON_DEGREES.to_radians().cos();
255    2.0 * ((MAX_ZOOM_LEVELS_ON_SCREEN - 1.0) / (numerator / denominator).log2() - 1.0)
256}
257
258fn integral_cos_power(power: f64, start: f64, end: f64) -> f64 {
259    let width = (end - start) / INTEGRATION_POINTS as f64;
260    (0..INTEGRATION_POINTS)
261        .map(|index| {
262            let x = start + (index as f64 + 0.5) * width;
263            width * x.cos().powf(power)
264        })
265        .sum()
266}
