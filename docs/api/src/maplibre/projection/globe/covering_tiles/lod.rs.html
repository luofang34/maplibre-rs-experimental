17const MAX_MERCATOR_HORIZON_DEGREES: f64 = 89.25;
18/// Latitude beyond which Mercator has no coordinates.
19const MAX_MERCATOR_LATITUDE_DEGREES: f64 = 85.051_129;

40impl LodContext {
41    pub(crate) fn new(camera: &GlobeCameraState, requested_zoom: f64) -> Self {
42        // Measure distance and foreshortening directly on S^2 from the camera's 3D unit-globe
43        // position and SO(3) focus direction so orbital, FreeGlobe, and external-eye cameras
44        // are all singularity-free across ±90° poles and ±180° antimeridian without Mercator dx.
45        let position = camera.camera_position();
46        let len = position.magnitude();
47        let eye_hat = if len > f64::EPSILON {
48            position / len
49        } else {
50            Vector3::unit_z()
51        };
52        let target_radius = if camera.is_external_eye() {
53            1.0
54        } else {
55            camera.target().magnitude()
56        };
57        let min_clearance = 1.0 / camera.body().radius_meters;
58        let height_radii = (len - target_radius).max(min_clearance);
59        let focus_chord = (eye_hat - camera.focus_unit_sphere()).magnitude();
60        Self {
61            camera: Point2::new(0.5, 0.5),
62            eye_sphere: Some((eye_hat, height_radii)),
63            eye_focal_pixels: None,
64            distance_z: height_radii,
65            distance_to_center_3d: focus_chord.hypot(height_radii),
66            requested_zoom,
67            field_of_view_degrees: camera.field_of_view_degrees(),
68            planar: false,
69        }
70    }
71
72    /// The level whose texels match the screen's pixels where the view meets each tile, seen
73    /// from where `camera` actually stands, for a viewport `viewport_height` pixels tall.
74    /// `level_offset` shifts every level, as a source's tile size shifts its request.
75    ///
76    /// A free-globe pose takes this instead of GL JS's pitch rule: a gaze at the horizon
77    /// derives a pitch near ninety degrees, at which that rule drops every distant tile
78    /// several levels for foreshortening, though a horizon's silhouette runs across the
79    /// line of sight and needs the full detail its pixels resolve.
80    pub(crate) fn screen_resolution(
81        camera: &GlobeCameraState,
82        viewport_height: f64,
83        level_offset: f64,
84    ) -> Self {
85        let focal =
86            0.5 * viewport_height / (0.5 * camera.field_of_view_degrees()).to_radians().tan();
87        Self::screen_resolution_focal(camera, focal, level_offset, 0.0)
88    }
89
90    /// Screen-pixel texel resolution using an explicit focal length in pixels (such as an
91    /// [`ExternalView`](crate::render::view_state::ExternalView)'s `lod_focal_pixels`) and local
92    /// ground elevation beneath the eye.
93    pub(crate) fn screen_resolution_focal(
94        camera: &GlobeCameraState,
95        focal_pixels: f64,
96        level_offset: f64,
97        ground_elevation_meters: f64,
98    ) -> Self {
99        let position = camera.camera_position();
100        let len = position.magnitude();
101        let eye_hat = if len > f64::EPSILON {
102            position / len
103        } else {
104            Vector3::unit_z()
105        };
106        let body = camera.body();
107        let ground_radius = body.unit_radius_at(ground_elevation_meters);
108        let min_clearance = 1.0 / body.radius_meters;
109        let height_radii = (len - ground_radius).max(min_clearance);
110        let parallel_cos = parallel_cos_from_sin_lat(eye_hat.y);
111        let height = height_radii / (std::f64::consts::TAU * parallel_cos);
112        Self {
113            camera: Point2::new(0.5, 0.5),
114            eye_sphere: Some((eye_hat, height_radii)),
115            distance_z: height,
116            eye_focal_pixels: Some(focal_pixels * 2_f64.powf(level_offset)),
117            distance_to_center_3d: height,
118            requested_zoom: 0.0,
119            field_of_view_degrees: 0.0,
120            planar: false,
121        }
122    }

124    /// Builds the context from the camera's and the center's positions in Mercator `0..1`

126    pub(crate) fn from_positions(

138            distance_z,
139            distance_to_center_3d: distance_to_center_2d.hypot(distance_z),
140            requested_zoom,
141            field_of_view_degrees,
142            planar: false,
143        }
144    }

179    pub(crate) fn stable_zoom_for_tile(
180        &self,
181        tile: TileCoords,
182        rounding: ZoomRounding,
183        history: Option<&crate::projection::lod_history::LodHistory>,
184    ) -> ZoomLevel {
185        let desired = rounding
186            .apply(self.desired_zoom(tile) + history.map_or(0.0, |h| h.bias(tile)))
187            .clamp(0.0, (MAX_ZOOM - 1) as f64);
188        ZoomLevel::new(desired as u8)
189    }
190
191    /// The fractional level `tile` should be drawn at, before rounding.
192    fn desired_zoom(&self, tile: TileCoords) -> f64 {
193        if !self.planar {
194            if let Some((eye_hat, height_radii)) = self.eye_sphere {
195                if let Some(focal) = self.eye_focal_pixels {
196                    let (q_hat, _) = closest_unit_sphere_point_on_tile(eye_hat, tile);
197                    let chord_vec = eye_hat - q_hat;
198                    let d_radii_3d = (chord_vec + eye_hat * height_radii).magnitude();
199                    let parallel_cos = if u8::from(tile.z) == 0 {
200                        1.0
201                    } else {
202                        parallel_cos_from_sin_lat(q_hat.y)
203                    };
204                    let distance =
205                        (d_radii_3d / (std::f64::consts::TAU * parallel_cos)).max(f64::EPSILON);
206                    return (focal / (crate::coords::TILE_SIZE * distance)).log2();
207                }
208                // Source tiles carry no raster/DEM/MVT texels or vertices poleward of ±85.051129°
209                // (the polar fan to ±90° is synthetic), so measure S^2 distance to the tile's
210                // actual Web Mercator data domain rather than demanding 2^z empty polar wedges.
211                let (_, chebyshev_chord) = closest_mercator_data_point_on_tile(eye_hat, tile);
212                return calculate_tile_zoom(
213                    self.requested_zoom,
214                    chebyshev_chord,
215                    self.distance_z,
216                    self.distance_to_center_3d,
217                    self.field_of_view_degrees,
218                );
219            }
220        }
221        let distance_2d = if self.planar {
222            distance_to_tile_planar(self.camera, tile)
223        } else {
224            distance_to_tile_2d(self.camera, tile)
225        };
226        if let Some(focal) = self.eye_focal_pixels {
227            let distance = distance_2d.hypot(self.distance_z).max(f64::EPSILON);
228            // Bound the longest projected texel axis. A grazing-angle area estimate
229            // would erase cross-slope detail on terrain and blur roads toward the horizon.
230            (focal / (crate::coords::TILE_SIZE * distance)).log2()
231        } else {
232            calculate_tile_zoom(
233                self.requested_zoom,
234                distance_2d,
235                self.distance_z,
236                self.distance_to_center_3d,
237                self.field_of_view_degrees,
238            )
239        }
240    }
241}

243fn parallel_cos_from_sin_lat(sin_lat: f64) -> f64 {
244    let min_cos = MAX_MERCATOR_LATITUDE_DEGREES.to_radians().cos();
245    (1.0 - sin_lat * sin_lat).max(0.0).sqrt().max(min_cos)
246}

248/// The straight distance from a point to a tile's rectangle in Mercator `0..1` units.
249fn distance_to_tile_planar(point: Point2<f64>, tile: TileCoords) -> f64 {
250    let scale = tile.z.grid_scale();
251    let size = 1.0 / scale;
252    let axis = |point: f64, corner: f64| {
253        let delta = point - corner;
254        if delta < 0.0 {
255            -delta
256        } else {
257            (delta - size).max(0.0)
258        }
259    };
260    axis(point.x, f64::from(tile.x) / scale).hypot(axis(point.y, f64::from(tile.y) / scale))
261}

263/// A location in Mercator `0..1` units.
264#[cfg(test)]
265fn mercator_point(location: LatLon) -> Point2<f64> {
266    let x = location.longitude / 360.0 + 0.5;
267    let latitude = location
268        .latitude
269        .clamp(
270            -MAX_MERCATOR_LATITUDE_DEGREES,
271            MAX_MERCATOR_LATITUDE_DEGREES,
272        )
273        .to_radians();
274    let y = (1.0 - latitude.tan().asinh() / std::f64::consts::PI) * 0.5;
275    Point2::new(x, y)
276}

278/// The globe's radius in Mercator `0..1` units at a latitude, where a world unit spans the
279/// circumference scaled by the Mercator stretch there.
280#[cfg(test)]
281fn radius_world_units(latitude_degrees: f64) -> f64 {
282    let latitude = latitude_degrees
283        .clamp(
284            -MAX_MERCATOR_LATITUDE_DEGREES,
285            MAX_MERCATOR_LATITUDE_DEGREES,
286        )
287        .to_radians();
288    1.0 / (2.0 * std::f64::consts::PI * latitude.cos())
289}

291fn calculate_tile_zoom(
292    requested_center_zoom: f64,
293    distance_to_tile_2d: f64,
294    distance_to_tile_z: f64,
295    distance_to_center_3d: f64,
296    field_of_view_degrees: f64,
297) -> f64 {
298    // GL JS's factor grows without bound as the field narrows, so a sensor's view of a few
299    // degrees, tilted, would load tiles levels coarser than its screen resolves. The factor
300    // holds back the tile count of a view spanning much of the ground toward the horizon; a
301    // field narrower than the normal one sees less ground and needs no more restraint.
302    let pitch_behavior =
303        pitch_tile_loading_behavior(field_of_view_degrees.max(NORMAL_FIELD_OF_VIEW_DEGREES));
304    let center_pitch = (distance_to_tile_z / distance_to_center_3d).acos();
305    let half_fov = (field_of_view_degrees * 0.5).to_radians();
306    let tile_count_pitch_zero = 2.0 * integral_cos_power(pitch_behavior - 1.0, 0.0, half_fov);
307    let highest_pitch = (center_pitch + half_fov).min(MAX_MERCATOR_HORIZON_DEGREES.to_radians());
308    let lowest_pitch = (center_pitch - half_fov).min(highest_pitch);
309    let tile_count = integral_cos_power(pitch_behavior - 1.0, lowest_pitch, highest_pitch);
310    let tile_pitch = (distance_to_tile_2d / distance_to_tile_z).atan();
311    let distance_to_tile_3d = distance_to_tile_2d.hypot(distance_to_tile_z);
312    let distance_scale = distance_to_center_3d / distance_to_tile_3d / 0.5_f64.max(half_fov.cos());
313    requested_center_zoom + distance_scale.log2() + pitch_behavior * tile_pitch.cos().log2() * 0.5
314        - (tile_count / tile_count_pitch_zero / TILE_COUNT_MAX_MIN_RATIO)
315            .max(1.0)
316            .log2()
317            * 0.5
318}
319
320fn pitch_tile_loading_behavior(field_of_view_degrees: f64) -> f64 {
321    let numerator = (MAX_MERCATOR_HORIZON_DEGREES - field_of_view_degrees)
322        .to_radians()
323        .cos();
324    let denominator = MAX_MERCATOR_HORIZON_DEGREES.to_radians().cos();
325    2.0 * ((MAX_ZOOM_LEVELS_ON_SCREEN - 1.0) / (numerator / denominator).log2() - 1.0)
326}
327
328fn integral_cos_power(power: f64, start: f64, end: f64) -> f64 {
329    let width = (end - start) / INTEGRATION_POINTS as f64;
330    (0..INTEGRATION_POINTS)
331        .map(|index| {
332            let x = start + (index as f64 + 0.5) * width;
333            width * x.cos().powf(power)
334        })
335        .sum()
336}
