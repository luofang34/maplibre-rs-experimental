1//! Extrudes clipped polar tile boundaries of subdivided fill polygons to the globe poles.
2
3use std::{collections::HashMap, ops::Range};
4
5use lyon::tessellation::{VertexBuffers, VertexId};
6
7use super::{vertex_index, FillSubdivisionOptions, NORTH_POLE_Y, SOUTH_POLE_Y};
8use crate::{coords::EXTENT_UINT, render::ShaderVertex};
9
10/// Minimum horizontal span (in 4096-extent tile units) required to treat an unpaired horizontal
11/// edge inside the tile buffer (`y < 0` or `y > EXTENT`) as a tile-clipped polar boundary rather
12/// than a quantized natural tangent at the tip of a closed coastal bay or island.
13const MIN_BUFFERED_POLE_CLIP_SPAN: f32 = 8.0;
14
15pub(super) fn append_pole_quads<I>(
16    buffer: &mut VertexBuffers<ShaderVertex, I>,
17    index_start: usize,
18    options: FillSubdivisionOptions,
19    cache: &mut HashMap<(i32, i32), usize>,
20) where
21    I: Copy + Into<u32> + From<VertexId>,
22{
23    if !options.extend_to_north_pole && !options.extend_to_south_pole {
24        return;
25    }
26    let input_end = buffer.indices.len();
27    if index_start >= input_end {
28        return;
29    }
30
31    let mut min_y = f32::INFINITY;
32    let mut max_y = f32::NEG_INFINITY;
33    let mut edge_counts: HashMap<(usize, usize), u32> =
34        HashMap::with_capacity(input_end - index_start);
35
36    for offset in (index_start..input_end).step_by(3) {
37        let triangle = [
38            buffer.indices[offset].into() as usize,
39            buffer.indices[offset + 1].into() as usize,
40            buffer.indices[offset + 2].into() as usize,
41        ];
42        for &vertex in &triangle {
43            let y = buffer.vertices[vertex].position[1];
44            min_y = min_y.min(y);
45            max_y = max_y.max(y);
46        }
47        for edge in 0..3 {
48            let a = triangle[edge];
49            let b = triangle[(edge + 1) % 3];
50            if a != b {
51                let unordered = if a < b { (a, b) } else { (b, a) };
52                *edge_counts.entry(unordered).or_insert(0) += 1;
53            }
54        }
55    }
56
57    if options.extend_to_north_pole && min_y <= 0.0 {
58        extrude_boundary_at_y(
59            buffer,
60            index_start..input_end,
61            min_y,
62            min_y < 0.0,
63            NORTH_POLE_Y,
64            &edge_counts,
65            cache,
66        );
67    }
68
69    let south_extent = EXTENT_UINT as f32;
70    if options.extend_to_south_pole && max_y >= south_extent {
71        extrude_boundary_at_y(
72            buffer,
73            index_start..input_end,
74            max_y,
75            max_y > south_extent,
76            SOUTH_POLE_Y,
77            &edge_counts,
78            cache,
79        );
80    }
81}

83fn extrude_boundary_at_y<I>(
84    buffer: &mut VertexBuffers<ShaderVertex, I>,
85    index_range: Range<usize>,
86    target_y: f32,
87    in_buffer: bool,
88    pole_y: f32,
89    edge_counts: &HashMap<(usize, usize), u32>,
90    cache: &mut HashMap<(i32, i32), usize>,
91) where
92    I: Copy + Into<u32> + From<VertexId>,
93{
94    let mut boundary_edges = Vec::new();
95    let mut total_span = 0.0_f32;
96    for offset in index_range.step_by(3) {
97        let triangle = [
98            buffer.indices[offset].into() as usize,
99            buffer.indices[offset + 1].into() as usize,
100            buffer.indices[offset + 2].into() as usize,
101        ];
102        for edge in 0..3 {
103            let first = triangle[edge];
104            let second = triangle[(edge + 1) % 3];
105            if first == second {
106                continue;
107            }
108            let first_pos = buffer.vertices[first].position;
109            let second_pos = buffer.vertices[second].position;
110            if first_pos[1] != target_y || second_pos[1] != target_y {
111                continue;
112            }
113            let unordered = if first < second {
114                (first, second)
115            } else {
116                (second, first)
117            };
118            if edge_counts.get(&unordered).copied() == Some(1) {
119                total_span += (second_pos[0] - first_pos[0]).abs();
120                boundary_edges.push((first, second, first_pos, second_pos));
121            }
122        }
123    }
124
125    if boundary_edges.is_empty() || (in_buffer && total_span < MIN_BUFFERED_POLE_CLIP_SPAN) {
126        return;
127    }
128
129    for (first, second, first_position, second_position) in boundary_edges {
130        append_pole_quad(
131            buffer,
132            first,
133            second,
134            first_position,
135            second_position,
136            pole_y,
137            cache,
138        );
139    }
140}

142fn append_pole_quad<I>(
143    buffer: &mut VertexBuffers<ShaderVertex, I>,
144    first: usize,
145    second: usize,
146    first_position: [f32; 2],
147    second_position: [f32; 2],
148    pole_y: f32,
149    cache: &mut HashMap<(i32, i32), usize>,
150) where
151    I: From<VertexId>,
152{
153    let first_pole = vertex_index(buffer, [first_position[0], pole_y], cache);
154    let second_pole = vertex_index(buffer, [second_position[0], pole_y], cache);
155    let flip = (first_position[0] > second_position[0]) != (pole_y == NORTH_POLE_Y);
156    let triangles = if flip {
157        [
158            [first, second, first_pole],
159            [second, second_pole, first_pole],
160        ]
161    } else {
162        [
163            [second, first, first_pole],
164            [second_pole, second, first_pole],
165        ]
166    };
167    for triangle in triangles {
168        buffer
169            .indices
170            .extend(triangle.map(|index| I::from(VertexId::from_usize(index))));
171    }
