13const CAP_CACHE_LEN: usize = 2048;

15thread_local! {
16    static CAP_CACHE: RefCell<Vec<(TileCoords, TileSphericalCap)>> = const { RefCell::new(Vec::new()) };

19#[inline(always)]
20fn cap_cache_slot(tile: TileCoords) -> usize {
21    let mut h = u64::from(tile.x).wrapping_mul(0x517c_c1b7_2722_0a95)
22        ^ u64::from(tile.y).wrapping_mul(0x9e37_79b9_7f4a_7c15)
23        ^ u64::from(u8::from(tile.z)).wrapping_mul(0xbf58_476d_1ce4_e5b9);
24    h ^= h >> 32;
25    (h as usize) & (CAP_CACHE_LEN - 1)
26}

28/// Singularity-free unit-sphere spherical cap `{ x in S^2 : x . center >= cos_alpha }`
29/// enclosing a canonical globe tile (including its polar fan at `y == 0` or `y == 2^z - 1`).
30#[derive(Clone, Copy, Debug, PartialEq)]
31pub struct TileSphericalCap {
32    /// Outward unit-sphere axis of the cap (`||center|| = 1`).
33    pub center: Vector3<f64>,
34    /// Cosine of the angular half-aperture `alpha` (`-1.0` for `z == 0`, hemisphere `~0.0` for
35    /// `z == 1`, and strictly positive for `z >= 2`).
36    pub cos_alpha: f64,
37    /// Sine of the angular half-aperture `alpha` (`0.0` for `z == 0`, `in [0.0, 1.0]` for `z >= 1`).
38    pub sin_alpha: f64,
39}
40
41impl TileSphericalCap {
42    /// Classifies the radial cap shell `[min_radius, max_radius] * Cap` against the half-space
43    /// `plane.truncate().dot(p) + plane.w >= 0.0`.
44    pub(crate) fn shell_intersects_plane(
45        &self,
46        plane: Vector4<f64>,
47        min_radius: f64,
48        max_radius: f64,
49    ) -> Intersection {
50        let normal = plane.truncate();
51        let len = normal.magnitude();
52        if len <= f64::EPSILON {
53            return if plane.w >= 0.0 {
54                Intersection::Full
55            } else {
56                Intersection::None
57            };
58        }
59        let n = normal / len;
60        let d = plane.w / len;
61        let (unit_max, unit_min) = if self.cos_alpha <= -1.0 {
62            (1.0, -1.0)
63        } else {
64            let u = n.dot(self.center).clamp(-1.0, 1.0);
65            let s = (1.0 - u * u).max(0.0).sqrt();
66            let hi = if u >= self.cos_alpha {
67                1.0
68            } else {
69                u * self.cos_alpha + s * self.sin_alpha
70            };
71            let lo = if -u >= self.cos_alpha {
72                -1.0
73            } else {
74                u * self.cos_alpha - s * self.sin_alpha
75            };
76            (hi, lo)
77        };
78        let max_val = if unit_max >= 0.0 {
79            max_radius * unit_max
80        } else {
81            min_radius * unit_max
82        } + d;
83        if max_val < 0.0 {
84            return Intersection::None;
85        }
86        let min_val = if unit_min >= 0.0 {
87            min_radius * unit_min
88        } else {
89            max_radius * unit_min
90        } + d;
91        if min_val >= 0.0 {
92            Intersection::Full
93        } else {
94            Intersection::Partial
95        }
96    }
97}

99/// Computes the tight unit-sphere spherical cap enclosing `tile` (including polar fans).
100pub fn tile_spherical_cap(tile: TileCoords) -> TileSphericalCap {
101    let zoom = u8::from(tile.z);
102    if zoom == 0 {
103        return TileSphericalCap {
104            center: Vector3::unit_z(),
105            cos_alpha: -1.0,
106            sin_alpha: 0.0,
107        };
108    }
109    if zoom < 32 {
110        let slot = cap_cache_slot(tile);
111        if let Some(hit) = CAP_CACHE.with(|cache| {
112            let borrow = cache.borrow();
113            borrow
114                .get(slot)
115                .filter(|(cached, _)| *cached == tile)
116                .map(|(_, cap)| *cap)
117        }) {
118            return hit;
119        }
120        let computed = compute_tile_spherical_cap(tile, zoom);
121        CAP_CACHE.with(|cache| {
122            let mut borrow = cache.borrow_mut();
123            if borrow.is_empty() {
124                let sentinel = (
125                    TileCoords {
126                        x: u32::MAX,
127                        y: u32::MAX,
128                        z: ZoomLevel::new(u8::MAX),
129                    },
130                    TileSphericalCap {
131                        center: Vector3::unit_z(),
132                        cos_alpha: -1.0,
133                        sin_alpha: 0.0,
134                    },
135                );
136                borrow.resize(CAP_CACHE_LEN, sentinel);
137            }
138            borrow[slot] = (tile, computed);
139        });
140        return computed;
141    }
142    compute_tile_spherical_cap(tile, zoom)
143}

145fn compute_tile_spherical_cap(tile: TileCoords, zoom: u8) -> TileSphericalCap {
146    let tile_count = 1_u32 << u32::from(zoom);
147    let scale = f64::from(tile_count);
148    let (lon_mid, _) = mercator_to_angular_radians((f64::from(tile.x) + 0.5) / scale, 0.5);
149    let lat_north = if tile.y == 0 {
150        std::f64::consts::FRAC_PI_2
151    } else {
152        mercator_to_angular_radians(0.5, f64::from(tile.y) / scale).1
153    };
154    let lat_south = if tile.y == tile_count - 1 {
155        -std::f64::consts::FRAC_PI_2
156    } else {
157        mercator_to_angular_radians(0.5, f64::from(tile.y + 1) / scale).1
158    };
159    let center = angular_radians_to_unit_sphere(lon_mid, 0.5 * (lat_north + lat_south));
160    let corners = super::tile_corners(tile);
161    let north_mid =
162        project_tile_coordinates_to_unit_sphere(tile.x, tile.y, zoom, EXTENT * 0.5, 0.0);
163    let south_mid =
164        project_tile_coordinates_to_unit_sphere(tile.x, tile.y, zoom, EXTENT * 0.5, EXTENT);
165    let mut cos_alpha = corners
166        .into_iter()
167        .chain([north_mid, south_mid])
168        .map(|point| center.dot(point))
169        .fold(1.0_f64, f64::min);
170    if tile.y == 0 {
171        cos_alpha = cos_alpha.min(center.dot(Vector3::unit_y()));
172    }
173    if tile.y == tile_count - 1 {
174        cos_alpha = cos_alpha.min(center.dot(-Vector3::unit_y()));
175    }
176    let cos_alpha = (cos_alpha - 1e-12).clamp(-1.0, 1.0);
177    let sin_alpha = (1.0 - cos_alpha * cos_alpha).max(0.0).sqrt();
178    TileSphericalCap {
179        center,
180        cos_alpha,
181        sin_alpha,
182    }
183}

185/// Closest point on `tile`'s spherical quadrangle (including polar extensions at `y == 0` and
186/// `y == 2^z - 1`) to `eye_hat` on `S^2`, returning `(closest_point, chebyshev_chord)` where
187/// `chebyshev_chord = max(chord_lon, chord_lat)` decomposes `||eye_hat - closest_point||` cleanly.
188pub(crate) fn closest_unit_sphere_point_on_tile(
189    eye_hat: Vector3<f64>,
190    tile: TileCoords,
191) -> (Vector3<f64>, f64) {
192    closest_unit_sphere_point_in_lat_band(eye_hat, tile, true)
193}

195/// Closest point on `tile`'s actual Web Mercator data quadrangle (`|lat| <= 85.051129°`, excluding
196/// the synthetic data-free polar fan) to `eye_hat` on `S^2`, returning
197/// `(closest_point, chebyshev_chord)`.
198pub(crate) fn closest_mercator_data_point_on_tile(
199    eye_hat: Vector3<f64>,
200    tile: TileCoords,
201) -> (Vector3<f64>, f64) {
202    closest_unit_sphere_point_in_lat_band(eye_hat, tile, false)
203}

205fn closest_unit_sphere_point_in_lat_band(
206    eye_hat: Vector3<f64>,
207    tile: TileCoords,
208    include_poles: bool,
209) -> (Vector3<f64>, f64) {
210    let zoom = u8::from(tile.z);
211    if zoom == 0 && include_poles {
212        return (eye_hat, 0.0);
213    }
214    let tile_count = 1_u32 << u32::from(zoom);
215    let scale = f64::from(tile_count);
216    let (sin_north, cos_north) = if include_poles && tile.y == 0 {
217        (1.0, 0.0)
218    } else {
219        mercator_to_angular_radians(0.5, f64::from(tile.y) / scale)
220            .1
221            .sin_cos()
222    };
223    let (sin_south, cos_south) = if include_poles && tile.y == tile_count - 1 {
224        (-1.0, 0.0)
225    } else {
226        mercator_to_angular_radians(0.5, f64::from(tile.y + 1) / scale)
227            .1
228            .sin_cos()
229    };
230    let r_xz = eye_hat.x.hypot(eye_hat.z);
231    let (hx, hz, a) = closest_meridian(eye_hat, tile.x, zoom, scale, r_xz);
232    let m = a.hypot(eye_hat.y);
233    let (sin_lat, cos_lat) =
234        if a >= 0.0 && m > f64::EPSILON && eye_hat.y >= m * sin_south && eye_hat.y <= m * sin_north
235        {
236            (eye_hat.y / m, a / m)
237        } else if a * cos_north + eye_hat.y * sin_north >= a * cos_south + eye_hat.y * sin_south {
238            (sin_north, cos_north)
239        } else {
240            (sin_south, cos_south)
241        };
242    let q_hat = Vector3::new(hx * cos_lat, sin_lat, hz * cos_lat);
243    let chord_lat = (r_xz - cos_lat).hypot(eye_hat.y - sin_lat);
244    let chord_lon = (2.0 * cos_lat * (r_xz - a)).max(0.0).sqrt();
245    (q_hat, chord_lon.max(chord_lat))
246}

248fn closest_meridian(
249    eye_hat: Vector3<f64>,
250    tile_x: u32,
251    zoom: u8,
252    scale: f64,
253    r_xz: f64,
254) -> (f64, f64, f64) {
255    if zoom == 0 {
256        return if r_xz <= f64::EPSILON {
257            (0.0, 1.0, 0.0)
258        } else {
259            (eye_hat.x / r_xz, eye_hat.z / r_xz, r_xz)
260        };
261    }
262    let lon_min = (f64::from(tile_x) / scale) * std::f64::consts::TAU + std::f64::consts::PI;
263    let lon_max = (f64::from(tile_x + 1) / scale) * std::f64::consts::TAU + std::f64::consts::PI;
264    let lon_mid = 0.5 * (lon_min + lon_max);
265    let half_dlon = std::f64::consts::PI / scale;
266    if r_xz <= f64::EPSILON {
267        let (s, c) = lon_mid.sin_cos();
268        (s, c, 0.0)
269    } else {
270        let (sin_mid, cos_mid) = lon_mid.sin_cos();
271        if eye_hat.x * sin_mid + eye_hat.z * cos_mid >= r_xz * half_dlon.cos() {
272            (eye_hat.x / r_xz, eye_hat.z / r_xz, r_xz)
273        } else {
274            let (sin_min, cos_min) = lon_min.sin_cos();
275            let (sin_max, cos_max) = lon_max.sin_cos();
276            let g_min = eye_hat.x * sin_min + eye_hat.z * cos_min;
277            let g_max = eye_hat.x * sin_max + eye_hat.z * cos_max;
278            if g_min >= g_max {
279                (sin_min, cos_min, g_min)
280            } else {
281                (sin_max, cos_max, g_max)
282            }
283        }
284    }
