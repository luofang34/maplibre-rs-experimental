25const ASSUMED_MAX_FEATURE_HEIGHT_METERS: f64 = 500.0;
26const MAX_MERCATOR_HORIZON_DEGREES: f64 = 89.25;

29/// How a fractional per-tile zoom becomes a tile level: floored for the view and for vector
30/// sources, rounded for raster sources, as GL JS `roundZoom`.
31#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
32pub enum ZoomRounding {
33    /// Take the level at or below the zoom.
34    #[default]
35    Floor,
36    /// Take the nearest level.
37    Round,
38}
39
40impl ZoomRounding {
41    /// Applies the rounding to a fractional zoom.
42    pub fn apply(self, zoom: f64) -> f64 {
43        match self {
44            Self::Floor => zoom.floor(),
45            Self::Round => zoom.round(),
46        }
47    }
48}

50/// Zoom levels a source serves. The covering never descends past `max`, and drops tiles it
51/// would select below `min`, as GL JS does with a source's `minzoom` and `maxzoom`.
52#[derive(Clone, Copy, Debug, PartialEq, Eq)]
53pub struct SourceZoomRange {
54    /// Lowest level with tiles.
55    pub min: u8,
56    /// Highest level with tiles; the covering stops descending here.
57    pub max: u8,
58}
59
60impl Default for SourceZoomRange {
61    fn default() -> Self {
62        Self {
63            min: 0,
64            max: (MAX_ZOOM - 1) as u8,
65        }
66    }
67}
68
69impl SourceZoomRange {
70    /// The range a style source declares, with the crate limits where it declares none.
71    pub fn from_style(minzoom: Option<u8>, maxzoom: Option<u8>) -> Self {
72        let default = Self::default();
73        Self {
74            min: minzoom.unwrap_or(default.min),
75            max: maxzoom.unwrap_or(default.max).min(default.max),
76        }
77    }
78
79    /// The level to descend to for a tile whose desired level is `desired`.
80    pub fn cap(self, desired: ZoomLevel) -> ZoomLevel {
81        ZoomLevel::new(u8::from(desired).min(self.max))
82    }
83
84    /// Whether a selected tile at `level` exists in the source.
85    pub fn serves(self, level: ZoomLevel) -> bool {
86        u8::from(level) >= self.min
87    }
88}

92pub struct GlobeCoveringOptions {
93    /// Canonical zoom level to select.
94    pub zoom: ZoomLevel,
95    /// Fractional map zoom used by the per-tile LOD calculation.
96    pub requested_zoom: f64,
97    /// Enables per-tile zoom variation at high map zooms.

104    pub padding: i32,

106    pub max_tiles: usize,
107}

109/// Failure while selecting visible globe tiles.
110#[derive(Clone, Copy, Debug, Error, PartialEq)]
111pub enum GlobeCoveringError {
112    /// The coordinate model only supports canonical zoom levels through 31.
113    #[error("globe covering zoom {zoom} exceeds the supported maximum")]
114    UnsupportedZoom {
115        /// Unsupported zoom level.
116        zoom: u8,
117    },
118    /// A tile bounding volume could not be constructed.
119    #[error("failed to construct globe tile bounds")]
120    TileBounds {
121        /// Underlying bounds error.
122        #[source]
123        source: GlobeTileBoundsError,
124    },
125}

134/// Selects canonical tiles intersecting both the camera frustum and visible globe hemisphere.
135///
136/// `elevation` supplies the elevation range of each tile's culling volume.
137pub fn covering_tiles(
138    camera: &GlobeCameraState,
139    options: GlobeCoveringOptions,
140    elevation: &dyn TileElevationProvider,
141) -> Result<Vec<WorldTileCoords>, GlobeCoveringError> {
142    let lod = lod::LodContext::new(camera, options.requested_zoom);
143    covering_tiles_with_history(camera, options, (lod, FarPlane::Horizon), elevation, None)
144}

146/// Selects the tiles as [`covering_tiles`] does, each at the level `lod` asks for it, out
147/// to `far`.
148pub(crate) fn covering_tiles_with_history(
149    camera: &GlobeCameraState,
150    options: GlobeCoveringOptions,
151    (lod, far): (lod::LodContext, FarPlane),
152    elevation: &dyn TileElevationProvider,
153    history: Option<&crate::projection::lod_history::LodHistory>,
154) -> Result<Vec<WorldTileCoords>, GlobeCoveringError> {
155    if usize::from(u8::from(options.zoom)) >= MAX_ZOOM {
156        return Err(GlobeCoveringError::UnsupportedZoom {
157            zoom: u8::from(options.zoom),
158        });
159    }
160    if options.max_tiles == 0 {
161        return Ok(Vec::new());
162    }
163    let frustum = GlobeFrustum::from_camera(camera, far);
164    let focus_unit = camera.focus_unit_sphere();
165    let inspect = |tile: WorldTileCoords, fully_visible| -> Result<_, GlobeCoveringError> {
166        let tile = TileCoords::from((tile.x as u32, tile.y as u32, tile.z));
167        let range = elevation.elevation_range(tile);
168        validate_elevation(range).map_err(|source| GlobeCoveringError::TileBounds { source })?;
169        let intersection = if fully_visible {
170            Intersection::Full
171        } else {
172            let min_radius = camera.body().unit_radius_at(range.min_meters.min(0.0));
173            let max_radius = camera.body().unit_radius_at(range.max_meters.max(0.0));
174            let cap = tile_spherical_cap(tile);
175            let cap_intersection = combine_intersections(
176                frustum.intersects_cap(&cap, min_radius, max_radius),
177                cap.shell_intersects_plane(camera.clipping_plane(), min_radius, max_radius),
178            );
179            match cap_intersection {
180                Intersection::None => Intersection::None,
181                Intersection::Full => Intersection::Full,
182                Intersection::Partial => {
183                    let bounds = compact_tile_volume(tile, range, camera.body())
184                        .map_err(|source| GlobeCoveringError::TileBounds { source })?;
185                    tile_intersection(&frustum, camera.clipping_plane(), &bounds)
186                }
187            }
188        };
189        if intersection == Intersection::None {
190            return Ok(None);
191        }
192        let target = options.zoom_range.cap(if options.variable_zoom {
193            lod.stable_zoom_for_tile(tile, options.rounding, history)
194        } else {
195            options.zoom
196        });
197        Ok(Some(crate::projection::tile_covering::Refinement {
198            target,
199            fully_visible: intersection == Intersection::Full,
200        }))
201    };
202    let mut visible = crate::projection::tile_covering::bounded_spherical(
203        options.max_tiles,
204        options.zoom_range.min,
205        focus_unit,
206        inspect,
207    )?;
208    sort_by_focus(&mut visible, focus_unit);
209    Ok(add_padding(visible, options.padding, options.max_tiles))
210}

212/// Returns the conservative elevation used to retain features near the frustum horizon.
213pub fn elevation_for_tile_culling(camera: &GlobeCameraState, center_elevation: f64) -> f64 {
214    let bottom_edge_above_horizontal = MAX_MERCATOR_HORIZON_DEGREES
215        - camera.pitch_degrees()
216        - camera.field_of_view_degrees() * 0.5;
217    let proximity = ((TILE_CULLING_HORIZON_ONSET_DEGREES - bottom_edge_above_horizontal)
218        / TILE_CULLING_HORIZON_ONSET_DEGREES)
219        .clamp(0.0, 1.0);
220    center_elevation + proximity * ASSUMED_MAX_FEATURE_HEIGHT_METERS
221}

223fn tile_intersection(
224    frustum: &GlobeFrustum,
225    clipping_plane: Vector4<f64>,
226    bounds: &CompactTileVolume,
227) -> Intersection {
228    let points = bounds.points_slice();
229    let result = frustum.intersects_volume(points, &bounds.planes);
230    if result == Intersection::None {
231        return result;
232    }
233    combine_intersections(
234        result,
235        classify_points(points, |point| clipping_plane.dot(point.extend(1.0))),
236    )
237}

239pub(crate) fn classify_points<T>(points: &[T], distance: impl Fn(&T) -> f64) -> Intersection {
240    let inside = points.iter().filter(|point| distance(point) >= 0.0).count();
241    if inside == 0 {
242        Intersection::None
243    } else if inside == points.len() {
244        Intersection::Full
245    } else {
246        Intersection::Partial
247    }
248}
249
250fn combine_intersections(left: Intersection, right: Intersection) -> Intersection {
251    match (left, right) {
252        (Intersection::None, _) | (_, Intersection::None) => Intersection::None,
253        (Intersection::Full, Intersection::Full) => Intersection::Full,
254        _ => Intersection::Partial,
255    }
256}

258pub(crate) fn sort_by_focus(tiles: &mut [WorldTileCoords], focus_unit: Vector3<f64>) {
259    tiles.sort_by(|left, right| {
260        crate::projection::tile_covering::spherical_tile_distance(*left, focus_unit)
261            .total_cmp(&crate::projection::tile_covering::spherical_tile_distance(
262                *right, focus_unit,
263            ))
264            .then_with(|| left.cmp(right))
265    });
266}

273pub(crate) fn add_padding(
274    visible: Vec<WorldTileCoords>,
275    padding: i32,
276    max_tiles: usize,
277) -> Vec<WorldTileCoords> {
278    if max_tiles == 0 || visible.is_empty() {
279        return Vec::new();
280    }
281    if padding <= 0 || visible.len() >= max_tiles {
282        return visible;
283    }
284    let mut padded = visible.clone();
285    let mut seen: HashSet<_> = padded.iter().copied().collect();
286    for tile in visible {
287        let count = 1_i64 << u8::from(tile.z);
288        for delta_x in -padding..=padding {
289            for delta_y in -padding..=padding {
290                let y = i64::from(tile.y) + i64::from(delta_y);
291                if !(0..count).contains(&y) {
292                    continue;
293                }
294                let candidate = WorldTileCoords {
295                    x: (i64::from(tile.x) + i64::from(delta_x)).rem_euclid(count) as i32,
296                    y: y as i32,
297                    z: tile.z,
298                };
299                if seen.insert(candidate) {
300                    padded.push(candidate);
301                    if padded.len() == max_tiles {
302                        return padded;
303                    }
304                }
305            }
306        }
307    }
308    padded
309}
