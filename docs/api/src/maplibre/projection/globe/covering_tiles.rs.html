26/// How a fractional per-tile zoom becomes a tile level: floored for the view and for vector
27/// sources, rounded for raster sources, as GL JS `roundZoom`.
28#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
29pub enum ZoomRounding {
30    /// Take the level at or below the zoom.
31    #[default]
32    Floor,
33    /// Take the nearest level.
34    Round,
35}
36
37impl ZoomRounding {
38    /// Applies the rounding to a fractional zoom.
39    pub fn apply(self, zoom: f64) -> f64 {
40        match self {
41            Self::Floor => zoom.floor(),
42            Self::Round => zoom.round(),
43        }
44    }
45}

47/// Zoom levels a source serves. The covering never descends past `max`, and drops tiles it
48/// would select below `min`, as GL JS does with a source's `minzoom` and `maxzoom`.
49#[derive(Clone, Copy, Debug, PartialEq, Eq)]
50pub struct SourceZoomRange {
51    /// Lowest level with tiles.
52    pub min: u8,
53    /// Highest level with tiles; the covering stops descending here.
54    pub max: u8,
55}
56
57impl Default for SourceZoomRange {
58    fn default() -> Self {
59        Self {
60            min: 0,
61            max: (MAX_ZOOM - 1) as u8,
62        }
63    }
64}
65
66impl SourceZoomRange {
67    /// The range a style source declares, with the crate limits where it declares none.
68    pub fn from_style(minzoom: Option<u8>, maxzoom: Option<u8>) -> Self {
69        let default = Self::default();
70        Self {
71            min: minzoom.unwrap_or(default.min),
72            max: maxzoom.unwrap_or(default.max).min(default.max),
73        }
74    }
75
76    /// The level to descend to for a tile whose desired level is `desired`.
77    pub fn cap(self, desired: ZoomLevel) -> ZoomLevel {
78        ZoomLevel::new(u8::from(desired).min(self.max))
79    }
80
81    /// Whether a selected tile at `level` exists in the source.
82    pub fn serves(self, level: ZoomLevel) -> bool {
83        u8::from(level) >= self.min
84    }
85}

89pub struct GlobeCoveringOptions {
90    /// Canonical zoom level to select.
91    pub zoom: ZoomLevel,
92    /// Fractional map zoom used by the per-tile LOD calculation.
93    pub requested_zoom: f64,
94    /// Enables per-tile zoom variation at high map zooms.
95    pub variable_zoom: bool,
96    /// How the per-tile zoom becomes a level.
97    pub rounding: ZoomRounding,
98    /// Levels the source serves.
99    pub zoom_range: SourceZoomRange,
100    /// Number of canonical neighbors to add around visible tiles.
101    pub padding: i32,
102    /// Maximum number of returned tiles after padding.
103    pub max_tiles: usize,
104}

106/// Failure while selecting visible globe tiles.
107#[derive(Clone, Copy, Debug, Error, PartialEq)]
108pub enum GlobeCoveringError {
109    /// The coordinate model only supports canonical zoom levels through 31.
110    #[error("globe covering zoom {zoom} exceeds the supported maximum")]
111    UnsupportedZoom {
112        /// Unsupported zoom level.
113        zoom: u8,
114    },
115    /// A tile bounding volume could not be constructed.
116    #[error("failed to construct globe tile bounds")]
117    TileBounds {
118        /// Underlying bounds error.
119        #[source]
120        source: GlobeTileBoundsError,
121    },
122}

131/// Selects canonical tiles intersecting both the camera frustum and visible globe hemisphere.
132///
133/// `elevation` supplies the elevation range of each tile's culling volume.
134pub fn covering_tiles(
135    camera: &GlobeCameraState,
136    options: GlobeCoveringOptions,
137    elevation: &dyn TileElevationProvider,
138) -> Result<Vec<WorldTileCoords>, GlobeCoveringError> {
139    covering_tiles_with_history(camera, options, elevation, None)
140}

142pub(crate) fn covering_tiles_with_history(
143    camera: &GlobeCameraState,
144    options: GlobeCoveringOptions,
145    elevation: &dyn TileElevationProvider,
146    history: Option<&crate::projection::lod_history::LodHistory>,
147) -> Result<Vec<WorldTileCoords>, GlobeCoveringError> {
148    if usize::from(u8::from(options.zoom)) >= MAX_ZOOM {
149        return Err(GlobeCoveringError::UnsupportedZoom {
150            zoom: u8::from(options.zoom),
151        });
152    }
153    if options.max_tiles == 0 {
154        return Ok(Vec::new());
155    }
156    let lod = lod::LodContext::new(camera, options.requested_zoom);
157    let frustum = GlobeFrustum::from_camera(camera);
158    let priority = if camera.is_external_eye() {
159        super::unit_sphere_to_lat_lon(camera.camera_position())
160    } else {
161        camera.center()
162    };
163    let inspect = |tile: WorldTileCoords, fully_visible| -> Result<_, GlobeCoveringError> {
164        let tile = TileCoords::from((tile.x as u32, tile.y as u32, tile.z));
165        let bounds =
166            globe_tile_bounding_volume(tile, elevation.elevation_range(tile), camera.body())
167                .map_err(|source| GlobeCoveringError::TileBounds { source })?;
168        let intersection = if fully_visible {
169            Intersection::Full
170        } else {
171            tile_intersection(&frustum, camera.clipping_plane(), &bounds)
172        };
173        if intersection == Intersection::None {
174            return Ok(None);
175        }
176        let target = options.zoom_range.cap(if options.variable_zoom {
177            lod.stable_zoom_for_tile(tile, options.rounding, history)
178        } else {
179            options.zoom
180        });
181        Ok(Some(crate::projection::tile_covering::Refinement {
182            target,
183            fully_visible: intersection == Intersection::Full,
184        }))
185    };
186    let mut visible = if camera.is_external_eye() {
187        crate::projection::tile_covering::bounded(
188            options.max_tiles,
189            options.zoom_range.min,
190            priority,
191            inspect,
192        )?
193    } else {
194        unbounded_covering(options.zoom_range.min, 0, inspect)?
195    };
196    sort_by_center(&mut visible, priority);
197    Ok(add_padding(visible, options.padding, options.max_tiles))
198}

200pub(crate) fn unbounded_covering<E>(
201    min_zoom: u8,
202    copies: i32,
203    mut inspect: impl FnMut(
204        WorldTileCoords,
205        bool,
206    ) -> Result<Option<crate::projection::tile_covering::Refinement>, E>,
207) -> Result<Vec<WorldTileCoords>, E> {
208    // Copies of the world to either side start from their own roots, as GL JS renders them.
209    let mut stack: Vec<_> = (-copies..=copies)
210        .map(|wrap| (WorldTileCoords::from((wrap, 0, ZoomLevel::new(0))), false))
211        .collect();
212    let mut visible = Vec::new();
213    while let Some((tile, fully_visible)) = stack.pop() {
214        let Some(refinement) = inspect(tile, fully_visible)? else {
215            continue;
216        };
217        if tile.z >= refinement.target {
218            if u8::from(tile.z) >= min_zoom {
219                visible.push(tile);
220            }
221        } else {
222            stack.extend(
223                tile.get_children()
224                    .map(|child| (child, refinement.fully_visible)),
225            );
226        }
227    }
228    Ok(visible)
229}

231/// Returns the conservative elevation used to retain features near the frustum horizon.
232pub fn elevation_for_tile_culling(camera: &GlobeCameraState, center_elevation: f64) -> f64 {
233    let bottom_edge_above_horizontal = MAX_MERCATOR_HORIZON_DEGREES
234        - camera.pitch_degrees()
235        - camera.field_of_view_degrees() * 0.5;
236    let proximity = ((TILE_CULLING_HORIZON_ONSET_DEGREES - bottom_edge_above_horizontal)
237        / TILE_CULLING_HORIZON_ONSET_DEGREES)
238        .clamp(0.0, 1.0);
239    center_elevation + proximity * ASSUMED_MAX_FEATURE_HEIGHT_METERS
240}

242fn tile_intersection(
243    frustum: &GlobeFrustum,
244    clipping_plane: Vector4<f64>,
245    bounds: &GlobeTileBoundingVolume,
246) -> Intersection {
247    let result = frustum.intersects(bounds);
248    if result == Intersection::None {
249        return result;
250    }
251    combine_intersections(
252        result,
253        classify_points(&bounds.points, |point| {
254            clipping_plane.dot(point.extend(1.0))
255        }),
256    )
257}
258
259pub(crate) fn classify_points<T>(points: &[T], distance: impl Fn(&T) -> f64) -> Intersection {
260    let inside = points.iter().filter(|point| distance(point) >= 0.0).count();
261    if inside == 0 {
262        Intersection::None
263    } else if inside == points.len() {
264        Intersection::Full
265    } else {
266        Intersection::Partial
267    }
268}
269
270fn combine_intersections(left: Intersection, right: Intersection) -> Intersection {
271    match (left, right) {
272        (Intersection::None, _) | (_, Intersection::None) => Intersection::None,
273        (Intersection::Full, Intersection::Full) => Intersection::Full,
274        _ => Intersection::Partial,
275    }
276}

278pub(crate) fn sort_by_center(tiles: &mut [WorldTileCoords], center: LatLon) {
279    let center_x = center.longitude / 360.0 + 0.5;
280    let latitude = center.latitude.to_radians();
281    let center_y = (1.0 - latitude.tan().asinh() / std::f64::consts::PI) * 0.5;
282    tiles.sort_by(|left, right| {
283        distance_squared(*left, center_x, center_y)
284            .total_cmp(&distance_squared(*right, center_x, center_y))
285            .then_with(|| left.cmp(right))
286    });
287}
288
289fn distance_squared(tile: WorldTileCoords, center_x: f64, center_y: f64) -> f64 {
290    let count = 2_f64.powi(i32::from(u8::from(tile.z)));
291    let dx = center_x - (f64::from(tile.x) + 0.5) / count;
292    let dx = dx - dx.round();
293    let dy = center_y - (f64::from(tile.y) + 0.5) / count;
294    dx * dx + dy * dy
295}

297pub(crate) fn add_padding(
298    visible: Vec<WorldTileCoords>,
299    padding: i32,
300    max_tiles: usize,
301) -> Vec<WorldTileCoords> {
302    if max_tiles == 0 {
303        return Vec::new();
304    }
305    if padding <= 0 {
306        return visible.into_iter().take(max_tiles).collect();
307    }
308    let mut padded: Vec<_> = visible.iter().take(max_tiles).copied().collect();
309    let mut seen: HashSet<_> = padded.iter().copied().collect();
310    if padded.len() == max_tiles {
311        return padded;
312    }
313    for tile in visible {
314        let count = 1_i64 << u8::from(tile.z);
315        for delta_x in -padding..=padding {
316            for delta_y in -padding..=padding {
317                let y = i64::from(tile.y) + i64::from(delta_y);
318                if !(0..count).contains(&y) {
319                    continue;
320                }
321                let candidate = WorldTileCoords {
322                    x: (i64::from(tile.x) + i64::from(delta_x)).rem_euclid(count) as i32,
323                    y: y as i32,
324                    z: tile.z,
325                };
326                if seen.insert(candidate) {
327                    padded.push(candidate);
328                    if padded.len() == max_tiles {
329                        return padded;
330                    }
331                }
332            }
333        }
334    }
335    padded
336}
