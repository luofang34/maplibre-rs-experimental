10const MAX_GENERATED_VERTICES_PER_SEGMENT: usize = 16_384;
11const MAX_CELLS_PER_TRIANGLE: usize = 65_536;
12const NORTH_POLE_Y: f32 = i16::MIN as f32;
13const SOUTH_POLE_Y: f32 = i16::MAX as f32;
14
15/// Controls fill-mesh subdivision and globe-only edge geometry.
16#[derive(Clone, Copy, Debug)]
17pub struct FillSubdivisionOptions {
18    /// Number of grid cells along each tile axis.
19    pub granularity: u32,
20    /// Rejects buffered geometry beyond the z0 antimeridian edges.
21    pub clip_x_to_tile: bool,
22    /// Extends fill edges at y=0 to the north pole.
23    pub extend_to_north_pole: bool,
24    /// Extends fill edges at y=extent to the south pole.
25    pub extend_to_south_pole: bool,
26}
27
28/// Failure while subdividing untrusted tile geometry.
29#[derive(Clone, Copy, Debug, Error, PartialEq)]
30pub enum SubdivisionError {
31    /// A line would generate an unreasonable number of vertices.
32    #[error("globe line subdivision exceeds {limit} generated vertices")]
33    LineVertexLimit {
34        /// Maximum allowed generated vertices.
35        limit: usize,
36    },
37    /// A triangle overlaps an unreasonable number of subdivision cells.
38    #[error("globe fill subdivision exceeds {limit} cells for one triangle")]
39    TriangleCellLimit {
40        /// Maximum allowed intersected cells.
41        limit: usize,
42    },
43    /// A tessellation index does not address an existing vertex.
44    #[error("globe subdivision index {index} is outside {vertex_count} vertices")]
45    InvalidIndex {
46        /// Invalid vertex index.
47        index: usize,
48        /// Number of available vertices.
49        vertex_count: usize,
50    },
51}

53/// Returns the GL JS subdivision policy for a canonical zoom level.
54pub fn granularity_for_zoom(base: u32, minimum: u32, zoom: u8) -> u32 {
55    let scaled = base.checked_shr(u32::from(zoom)).unwrap_or(0);
56    scaled.max(minimum).max(1)
57}

59/// Inserts every grid-axis intersection along one line segment.
60pub fn subdivide_line_segment(
61    start: [f32; 2],
62    end: [f32; 2],
63    granularity: u32,
64) -> Result<Vec<[f32; 2]>, SubdivisionError> {
65    if granularity < 2 || start == end {
66        return Ok(vec![end]);
67    }
68    let cell_size = (EXTENT_UINT / granularity).max(1) as f32;
69    let delta = [end[0] - start[0], end[1] - start[1]];
70    let mut crossings = Vec::new();
71    collect_axis_crossings(start[0], end[0], cell_size, &mut crossings);
72    collect_axis_crossings(start[1], end[1], cell_size, &mut crossings);
73    crossings.sort_by(f32::total_cmp);
74    crossings.dedup_by(|left, right| (*left - *right).abs() <= f32::EPSILON);
75    if crossings.len() > MAX_GENERATED_VERTICES_PER_SEGMENT {
76        return Err(SubdivisionError::LineVertexLimit {
77            limit: MAX_GENERATED_VERTICES_PER_SEGMENT,
78        });
79    }
80    let mut points = crossings
81        .into_iter()
82        .filter(|fraction| *fraction > 0.0 && *fraction < 1.0)
83        .map(|fraction| {
84            [
85                start[0] + delta[0] * fraction,
86                start[1] + delta[1] * fraction,
87            ]
88        })
89        .collect::<Vec<_>>();
90    points.dedup();
91    if points.last().copied() != Some(end) {
92        points.push(end);
93    }
94    Ok(points)
95}

97/// Clips newly tessellated triangles to the regular subdivision grid.
98pub fn subdivide_triangles<I>(
99    buffer: &mut VertexBuffers<ShaderVertex, I>,
100    index_start: usize,
101    options: FillSubdivisionOptions,
102) -> Result<(), SubdivisionError>
103where
104    I: Copy + Into<u32> + From<VertexId>,
105{
106    if options.granularity < 2 {
107        return Ok(());
108    }
109    sanitize_pole_sentinels(&mut buffer.vertices);
110    let input = buffer.indices[index_start..].to_vec();
111    buffer.indices.truncate(index_start);
112    let mut vertices = HashMap::new();
113    for triangle in input.as_chunks::<3>().0 {
114        let points = triangle_points(&buffer.vertices, triangle)?;
115        subdivide_triangle(
116            buffer,
117            points,
118            options.granularity,
119            options.clip_x_to_tile,
120            &mut vertices,
121        )?;
122    }
123    append_pole_quads(buffer, index_start, options, &mut vertices);
124    Ok(())
125}

127fn sanitize_pole_sentinels(vertices: &mut [ShaderVertex]) {
128    for vertex in vertices {
129        if vertex.position[1] == NORTH_POLE_Y {
130            vertex.position[1] = NORTH_POLE_Y + 1.0;
131        } else if vertex.position[1] == SOUTH_POLE_Y {
132            vertex.position[1] = SOUTH_POLE_Y - 1.0;
133        }
134    }
135}

137fn collect_axis_crossings(start: f32, end: f32, cell_size: f32, output: &mut Vec<f32>) {
138    if start == end {
139        return;
140    }
141    let low = start.min(end);
142    let high = start.max(end);
143    let first = (low / cell_size).floor() as i64 + 1;
144    let last = (high / cell_size).ceil() as i64;
145    for axis in first..last {
146        let coordinate = axis as f32 * cell_size;
147        output.push((coordinate - start) / (end - start));
148    }
149}

151fn triangle_points<I: Copy + Into<u32>>(
152    vertices: &[ShaderVertex],
153    indices: &[I],
154) -> Result<[[f32; 2]; 3], SubdivisionError> {
155    let mut points = [[0.0; 2]; 3];
156    for (slot, index) in points.iter_mut().zip(indices) {
157        let index = (*index).into() as usize;
158        let vertex = vertices.get(index).ok_or(SubdivisionError::InvalidIndex {
159            index,
160            vertex_count: vertices.len(),
161        })?;
162        *slot = vertex.position;
163    }
164    Ok(points)
165}

167fn subdivide_triangle<I>(
168    buffer: &mut VertexBuffers<ShaderVertex, I>,
169    triangle: [[f32; 2]; 3],
170    granularity: u32,
171    clip_x_to_tile: bool,
172    cache: &mut HashMap<(i32, i32), usize>,
173) -> Result<(), SubdivisionError>
174where
175    I: From<VertexId>,
176{
177    let cell_size = (EXTENT_UINT / granularity).max(1) as f32;
178    let bounds = triangle_bounds(triangle, cell_size);
179    let cell_count = (bounds.1 - bounds.0) as usize * (bounds.3 - bounds.2) as usize;
180    if cell_count > MAX_CELLS_PER_TRIANGLE {
181        return Err(SubdivisionError::TriangleCellLimit {
182            limit: MAX_CELLS_PER_TRIANGLE,
183        });
184    }
185    for y in bounds.2..bounds.3 {
186        for x in bounds.0..bounds.1 {
187            let mut polygon = triangle.to_vec();
188            polygon = clip_axis(polygon, 0, x as f32 * cell_size, true);
189            polygon = clip_axis(polygon, 0, (x + 1) as f32 * cell_size, false);
190            polygon = clip_axis(polygon, 1, y as f32 * cell_size, true);
191            polygon = clip_axis(polygon, 1, (y + 1) as f32 * cell_size, false);
192            append_polygon(buffer, polygon, clip_x_to_tile, cache);
193        }
194    }
195    Ok(())
196}

198fn triangle_bounds(triangle: [[f32; 2]; 3], cell_size: f32) -> (i64, i64, i64, i64) {
199    let min_x = triangle
200        .iter()
201        .map(|point| point[0])
202        .fold(f32::INFINITY, f32::min);
203    let max_x = triangle
204        .iter()
205        .map(|point| point[0])
206        .fold(f32::NEG_INFINITY, f32::max);
207    let min_y = triangle
208        .iter()
209        .map(|point| point[1])
210        .fold(f32::INFINITY, f32::min);
211    let max_y = triangle
212        .iter()
213        .map(|point| point[1])
214        .fold(f32::NEG_INFINITY, f32::max);
215    (
216        (min_x / cell_size).floor() as i64,
217        (max_x / cell_size).ceil() as i64,
218        (min_y / cell_size).floor() as i64,
219        (max_y / cell_size).ceil() as i64,
220    )
221}

223fn clip_axis(
224    input: Vec<[f32; 2]>,
225    axis: usize,
226    boundary: f32,
227    keep_greater: bool,
228) -> Vec<[f32; 2]> {
229    let Some(mut previous) = input.last().copied() else {
230        return input;
231    };
232    let mut output = Vec::with_capacity(input.len() + 1);
233    for current in input {
234        let previous_inside = (previous[axis] >= boundary) == keep_greater;
235        let current_inside = (current[axis] >= boundary) == keep_greater;
236        if previous_inside != current_inside {
237            output.push(axis_intersection(previous, current, axis, boundary));
238        }
239        if current_inside {
240            output.push(current);
241        }
242        previous = current;
243    }
244    output
245}

247fn axis_intersection(start: [f32; 2], end: [f32; 2], axis: usize, boundary: f32) -> [f32; 2] {
248    let fraction = (boundary - start[axis]) / (end[axis] - start[axis]);
249    let mut point = [
250        (start[0] + (end[0] - start[0]) * fraction).round(),
251        (start[1] + (end[1] - start[1]) * fraction).round(),
252    ];
253    point[axis] = boundary;
254    point
255}

257fn append_polygon<I: From<VertexId>>(
258    buffer: &mut VertexBuffers<ShaderVertex, I>,
259    polygon: Vec<[f32; 2]>,
260    clip_x_to_tile: bool,
261    cache: &mut HashMap<(i32, i32), usize>,
262) {
263    if polygon.len() < 3
264        || (clip_x_to_tile
265            && polygon
266                .iter()
267                .any(|point| point[0] < 0.0 || point[0] > EXTENT_UINT as f32))
268    {
269        return;
270    }
271    let first = vertex_index(buffer, polygon[0], cache);
272    for edge in 1..polygon.len() - 1 {
273        let second = vertex_index(buffer, polygon[edge], cache);
274        let third = vertex_index(buffer, polygon[edge + 1], cache);
275        if first != second && second != third && third != first {
276            buffer
277                .indices
278                .extend([first, second, third].map(|index| I::from(VertexId::from_usize(index))));
279        }
280    }
281}

283fn append_pole_quads<I>(
284    buffer: &mut VertexBuffers<ShaderVertex, I>,
285    index_start: usize,
286    options: FillSubdivisionOptions,
287    cache: &mut HashMap<(i32, i32), usize>,
288) where
289    I: Copy + Into<u32> + From<VertexId>,
290{
291    let input_end = buffer.indices.len();
292    for offset in (index_start..input_end).step_by(3) {
293        let triangle = [
294            buffer.indices[offset].into() as usize,
295            buffer.indices[offset + 1].into() as usize,
296            buffer.indices[offset + 2].into() as usize,
297        ];
298        for edge in 0..3 {
299            let first = triangle[edge];
300            let second = triangle[(edge + 1) % 3];
301            let first_position = buffer.vertices[first].position;
302            let second_position = buffer.vertices[second].position;
303            if options.extend_to_north_pole && first_position[1] == 0.0 && second_position[1] == 0.0
304            {
305                append_pole_quad(
306                    buffer,
307                    first,
308                    second,
309                    first_position,
310                    second_position,
311                    NORTH_POLE_Y,
312                    cache,
313                );
314            }
315            if options.extend_to_south_pole
316                && first_position[1] == EXTENT_UINT as f32
317                && second_position[1] == EXTENT_UINT as f32
318            {
319                append_pole_quad(
320                    buffer,
321                    first,
322                    second,
323                    first_position,
324                    second_position,
325                    SOUTH_POLE_Y,
326                    cache,
327                );
328            }
329        }
330    }
331}

333fn append_pole_quad<I>(
334    buffer: &mut VertexBuffers<ShaderVertex, I>,
335    first: usize,
336    second: usize,
337    first_position: [f32; 2],
338    second_position: [f32; 2],
339    pole_y: f32,
340    cache: &mut HashMap<(i32, i32), usize>,
341) where
342    I: From<VertexId>,
343{
344    let first_pole = vertex_index(buffer, [first_position[0], pole_y], cache);
345    let second_pole = vertex_index(buffer, [second_position[0], pole_y], cache);
346    let flip = (first_position[0] > second_position[0]) != (pole_y == NORTH_POLE_Y);
347    let triangles = if flip {
348        [
349            [first, second, first_pole],
350            [second, second_pole, first_pole],
351        ]
352    } else {
353        [
354            [second, first, first_pole],
355            [second_pole, second, first_pole],
356        ]
357    };
358    for triangle in triangles {
359        buffer
360            .indices
361            .extend(triangle.map(|index| I::from(VertexId::from_usize(index))));
362    }
363}

365fn vertex_index<I>(
366    buffer: &mut VertexBuffers<ShaderVertex, I>,
367    point: [f32; 2],
368    cache: &mut HashMap<(i32, i32), usize>,
369) -> usize {
370    let point = [point[0].round(), point[1].round()];
371    let key = (point[0] as i32, point[1] as i32);
372    if let Some(index) = cache.get(&key) {
373        return *index;
374    }
375    let index = buffer.vertices.len();
376    buffer.vertices.push(ShaderVertex::new(point, [0.0, 0.0]));
377    cache.insert(key, index);
378    index
379}
