10const MAX_GENERATED_VERTICES_PER_SEGMENT: usize = 16_384;
11const MAX_CELLS_PER_TRIANGLE: usize = 65_536;
12const NORTH_POLE_Y: f32 = i16::MIN as f32;
13const SOUTH_POLE_Y: f32 = i16::MAX as f32;

15/// Controls fill-mesh subdivision and globe-only edge geometry.
16#[derive(Clone, Copy, Debug)]
17pub struct FillSubdivisionOptions {
18    /// Number of grid cells along each tile axis.
19    pub granularity: u32,
20    /// Rejects buffered geometry beyond the z0 antimeridian edges.
21    pub clip_x_to_tile: bool,
22    /// Extends fill edges at y=0 to the north pole.
23    pub extend_to_north_pole: bool,
24    /// Extends fill edges at y=extent to the south pole.
25    pub extend_to_south_pole: bool,
26}
27
28/// Failure while subdividing untrusted tile geometry.
29#[derive(Clone, Copy, Debug, Error, PartialEq)]
30pub enum SubdivisionError {
31    /// A line would generate an unreasonable number of vertices.
32    #[error("globe line subdivision exceeds {limit} generated vertices")]
33    LineVertexLimit {
34        /// Maximum allowed generated vertices.
35        limit: usize,
36    },
37    /// A triangle overlaps an unreasonable number of subdivision cells.
38    #[error("globe fill subdivision exceeds {limit} cells for one triangle")]
39    TriangleCellLimit {
40        /// Maximum allowed intersected cells.
41        limit: usize,
42    },
43    /// A tessellation index does not address an existing vertex.
44    #[error("globe subdivision index {index} is outside {vertex_count} vertices")]
45    InvalidIndex {
46        /// Invalid vertex index.
47        index: usize,
48        /// Number of available vertices.
49        vertex_count: usize,
50    },
51}

53/// Returns the GL JS subdivision policy for a canonical zoom level.
54pub fn granularity_for_zoom(base: u32, minimum: u32, zoom: u8) -> u32 {
55    let scaled = base.checked_shr(u32::from(zoom)).unwrap_or(0);
56    scaled.max(minimum).max(1)
57}

59/// Inserts every grid-axis intersection along one line segment.
60pub fn subdivide_line_segment(
61    start: [f32; 2],
62    end: [f32; 2],
63    granularity: u32,
64) -> Result<Vec<[f32; 2]>, SubdivisionError> {
65    subdivide_line_segment_on_grid(start, end, granularity, false)
66}
67
68/// Inserts every grid-axis intersection along one line segment and, with `diagonals`, every
69/// crossing of a cell diagonal from its top-left to its bottom-right corner as well, where a
70/// terrain mesh splits its cells into triangles.
71pub fn subdivide_line_segment_on_grid(
72    start: [f32; 2],
73    end: [f32; 2],
74    granularity: u32,
75    diagonals: bool,
76) -> Result<Vec<[f32; 2]>, SubdivisionError> {
77    if granularity < 2 || start == end {
78        return Ok(vec![end]);
79    }
80    let cell_size = (EXTENT_UINT / granularity).max(1) as f32;
81    let delta = [end[0] - start[0], end[1] - start[1]];
82    let mut crossings = Vec::new();
83    collect_axis_crossings(start[0], end[0], cell_size, &mut crossings);
84    collect_axis_crossings(start[1], end[1], cell_size, &mut crossings);
85    if diagonals {
86        // The diagonals are the lines on which x - y is a whole number of cells.
87        collect_axis_crossings(
88            start[0] - start[1],
89            end[0] - end[1],
90            cell_size,
91            &mut crossings,
92        );
93    }
94    crossings.sort_by(f32::total_cmp);
95    crossings.dedup_by(|left, right| (*left - *right).abs() <= f32::EPSILON);
96    if crossings.len() > MAX_GENERATED_VERTICES_PER_SEGMENT {
97        return Err(SubdivisionError::LineVertexLimit {
98            limit: MAX_GENERATED_VERTICES_PER_SEGMENT,
99        });
100    }
101    let mut points = crossings
102        .into_iter()
103        .filter(|fraction| *fraction > 0.0 && *fraction < 1.0)
104        .map(|fraction| {
105            [
106                start[0] + delta[0] * fraction,
107                start[1] + delta[1] * fraction,
108            ]
109        })
110        .collect::<Vec<_>>();
111    points.dedup();
112    if points.last().copied() != Some(end) {
113        points.push(end);
114    }
115    Ok(points)
116}

118/// Clips newly tessellated triangles to the regular subdivision grid.
119pub fn subdivide_triangles<I>(
120    buffer: &mut VertexBuffers<ShaderVertex, I>,
121    index_start: usize,
122    options: FillSubdivisionOptions,
123) -> Result<(), SubdivisionError>
124where
125    I: Copy + Into<u32> + From<VertexId>,
126{
127    if options.granularity < 2 {
128        return Ok(());
129    }
130    sanitize_pole_sentinels(&mut buffer.vertices);
131    let input = buffer.indices[index_start..].to_vec();
132    buffer.indices.truncate(index_start);
133    let mut vertices = HashMap::with_capacity(input.len());
134    for triangle in input.as_chunks::<3>().0 {
135        let points = triangle_points(&buffer.vertices, triangle)?;
136        subdivide_triangle(
137            buffer,
138            points,
139            options.granularity,
140            options.clip_x_to_tile,
141            &mut vertices,
142        )?;
143    }
144    append_pole_quads(buffer, index_start, options, &mut vertices);
145    Ok(())
146}
147
148fn sanitize_pole_sentinels(vertices: &mut [ShaderVertex]) {
149    for vertex in vertices {
150        if vertex.position[1] == NORTH_POLE_Y {
151            vertex.position[1] = NORTH_POLE_Y + 1.0;
152        } else if vertex.position[1] == SOUTH_POLE_Y {
153            vertex.position[1] = SOUTH_POLE_Y - 1.0;
154        }
155    }
156}
157
158fn collect_axis_crossings(start: f32, end: f32, cell_size: f32, output: &mut Vec<f32>) {
159    if start == end {
160        return;
161    }
162    let low = start.min(end);
163    let high = start.max(end);
164    let first = (low / cell_size).floor() as i64 + 1;
165    let last = (high / cell_size).ceil() as i64;
166    for axis in first..last {
167        let coordinate = axis as f32 * cell_size;
168        output.push((coordinate - start) / (end - start));
169    }
170}

172fn triangle_points<I: Copy + Into<u32>>(
173    vertices: &[ShaderVertex],
174    indices: &[I],
175) -> Result<[[f32; 2]; 3], SubdivisionError> {
176    let mut points = [[0.0; 2]; 3];
177    for (slot, index) in points.iter_mut().zip(indices) {
178        let index = (*index).into() as usize;
179        let vertex = vertices.get(index).ok_or(SubdivisionError::InvalidIndex {
180            index,
181            vertex_count: vertices.len(),
182        })?;
183        *slot = vertex.position;
184    }
185    Ok(points)
186}

188const MAX_CLIPPED_POLYGON_VERTICES: usize = 16;
189
190/// Stack-allocated polygon buffer avoiding heap allocations during Sutherland-Hodgman clipping.
191#[derive(Clone, Copy, Debug)]
192pub(crate) struct InlinePolygon {
193    vertices: [[f32; 2]; MAX_CLIPPED_POLYGON_VERTICES],
194    len: usize,
195}
196
197impl InlinePolygon {
198    #[inline]
199    pub(crate) fn empty() -> Self {
200        Self {
201            vertices: [[0.0; 2]; MAX_CLIPPED_POLYGON_VERTICES],
202            len: 0,
203        }
204    }
205
206    #[inline]
207    pub(crate) fn from_triangle(triangle: [[f32; 2]; 3]) -> Self {
208        let mut vertices = [[0.0; 2]; MAX_CLIPPED_POLYGON_VERTICES];
209        vertices[0] = triangle[0];
210        vertices[1] = triangle[1];
211        vertices[2] = triangle[2];
212        Self { vertices, len: 3 }
213    }
214
215    #[inline]
216    pub(crate) fn as_slice(&self) -> &[[f32; 2]] {
217        &self.vertices[..self.len]
218    }
219
220    #[inline]
221    pub(crate) fn len(&self) -> usize {
222        self.len
223    }
224
225    #[inline]
226    pub(crate) fn push(&mut self, point: [f32; 2]) {
227        if self.len < MAX_CLIPPED_POLYGON_VERTICES {
228            self.vertices[self.len] = point;
229            self.len += 1;
230        }
231    }
232
233    #[inline]
234    pub(crate) fn clear(&mut self) {
235        self.len = 0;
236    }
237}

239fn subdivide_triangle<I>(
240    buffer: &mut VertexBuffers<ShaderVertex, I>,
241    triangle: [[f32; 2]; 3],
242    granularity: u32,
243    clip_x_to_tile: bool,
244    cache: &mut HashMap<(i32, i32), usize>,
245) -> Result<(), SubdivisionError>
246where
247    I: From<VertexId>,
248{
249    let cell_size = (EXTENT_UINT / granularity).max(1) as f32;
250    let bounds = triangle_bounds(triangle, cell_size);
251    let cell_count = (bounds.1 - bounds.0) as usize * (bounds.3 - bounds.2) as usize;
252    if cell_count > MAX_CELLS_PER_TRIANGLE {
253        return Err(SubdivisionError::TriangleCellLimit {
254            limit: MAX_CELLS_PER_TRIANGLE,
255        });
256    }
257    let mut poly_b = InlinePolygon::empty();
258    for y in bounds.2..bounds.3 {
259        for x in bounds.0..bounds.1 {
260            let mut poly_a = InlinePolygon::from_triangle(triangle);
261            clip_axis_into(&poly_a, &mut poly_b, 0, x as f32 * cell_size, true);
262            if poly_b.len() < 3 {
263                continue;
264            }
265            clip_axis_into(&poly_b, &mut poly_a, 0, (x + 1) as f32 * cell_size, false);
266            if poly_a.len() < 3 {
267                continue;
268            }
269            clip_axis_into(&poly_a, &mut poly_b, 1, y as f32 * cell_size, true);
270            if poly_b.len() < 3 {
271                continue;
272            }
273            clip_axis_into(&poly_b, &mut poly_a, 1, (y + 1) as f32 * cell_size, false);
274            append_polygon(buffer, poly_a.as_slice(), clip_x_to_tile, cache);
275        }
276    }
277    Ok(())
278}

280fn triangle_bounds(triangle: [[f32; 2]; 3], cell_size: f32) -> (i64, i64, i64, i64) {
281    let min_x = triangle
282        .iter()
283        .map(|point| point[0])
284        .fold(f32::INFINITY, f32::min);
285    let max_x = triangle
286        .iter()
287        .map(|point| point[0])
288        .fold(f32::NEG_INFINITY, f32::max);
289    let min_y = triangle
290        .iter()
291        .map(|point| point[1])
292        .fold(f32::INFINITY, f32::min);
293    let max_y = triangle
294        .iter()
295        .map(|point| point[1])
296        .fold(f32::NEG_INFINITY, f32::max);
297    (
298        (min_x / cell_size).floor() as i64,
299        (max_x / cell_size).ceil() as i64,
300        (min_y / cell_size).floor() as i64,
301        (max_y / cell_size).ceil() as i64,
302    )
303}

305pub(crate) fn clip_axis_into(
306    input: &InlinePolygon,
307    output: &mut InlinePolygon,
308    axis: usize,
309    boundary: f32,
310    keep_greater: bool,
311) {
312    output.clear();
313    let slice = input.as_slice();
314    let Some(mut previous) = slice.last().copied() else {
315        return;
316    };
317    for &current in slice {
318        let previous_inside = (previous[axis] >= boundary) == keep_greater;
319        let current_inside = (current[axis] >= boundary) == keep_greater;
320        if previous_inside != current_inside {
321            output.push(axis_intersection(previous, current, axis, boundary));
322        }
323        if current_inside {
324            output.push(current);
325        }
326        previous = current;
327    }
328}

330fn axis_intersection(start: [f32; 2], end: [f32; 2], axis: usize, boundary: f32) -> [f32; 2] {
331    let fraction = (boundary - start[axis]) / (end[axis] - start[axis]);
332    let mut point = [
333        (start[0] + (end[0] - start[0]) * fraction).round(),
334        (start[1] + (end[1] - start[1]) * fraction).round(),
335    ];
336    point[axis] = boundary;
337    point
338}

340fn append_polygon<I: From<VertexId>>(
341    buffer: &mut VertexBuffers<ShaderVertex, I>,
342    polygon: &[[f32; 2]],
343    clip_x_to_tile: bool,
344    cache: &mut HashMap<(i32, i32), usize>,
345) {
346    if polygon.len() < 3
347        || (clip_x_to_tile
348            && polygon
349                .iter()
350                .any(|point| point[0] < 0.0 || point[0] > EXTENT_UINT as f32))
351    {
352        return;
353    }
354    let first = vertex_index(buffer, polygon[0], cache);
355    for edge in 1..polygon.len() - 1 {
356        let second = vertex_index(buffer, polygon[edge], cache);
357        let third = vertex_index(buffer, polygon[edge + 1], cache);
358        if first != second && second != third && third != first {
359            buffer
360                .indices
361                .extend([first, second, third].map(|index| I::from(VertexId::from_usize(index))));
362        }
363    }
364}

366fn append_pole_quads<I>(
367    buffer: &mut VertexBuffers<ShaderVertex, I>,
368    index_start: usize,
369    options: FillSubdivisionOptions,
370    cache: &mut HashMap<(i32, i32), usize>,
371) where
372    I: Copy + Into<u32> + From<VertexId>,
373{
374    let input_end = buffer.indices.len();
375    for offset in (index_start..input_end).step_by(3) {
376        let triangle = [
377            buffer.indices[offset].into() as usize,
378            buffer.indices[offset + 1].into() as usize,
379            buffer.indices[offset + 2].into() as usize,
380        ];
381        for edge in 0..3 {
382            let first = triangle[edge];
383            let second = triangle[(edge + 1) % 3];
384            let first_position = buffer.vertices[first].position;
385            let second_position = buffer.vertices[second].position;
386            if options.extend_to_north_pole && first_position[1] == 0.0 && second_position[1] == 0.0
387            {
388                append_pole_quad(
389                    buffer,
390                    first,
391                    second,
392                    first_position,
393                    second_position,
394                    NORTH_POLE_Y,
395                    cache,
396                );
397            }
398            if options.extend_to_south_pole
399                && first_position[1] == EXTENT_UINT as f32
400                && second_position[1] == EXTENT_UINT as f32
401            {
402                append_pole_quad(
403                    buffer,
404                    first,
405                    second,
406                    first_position,
407                    second_position,
408                    SOUTH_POLE_Y,
409                    cache,
410                );
411            }
412        }
413    }
414}

416fn append_pole_quad<I>(
417    buffer: &mut VertexBuffers<ShaderVertex, I>,
418    first: usize,
419    second: usize,
420    first_position: [f32; 2],
421    second_position: [f32; 2],
422    pole_y: f32,
423    cache: &mut HashMap<(i32, i32), usize>,
424) where
425    I: From<VertexId>,
426{
427    let first_pole = vertex_index(buffer, [first_position[0], pole_y], cache);
428    let second_pole = vertex_index(buffer, [second_position[0], pole_y], cache);
429    let flip = (first_position[0] > second_position[0]) != (pole_y == NORTH_POLE_Y);
430    let triangles = if flip {
431        [
432            [first, second, first_pole],
433            [second, second_pole, first_pole],
434        ]
435    } else {
436        [
437            [second, first, first_pole],
438            [second_pole, second, first_pole],
439        ]
440    };
441    for triangle in triangles {
442        buffer
443            .indices
444            .extend(triangle.map(|index| I::from(VertexId::from_usize(index))));
445    }
446}

448fn vertex_index<I>(
449    buffer: &mut VertexBuffers<ShaderVertex, I>,
450    point: [f32; 2],
451    cache: &mut HashMap<(i32, i32), usize>,
452) -> usize {
453    let point = [point[0].round(), point[1].round()];
454    let key = (point[0] as i32, point[1] as i32);
455    if let Some(index) = cache.get(&key) {
456        return *index;
457    }
458    let index = buffer.vertices.len();
459    buffer.vertices.push(ShaderVertex::new(point, [0.0, 0.0]));
460    cache.insert(key, index);
461    index
462}
