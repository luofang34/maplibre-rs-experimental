10const MAX_GENERATED_VERTICES_PER_SEGMENT: usize = 16_384;
11const MAX_CELLS_PER_TRIANGLE: usize = 65_536;
12const NORTH_POLE_Y: f32 = i16::MIN as f32;
13const SOUTH_POLE_Y: f32 = i16::MAX as f32;
14
15/// Controls fill-mesh subdivision and globe-only edge geometry.
16#[derive(Clone, Copy, Debug)]
17pub struct FillSubdivisionOptions {
18    /// Number of grid cells along each tile axis.
19    pub granularity: u32,
20    /// Rejects buffered geometry beyond the z0 antimeridian edges.
21    pub clip_x_to_tile: bool,
22    /// Extends fill edges at y=0 to the north pole.
23    pub extend_to_north_pole: bool,
24    /// Extends fill edges at y=extent to the south pole.
25    pub extend_to_south_pole: bool,
26}
27
28/// Failure while subdividing untrusted tile geometry.
29#[derive(Clone, Copy, Debug, Error, PartialEq)]
30pub enum SubdivisionError {
31    /// A line would generate an unreasonable number of vertices.
32    #[error("globe line subdivision exceeds {limit} generated vertices")]
33    LineVertexLimit {
34        /// Maximum allowed generated vertices.
35        limit: usize,
36    },
37    /// A triangle overlaps an unreasonable number of subdivision cells.
38    #[error("globe fill subdivision exceeds {limit} cells for one triangle")]
39    TriangleCellLimit {
40        /// Maximum allowed intersected cells.
41        limit: usize,
42    },
43    /// A tessellation index does not address an existing vertex.
44    #[error("globe subdivision index {index} is outside {vertex_count} vertices")]
45    InvalidIndex {
46        /// Invalid vertex index.
47        index: usize,
48        /// Number of available vertices.
49        vertex_count: usize,
50    },
51}

53/// Returns the GL JS subdivision policy for a canonical zoom level.
54pub fn granularity_for_zoom(base: u32, minimum: u32, zoom: u8) -> u32 {
55    let scaled = base.checked_shr(u32::from(zoom)).unwrap_or(0);
56    scaled.max(minimum).max(1)
57}

59/// Inserts every grid-axis intersection along one line segment.
60pub fn subdivide_line_segment(
61    start: [f32; 2],
62    end: [f32; 2],
63    granularity: u32,
64) -> Result<Vec<[f32; 2]>, SubdivisionError> {
65    subdivide_line_segment_on_grid(start, end, granularity, false)
66}
67
68/// Inserts every grid-axis intersection along one line segment and, with `diagonals`, every
69/// crossing of a cell diagonal from its top-left to its bottom-right corner as well, where a
70/// terrain mesh splits its cells into triangles.
71pub fn subdivide_line_segment_on_grid(
72    start: [f32; 2],
73    end: [f32; 2],
74    granularity: u32,
75    diagonals: bool,
76) -> Result<Vec<[f32; 2]>, SubdivisionError> {
77    if granularity < 2 || start == end {
78        return Ok(vec![end]);
79    }
80    let cell_size = (EXTENT_UINT / granularity).max(1) as f32;
81    let delta = [end[0] - start[0], end[1] - start[1]];
82    let mut crossings = Vec::new();
83    collect_axis_crossings(start[0], end[0], cell_size, &mut crossings);
84    collect_axis_crossings(start[1], end[1], cell_size, &mut crossings);
85    if diagonals {
86        // The diagonals are the lines on which x - y is a whole number of cells.
87        collect_axis_crossings(
88            start[0] - start[1],
89            end[0] - end[1],
90            cell_size,
91            &mut crossings,
92        );
93    }
94    crossings.sort_by(f32::total_cmp);
95    crossings.dedup_by(|left, right| (*left - *right).abs() <= f32::EPSILON);
96    if crossings.len() > MAX_GENERATED_VERTICES_PER_SEGMENT {
97        return Err(SubdivisionError::LineVertexLimit {
98            limit: MAX_GENERATED_VERTICES_PER_SEGMENT,
99        });
100    }
101    let mut points = crossings
102        .into_iter()
103        .filter(|fraction| *fraction > 0.0 && *fraction < 1.0)
104        .map(|fraction| {
105            [
106                start[0] + delta[0] * fraction,
107                start[1] + delta[1] * fraction,
108            ]
109        })
110        .collect::<Vec<_>>();
111    points.dedup();
112    if points.last().copied() != Some(end) {
113        points.push(end);
114    }
115    Ok(points)
116}

118/// Clips newly tessellated triangles to the regular subdivision grid.
119pub fn subdivide_triangles<I>(
120    buffer: &mut VertexBuffers<ShaderVertex, I>,
121    index_start: usize,
122    options: FillSubdivisionOptions,
123) -> Result<(), SubdivisionError>
124where
125    I: Copy + Into<u32> + From<VertexId>,
126{
127    if options.granularity < 2 {
128        return Ok(());
129    }
130    sanitize_pole_sentinels(&mut buffer.vertices);
131    let input = buffer.indices[index_start..].to_vec();
132    buffer.indices.truncate(index_start);
133    let mut vertices = HashMap::new();
134    for triangle in input.as_chunks::<3>().0 {
135        let points = triangle_points(&buffer.vertices, triangle)?;
136        subdivide_triangle(
137            buffer,
138            points,
139            options.granularity,
140            options.clip_x_to_tile,
141            &mut vertices,
142        )?;
143    }
144    append_pole_quads(buffer, index_start, options, &mut vertices);
145    Ok(())
146}

148fn sanitize_pole_sentinels(vertices: &mut [ShaderVertex]) {
149    for vertex in vertices {
150        if vertex.position[1] == NORTH_POLE_Y {
151            vertex.position[1] = NORTH_POLE_Y + 1.0;
152        } else if vertex.position[1] == SOUTH_POLE_Y {
153            vertex.position[1] = SOUTH_POLE_Y - 1.0;
154        }
155    }
156}

158fn collect_axis_crossings(start: f32, end: f32, cell_size: f32, output: &mut Vec<f32>) {
159    if start == end {
160        return;
161    }
162    let low = start.min(end);
163    let high = start.max(end);
164    let first = (low / cell_size).floor() as i64 + 1;
165    let last = (high / cell_size).ceil() as i64;
166    for axis in first..last {
167        let coordinate = axis as f32 * cell_size;
168        output.push((coordinate - start) / (end - start));
169    }
170}

172fn triangle_points<I: Copy + Into<u32>>(
173    vertices: &[ShaderVertex],
174    indices: &[I],
175) -> Result<[[f32; 2]; 3], SubdivisionError> {
176    let mut points = [[0.0; 2]; 3];
177    for (slot, index) in points.iter_mut().zip(indices) {
178        let index = (*index).into() as usize;
179        let vertex = vertices.get(index).ok_or(SubdivisionError::InvalidIndex {
180            index,
181            vertex_count: vertices.len(),
182        })?;
183        *slot = vertex.position;
184    }
185    Ok(points)
186}

188fn subdivide_triangle<I>(
189    buffer: &mut VertexBuffers<ShaderVertex, I>,
190    triangle: [[f32; 2]; 3],
191    granularity: u32,
192    clip_x_to_tile: bool,
193    cache: &mut HashMap<(i32, i32), usize>,
194) -> Result<(), SubdivisionError>
195where
196    I: From<VertexId>,
197{
198    let cell_size = (EXTENT_UINT / granularity).max(1) as f32;
199    let bounds = triangle_bounds(triangle, cell_size);
200    let cell_count = (bounds.1 - bounds.0) as usize * (bounds.3 - bounds.2) as usize;
201    if cell_count > MAX_CELLS_PER_TRIANGLE {
202        return Err(SubdivisionError::TriangleCellLimit {
203            limit: MAX_CELLS_PER_TRIANGLE,
204        });
205    }
206    for y in bounds.2..bounds.3 {
207        for x in bounds.0..bounds.1 {
208            let mut polygon = triangle.to_vec();
209            polygon = clip_axis(polygon, 0, x as f32 * cell_size, true);
210            polygon = clip_axis(polygon, 0, (x + 1) as f32 * cell_size, false);
211            polygon = clip_axis(polygon, 1, y as f32 * cell_size, true);
212            polygon = clip_axis(polygon, 1, (y + 1) as f32 * cell_size, false);
213            append_polygon(buffer, polygon, clip_x_to_tile, cache);
214        }
215    }
216    Ok(())
217}

219fn triangle_bounds(triangle: [[f32; 2]; 3], cell_size: f32) -> (i64, i64, i64, i64) {
220    let min_x = triangle
221        .iter()
222        .map(|point| point[0])
223        .fold(f32::INFINITY, f32::min);
224    let max_x = triangle
225        .iter()
226        .map(|point| point[0])
227        .fold(f32::NEG_INFINITY, f32::max);
228    let min_y = triangle
229        .iter()
230        .map(|point| point[1])
231        .fold(f32::INFINITY, f32::min);
232    let max_y = triangle
233        .iter()
234        .map(|point| point[1])
235        .fold(f32::NEG_INFINITY, f32::max);
236    (
237        (min_x / cell_size).floor() as i64,
238        (max_x / cell_size).ceil() as i64,
239        (min_y / cell_size).floor() as i64,
240        (max_y / cell_size).ceil() as i64,
241    )
242}

244fn clip_axis(
245    input: Vec<[f32; 2]>,
246    axis: usize,
247    boundary: f32,
248    keep_greater: bool,
249) -> Vec<[f32; 2]> {
250    let Some(mut previous) = input.last().copied() else {
251        return input;
252    };
253    let mut output = Vec::with_capacity(input.len() + 1);
254    for current in input {
255        let previous_inside = (previous[axis] >= boundary) == keep_greater;
256        let current_inside = (current[axis] >= boundary) == keep_greater;
257        if previous_inside != current_inside {
258            output.push(axis_intersection(previous, current, axis, boundary));
259        }
260        if current_inside {
261            output.push(current);
262        }
263        previous = current;
264    }
265    output
266}

268fn axis_intersection(start: [f32; 2], end: [f32; 2], axis: usize, boundary: f32) -> [f32; 2] {
269    let fraction = (boundary - start[axis]) / (end[axis] - start[axis]);
270    let mut point = [
271        (start[0] + (end[0] - start[0]) * fraction).round(),
272        (start[1] + (end[1] - start[1]) * fraction).round(),
273    ];
274    point[axis] = boundary;
275    point
276}

278fn append_polygon<I: From<VertexId>>(
279    buffer: &mut VertexBuffers<ShaderVertex, I>,
280    polygon: Vec<[f32; 2]>,
281    clip_x_to_tile: bool,
282    cache: &mut HashMap<(i32, i32), usize>,
283) {
284    if polygon.len() < 3
285        || (clip_x_to_tile
286            && polygon
287                .iter()
288                .any(|point| point[0] < 0.0 || point[0] > EXTENT_UINT as f32))
289    {
290        return;
291    }
292    let first = vertex_index(buffer, polygon[0], cache);
293    for edge in 1..polygon.len() - 1 {
294        let second = vertex_index(buffer, polygon[edge], cache);
295        let third = vertex_index(buffer, polygon[edge + 1], cache);
296        if first != second && second != third && third != first {
297            buffer
298                .indices
299                .extend([first, second, third].map(|index| I::from(VertexId::from_usize(index))));
300        }
301    }
302}

304fn append_pole_quads<I>(
305    buffer: &mut VertexBuffers<ShaderVertex, I>,
306    index_start: usize,
307    options: FillSubdivisionOptions,
308    cache: &mut HashMap<(i32, i32), usize>,
309) where
310    I: Copy + Into<u32> + From<VertexId>,
311{
312    let input_end = buffer.indices.len();
313    for offset in (index_start..input_end).step_by(3) {
314        let triangle = [
315            buffer.indices[offset].into() as usize,
316            buffer.indices[offset + 1].into() as usize,
317            buffer.indices[offset + 2].into() as usize,
318        ];
319        for edge in 0..3 {
320            let first = triangle[edge];
321            let second = triangle[(edge + 1) % 3];
322            let first_position = buffer.vertices[first].position;
323            let second_position = buffer.vertices[second].position;
324            if options.extend_to_north_pole && first_position[1] == 0.0 && second_position[1] == 0.0
325            {
326                append_pole_quad(
327                    buffer,
328                    first,
329                    second,
330                    first_position,
331                    second_position,
332                    NORTH_POLE_Y,
333                    cache,
334                );
335            }
336            if options.extend_to_south_pole
337                && first_position[1] == EXTENT_UINT as f32
338                && second_position[1] == EXTENT_UINT as f32
339            {
340                append_pole_quad(
341                    buffer,
342                    first,
343                    second,
344                    first_position,
345                    second_position,
346                    SOUTH_POLE_Y,
347                    cache,
348                );
349            }
350        }
351    }
352}

354fn append_pole_quad<I>(
355    buffer: &mut VertexBuffers<ShaderVertex, I>,
356    first: usize,
357    second: usize,
358    first_position: [f32; 2],
359    second_position: [f32; 2],
360    pole_y: f32,
361    cache: &mut HashMap<(i32, i32), usize>,
362) where
363    I: From<VertexId>,
364{
365    let first_pole = vertex_index(buffer, [first_position[0], pole_y], cache);
366    let second_pole = vertex_index(buffer, [second_position[0], pole_y], cache);
367    let flip = (first_position[0] > second_position[0]) != (pole_y == NORTH_POLE_Y);
368    let triangles = if flip {
369        [
370            [first, second, first_pole],
371            [second, second_pole, first_pole],
372        ]
373    } else {
374        [
375            [second, first, first_pole],
376            [second_pole, second, first_pole],
377        ]
378    };
379    for triangle in triangles {
380        buffer
381            .indices
382            .extend(triangle.map(|index| I::from(VertexId::from_usize(index))));
383    }
384}

386fn vertex_index<I>(
387    buffer: &mut VertexBuffers<ShaderVertex, I>,
388    point: [f32; 2],
389    cache: &mut HashMap<(i32, i32), usize>,
390) -> usize {
391    let point = [point[0].round(), point[1].round()];
392    let key = (point[0] as i32, point[1] as i32);
393    if let Some(index) = cache.get(&key) {
394        return *index;
395    }
396    let index = buffer.vertices.len();
397    buffer.vertices.push(ShaderVertex::new(point, [0.0, 0.0]));
398    cache.insert(key, index);
399    index
400}
