18const MIN_ROTATION_AXIS_SQUARED: f64 = 1e-24;
19const MAX_INERTIA_LONGITUDE_DELTA: f64 = 179.5;
20const MAX_VALID_LATITUDE: f64 = 85.051_128_779_806_59;
21const PAN_FALLOFF_BAND: f64 = 0.1;
22const PAN_MAX_ANGLE: f64 = PI * 0.98;
23const DIAL_MIN_RADIUS_PIXELS: f64 = 20.0;
24
25/// Camera-center change that keeps a geographic anchor under the cursor.
26#[derive(Clone, Copy, Debug)]
27pub struct GlobePanUpdate {
28    /// Rotated geographic map center.
29    pub center: LatLon,
30    /// Zoom delta preserving the globe's apparent radius across latitude changes.
31    pub zoom_adjustment: f64,
32}

34/// Limits an inertial pan target so an ease animation cannot choose the opposite direction.
35pub fn clamp_pan_inertia_center(current_center: LatLon, target_center: LatLon) -> LatLon {
36    let longitude_delta = target_center.longitude - current_center.longitude;
37    if longitude_delta.abs() <= 180.0 {
38        return target_center;
39    }
40    LatLon::new(
41        target_center.latitude,
42        current_center.longitude + longitude_delta.signum() * MAX_INERTIA_LONGITUDE_DELTA,
43    )
44}

46/// Maps a pointer ray to the virtual globe trackball used during drag panning.
47pub fn pan_surface_location(camera: &GlobeCameraState, pixel: Point2<f64>) -> Option<LatLon> {
48    let origin = camera.camera_position();
49    let distance = origin.magnitude();
50    if distance <= 1.0 {
51        return camera.screen_point_to_location(pixel);
52    }
53    let up = origin / distance;
54    let direction = camera.ray_direction_from_pixel(pixel)?;
55    let forward = -direction.dot(up);
56    let lateral = direction + up * forward;
57    let lateral_length = lateral.magnitude();
58    if lateral_length < 1e-9 {
59        return camera.screen_point_to_location(pixel);
60    }
61    let angle = lateral_length.atan2(forward);
62    let horizon_angle = (1.0 / distance).asin();
63    let handover_angle = horizon_angle * (1.0 - PAN_FALLOFF_BAND);
64    if angle < handover_angle {
65        return camera.screen_point_to_location(pixel);
66    }
67    let sin_handover = distance * handover_angle.sin();
68    let target_at_handover = sin_handover.clamp(-1.0, 1.0).asin() - handover_angle;
69    let slope_at_handover = distance * handover_angle.cos()
70        / (1.0 - sin_handover * sin_handover).max(1e-12).sqrt()
71        - 1.0;
72    let room = PAN_MAX_ANGLE - target_at_handover;
73    let excess = angle - handover_angle;
74    let target = target_at_handover
75        + room * (slope_at_handover * excess) / (room + slope_at_handover * excess);
76    let target = target.clamp(0.0, PAN_MAX_ANGLE);
77    let surface = up * target.cos() + lateral / lateral_length * target.sin();
78    Some(unit_sphere_to_lat_lon(surface.normalize()))
79}

81/// Applies one pixel drag using GL JS's surface fallback and pole-dial behavior.
82pub fn pan_camera_by_pixels(
83    camera: &GlobeCameraState,
84    requested_anchor: Point2<f64>,
85    center_pixel: Point2<f64>,
86    pan_delta: Vector2<f64>,
87) -> Option<GlobePanUpdate> {
88    let anchor = if camera.is_point_on_map_surface(requested_anchor) {
89        requested_anchor
90    } else {
91        center_pixel
92    };
93    let current = pan_surface_location(camera, anchor)?;
94    let target = pan_surface_location(camera, anchor - pan_delta)?;
95    let mut update =
96        pan_center_to_anchor(camera.center(), camera.bearing_degrees(), target, current);
97    let pole_latitude = if camera.center().latitude >= 0.0 {
98        90.0
99    } else {
100        -90.0
101    };
102    let pole = LatLon::new(pole_latitude, 0.0);
103    let pole_pixel = camera.location_to_screen(pole, 0.0);
104    update.center.longitude = fixed_bearing_longitude(
105        camera.center(),
106        anchor,
107        pan_delta,
108        pole_pixel,
109        update.center.longitude,
110    );
111    update.zoom_adjustment =
112        globe_zoom_adjustment(camera.center().latitude, update.center.latitude);
113    Some(update)
114}

116fn fixed_bearing_longitude(
117    old_center: LatLon,
118    cursor: Point2<f64>,
119    pan_delta: Vector2<f64>,
120    pole_pixel: Point2<f64>,
121    swing_longitude: f64,
122) -> f64 {
123    let radial = cursor - pole_pixel;
124    let radius_squared = radial.magnitude2();
125    let ramp = (1.0 - (MAX_VALID_LATITUDE - old_center.latitude.abs()) / 12.0).clamp(0.0, 1.0);
126    let dial = ramp * ramp * (3.0 - 2.0 * ramp);
127    let swing = wrap_longitude(swing_longitude - old_center.longitude);
128    let sweep = (radial.x * pan_delta.y - radial.y * pan_delta.x)
129        / radius_squared.max(DIAL_MIN_RADIUS_PIXELS * DIAL_MIN_RADIUS_PIXELS);
130    let pole_sign = if old_center.latitude >= 0.0 {
131        1.0
132    } else {
133        -1.0
134    };
135    let dial_longitude = pole_sign * sweep.to_degrees();
136    wrap_longitude(old_center.longitude + (1.0 - dial) * swing + dial * dial_longitude)
137}

139/// Rotates a globe so `anchor` moves to the surface location currently under the cursor.
140///
141/// The rotation uses the same surface-vector and quaternion frames as MapLibre GL JS. Bearing is
142/// intentionally fixed for drag panning.
143pub fn pan_center_to_anchor(
144    current_center: LatLon,
145    bearing_degrees: f64,
146    anchor: LatLon,
147    cursor_location: LatLon,
148) -> GlobePanUpdate {
149    let target = lat_lon_to_unit_sphere(anchor);
150    let current = lat_lon_to_unit_sphere(cursor_location);
151    let axis = target.cross(current);
152    let axis_length_squared = axis.magnitude2();
153    let dot = target.dot(current).clamp(-1.0, 1.0);
154    let half_angle = dot.acos() * 0.5;
155    let delta = if axis_length_squared > MIN_ROTATION_AXIS_SQUARED {
156        let scaled_axis = axis * (half_angle.sin() / axis_length_squared.sqrt());
157        Quaternion::new(
158            half_angle.cos(),
159            scaled_axis.y,
160            -scaled_axis.x,
161            scaled_axis.z,
162        )
163    } else {
164        Quaternion::new(1.0, 0.0, 0.0, 0.0)
165    };
166    let orientation = orientation_from_lat_lon_bearing(current_center, bearing_degrees) * delta;
167    let rotated = lat_lon_bearing_from_orientation(orientation);
168    let center = LatLon::new(
169        rotated.center.latitude.clamp(-90.0, 90.0),
170        wrap_longitude(rotated.center.longitude),
171    );
172
173    GlobePanUpdate {
174        center,
175        zoom_adjustment: globe_zoom_adjustment(current_center.latitude, center.latitude),
176    }
177}
