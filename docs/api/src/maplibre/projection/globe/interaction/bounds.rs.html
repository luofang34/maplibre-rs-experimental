1//! Globe scale solving used by camera-for-bounds calculations.
2
3use cgmath::{InnerSpace, Matrix4, Vector3, Vector4};
4
5/// Clip-space dimension constrained by a bounds edge.
6#[derive(Clone, Copy, Debug, Eq, PartialEq)]
7pub enum ClipDimension {
8    /// Horizontal clip-space coordinate.
9    X,
10    /// Vertical clip-space coordinate.
11    Y,
12}
13
14/// Solves the globe scale that projects `surface` onto one clip-space edge.
15///
16/// `center_surface` accounts for the camera moving toward the geographic center while zooming.
17pub fn solve_vector_scale(
18    surface: Vector3<f64>,
19    center_surface: Vector3<f64>,
20    projection: Matrix4<f64>,
21    dimension: ClipDimension,
22    target: f64,
23) -> Option<f64> {
24    let axis = match dimension {
25        ClipDimension::X => Vector4::new(
26            projection.x.x,
27            projection.y.x,
28            projection.z.x,
29            projection.w.x,
30        ),
31        ClipDimension::Y => Vector4::new(
32            projection.x.y,
33            projection.y.y,
34            projection.z.y,
35            projection.w.y,
36        ),
37    };
38    let homogeneous = Vector4::new(
39        projection.x.w,
40        projection.y.w,
41        projection.z.w,
42        projection.w.w,
43    );
44    let surface_axis = surface.dot(axis.truncate());
45    let surface_w = surface.dot(homogeneous.truncate());
46    let center_axis = center_surface.dot(axis.truncate());
47    let center_w = center_surface.dot(homogeneous.truncate());
48    let numerator = center_axis + axis.w - target * center_w - target * homogeneous.w;
49    let denominator = center_axis - surface_axis - target * center_w + target * surface_w;
50    let invalid_first = center_axis + target * surface_w == surface_axis + target * center_w;
51    let invalid_second = homogeneous.w * (surface_axis - center_axis)
52        + axis.w * (center_w - surface_w)
53        + surface_axis * center_w
54        == center_axis * surface_w;
55    if invalid_first || invalid_second || denominator == 0.0 {
56        return None;
57    }
58    let scale = numerator / denominator;
59    scale.is_finite().then_some(scale)
60}
61
62/// Keeps the smaller non-negative candidate scale.
63pub fn lesser_non_negative(current: f64, candidate: Option<f64>) -> f64 {
64    match candidate {
65        Some(candidate) if candidate >= 0.0 && candidate < current => candidate,
66        _ => current,
67    }
68}
