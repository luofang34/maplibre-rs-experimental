12/// Minimum and maximum elevation included in a tile bounding volume.

14pub struct TileElevationRange {

33/// Convex volume approximating a curved globe tile and its elevation range.
34#[derive(Clone, Debug, PartialEq)]
35pub struct GlobeTileBoundingVolume {
36    /// Points defining the convex shape.
37    pub points: Vec<Vector3<f64>>,
38    /// Inward-facing half-space planes containing the volume.
39    pub planes: [Vector4<f64>; 6],
40    /// Minimum corner of the tile surface's rejection AABB.
41    pub min: Vector3<f64>,
42    /// Maximum corner of the tile surface's rejection AABB.
43    pub max: Vector3<f64>,
44}
45
46/// Failure while constructing a globe tile bounding volume.
47#[derive(Clone, Copy, Debug, Error, PartialEq)]
48pub enum GlobeTileBoundsError {
49    /// Elevation values must be finite.
50    #[error("tile elevation range must be finite, got {min_meters}..{max_meters} metres")]
51    InvalidElevation {
52        /// Invalid minimum elevation.
53        min_meters: f64,
54        /// Invalid maximum elevation.
55        max_meters: f64,
56    },
57    /// The tile's boundary planes do not have a unique intersection.
58    #[error("globe tile planes are degenerate for z={z}, x={x}, y={y}")]
59    DegeneratePlanes {
60        /// Tile zoom.
61        z: u8,
62        /// Tile X coordinate.
63        x: u32,
64        /// Tile Y coordinate.
65        y: u32,
66    },
67}

69/// Returns the nearest wrapped, pole-aware Mercator distance from a point to a tile.
70pub fn distance_to_tile_2d(point: Point2<f64>, tile: TileCoords) -> f64 {
71    let scale = 2_f64.powi(i32::from(u8::from(tile.z)));
72    let size = 1.0 / scale;
73    let corner_x = f64::from(tile.x) / scale;
74    let corner_y = f64::from(tile.y) / scale;
75    let original = distance_to_tile_wrap_x(point, corner_x, corner_y, size);
76    let north = distance_to_tile_wrap_x(point, corner_x + 0.5, -corner_y - size, size);
77    let south = distance_to_tile_wrap_x(point, corner_x + 0.5, 2.0 - corner_y - size, size);
78    original.min(north).min(south)
79}

81/// Selects the wrap nearest to a Mercator center so tiles persist across the antimeridian.
82pub fn nearest_tile_wrap(center_x: f64, tile: TileCoords) -> i8 {
83    let scale = 2_f64.powi(i32::from(u8::from(tile.z)));
84    let size = 1.0 / scale;
85    let tile_x = f64::from(tile.x) / scale;
86    let current = distance_to_interval(center_x, tile_x, size);
87    let left = distance_to_interval(center_x, tile_x - 1.0, size);
88    let right = distance_to_interval(center_x, tile_x + 1.0, size);
89    let smallest = current.min(left).min(right);
90    if smallest == right {
91        1
92    } else if smallest == left {
93        -1
94    } else {
95        0
96    }
97}

99/// Computes the convex bounding volume for a globe tile and elevation range.
100pub fn globe_tile_bounding_volume(
101    tile: TileCoords,
102    elevation: TileElevationRange,
103    body: Body,
104) -> Result<GlobeTileBoundingVolume, GlobeTileBoundsError> {
105    validate_elevation(elevation)?;
106    let min_radius = body.unit_radius_at(elevation.min_meters.min(0.0));
107    let max_radius = body.unit_radius_at(elevation.max_meters.max(0.0));
108    match u8::from(tile.z) {
109        0 => Ok(aabb_volume(
110            Vector3::new(-max_radius, -max_radius, -max_radius),
111            Vector3::new(max_radius, max_radius, max_radius),
112        )),
113        1 => Ok(zoom_one_volume(tile, max_radius)),
114        _ => curved_tile_volume(tile, min_radius, max_radius),
115    }
116}
117
118fn validate_elevation(elevation: TileElevationRange) -> Result<(), GlobeTileBoundsError> {
119    if !elevation.min_meters.is_finite() || !elevation.max_meters.is_finite() {
120        return Err(GlobeTileBoundsError::InvalidElevation {
121            min_meters: elevation.min_meters,
122            max_meters: elevation.max_meters,
123        });
124    }
125    Ok(())
126}

128fn distance_to_interval(point: f64, start: f64, size: f64) -> f64 {
129    let delta = point - start;
130    if delta < 0.0 {
131        -delta
132    } else {
133        (delta - size).max(0.0)
134    }
135}
136
137fn distance_to_tile_wrap_x(point: Point2<f64>, corner_x: f64, corner_y: f64, size: f64) -> f64 {
138    let delta_x = point.x - corner_x;
139    let distance_x = if delta_x < 0.0 {
140        (-delta_x).min(1.0 + delta_x - size)
141    } else if delta_x > size {
142        (delta_x - size).max(0.0).min(1.0 - delta_x)
143    } else {
144        0.0
145    };
146    distance_x.max(distance_to_interval(point.y, corner_y, size))
147}

149fn zoom_one_volume(tile: TileCoords, radius: f64) -> GlobeTileBoundingVolume {
150    let min = Vector3::new(
151        if tile.x == 0 { -radius } else { 0.0 },
152        if tile.y == 0 { 0.0 } else { -radius },
153        -radius,
154    );
155    let max = Vector3::new(
156        if tile.x == 0 { 0.0 } else { radius },
157        if tile.y == 0 { radius } else { 0.0 },
158        radius,
159    );
160    aabb_volume(min, max)
161}

163pub(crate) fn aabb_volume(min: Vector3<f64>, max: Vector3<f64>) -> GlobeTileBoundingVolume {
164    let mut points = Vec::with_capacity(8);
165    for index in 0..8 {
166        points.push(Vector3::new(
167            if index & 1 == 0 { min.x } else { max.x },
168            if index & 2 == 0 { min.y } else { max.y },
169            if index & 4 == 0 { min.z } else { max.z },
170        ));
171    }
172    GlobeTileBoundingVolume {
173        points,
174        planes: [
175            Vector4::new(-1.0, 0.0, 0.0, max.x),
176            Vector4::new(1.0, 0.0, 0.0, -min.x),
177            Vector4::new(0.0, -1.0, 0.0, max.y),
178            Vector4::new(0.0, 1.0, 0.0, -min.y),
179            Vector4::new(0.0, 0.0, -1.0, max.z),
180            Vector4::new(0.0, 0.0, 1.0, -min.z),
181        ],
182        min,
183        max,
184    }
185}

187fn curved_tile_volume(
188    tile: TileCoords,
189    min_radius: f64,
190    max_radius: f64,
191) -> Result<GlobeTileBoundingVolume, GlobeTileBoundsError> {
192    let corners = tile_corners(tile);
193    let mut extremes: Vec<_> = corners.iter().map(|corner| *corner * max_radius).collect();
194    if min_radius != max_radius {
195        extremes.extend(corners.iter().map(|corner| *corner * min_radius));
196    }
197    extend_pole_extremes(&mut extremes, tile);
198    let (min, max) = bounds(&extremes);
199    let axes = tile_axes(tile, corners);
200    extend_curved_edge_extremes(&mut extremes, tile, axes.center, max_radius);
201    let planes = tile_planes(&extremes, axes);
202    let points = volume_points(tile, planes)?;
203    Ok(GlobeTileBoundingVolume {
204        points,
205        planes,
206        min,
207        max,
208    })
209}

211fn tile_corners(tile: TileCoords) -> [Vector3<f64>; 4] {
212    let zoom = u8::from(tile.z);
213    [
214        project_tile_coordinates_to_unit_sphere(tile.x, tile.y, zoom, 0.0, 0.0),
215        project_tile_coordinates_to_unit_sphere(tile.x, tile.y, zoom, EXTENT, 0.0),
216        project_tile_coordinates_to_unit_sphere(tile.x, tile.y, zoom, EXTENT, EXTENT),
217        project_tile_coordinates_to_unit_sphere(tile.x, tile.y, zoom, 0.0, EXTENT),
218    ]
219}
220
221fn extend_pole_extremes(extremes: &mut Vec<Vector3<f64>>, tile: TileCoords) {
222    let tile_count = 2_u32.pow(u32::from(u8::from(tile.z)));
223    if tile.y == 0 {
224        extremes.push(Vector3::unit_y());
225    }
226    if tile.y == tile_count - 1 {
227        extremes.push(-Vector3::unit_y());
228    }
229}

231fn bounds(points: &[Vector3<f64>]) -> (Vector3<f64>, Vector3<f64>) {
232    let mut min = Vector3::new(f64::INFINITY, f64::INFINITY, f64::INFINITY);
233    let mut max = Vector3::new(f64::NEG_INFINITY, f64::NEG_INFINITY, f64::NEG_INFINITY);
234    for point in points {
235        min.x = min.x.min(point.x);
236        min.y = min.y.min(point.y);
237        min.z = min.z.min(point.z);
238        max.x = max.x.max(point.x);
239        max.y = max.y.max(point.y);
240        max.z = max.z.max(point.z);
241    }
242    (min, max)
243}

245#[derive(Clone, Copy)]
246struct TileAxes {
247    center: Vector3<f64>,
248    north: Vector3<f64>,
249    east: Vector3<f64>,
250    west: Vector3<f64>,
251}
252
253fn tile_axes(tile: TileCoords, corners: [Vector3<f64>; 4]) -> TileAxes {
254    let center = project_tile_coordinates_to_unit_sphere(
255        tile.x,
256        tile.y,
257        u8::from(tile.z),
258        EXTENT / 2.0,
259        EXTENT / 2.0,
260    );
261    let center_east = Vector3::unit_y().cross(center).normalize();
262    TileAxes {
263        center,
264        north: center.cross(center_east).normalize(),
265        east: corners[2].cross(corners[1]).normalize(),
266        west: corners[0].cross(corners[3]).normalize(),
267    }
268}

270fn extend_curved_edge_extremes(
271    extremes: &mut Vec<Vector3<f64>>,
272    tile: TileCoords,
273    center: Vector3<f64>,
274    max_radius: f64,
275) {
276    extremes.push(center * max_radius);
277    let tile_count = 2_u32.pow(u32::from(u8::from(tile.z)));
278    let edge_y = if tile.y >= tile_count / 2 {
279        0.0
280    } else {
281        EXTENT
282    };
283    let edge_midpoint = project_tile_coordinates_to_unit_sphere(
284        tile.x,
285        tile.y,
286        u8::from(tile.z),
287        EXTENT / 2.0,
288        edge_y,
289    );
290    extremes.push(edge_midpoint * max_radius);
291}

293fn tile_planes(points: &[Vector3<f64>], axes: TileAxes) -> [Vector4<f64>; 6] {
294    let (up_min, up_max) = axis_min_max(axes.center, points);
295    let (north_min, north_max) = axis_min_max(axes.north, points);
296    [
297        (-axes.center).extend(up_max),
298        axes.center.extend(-up_min),
299        (-axes.north).extend(north_max),
300        axes.north.extend(-north_min),
301        axes.east.extend(0.0),
302        axes.west.extend(0.0),
303    ]
304}
305
306fn axis_min_max(axis: Vector3<f64>, points: &[Vector3<f64>]) -> (f64, f64) {
307    points
308        .iter()
309        .fold((f64::INFINITY, f64::NEG_INFINITY), |(min, max), point| {
310            let value = axis.dot(*point);
311            (min.min(value), max.max(value))
312        })
313}

315fn volume_points(
316    tile: TileCoords,
317    planes: [Vector4<f64>; 6],
318) -> Result<Vec<Vector3<f64>>, GlobeTileBoundsError> {
319    let [up, down, north, south, east, west] = planes;
320    let tile_count = 2_u32.pow(u32::from(u8::from(tile.z)));
321    let mut triples = Vec::with_capacity(8);
322    if tile.y == 0 {
323        triples.extend([(west, east, up), (west, east, down)]);
324    } else {
325        triples.extend([
326            (north, east, up),
327            (north, east, down),
328            (north, west, up),
329            (north, west, down),
330        ]);
331    }
332    if tile.y == tile_count - 1 {
333        triples.extend([(west, east, up), (west, east, down)]);
334    } else {
335        triples.extend([
336            (south, east, up),
337            (south, east, down),
338            (south, west, up),
339            (south, west, down),
340        ]);
341    }
342    triples
343        .into_iter()
344        .map(|(first, second, third)| {
345            three_plane_intersection(first, second, third).ok_or_else(|| bounds_error(tile))
346        })
347        .collect()
348}
349
350fn three_plane_intersection(
351    first: Vector4<f64>,
352    second: Vector4<f64>,
353    third: Vector4<f64>,
354) -> Option<Vector3<f64>> {
355    let first_normal = first.truncate();
356    let second_normal = second.truncate();
357    let third_normal = third.truncate();
358    let determinant = first_normal.dot(second_normal.cross(third_normal));
359    if determinant.abs() <= f64::EPSILON {
360        return None;
361    }
362    Some(
363        (second_normal.cross(third_normal) * -first.w
364            + third_normal.cross(first_normal) * -second.w
365            + first_normal.cross(second_normal) * -third.w)
366            / determinant,
367    )
368}
369
370fn bounds_error(tile: TileCoords) -> GlobeTileBoundsError {
371    GlobeTileBoundsError::DegeneratePlanes {
372        z: u8::from(tile.z),
373        x: tile.x,
374        y: tile.y,
375    }
