37    pub pitch_degrees: f64,

39    pub roll_degrees: f64,

41    pub center_offset: Point2<f64>,

48    /// `None` derives it from the world size at the center's latitude, which grows without

64/// Failure while constructing or querying a globe camera.
65#[derive(Clone, Copy, Debug, Error, PartialEq)]
66pub enum GlobeCameraError {
67    /// Viewport dimensions must be finite and positive.
68    #[error("globe viewport must be finite and positive, got {width}x{height}")]
69    InvalidViewport {
70        /// Invalid viewport width.
71        width: f64,
72        /// Invalid viewport height.
73        height: f64,
74    },
75    /// Field of view must be finite and strictly between zero and 180 degrees.
76    #[error("globe field of view must be between 0 and 180 degrees, got {degrees}")]
77    InvalidFieldOfView {
78        /// Invalid field of view in degrees.
79        degrees: f64,
80    },
81    /// World size must be finite and positive.
82    #[error("globe world size must be finite and positive, got {world_size}")]
83    InvalidWorldSize {
84        /// Invalid world size in pixels.
85        world_size: f64,
86    },
87    /// A camera angle or center component must be finite.
88    #[error("{name} must be finite, got {degrees}")]
89    InvalidAngle {
90        /// Name of the invalid angle.
91        name: &'static str,
92        /// Invalid angle in degrees.
93        degrees: f64,
94    },
95    /// Perspective center offset must be finite.
96    #[error("globe perspective center offset must be finite, got ({x}, {y})")]
97    InvalidCenterOffset {
98        /// Invalid horizontal offset.
99        x: f64,
100        /// Invalid vertical offset.
101        y: f64,
102    },
103    /// The orbit target lies at or below the center of the body.
104    #[error("the camera target lies {meters} m above sea level, at or below the body's center")]
105    InvalidTargetElevation {
106        /// Invalid target elevation in metres.
107        meters: f64,
108    },
109    /// The eye sits on or under the surface its horizon is measured on, where it has none.
110    #[error("the eye is {distance} radii from the globe center, on or below the surface")]
111    EyeBelowSurface {
112        /// Distance of the eye from the globe center in radii.
113        distance: f64,
114    },
115    /// The derived projection matrix cannot be inverted.
116    #[error("globe view-projection matrix is not invertible")]
117    NonInvertibleViewProjection,
118    /// Projection-data geometry is invalid.
119    #[error("failed to compute globe projection data")]
120    ProjectionData {
121        /// Underlying projection-data error.
122        #[source]
123        source: ProjectionDataError,
124    },
125}

127/// Immutable matrices and derived geometry for one globe camera configuration.

129pub struct GlobeCameraState {
130    options: GlobeCameraOptions,
131    projection: Matrix4<f64>,
132    inverse_projection: Matrix4<f64>,
133    view: Matrix4<f64>,
134    view_projection: Matrix4<f64>,
135    inverse_view_projection: Matrix4<f64>,
136    camera_position: Vector3<f64>,
137    clipping_plane: Vector4<f64>,
138    globe_radius_pixels: f64,
139    camera_to_center_distance: f64,
140    near_z: f64,
141    far_z: f64,
142    /// Whether the camera is an eye a host placed rather than one derived from map angles.
143    external_eye: bool,
144}

146impl GlobeCameraState {
147    /// Builds camera matrices and horizon geometry from validated options.
148    pub fn new(options: GlobeCameraOptions) -> Result<Self, GlobeCameraError> {
149        validate_options(options)?;
150        let camera_to_center_distance = camera_to_center_distance(options);
151        let radius = radius_pixels(options);
152        let target = target_scale(options)?;
153        // The line of sight to the horizon of a sphere scaled by `target` is shorter than the
154        // eye's distance to the globe center, and the tallest terrain beyond that horizon adds
155        // a small part of a radius, so this far plane holds everything in front of the eye.
156        let far_z = camera_to_center_distance + radius * 2.0 * target;
157        let projection = projection_matrix(options, far_z);
158        let inverse_projection = projection
159            .invert()
160            .ok_or(GlobeCameraError::NonInvertibleViewProjection)?;
161        let view = globe_view_matrix(options, camera_to_center_distance, radius, target);
162        let view_projection = projection * view;
163        let inverse_view_projection = view_projection
164            .invert()
165            .ok_or(GlobeCameraError::NonInvertibleViewProjection)?;
166        let camera_position = camera_position(options, camera_to_center_distance, radius, target);
167        // Below sea level, as in a depression, the eye can sit inside the sea-level sphere,
168        // which then hides nothing; the ground it stands over does.
169        let clipping_plane = horizon_plane(camera_position, target.min(1.0))?;
170
171        Ok(Self {
172            options,
173            projection,
174            inverse_projection,
175            view,
176            view_projection,
177            inverse_view_projection,
178            camera_position,
179            clipping_plane,
180            globe_radius_pixels: radius,
181            camera_to_center_distance,
182            near_z: NEAR_Z,
183            far_z,
184            external_eye: false,
185        })
186    }

292    /// Returns text-size correction for a tile-local anchor under globe projection.
293    pub fn pitched_text_correction(&self, _x: f64, y: f64, tile: TileCoords) -> f64 {
294        let tile_count = 2_f64.powi(i32::from(u8::from(tile.z)));
295        let mercator_y = (f64::from(tile.y) + y / EXTENT) / tile_count;
296        let (_, latitude) = mercator_to_angular_radians(0.0, mercator_y);
297        self.circle_radius_correction() / latitude.cos()
298    }

300    /// Transforms a map-local light direction into globe axes at the map center.
301    pub fn transform_light_direction(&self, direction: Vector3<f64>) -> Option<Vector3<f64>> {
302        let sphere = lat_lon_to_unit_sphere(self.options.center);
303        let right = Vector3::new(sphere.z, 0.0, -sphere.x);
304        if right.magnitude2() <= MIN_DIRECTION_LENGTH_SQUARED {
305            return None;
306        }
307        let right = right.normalize();
308        let down = right.cross(sphere).normalize();
309        let transformed = right * direction.x + down * direction.y + sphere * direction.z;
310        (transformed.magnitude2() > MIN_DIRECTION_LENGTH_SQUARED).then(|| transformed.normalize())
311    }

313    /// Transforms a globe-world direction into view axes without applying translation.
314    pub fn world_direction_to_view(&self, direction: Vector3<f64>) -> Option<Vector3<f64>> {
315        let linear = Matrix3::from_cols(
316            self.view.x.truncate(),
317            self.view.y.truncate(),
318            self.view.z.truncate(),
319        );
320        let transformed = linear * direction;
321        (transformed.magnitude2() > MIN_DIRECTION_LENGTH_SQUARED).then(|| transformed.normalize())
322    }

351    /// Projects a geographic location to viewport pixels.
352    pub fn location_to_screen(&self, location: LatLon, elevation_meters: f64) -> Point2<f64> {
353        let surface = lat_lon_to_unit_sphere(location);
354        let projected = self.project_surface_point(
355            elevate_surface_point(surface, elevation_meters, self.options.body),
356            surface,
357        );
358        Point2::new(
359            (projected.point.x * 0.5 + 0.5) * self.options.width,
360            (-projected.point.y * 0.5 + 0.5) * self.options.height,
361        )
362    }

364    /// Returns whether a geographic location is hidden behind the planet.
365    pub fn is_location_occluded(&self, location: LatLon) -> bool {

367    }
368
369    fn project_surface_point(
370        &self,
371        position: Vector3<f64>,
372        unelevated_surface: Vector3<f64>,
373    ) -> GlobePointProjection {
374        let clip = self.view_projection * position.extend(1.0);
375        GlobePointProjection {
376            point: Point2::new(clip.x / clip.w, clip.y / clip.w),
377            signed_distance_from_camera: clip.w,
378            is_occluded: !self.is_surface_point_visible(unelevated_surface),
379        }
380    }
381
382    fn is_surface_point_visible(&self, surface: Vector3<f64>) -> bool {
383        self.clipping_plane.truncate().dot(surface) + self.clipping_plane.w >= 0.0
384    }

395/// The orbit target's distance from the globe center in radii.
396fn target_scale(options: GlobeCameraOptions) -> Result<f64, GlobeCameraError> {
397    let scale = 1.0 + options.target_elevation_meters / options.body.radius_meters;
398    if scale.is_finite() && scale > 0.0 {
399        Ok(scale)
400    } else {
401        Err(GlobeCameraError::InvalidTargetElevation {
402            meters: options.target_elevation_meters,
403        })
404    }
405}

407/// The plane through the points where lines from `eye` touch the sphere of `radius` about
408/// the globe center: the points `P` of that sphere with `dot(eye, P) = radius²`, normalised so
409/// the visible side is positive. On the unit sphere this is `dot(eye, P) = 1`.
410pub(super) fn horizon_plane(
411    eye: Vector3<f64>,
412    radius: f64,
413) -> Result<Vector4<f64>, GlobeCameraError> {
414    let distance = eye.magnitude();
415    if distance.is_nan() || distance <= radius {
416        return Err(GlobeCameraError::EyeBelowSurface { distance });
417    }
418    Ok((eye / distance).extend(-radius * radius / distance))
419}

421fn camera_to_center_distance(options: GlobeCameraOptions) -> f64 {
422    (options.height * 0.5) / (options.field_of_view_degrees.to_radians() * 0.5).tan()
423}
424
425fn projection_matrix(options: GlobeCameraOptions, far_z: f64) -> Matrix4<f64> {
426    let mut projection = perspective(
427        Deg(options.field_of_view_degrees),
428        options.width / options.height,
429        NEAR_Z,
430        far_z,
431    );
432    projection.z.x = -options.center_offset.x * 2.0 / options.width;
433    projection.z.y = options.center_offset.y * 2.0 / options.height;
434    projection
435}

437fn globe_view_matrix(
438    options: GlobeCameraOptions,
439    camera_distance: f64,
440    radius: f64,
441    target: f64,
442) -> Matrix4<f64> {
443    Matrix4::from_translation(Vector3::new(0.0, 0.0, -camera_distance))
444        * Matrix4::from_angle_z(Deg(options.roll_degrees))
445        * Matrix4::from_angle_x(Deg(-options.pitch_degrees))
446        * Matrix4::from_angle_z(Deg(options.bearing_degrees))
447        * Matrix4::from_translation(Vector3::new(0.0, 0.0, -radius * target))
448        * Matrix4::from_angle_x(Deg(options.center.latitude))
449        * Matrix4::from_angle_y(Deg(-options.center.longitude))
450        * Matrix4::from_scale(radius)
451}
452
453fn camera_position(
454    options: GlobeCameraOptions,
455    camera_distance: f64,
456    radius: f64,
457    target: f64,
458) -> Vector3<f64> {
459    let mut position = Vector3::new(0.0, 0.0, camera_distance / radius);
460    position = Matrix3::from_angle_z(Deg(-options.roll_degrees)) * position;
461    position = Matrix3::from_angle_x(Deg(options.pitch_degrees)) * position;
462    position = Matrix3::from_angle_z(Deg(-options.bearing_degrees)) * position;
463    position += Vector3::unit_z() * target;
464    position = Matrix3::from_angle_x(Deg(-options.center.latitude)) * position;
465    Matrix3::from_angle_y(Rad(options.center.longitude.to_radians())) * position
466}
