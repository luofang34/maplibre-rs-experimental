1//! Visible-tile traversal for the flat Mercator projection with frustum culling and LOD.
2//!
3//! Mirrors GL JS `coveringTiles` with its Mercator details provider: tiles are axis-aligned
4//! boxes spanning world pixels horizontally and metres of elevation vertically, the space the
5//! view projection unprojects into, tested against the camera frustum; far tiles get a lower
6//! zoom from the same distance rule as the globe.
7

23/// Inputs controlling Mercator tile selection.
24#[derive(Clone, Copy, Debug)]
25pub struct MercatorCoveringOptions {
26    /// Canonical zoom level to select when zoom does not vary per tile.
27    pub zoom: ZoomLevel,
28    /// Fractional map zoom used by the per-tile LOD calculation.
29    pub requested_zoom: f64,
30    /// Lowers the zoom of distant tiles, as GL JS does for pitched or terrain views.
31    pub variable_zoom: bool,
32    /// How the per-tile zoom becomes a level.
33    pub rounding: ZoomRounding,
34    /// Levels the source serves.
35    pub zoom_range: SourceZoomRange,
36    /// How many copies of the world to either side to cover; GL JS draws three.
37    pub world_copies: i32,
38    /// Number of canonical neighbors to add around visible tiles.
39    pub padding: i32,
40    /// Maximum number of returned tiles after padding.
41    pub max_tiles: usize,
42}

44/// Failure while selecting visible Mercator tiles.
45#[derive(Clone, Copy, Debug, Error, PartialEq, Eq)]
46pub enum MercatorCoveringError {
47    /// The current camera cannot produce finite frustum coordinates.
48    #[error("cannot unproject the Mercator covering frustum")]
49    ViewProjection {
50        /// Invalid projection supplied by the view.
51        #[from]
52        source: crate::render::camera::ViewProjectionError,
53    },
54    /// The coordinate model only supports canonical zoom levels through 31.
55    #[error("mercator covering zoom {zoom} exceeds the supported maximum")]
56    UnsupportedZoom {
57        /// Unsupported zoom level.
58        zoom: u8,
59    },
60}

62/// Selects canonical tiles whose elevated box intersects the camera frustum.
63///
64/// `elevation` supplies the height of each tile's box.
65pub fn covering_tiles(
66    view_state: &ViewState,
67    options: MercatorCoveringOptions,
68    elevation: &dyn TileElevationProvider,
69) -> Result<Vec<WorldTileCoords>, MercatorCoveringError> {
70    covering_tiles_with_history(view_state, options, elevation, None)
71}

73pub(crate) fn covering_tiles_with_history(
74    view_state: &ViewState,
75    options: MercatorCoveringOptions,
76    elevation: &dyn TileElevationProvider,
77    history: Option<&crate::projection::lod_history::LodHistory>,
78) -> Result<Vec<WorldTileCoords>, MercatorCoveringError> {
79    if usize::from(u8::from(options.zoom)) >= MAX_ZOOM {
80        return Err(MercatorCoveringError::UnsupportedZoom {
81            zoom: u8::from(options.zoom),
82        });
83    }
84    let frustum = GlobeFrustum::from_points_oriented(view_state.frustum_corners()?);
85    let world_size = TILE_SIZE * 2_f64.powf(view_state.zoom().value());
86    let center = view_state.camera().position();
87    let eye = view_state.eye_position();
88    let lod = lod_context(view_state, options.requested_zoom, world_size);
89    let priority = if view_state.has_external_view() {
90        Point2::new(eye.x, eye.y)
91    } else {
92        center
93    };
94    let priority = mercator_world_to_lat_lon(priority.x, priority.y, world_size);
95    let inspect = |placed: WorldTileCoords, fully_visible| -> Result<_, MercatorCoveringError> {
96        let Some((canonical, wrap)) = placed.wrapped() else {
97            return Ok(None);
98        };
99        let tile = TileCoords::from((canonical.x as u32, canonical.y as u32, canonical.z));
100        let intersection = if fully_visible {
101            Intersection::Full
102        } else {
103            tile_intersection(&frustum, (tile, wrap), world_size, elevation)
104        };
105        if intersection == Intersection::None {
106            return Ok(None);
107        }
108        let target = options.zoom_range.cap(if options.variable_zoom {
109            lod.shifted(f64::from(wrap))
110                .stable_zoom_for_tile(tile, options.rounding, history)
111        } else {
112            options.zoom
113        });
114        Ok(Some(crate::projection::tile_covering::Refinement {
115            target,
116            fully_visible: intersection == Intersection::Full,
117        }))
118    };
119    let mut visible = if view_state.has_external_view() {
120        crate::projection::tile_covering::bounded(
121            options.max_tiles,
122            options.zoom_range.min,
123            priority,
124            inspect,
125        )?
126    } else {
127        unbounded_covering(options.zoom_range.min, options.world_copies, inspect)?
128    };
129    sort_by_center(&mut visible, priority);
130    Ok(add_padding(visible, options.padding, options.max_tiles))
131}

140fn lod_context(view_state: &ViewState, requested_zoom: f64, world_size: f64) -> LodContext {
141    // The camera position comes from the view transform itself, so the distance rule sees
142    // the camera where the frustum has it under any bearing.
143    let eye = view_state.eye_position();
144    let center = view_state.camera().position();
145    view_state
146        .eye_lod_context(requested_zoom)
147        .unwrap_or_else(|| {
148            LodContext::from_positions(
149                Point2::new(eye.x / world_size, eye.y / world_size),
150                Point2::new(center.x / world_size, center.y / world_size),
151                (eye.z - view_state.center_elevation()) * view_state.pixels_per_meter()
152                    / world_size,
153                view_state.field_of_view().0.to_degrees(),
154                requested_zoom,
155            )
156            .planar()
157        })
158}

160fn tile_intersection(
161    frustum: &GlobeFrustum,
162    (tile, wrap): (TileCoords, i32),
163    world_size: f64,
164    elevation: &dyn TileElevationProvider,
165) -> Intersection {
166    let count = 2_f64.powi(i32::from(u8::from(tile.z)));
167    let size = world_size / count;
168    let range = elevation.elevation_range(tile);
169    let min = Vector3::new(
170        (f64::from(tile.x) + f64::from(wrap) * count) * size,
171        f64::from(tile.y) * size,
172        range.min_meters.min(0.0),
173    );
174    let max = Vector3::new(min.x + size, min.y + size, range.max_meters.max(0.0));
175    frustum.intersects(&aabb_volume(min, max))
176}
