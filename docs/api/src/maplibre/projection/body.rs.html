1//! The body the map is drawn on. Earth by default; the radius is the figure the globe
2//! projection, terrain and the atmosphere need, and the height of the highest ground bounds
3//! terrain no elevation data has described yet, so a second body only changes them here.
4
5/// A spherical body the map projects onto.
6#[derive(Clone, Copy, Debug, PartialEq)]
7pub struct Body {
8    /// Mean radius in metres.
9    pub radius_meters: f64,
10    /// Height in metres of the body's highest ground above the mean radius: where no
11    /// elevation data has loaded, ground anywhere below it may hide what lies beyond.
12    pub highest_ground_meters: f64,
13}
14
15impl Body {
16    /// Earth, with the mean radius GL JS uses.
17    pub const EARTH: Self = Self {
18        radius_meters: 6_371_008.8,
19        highest_ground_meters: 8_849.0,
20    };
21
22    /// Metres around the equator, the length one Mercator unit covers there.
23    pub fn circumference_meters(self) -> f64 {
24        2.0 * std::f64::consts::PI * self.radius_meters
25    }
26
27    /// Metres around the parallel at a latitude.
28    pub fn circumference_at_latitude(self, latitude_degrees: f64) -> f64 {
29        self.circumference_meters() * latitude_degrees.to_radians().cos()
30    }
31
32    /// Radius of the unit sphere at an elevation in metres above the surface.
33    pub fn unit_radius_at(self, elevation_meters: f64) -> f64 {
34        1.0 + elevation_meters / self.radius_meters
35    }
36}
37
38impl Default for Body {
39    fn default() -> Self {
40        Self::EARTH
41    }
42}
