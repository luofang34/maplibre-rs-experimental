13#[derive(Clone, Copy, Debug, PartialEq)]
14pub(crate) enum CoveringFocus {
15    /// Unit-sphere focus vector `f_hat in S^2` for 3D Globe covering (`1.0 - c_hat.dot(f_hat)`).
16    Spherical(Vector3<f64>),
17    /// Planar Web Mercator focus in `0..1` world units (preserving `wrap = -1, 0, +1` copies).
18    Planar(Point2<f64>),
19}
20
21pub(crate) struct Refinement {
22    pub(crate) target: ZoomLevel,
23    pub(crate) fully_visible: bool,
24}
25
26struct Candidate {
27    tile: WorldTileCoords,
28    refinement: Refinement,
29    distance: f64,
30    rank: usize,
31}
32
33impl Candidate {
34    fn priority(&self) -> u8 {
35        u8::from(self.refinement.target).saturating_sub(u8::from(self.tile.z))
36    }
37}
38
39impl PartialEq for Candidate {
40    fn eq(&self, other: &Self) -> bool {
41        self.cmp(other) == Ordering::Equal
42    }
43}
44impl Eq for Candidate {}
45impl PartialOrd for Candidate {
46    fn partial_cmp(&self, other: &Self) -> Option<Ordering> {
47        Some(self.cmp(other))
48    }
49}
50impl Ord for Candidate {
51    fn cmp(&self, other: &Self) -> Ordering {
52        self.rank
53            .cmp(&other.rank)
54            .then_with(|| other.distance.total_cmp(&self.distance))
55            .then_with(|| self.tile.cmp(&other.tile))
56    }
57}

59/// Refine the most undersampled visible globe tile first on `S^2` (`1.0 - c_hat.dot(focus_unit)`).
60/// A split replaces its parent only when all visible children fit, so the budget cannot cut holes.
61pub(crate) fn bounded_spherical<E>(
62    limit: usize,
63    min_zoom: u8,
64    focus_unit: Vector3<f64>,
65    mut inspect: impl FnMut(WorldTileCoords, bool) -> Result<Option<Refinement>, E>,
66) -> Result<Vec<WorldTileCoords>, E> {
67    bounded_by(
68        limit,
69        min_zoom,
70        0,
71        CoveringFocus::Spherical(focus_unit),
72        &mut inspect,
73        |tile, refinement| {
74            usize::from(u8::from(refinement.target).saturating_sub(u8::from(tile.z)))
75        },
76    )
77}

79/// Refine the most undersampled visible planar Mercator tile first, keeping distinct 2D world-copy
80/// distances across `[-world_copies..=world_copies]`.
81pub(crate) fn bounded_planar<E>(
82    limit: usize,
83    min_zoom: u8,
84    world_copies: i32,
85    focus_2d: Point2<f64>,
86    mut inspect: impl FnMut(WorldTileCoords, bool) -> Result<Option<Refinement>, E>,
87) -> Result<Vec<WorldTileCoords>, E> {
88    bounded_by(
89        limit,
90        min_zoom,
91        world_copies.max(0),
92        CoveringFocus::Planar(focus_2d),
93        &mut inspect,
94        |tile, refinement| {
95            usize::from(u8::from(refinement.target).saturating_sub(u8::from(tile.z)))
96        },
97    )
98}

100#[cfg(test)]
101pub(crate) fn bounded<E>(
102    limit: usize,
103    min_zoom: u8,
104    center: LatLon,
105    inspect: impl FnMut(WorldTileCoords, bool) -> Result<Option<Refinement>, E>,
106) -> Result<Vec<WorldTileCoords>, E> {
107    bounded_spherical(
108        limit,
109        min_zoom,
110        crate::projection::globe::lat_lon_to_unit_sphere(center),
111        inspect,
112    )
113}

115fn bounded_by<E>(
116    limit: usize,
117    min_zoom: u8,
118    world_copies: i32,
119    focus: CoveringFocus,
120    mut inspect: impl FnMut(WorldTileCoords, bool) -> Result<Option<Refinement>, E>,
121    rank: impl Fn(WorldTileCoords, &Refinement) -> usize,
122) -> Result<Vec<WorldTileCoords>, E> {
123    if limit == 0 {
124        return Ok(Vec::new());
125    }
126    let mut roots = Vec::new();
127    for wrap in -world_copies..=world_copies {
128        let root = WorldTileCoords::from((wrap, 0, ZoomLevel::new(0)));
129        if let Some(refinement) = inspect(root, false)? {
130            roots.push(candidate(root, refinement, focus, &rank));
131        }
132    }
133    if roots.is_empty() {
134        return Ok(Vec::new());
135    }
136    roots.sort_by(|a, b| b.cmp(a));
137    let mut pending: BinaryHeap<_> = roots.into_iter().take(limit).collect();
138    let mut visible = Vec::new();
139    // Conservative bounds may expose parents whose children all cull away. Count those
140    // splits too, so freeing a frontier slot cannot turn an empty view into unbounded work.
141    let mut remaining_splits = limit.saturating_mul(32);
142    while let Some(parent) = pending.pop() {
143        if parent.priority() == 0 || remaining_splits == 0 {
144            visible.push(parent.tile);
145            continue;
146        }
147        remaining_splits -= 1;
148        let mut children = Vec::with_capacity(4);
149        for tile in parent.tile.get_children() {
150            if let Some(refinement) = inspect(tile, parent.refinement.fully_visible)? {
151                children.push(candidate(tile, refinement, focus, &rank));
152            }
153        }
154        if visible.len() + pending.len() + children.len() <= limit {
155            pending.extend(children);
156        } else {
157            visible.push(parent.tile);
158        }
159    }
160    Ok(visible
161        .into_iter()
162        .filter(|tile| u8::from(tile.z) >= min_zoom)
163        .collect())
164}

166/// Singularity-free `S^2` angular priority key `1.0 - c_hat.dot(focus_unit)` in `[0, 2]`.
167pub(crate) fn spherical_tile_distance(tile: WorldTileCoords, focus_unit: Vector3<f64>) -> f64 {
168    let count = 1_i64 << u32::from(u8::from(tile.z));
169    let canonical = TileCoords::from((
170        i64::from(tile.x).rem_euclid(count) as u32,
171        i64::from(tile.y).clamp(0, count - 1) as u32,
172        tile.z,
173    ));
174    let cap = crate::projection::globe::covering::tile_spherical_cap(canonical);
175    1.0 - cap.center.dot(focus_unit)
176}

178/// Planar 2D Web Mercator squared distance in world-copy coordinates (no wrap folding).
179pub(crate) fn planar_tile_distance_squared(tile: WorldTileCoords, focus_2d: Point2<f64>) -> f64 {
180    let count = 2_f64.powi(i32::from(u8::from(tile.z)));
181    let dx = focus_2d.x - (f64::from(tile.x) + 0.5) / count;
182    let dy = focus_2d.y - (f64::from(tile.y) + 0.5) / count;
183    dx * dx + dy * dy
184}

186fn candidate(
187    tile: WorldTileCoords,
188    refinement: Refinement,
189    focus: CoveringFocus,
190    rank: &impl Fn(WorldTileCoords, &Refinement) -> usize,
191) -> Candidate {
192    let distance = match focus {
193        CoveringFocus::Spherical(focus_unit) => spherical_tile_distance(tile, focus_unit),
194        CoveringFocus::Planar(focus_2d) => planar_tile_distance_squared(tile, focus_2d),
195    };
196    Candidate {
197        rank: rank(tile, &refinement),
198        tile,
199        refinement,
200        distance,
201    }
202}

204/// Fits a covering to the drape budget by refining the largest remaining detail deficit.
205/// Distance breaks ties; a single nearby tile cannot consume every refinement slot.
206#[cfg(test)]
207pub(crate) fn coarsen(
208    tiles: impl Iterator<Item = WorldTileCoords>,
209    limit: usize,
210    min_zoom: u8,
211) -> Vec<WorldTileCoords> {
212    coarsen_with_focus(tiles, limit, min_zoom, None)
213}

215pub(crate) fn coarsen_with_focus(
216    tiles: impl Iterator<Item = WorldTileCoords>,
217    limit: usize,
218    min_zoom: u8,
219    focus: Option<CoveringFocus>,
220) -> Vec<WorldTileCoords> {
221    let tiles: Vec<_> = tiles.collect();
222    if tiles.len() <= limit {
223        return tiles;
224    }
225    let mut ancestors = HashMap::new();
226    let mut world_copies = 0_i32;
227    for (index, tile) in tiles.iter().enumerate() {
228        let mut ancestor = Some(*tile);
229        while let Some(coords) = ancestor {
230            if u8::from(coords.z) == 0 {
231                world_copies = world_copies.max(coords.x.abs());
232            }
233            let entry = ancestors.entry(coords).or_insert((tile.z, index));
234            entry.0 = entry.0.max(tile.z);
235            ancestor = coords.get_parent();
236        }
237    }
238    let focus = focus.unwrap_or_else(|| {
239        let first = tiles[0];
240        if world_copies == 0 {
241            let count = 1_i64 << u32::from(u8::from(first.z));
242            let canonical = TileCoords::from((
243                i64::from(first.x).rem_euclid(count) as u32,
244                i64::from(first.y).clamp(0, count - 1) as u32,
245                first.z,
246            ));
247            CoveringFocus::Spherical(
248                crate::projection::globe::covering::tile_spherical_cap(canonical).center,
249            )
250        } else {
251            let count = 2_f64.powi(i32::from(u8::from(first.z)));
252            CoveringFocus::Planar(Point2::new(
253                (f64::from(first.x) + 0.5) / count,
254                (f64::from(first.y) + 0.5) / count,
255            ))
256        }
257    });
258    let result: Result<_, std::convert::Infallible> = bounded_by(
259        limit,
260        min_zoom,
261        world_copies,
262        focus,
263        |tile, _| {
264            Ok(ancestors.get(&tile).map(|(target, _)| Refinement {
265                target: *target,
266                fully_visible: false,
267            }))
268        },
269        |tile, refinement| {
270            usize::from(u8::from(refinement.target).saturating_sub(u8::from(tile.z)))
271        },
272    );
273    let mut result = match result {
274        Ok(tiles) => tiles,
275        Err(never) => match never {},
276    };
277    result.sort_by_key(|tile| (ancestors.get(tile).map(|(_, index)| *index), *tile));
278    result
279}

281pub(crate) fn covers(parent: WorldTileCoords, tile: WorldTileCoords) -> bool {
282    let delta = i32::from(u8::from(tile.z)) - i32::from(u8::from(parent.z));
283    delta >= 0 && (tile.x >> delta) == parent.x && (tile.y >> delta) == parent.y
284}
