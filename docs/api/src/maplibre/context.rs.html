18/// Mutable frame state shared by scheduled systems, independent of host service types.
19pub struct MapContext {
20    /// Style evaluated by rendering and tile request systems.
21    pub style: Style,
22    /// Loaded tile components and shared system resources.
23    pub world: World,
24    /// Camera, viewport and projection inputs for the current frame.
25    pub view_state: ViewState,
26    /// GPU state and render graph used to draw this map.
27    pub renderer: Renderer,
28}
29
30impl MapContext {
31    /// The time of the frame being rendered, as the host wrote it; zero until it does.
32    pub fn frame_time(&self) -> std::time::Duration {
33        crate::render::frame_input::frame_time(&self.world)
34    }
35
36    /// The credits of the sources drawn at the camera's zoom, for a host's credits panel; see
37    /// [`Style::attributions`].
38    pub fn attributions(&self) -> Vec<String> {
39        self.style.attributions(self.view_state.zoom().value())
40    }
41
42    /// Updates the logical viewport and resizes the physical presentation surface.
43    /// `scale_factor` is the number of physical pixels per logical unit.
44    /// Logical dimensions remain at least one unit even below the device scale.
45    pub fn resize(&mut self, size: PhysicalSize, scale_factor: f64) {
46        self.view_state.resize(size.to_logical(scale_factor));
47        crate::sdf::provided::set_pixel_ratio(&mut self.world, scale_factor);
48        self.renderer.resize_surface(size);
49        // A new surface holds nothing, even when the logical size and so the camera are unchanged.
50        crate::render::frame_signals::mark_dirty(&mut self.world);
51    }
52
53    /// Sets the direction towards the sun in ECEF axes, which drives the atmosphere in place
54    /// of the style's root `light` and darkens the terrain's night side; `None` returns the
55    /// atmosphere to the style's light and lights all terrain as day.
56    pub fn set_sun_direction(&mut self, sun: Option<crate::celestial::EcefDirection>) {
57        if crate::celestial::sun_direction(&self.world) != sun {
58            crate::celestial::set_sun_direction(&mut self.world, sun);
59            crate::render::frame_signals::mark_dirty(&mut self.world);
60        }
61    }
62
63    /// Sets the solar elevations over which the ground turns from night to day while a sun
64    /// direction is set.
65    pub fn set_twilight_band(&mut self, twilight: crate::celestial::TwilightBand) {
66        if crate::celestial::lighting(&self.world).twilight != twilight {
67            crate::celestial::set_twilight_band(&mut self.world, twilight);
68            crate::render::frame_signals::mark_dirty(&mut self.world);
69        }
70    }
71
72    /// Draws the globe's atmosphere in full, lit by the sun direction while one is set, even
73    /// when the style has no `sky`; `false` leaves the atmosphere to the style's `sky`.
74    pub fn set_atmosphere(&mut self, atmosphere: bool) {
75        if crate::celestial::lighting(&self.world).atmosphere != atmosphere {
76            crate::celestial::set_atmosphere(&mut self.world, atmosphere);
77            crate::render::frame_signals::mark_dirty(&mut self.world);
78        }
79    }
80
81    /// Replaces the stars drawn behind a globe; the starfield plugin draws them.
82    pub fn set_stars(&mut self, stars: Vec<crate::celestial::Star>) {
83        crate::celestial::set_stars(&mut self.world, stars);
84        crate::render::frame_signals::mark_dirty(&mut self.world);
85    }
86
87    /// Sets or clears the sun's angular radius. The disc stands where
88    /// [`Self::set_sun_direction`] points, with a glare that follows how much of it shows; with no
89    /// sun direction it is not drawn. The starfield plugin draws it.
90    pub fn set_sun_angular_radius(&mut self, radius: Option<crate::celestial::AngularRadius>) {
91        if crate::celestial::discs(&self.world).sun_radius != radius {
92            crate::celestial::set_sun_angular_radius(&mut self.world, radius);
93            crate::render::frame_signals::mark_dirty(&mut self.world);
94        }
95    }
96
97    /// Sets or clears the moon's disc. Its direction is the one seen from the camera's eye: the
98    /// moon is near enough that the Earth's centre sees it elsewhere. Its phase follows
99    /// [`Self::set_sun_direction`]; with no sun set, the whole disc is drawn lit, a fallback
100    /// rather than a phase. The starfield plugin draws it.
101    pub fn set_moon_disc(&mut self, moon: Option<crate::celestial::Disc>) {
102        if crate::celestial::discs(&self.world).moon != moon {
103            crate::celestial::set_moon_disc(&mut self.world, moon);
104            crate::render::frame_signals::mark_dirty(&mut self.world);
105        }
106    }
107
108    /// Sets or clears the Earth's shadow the moon is lit through during a lunar eclipse, as
109    /// the host computes it: the moon dims and turns copper in the umbra, as dark as the
110    /// shadow's Danjon value, and its light on the night side dims with it.
111    pub fn set_earth_shadow(&mut self, shadow: Option<crate::celestial::EarthShadow>) {
112        if crate::celestial::discs(&self.world).earth_shadow != shadow {
113            crate::celestial::set_earth_shadow(&mut self.world, shadow);
114            crate::render::frame_signals::mark_dirty(&mut self.world);
115        }
116    }
117
118    /// The sun direction set with [`Self::set_sun_direction`].
119    pub fn sun_direction(&self) -> Option<crate::celestial::EcefDirection> {
120        crate::celestial::sun_direction(&self.world)
121    }
122
123    /// Switches how the camera navigates the globe, for the style's projection: free-globe
124    /// navigation needs the pure `vertical-perspective` projection. A center held by
125    /// terrain stands on the loaded ground under it at once, as
126    /// [`crate::terrain::elevation::ground_center`] stands it.
127    pub fn set_navigation_mode(
128        &mut self,
129        mode: crate::render::view_state::NavigationMode,
130    ) -> Result<(), crate::render::view_state::NavigationError> {
131        let projection = self
132            .style
133            .projection
134            .as_ref()
135            .map_or_else(Default::default, |projection| {
136                projection.projection_type.clone()
137            });
138        self.view_state.set_navigation_mode(mode, &projection)?;
139        crate::terrain::elevation::ground_center(&mut self.view_state, &self.world);
140        Ok(())
141    }
142
143    /// Moves the map at once, as GL JS `jumpTo`, in either navigation mode, as
144    /// [`ViewState::jump_to`] moves it. A center held by terrain stands on the loaded ground
145    /// under it at once, or at sea level where none has loaded, never on the ground it left.
146    pub fn jump_to(
147        &mut self,
148        options: crate::render::view_state::CameraOptions,
149    ) -> Result<(), crate::render::view_state::NavigationError> {
150        self.view_state.jump_to(options)?;
151        crate::terrain::elevation::ground_center(&mut self.view_state, &self.world);
152        Ok(())
153    }
154
155    /// Restores a stored free-globe pose, for the style's projection: it must be the pure
156    /// `vertical-perspective` projection. A center held by terrain stands on the loaded
157    /// ground under the pose's target at once, rather than at the pose's own target height, as
158    /// [`crate::terrain::elevation::ground_center`] stands it.
159    pub fn restore_globe_pose(
160        &mut self,
161        pose: crate::render::view_state::GlobePose,
162    ) -> Result<(), crate::render::view_state::NavigationError> {
163        let projection = self
164            .style
165            .projection
166            .as_ref()
167            .map_or_else(Default::default, |projection| {
168                projection.projection_type.clone()
169            });
170        self.view_state.restore_globe_pose(pose, &projection)?;
171        crate::terrain::elevation::ground_center(&mut self.view_state, &self.world);
172        Ok(())
173    }
174}
