1//! Style, tile storage, view and renderer state for a map frame.
2
3#![deny(missing_docs)]
4
5use crate::{
6    render::{view_state::ViewState, Renderer},
7    style::Style,
8    tcs::world::World,
9    window::PhysicalSize,
10};
11
12mod ground;
13
14/// Mutable frame state shared by scheduled systems, independent of host service types.
15pub struct MapContext {
16    /// Style evaluated by rendering and tile request systems.
17    pub style: Style,
18    /// Loaded tile components and shared system resources.
19    pub world: World,
20    /// Camera, viewport and projection inputs for the current frame.
21    pub view_state: ViewState,
22    /// GPU state and render graph used to draw this map.
23    pub renderer: Renderer,
24}
25
26impl MapContext {
27    /// The time of the frame being rendered, as the host wrote it; zero until it does.
28    pub fn frame_time(&self) -> std::time::Duration {
29        crate::render::frame_input::frame_time(&self.world)
30    }
31
32    /// Updates the logical viewport and resizes the physical presentation surface.
33    /// `scale_factor` is the number of physical pixels per logical unit.
34    /// Logical dimensions remain at least one unit even below the device scale.
35    pub fn resize(&mut self, size: PhysicalSize, scale_factor: f64) {
36        self.view_state.resize(size.to_logical(scale_factor));
37        crate::sdf::provided::set_pixel_ratio(&mut self.world, scale_factor);
38        self.renderer.resize_surface(size);
39        // A new surface holds nothing, even when the logical size and so the camera are unchanged.
40        crate::render::frame_signals::mark_dirty(&mut self.world);
41    }
42
43    /// Switches how the camera navigates the globe, for the style's projection: free-globe
44    /// navigation needs the pure `vertical-perspective` projection. A center held by
45    /// terrain stands on the loaded ground under it at once, as
46    /// [`crate::terrain::elevation::ground_center`] stands it.
47    pub fn set_navigation_mode(
48        &mut self,
49        mode: crate::render::view_state::NavigationMode,
50    ) -> Result<(), crate::render::view_state::NavigationError> {
51        let projection = self
52            .style
53            .projection
54            .as_ref()
55            .map_or_else(Default::default, |projection| {
56                projection.projection_type.clone()
57            });
58        self.view_state.set_navigation_mode(mode, &projection)?;
59        crate::terrain::elevation::ground_center(&mut self.view_state, &self.world);
60        Ok(())
61    }
62
63    /// Moves the map at once, as GL JS `jumpTo`, in either navigation mode, as
64    /// [`ViewState::jump_to`] moves it. A center held by terrain stands on the loaded ground
65    /// under it at once, or at sea level where none has loaded, never on the ground it left.
66    pub fn jump_to(
67        &mut self,
68        options: crate::render::view_state::CameraOptions,
69    ) -> Result<(), crate::render::view_state::NavigationError> {
70        self.view_state.jump_to(options)?;
71        crate::terrain::elevation::ground_center(&mut self.view_state, &self.world);
72        Ok(())
73    }
74
75    /// Restores a stored free-globe pose, for the style's projection: it must be the pure
76    /// `vertical-perspective` projection. A center held by terrain stands on the loaded
77    /// ground under the pose's target at once, rather than at the pose's own target height, as
78    /// [`crate::terrain::elevation::ground_center`] stands it.
79    pub fn restore_globe_pose(
80        &mut self,
81        pose: crate::render::view_state::GlobePose,
82    ) -> Result<(), crate::render::view_state::NavigationError> {
83        let projection = self
84            .style
85            .projection
86            .as_ref()
87            .map_or_else(Default::default, |projection| {
88                projection.projection_type.clone()
89            });
90        self.view_state.restore_globe_pose(pose, &projection)?;
91        crate::terrain::elevation::ground_center(&mut self.view_state, &self.world);
92        Ok(())
93    }
94}
