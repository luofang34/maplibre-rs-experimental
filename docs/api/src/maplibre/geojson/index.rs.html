1//! A GeoJSON document parsed once into a spatial index, tiled into vector tile bytes on demand.
2//!
3//! Tiles are cut from the index instead of re-reading the document, so a source is parsed once
4//! per generation however many tiles are requested. Each tile carries its features' geometry
5//! cut to the tile and its buffer, unsimplified.
6

17/// Vector tile grid the features are quantized to.
18pub const EXTENT: u32 = 4096;
19/// Distance beyond the tile edge whose features are included, in tile grid units (GL JS default).
20const BUFFER: f64 = 128.0;
21/// Mercator latitude limit in degrees.
22const MAX_LATITUDE: f64 = 85.051_128_78;
23
24/// Why a document could not be indexed.
25#[derive(Debug, Error)]
26pub enum GeoJsonError {
27    /// The text is not JSON.
28    #[error("GeoJSON is not valid JSON")]
29    Json(#[source] serde_json::Error),
30    /// The document is neither a feature collection, a feature nor a geometry.
31    #[error("GeoJSON root has unsupported type `{found}`")]
32    Root {
33        /// The `type` member found, or an empty string.
34        found: String,
35    },
36    /// A feature's geometry cannot be read.
37    #[error("GeoJSON feature {index} has an invalid geometry: {reason}")]
38    Geometry {
39        /// Position of the feature in the document.
40        index: usize,
41        /// What is wrong with it.
42        reason: &'static str,
43    },
44}

46type Point = [f64; 2];
47type Ring = Vec<Point>;

49/// Points, lines and polygons of one feature in world coordinates, where the world is `0..1`.
50#[derive(Default, Clone)]
51struct Geometry {
52    points: Vec<Point>,
53    lines: Vec<Ring>,
54    polygons: Vec<Vec<Ring>>,
55}

57#[derive(Clone)]
58struct IndexedFeature {
59    id: Option<u64>,
60    geometry: Geometry,
61    properties: Vec<(String, Value)>,
62}

64struct Entry {
65    bounds: [f64; 4],
66    feature: u32,
67}

69impl RTreeObject for Entry {

72    fn envelope(&self) -> Self::Envelope {
73        AABB::from_corners(
74            [self.bounds[0], self.bounds[1]],
75            [self.bounds[2], self.bounds[3]],
76        )
77    }

80/// Features of one GeoJSON document, findable by the tile they fall in.
81pub struct GeoJsonIndex {
82    features: Vec<IndexedFeature>,
83    /// The points of a clustering source, and how they group at each zoom.
84    clustered: Option<(Vec<IndexedFeature>, Clusters)>,
85    /// Whether the source filters its features before they are tiled.
86    filtered: bool,
87    tree: RTree<Entry>,
88    approximate_bytes: usize,
89}

91impl GeoJsonIndex {
92    /// Indexes the document text a source URL returned.
93    pub fn from_text(text: &[u8], source: &GeoJsonSource) -> Result<Self, GeoJsonError> {
94        let value = serde_json::from_slice(text).map_err(GeoJsonError::Json)?;
95        Self::from_value(&value, source)
96    }

98    /// Indexes an inline document, applying the source's id options.
99    pub fn from_value(document: &Value, source: &GeoJsonSource) -> Result<Self, GeoJsonError> {
100        let promote = promoted_property(source.promote_id.as_ref());
101        let mut features = Vec::new();
102        let mut entries = Vec::new();
103        let mut bytes = 0;
104        let mut position = 0_usize;
105        let filter = source_filter(source);
106        let filter_declared = filter.is_some();
107        let mut points = Vec::new();
108        let mut add = |feature: &Value| -> Result<(), GeoJsonError> {
109            let index = position;
110            position += 1;
111            if filter
112                .as_ref()
113                .is_some_and(|filter| !feature_passes(feature, filter, 0.0))
114            {
115                return Ok(());
116            }
117            let Some(geometry) = feature.get("geometry").filter(|value| !value.is_null()) else {
118                return Ok(());
119            };
120            let mut parsed = Geometry::default();
121            read_geometry(geometry, &mut parsed)
122                .map_err(|reason| GeoJsonError::Geometry { index, reason })?;
123            let Some(bounds) = bounds_of(&parsed) else {
124                return Ok(());
125            };
126            let properties: Vec<(String, Value)> = feature
127                .get("properties")
128                .and_then(Value::as_object)
129                .map(|map| map.iter().map(|(k, v)| (k.clone(), v.clone())).collect())
130                .unwrap_or_default();
131            let id = numeric_feature_id(feature, promote, source.generate_id.then_some(index));
132            if source.cluster && is_single_point(geometry, &parsed) {
133                points.push(IndexedFeature {
134                    id,
135                    geometry: parsed,
136                    properties,
137                });
138                return Ok(());
139            }
140            bytes += coordinate_count(&parsed) * 16
141                + properties
142                    .iter()
143                    .map(|(key, value)| key.len() + value.to_string().len())
144                    .sum::<usize>()
145                + 64;
146            entries.push(Entry {
147                bounds,
148                feature: features.len() as u32,
149            });
150            features.push(IndexedFeature {
151                id,
152                geometry: parsed,
153                properties,
154            });
155            Ok(())
156        };
157        match document.get("type").and_then(Value::as_str) {
158            Some("FeatureCollection") => {
159                for feature in document
160                    .get("features")
161                    .and_then(Value::as_array)
162                    .into_iter()
163                    .flatten()
164                {
165                    add(feature)?;
166                }
167            }
168            Some("Feature") => add(document)?,
169            Some(
170                "Point" | "MultiPoint" | "LineString" | "MultiLineString" | "Polygon"
171                | "MultiPolygon" | "GeometryCollection",
172            ) => add(&serde_json::json!({ "geometry": document }))?,
173            other => {
174                return Err(GeoJsonError::Root {
175                    found: other.unwrap_or_default().to_owned(),
176                })
177            }
178        }
179        let clustered = source.cluster.then(|| {
180            let inputs: Vec<ClusterInput<'_>> = points
181                .iter()
182                .filter_map(|feature| {
183                    Some((
184                        *feature.geometry.points.first()?,
185                        feature.properties.as_slice(),
186                    ))
187                })
188                .collect();
189            let options = ClusterOptions::new(
190                source.cluster_radius,
191                source.cluster_max_zoom,
192                source.cluster_min_points,
193                source.cluster_properties.as_ref(),
194                source.maxzoom,
195            );
196            let clusters = Clusters::new(&inputs, options);
197            (points, clusters)
198        });
199        Ok(Self {
200            features,
201            clustered,
202            filtered: filter_declared,
203            tree: RTree::bulk_load(entries),
204            approximate_bytes: bytes,
205        })
206    }

223    /// The GeoJSON a source with a filter or clusters presents to `coords`: the features of
224    /// `document` that pass the filter, with its points replaced by the clusters they form at
225    /// that zoom. `None` for a source that presents its document unchanged.
226    pub fn source_document(
227        &self,
228        document: &Value,
229        source: &GeoJsonSource,
230        coords: WorldTileCoords,
231    ) -> Option<Value> {
232        let clusters = self.clustered.as_ref().map(|(_, clusters)| clusters);
233        if clusters.is_none() && !self.filtered {
234            return None;
235        }
236        let filter = source_filter(source);
237        let members: Vec<&Value> = match document.get("type").and_then(Value::as_str) {
238            Some("FeatureCollection") => document
239                .get("features")
240                .and_then(Value::as_array)
241                .map(|features| features.iter().collect())
242                .unwrap_or_default(),
243            _ => vec![document],
244        };
245        let mut features: Vec<Value> = members
246            .into_iter()
247            .filter(|feature| {
248                filter
249                    .as_ref()
250                    .is_none_or(|filter| feature_passes(feature, filter, 0.0))
251            })
252            .filter(|feature| {
253                clusters.is_none()
254                    || !feature.get("geometry").is_some_and(|geometry| {
255                        geometry.get("type").and_then(Value::as_str) == Some("Point")
256                    })
257            })
258            .cloned()
259            .collect();
260        if let Some(clusters) = clusters {
261            let zoom = u32::from(u8::from(coords.z));
262            let window = TileWindow::of(coords).bounds;
263            let points = &self.clustered.as_ref()?.0;
264            for item in clusters.within(zoom.min(24) as u8, window) {
265                let (position, id, properties) = match item {
266                    Clustered::Point(index, position) => {
267                        (position, points[index].id, points[index].properties.clone())
268                    }
269                    Clustered::Cluster(position, id, properties) => {
270                        (position, Some(id), properties)
271                    }
272                };
273                let mut feature = serde_json::json!({
274                    "type": "Feature",
275                    "properties": properties.into_iter().collect::<serde_json::Map<String, Value>>(),
276                    "geometry": {"type": "Point", "coordinates": super::cluster::unproject(position)},
277                });
278                if let Some(id) = id {
279                    feature["id"] = serde_json::json!(id);
280                }
281                features.push(feature);
282            }
283        }
284        Some(serde_json::json!({"type": "FeatureCollection", "features": features}))
285    }

287    /// Vector tile bytes with the features that touch `coords`, in document order.
288    pub fn tile(&self, coords: WorldTileCoords) -> Vec<u8> {
289        let zoom = u32::from(u8::from(coords.z));
290        let TileWindow {
291            tiles,
292            west,
293            north,
294            bounds,
295        } = TileWindow::of(coords);
296        let window = AABB::from_corners([bounds[0], bounds[1]], [bounds[2], bounds[3]]);
297        let mut hits: Vec<u32> = self
298            .tree
299            .locate_in_envelope_intersecting(&window)
300            .map(|entry| entry.feature)
301            .collect();
302        hits.sort_unstable();
303        let mut encoder = TileEncoder::default();
304        let to_tile = |point: &Point| -> (i32, i32) {
305            (
306                ((point[0] - west) * tiles * f64::from(EXTENT)).round() as i32,
307                ((point[1] - north) * tiles * f64::from(EXTENT)).round() as i32,
308            )
309        };
310        for index in hits {
311            let feature = &self.features[index as usize];
312            encoder.feature(feature, &feature.geometry.within(bounds), &to_tile);
313        }
314        if let Some((points, clusters)) = &self.clustered {
315            for item in clusters.within(zoom.min(24) as u8, bounds) {
316                let feature = match item {
317                    Clustered::Point(index, _) => points[index].clone(),
318                    Clustered::Cluster(position, id, properties) => IndexedFeature {
319                        id: Some(id),
320                        geometry: Geometry {
321                            points: vec![position],
322                            ..Geometry::default()
323                        },
324                        properties,
325                    },
326                };
327                encoder.feature(&feature, &feature.geometry, &to_tile);
328            }
329        }
330        Tile {
331            layers: vec![encoder.finish()],
332        }
333        .encode_to_vec()
334    }
335}

337fn promoted_property(promote: Option<&PromoteId>) -> Option<&str> {
338    match promote? {
339        PromoteId::Property(name) => Some(name),
340        PromoteId::PerLayer(map) => map.get(GEOJSON_LAYER).map(String::as_str),
341    }

344/// The id a feature carries into tiles and queries: the promoted property or `id` when it is a
345/// whole non-negative number, else its position in the document with `generateId`.
346/// The id of each feature of a collection, or of a bare feature, in the order a geometry
347/// processor meets them.
348pub(super) fn feature_ids(geojson: &Value) -> Vec<Option<u64>> {
349    geojson
350        .get("features")
351        .and_then(Value::as_array)
352        .map(Vec::as_slice)
353        .unwrap_or(std::slice::from_ref(geojson))
354        .iter()
355        .map(|feature| numeric_feature_id(feature, None, None))
356        .collect()
357}
358
359pub(super) fn numeric_feature_id(
360    feature: &Value,
361    promoted: Option<&str>,
362    generated: Option<usize>,
363) -> Option<u64> {
364    promoted
365        .and_then(|name| feature.get("properties")?.get(name))
366        .and_then(integer_id)
367        .or_else(|| feature.get("id").and_then(integer_id))
368        .or(generated.map(|index| index as u64))
369}
370
371fn integer_id(value: &Value) -> Option<u64> {
372    value.as_u64().or_else(|| {
373        let number = value.as_f64()?;
374        (number >= 0.0 && number.fract() == 0.0 && number < 9.0e15).then_some(number as u64)
375    })
376}

378fn project(position: &Value) -> Result<Point, &'static str> {
379    let coordinates = position.as_array().ok_or("position is not an array")?;
380    let longitude = coordinates
381        .first()
382        .and_then(Value::as_f64)
383        .ok_or("position has no longitude")?;
384    let latitude = coordinates
385        .get(1)
386        .and_then(Value::as_f64)
387        .ok_or("position has no latitude")?;
388    let latitude = latitude.clamp(-MAX_LATITUDE, MAX_LATITUDE).to_radians();
389    Ok([
390        (longitude + 180.0) / 360.0,
391        0.5 - (std::f64::consts::FRAC_PI_4 + latitude / 2.0).tan().ln()
392            / (2.0 * std::f64::consts::PI),
393    ])
394}

396fn read_ring(value: &Value) -> Result<Ring, &'static str> {
397    value
398        .as_array()
399        .ok_or("coordinates are not an array")?
400        .iter()
401        .map(project)
402        .collect()
403}
404
405fn read_rings(value: &Value) -> Result<Vec<Ring>, &'static str> {
406    value
407        .as_array()
408        .ok_or("coordinates are not an array")?
409        .iter()
410        .map(read_ring)
411        .collect()
412}

414fn read_geometry(geometry: &Value, out: &mut Geometry) -> Result<(), &'static str> {
415    let coordinates = geometry.get("coordinates");
416    let need = || coordinates.ok_or("geometry has no coordinates");
417    match geometry.get("type").and_then(Value::as_str) {
418        Some("Point") => out.points.push(project(need()?)?),
419        Some("MultiPoint") => out.points.extend(read_ring(need()?)?),
420        Some("LineString") => out.lines.push(read_ring(need()?)?),
421        Some("MultiLineString") => out.lines.extend(read_rings(need()?)?),
422        Some("Polygon") => out.polygons.push(read_rings(need()?)?),
423        Some("MultiPolygon") => {
424            for polygon in need()?.as_array().ok_or("coordinates are not an array")? {
425                out.polygons.push(read_rings(polygon)?);
426            }
427        }
428        Some("GeometryCollection") => {
429            for member in geometry
430                .get("geometries")
431                .and_then(Value::as_array)
432                .ok_or("collection has no geometries")?
433            {
434                read_geometry(member, out)?;
435            }
436        }
437        _ => return Err("unsupported geometry type"),
438    }
439    Ok(())
440}
441
442fn positions(geometry: &Geometry) -> impl Iterator<Item = &Point> {
443    geometry
444        .points
445        .iter()
446        .chain(geometry.lines.iter().flatten())
447        .chain(geometry.polygons.iter().flatten().flatten())
448}
449
450fn coordinate_count(geometry: &Geometry) -> usize {
451    positions(geometry).count()
452}
453
454fn bounds_of(geometry: &Geometry) -> Option<[f64; 4]> {
455    positions(geometry).fold(None, |bounds, point| {
456        let [min_x, min_y, max_x, max_y] =
457            bounds.unwrap_or([point[0], point[1], point[0], point[1]]);
458        Some([
459            min_x.min(point[0]),
460            min_y.min(point[1]),
461            max_x.max(point[0]),
462            max_y.max(point[1]),
463        ])
464    })

483/// Whether a feature is one point, the only geometry that clusters.
484fn is_single_point(geometry: &Value, parsed: &Geometry) -> bool {
485    geometry.get("type").and_then(Value::as_str) == Some("Point")
486        && parsed.points.len() == 1
487        && parsed.lines.is_empty()
488        && parsed.polygons.is_empty()
489}
