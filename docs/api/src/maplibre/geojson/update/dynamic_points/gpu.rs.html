27/// Synchronizes tracked dynamic-point GeoJSON sources when `style.revision()` changes (`O(1)` when
28/// revision is unchanged).
29pub(crate) fn ensure_seeded_for_style(world: &mut World, style: &Style) {
30    let current_rev = style.revision();
31    if world
32        .resources
33        .get::<DynamicPointGeoJson>()
34        .is_some_and(|s| s.last_style_revision == Some(current_rev))
35    {
36        return;
37    }
38    let eligible: Vec<String> = style
39        .sources
40        .keys()
41        .filter(|name| source_supports_dynamic_points(style, name))
42        .cloned()
43        .collect();
44
45    if eligible.is_empty() && world.resources.get::<DynamicPointGeoJson>().is_none() {
46        return;
47    }
48    {
49        let state = world.resources.get_or_init_mut::<DynamicPointGeoJson>();
50        state
51            .sources
52            .retain(|name, _| eligible.iter().any(|e| e == name));
53        state.last_style_revision = Some(current_rev);
54    }
55    for name in eligible {
56        let already = world
57            .resources
58            .get::<DynamicPointGeoJson>()
59            .is_some_and(|s| s.sources.contains_key(&name));
60        if !already {
61            seed_from_inline(style, world, &name);
62        }
63    }
64}

66/// Builds `2 * vertices` `[fill_rgba8, dyn_radius_u32]` records for a newly uploaded bucket.
67/// Returns `Some(seq)` when populated directly from `DynamicPointGeoJson`.
68#[allow(clippy::too_many_arguments)]
69pub(crate) fn populate_dynamic_circle_bucket(
70    dyn_points: Option<&mut DynamicPointGeoJson>,
71    style_layer: &StyleLayer,
72    bucket: &AvailableVectorLayerBucket,
73    fallback_vertices: Option<&[crate::render::ShaderVertex]>,
74    active_colors: &[[f32; 4]],
75    fallback_color: [f32; 4],
76    overscaled_zoom: u8,
77    out: &mut Vec<FillShaderFeatureMetadata>,
78) -> Option<u64> {
79    let coords = bucket.coords;
80    let vertices = if bucket.buffer.buffer.vertices.is_empty() {
81        fallback_vertices.unwrap_or(&[])
82    } else {
83        bucket.buffer.buffer.vertices.as_slice()
84    };
85    let vertex_count = vertices.len();
86    out.clear();
87
88    if let (Some(state), Some(source_name), Some(LayerPaint::Circle(circle_paint))) = (
89        dyn_points,
90        style_layer.source.as_deref(),
91        style_layer.paint.as_ref(),
92    ) {
93        if let Some(src_state) = state.sources.get_mut(source_name) {
94            let seq = src_state.seq;
95            if vertex_count == 0 {
96                return Some(seq);
97            }
98            let hits: &[u32] = src_state
99                .tile_hits
100                .entry(coords)
101                .or_insert_with(|| super::compute_tile_hits(&src_state.world_points, coords));
102            let layer_id = style_layer.layer_id();
103            if !hits.is_empty()
104                && (vertex_count == hits.len() * 4 || vertex_count == hits.len() * 16)
105            {
106                let verts_per_circle = vertex_count / hits.len();
107                let tile_zoom = f64::from(overscaled_zoom.max(u8::from(coords.z)));
108                evaluate_circle_metadata_into(
109                    out,
110                    circle_paint,
111                    &src_state.properties,
112                    hits,
113                    verts_per_circle,
114                    tile_zoom,
115                    (seq, layer_id, &mut src_state.eval_cache),
116                );
117                return Some(seq);
118            }
119        }
120    }
121
122    fallback_dynamic_circle_from_vertices(
123        out,
124        &bucket.feature_indices,
125        active_colors,
126        fallback_color,
127        vertices,
128    );
129    None
130}

132/// Records `(seq, allocation_id)` after `upload_bucket` commits a dynamic circle allocation.
133pub(crate) fn record_uploaded_dynamic_circle(
134    dyn_points: Option<&mut DynamicPointGeoJson>,
135    buffer_pool: &VectorBufferPool,
136    coords: WorldTileCoords,
137    style_layer: &StyleLayer,
138    seq: Option<u64>,
139) {
140    let (Some(state), Some(source_name), Some(seq)) =
141        (dyn_points, style_layer.source.as_deref(), seq)
142    else {
143        return;
144    };
145    let Some(src_state) = state.sources.get_mut(source_name) else {
146        return;
147    };
148    let style_layer_id = style_layer.layer_id();
149    let Some(alloc_id) = buffer_pool
150        .index()
151        .get_layers(coords)
152        .and_then(|layers| {
153            layers
154                .iter()
155                .rev()
156                .find(|e| e.style_layer_id == style_layer_id)
157        })
158        .map(|e| e.allocation_id())
159    else {
160        return;
161    };
162    src_state
163        .applied_seq
164        .insert((coords, style_layer_id), (seq, alloc_id));
165}

167/// Updates stale visible `CircleDynamicRadiusShader` allocations in-place, self-healing via tile
168/// refresh if any resident tile's vertex count diverges from the stationary point index.
169pub(crate) fn refresh_dynamic_point_layers(
170    world: &mut World,
171    style: &mut Style,
172    queue: &UploadQueue,
173    painted_tiles: &[WorldTileCoords],
174) {
175    if world
176        .resources
177        .get::<DynamicPointGeoJson>()
178        .is_none_or(|s| s.sources.is_empty())
179    {
180        return;
181    }
182    let mut mismatch_sources: Vec<String> = Vec::new();
183    {
184        let tiles = &world.tiles;
185        let Some((Initialized(buffer_pool), dyn_points)) = world
186            .resources
187            .query_mut::<(&mut Eventually<VectorBufferPool>, &mut DynamicPointGeoJson)>()
188        else {
189            return;
190        };
191        let DynamicPointGeoJson {
192            sources,
193            gpu_scratch,
194            ..
195        } = dyn_points;
196
197        for style_layer in &style.layers {
198            if style_layer.is_hidden() {
199                continue;
200            }
201            let (Some(source_name), Some(LayerPaint::Circle(circle_paint))) =
202                (style_layer.source.as_deref(), style_layer.paint.as_ref())
203            else {
204                continue;
205            };
206            let Some(src_state) = sources.get_mut(source_name) else {
207                continue;
208            };
209            let seq = src_state.seq;
210            let style_layer_id = style_layer.layer_id();
211
212            for &coords in painted_tiles {
213                let entry_info = buffer_pool
214                    .index()
215                    .get_layers(coords)
216                    .and_then(|layers| {
217                        layers.iter().find(|e| {
218                            e.style_layer_id == style_layer_id && e.is_dynamic_circle_metadata()
219                        })
220                    })
221                    .map(|e| {
222                        let bytes = e.feature_metadata_buffer_range().end
223                            - e.feature_metadata_buffer_range().start;
224                        (
225                            e.allocation_id(),
226                            bytes as usize / std::mem::size_of::<FillShaderFeatureMetadata>(),
227                        )
228                    });
229                let Some((alloc_id, allocated_records)) = entry_info else {
230                    continue;
231                };
232                if src_state.applied_seq.get(&(coords, style_layer_id)) == Some(&(seq, alloc_id)) {
233                    continue;
234                }
235
236                let hits: &[u32] = src_state
237                    .tile_hits
238                    .entry(coords)
239                    .or_insert_with(|| super::compute_tile_hits(&src_state.world_points, coords));
240                if hits.is_empty() && allocated_records == 0 {
241                    src_state
242                        .applied_seq
243                        .insert((coords, style_layer_id), (seq, alloc_id));
244                    continue;
245                }
246                let verts = allocated_records / 2;
247                if allocated_records % 2 != 0
248                    || hits.is_empty()
249                    || !(verts == hits.len() * 4 || verts == hits.len() * 16)
250                {
251                    if tile_retry::is_waiting_or_pending(tiles, coords, RequestKind::Vector)
252                        || tiles
253                            .query::<&crate::vector::content::LayerReplacements>(coords)
254                            .is_some()
255                    {
256                        continue;
257                    }
258                    if !mismatch_sources.iter().any(|s| s == source_name) {
259                        mismatch_sources.push(source_name.to_owned());
260                    }
261                    break;
262                }
263
264                let verts_per_circle = verts / hits.len();
265                let overscaled_z = tiles
266                    .query::<&VectorLayerBucketComponent>(coords)
267                    .map_or(u8::from(coords.z), |b| {
268                        b.overscaled_zoom.max(u8::from(coords.z))
269                    });
270                evaluate_circle_metadata_into(
271                    gpu_scratch,
272                    circle_paint,
273                    &src_state.properties,
274                    hits,
275                    verts_per_circle,
276                    f64::from(overscaled_z),
277                    (seq, style_layer_id, &mut src_state.eval_cache),
278                );
279                if let Some(entry) = buffer_pool.index().get_layers(coords).and_then(|layers| {
280                    layers.iter().find(|e| {
281                        e.style_layer_id == style_layer_id && e.allocation_id() == alloc_id
282                    })
283                }) {
284                    buffer_pool.update_feature_metadata(queue, entry, gpu_scratch);
285                }
286                if style.terrain.is_some() {
287                    buffer_pool.bump_layer_paint_revision(coords, &style_layer.id);
288                }
289                src_state
290                    .applied_seq
291                    .insert((coords, style_layer_id), (seq, alloc_id));
292            }
293        }
294    }
295
296    if !mismatch_sources.is_empty() {
297        flush_deferred_geojson_bytes(style, world);
298        if let Some(state) = world.resources.get_mut::<DynamicPointGeoJson>() {
299            for name in &mismatch_sources {
300                state.sources.remove(name);
301            }
302        }
303        let all_tiles: Vec<WorldTileCoords> =
304            world.tiles.pyramid().coords_coarse_to_fine().collect();
305        for name in &mismatch_sources {
306            if let Some(Source::GeoJson(src)) = style.sources.get_mut(name) {
307                src.generation = fresh_generation();
308            }
309            if !all_tiles.is_empty() {
310                tile_retry::refresh_tiles_for_sources(
311                    world,
312                    RequestKind::Vector,
313                    &all_tiles,
314                    std::slice::from_ref(name),
315                );
316            }
317        }
318        style.bump_revision();
319        crate::render::frame_signals::mark_dirty(world);
320    }
321}

323fn evaluate_circle_metadata_into(
324    out: &mut Vec<FillShaderFeatureMetadata>,
325    circle_paint: &CirclePaint,
326    all_props: &[FeatureProperties],
327    hits: &[u32],
328    verts_per_circle: usize,
329    tile_zoom: f64,
330    (seq, style_layer_id, cache): (u64, crate::style::StyleLayerId, &mut EvalCache),
331) {
332    out.clear();
333    let needed = hits.len() * verts_per_circle * 2;
334    if out.capacity() < needed {
335        out.reserve(needed - out.len());
336    }
337
338    let cache_key = (seq, style_layer_id, tile_zoom.to_bits());
339    if cache.key != Some(cache_key) {
340        cache.slots.clear();
341        cache.slots.resize(all_props.len(), None);
342        cache.key = Some(cache_key);
343    } else if cache.slots.len() != all_props.len() {
344        cache.slots.resize(all_props.len(), None);
345    }
346
347    let radius_prop = circle_paint.radius();
348    let radius_const = radius_prop.is_feature_constant().then(|| {
349        radius_prop
350            .evaluate_at_zoom(tile_zoom)
351            .unwrap_or(CirclePaint::DEFAULT_RADIUS)
352            .max(0.0)
353    });
354    let radius_ramp = ScalarRamp::compile(&radius_prop);
355    let radius_zoom_const = radius_prop.is_zoom_constant();
356
357    let color_prop = circle_paint.circle_color.as_ref();
358    let color_const = match color_prop {
359        None => Some([0.0, 0.0, 0.0, 1.0]),
360        Some(prop) if prop.is_feature_constant() => Some(
361            prop.evaluate_at_zoom(tile_zoom)
362                .map_or([0.0, 0.0, 0.0, 1.0], |c| c.to_array()),
363        ),
364        Some(_) => None,
365    };
366    let color_ramp = color_prop.and_then(ColorRamp::compile);
367
368    let opacity_prop = circle_paint.circle_opacity.as_ref();
369    let opacity_const = match opacity_prop {
370        None => Some(1.0_f32),
371        Some(prop) if prop.is_feature_constant() => Some(
372            prop.evaluate_at_zoom(tile_zoom)
373                .unwrap_or(1.0)
374                .clamp(0.0, 1.0),
375        ),
376        Some(_) => None,
377    };
378    let opacity_ramp = opacity_prop.and_then(ScalarRamp::compile);
379
380    for &feat_idx in hits {
381        let idx = feat_idx as usize;
382        if let Some(pair) = cache.slots.get(idx).and_then(Option::as_ref) {
383            for _ in 0..verts_per_circle {
384                out.extend_from_slice(pair);
385            }
386            continue;
387        }
388        let Some(props) = all_props.get(idx) else {
389            continue;
390        };
391        let mut rgba = color_const.unwrap_or_else(|| {
392            color_ramp
393                .as_ref()
394                .and_then(|r| r.eval(props))
395                .or_else(|| {
396                    color_prop
397                        .and_then(|p| p.evaluate_for(props, tile_zoom))
398                        .map(|c| c.to_array())
399                })
400                .unwrap_or([0.0, 0.0, 0.0, 1.0])
401        });
402        let opacity = opacity_const.unwrap_or_else(|| {
403            opacity_ramp
404                .as_ref()
405                .and_then(|r| r.eval(props))
406                .or_else(|| opacity_prop.and_then(|p| p.evaluate_for(props, tile_zoom)))
407                .unwrap_or(1.0)
408                .clamp(0.0, 1.0)
409        });
410        rgba[3] *= opacity;
411
412        let (radius, radius_next) = if let Some(r) = radius_const {
413            (r, r)
414        } else {
415            let r0 = radius_ramp
416                .as_ref()
417                .and_then(|r| r.eval(props))
418                .or_else(|| radius_prop.evaluate_for(props, tile_zoom))
419                .unwrap_or(CirclePaint::DEFAULT_RADIUS)
420                .max(0.0);
421            let r1 = if radius_zoom_const {
422                r0
423            } else {
424                radius_prop
425                    .evaluate_for(props, tile_zoom + 1.0)
426                    .unwrap_or(CirclePaint::DEFAULT_RADIUS)
427                    .max(0.0)
428            };
429            (r0, r1)
430        };
431
432        let pair = [
433            FillShaderFeatureMetadata::from_rgba(rgba),
434            pack_dynamic_circle_radii(radius, radius_next),
435        ];
436        if let Some(slot) = cache.slots.get_mut(idx) {
437            *slot = Some(pair);
438        }
439        for _ in 0..verts_per_circle {
440            out.extend_from_slice(&pair);
441        }
442    }
443}

445fn fallback_dynamic_circle_from_vertices(
446    out: &mut Vec<FillShaderFeatureMetadata>,
447    feature_indices: &[u32],
448    active_colors: &[[f32; 4]],
449    fallback_color: [f32; 4],
450    vertices: &[crate::render::ShaderVertex],
451) {
452    out.clear();
453    let needed = vertices.len() * 2;
454    if out.capacity() < needed {
455        out.reserve(needed - out.len());
456    }
457    let mut vert_cursor = 0_usize;
458    for (feat_idx, &count) in feature_indices.iter().enumerate() {
459        let color = active_colors
460            .get(feat_idx)
461            .copied()
462            .unwrap_or(fallback_color);
463        let fill_meta = FillShaderFeatureMetadata::from_rgba(color);
464        for _ in 0..count {
465            let Some(v) = vertices.get(vert_cursor) else {
466                break;
467            };
468            vert_cursor += 1;
469            out.extend_from_slice(&[
470                fill_meta,
471                pack_dynamic_circle_radii(v.normal[0], v.edge_distance),
472            ]);
473        }
474    }
475    while vert_cursor < vertices.len() {
476        let v = &vertices[vert_cursor];
477        vert_cursor += 1;
478        out.extend_from_slice(&[
479            FillShaderFeatureMetadata::from_rgba(fallback_color),
480            pack_dynamic_circle_radii(v.normal[0], v.edge_distance),
481        ]);
482    }
