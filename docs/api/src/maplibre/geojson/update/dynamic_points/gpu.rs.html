26/// Synchronizes tracked dynamic-point GeoJSON sources when `style.revision()` changes (`O(1)` when
27/// revision is unchanged).
28pub(crate) fn ensure_seeded_for_style(world: &mut World, style: &Style) {
29    let current_rev = style.revision();
30    if world
31        .resources
32        .get::<DynamicPointGeoJson>()
33        .is_some_and(|s| s.last_style_revision == Some(current_rev))
34    {
35        return;
36    }
37    let eligible: Vec<String> = style
38        .sources
39        .keys()
40        .filter(|name| source_supports_dynamic_points(style, name))
41        .cloned()
42        .collect();
43
44    if eligible.is_empty() && world.resources.get::<DynamicPointGeoJson>().is_none() {
45        return;
46    }
47    {
48        let state = world.resources.get_or_init_mut::<DynamicPointGeoJson>();
49        state
50            .sources
51            .retain(|name, _| eligible.iter().any(|e| e == name));
52        state.last_style_revision = Some(current_rev);
53    }
54    for name in eligible {
55        let already = world
56            .resources
57            .get::<DynamicPointGeoJson>()
58            .is_some_and(|s| s.sources.contains_key(&name));
59        if !already {
60            seed_from_inline(style, world, &name);
61        }
62    }
63}

65/// Builds `2 * vertices` `[fill_rgba8, dyn_radius_u32]` records for a newly uploaded bucket.
66/// Returns `Some(seq)` when populated directly from `DynamicPointGeoJson`.
67#[allow(clippy::too_many_arguments)]
68pub(crate) fn populate_dynamic_circle_bucket(
69    dyn_points: Option<&mut DynamicPointGeoJson>,
70    style_layer: &StyleLayer,
71    bucket: &AvailableVectorLayerBucket,
72    fallback_vertices: Option<&[crate::render::ShaderVertex]>,
73    active_colors: &[[f32; 4]],
74    fallback_color: [f32; 4],
75    overscaled_zoom: u8,
76    out: &mut Vec<FillShaderFeatureMetadata>,
77) -> Option<u64> {
78    let coords = bucket.coords;
79    let vertices = if bucket.buffer.buffer.vertices.is_empty() {
80        fallback_vertices.unwrap_or(&[])
81    } else {
82        bucket.buffer.buffer.vertices.as_slice()
83    };
84    let vertex_count = vertices.len();
85    out.clear();
86
87    if let (Some(state), Some(source_name), Some(LayerPaint::Circle(circle_paint))) = (
88        dyn_points,
89        style_layer.source.as_deref(),
90        style_layer.paint.as_ref(),
91    ) {
92        if let Some(src_state) = state.sources.get_mut(source_name) {
93            let seq = src_state.seq;
94            if vertex_count == 0 {
95                return Some(seq);
96            }
97            let hits: &[u32] = src_state
98                .tile_hits
99                .entry(coords)
100                .or_insert_with(|| super::compute_tile_hits(&src_state.world_points, coords));
101            if !hits.is_empty()
102                && (vertex_count == hits.len() * 4 || vertex_count == hits.len() * 16)
103            {
104                let verts_per_circle = vertex_count / hits.len();
105                let tile_zoom = f64::from(overscaled_zoom.max(u8::from(coords.z)));
106                evaluate_circle_metadata_into(
107                    out,
108                    circle_paint,
109                    &src_state.properties,
110                    hits,
111                    verts_per_circle,
112                    tile_zoom,
113                );
114                return Some(seq);
115            }
116        }
117    }
118
119    fallback_dynamic_circle_from_vertices(
120        out,
121        &bucket.feature_indices,
122        active_colors,
123        fallback_color,
124        vertices,
125    );
126    None
127}

129/// Records `(seq, allocation_id)` after `upload_bucket` commits a dynamic circle allocation.
130pub(crate) fn record_uploaded_dynamic_circle(
131    dyn_points: Option<&mut DynamicPointGeoJson>,
132    buffer_pool: &VectorBufferPool,
133    coords: WorldTileCoords,
134    style_layer: &StyleLayer,
135    seq: Option<u64>,
136) {
137    let (Some(state), Some(source_name), Some(seq)) =
138        (dyn_points, style_layer.source.as_deref(), seq)
139    else {
140        return;
141    };
142    let Some(src_state) = state.sources.get_mut(source_name) else {
143        return;
144    };
145    let style_layer_id = style_layer.layer_id();
146    let Some(alloc_id) = buffer_pool
147        .index()
148        .get_layers(coords)
149        .and_then(|layers| {
150            layers
151                .iter()
152                .rev()
153                .find(|e| e.style_layer_id == style_layer_id)
154        })
155        .map(|e| e.allocation_id())
156    else {
157        return;
158    };
159    src_state
160        .applied_seq
161        .insert((coords, style_layer_id), (seq, alloc_id));
162}

164/// Updates stale visible `CircleDynamicRadiusShader` allocations in-place, self-healing via tile
165/// refresh if any resident tile's vertex count diverges from the stationary point index.
166pub(crate) fn refresh_dynamic_point_layers(
167    world: &mut World,
168    style: &mut Style,
169    queue: &UploadQueue,
170    painted_tiles: &[WorldTileCoords],
171) {
172    if world
173        .resources
174        .get::<DynamicPointGeoJson>()
175        .is_none_or(|s| s.sources.is_empty())
176    {
177        return;
178    }
179    let mut mismatch_sources: Vec<String> = Vec::new();
180    {
181        let tiles = &world.tiles;
182        let Some((Initialized(buffer_pool), dyn_points)) = world
183            .resources
184            .query_mut::<(&mut Eventually<VectorBufferPool>, &mut DynamicPointGeoJson)>()
185        else {
186            return;
187        };
188        let DynamicPointGeoJson {
189            sources,
190            gpu_scratch,
191            ..
192        } = dyn_points;
193
194        for style_layer in &style.layers {
195            if style_layer.is_hidden() {
196                continue;
197            }
198            let (Some(source_name), Some(LayerPaint::Circle(circle_paint))) =
199                (style_layer.source.as_deref(), style_layer.paint.as_ref())
200            else {
201                continue;
202            };
203            let Some(src_state) = sources.get_mut(source_name) else {
204                continue;
205            };
206            let seq = src_state.seq;
207            let style_layer_id = style_layer.layer_id();
208
209            for &coords in painted_tiles {
210                let entry_info = buffer_pool
211                    .index()
212                    .get_layers(coords)
213                    .and_then(|layers| {
214                        layers.iter().find(|e| {
215                            e.style_layer_id == style_layer_id && e.is_dynamic_circle_metadata()
216                        })
217                    })
218                    .map(|e| {
219                        let bytes = e.feature_metadata_buffer_range().end
220                            - e.feature_metadata_buffer_range().start;
221                        (
222                            e.allocation_id(),
223                            bytes as usize / std::mem::size_of::<FillShaderFeatureMetadata>(),
224                        )
225                    });
226                let Some((alloc_id, allocated_records)) = entry_info else {
227                    continue;
228                };
229                if src_state.applied_seq.get(&(coords, style_layer_id)) == Some(&(seq, alloc_id)) {
230                    continue;
231                }
232
233                let hits: &[u32] = src_state
234                    .tile_hits
235                    .entry(coords)
236                    .or_insert_with(|| super::compute_tile_hits(&src_state.world_points, coords));
237                if hits.is_empty() && allocated_records == 0 {
238                    src_state
239                        .applied_seq
240                        .insert((coords, style_layer_id), (seq, alloc_id));
241                    continue;
242                }
243                let verts = allocated_records / 2;
244                if allocated_records % 2 != 0
245                    || hits.is_empty()
246                    || !(verts == hits.len() * 4 || verts == hits.len() * 16)
247                {
248                    if !mismatch_sources.iter().any(|s| s == source_name) {
249                        mismatch_sources.push(source_name.to_owned());
250                    }
251                    break;
252                }
253
254                let verts_per_circle = verts / hits.len();
255                let overscaled_z = tiles
256                    .query::<&VectorLayerBucketComponent>(coords)
257                    .map_or(u8::from(coords.z), |b| {
258                        b.overscaled_zoom.max(u8::from(coords.z))
259                    });
260                evaluate_circle_metadata_into(
261                    gpu_scratch,
262                    circle_paint,
263                    &src_state.properties,
264                    hits,
265                    verts_per_circle,
266                    f64::from(overscaled_z),
267                );
268                if let Some(entry) = buffer_pool.index().get_layers(coords).and_then(|layers| {
269                    layers.iter().find(|e| {
270                        e.style_layer_id == style_layer_id && e.allocation_id() == alloc_id
271                    })
272                }) {
273                    buffer_pool.update_feature_metadata(queue, entry, gpu_scratch);
274                }
275                buffer_pool.bump_layer_paint_revision(coords, &style_layer.id);
276                src_state
277                    .applied_seq
278                    .insert((coords, style_layer_id), (seq, alloc_id));
279            }
280        }
281    }
282
283    if !mismatch_sources.is_empty() {
284        flush_deferred_geojson_bytes(style, world);
285        if let Some(state) = world.resources.get_mut::<DynamicPointGeoJson>() {
286            for name in &mismatch_sources {
287                state.sources.remove(name);
288            }
289        }
290        let all_tiles: Vec<WorldTileCoords> =
291            world.tiles.pyramid().coords_coarse_to_fine().collect();
292        for name in &mismatch_sources {
293            if let Some(Source::GeoJson(src)) = style.sources.get_mut(name) {
294                src.generation = fresh_generation();
295            }
296            if !all_tiles.is_empty() {
297                tile_retry::refresh_tiles_for_sources(
298                    world,
299                    RequestKind::Vector,
300                    &all_tiles,
301                    std::slice::from_ref(name),
302                );
303            }
304        }
305        style.bump_revision();
306        crate::render::frame_signals::mark_dirty(world);
307    }
308}

310fn evaluate_circle_metadata_into(
311    out: &mut Vec<FillShaderFeatureMetadata>,
312    circle_paint: &CirclePaint,
313    all_props: &[FeatureProperties],
314    hits: &[u32],
315    verts_per_circle: usize,
316    tile_zoom: f64,
317) {
318    out.clear();
319    let needed = hits.len() * verts_per_circle * 2;
320    if out.capacity() < needed {
321        out.reserve(needed - out.len());
322    }
323
324    let radius_prop = circle_paint.radius();
325    let radius_const = radius_prop.is_feature_constant().then(|| {
326        radius_prop
327            .evaluate_at_zoom(tile_zoom)
328            .unwrap_or(CirclePaint::DEFAULT_RADIUS)
329            .max(0.0)
330    });
331    let radius_zoom_const = radius_prop.is_zoom_constant();
332
333    let color_prop = circle_paint.circle_color.as_ref();
334    let color_const = match color_prop {
335        None => Some([0.0, 0.0, 0.0, 1.0]),
336        Some(prop) if prop.is_feature_constant() => Some(
337            prop.evaluate_at_zoom(tile_zoom)
338                .map_or([0.0, 0.0, 0.0, 1.0], |c| c.to_array()),
339        ),
340        Some(_) => None,
341    };
342
343    let opacity_prop = circle_paint.circle_opacity.as_ref();
344    let opacity_const = match opacity_prop {
345        None => Some(1.0_f32),
346        Some(prop) if prop.is_feature_constant() => Some(
347            prop.evaluate_at_zoom(tile_zoom)
348                .unwrap_or(1.0)
349                .clamp(0.0, 1.0),
350        ),
351        Some(_) => None,
352    };
353
354    for &feat_idx in hits {
355        let Some(props) = all_props.get(feat_idx as usize) else {
356            continue;
357        };
358        let mut rgba = color_const.unwrap_or_else(|| {
359            color_prop
360                .and_then(|p| p.evaluate_for(props, tile_zoom))
361                .map_or([0.0, 0.0, 0.0, 1.0], |c| c.to_array())
362        });
363        let opacity = opacity_const.unwrap_or_else(|| {
364            opacity_prop
365                .and_then(|p| p.evaluate_for(props, tile_zoom))
366                .unwrap_or(1.0)
367                .clamp(0.0, 1.0)
368        });
369        rgba[3] *= opacity;
370
371        let (radius, radius_next) = if let Some(r) = radius_const {
372            (r, r)
373        } else {
374            let r0 = radius_prop
375                .evaluate_for(props, tile_zoom)
376                .unwrap_or(CirclePaint::DEFAULT_RADIUS)
377                .max(0.0);
378            let r1 = if radius_zoom_const {
379                r0
380            } else {
381                radius_prop
382                    .evaluate_for(props, tile_zoom + 1.0)
383                    .unwrap_or(CirclePaint::DEFAULT_RADIUS)
384                    .max(0.0)
385            };
386            (r0, r1)
387        };
388
389        let pair = [
390            FillShaderFeatureMetadata::from_rgba(rgba),
391            pack_dynamic_circle_radii(radius, radius_next),
392        ];
393        for _ in 0..verts_per_circle {
394            out.extend_from_slice(&pair);
395        }
396    }
397}

399fn fallback_dynamic_circle_from_vertices(
400    out: &mut Vec<FillShaderFeatureMetadata>,
401    feature_indices: &[u32],
402    active_colors: &[[f32; 4]],
403    fallback_color: [f32; 4],
404    vertices: &[crate::render::ShaderVertex],
405) {
406    out.clear();
407    let needed = vertices.len() * 2;
408    if out.capacity() < needed {
409        out.reserve(needed - out.len());
410    }
411    let mut vert_cursor = 0_usize;
412    for (feat_idx, &count) in feature_indices.iter().enumerate() {
413        let color = active_colors
414            .get(feat_idx)
415            .copied()
416            .unwrap_or(fallback_color);
417        let fill_meta = FillShaderFeatureMetadata::from_rgba(color);
418        for _ in 0..count {
419            let Some(v) = vertices.get(vert_cursor) else {
420                break;
421            };
422            vert_cursor += 1;
423            out.extend_from_slice(&[
424                fill_meta,
425                pack_dynamic_circle_radii(v.normal[0], v.edge_distance),
426            ]);
427        }
428    }
429    while vert_cursor < vertices.len() {
430        let v = &vertices[vert_cursor];
431        vert_cursor += 1;
432        out.extend_from_slice(&[
433            FillShaderFeatureMetadata::from_rgba(fallback_color),
434            pack_dynamic_circle_radii(v.normal[0], v.edge_distance),
435        ]);
436    }
