1//! Fast compiled 1D linear ramp evaluators for zoom-constant per-feature expressions.

3use crate::style::{

8struct PropertyInput<'a> {
9    key: &'a str,
10    fallback: Option<f64>,
11}

13/// Zero-allocation 1D linear scalar interpolation ramp (`["interpolate", ["linear"], input, ...]`).
14pub(super) struct ScalarRamp<'a> {
15    input: PropertyInput<'a>,
16    stops: &'a [(f64, Expression)],
17}

19/// Zero-allocation 1D linear RGBA interpolation ramp (`["interpolate", ["linear"], input, ...]`).
20pub(super) struct ColorRamp<'a> {

25impl<'a> ScalarRamp<'a> {
26    /// Returns `Some` when `prop` is a simple 1D linear scalar ramp over one feature property.
27    pub(super) fn compile(prop: &'a StyleProperty<f32>) -> Option<Self> {
28        let StyleProperty::Expression(expr_prop) = prop else {
29            return None;
30        };
31        let Expression::Interpolate {
32            interpolation: Interpolation::Linear,
33            input,
34            stops,
35            ..
36        } = expr_prop.expression()
37        else {
38            return None;
39        };
40        if stops.is_empty() || !stops.iter().all(|(_, out)| stop_number(out).is_some()) {
41            return None;
42        }
43        Some(Self {
44            input: compile_input(input)?,
45            stops,
46        })
47    }
48
49    #[inline]
50    pub(super) fn eval(&self, props: &FeatureProperties) -> Option<f32> {
51        let x = eval_input(props, &self.input)?;
52        if self.stops.len() == 1 || x <= self.stops[0].0 {
53            return stop_number(&self.stops[0].1).map(|v| v as f32);
54        }
55        let last = self.stops.len() - 1;
56        if x >= self.stops[last].0 {
57            return stop_number(&self.stops[last].1).map(|v| v as f32);
58        }
59        for pair in self.stops.windows(2) {
60            let (x0, e0) = &pair[0];
61            let (x1, e1) = &pair[1];
62            if x <= *x1 {
63                let y0 = stop_number(e0)?;
64                let y1 = stop_number(e1)?;
65                let span = *x1 - *x0;
66                let t = if span > 0.0 {
67                    ((x - *x0) / span).clamp(0.0, 1.0)
68                } else {
69                    0.0
70                };
71                return Some((y0 + (y1 - y0) * t) as f32);
72            }
73        }
74        stop_number(&self.stops[last].1).map(|v| v as f32)
75    }
76}

78impl<'a> ColorRamp<'a> {
79    /// Returns `Some` when `prop` is a simple 1D linear sRGB color ramp over one feature property.
80    pub(super) fn compile<T>(prop: &'a StyleProperty<T>) -> Option<Self> {
81        let StyleProperty::Expression(expr_prop) = prop else {
82            return None;
83        };
84        let Expression::Interpolate {
85            interpolation: Interpolation::Linear,
86            space: ColorSpace::Rgb,
87            input,
88            stops,
89            ..
90        } = expr_prop.expression()
91        else {
92            return None;
93        };
94        if stops.is_empty() || !stops.iter().all(|(_, out)| stop_color(out).is_some()) {
95            return None;
96        }
97        Some(Self {
98            input: compile_input(input)?,
99            stops,
100        })
101    }
102
103    #[inline]
104    pub(super) fn eval(&self, props: &FeatureProperties) -> Option<[f32; 4]> {
105        let x = eval_input(props, &self.input)?;
106        let to_f32 = |c: Color| {
107            let [r, g, b, a] = c.straight();
108            [r as f32, g as f32, b as f32, a as f32]
109        };
110        if self.stops.len() == 1 || x <= self.stops[0].0 {
111            return stop_color(&self.stops[0].1).map(to_f32);
112        }
113        let last = self.stops.len() - 1;
114        if x >= self.stops[last].0 {
115            return stop_color(&self.stops[last].1).map(to_f32);
116        }
117        for pair in self.stops.windows(2) {
118            let (x0, e0) = &pair[0];
119            let (x1, e1) = &pair[1];
120            if x <= *x1 {
121                let c0 = stop_color(e0)?.straight();
122                let c1 = stop_color(e1)?.straight();
123                let span = *x1 - *x0;
124                let t = if span > 0.0 {
125                    ((x - *x0) / span).clamp(0.0, 1.0)
126                } else {
127                    0.0
128                };
129                return Some([
130                    (c0[0] + (c1[0] - c0[0]) * t) as f32,
131                    (c0[1] + (c1[1] - c0[1]) * t) as f32,
132                    (c0[2] + (c1[2] - c0[2]) * t) as f32,
133                    (c0[3] + (c1[3] - c0[3]) * t) as f32,
134                ]);
135            }
136        }
137        stop_color(&self.stops[last].1).map(to_f32)
138    }
139}

141fn compile_input(expr: &Expression) -> Option<PropertyInput<'_>> {
142    match expr {
143        Expression::Get { key, object: None } => Some(PropertyInput {
144            key: literal_str(key)?,
145            fallback: None,
146        }),
147        Expression::Coalesce { operands, .. } if operands.len() == 2 => {
148            let Expression::Get { key, object: None } = &operands[0] else {
149                return None;
150            };
151            Some(PropertyInput {
152                key: literal_str(key)?,
153                fallback: Some(stop_number(&operands[1])?),
154            })
155        }
156        _ => None,
157    }
158}
159
160#[inline]
161fn eval_input(props: &FeatureProperties, input: &PropertyInput<'_>) -> Option<f64> {
162    props
163        .get(input.key)
164        .and_then(Value::as_number)
165        .or(input.fallback)
166}
167
168#[inline]
169fn literal_str(expr: &Expression) -> Option<&str> {
170    match expr {
171        Expression::Literal(Value::String(s))
172        | Expression::Folded {
173            value: Value::String(s),
174            ..
175        } => Some(s.as_str()),
176        _ => None,
177    }
178}
179
180#[inline]
181fn stop_number(expr: &Expression) -> Option<f64> {
182    match expr {
183        Expression::Literal(Value::Number(v))
184        | Expression::Folded {
185            value: Value::Number(v),
186            ..
187        } => Some(*v),
188        _ => None,
189    }
190}
191
192#[inline]
193fn stop_color(expr: &Expression) -> Option<Color> {
194    match expr {
195        Expression::Literal(Value::Color(c))
196        | Expression::Folded {
197            value: Value::Color(c),
198            ..
199        } => Some(*c),
200        Expression::Literal(Value::String(s))
201        | Expression::Folded {
202            value: Value::String(s),
203            ..
204        } => Color::parse(s),
205        _ => None,
206    }
