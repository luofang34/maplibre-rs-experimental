13#[inline]
14fn col_index(keys: &[String], key: &str) -> Option<usize> {
15    keys.iter().position(|k| k == key)
16}

18#[allow(clippy::too_many_arguments)]
19pub(super) fn evaluate_circle_metadata_into(
20    out: &mut Vec<FillShaderFeatureMetadata>,
21    circle_paint: &CirclePaint,
22    all_props: &[FeatureProperties],
23    fast_columns: &FastColumnStore,
24    hits: &[u32],
25    verts_per_circle: usize,
26    tile_zoom: f64,
27    (seq, style_layer_id, cache): (u64, StyleLayerId, &mut EvalCache),
28) {
29    out.clear();
30    let needed = hits.len() * verts_per_circle * 2;
31    if out.capacity() < needed {
32        out.reserve(needed - out.len());
33    }
34
35    let use_fast =
36        fast_columns.active_seq == Some(seq) && fast_columns.values.len() == all_props.len();
37    let cache_key = (seq, style_layer_id, tile_zoom.to_bits());
38    if cache.key != Some(cache_key) || cache.pairs.len() != all_props.len() {
39        cache.pairs.clear();
40        cache.pairs.reserve(all_props.len());
41        let radius_prop = circle_paint.radius();
42        let radius_const = radius_prop.is_feature_constant().then(|| {
43            radius_prop
44                .evaluate_at_zoom(tile_zoom)
45                .unwrap_or(CirclePaint::DEFAULT_RADIUS)
46                .max(0.0)
47        });
48        let radius_ramp = ScalarRamp::compile(&radius_prop);
49        let radius_col = if use_fast {
50            radius_ramp
51                .as_ref()
52                .and_then(|r| col_index(&fast_columns.keys, r.key()))
53        } else {
54            None
55        };
56        let radius_zoom_const = radius_prop.is_zoom_constant();
57
58        let color_prop = circle_paint.circle_color.as_ref();
59        let color_const = match color_prop {
60            None => Some([0.0, 0.0, 0.0, 1.0]),
61            Some(prop) if prop.is_feature_constant() => Some(
62                prop.evaluate_at_zoom(tile_zoom)
63                    .map_or([0.0, 0.0, 0.0, 1.0], |c| c.to_array()),
64            ),
65            Some(_) => None,
66        };
67        let color_ramp = color_prop.and_then(ColorRamp::compile);
68        let color_col = if use_fast {
69            color_ramp
70                .as_ref()
71                .and_then(|r| col_index(&fast_columns.keys, r.key()))
72        } else {
73            None
74        };
75
76        let opacity_prop = circle_paint.circle_opacity.as_ref();
77        let opacity_const = match opacity_prop {
78            None => Some(1.0_f32),
79            Some(prop) if prop.is_feature_constant() => Some(
80                prop.evaluate_at_zoom(tile_zoom)
81                    .unwrap_or(1.0)
82                    .clamp(0.0, 1.0),
83            ),
84            Some(_) => None,
85        };
86        let opacity_ramp = opacity_prop.and_then(ScalarRamp::compile);
87        let opacity_col = if use_fast {
88            opacity_ramp
89                .as_ref()
90                .and_then(|r| col_index(&fast_columns.keys, r.key()))
91        } else {
92            None
93        };
94
95        if use_fast {
96            for row in &fast_columns.values {
97                let mut rgba = color_const.unwrap_or_else(|| {
98                    color_ramp
99                        .as_ref()
100                        .and_then(|r| r.eval_f64(color_col.and_then(|c| row[c])))
101                        .unwrap_or([0.0, 0.0, 0.0, 1.0])
102                });
103                let opacity = opacity_const.unwrap_or_else(|| {
104                    opacity_ramp
105                        .as_ref()
106                        .and_then(|r| r.eval_f64(opacity_col.and_then(|c| row[c])))
107                        .unwrap_or(1.0)
108                        .clamp(0.0, 1.0)
109                });
110                rgba[3] *= opacity;
111
112                let (radius, radius_next) = if let Some(r) = radius_const {
113                    (r, r)
114                } else {
115                    let r0 = radius_ramp
116                        .as_ref()
117                        .and_then(|r| r.eval_f64(radius_col.and_then(|c| row[c])))
118                        .unwrap_or(CirclePaint::DEFAULT_RADIUS)
119                        .max(0.0);
120                    (r0, r0)
121                };
122                cache.pairs.push([
123                    FillShaderFeatureMetadata::from_rgba(rgba),
124                    pack_dynamic_circle_radii(radius, radius_next),
125                ]);
126            }
127        } else {
128            for props in all_props {
129                let mut rgba = color_const.unwrap_or_else(|| {
130                    color_ramp
131                        .as_ref()
132                        .and_then(|r| r.eval(props))
133                        .or_else(|| {
134                            color_prop
135                                .and_then(|p| p.evaluate_for(props, tile_zoom))
136                                .map(|c| c.to_array())
137                        })
138                        .unwrap_or([0.0, 0.0, 0.0, 1.0])
139                });
140                let opacity = opacity_const.unwrap_or_else(|| {
141                    opacity_ramp
142                        .as_ref()
143                        .and_then(|r| r.eval(props))
144                        .or_else(|| opacity_prop.and_then(|p| p.evaluate_for(props, tile_zoom)))
145                        .unwrap_or(1.0)
146                        .clamp(0.0, 1.0)
147                });
148                rgba[3] *= opacity;
149
150                let (radius, radius_next) = if let Some(r) = radius_const {
151                    (r, r)
152                } else {
153                    let r0 = radius_ramp
154                        .as_ref()
155                        .and_then(|r| r.eval(props))
156                        .or_else(|| radius_prop.evaluate_for(props, tile_zoom))
157                        .unwrap_or(CirclePaint::DEFAULT_RADIUS)
158                        .max(0.0);
159                    let r1 = if radius_zoom_const {
160                        r0
161                    } else {
162                        radius_prop
163                            .evaluate_for(props, tile_zoom + 1.0)
164                            .unwrap_or(CirclePaint::DEFAULT_RADIUS)
165                            .max(0.0)
166                    };
167                    (r0, r1)
168                };
169                cache.pairs.push([
170                    FillShaderFeatureMetadata::from_rgba(rgba),
171                    pack_dynamic_circle_radii(radius, radius_next),
172                ]);
173            }
174        }
175        cache.key = Some(cache_key);
176    }
177
178    let pairs = &cache.pairs;
179    if verts_per_circle == 4 {
180        for &feat_idx in hits {
181            if let Some(&pair) = pairs.get(feat_idx as usize) {
182                out.extend_from_slice(&[
183                    pair[0], pair[1], pair[0], pair[1], pair[0], pair[1], pair[0], pair[1],
184                ]);
185            }
186        }
187    } else {
188        for &feat_idx in hits {
189            if let Some(&pair) = pairs.get(feat_idx as usize) {
190                for _ in 0..verts_per_circle {
191                    out.extend_from_slice(&pair);
192                }
193            }
194        }
195    }
196}

198pub(super) fn fallback_dynamic_circle_from_vertices(
199    out: &mut Vec<FillShaderFeatureMetadata>,
200    feature_indices: &[u32],
201    active_colors: &[[f32; 4]],
202    fallback_color: [f32; 4],
203    vertices: &[crate::render::ShaderVertex],
204) {
205    out.clear();
206    let needed = vertices.len() * 2;
207    if out.capacity() < needed {
208        out.reserve(needed - out.len());
209    }
210    let mut vert_cursor = 0_usize;
211    for (feat_idx, &count) in feature_indices.iter().enumerate() {
212        let color = active_colors
213            .get(feat_idx)
214            .copied()
215            .unwrap_or(fallback_color);
216        let fill_meta = FillShaderFeatureMetadata::from_rgba(color);
217        for _ in 0..count {
218            let Some(v) = vertices.get(vert_cursor) else {
219                break;
220            };
221            vert_cursor += 1;
222            out.extend_from_slice(&[
223                fill_meta,
224                pack_dynamic_circle_radii(v.normal[0], v.edge_distance),
225            ]);
226        }
227    }
228    while vert_cursor < vertices.len() {
229        let v = &vertices[vert_cursor];
230        vert_cursor += 1;
231        out.extend_from_slice(&[
232            FillShaderFeatureMetadata::from_rgba(fallback_color),
233            pack_dynamic_circle_radii(v.normal[0], v.edge_distance),
234        ]);
235    }
