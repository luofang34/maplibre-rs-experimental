1//! In-place GPU metadata fast path for stationary dynamic-point GeoJSON circle sources.
2//!
3//! When a GeoJSON `FeatureCollection` of `Point`s keeps identical feature IDs and coordinates
4//! across updates (such as real-time sensor telemetry), this module parses incoming UTF-8 JSON
5//! directly into [`FeatureProperties`] without `serde_json::Value` DOM allocation, skips bumping
6//! `source.generation` / `style.revision()` (zero worker re-tiling), and updates only visible
7//! tiles' `[fill_rgba8, packed_radius_f16x2]` GPU feature metadata in-place.

37/// World resource tracking stationary `Point` GeoJSON sources for zero-retiling GPU updates.
38#[derive(Default)]
39pub(crate) struct DynamicPointGeoJson {
40    last_style_revision: Option<u64>,
41    sources: FxHashMap<String, DynamicPointSource>,
42    gpu_scratch: Vec<FillShaderFeatureMetadata>,
43}

45struct DynamicPointSource {
46    keys: Vec<(RawFeatureId, [u64; 2])>,
47    world_points: Vec<[f64; 2]>,
48    properties: Vec<FeatureProperties>,
49    scratch_props: Vec<FeatureProperties>,
50    promoted: Option<String>,
51    generate_id: bool,
52    inject_identity: bool,
53    seq: u64,
54    tile_hits: FxHashMap<WorldTileCoords, Vec<u32>>,
55    applied_seq: FxHashMap<(WorldTileCoords, StyleLayerId), (u64, u64)>,
56    deferred_bytes: Option<Arc<[u8]>>,
57}

59fn compute_tile_hits(world_points: &[[f64; 2]], coords: WorldTileCoords) -> Vec<u32> {
60    let TileWindow {
61        tiles,
62        west,
63        north,
64        bounds,
65    } = TileWindow::of(coords);
66    let extent = f64::from(EXTENT);
67    let extent_i32 = EXTENT as i32;
68    let mut hits = Vec::new();
69    for (idx, pt) in world_points.iter().enumerate() {
70        if !(bounds[0]..=bounds[2]).contains(&pt[0]) || !(bounds[1]..=bounds[3]).contains(&pt[1]) {
71            continue;
72        }
73        let tx = ((pt[0] - west) * tiles * extent).round() as i32;
74        let ty = ((pt[1] - north) * tiles * extent).round() as i32;
75        if (0..extent_i32).contains(&tx) && (0..extent_i32).contains(&ty) {
76            hits.push(idx as u32);
77        }
78    }
79    hits
80}

82/// Returns true when `style_layer` uses [`CircleDynamicRadiusShader`] (`2 * vertices` metadata).
83pub(crate) fn uses_dynamic_circle_metadata(style: &Style, style_layer: &StyleLayer) -> bool {
84    let Some(source_name) = style_layer.source.as_deref() else {
85        return false;
86    };
87    layer_supports_dynamic_points(style_layer) && source_supports_dynamic_points(style, source_name)
88}

90/// Returns true when `source_name` is an unclustered, unfiltered GeoJSON source whose referencing
91/// style layers all support in-place dynamic circle metadata updates.
92pub(crate) fn source_supports_dynamic_points(style: &Style, source_name: &str) -> bool {
93    let Some(Source::GeoJson(src)) = style.sources.get(source_name) else {
94        return false;
95    };
96    if src.cluster || src.filter.is_some() || src.line_metrics {
97        return false;
98    }
99    let mut has_any_layer = false;
100    for layer in &style.layers {
101        if layer.source.as_deref() != Some(source_name) {
102            continue;
103        }
104        has_any_layer = true;
105        if !layer_supports_dynamic_points(layer) {
106            return false;
107        }
108    }
109    has_any_layer
110}

112fn layer_supports_dynamic_points(style_layer: &StyleLayer) -> bool {
113    if style_layer.type_ != "circle"
114        || style_layer.filter.is_some()
115        || style_layer
116            .unrecognized
117            .layout
118            .contains_key("circle-sort-key")
119        || layer_paint_reads_feature_state(style_layer)
120        || layer_geometry_reads_feature_state(style_layer)
121    {
122        return false;
123    }
124    let Some(LayerPaint::Circle(paint)) = style_layer.paint.as_ref() else {
125        return false;
126    };
127    let radius_varies = !paint.radius().is_feature_constant();
128    let color_varies = paint
129        .circle_color
130        .as_ref()
131        .is_some_and(|c| !c.is_feature_constant());
132    let opacity_varies = paint
133        .circle_opacity
134        .as_ref()
135        .is_some_and(|o| !o.is_feature_constant());
136    if !(radius_varies || color_varies || opacity_varies) {
137        return false;
138    }
139    paint.stroke_color_per_feature().is_none()
140        && paint
141            .circle_stroke_width
142            .as_ref()
143            .is_none_or(StyleProperty::is_feature_constant)
144        && paint
145            .circle_blur
146            .as_ref()
147            .is_none_or(StyleProperty::is_feature_constant)
148        && paint
149            .circle_stroke_opacity
150            .as_ref()
151            .is_none_or(StyleProperty::is_feature_constant)
152        && !prop_reads_spatial(paint.circle_radius.as_ref())
153        && !prop_reads_spatial(paint.circle_color.as_ref())
154        && !prop_reads_spatial(paint.circle_opacity.as_ref())
155}

157fn prop_reads_spatial<T>(prop: Option<&StyleProperty<T>>) -> bool {
158    match prop {
159        Some(StyleProperty::Expression(expr)) => json_reads_spatial(expr.source()),
160        _ => false,
161    }
162}
163
164fn json_reads_spatial(value: &JsonValue) -> bool {
165    match value {
166        JsonValue::Array(items) => {
167            matches!(
168                items.first().and_then(JsonValue::as_str),
169                Some("within" | "distance")
170            ) || items.iter().any(json_reads_spatial)
171        }
172        JsonValue::Object(map) => map.values().any(json_reads_spatial),
173        _ => false,
174    }
175}

177fn source_layers_read_identity(style: &Style, source_name: &str) -> bool {
178    style.layers.iter().any(|layer| {
179        layer.source.as_deref() == Some(source_name)
180            && match layer.paint.as_ref() {
181                Some(LayerPaint::Circle(p)) => {
182                    prop_reads_identity(p.circle_radius.as_ref())
183                        || prop_reads_identity(p.circle_color.as_ref())
184                        || prop_reads_identity(p.circle_opacity.as_ref())
185                }
186                _ => false,
187            }
188    })
189}
190
191fn prop_reads_identity<T>(prop: Option<&StyleProperty<T>>) -> bool {
192    match prop {
193        Some(StyleProperty::Expression(expr)) => reads_identity(expr.source()),
194        _ => false,
195    }
196}

198/// Materializes any deferred stationary `bytes` back into `GeoJsonSource::data` before a
199/// non-stationary style/source mutation or diff.
200pub(crate) fn flush_deferred_geojson_bytes(style: &mut Style, world: &mut World) {
201    let Some(state) = world.resources.get_mut::<DynamicPointGeoJson>() else {
202        return;
203    };
204    for (name, src_state) in &mut state.sources {
205        let Some(bytes) = src_state.deferred_bytes.take() else {
206            continue;
207        };
208        if let Some(Source::GeoJson(src)) = style.sources.get_mut(name) {
209            if let Ok(doc) = serde_json::from_slice::<JsonValue>(&bytes) {
210                src.data = GeoJsonData::Inline(Arc::new(doc));
211            }
212        }
213    }
214}

216/// Invalidates `applied_seq` across tracked dynamic sources when style paint or layer order edits
217/// occur so resident `CircleDynamicRadiusShader` allocations re-evaluate in-place on next frame.
218pub(crate) fn invalidate_after_style_change(style: &Style, world: &mut World) {
219    let Some(state) = world.resources.get_mut::<DynamicPointGeoJson>() else {
220        return;
221    };
222    state.last_style_revision = None;
223    state
224        .sources
225        .retain(|name, _| source_supports_dynamic_points(style, name));
226    for (name, src_state) in &mut state.sources {
227        src_state.inject_identity = source_layers_read_identity(style, name);
228        src_state.seq = src_state.seq.wrapping_add(1);
229        src_state.applied_seq.clear();
230    }
231}

233/// Parses deferred stationary UTF-8 bytes on demand for read-only [`MapContext::query_source_features`].
234pub(crate) fn deferred_inline_value(world: &World, source_name: &str) -> Option<JsonValue> {
235    let bytes = world
236        .resources
237        .get::<DynamicPointGeoJson>()?
238        .sources
239        .get(source_name)?
240        .deferred_bytes
241        .as_ref()?;
242    serde_json::from_slice(bytes).ok()
243}

245/// Seeds or removes `source_name` in [`DynamicPointGeoJson`] from `style`'s current inline data.
246pub(crate) fn seed_from_inline(style: &Style, world: &mut World, source_name: &str) {
247    if !source_supports_dynamic_points(style, source_name) {
248        if let Some(state) = world.resources.get_mut::<DynamicPointGeoJson>() {
249            state.sources.remove(source_name);
250        }
251        return;
252    }
253    let Some(Source::GeoJson(src)) = style.sources.get(source_name) else {
254        return;
255    };
256    let GeoJsonData::Inline(doc) = &src.data else {
257        if let Some(state) = world.resources.get_mut::<DynamicPointGeoJson>() {
258            state.sources.remove(source_name);
259        }
260        return;
261    };
262    let inject_identity = source_layers_read_identity(style, source_name);
263    let Some(built) = build_source_from_value(doc, src, inject_identity) else {
264        if let Some(state) = world.resources.get_mut::<DynamicPointGeoJson>() {
265            state.sources.remove(source_name);
266        }
267        return;
268    };
269    let state = world.resources.get_or_init_mut::<DynamicPointGeoJson>();
270    state.sources.insert(source_name.to_owned(), built);
271}

273fn build_source_from_value(
274    doc: &JsonValue,
275    src: &GeoJsonSource,
276    inject_identity: bool,
277) -> Option<DynamicPointSource> {
278    if doc.get("type").and_then(JsonValue::as_str) != Some("FeatureCollection") {
279        return None;
280    }
281    let features = doc.get("features").and_then(JsonValue::as_array)?;
282    if features.is_empty() {
283        return None;
284    }
285    let promoted = promoted_property(src);
286    let n = features.len();
287    let mut keys = Vec::with_capacity(n);
288    let mut world_points = Vec::with_capacity(n);
289    let mut properties = Vec::with_capacity(n);
290
291    for (idx, feature) in features.iter().enumerate() {
292        if feature.get("type").and_then(JsonValue::as_str) != Some("Feature") {
293            return None;
294        }
295        let geom = feature.get("geometry")?;
296        if geom.get("type").and_then(JsonValue::as_str) != Some("Point") {
297            return None;
298        }
299        let coords = geom.get("coordinates").and_then(JsonValue::as_array)?;
300        let lon = coords.first().and_then(JsonValue::as_f64)?;
301        let lat = coords.get(1).and_then(JsonValue::as_f64)?;
302        let projected = project_lon_lat(lon, lat)?;
303        let raw_id = RawFeatureId::from_json(feature.get("id"));
304        let mut props = FeatureProperties::new();
305        fill_properties_from_json(
306            &mut props,
307            feature.get("properties"),
308            &raw_id,
309            promoted.as_deref(),
310            src.generate_id.then_some(idx),
311            inject_identity,
312        );
313        keys.push((raw_id, [lon.to_bits(), lat.to_bits()]));
314        world_points.push(projected);
315        properties.push(props);
316    }
317
318    let scratch_props = properties.clone();
319    Some(DynamicPointSource {
320        keys,
321        world_points,
322        properties,
323        scratch_props,
324        promoted,
325        generate_id: src.generate_id,
326        inject_identity,
327        seq: 1,
328        tile_hits: FxHashMap::default(),
329        applied_seq: FxHashMap::default(),
330        deferred_bytes: None,
331    })
332}

334/// Attempts a zero-DOM-allocation stationary update from UTF-8 JSON `bytes`.
335pub(crate) fn try_apply_stationary_bytes(
336    style: &Style,
337    world: &mut World,
338    source_name: &str,
339    bytes: &[u8],
340) -> bool {
341    if !source_supports_dynamic_points(style, source_name) {
342        return false;
343    }
344    let needs_seed = world
345        .resources
346        .get::<DynamicPointGeoJson>()
347        .is_none_or(|s| !s.sources.contains_key(source_name));
348    if needs_seed {
349        seed_from_inline(style, world, source_name);
350    }
351    let Some(state) = world.resources.get_mut::<DynamicPointGeoJson>() else {
352        return false;
353    };
354    let Some(src_state) = state.sources.get_mut(source_name) else {
355        return false;
356    };
357    if !try_parse_stationary_bytes(
358        bytes,
359        &src_state.keys,
360        &mut src_state.scratch_props,
361        src_state.promoted.as_deref(),
362        src_state.generate_id,
363        src_state.inject_identity,
364    ) {
365        return false;
366    }
367    std::mem::swap(&mut src_state.properties, &mut src_state.scratch_props);
368    src_state.seq = src_state.seq.wrapping_add(1);
369    src_state.deferred_bytes = Some(Arc::from(bytes));
370    crate::render::frame_signals::mark_dirty(world);
371    true
372}

374/// Attempts an in-place stationary update from an inline `serde_json::Value` document.
375pub(crate) fn try_apply_stationary_inline(
376    style: &mut Style,
377    world: &mut World,
378    source_name: &str,
379    doc: &Arc<JsonValue>,
380) -> bool {
381    if !source_supports_dynamic_points(style, source_name) {
382        return false;
383    }
384    let needs_seed = world
385        .resources
386        .get::<DynamicPointGeoJson>()
387        .is_none_or(|s| !s.sources.contains_key(source_name));
388    if needs_seed {
389        seed_from_inline(style, world, source_name);
390    }
391    let Some(state) = world.resources.get_mut::<DynamicPointGeoJson>() else {
392        return false;
393    };
394    let Some(src_state) = state.sources.get_mut(source_name) else {
395        return false;
396    };
397    if doc.get("type").and_then(JsonValue::as_str) != Some("FeatureCollection") {
398        return false;
399    }
400    let Some(features) = doc.get("features").and_then(JsonValue::as_array) else {
401        return false;
402    };
403    if features.len() != src_state.keys.len() || features.is_empty() {
404        return false;
405    }
406    for (idx, (feature, expected)) in features.iter().zip(src_state.keys.iter()).enumerate() {
407        if feature.get("type").and_then(JsonValue::as_str) != Some("Feature") {
408            return false;
409        }
410        let Some(geom) = feature.get("geometry") else {
411            return false;
412        };
413        if geom.get("type").and_then(JsonValue::as_str) != Some("Point") {
414            return false;
415        }
416        let Some(coords) = geom.get("coordinates").and_then(JsonValue::as_array) else {
417            return false;
418        };
419        let (Some(lon), Some(lat)) = (
420            coords.first().and_then(JsonValue::as_f64),
421            coords.get(1).and_then(JsonValue::as_f64),
422        ) else {
423            return false;
424        };
425        if !lon.is_finite()
426            || !lat.is_finite()
427            || [lon.to_bits(), lat.to_bits()] != expected.1
428            || RawFeatureId::from_json(feature.get("id")) != expected.0
429        {
430            return false;
431        }
432        fill_properties_from_json(
433            &mut src_state.scratch_props[idx],
434            feature.get("properties"),
435            &expected.0,
436            src_state.promoted.as_deref(),
437            src_state.generate_id.then_some(idx),
438            src_state.inject_identity,
439        );
440    }
441    std::mem::swap(&mut src_state.properties, &mut src_state.scratch_props);
442    src_state.seq = src_state.seq.wrapping_add(1);
443    src_state.deferred_bytes = None;
444    if let Some(Source::GeoJson(src)) = style.sources.get_mut(source_name) {
445        src.data = GeoJsonData::Inline(Arc::clone(doc));
446    }
447    crate::render::frame_signals::mark_dirty(world);
448    true
