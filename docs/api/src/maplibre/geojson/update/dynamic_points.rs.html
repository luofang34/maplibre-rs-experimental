1//! In-place GPU metadata fast path for stationary dynamic-point GeoJSON circle sources.
2//!
3//! When a GeoJSON `FeatureCollection` of `Point`s keeps identical feature IDs and coordinates
4//! across updates (such as real-time sensor telemetry), this module parses incoming UTF-8 JSON
5//! directly into [`FeatureProperties`] without `serde_json::Value` DOM allocation, skips bumping
6//! `source.generation` / `style.revision()` (zero worker re-tiling), and updates only visible
7//! tiles' `[fill_rgba8, packed_radius_f16x2]` GPU feature metadata in-place.

71fn compute_tile_hits(world_points: &[[f64; 2]], coords: WorldTileCoords) -> Vec<u32> {
72    let TileWindow {
73        tiles,
74        west,
75        north,
76        bounds,
77    } = TileWindow::of(coords);
78    let extent = f64::from(EXTENT);
79    let extent_i32 = EXTENT as i32;
80    let mut hits = Vec::new();
81    for (idx, pt) in world_points.iter().enumerate() {
82        if !(bounds[0]..=bounds[2]).contains(&pt[0]) || !(bounds[1]..=bounds[3]).contains(&pt[1]) {
83            continue;
84        }
85        let tx = ((pt[0] - west) * tiles * extent).round() as i32;
86        let ty = ((pt[1] - north) * tiles * extent).round() as i32;
87        if (0..extent_i32).contains(&tx) && (0..extent_i32).contains(&ty) {
88            hits.push(idx as u32);
89        }
90    }
91    hits
92}

94/// Materializes any deferred stationary `bytes` back into `GeoJsonSource::data` before a
95/// non-stationary style/source mutation or diff.
96pub(crate) fn flush_deferred_geojson_bytes(style: &mut Style, world: &mut World) {
97    let Some(state) = world.resources.get_mut::<DynamicPointGeoJson>() else {
98        return;
99    };
100    for (name, src_state) in &mut state.sources {
101        let Some(bytes) = src_state.deferred_bytes.take() else {
102            continue;
103        };
104        if let Some(Source::GeoJson(src)) = style.sources.get_mut(name) {
105            if let Ok(doc) = serde_json::from_slice::<JsonValue>(&bytes) {
106                src.data = GeoJsonData::Inline(Arc::new(doc));
107            }
108        }
109    }
110}

112/// Invalidates `applied_seq` across tracked dynamic sources when style paint or layer order edits
113/// occur so resident `CircleDynamicRadiusShader` allocations re-evaluate in-place on next frame.
114pub(crate) fn invalidate_after_style_change(style: &Style, world: &mut World) {
115    let Some(state) = world.resources.get_mut::<DynamicPointGeoJson>() else {
116        return;
117    };
118    state.last_style_revision = None;
119    state
120        .sources
121        .retain(|name, _| source_supports_dynamic_points(style, name));
122    for (name, src_state) in &mut state.sources {
123        if src_state.fast_columns.active_seq == Some(src_state.seq) {
124            if let Some(bytes) = src_state.deferred_bytes.as_deref() {
125                if try_parse_stationary_bytes(
126                    bytes,
127                    &src_state.keys,
128                    &mut src_state.scratch_props,
129                    src_state.promoted.as_deref(),
130                    src_state.generate_id,
131                    src_state.inject_identity,
132                ) {
133                    std::mem::swap(&mut src_state.properties, &mut src_state.scratch_props);
134                }
135            }
136            src_state.fast_columns.active_seq = None;
137        }
138        src_state.inject_identity = source_layers_read_identity(style, name);
139        src_state.seq = src_state.seq.wrapping_add(1);
140        src_state.applied_seq.clear();
141    }
142}

144/// Parses deferred stationary UTF-8 bytes on demand for read-only [`MapContext::query_source_features`].
145pub(crate) fn deferred_inline_value(world: &World, source_name: &str) -> Option<JsonValue> {
146    let src = world
147        .resources
148        .get::<DynamicPointGeoJson>()?
149        .sources
150        .get(source_name)?;
151    src.deferred_bytes
152        .as_deref()
153        .and_then(|bytes| serde_json::from_slice(bytes).ok())
154}

156/// Seeds or removes `source_name` in [`DynamicPointGeoJson`] from `style`'s current inline data.
157pub(crate) fn seed_from_inline(style: &Style, world: &mut World, source_name: &str) {
158    if !source_supports_dynamic_points(style, source_name) {
159        if let Some(state) = world.resources.get_mut::<DynamicPointGeoJson>() {
160            state.sources.remove(source_name);
161        }
162        return;
163    }
164    let Some(Source::GeoJson(src)) = style.sources.get(source_name) else {
165        return;
166    };
167    let GeoJsonData::Inline(doc) = &src.data else {
168        if let Some(state) = world.resources.get_mut::<DynamicPointGeoJson>() {
169            state.sources.remove(source_name);
170        }
171        return;
172    };
173    let inject_identity = source_layers_read_identity(style, source_name);
174    let Some(built) = build_source_from_value(doc, src, inject_identity) else {
175        if let Some(state) = world.resources.get_mut::<DynamicPointGeoJson>() {
176            state.sources.remove(source_name);
177        }
178        return;
179    };
180    let state = world.resources.get_or_init_mut::<DynamicPointGeoJson>();
181    state.sources.insert(source_name.to_owned(), built);
182}

184fn build_source_from_value(
185    doc: &JsonValue,
186    src: &GeoJsonSource,
187    inject_identity: bool,
188) -> Option<DynamicPointSource> {
189    if doc.get("type").and_then(JsonValue::as_str) != Some("FeatureCollection") {
190        return None;
191    }
192    let features = doc.get("features").and_then(JsonValue::as_array)?;
193    if features.is_empty() {
194        return None;
195    }
196    let promoted = promoted_property(src);
197    let n = features.len();
198    let mut keys = Vec::with_capacity(n);
199    let mut world_points = Vec::with_capacity(n);
200    let mut properties = Vec::with_capacity(n);
201
202    for (idx, feature) in features.iter().enumerate() {
203        if feature.get("type").and_then(JsonValue::as_str) != Some("Feature") {
204            return None;
205        }
206        let geom = feature.get("geometry")?;
207        if geom.get("type").and_then(JsonValue::as_str) != Some("Point") {
208            return None;
209        }
210        let coords = geom.get("coordinates").and_then(JsonValue::as_array)?;
211        let lon = coords.first().and_then(JsonValue::as_f64)?;
212        let lat = coords.get(1).and_then(JsonValue::as_f64)?;
213        let projected = project_lon_lat(lon, lat)?;
214        let raw_id = RawFeatureId::from_json(feature.get("id"));
215        let mut props = FeatureProperties::default();
216        fill_properties_from_json(
217            &mut props,
218            feature.get("properties"),
219            &raw_id,
220            promoted.as_deref(),
221            src.generate_id.then_some(idx),
222            inject_identity,
223        );
224        keys.push((raw_id, [lon.to_bits(), lat.to_bits()]));
225        world_points.push(projected);
226        properties.push(props);
227    }
228
229    let scratch_props = properties.clone();
230    Some(DynamicPointSource {
231        keys,
232        world_points,
233        properties,
234        scratch_props,
235        promoted,
236        generate_id: src.generate_id,
237        inject_identity,
238        seq: 1,
239        tile_hits: FxHashMap::default(),
240        applied_seq: FxHashMap::default(),
241        deferred_bytes: None,
242        fast_columns: FastColumnStore::default(),
243        eval_cache: EvalCache::default(),
244    })
245}

247fn store_deferred_bytes(slot: &mut Option<Vec<u8>>, bytes: Cow<'_, [u8]>) {
248    match bytes {
249        Cow::Owned(owned) => *slot = Some(owned),
250        Cow::Borrowed(borrowed) => {
251            let buf = slot.get_or_insert_with(Vec::new);
252            buf.clear();
253            buf.extend_from_slice(borrowed);
254        }
255    }
256}

258/// Attempts a zero-DOM-allocation stationary update from UTF-8 JSON `bytes`, moving owned buffers
259/// directly without copying on success and returning the buffer back on `Err` otherwise.
260pub(crate) fn try_apply_stationary_bytes<'a>(
261    style: &Style,
262    world: &mut World,
263    source_name: &str,
264    bytes: Cow<'a, [u8]>,
265) -> Result<(), Cow<'a, [u8]>> {
266    if !source_supports_dynamic_points(style, source_name) {
267        return Err(bytes);
268    }
269    let needs_seed = world
270        .resources
271        .get::<DynamicPointGeoJson>()
272        .is_none_or(|s| !s.sources.contains_key(source_name));
273    if needs_seed {
274        seed_from_inline(style, world, source_name);
275    }
276    let Some(state) = world.resources.get_mut::<DynamicPointGeoJson>() else {
277        return Err(bytes);
278    };
279    let Some(src_state) = state.sources.get_mut(source_name) else {
280        return Err(bytes);
281    };
282    if !src_state.inject_identity
283        && src_state.promoted.is_none()
284        && collect_fast_ramp_keys(style, source_name, &mut src_state.fast_columns.keys)
285        && try_fast_scan_stationary_bytes(&bytes, &src_state.keys, &mut src_state.fast_columns)
286    {
287        src_state.seq = src_state.seq.wrapping_add(1);
288        src_state.fast_columns.active_seq = Some(src_state.seq);
289        store_deferred_bytes(&mut src_state.deferred_bytes, bytes);
290        crate::render::frame_signals::mark_dirty(world);
291        return Ok(());
292    }
293    src_state.fast_columns.active_seq = None;
294    if !try_parse_stationary_bytes(
295        &bytes,
296        &src_state.keys,
297        &mut src_state.scratch_props,
298        src_state.promoted.as_deref(),
299        src_state.generate_id,
300        src_state.inject_identity,
301    ) {
302        return Err(bytes);
303    }
304    std::mem::swap(&mut src_state.properties, &mut src_state.scratch_props);
305    src_state.seq = src_state.seq.wrapping_add(1);
306    store_deferred_bytes(&mut src_state.deferred_bytes, bytes);
307    crate::render::frame_signals::mark_dirty(world);
308    Ok(())
309}

311/// Attempts an in-place stationary update from an inline `serde_json::Value` document.
312pub(crate) fn try_apply_stationary_inline(
313    style: &mut Style,
314    world: &mut World,
315    source_name: &str,
316    doc: &Arc<JsonValue>,
317) -> bool {
318    if !source_supports_dynamic_points(style, source_name) {
319        return false;
320    }
321    let needs_seed = world
322        .resources
323        .get::<DynamicPointGeoJson>()
324        .is_none_or(|s| !s.sources.contains_key(source_name));
325    if needs_seed {
326        seed_from_inline(style, world, source_name);
327    }
328    let Some(state) = world.resources.get_mut::<DynamicPointGeoJson>() else {
329        return false;
330    };
331    let Some(src_state) = state.sources.get_mut(source_name) else {
332        return false;
333    };
334    if doc.get("type").and_then(JsonValue::as_str) != Some("FeatureCollection") {
335        return false;
336    }
337    let Some(features) = doc.get("features").and_then(JsonValue::as_array) else {
338        return false;
339    };
340    if features.len() != src_state.keys.len() || features.is_empty() {
341        return false;
342    }
343    for (idx, (feature, expected)) in features.iter().zip(src_state.keys.iter()).enumerate() {
344        if feature.get("type").and_then(JsonValue::as_str) != Some("Feature") {
345            return false;
346        }
347        let Some(geom) = feature.get("geometry") else {
348            return false;
349        };
350        if geom.get("type").and_then(JsonValue::as_str) != Some("Point") {
351            return false;
352        }
353        let Some(coords) = geom.get("coordinates").and_then(JsonValue::as_array) else {
354            return false;
355        };
356        let (Some(lon), Some(lat)) = (
357            coords.first().and_then(JsonValue::as_f64),
358            coords.get(1).and_then(JsonValue::as_f64),
359        ) else {
360            return false;
361        };
362        if !lon.is_finite()
363            || !lat.is_finite()
364            || [lon.to_bits(), lat.to_bits()] != expected.1
365            || RawFeatureId::from_json(feature.get("id")) != expected.0
366        {
367            return false;
368        }
369        fill_properties_from_json(
370            &mut src_state.scratch_props[idx],
371            feature.get("properties"),
372            &expected.0,
373            src_state.promoted.as_deref(),
374            src_state.generate_id.then_some(idx),
375            src_state.inject_identity,
376        );
377    }
378    std::mem::swap(&mut src_state.properties, &mut src_state.scratch_props);
379    src_state.seq = src_state.seq.wrapping_add(1);
380    src_state.fast_columns.active_seq = None;
381    src_state.deferred_bytes = None;
382    if let Some(Source::GeoJson(src)) = style.sources.get_mut(source_name) {
383        src.data = GeoJsonData::Inline(Arc::clone(doc));
384    }
385    crate::render::frame_signals::mark_dirty(world);
386    true
