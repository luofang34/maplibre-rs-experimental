1//! Fetching again only the loaded tiles that new GeoJSON data changes.
2//!
3//! A tile holds the features whose bounds meet its window, the tile and the index's buffer around
4//! it, and its content is made from those features alone: line widths, labels and the source's
5//! own `buffer` act on them inside the tile. So a feature's change reaches exactly the tiles
6//! whose window meets its old or its new bounds, at each tile's own zoom, and the other tiles
7//! would come back unchanged.
8

25/// The GeoJSON sources of a style as they were before a change.
26pub(crate) struct GeoJsonVersions(Vec<(String, GeoJsonSource)>);
27
28impl GeoJsonVersions {
29    /// Inline documents are shared, so this copies no feature.
30    pub(crate) fn of(style: &Style) -> Self {
31        Self(
32            style
33                .sources
34                .iter()
35                .filter_map(|(name, source)| match source {
36                    Source::GeoJson(source) => Some((name.clone(), source.clone())),
37                    _ => None,
38                })
39                .collect(),
40        )
41    }
42
43    /// Requests again the loaded vector tiles whose content the sources' new data changes; the
44    /// old content stays until the new tiles arrive. Sources added or removed since are left to
45    /// the layer changes that draw them.
46    pub(crate) fn refresh_changed(self, style: &Style, world: &mut World) {
47        let mut reached = Vec::new();
48        for (name, before) in &self.0 {
49            let Some(Source::GeoJson(after)) = style.sources.get(name) else {
50                continue;
51            };
52            if before.generation == after.generation {
53                continue;
54            }
55            match reach(before, after) {
56                Some(bounds) => reached.extend(bounds),
57                None => {
58                    tracing::debug!(source = %name, "GeoJSON change refreshes every loaded tile");
59                    tile_retry::refresh(world, RequestKind::Vector);
60                    return;
61                }
62            }
63        }
64        let tiles = tiles_meeting(world, &reached);
65        if !tiles.is_empty() {
66            tracing::debug!(
67                tiles = tiles.len(),
68                "GeoJSON change refreshes the tiles it reaches"
69            );
70            tile_retry::refresh_tiles(world, RequestKind::Vector, &tiles);
71        }
72    }
73}

75/// The loaded tiles whose window meets any of `bounds`.
76fn tiles_meeting(world: &World, bounds: &[[f64; 4]]) -> Vec<WorldTileCoords> {
77    if bounds.is_empty() {
78        return Vec::new();
79    }
80    world
81        .tiles
82        .tiles
83        .values()
84        .map(|tile| tile.coords)
85        .filter(|coords| {
86            let window = TileWindow::of(*coords);
87            bounds.iter().any(|bounds| window.meets(bounds))
88        })
89        .collect()
90}

92/// The bounds of every feature version a data change adds, alters or drops; `None` when the
93/// change can reach any tile: data behind a URL, clusters, which a moved point can regroup
94/// anywhere, other tiling options, or a document that cannot be read feature by feature.
95fn reach(before: &GeoJsonSource, after: &GeoJsonSource) -> Option<Vec<[f64; 4]>> {
96    let (GeoJsonData::Inline(old), GeoJsonData::Inline(new)) = (&before.data, &after.data) else {
97        return None;
98    };
99    if before.cluster || after.cluster || !same_tiling(before, after) {
100        return None;
101    }
102    let (old, new) = (features(old)?, features(new)?);
103    // A generated id is a feature's position, so every feature after a change of position moves.
104    let promoted = promoted_property(after);
105    let pairs = (!after.generate_id)
106        .then(|| pairs_by_id(&old, &new, promoted.as_deref()))
107        .flatten()
108        .unwrap_or_else(|| pairs_by_position(&old, &new));
109    let mut bounds = Vec::new();
110    for (old, new) in pairs.into_iter().filter(|(old, new)| old != new) {
111        for feature in [old, new].into_iter().flatten() {
112            bounds.extend(feature_bounds(feature).ok()?);
113        }
114    }
115    Some(bounds)
116}

118/// Whether every option that decides which features a tile holds, and with which ids, is the
119/// same.
120fn same_tiling(before: &GeoJsonSource, after: &GeoJsonSource) -> bool {
121    before.promote_id == after.promote_id
122        && before.generate_id == after.generate_id
123        && before.filter == after.filter
124        && before.buffer == after.buffer
125        && before.minzoom == after.minzoom
126        && before.maxzoom == after.maxzoom
127}

129/// The features of a document the index reads, or `None` for a root it refuses.
130fn features(document: &Value) -> Option<std::borrow::Cow<'_, [Value]>> {
131    match document.get("type").and_then(Value::as_str)? {
132        "FeatureCollection" | "Feature" | "Point" | "MultiPoint" | "LineString"
133        | "MultiLineString" | "Polygon" | "MultiPolygon" | "GeometryCollection" => {
134            Some(feature_list(document))
135        }
136        _ => None,
137    }
138}

140type Pair<'a> = (Option<&'a Value>, Option<&'a Value>);
141
142fn pairs_by_position<'a>(old: &'a [Value], new: &'a [Value]) -> Vec<Pair<'a>> {
143    (0..old.len().max(new.len()))
144        .map(|index| (old.get(index), new.get(index)))
145        .collect()
146}
147
148/// Each feature with the one of the same id, as feature edits address them; `None` unless every
149/// feature has an id of its own and the features kept are still in the order tiles draw them in.
150fn pairs_by_id<'a>(
151    old: &'a [Value],
152    new: &'a [Value],
153    promoted: Option<&str>,
154) -> Option<Vec<Pair<'a>>> {
155    let ids = |features: &[Value]| -> Option<Vec<String>> {
156        features
157            .iter()
158            .map(|feature| feature_id(feature, promoted).map(|id| id.to_string()))
159            .collect()
160    };
161    let (old_ids, new_ids) = (ids(old)?, ids(new)?);
162    let (old_at, new_at) = (positions(&old_ids), positions(&new_ids));
163    if old_at.len() != old_ids.len() || new_at.len() != new_ids.len() {
164        return None;
165    }
166    if kept(&old_ids, &new_at) != kept(&new_ids, &old_at) {
167        return None;
168    }
169    let changed = old_ids.iter().zip(old).map(|(id, feature)| {
170        let replacement = new_at.get(id.as_str()).and_then(|&index| new.get(index));
171        (Some(feature), replacement)
172    });
173    let added = new_ids
174        .iter()
175        .zip(new)
176        .filter(|(id, _)| !old_at.contains_key(id.as_str()))
177        .map(|(_, feature)| (None, Some(feature)));
178    Some(changed.chain(added).collect())
179}
180
181fn positions(ids: &[String]) -> HashMap<&str, usize> {
182    ids.iter()
183        .enumerate()
184        .map(|(index, id)| (id.as_str(), index))
185        .collect()
186}
187
188fn kept<'a>(ids: &'a [String], other: &HashMap<&str, usize>) -> Vec<&'a str> {
189    ids.iter()
190        .map(String::as_str)
191        .filter(|id| other.contains_key(id))
192        .collect()
193}
