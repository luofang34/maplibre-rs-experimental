1//! The shared, parse-once index of each GeoJSON source generation.
2
3use std::sync::Arc;
4
5use super::index::GeoJsonIndex;
6use crate::{
7    io::source_client::{HttpClient, SourceClient},
8    sdf::assets::{fetch, AssetCache, AssetFailure},
9    style::source::{GeoJsonData, GeoJsonSource},
10};
11
12impl AssetCache {
13    /// The index of a GeoJSON source, parsed and, for URL data, fetched once per generation
14    /// however many tiles ask for it at the same time.
15    pub(crate) async fn geojson_index<HC: HttpClient>(
16        &self,
17        client: &SourceClient<HC>,
18        name: &str,
19        source: &GeoJsonSource,
20    ) -> Result<Arc<GeoJsonIndex>, AssetFailure> {
21        let key = format!("geojson:{name}:{}", source.generation);
22        self.load(key, || async {
23            let index = match &source.data {
24                GeoJsonData::Inline(document) => GeoJsonIndex::from_value(document, source),
25                GeoJsonData::Url(url) => {
26                    let text = fetch(client, url, "GeoJSON document").await?;
27                    GeoJsonIndex::from_text(&text, source)
28                }
29            }
30            .map_err(|error| {
31                tracing::warn!(source = name, %error, "GeoJSON source cannot be indexed");
32                AssetFailure::Terminal(format!("GeoJSON source `{name}`: {error}"))
33            })?;
34            let bytes = index.approximate_bytes();
35            Ok((index, bytes))
36        })
37        .await
38    }
39}
