3use std::{

9use crate::{

11    sdf::assets::{fetch, AssetCache, AssetFailure},
12    style::source::{GeoJsonData, GeoJsonSource},
13};

15const MAX_PRIMED_INDICES: usize = 16;
16
17type PrimedQueue = VecDeque<(u64, Arc<GeoJsonIndex>)>;
18
19static PRIMED_INDICES: LazyLock<Mutex<PrimedQueue>> =
20    LazyLock::new(|| Mutex::new(VecDeque::with_capacity(MAX_PRIMED_INDICES)));
21
22/// Retains a pre-validated [`GeoJsonIndex`] built during [`crate::style::Style::set_geojson_data`]
23/// or [`crate::style::Style::update_geojson_data`] so in-process tiling reuses it without parsing
24/// and bulk-loading the R-tree a second time.
25pub(crate) fn prime_geojson_index(generation: u64, index: Arc<GeoJsonIndex>) {
26    let mut guard = PRIMED_INDICES.lock().unwrap_or_else(|e| e.into_inner());
27    if guard.len() >= MAX_PRIMED_INDICES {
28        guard.pop_front();
29    }
30    guard.push_back((generation, index));
31}
32
33fn primed_geojson_index(generation: u64) -> Option<Arc<GeoJsonIndex>> {
34    let guard = PRIMED_INDICES.lock().unwrap_or_else(|e| e.into_inner());
35    guard
36        .iter()
37        .rev()
38        .find_map(|(gen_id, idx)| (*gen_id == generation).then(|| Arc::clone(idx)))
39}
40
41impl AssetCache {
42    /// The index of a GeoJSON source, parsed and, for URL data, fetched once per generation
43    /// however many tiles ask for it at the same time.
44    pub(crate) async fn geojson_index<HC: HttpClient>(
45        &self,
46        client: &SourceClient<HC>,
47        name: &str,
48        source: &GeoJsonSource,
49    ) -> Result<Arc<GeoJsonIndex>, AssetFailure> {
50        if let Some(cached) = primed_geojson_index(source.generation) {
51            return Ok(cached);
52        }
53        let key = format!("geojson:{name}:{}", source.generation);
54        self.load(key, || async {
55            let index = match &source.data {
56                GeoJsonData::Inline(document) => GeoJsonIndex::from_value(document, source),
57                GeoJsonData::Url(url) => {
58                    let text = fetch(client, url, "GeoJSON document").await?;
59                    GeoJsonIndex::from_text(&text, source)
60                }
61            }
62            .map_err(|error| {
63                tracing::warn!(source = name, %error, "GeoJSON source cannot be indexed");
64                AssetFailure::Terminal(format!("GeoJSON source `{name}`: {error}"))
65            })?;
66            let bytes = index.approximate_bytes();
67            Ok((index, bytes))
68        })
69        .await
70    }
71}
