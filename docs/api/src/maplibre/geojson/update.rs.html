22/// Why a source's data could not be changed. The source keeps its previous data.
23#[derive(Debug, Error)]
24pub enum SourceUpdateError {
25    /// The style declares no source with this name.
26    #[error("style has no source `{source_name}`")]
27    UnknownSource {
28        /// The requested name.
29        source_name: String,
30    },
31    /// The source is not a GeoJSON source.
32    #[error("source `{source_name}` is not a GeoJSON source")]
33    NotGeoJson {
34        /// The requested name.
35        source_name: String,
36    },
37    /// A feature diff needs the document itself, which a URL source does not hold.
38    #[error(
39        "source `{source_name}` loads its data from a URL, so it cannot be updated by feature"
40    )]
41    NotInline {
42        /// The requested name.
43        source_name: String,
44    },
45    /// A feature to add carries no id to identify it by.
46    #[error("a feature added to `{source_name}` has no id")]
47    MissingId {
48        /// The source being changed.
49        source_name: String,
50    },
51    /// An update names a feature the source does not hold.
52    #[error("source `{source_name}` has no feature with id {id}")]
53    UnknownFeature {
54        /// The source being changed.
55        source_name: String,
56        /// The id that matched nothing.
57        id: Value,
58    },
59    /// The resulting document cannot be indexed.
60    #[error("the new data of `{source_name}` is invalid")]
61    Invalid {
62        /// The source being changed.
63        source_name: String,
64        /// What is wrong with the document.
65        #[source]
66        source: GeoJsonError,
67    },
68}

79/// A change to one feature, as GL JS `GeoJSONFeatureDiff`.
80#[derive(Debug, Clone, Default, Deserialize)]
81#[serde(rename_all = "camelCase")]
82pub struct FeatureUpdate {
83    /// Id of the feature to change.
84    pub id: Value,
85    /// Replaces the geometry.
86    #[serde(default)]
87    pub new_geometry: Option<Value>,
88    /// Properties to set.
89    #[serde(default)]
90    pub add_or_update_properties: Vec<PropertyValue>,
91    /// Properties to drop.
92    #[serde(default)]
93    pub remove_properties: Vec<String>,
94    /// Drops every property before the others are set.
95    #[serde(default)]
96    pub remove_all_properties: bool,
97}

99/// Feature-level changes to an inline source, as GL JS `GeoJSONSourceDiff`.
100#[derive(Debug, Clone, Default, Deserialize)]
101#[serde(rename_all = "camelCase")]
102pub struct GeoJsonDiff {
103    /// Drops every feature first.
104    #[serde(default)]
105    pub remove_all: bool,
106    /// Ids of features to drop.
107    #[serde(default)]
108    pub remove: Vec<Value>,
109    /// Features to add; one with an existing id replaces it.
110    #[serde(default)]
111    pub add: Vec<Value>,
112    /// Features to change.
113    #[serde(default)]
114    pub update: Vec<FeatureUpdate>,
115}

117impl Style {
118    /// Replaces the data of a GeoJSON source, as GL JS `setData`.
119    pub fn set_geojson_data(
120        &mut self,
121        source_name: &str,
122        data: GeoJsonData,
123    ) -> Result<(), SourceUpdateError> {
124        let source = geojson_source_mut(self, source_name)?;
125        if let GeoJsonData::Inline(document) = &data {
126            GeoJsonIndex::from_value(document, source).map_err(|source| {
127                SourceUpdateError::Invalid {
128                    source_name: source_name.to_owned(),
129                    source,
130                }
131            })?;
132        }
133        source.data = data;
134        source.generation = fresh_generation();
135        Ok(())
136    }
137
138    /// Adds, changes and removes features of an inline source, as GL JS `updateData`.
139    pub fn update_geojson_data(
140        &mut self,
141        source_name: &str,
142        diff: &GeoJsonDiff,
143    ) -> Result<(), SourceUpdateError> {
144        let source = geojson_source_mut(self, source_name)?;
145        let GeoJsonData::Inline(document) = &source.data else {
146            return Err(SourceUpdateError::NotInline {
147                source_name: source_name.to_owned(),
148            });
149        };
150        let promoted = promoted_property(source);
151        let mut features = feature_list(document).into_owned();
152        apply(&mut features, diff, promoted.as_deref(), source_name)?;
153        let updated = json!({"type": "FeatureCollection", "features": features});
154        GeoJsonIndex::from_value(&updated, source).map_err(|source| {
155            SourceUpdateError::Invalid {
156                source_name: source_name.to_owned(),
157                source,
158            }
159        })?;
160        source.data = GeoJsonData::Inline(Arc::new(updated));
161        source.generation = fresh_generation();
162        Ok(())
163    }
164}

166impl MapContext {
167    /// Replaces a GeoJSON source's data and refreshes the tiles drawn from it. The old tiles stay
168    /// on screen until the new ones arrive.
169    pub fn set_geojson_data(
170        &mut self,
171        source_name: &str,
172        data: GeoJsonData,
173    ) -> Result<(), SourceUpdateError> {
174        self.style.set_geojson_data(source_name, data)?;
175        tile_retry::refresh(&mut self.world, RequestKind::Vector);
176        Ok(())
177    }

179    /// Applies a feature diff to an inline GeoJSON source and refreshes its tiles.
180    pub fn update_geojson_data(
181        &mut self,
182        source_name: &str,
183        diff: &GeoJsonDiff,
184    ) -> Result<(), SourceUpdateError> {
185        self.style.update_geojson_data(source_name, diff)?;
186        tile_retry::refresh(&mut self.world, RequestKind::Vector);
187        Ok(())
188    }

191fn geojson_source_mut<'a>(
192    style: &'a mut Style,
193    name: &str,
194) -> Result<&'a mut GeoJsonSource, SourceUpdateError> {
195    match style.sources.get_mut(name) {
196        Some(Source::GeoJson(source)) => Ok(source),
197        Some(_) => Err(SourceUpdateError::NotGeoJson {
198            source_name: name.to_owned(),
199        }),
200        None => Err(SourceUpdateError::UnknownSource {
201            source_name: name.to_owned(),
202        }),
203    }
204}

206pub(super) fn promoted_property(source: &GeoJsonSource) -> Option<String> {
207    match source.promote_id.as_ref()? {
208        PromoteId::Property(name) => Some(name.clone()),
209        PromoteId::PerLayer(map) => map.get(GEOJSON_LAYER).cloned(),
210    }
211}

213/// The features of a document, whichever GeoJSON root it has.
214pub(super) fn feature_list(document: &Value) -> Cow<'_, [Value]> {
215    match document.get("type").and_then(Value::as_str) {
216        Some("FeatureCollection") => document
217            .get("features")
218            .and_then(Value::as_array)
219            .map_or(Cow::Owned(Vec::new()), |features| {
220                Cow::Borrowed(features.as_slice())
221            }),
222        Some("Feature") => Cow::Owned(vec![document.clone()]),
223        Some(_) => Cow::Owned(vec![
224            json!({"type": "Feature", "properties": {}, "geometry": document}),
225        ]),
226        None => Cow::Owned(Vec::new()),
227    }
228}

230/// The value a feature is addressed by: the promoted property, else its `id`.
231pub(super) fn feature_id(feature: &Value, promoted: Option<&str>) -> Option<Value> {
232    promoted
233        .and_then(|name| feature.get("properties")?.get(name))
234        .filter(|value| !value.is_null())
235        .or_else(|| feature.get("id"))
236        .cloned()
237}

239fn apply(
240    features: &mut Vec<Value>,
241    diff: &GeoJsonDiff,
242    promoted: Option<&str>,
243    source_name: &str,
244) -> Result<(), SourceUpdateError> {
245    if diff.remove_all {
246        features.clear();
247    }
248    features.retain(|feature| {
249        feature_id(feature, promoted).is_none_or(|id| !diff.remove.contains(&id))
250    });
251    for added in &diff.add {
252        let id = feature_id(added, promoted).ok_or_else(|| SourceUpdateError::MissingId {
253            source_name: source_name.to_owned(),
254        })?;
255        match features
256            .iter_mut()
257            .find(|feature| feature_id(feature, promoted).as_ref() == Some(&id))
258        {
259            Some(existing) => *existing = added.clone(),
260            None => features.push(added.clone()),
261        }
262    }
263    for update in &diff.update {
264        let feature = features
265            .iter_mut()
266            .find(|feature| feature_id(feature, promoted).as_ref() == Some(&update.id))
267            .ok_or_else(|| SourceUpdateError::UnknownFeature {
268                source_name: source_name.to_owned(),
269                id: update.id.clone(),
270            })?;
271        if let Some(geometry) = &update.new_geometry {
272            feature["geometry"] = geometry.clone();
273        }
274        if update.remove_all_properties {
275            feature["properties"] = json!({});
276        }
277        if !feature["properties"].is_object() {
278            feature["properties"] = json!({});
279        }
280        if let Some(properties) = feature["properties"].as_object_mut() {
281            for name in &update.remove_properties {
282                properties.remove(name);
283            }
284            for property in &update.add_or_update_properties {
285                properties.insert(property.key.clone(), property.value.clone());
286            }
287        }
288    }
289    Ok(())
290}
