1//! Direct `&serde_json::Value` streamer for `geozero::FeatureProcessor` without `geojson` re-parsing.
2
3use geozero::{
4    error::GeozeroError, ColumnValue, FeatureProcessor, GeomProcessor, PropertyProcessor,
5};
6use serde_json::{Map, Value};
7
8fn err(msg: impl Into<String>) -> GeozeroError {
9    GeozeroError::Geometry(msg.into())
10}
11
12/// Streams a GeoJSON `FeatureCollection`, `Feature`, or bare geometry directly from `value` into
13/// `processor`, emitting feature properties before geometry.
14pub(crate) fn process_geojson_value<P: FeatureProcessor>(
15    value: &Value,
16    processor: &mut P,
17) -> Result<(), GeozeroError> {
18    let kind = value
19        .get("type")
20        .and_then(Value::as_str)
21        .ok_or_else(|| err("GeoJSON object missing string `type`"))?;
22    match kind {
23        "FeatureCollection" => {
24            let features = value
25                .get("features")
26                .and_then(Value::as_array)
27                .ok_or_else(|| err("FeatureCollection missing `features` array"))?;
28            processor.dataset_begin(None)?;
29            for (idx, feature) in features.iter().enumerate() {
30                emit_feature(feature, idx as u64, processor)?;
31            }
32            processor.dataset_end()?;
33        }
34        "Feature" => {
35            processor.dataset_begin(None)?;
36            emit_feature(value, 0, processor)?;
37            processor.dataset_end()?;
38        }
39        _ => emit_geometry(value, processor, 0)?,
40    }
41    Ok(())
42}

44fn emit_feature<P: FeatureProcessor>(
45    feature: &Value,
46    idx: u64,
47    processor: &mut P,
48) -> Result<(), GeozeroError> {
49    processor.feature_begin(idx)?;
50    processor.properties_begin()?;
51    if let Some(props) = feature.get("properties").and_then(Value::as_object) {
52        emit_properties(props, processor)?;
53    }
54    processor.properties_end()?;
55    processor.geometry_begin()?;
56    if let Some(geometry) = feature.get("geometry").filter(|g| !g.is_null()) {
57        emit_geometry(geometry, processor, 0)?;
58    }
59    processor.geometry_end()?;
60    processor.feature_end(idx)?;
61    Ok(())
62}

64fn emit_properties<P: PropertyProcessor>(
65    props: &Map<String, Value>,
66    processor: &mut P,
67) -> Result<(), GeozeroError> {
68    for (idx, (key, value)) in props.iter().enumerate() {
69        let stop = match value {
70            Value::Null => false,
71            Value::Bool(flag) => processor.property(idx, key, &ColumnValue::Bool(*flag))?,
72            Value::Number(num) => {
73                if let Some(u) = num.as_u64() {
74                    processor.property(idx, key, &ColumnValue::ULong(u))?
75                } else if let Some(i) = num.as_i64() {
76                    processor.property(idx, key, &ColumnValue::Long(i))?
77                } else if let Some(f) = num.as_f64() {
78                    processor.property(idx, key, &ColumnValue::Double(f))?
79                } else {
80                    false
81                }
82            }
83            Value::String(text) => processor.property(idx, key, &ColumnValue::String(text))?,
84            Value::Array(_) | Value::Object(_) => {
85                let encoded = value.to_string();
86                processor.property(idx, key, &ColumnValue::Json(&encoded))?
87            }
88        };
89        if stop {
90            break;
91        }
92    }
93    Ok(())
94}

96fn emit_point_coords<P: GeomProcessor>(
97    coord: &Value,
98    idx: usize,
99    processor: &mut P,
100) -> Result<(), GeozeroError> {
101    let arr = coord
102        .as_array()
103        .ok_or_else(|| err("expected coordinate array"))?;
104    let x = arr
105        .first()
106        .and_then(Value::as_f64)
107        .ok_or_else(|| err("coordinate missing x"))?;
108    let y = arr
109        .get(1)
110        .and_then(Value::as_f64)
111        .ok_or_else(|| err("coordinate missing y"))?;
112    processor.xy(x, y, idx)
113}
114
115fn emit_coord_seq<P: GeomProcessor>(
116    coords: &[Value],
117    processor: &mut P,
118) -> Result<(), GeozeroError> {
119    for (i, pt) in coords.iter().enumerate() {
120        emit_point_coords(pt, i, processor)?;
121    }
122    Ok(())
123}
124
125fn emit_rings<P: GeomProcessor>(rings: &[Value], processor: &mut P) -> Result<(), GeozeroError> {
126    for (ri, ring) in rings.iter().enumerate() {
127        let pts = ring
128            .as_array()
129            .ok_or_else(|| err("polygon ring is not an array"))?;
130        processor.linestring_begin(false, pts.len(), ri)?;
131        emit_coord_seq(pts, processor)?;
132        processor.linestring_end(false, ri)?;
133    }
134    Ok(())
135}

137fn emit_geometry<P: GeomProcessor>(
138    geometry: &Value,
139    processor: &mut P,
140    idx: usize,
141) -> Result<(), GeozeroError> {
142    let kind = geometry
143        .get("type")
144        .and_then(Value::as_str)
145        .ok_or_else(|| err("geometry missing `type`"))?;
146    if kind == "GeometryCollection" {
147        let geoms = geometry
148            .get("geometries")
149            .and_then(Value::as_array)
150            .ok_or_else(|| err("GeometryCollection missing `geometries`"))?;
151        processor.geometrycollection_begin(geoms.len(), idx)?;
152        for (gi, child) in geoms.iter().enumerate() {
153            emit_geometry(child, processor, gi)?;
154        }
155        return processor.geometrycollection_end(idx);
156    }
157    let coords = geometry
158        .get("coordinates")
159        .ok_or_else(|| err(format!("{kind} missing `coordinates`")))?;
160    match kind {
161        "Point" => {
162            processor.point_begin(idx)?;
163            emit_point_coords(coords, 0, processor)?;
164            processor.point_end(idx)?;
165        }
166        "MultiPoint" => {
167            let pts = coords
168                .as_array()
169                .ok_or_else(|| err("MultiPoint coordinates not array"))?;
170            processor.multipoint_begin(pts.len(), idx)?;
171            emit_coord_seq(pts, processor)?;
172            processor.multipoint_end(idx)?;
173        }
174        "LineString" => {
175            let pts = coords
176                .as_array()
177                .ok_or_else(|| err("LineString coordinates not array"))?;
178            processor.linestring_begin(true, pts.len(), idx)?;
179            emit_coord_seq(pts, processor)?;
180            processor.linestring_end(true, idx)?;
181        }
182        "MultiLineString" => {
183            let lines = coords
184                .as_array()
185                .ok_or_else(|| err("MultiLineString coordinates not array"))?;
186            processor.multilinestring_begin(lines.len(), idx)?;
187            for (li, line) in lines.iter().enumerate() {
188                let pts = line
189                    .as_array()
190                    .ok_or_else(|| err("MultiLineString element not array"))?;
191                processor.linestring_begin(false, pts.len(), li)?;
192                emit_coord_seq(pts, processor)?;
193                processor.linestring_end(false, li)?;
194            }
195            processor.multilinestring_end(idx)?;
196        }
197        "Polygon" => {
198            let rings = coords
199                .as_array()
200                .ok_or_else(|| err("Polygon coordinates not array"))?;
201            processor.polygon_begin(true, rings.len(), idx)?;
202            emit_rings(rings, processor)?;
203            processor.polygon_end(true, idx)?;
204        }
205        "MultiPolygon" => {
206            let polys = coords
207                .as_array()
208                .ok_or_else(|| err("MultiPolygon coordinates not array"))?;
209            processor.multipolygon_begin(polys.len(), idx)?;
210            for (pi, poly) in polys.iter().enumerate() {
211                let rings = poly
212                    .as_array()
213                    .ok_or_else(|| err("MultiPolygon element not array"))?;
214                processor.polygon_begin(false, rings.len(), pi)?;
215                emit_rings(rings, processor)?;
216                processor.polygon_end(false, pi)?;
217            }
218            processor.multipolygon_end(idx)?;
219        }
220        _ => return Err(err(format!("unsupported GeoJSON geometry type `{kind}`"))),
221    }
222    Ok(())
