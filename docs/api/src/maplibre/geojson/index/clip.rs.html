1//! A feature's geometry cut to the window of the tile it is drawn in.
2//!
3//! The renderer tessellates and, on the globe, subdivides everything a tile carries, so a
4//! polygon left whole would cost each tile it touches as much as the polygon's whole area.
5
6use geo_types::{Coord, Rect};
7
8use super::{Geometry, Point, Ring};
9use crate::coords::clip::{line_within, ring_within};
10
11impl Geometry {
12    /// The parts of the geometry within `bounds`, `[west, north, east, south]` in world units.
13    pub(super) fn within(&self, bounds: [f64; 4]) -> Self {
14        let [west, north, east, south] = bounds;
15        let area = Rect::new(Coord { x: west, y: north }, Coord { x: east, y: south });
16        let polygons = self
17            .polygons
18            .iter()
19            .filter_map(|rings| {
20                let mut cut = rings
21                    .iter()
22                    .map(|ring| ring_within(&coords(ring), area))
23                    .map(|ring| (ring.len() >= 4).then(|| points(&ring)));
24                // Without its exterior ring a polygon has nothing for its holes to cut into.
25                let exterior = cut.next()??;
26                Some(std::iter::once(exterior).chain(cut.flatten()).collect())
27            })
28            .collect();
29        let lines = self
30            .lines
31            .iter()
32            .flat_map(|line| line_within(coords(line), area))
33            .map(|run| points(&run))
34            .collect();
35        let points = self
36            .points
37            .iter()
38            .filter(|point| {
39                (west..=east).contains(&point[0]) && (north..=south).contains(&point[1])
40            })
41            .copied()
42            .collect();
43        Self {
44            points,
45            lines,
46            polygons,
47        }
48    }
49}
50
51fn coords(ring: &[Point]) -> Vec<Coord<f64>> {
52    ring.iter()
53        .map(|point| Coord {
54            x: point[0],
55            y: point[1],
56        })
57        .collect()
58}
59
60fn points(coords: &[Coord<f64>]) -> Ring {
61    coords.iter().map(|coord| [coord.x, coord.y]).collect()
62}
