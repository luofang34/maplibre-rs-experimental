1//! Builds configured [`ZeroTessellator`] instances for GeoJSON style layers.
2
3use crate::{
4    coords::{WorldTileCoords, EXTENT},
5    projection::{globe::subdivision::granularity_for_zoom, ProjectionType},
6    style::layer::{LayerPaint, StyleLayer},
7    vector::tessellation::{
8        sharp_corner_offset, sort_key_of, CircleOptions, ExtrusionOptions, IndexDataType,
9        LineElevationOptions, LineFeatureStyle, SortKeys, ZeroTessellator,
10    },
11};
12
13pub(super) fn build_tessellator(
14    paint: &LayerPaint,
15    style_layer: &StyleLayer,
16    coords: WorldTileCoords,
17    projection: &ProjectionType,
18    (style_zoom, overscaling, buffer): (f64, f64, Option<u32>),
19) -> ZeroTessellator<IndexDataType> {
20    let zoom = u8::from(coords.z);
21    let granularity = match paint {
22        LayerPaint::Fill(_) => granularity_for_zoom(128, 2, zoom),
23        LayerPaint::Line(_) => granularity_for_zoom(512, 0, zoom),
24        _ => 1,
25    };
26    let use_globe_geometry = projection.uses_globe_rendering(f64::from(zoom));
27    let mut tessellator = match paint {
28        LayerPaint::Circle(circle) => {
29            ZeroTessellator::<IndexDataType>::default().with_circles(CircleOptions {
30                grid: use_globe_geometry
31                    && circle.circle_pitch_alignment
32                        == crate::style::circle::CirclePitchAlignment::Map,
33                ..CircleOptions::for_paint(circle, style_zoom)
34            })
35        }
36        LayerPaint::Heatmap(heatmap) => {
37            ZeroTessellator::<IndexDataType>::default().with_circles(CircleOptions {
38                grid: use_globe_geometry,
39                ..CircleOptions::for_heatmap(heatmap, style_zoom)
40            })
41        }
42        LayerPaint::FillExtrusion(extrusion) => {
43            let defaults = ExtrusionOptions::for_paint(extrusion, overscaling);
44            let clip_buffer = defaults.clip_buffer.or_else(|| {
45                buffer
46                    .filter(|_| extrusion.fill_extrusion_pattern.is_some())
47                    .map(|buf| buf as f32 * EXTENT as f32 / 512.0)
48            });
49            let tessellator =
50                ZeroTessellator::<IndexDataType>::default().with_extrusion(ExtrusionOptions {
51                    clip_buffer,
52                    ..defaults
53                });
54            let min_grid = if extrusion.may_use_vertex_agl() {
55                16
56            } else {
57                2
58            };
59            if use_globe_geometry {
60                let last_tile = i64::from(crate::coords::ZOOM_BOUNDS[usize::from(zoom)]) - 1;
61                tessellator.with_globe_subdivision(
62                    granularity_for_zoom(128, min_grid, zoom),
63                    zoom == 0,
64                    coords.y == 0,
65                    i64::from(coords.y) == last_tile,
66                )
67            } else if extrusion.may_use_vertex_agl() {
68                tessellator.with_globe_subdivision(16, false, false, false)
69            } else {
70                tessellator
71            }
72        }
73        _ if use_globe_geometry => {
74            let last_tile = i64::from(crate::coords::ZOOM_BOUNDS[usize::from(zoom)]) - 1;
75            ZeroTessellator::<IndexDataType>::default().with_globe_subdivision(
76                granularity,
77                zoom == 0,
78                coords.y == 0,
79                i64::from(coords.y) == last_tile,
80            )
81        }
82        _ => ZeroTessellator::<IndexDataType>::default(),
83    }
84    .with_feature_opacity(paint.opacity(), style_zoom);
85    tessellator.sort_key = SortKeys::by(sort_key_of(style_layer));
86    match paint {
87        LayerPaint::Fill(p) if p.fill_pattern.is_some() => {
88            tessellator.fallback_color = [1.0; 4];
89            tessellator.pattern_property = crate::vector::pattern::per_feature_pattern(paint);
90        }
91        LayerPaint::Fill(p) => {
92            tessellator.style_property = p.fill_color.clone();
93            tessellator.outline_property = p.outline_color();
94        }
95        LayerPaint::FillExtrusion(p) if p.fill_extrusion_pattern.is_some() => {
96            tessellator.fallback_color = [1.0; 4];
97            tessellator.pattern_property = crate::vector::pattern::per_feature_pattern(paint);
98        }
99        LayerPaint::FillExtrusion(p) => {
100            tessellator.style_property = p.fill_extrusion_color.clone();
101        }
102        LayerPaint::Circle(p) => tessellator.style_property = p.circle_color.clone(),
103        LayerPaint::Line(p) => {
104            tessellator.pattern_property = crate::vector::pattern::per_feature_pattern(paint);
105            tessellator.style_property = p.line_color.clone();
106            tessellator.is_line_layer = true;
107            tessellator.line_gradient = p.line_gradient.is_some();
108            tessellator.split_corners =
109                p.line_gradient.is_some() || p.line_pattern.is_some() || p.line_dasharray.is_some();
110            tessellator.line_feature_style = LineFeatureStyle::for_paint(p, style_zoom);
111            tessellator.line_elevation = LineElevationOptions::for_paint(p, style_zoom);
112            tessellator.sharp_corner_offset = sharp_corner_offset(overscaling);
113            tessellator.set_layer_strokes(style_layer);
114        }
115        LayerPaint::Background(p) => {
116            tessellator.style_property = p.background_color.clone();
117        }
118        _ => {}
119    }
120    tessellator
121}
