1//! Groups the points of a GeoJSON source into clusters for each zoom, as GL JS does with
2//! supercluster: points within a pixel radius of one another merge, level by level from the
3//! deepest zoom up, into a cluster at their weighted centre.
4
5use rstar::{primitives::GeomWithData, RTree};
6use serde_json::{json, Value};
7
8use crate::style::expression::{EvaluationContext, Expression, FeatureProperties, Value as Datum};
9
10/// Side of the tile the radius is measured on, in pixels.
11const TILE_PIXELS: f64 = 512.0;
12
13/// How a cluster property combines the values of the points and sub-clusters it holds.
14#[derive(Clone, Debug)]
15enum Reduce {
16    Sum,
17    Product,
18    Max,
19    Min,
20    Any,
21    All,
22    Expression(Expression),
23}
24
25impl Reduce {
26    fn parse_builtin(operator: &str) -> Option<Self> {
27        Some(match operator {
28            "+" => Self::Sum,
29            "*" => Self::Product,
30            "max" => Self::Max,
31            "min" => Self::Min,
32            "any" => Self::Any,
33            "all" => Self::All,
34            _ => return None,
35        })
36    }
37
38    fn combine(&self, left: &Datum, right: &Datum, context: &EvaluationContext<'_>) -> Datum {
39        let numbers = || Some((left.as_number()?, right.as_number()?));
40        let flags = || {
41            (
42                left.as_bool().unwrap_or(false),
43                right.as_bool().unwrap_or(false),
44            )
45        };
46        match self {
47            Self::Sum => numbers().map_or(Datum::Null, |(a, b)| Datum::Number(a + b)),
48            Self::Product => numbers().map_or(Datum::Null, |(a, b)| Datum::Number(a * b)),
49            Self::Max => numbers().map_or(Datum::Null, |(a, b)| Datum::Number(a.max(b))),
50            Self::Min => numbers().map_or(Datum::Null, |(a, b)| Datum::Number(a.min(b))),
51            Self::Any => Datum::Bool(flags().0 || flags().1),
52            Self::All => Datum::Bool(flags().0 && flags().1),
53            Self::Expression(expr) => expr.evaluate(context).unwrap_or(Datum::Null),
54        }
55    }
56}

58/// One cluster property: the expression mapping a point to its initial value and how values combine.
59struct Property {
60    name: String,
61    map: Expression,
62    source_key: Option<String>,
63    reduce: Reduce,
64}
65
66fn direct_property_key(expr: &Expression) -> Option<String> {
67    match expr {
68        Expression::Get { key, object: None } => match key.as_ref() {
69            Expression::Literal(Datum::String(name)) => Some(name.clone()),
70            _ => None,
71        },
72        _ => None,
73    }
74}

76/// The clustering a source asks for.
77pub struct ClusterOptions {
78    radius: f64,
79    max_zoom: u8,
80    min_points: usize,
81    properties: Vec<Property>,
82}
83
84impl ClusterOptions {
85    /// The options from a source declaration; properties that cannot be read are left out.
86    pub fn new(
87        radius: Option<f64>,
88        max_zoom: Option<u8>,
89        min_points: Option<usize>,
90        properties: Option<&Value>,
91        source_max_zoom: Option<u8>,
92    ) -> Self {
93        let max_zoom = max_zoom.unwrap_or_else(|| source_max_zoom.unwrap_or(18).saturating_sub(1));
94        Self {
95            radius: radius.unwrap_or(50.0),
96            max_zoom: max_zoom.min(24),
97            min_points: min_points.unwrap_or(2),
98            properties: properties
99                .and_then(Value::as_object)
100                .map(|map| {
101                    map.iter()
102                        .filter_map(|(name, spec)| Self::property(name, spec))
103                        .collect()
104                })
105                .unwrap_or_default(),
106        }
107    }
108
109    /// Parses `[operator, map]`, `[operator, ["accumulated"], map]`, or `[reduce_expr, map_expr]`.
110    fn property(name: &str, spec: &Value) -> Option<Property> {
111        let items = spec.as_array()?;
112        let (reduce, map_json) = match items.as_slice() {
113            [Value::String(op), map_json] => (Reduce::parse_builtin(op)?, map_json),
114            [Value::String(op), acc, map_json]
115                if acc
116                    .as_array()
117                    .is_some_and(|a| a.as_slice() == [json!("accumulated")]) =>
118            {
119                (Reduce::parse_builtin(op)?, map_json)
120            }
121            [reduce_json, map_json] => {
122                let reduce_expr = Expression::parse(reduce_json).ok()?;
123                (Reduce::Expression(reduce_expr), map_json)
124            }
125            _ => return None,
126        };
127        let map = Expression::parse(map_json).ok()?;
128        let source_key = direct_property_key(&map);
129        Some(Property {
130            name: name.to_owned(),
131            map,
132            source_key,
133            reduce,
134        })
135    }
136
137    fn values(&self, properties: &[(String, Value)]) -> (FeatureProperties, Vec<Datum>) {
138        let feature: FeatureProperties = properties
139            .iter()
140            .map(|(key, value)| (key.clone(), Datum::from_json(value)))
141            .collect();
142        let mapped = self
143            .properties
144            .iter()
145            .map(|property| {
146                property
147                    .map
148                    .evaluate(&EvaluationContext::for_feature(0.0, &feature))
149                    .unwrap_or(Datum::Null)
150            })
151            .collect();
152        (feature, mapped)
153    }
154}

156/// A point or a cluster on one level.
157#[derive(Clone)]
158struct Item {
159    position: [f64; 2],
160    /// Number of points a cluster holds; zero for a point.
161    count: usize,
162    /// The point's index, or the cluster's identifier.
163    id: usize,
164    /// Original feature properties when `count == 0`.
165    point_props: Option<FeatureProperties>,
166    /// Cluster property values, in the order of the options.
167    values: Vec<Datum>,
168}
169
170struct Level {
171    items: Vec<Item>,
172    tree: RTree<GeomWithData<[f64; 2], usize>>,
173}
174
175impl Level {
176    fn new(items: Vec<Item>) -> Self {
177        let entries = items
178            .iter()
179            .enumerate()
180            .map(|(index, item)| GeomWithData::new(item.position, index))
181            .collect();
182        Self {
183            items,
184            tree: RTree::bulk_load(entries),
185        }
186    }
187}

189/// What lies in a window at a zoom.
190pub enum Clustered {
191    /// A point that stayed alone: its index among the points given to [`Clusters::new`].
192    Point(usize, [f64; 2]),
193    /// A cluster: where it is and the properties it carries.
194    Cluster([f64; 2], u64, Vec<(String, Value)>),
195}
196
197/// A point to cluster: its world position and its properties.
198pub type ClusterInput<'a> = ([f64; 2], &'a [(String, Value)]);
199
200/// The points of a source clustered at every zoom.
201pub struct Clusters {
202    options: ClusterOptions,
203    levels: Vec<Level>,
204}
205
206impl Clusters {
207    /// Clusters `points`, given as world positions with their properties.
208    pub fn new(points: &[ClusterInput<'_>], options: ClusterOptions) -> Self {
209        let has_custom_reduce = options
210            .properties
211            .iter()
212            .any(|prop| matches!(prop.reduce, Reduce::Expression(_)));
213        let mut items: Vec<Item> = points
214            .iter()
215            .enumerate()
216            .map(|(index, (position, properties))| {
217                let (feature, values) = options.values(properties);
218                Item {
219                    position: *position,
220                    count: 0,
221                    id: index,
222                    point_props: has_custom_reduce.then_some(feature),
223                    values,
224                }
225            })
226            .collect();
227        let max_zoom = usize::from(options.max_zoom);
228        let mut levels: Vec<Option<Level>> = (0..=max_zoom + 1).map(|_| None).collect();
229        levels[max_zoom + 1] = Some(Level::new(items.clone()));
230        for zoom in (0..=max_zoom).rev() {
231            let Some(previous) = levels[zoom + 1].as_ref() else {
232                break;
233            };
234            items = Self::merge(previous, zoom, &options);
235            levels[zoom] = Some(Level::new(items.clone()));
236        }
237        Self {
238            options,
239            levels: levels.into_iter().flatten().collect(),
240        }
241    }
242
243    fn merge(previous: &Level, zoom: usize, options: &ClusterOptions) -> Vec<Item> {
244        let radius = options.radius / (TILE_PIXELS * 2.0_f64.powi(zoom as i32));
245        let mut merged_at = vec![usize::MAX; previous.items.len()];
246        let mut result = Vec::new();
247        for (index, item) in previous.items.iter().enumerate() {
248            if merged_at[index] <= zoom {
249                continue;
250            }
251            merged_at[index] = zoom;
252            let neighbours: Vec<usize> = previous
253                .tree
254                .locate_within_distance(item.position, radius * radius)
255                .map(|entry| entry.data)
256                .collect();
257            let own = item.count.max(1);
258            let total = own
259                + neighbours
260                    .iter()
261                    .filter(|neighbour| merged_at[**neighbour] > zoom)
262                    .map(|neighbour| previous.items[*neighbour].count.max(1))
263                    .sum::<usize>();
264            if total > own && total >= options.min_points {
265                let mut weighted = [item.position[0] * own as f64, item.position[1] * own as f64];
266                let mut values = item.values.clone();
267                for neighbour in neighbours {
268                    if merged_at[neighbour] <= zoom {
269                        continue;
270                    }
271                    merged_at[neighbour] = zoom;
272                    let other = &previous.items[neighbour];
273                    let weight = other.count.max(1) as f64;
274                    weighted[0] += other.position[0] * weight;
275                    weighted[1] += other.position[1] * weight;
276                    for (prop_idx, property) in options.properties.iter().enumerate() {
277                        if matches!(property.reduce, Reduce::Expression(_)) {
278                            let mut props = other.point_props.clone().unwrap_or_default();
279                            for (j, other_prop) in options.properties.iter().enumerate() {
280                                props.insert(other_prop.name.clone(), other.values[j].clone());
281                            }
282                            if let Some(src_key) = &property.source_key {
283                                props
284                                    .entry(src_key.clone())
285                                    .or_insert_with(|| other.values[prop_idx].clone());
286                            }
287                            props.insert(property.name.clone(), other.values[prop_idx].clone());
288                            let ctx = EvaluationContext {
289                                zoom: zoom as f64,
290                                properties: Some(&props),
291                                accumulated: Some(&values[prop_idx]),
292                                ..EvaluationContext::default()
293                            };
294                            values[prop_idx] = property.reduce.combine(
295                                &values[prop_idx],
296                                &other.values[prop_idx],
297                                &ctx,
298                            );
299                        } else {
300                            values[prop_idx] = property.reduce.combine(
301                                &values[prop_idx],
302                                &other.values[prop_idx],
303                                &EvaluationContext::default(),
304                            );
305                        }
306                    }
307                }
308                result.push(Item {
309                    position: [weighted[0] / total as f64, weighted[1] / total as f64],
310                    count: total,
311                    id: (index << 5) + zoom + 1 + previous.items.len(),
312                    point_props: None,
313                    values,
314                });
315            } else {
316                result.push(item.clone());
317                if total > 1 {
318                    for neighbour in neighbours {
319                        if merged_at[neighbour] > zoom {
320                            merged_at[neighbour] = zoom;
321                            result.push(previous.items[neighbour].clone());
322                        }
323                    }
324                }
325            }
326        }
327        result
328    }

330    /// The points and clusters inside `window` (`[west, north, east, south]` in world units)
331    /// at a tile zoom, in the order they were made.
332    pub fn within(&self, zoom: u8, window: [f64; 4]) -> Vec<Clustered> {
333        let level = &self.levels[usize::from(zoom).min(self.levels.len() - 1)];
334        let corners = ([window[0], window[1]], [window[2], window[3]]);
335        let mut hits: Vec<usize> = level
336            .tree
337            .locate_in_envelope_intersecting(&rstar::AABB::from_corners(corners.0, corners.1))
338            .map(|entry| entry.data)
339            .collect();
340        hits.sort_unstable();
341        hits.into_iter()
342            .map(|index| {
343                let item = &level.items[index];
344                if item.count == 0 {
345                    return Clustered::Point(item.id, item.position);
346                }
347                let mut properties = vec![
348                    ("cluster".to_owned(), json!(true)),
349                    ("cluster_id".to_owned(), json!(item.id)),
350                    ("point_count".to_owned(), json!(item.count)),
351                    (
352                        "point_count_abbreviated".to_owned(),
353                        abbreviated(item.count),
354                    ),
355                ];
356                for (property, value) in self.options.properties.iter().zip(&item.values) {
357                    properties.push((property.name.clone(), value.to_json()));
358                }
359                Clustered::Cluster(item.position, item.id as u64, properties)
360            })
361            .collect()
362    }
363}
364
365fn abbreviated(count: usize) -> Value {
366    if count >= 10_000 {
367        json!(format!("{}k", (count as f64 / 1000.0).round()))
368    } else if count >= 1000 {
369        json!(format!("{}k", (count as f64 / 100.0).round() / 10.0))
370    } else {
371        json!(count)
372    }
373}

375/// Longitude and latitude in degrees of a world position where the world is `0..1`.
376pub fn unproject(position: [f64; 2]) -> [f64; 2] {
377    let longitude = (position[0] - 0.5) * 360.0;
378    let y = (180.0 - position[1] * 360.0).to_radians();
379    [
380        longitude,
381        360.0 * y.exp().atan() / std::f64::consts::PI - 90.0,
382    ]
383}

395    #[test]
396    fn points_within_the_radius_merge_at_low_zoom_and_part_at_high_zoom() {
397        let properties: [(String, Value); 0] = [];
398        let points = [
399            ([0.5, 0.5], &properties[..]),
400            ([0.5 + 0.0001, 0.5], &properties[..]),
401            ([0.9, 0.9], &properties[..]),
402        ];
403        let clusters = Clusters::new(&points, options());
404        let world = [0.0, 0.0, 1.0, 1.0];
405        let low = clusters.within(0, world);
406        assert_eq!(low.len(), 2, "the two close points merge");
407        assert!(low
408            .iter()
409            .any(|item| matches!(item, Clustered::Cluster(_, _, p)
410            if p.iter().any(|(name, value)| name == "point_count" && *value == json!(2)))));
411        assert_eq!(
412            clusters.within(5, world).len(),
413            3,
414            "past the cluster zoom nothing merges"
415        );
416    }

425    #[test]
426    fn cluster_properties_support_shorthand_and_arbitrary_accumulated_reduce_expressions() {
427        let prop_defs = json!({
428            "sum_short": ["+", ["get", "mag"]],
429            "max_legacy": ["max", ["accumulated"], ["get", "mag"]],
430            "sum_expr": [["+", ["accumulated"], ["get", "sum_expr"]], ["get", "mag"]],
431            "sum_by_field": [["+", ["accumulated"], ["get", "mag"]], ["get", "mag"]],
432            "capped_max": [
433                ["case", [">", ["get", "capped_max"], ["accumulated"]], ["get", "capped_max"], ["accumulated"]],
434                ["min", ["get", "mag"], 8]
435            ]
436        });
437        let opts = ClusterOptions::new(Some(60.0), Some(6), None, Some(&prop_defs), None);
438        let p1 = [("mag".to_owned(), json!(3))];
439        let p2 = [("mag".to_owned(), json!(10))];
440        let p3 = [("mag".to_owned(), json!(5))];
441        // p1 and p2 merge at zoom 4, then merge with p3 at zoom 0 (hierarchical multi-level merge!).
442        let points = [
443            ([0.5, 0.5], &p1[..]),
444            ([0.5001, 0.5], &p2[..]),
445            ([0.51, 0.5], &p3[..]),
446        ];
447        let clusters = Clusters::new(&points, opts);
448        let items = clusters.within(0, [0.0, 0.0, 1.0, 1.0]);
449        assert_eq!(items.len(), 1);
450        let Clustered::Cluster(_, _, cluster_props) = &items[0] else {
451            panic!("expected single merged cluster");
452        };
453        let lookup = |key: &str| {
454            cluster_props
455                .iter()
456                .find(|(k, _)| k == key)
457                .map(|(_, v)| v.clone())
458                .expect("cluster property present")
459        };
460        assert_eq!(lookup("point_count"), json!(3));
461        assert_eq!(lookup("sum_short"), json!(18));
462        assert_eq!(lookup("max_legacy"), json!(10));
463        assert_eq!(lookup("sum_expr"), json!(18));
464        assert_eq!(lookup("sum_by_field"), json!(18));
465        assert_eq!(lookup("capped_max"), json!(8));
466    }
