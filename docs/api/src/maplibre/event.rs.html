31/// Category of a runtime map error delivered via [`MapEvent::Error`].
32#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
33pub enum MapErrorKind {
34    /// Style parsing, validation, or runtime mutation failure.
35    Style,
36    /// Source metadata, tile fetch, or source data update failure.
37    Source,
38    /// Renderer, surface, or GPU pipeline failure.
39    Renderer,
40    /// Schedule or plugin execution failure.
41    Schedule,
42    /// Camera navigation or feature query failure.
43    Operation,
44}
45
46/// Typed lifecycle, data, and error events emitted by a map.
47#[derive(Clone, Debug, PartialEq, Eq)]
48pub enum MapEvent {
49    /// Emitted once when the initial style and visible frame first reach a settled state.
50    MapLoaded,
51    /// Emitted when a frame transitions from busy (pending tiles, animations, or redraws) to
52    /// settled idle.
53    MapIdle,
54    /// Emitted when a style, source, renderer, schedule, or operation error occurs.
55    Error {
56        /// Category of the failure.
57        kind: MapErrorKind,
58        /// Human-readable description of the error.
59        message: String,
60    },
61    /// Emitted when the initial style loads or when a runtime style mutation succeeds.
62    StyleDataLoaded,
63    /// Emitted when metadata or tile data for `source` becomes ready.
64    SourceDataLoaded {
65        /// Identifier of the style source whose data became ready.
66        source: String,
67    },
68    /// Emitted once per missing icon or pattern identifier when a layer requests an image
69    /// absent from the style's sprites, runtime images, and [`ImageProviders`].
70    StyleImageMissing {
71        /// Identifier of the requested missing style image.
72        id: String,
73    },
74}

76/// Opaque handle identifying a callback listener registered with [`MapEventBus::on_event`].
77#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
78pub struct ListenerId(u64);
79
80/// Pull-based async and non-blocking receiver for [`MapEvent`] notifications.
81#[derive(Debug)]
82pub struct MapEventReceiver {
83    receiver: UnboundedReceiver<MapEvent>,
84}
85
86impl MapEventReceiver {
87    /// Attempts to dequeue the next available [`MapEvent`] without blocking.
88    pub fn try_recv(&mut self) -> Option<MapEvent> {
89        self.receiver.try_recv().ok()
90    }
91
92    /// Drains all currently queued [`MapEvent`] items in FIFO order.
93    pub fn drain(&mut self) -> Vec<MapEvent> {
94        let mut out = Vec::new();
95        while let Some(event) = self.try_recv() {
96            out.push(event);
97        }
98        out
99    }
100
101    /// Awaits the next [`MapEvent`], returning `None` only if all [`MapEventBus`] handles drop.
102    pub async fn next(&mut self) -> Option<MapEvent> {
103        StreamExt::next(&mut self.receiver).await
104    }
105}

107impl Stream for MapEventReceiver {
108    type Item = MapEvent;
109
110    fn poll_next(mut self: Pin<&mut Self>, cx: &mut Context<'_>) -> Poll<Option<Self::Item>> {
111        Pin::new(&mut self.receiver).poll_next(cx)
112    }
113}

115type SharedCallback = Rc<RefCell<dyn FnMut(&MapEvent)>>;

117#[derive(Default)]
118struct MapEventBusInner {
119    senders: Vec<UnboundedSender<MapEvent>>,
120    listeners: Vec<(ListenerId, SharedCallback)>,
121    next_listener_id: u64,
122    initial_style_emitted: bool,
123    map_loaded_emitted: bool,
124    was_idle: bool,
125    emitted_missing_images: BTreeSet<String>,
126    buffered_ready_resources: VecDeque<ResourceReady>,
127}
128
129/// Shared event bus stored in `World::resources`, supporting both pull-based channel
130/// subscriptions ([`Self::subscribe`]) and callback listeners ([`Self::on_event`]).
131#[derive(Clone, Default)]
132pub struct MapEventBus {
133    inner: Rc<RefCell<MapEventBusInner>>,
134}
135
136impl std::fmt::Debug for MapEventBus {
137    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
138        let inner = self.inner.borrow();
139        f.debug_struct("MapEventBus")
140            .field("subscribers", &inner.senders.len())
141            .field("listeners", &inner.listeners.len())
142            .field("map_loaded_emitted", &inner.map_loaded_emitted)
143            .field("was_idle", &inner.was_idle)
144            .finish()
145    }
146}
147
148impl MapEventBus {
149    /// Creates a fresh empty event bus.
150    pub fn new() -> Self {
151        Self::default()
152    }
153
154    /// Subscribes a new pull-based [`MapEventReceiver`] that receives all subsequent events.
155    pub fn subscribe(&self) -> MapEventReceiver {
156        let (tx, rx) = unbounded();
157        self.inner.borrow_mut().senders.push(tx);
158        MapEventReceiver { receiver: rx }
159    }
160
161    /// Registers a callback invoked synchronously whenever a [`MapEvent`] is emitted.
162    pub fn on_event(&self, callback: impl FnMut(&MapEvent) + 'static) -> ListenerId {
163        let mut inner = self.inner.borrow_mut();
164        inner.next_listener_id = inner.next_listener_id.wrapping_add(1);
165        let id = ListenerId(inner.next_listener_id);
166        inner.listeners.push((id, Rc::new(RefCell::new(callback))));
167        id
168    }
169
170    /// Removes a previously registered callback listener; returns whether it was present.
171    pub fn remove_listener(&self, id: ListenerId) -> bool {
172        let mut inner = self.inner.borrow_mut();
173        let before = inner.listeners.len();
174        inner.listeners.retain(|(existing, _)| *existing != id);
175        inner.listeners.len() != before
176    }
177
178    /// Emits `event` to all active subscribers and callback listeners.
179    pub fn emit(&self, event: MapEvent) {
180        let callbacks: Vec<SharedCallback> = {
181            let mut inner = self.inner.borrow_mut();
182            inner
183                .senders
184                .retain(|sender| sender.unbounded_send(event.clone()).is_ok());
185            inner.listeners.iter().map(|(_, cb)| cb.clone()).collect()
186        };
187        for cb in callbacks {
188            if let Ok(mut f) = cb.try_borrow_mut() {
189                f(&event);
190            }
191        }
192    }
193
194    /// Emits [`MapEvent::Error`] with `kind` and `message`.
195    pub fn emit_error(&self, kind: MapErrorKind, message: impl Into<String>) {
196        self.emit(MapEvent::Error {
197            kind,
198            message: message.into(),
199        });
200    }
201
202    /// Emits [`MapEvent::StyleImageMissing`] for `id` if it has not yet been reported in the
203    /// current style session.
204    pub fn emit_missing_image(&self, id: impl Into<String>) -> bool {
205        let id = id.into();
206        if id.is_empty() {
207            return false;
208        }
209        let inserted = self
210            .inner
211            .borrow_mut()
212            .emitted_missing_images
213            .insert(id.clone());
214        if inserted {
215            self.emit(MapEvent::StyleImageMissing { id });
216        }
217        inserted
218    }
219
220    /// Clears `id` from the reported missing-image set so a future request emits
221    /// [`MapEvent::StyleImageMissing`] again.
222    pub fn clear_missing_image(&self, id: &str) {
223        self.inner.borrow_mut().emitted_missing_images.remove(id);
224    }
225
226    /// Resets lifecycle tracking when a map or style is reset, allowing missing images and idle
227    /// transitions to be emitted afresh.
228    pub fn reset_style_state(&self) {
229        let mut inner = self.inner.borrow_mut();
230        inner.initial_style_emitted = false;
231        inner.map_loaded_emitted = false;
232        inner.was_idle = false;
233        inner.emitted_missing_images.clear();
234    }
235
236    /// Marks the map as busy so the next settled frame emits [`MapEvent::MapIdle`].
237    pub fn mark_busy(&self) {
238        self.inner.borrow_mut().was_idle = false;
239    }
240
241    pub(crate) fn mark_initial_style_emitted(&self) -> bool {
242        let mut inner = self.inner.borrow_mut();
243        let already = inner.initial_style_emitted;
244        inner.initial_style_emitted = true;
245        already
246    }
247
248    pub(crate) fn observe_frame_settled(&self, settled: bool) {
249        if !settled {
250            self.inner.borrow_mut().was_idle = false;
251            return;
252        }
253        let (emit_loaded, emit_idle) = {
254            let mut inner = self.inner.borrow_mut();
255            let loaded = !inner.map_loaded_emitted;
256            let idle = !inner.was_idle;
257            inner.map_loaded_emitted = true;
258            inner.was_idle = true;
259            (loaded, idle)
260        };
261        if emit_loaded {
262            self.emit(MapEvent::MapLoaded);
263        }
264        if emit_idle {
265            self.emit(MapEvent::MapIdle);
266        }
267    }
268
269    pub(crate) fn push_ready_resources(&self, items: impl IntoIterator<Item = ResourceReady>) {
270        self.inner
271            .borrow_mut()
272            .buffered_ready_resources
273            .extend(items);
274    }
275
276    pub(crate) fn take_buffered_ready_resources(&self) -> Vec<ResourceReady> {
277        self.inner
278            .borrow_mut()
279            .buffered_ready_resources
280            .drain(..)
281            .collect()
282    }
283}

285impl MapContext {
286    /// Returns the shared [`MapEventBus`] stored in this context's world resources.
287    pub fn event_bus(&mut self) -> MapEventBus {
288        self.world
289            .resources
290            .get_or_init_mut::<MapEventBus>()
291            .clone()
292    }
293
294    /// Subscribes a pull-based [`MapEventReceiver`] to this map context's [`MapEventBus`].
295    pub fn subscribe_events(&mut self) -> MapEventReceiver {
296        self.event_bus().subscribe()
297    }
298
299    /// Registers a callback listener on this map context's [`MapEventBus`].
300    pub fn on_event(&mut self, listener: impl FnMut(&MapEvent) + 'static) -> ListenerId {
301        self.event_bus().on_event(listener)
302    }

304    /// Adds or replaces a runtime style image (`id`) and refreshes symbol and pattern layers so
305    /// loaded tiles can pack it into their atlases.
306    pub fn add_style_image(
307        &mut self,
308        id: &str,
309        image: StyleImage,
310    ) -> Result<StyleChange, StyleMutationError> {
311        let bus = self.event_bus();
312        match self.mutate_style(|style| {
313            let mut change = style.add_image(id, image);
314            for layer in &style.layers {
315                if matches!(
316                    layer.type_.as_str(),
317                    "fill" | "fill-extrusion" | "line" | "background"
318                ) && !change.layers.contains(&layer.id)
319                {
320                    change.layers.push(layer.id.clone());
321                    if layer.type_ != "background" {
322                        change.redraw_tiles = true;
323                    }
324                }
325            }
326            Ok(change)
327        }) {
328            Ok(change) => {
329                bus.mark_busy();
330                bus.emit(MapEvent::StyleDataLoaded);
331                Ok(change)
332            }
333            Err(error) => {
334                bus.emit_error(MapErrorKind::Style, error.to_string());
335                Err(error)
336            }
337        }
338    }
339
340    /// Removes a runtime style image (`id`) and clears its missing-image deduplication entry so
341    /// subsequent requests for `id` emit [`MapEvent::StyleImageMissing`] again.
342    pub fn remove_style_image(&mut self, id: &str) -> Result<StyleChange, StyleMutationError> {
343        let bus = self.event_bus();
344        bus.clear_missing_image(id);
345        match self.mutate_style(|style| Ok(style.remove_image(id))) {
346            Ok(change) => {
347                bus.mark_busy();
348                bus.emit(MapEvent::StyleDataLoaded);
349                Ok(change)
350            }
351            Err(error) => {
352                bus.emit_error(MapErrorKind::Style, error.to_string());
353                Err(error)
354            }
355        }
356    }

358    /// Marks [`MapEventBus`] busy if this context currently requires a redraw or `extra_busy` holds.
359    pub(crate) fn mark_busy_if_needed(&mut self, extra_busy: bool) {
360        if self.needs_redraw() || extra_busy {
361            self.event_bus().mark_busy();
362        }
363    }

365    /// Synchronizes frame-end lifecycle, ready-source, and missing-image events into [`MapEventBus`].
366    pub(crate) fn sync_frame_events(
367        &mut self,
368        image_providers: Option<&ImageProviders>,
369        extra_busy: bool,
370    ) {
371        let bus = self.event_bus();
372        if !bus.mark_initial_style_emitted() {
373            bus.emit(MapEvent::StyleDataLoaded);
374        }
375
376        let ready = self.take_ready_resources();
377        if !ready.is_empty() {
378            let mut loaded_sources = BTreeSet::new();
379            for item in &ready {
380                if let ResourceReady::Tile {
381                    kind, loaded: true, ..
382                } = item
383                {
384                    for (name, source) in &self.style.sources {
385                        let matches_kind = match kind {
386                            RequestKind::Vector => {
387                                matches!(source, Source::Vector(_) | Source::GeoJson(_))
388                            }
389                            RequestKind::Raster => {
390                                matches!(source, Source::Raster(_) | Source::Image(_))
391                            }
392                            RequestKind::Dem => matches!(source, Source::RasterDem(_)),
393                        };
394                        if matches_kind {
395                            loaded_sources.insert(name.clone());
396                        }
397                    }
398                }
399            }
400            bus.push_ready_resources(ready);
401            for source in loaded_sources {
402                bus.emit(MapEvent::SourceDataLoaded { source });
403            }
404        }
405
406        let mut missing_ids = BTreeSet::new();
407        if let Some(providers) = image_providers {
408            for id in providers.take_missing() {
409                missing_ids.insert(id);
410            }
411        }
412        collect_visible_missing_style_images(
413            &self.style,
414            self.view_state.zoom().value(),
415            image_providers,
416            &mut missing_ids,
417        );
418        for id in missing_ids {
419            if !self.style.images.contains_key(&id) {
420                bus.emit_missing_image(id);
421            }
422        }
423
424        let settled = !self.needs_redraw() && !extra_busy;
425        bus.observe_frame_settled(settled);
426    }
427}

429fn collect_visible_missing_style_images(
430    style: &Style,
431    zoom: f64,
432    providers: Option<&ImageProviders>,
433    out: &mut BTreeSet<String>,
434) {
435    let is_supplied = |name: &str| -> bool {
436        style.images.contains_key(name) || providers.is_some_and(|p| p.provides(name))
437    };
438    for layer in &style.layers {
439        if !layer.is_visible_at(zoom) {
440            continue;
441        }
442        let Some(paint) = &layer.paint else {
443            continue;
444        };
445        let pattern_val = match paint {
446            LayerPaint::Fill(p) => p.fill_pattern.as_ref(),
447            LayerPaint::FillExtrusion(p) => p.fill_extrusion_pattern.as_ref(),
448            LayerPaint::Background(p) => p.background_pattern.as_ref(),
449            LayerPaint::Line(p) => p.line_pattern.as_ref(),
450            _ => None,
451        };
452        if let Some(val) = pattern_val {
453            let prop = StyleProperty::<PatternName>::parse(val);
454            if prop.is_feature_constant()
455                && let Some(PatternName(name)) = prop.evaluate_at_zoom(zoom)
456                && !name.is_empty()
457                && style.sprite.is_none()
458                && !is_supplied(&name)
459            {
460                out.insert(name);
461            }
462        }
463        if style.sprite.is_none()
464            && let LayerPaint::Symbol(sym) = paint
465            && let Some(val) = sym.properties.get("icon-image")
466        {
467            let prop = StyleProperty::<ImageName>::parse(val);
468            if prop.is_feature_constant()
469                && let Some(ImageName(name)) = prop.evaluate_at_zoom(zoom)
470                && !name.is_empty()
471                && !is_supplied(&name)
472            {
473                out.insert(name);
474            }
475        }
476    }
