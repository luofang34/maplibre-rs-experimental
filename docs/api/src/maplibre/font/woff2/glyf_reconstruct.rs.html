13/// Decodes one WOFF2 variable-length `(dx, dy, on_curve)` point triplet (W3C WOFF 2.0 Section 5.1).
14fn decode_triplet(flag: u8, glyph_stream: &mut Cursor<'_>) -> Result<(i32, i32, bool), FontError> {
15    let on_curve = (flag & 0x80) == 0;
16    let f = flag & 0x7F;
17    let sign = |bit: u8| if bit != 0 { 1_i32 } else { -1_i32 };
18    let (dx, dy) = if f < 10 {
19        let b0 = i32::from(glyph_stream.read_u8()?);
20        let mag = ((i32::from(f & 0x0E)) << 7) + b0;
21        (0, mag * sign(f & 1))
22    } else if f < 20 {
23        let k = f - 10;
24        let b0 = i32::from(glyph_stream.read_u8()?);
25        let mag = ((i32::from(k & 0x0E)) << 7) + b0;
26        (mag * sign(k & 1), 0)
27    } else if f < 84 {
28        let k = f - 20;
29        let b0 = glyph_stream.read_u8()?;
30        let mag_x = 1 + i32::from(k & 0x30) + i32::from(b0 >> 4);
31        let mag_y = 1 + (i32::from(k & 0x0C) << 2) + i32::from(b0 & 0x0F);
32        (mag_x * sign(k & 1), mag_y * sign(k & 2))
33    } else if f < 120 {
34        let k = f - 84;
35        let group = i32::from(k / 4);
36        let b0 = i32::from(glyph_stream.read_u8()?);
37        let b1 = i32::from(glyph_stream.read_u8()?);
38        let mag_x = 1 + 256 * (group / 3) + b0;
39        let mag_y = 1 + 256 * (group % 3) + b1;
40        (mag_x * sign(k & 1), mag_y * sign(k & 2))
41    } else if f < 124 {
42        let k = f - 120;
43        let b0 = u16::from(glyph_stream.read_u8()?);
44        let b1 = u16::from(glyph_stream.read_u8()?);
45        let b2 = u16::from(glyph_stream.read_u8()?);
46        let mag_x = i32::from((b0 << 4) | (b1 >> 4));
47        let mag_y = i32::from(((b1 & 0x0F) << 8) | b2);
48        (mag_x * sign(k & 1), mag_y * sign(k & 2))
49    } else {
50        let k = f - 124;
51        let b0 = u16::from(glyph_stream.read_u8()?);
52        let b1 = u16::from(glyph_stream.read_u8()?);
53        let b2 = u16::from(glyph_stream.read_u8()?);
54        let b3 = u16::from(glyph_stream.read_u8()?);
55        let mag_x = i32::from((b0 << 8) | b1);
56        let mag_y = i32::from((b2 << 8) | b3);
57        (mag_x * sign(k & 1), mag_y * sign(k & 2))
58    };
59    Ok((dx, dy, on_curve))
60}

62/// Encodes decoded `(dx, dy, on_curve)` deltas into standard TrueType simple-glyph flag, X, and Y arrays.
63fn encode_truetype_points(
64    points: &[(i32, i32, bool)],
65    has_overlap: bool,
66    out: &mut Vec<u8>,
67) -> Result<(), FontError> {
68    let mut flags = Vec::with_capacity(points.len());
69    let mut x_bytes = Vec::with_capacity(points.len() * 2);
70    let mut y_bytes = Vec::with_capacity(points.len() * 2);
71
72    for (idx, &(dx, dy, on_curve)) in points.iter().enumerate() {
73        let dx_i16 = i16::try_from(dx)
74            .map_err(|_| FontError::InvalidContainer("WOFF2 glyph x delta overflow"))?;
75        let dy_i16 = i16::try_from(dy)
76            .map_err(|_| FontError::InvalidContainer("WOFF2 glyph y delta overflow"))?;
77        let mut f = if on_curve { 0x01_u8 } else { 0x00_u8 };
78        if idx == 0 && has_overlap {
79            f |= 0x40;
80        }
81        if dx_i16 == 0 {
82            f |= 0x10;
83        } else if (-255..=255).contains(&dx_i16) {
84            f |= 0x02;
85            if dx_i16 > 0 {
86                f |= 0x10;
87            }
88            x_bytes.push(dx_i16.unsigned_abs() as u8);
89        } else {
90            x_bytes.extend_from_slice(&dx_i16.to_be_bytes());
91        }
92
93        if dy_i16 == 0 {
94            f |= 0x20;
95        } else if (-255..=255).contains(&dy_i16) {
96            f |= 0x04;
97            if dy_i16 > 0 {
98                f |= 0x20;
99            }
100            y_bytes.push(dy_i16.unsigned_abs() as u8);
101        } else {
102            y_bytes.extend_from_slice(&dy_i16.to_be_bytes());
103        }
104        flags.push(f);
105    }
106
107    out.extend_from_slice(&flags);
108    out.extend_from_slice(&x_bytes);
109    out.extend_from_slice(&y_bytes);
110    Ok(())
111}

113/// Scans a WOFF2 composite glyph component stream to find its exact byte length and whether
114/// TrueType instructions follow (`WE_HAVE_INSTRUCTIONS = 0x0100`).
115fn scan_composite_components<'a>(
116    composite: &mut Cursor<'a>,
117) -> Result<(&'a [u8], bool), FontError> {
118    let start = composite.position();
119    let mut have_instructions = false;
120    loop {
121        let flags = composite.read_u16_be()?;
122        let _glyph_index = composite.read_u16_be()?;
123        // ARG_1_AND_2_ARE_WORDS (0x0001): 4 bytes vs 2 bytes.
124        let arg_bytes = if (flags & 0x0001) != 0 { 4 } else { 2 };
125        let _ = composite.read_slice(arg_bytes)?;
126        // WE_HAVE_A_SCALE (0x0008), WE_HAVE_AN_X_AND_Y_SCALE (0x0040), WE_HAVE_A_TWO_BY_TWO (0x0080).
127        let xform_bytes = if (flags & 0x0008) != 0 {
128            2
129        } else if (flags & 0x0040) != 0 {
130            4
131        } else if (flags & 0x0080) != 0 {
132            8
133        } else {
134            0
135        };
136        let _ = composite.read_slice(xform_bytes)?;
137        if (flags & 0x0100) != 0 {
138            have_instructions = true;
139        }
140        if (flags & 0x0020) == 0 {
141            break;
142        }
143    }
144    let end = composite.position();
145    Ok((&composite.bytes[start..end], have_instructions))
146}

148/// Reconstructs TrueType `glyf` and `loca` tables from a WOFF2 transformed `glyf` stream.
149pub(super) fn reconstruct_glyf_loca(stream: &[u8]) -> Result<ReconstructedGlyf, FontError> {
150    let mut header = Cursor::new(stream);
151    let _reserved = header.read_u16_be()?;
152    let option_flags = header.read_u16_be()?;
153    let num_glyphs = header.read_u16_be()? as usize;
154    let mut index_format = header.read_u16_be()?;
155    let n_contour_size = header.read_u32_be()? as usize;
156    let n_points_size = header.read_u32_be()? as usize;
157    let flag_size = header.read_u32_be()? as usize;
158    let glyph_size = header.read_u32_be()? as usize;
159    let composite_size = header.read_u32_be()? as usize;
160    let bbox_size = header.read_u32_be()? as usize;
161    let instruction_size = header.read_u32_be()? as usize;
162
163    let mut n_contours = Cursor::new(header.read_slice(n_contour_size)?);
164    let mut n_points = Cursor::new(header.read_slice(n_points_size)?);
165    let mut flags = Cursor::new(header.read_slice(flag_size)?);
166    let mut glyph_data = Cursor::new(header.read_slice(glyph_size)?);
167    let mut composites = Cursor::new(header.read_slice(composite_size)?);
168    let mut bboxes = Cursor::new(header.read_slice(bbox_size)?);
169    let mut instructions = Cursor::new(header.read_slice(instruction_size)?);
170
171    let bitmap_bytes = num_glyphs.div_ceil(32) * 4;
172    let bbox_bitmap = bboxes.read_slice(bitmap_bytes)?;
173    let overlap_bitmap = if (option_flags & 0x0001) != 0 {
174        Some(header.read_slice(num_glyphs.div_ceil(8))?)
175    } else {
176        None
177    };
178
179    let has_bit = |bitmap: &[u8], glyph_idx: usize| -> bool {
180        bitmap
181            .get(glyph_idx >> 3)
182            .is_some_and(|byte| (byte & (0x80 >> (glyph_idx & 7))) != 0)
183    };
184
185    let mut glyf = Vec::new();
186    let mut offsets = Vec::with_capacity(num_glyphs + 1);
187    let mut x_mins = vec![0_i16; num_glyphs];
188
189    for (glyph_idx, x_min_slot) in x_mins.iter_mut().enumerate() {
190        offsets.push(glyf.len());
191        let contour_count = n_contours.read_i16_be()?;
192        let explicit_bbox = has_bit(bbox_bitmap, glyph_idx);
193
194        if contour_count == 0 {
195            if explicit_bbox {
196                return Err(FontError::InvalidContainer(
197                    "WOFF2 empty glyph cannot carry an explicit bbox",
198                ));
199            }
200            continue;
201        }
202
203        if contour_count > 0 {
204            let contours = contour_count as usize;
205            let mut end_pts = Vec::with_capacity(contours);
206            let mut total_points = 0usize;
207            for _ in 0..contours {
208                let pts = usize::from(n_points.read_255_ushort()?);
209                total_points = total_points
210                    .checked_add(pts)
211                    .filter(|&sum| sum > 0 && sum <= u16::MAX as usize + 1)
212                    .ok_or(FontError::InvalidContainer(
213                        "WOFF2 contour point count invalid",
214                    ))?;
215                end_pts.push((total_points - 1) as u16);
216            }
217
218            let mut decoded_pts = Vec::with_capacity(total_points);
219            let (mut x, mut y) = (0_i32, 0_i32);
220            let (mut min_x, mut min_y) = (i32::MAX, i32::MAX);
221            let (mut max_x, mut max_y) = (i32::MIN, i32::MIN);
222            for _ in 0..total_points {
223                let raw_flag = flags.read_u8()?;
224                let (dx, dy, on_curve) = decode_triplet(raw_flag, &mut glyph_data)?;
225                x = x.saturating_add(dx);
226                y = y.saturating_add(dy);
227                min_x = min_x.min(x);
228                min_y = min_y.min(y);
229                max_x = max_x.max(x);
230                max_y = max_y.max(y);
231                decoded_pts.push((dx, dy, on_curve));
232            }
233
234            let instr_len = glyph_data.read_255_ushort()?;
235            let instr_bytes = instructions.read_slice(usize::from(instr_len))?;
236
237            let (bx_min, by_min, bx_max, by_max) = if explicit_bbox {
238                (
239                    bboxes.read_i16_be()?,
240                    bboxes.read_i16_be()?,
241                    bboxes.read_i16_be()?,
242                    bboxes.read_i16_be()?,
243                )
244            } else {
245                (
246                    i16::try_from(min_x).unwrap_or(i16::MIN),
247                    i16::try_from(min_y).unwrap_or(i16::MIN),
248                    i16::try_from(max_x).unwrap_or(i16::MAX),
249                    i16::try_from(max_y).unwrap_or(i16::MAX),
250                )
251            };
252            *x_min_slot = bx_min;
253
254            glyf.extend_from_slice(&contour_count.to_be_bytes());
255            glyf.extend_from_slice(&bx_min.to_be_bytes());
256            glyf.extend_from_slice(&by_min.to_be_bytes());
257            glyf.extend_from_slice(&bx_max.to_be_bytes());
258            glyf.extend_from_slice(&by_max.to_be_bytes());
259            for end_pt in end_pts {
260                glyf.extend_from_slice(&end_pt.to_be_bytes());
261            }
262            glyf.extend_from_slice(&instr_len.to_be_bytes());
263            glyf.extend_from_slice(instr_bytes);
264
265            let has_overlap = overlap_bitmap.is_some_and(|bm| has_bit(bm, glyph_idx));
266            encode_truetype_points(&decoded_pts, has_overlap, &mut glyf)?;
267        } else {
268            if !explicit_bbox {
269                return Err(FontError::InvalidContainer(
270                    "WOFF2 composite glyph requires an explicit bbox",
271                ));
272            }
273            let bx_min = bboxes.read_i16_be()?;
274            let by_min = bboxes.read_i16_be()?;
275            let bx_max = bboxes.read_i16_be()?;
276            let by_max = bboxes.read_i16_be()?;
277            *x_min_slot = bx_min;
278
279            let (comp_bytes, have_instr) = scan_composite_components(&mut composites)?;
280            let instr_bytes = if have_instr {
281                let len = glyph_data.read_255_ushort()?;
282                Some((len, instructions.read_slice(usize::from(len))?))
283            } else {
284                None
285            };
286
287            glyf.extend_from_slice(&(-1_i16).to_be_bytes());
288            glyf.extend_from_slice(&bx_min.to_be_bytes());
289            glyf.extend_from_slice(&by_min.to_be_bytes());
290            glyf.extend_from_slice(&bx_max.to_be_bytes());
291            glyf.extend_from_slice(&by_max.to_be_bytes());
292            glyf.extend_from_slice(comp_bytes);
293            if let Some((len, bytes)) = instr_bytes {
294                glyf.extend_from_slice(&len.to_be_bytes());
295                glyf.extend_from_slice(bytes);
296            }
297        }
298
299        // Pad each glyph to a 4-byte boundary so short `loca` (`offset / 2`) stays exact.
300        let padded = (glyf.len() + 3) & !3;
301        glyf.resize(padded, 0);
302    }
303    offsets.push(glyf.len());
304
305    if index_format == 0 && glyf.len() / 2 > u16::MAX as usize {
306        index_format = 1;
307    }
308
309    let mut loca = Vec::with_capacity(offsets.len() * if index_format == 0 { 2 } else { 4 });
310    if index_format == 0 {
311        for off in offsets {
312            let word = u16::try_from(off / 2)
313                .map_err(|_| FontError::InvalidContainer("WOFF2 short loca overflow"))?;
314            loca.extend_from_slice(&word.to_be_bytes());
315        }
316    } else {
317        for off in offsets {
318            let dword = u32::try_from(off)
319                .map_err(|_| FontError::InvalidContainer("WOFF2 long loca overflow"))?;
320            loca.extend_from_slice(&dword.to_be_bytes());
321        }
322    }
323
324    Ok(ReconstructedGlyf {
325        glyf,
326        loca,
327        index_format,
328        x_mins,
329    })
330}

332/// Reconstructs a WOFF2 transformed `hmtx` table (transform version 1, W3C WOFF 2.0 Section 5.4).
333pub(super) fn reconstruct_hmtx(
334    stream: &[u8],
335    hhea: Option<&[u8]>,
336    maxp: Option<&[u8]>,
337    x_mins: &[i16],
338) -> Result<Vec<u8>, FontError> {
339    let hhea = hhea
340        .filter(|b| b.len() >= 36)
341        .ok_or(FontError::InvalidContainer(
342            "WOFF2 transformed hmtx requires hhea table",
343        ))?;
344    let maxp = maxp
345        .filter(|b| b.len() >= 6)
346        .ok_or(FontError::InvalidContainer(
347            "WOFF2 transformed hmtx requires maxp table",
348        ))?;
349    let num_h_metrics = usize::from(u16::from_be_bytes([hhea[34], hhea[35]]));
350    let num_glyphs = usize::from(u16::from_be_bytes([maxp[4], maxp[5]]));
351    if num_h_metrics > num_glyphs {
352        return Err(FontError::InvalidContainer(
353            "WOFF2 hmtx numOfHMetrics > numGlyphs",
354        ));
355    }
356
357    let mut cur = Cursor::new(stream);
358    let flags = cur.read_u8()?;
359    if (flags & 0xFC) != 0 {
360        return Err(FontError::InvalidContainer("WOFF2 hmtx reserved flags set"));
361    }
362    let omit_proportional_lsb = (flags & 0x01) != 0;
363    let omit_monospace_lsb = (flags & 0x02) != 0;
364
365    let mut advances = Vec::with_capacity(num_h_metrics);
366    for _ in 0..num_h_metrics {
367        advances.push(cur.read_u16_be()?);
368    }
369
370    let mut out = Vec::with_capacity(num_h_metrics * 4 + (num_glyphs - num_h_metrics) * 2);
371    for (idx, adv) in advances.into_iter().enumerate() {
372        let lsb = if omit_proportional_lsb {
373            x_mins.get(idx).copied().unwrap_or(0)
374        } else {
375            cur.read_i16_be()?
376        };
377        out.extend_from_slice(&adv.to_be_bytes());
378        out.extend_from_slice(&lsb.to_be_bytes());
379    }
380    for idx in num_h_metrics..num_glyphs {
381        let lsb = if omit_monospace_lsb {
382            x_mins.get(idx).copied().unwrap_or(0)
383        } else {
384            cur.read_i16_be()?
385        };
386        out.extend_from_slice(&lsb.to_be_bytes());
387    }
388    Ok(out)
