16/// Standard 63 known table tags defined in W3C WOFF File Format 2.0, Section 3.2 (Table 1).
17const KNOWN_TAGS: [[u8; 4]; 63] = [
18    *b"cmap", *b"head", *b"hhea", *b"hmtx", *b"maxp", *b"name", *b"OS/2", *b"post", *b"cvt ",

27/// Bounds-checked big-endian byte cursor used across WOFF2 header and substream decoders.
28pub(super) struct Cursor<'a> {
29    bytes: &'a [u8],
30    pos: usize,
31}
32
33impl<'a> Cursor<'a> {
34    pub(super) fn new(bytes: &'a [u8]) -> Self {
35        Self { bytes, pos: 0 }
36    }
37
38    pub(super) fn position(&self) -> usize {
39        self.pos
40    }
41
42    pub(super) fn read_u8(&mut self) -> Result<u8, FontError> {
43        let &byte = self
44            .bytes
45            .get(self.pos)
46            .ok_or(FontError::InvalidContainer("WOFF2 unexpected EOF"))?;
47        self.pos += 1;
48        Ok(byte)
49    }
50
51    pub(super) fn read_slice(&mut self, len: usize) -> Result<&'a [u8], FontError> {
52        let end = self
53            .pos
54            .checked_add(len)
55            .filter(|&end| end <= self.bytes.len())
56            .ok_or(FontError::InvalidContainer("WOFF2 slice out of bounds"))?;
57        let slice = &self.bytes[self.pos..end];
58        self.pos = end;
59        Ok(slice)
60    }
61
62    pub(super) fn read_u16_be(&mut self) -> Result<u16, FontError> {
63        let s = self.read_slice(2)?;
64        Ok(u16::from_be_bytes([s[0], s[1]]))
65    }
66
67    pub(super) fn read_i16_be(&mut self) -> Result<i16, FontError> {
68        let s = self.read_slice(2)?;
69        Ok(i16::from_be_bytes([s[0], s[1]]))
70    }
71
72    pub(super) fn read_u32_be(&mut self) -> Result<u32, FontError> {
73        let s = self.read_slice(4)?;
74        Ok(u32::from_be_bytes([s[0], s[1], s[2], s[3]]))
75    }
76
77    /// Reads a WOFF2 `UIntBase128` variable-length integer (W3C WOFF 2.0 Section 2).
78    pub(super) fn read_uint_base128(&mut self) -> Result<u32, FontError> {
79        let mut accum = 0_u32;
80        for idx in 0..5 {
81            let byte = self.read_u8()?;
82            if idx == 0 && byte == 0x80 {
83                return Err(FontError::InvalidContainer(
84                    "WOFF2 UIntBase128 leading zero",
85                ));
86            }
87            if (accum & 0xFE00_0000) != 0 {
88                return Err(FontError::InvalidContainer("WOFF2 UIntBase128 overflow"));
89            }
90            accum = (accum << 7) | u32::from(byte & 0x7F);
91            if (byte & 0x80) == 0 {
92                return Ok(accum);
93            }
94        }
95        Err(FontError::InvalidContainer(
96            "WOFF2 UIntBase128 exceeds 5 bytes",
97        ))
98    }
99
100    /// Reads a WOFF2 `255UShort` variable-length integer (W3C WOFF 2.0 Section 2).
101    pub(super) fn read_255_ushort(&mut self) -> Result<u16, FontError> {
102        let code = self.read_u8()?;
103        match code {
104            253 => self.read_u16_be(),
105            255 => Ok(253_u16.saturating_add(u16::from(self.read_u8()?))),
106            254 => Ok(506_u16.saturating_add(u16::from(self.read_u8()?))),
107            value => Ok(u16::from(value)),
108        }
109    }
110}

112struct Woff2DirEntry {
113    tag: [u8; 4],
114    transformed: bool,
115    stream_offset: usize,
116    stream_len: usize,
117}
118
119/// Unpacks a WOFF 2.0 (`wOF2`) buffer into a standard OpenType/TrueType `sfnt` byte buffer.
120pub(super) fn unpack_woff2(bytes: &[u8]) -> Result<Vec<u8>, FontError> {
121    if bytes.len() < WOFF2_HEADER_SIZE || bytes.get(..4) != Some(b"wOF2") {
122        return Err(FontError::InvalidContainer(
123            "truncated or invalid WOFF2 header",
124        ));
125    }
126    let mut cursor = Cursor::new(bytes);
127    let _sig = cursor.read_u32_be()?;
128    let mut flavor = cursor.read_u32_be()?;
129    let declared_len = cursor.read_u32_be()? as usize;
130    if declared_len > bytes.len() {
131        return Err(FontError::InvalidContainer("WOFF2 length exceeds buffer"));
132    }
133    let num_tables = cursor.read_u16_be()? as usize;
134    let reserved = cursor.read_u16_be()?;
135    if num_tables == 0 || reserved != 0 {
136        return Err(FontError::InvalidContainer(
137            "WOFF2 header has zero tables or non-zero reserved field",
138        ));
139    }
140    let _total_sfnt_size = cursor.read_u32_be()?;
141    let total_compressed_size = cursor.read_u32_be()? as usize;
142    // Skip majorVersion, minorVersion, metaOffset/Length/OrigLength, privOffset/Length (24 bytes).
143    let _ = cursor.read_slice(24)?;
144
145    let mut entries = Vec::with_capacity(num_tables);
146    let mut stream_total = 0usize;
147    for _ in 0..num_tables {
148        let flags = cursor.read_u8()?;
149        let tag_index = usize::from(flags & 0x3F);
150        let tag = if tag_index == 63 {
151            let s = cursor.read_slice(4)?;
152            [s[0], s[1], s[2], s[3]]
153        } else {
154            KNOWN_TAGS[tag_index]
155        };
156        let xform_version = (flags >> 6) & 0x03;
157        let orig_length = cursor.read_uint_base128()? as usize;
158        let is_glyf_or_loca = &tag == b"glyf" || &tag == b"loca";
159        let transformed = if is_glyf_or_loca {
160            xform_version == 0
161        } else {
162            xform_version != 0
163        };
164        let stream_len = if transformed {
165            cursor.read_uint_base128()? as usize
166        } else {
167            orig_length
168        };
169        if &tag == b"loca" && transformed && stream_len != 0 {
170            return Err(FontError::InvalidContainer(
171                "WOFF2 transformed loca must have zero stream length",
172            ));
173        }
174        let stream_offset = stream_total;
175        stream_total = stream_total
176            .checked_add(stream_len)
177            .filter(|&sum| sum <= MAX_DECOMPRESSED_FONT_BYTES)
178            .ok_or(FontError::InvalidContainer(
179                "WOFF2 uncompressed stream too large",
180            ))?;
181        entries.push(Woff2DirEntry {
182            tag,
183            transformed,
184            stream_offset,
185            stream_len,
186        });
187    }
188
189    // Optional TTC collection directory (`ttcf` flavor): select the first face's flavor & tables.
190    let mut selected_indices: Option<Vec<usize>> = None;
191    if &flavor.to_be_bytes() == b"ttcf" {
192        let _ttc_version = cursor.read_u32_be()?;
193        let num_fonts = cursor.read_255_ushort()? as usize;
194        if num_fonts == 0 {
195            return Err(FontError::InvalidContainer(
196                "WOFF2 collection has zero fonts",
197            ));
198        }
199        for font_idx in 0..num_fonts {
200            let count = cursor.read_255_ushort()? as usize;
201            let face_flavor = cursor.read_u32_be()?;
202            let mut indices = Vec::with_capacity(count);
203            for _ in 0..count {
204                let idx = cursor.read_255_ushort()? as usize;
205                if idx >= entries.len() {
206                    return Err(FontError::InvalidContainer(
207                        "WOFF2 collection table index out of bounds",
208                    ));
209                }
210                indices.push(idx);
211            }
212            if font_idx == 0 {
213                flavor = face_flavor;
214                selected_indices = Some(indices);
215            }
216        }
217    }
218
219    let compressed = cursor.read_slice(total_compressed_size)?;
220    let mut stream = Vec::with_capacity(stream_total.min(131_072));
221    let limit = u64::try_from(stream_total)
222        .map_err(|_| FontError::InvalidContainer("WOFF2 stream too large"))?
223        .saturating_add(1);
224    Decompressor::new(compressed, 4096)
225        .take(limit)
226        .read_to_end(&mut stream)
227        .map_err(|_| FontError::InvalidContainer("WOFF2 Brotli decompression failed"))?;
228    if stream.len() < stream_total {
229        return Err(FontError::InvalidContainer(
230            "WOFF2 decompressed stream shorter than declared tables",
231        ));
232    }
233
234    let active_entries: Vec<&Woff2DirEntry> = match &selected_indices {
235        Some(indices) => indices.iter().map(|&idx| &entries[idx]).collect(),
236        None => entries.iter().collect(),
237    };
238
239    let mut raw_tables = Vec::with_capacity(active_entries.len());
240    let mut reconstructed_glyf_loca: Option<glyf_reconstruct::ReconstructedGlyf> = None;
241
242    for entry in &active_entries {
243        if &entry.tag == b"glyf" && entry.transformed {
244            let slice = &stream[entry.stream_offset..entry.stream_offset + entry.stream_len];
245            reconstructed_glyf_loca = Some(glyf_reconstruct::reconstruct_glyf_loca(slice)?);
246        }
247    }
248
249    // Locate `hhea` and `maxp` slices in case `hmtx` uses transform version 1.
250    let find_raw = |target: &[u8; 4]| -> Option<&[u8]> {
251        active_entries
252            .iter()
253            .find(|e| &e.tag == target && !e.transformed)
254            .map(|e| &stream[e.stream_offset..e.stream_offset + e.stream_len])
255    };
256    let hhea_slice = find_raw(b"hhea");
257    let maxp_slice = find_raw(b"maxp");
258
259    for entry in &active_entries {
260        let data = match (&entry.tag, entry.transformed) {
261            (b"glyf", true) => reconstructed_glyf_loca
262                .as_ref()
263                .map(|r| r.glyf.clone())
264                .ok_or(FontError::InvalidContainer("missing reconstructed glyf"))?,
265            (b"loca", true) => reconstructed_glyf_loca
266                .as_ref()
267                .map(|r| r.loca.clone())
268                .ok_or(FontError::InvalidContainer("missing reconstructed loca"))?,
269            (b"hmtx", true) => {
270                let slice = &stream[entry.stream_offset..entry.stream_offset + entry.stream_len];
271                let x_mins = reconstructed_glyf_loca
272                    .as_ref()
273                    .map(|r| r.x_mins.as_slice())
274                    .unwrap_or(&[]);
275                glyf_reconstruct::reconstruct_hmtx(slice, hhea_slice, maxp_slice, x_mins)?
276            }
277            (_, true) => {
278                return Err(FontError::InvalidContainer(
279                    "unsupported WOFF2 table transform",
280                ))
281            }
282            (b"head", false) => {
283                let mut head =
284                    stream[entry.stream_offset..entry.stream_offset + entry.stream_len].to_vec();
285                if let Some(reconstructed) = &reconstructed_glyf_loca {
286                    if head.len() >= 52 {
287                        head[50..52].copy_from_slice(&reconstructed.index_format.to_be_bytes());
288                    }
289                }
290                head
291            }
292            (_, false) => {
293                stream[entry.stream_offset..entry.stream_offset + entry.stream_len].to_vec()
294            }
295        };
296        raw_tables.push(SfntTable {
297            tag: entry.tag,
298            data,
299        });
300    }
301
302    assemble_sfnt(flavor, raw_tables)
303}
