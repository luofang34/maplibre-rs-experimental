1//! Unicode block classification for local CJK ideograph generation and complex script shaping.
2
3/// Returns `true` when `ch` belongs to a CJK ideograph, Kana, Hangul, Bopomofo, Yi, or CJK
4/// punctuation/symbol block that MapLibre rasterises locally when `localIdeographFontFamily`
5/// is configured.
6///
7/// Covers Basic Multilingual Plane blocks as well as Supplementary Ideographic Planes (SIP/TIP,
8/// `U+20000..=U+323AF`, including Extension B–I characters such as `𪚥` and `𦨭`).
9pub fn is_local_cjk_codepoint(ch: char) -> bool {
10    matches!(
11        u32::from(ch),
12        // Hangul Jamo
13        0x1100..=0x11FF
14            // CJK Radicals Supplement, Kangxi Radicals, Ideographic Description, CJK Symbols & Punctuation
15            | 0x2E80..=0x303F
16            // Hiragana & Katakana
17            | 0x3040..=0x30FF
18            // Bopomofo, Hangul Compatibility Jamo, Kanbun, Bopomofo Extended, CJK Strokes, Katakana Phonetic Extensions
19            | 0x3100..=0x31FF
20            // Enclosed CJK Letters and Months & CJK Compatibility
21            | 0x3200..=0x33FF
22            // CJK Unified Ideographs Extension A
23            | 0x3400..=0x4DBF
24            // CJK Unified Ideographs
25            | 0x4E00..=0x9FFF
26            // Yi Syllables & Yi Radicals
27            | 0xA000..=0xA4CF
28            // Hangul Jamo Extended-A
29            | 0xA960..=0xA97F
30            // Hangul Syllables & Hangul Jamo Extended-B
31            | 0xAC00..=0xD7FF
32            // CJK Compatibility Ideographs
33            | 0xF900..=0xFAFF
34            // Vertical Forms & CJK Compatibility Forms & Small Form Variants
35            | 0xFE10..=0xFE1F
36            | 0xFE30..=0xFE6F
37            // Halfwidth and Fullwidth Forms
38            | 0xFF00..=0xFFEF
39            // Kana Supplement, Kana Extended-A, Small Kana Extension
40            | 0x1B000..=0x1B16F
41            // Enclosed Ideographic Supplement
42            | 0x1F200..=0x1F2FF
43            // CJK Unified Ideographs Extensions B, C, D, E, F & CJK Compatibility Ideographs Supplement
44            | 0x20000..=0x2FA1F
45            // CJK Unified Ideographs Extensions G, H, I
46            | 0x30000..=0x323AF
47    )
48}

50/// Returns `true` when `ch` belongs to a script or combining-mark range whose typography
51/// benefits from OpenType `GSUB`/`GPOS` shaping (Arabic, Indic, Southeast Asian, Tibetan,
52/// or combining marks).
53pub fn is_complex_shaping_codepoint(ch: char) -> bool {
54    matches!(
55        u32::from(ch),
56        // Combining Diacritical Marks
57        0x0300..=0x036F
58            // Hebrew, Arabic, Syriac, Arabic Supplement, Thaana, NKo, Samaritan, Mandaic, Arabic Extended
59            | 0x0590..=0x08FF
60            // Devanagari, Bengali, Gurmukhi, Gujarati, Oriya, Tamil, Telugu, Kannada, Malayalam, Sinhala
61            | 0x0900..=0x0DFF
62            // Thai & Lao
63            | 0x0E00..=0x0EFF
64            // Tibetan, Myanmar
65            | 0x0F00..=0x109F
66            // Tagalog, Hanunoo, Buhid, Tagbanwa, Khmer, Mongolian
67            | 0x1700..=0x18AF
68            // Limbu, Tai Le, New Tai Lue, Khmer Symbols, Buginese, Tai Tham
69            | 0x1900..=0x1AAF
70            // Balinese, Sundanese, Batak, Lepcha
71            | 0x1B00..=0x1C4F
72            // Vedic Extensions, Combining Diacritical Marks Extended/Supplement
73            | 0x1CD0..=0x1DFF
74            // Zero-width non-joiner / joiner
75            | 0x200C..=0x200D
76            // Syloti Nagri, Devanagari Extended, Kayah Li, Rejang, Javanese, Myanmar Extended-B, Cham, Myanmar Extended-A, Tai Viet
77            | 0xA800..=0xAADF
78    )
79}
