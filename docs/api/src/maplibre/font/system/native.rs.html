15pub(in crate::font) struct NativeFontDb {
16    db: fontdb::Database,
17    tier: u8,
18    loaded_by_db_id: HashMap<fontdb::ID, u64>,
19    shared_files: HashMap<PathBuf, Arc<Vec<u8>>>,
20    codepoint_cache: HashMap<(String, u32), Option<u64>>,
21}

23impl FontSystem {
24    pub(super) fn ensure_native_db(&mut self) {
25        if self.native.is_some() {
26            return;
27        }
28        let mut db = fontdb::Database::new();
29        load_tier1_system_fonts(&mut db);
30        if db.is_empty() {
31            db.load_system_fonts();
32        }
33        self.native = Some(NativeFontDb {
34            db,
35            tier: 1,
36            loaded_by_db_id: HashMap::new(),
37            shared_files: HashMap::new(),
38            codepoint_cache: HashMap::new(),
39        });
40    }
41
42    fn escalate_native_db_tier(&mut self, target_tier: u8) -> bool {
43        let Some(native) = self.native.as_mut() else {
44            return false;
45        };
46        if native.tier >= target_tier {
47            return false;
48        }
49        if native.tier < 2 && target_tier >= 2 {
50            load_tier2_cjk_system_fonts(&mut native.db);
51            native.tier = 2;
52        }
53        if native.tier < 3 && target_tier >= 3 {
54            native.db.load_system_fonts();
55            #[cfg(target_os = "macos")]
56            load_macos_mobile_asset_fonts(&mut native.db);
57            #[cfg(target_os = "visionos")]
58            native.db.load_fonts_dir("/System/Library/Fonts");
59            #[cfg(target_os = "android")]
60            native.db.load_fonts_dir("/system/fonts");
61            native.tier = 3;
62        }
63        native.codepoint_cache.retain(|_, v| v.is_some());
64        true
65    }

67    pub(super) fn resolve_native_face(
68        &mut self,
69        font_stack: &str,
70        ch: char,
71    ) -> Option<(u64, Arc<LoadedFace>)> {
72        self.ensure_native_db();
73        if is_cjk_or_hangul_or_kana(ch) {
74            let _ = self.escalate_native_db_tier(2);
75        }
76        let cache_key = (font_stack.to_owned(), u32::from(ch));
77        if let Some(&cached) = self
78            .native
79            .as_ref()
80            .and_then(|n| n.codepoint_cache.get(&cache_key))
81        {
82            let id = cached?;
83            return self.faces.get(&id).map(|f| (id, Arc::clone(f)));
84        }
85
86        let mut found_db_id = self.find_matching_db_face(font_stack, ch);
87        if found_db_id.is_none() && !ch.is_control() {
88            if self.escalate_native_db_tier(2) {
89                found_db_id = self.find_matching_db_face(font_stack, ch);
90            }
91            if found_db_id.is_none() && self.escalate_native_db_tier(3) {
92                found_db_id = self.find_matching_db_face(font_stack, ch);
93            }
94        }
95        let loaded_id = found_db_id.and_then(|db_id| self.materialize_db_face(db_id));
96        if let Some(native) = self.native.as_mut() {
97            native.codepoint_cache.insert(cache_key, loaded_id);
98        }
99        let id = loaded_id?;
100        self.faces.get(&id).map(|f| (id, Arc::clone(f)))
101    }

103    fn find_matching_db_face(&self, font_stack: &str, ch: char) -> Option<fontdb::ID> {
104        let native = self.native.as_ref()?;
105        let db = &native.db;
106        let face_has_char = |id: fontdb::ID| -> bool {
107            if db.face(id).is_some_and(|info| {
108                info.post_script_name.contains("LastResort")
109                    || info
110                        .families
111                        .iter()
112                        .any(|(name, _)| name.contains("LastResort"))
113            }) {
114                return false;
115            }
116            db.with_face_data(id, |data, idx| {
117                FontRef::from_index(data, idx)
118                    .ok()
119                    .and_then(|f| f.charmap().map(ch))
120                    .is_some_and(|gid| gid != skrifa::GlyphId::NOTDEF)
121            })
122            .unwrap_or(false)
123        };
124
125        let (_, primary_weight, primary_style) =
126            parse_font_stack_entry(font_stack.split(',').next().unwrap_or(font_stack));
127        let primary_db_style = match primary_style {
128            FontStyle::Normal => fontdb::Style::Normal,
129            FontStyle::Italic => fontdb::Style::Italic,
130            FontStyle::Oblique => fontdb::Style::Oblique,
131        };
132
133        for raw_entry in font_stack.split(',') {
134            let (family, weight, style) = parse_font_stack_entry(raw_entry);
135            let db_style = match style {
136                FontStyle::Normal => fontdb::Style::Normal,
137                FontStyle::Italic => fontdb::Style::Italic,
138                FontStyle::Oblique => fontdb::Style::Oblique,
139            };
140            let lower = family.to_ascii_lowercase();
141            if matches!(
142                lower.as_str(),
143                "sans-serif" | "system-ui" | "serif" | "monospace"
144            ) {
145                for fallback_name in preferred_system_fallbacks(ch) {
146                    let query = fontdb::Query {
147                        families: &[fontdb::Family::Name(fallback_name)],
148                        weight: fontdb::Weight(weight.0),
149                        stretch: fontdb::Stretch::Normal,
150                        style: db_style,
151                    };
152                    if let Some(id) = db.query(&query).filter(|&id| face_has_char(id)) {
153                        return Some(id);
154                    }
155                }
156            }
157            let fam = match lower.as_str() {
158                "sans-serif" | "system-ui" => fontdb::Family::SansSerif,
159                "serif" => fontdb::Family::Serif,
160                "monospace" => fontdb::Family::Monospace,
161                _ => fontdb::Family::Name(&family),
162            };
163            let query = fontdb::Query {
164                families: &[fam],
165                weight: fontdb::Weight(weight.0),
166                stretch: fontdb::Stretch::Normal,
167                style: db_style,
168            };
169            if let Some(id) = db.query(&query).filter(|&id| face_has_char(id)) {
170                return Some(id);
171            }
172        }
173
174        for fallback_name in preferred_system_fallbacks(ch) {
175            let query = fontdb::Query {
176                families: &[fontdb::Family::Name(fallback_name)],
177                weight: fontdb::Weight(primary_weight.0),
178                stretch: fontdb::Stretch::Normal,
179                style: primary_db_style,
180            };
181            if let Some(id) = db.query(&query).filter(|&id| face_has_char(id)) {
182                return Some(id);
183            }
184        }
185
186        db.faces().map(|info| info.id).find(|&id| face_has_char(id))
187    }

189    fn materialize_db_face(&mut self, db_id: fontdb::ID) -> Option<u64> {
190        if let Some(&existing) = self
191            .native
192            .as_ref()
193            .and_then(|n| n.loaded_by_db_id.get(&db_id))
194        {
195            return Some(existing);
196        }
197        let (bytes, face_index) = {
198            let native = self.native.as_mut()?;
199            let (source, index) = native.db.face_source(db_id)?;
200            let arc_bytes = match source {
201                fontdb::Source::Binary(bin) => Arc::new((*bin).as_ref().to_vec()),
202                fontdb::Source::File(path) => {
203                    if let Some(shared) = native.shared_files.get(&path) {
204                        Arc::clone(shared)
205                    } else {
206                        let data = std::fs::read(&path).ok()?;
207                        let arc = Arc::new(data);
208                        native.shared_files.insert(path, Arc::clone(&arc));
209                        arc
210                    }
211                }
212                _ => {
213                    let copied = native.db.with_face_data(db_id, |slice, _| slice.to_vec())?;
214                    Arc::new(copied)
215                }
216            };
217            (arc_bytes, index)
218        };
219
220        let font_ref = FontRef::from_index(bytes.as_slice(), face_index).ok()?;
221        let blob: Arc<dyn AsRef<[u8]> + Send + Sync> = Arc::clone(&bytes) as _;
222        let hb_font = HbFont::new(blob, face_index)?;
223        let (families, weight, style) = inspect_face_metadata(&font_ref);
224        self.next_id += 1;
225        let id = self.next_id;
226        self.faces.insert(
227            id,
228            Arc::new(LoadedFace {
229                id,
230                bytes,
231                face_index,
232                hb_font,
233                families,
234                weight,
235                style,
236                unicode_ranges: Vec::new(),
237            }),
238        );
239        if let Some(native) = self.native.as_mut() {
240            native.loaded_by_db_id.insert(db_id, id);
241        }
242        Some(id)
243    }
244}

246fn is_cjk_or_hangul_or_kana(ch: char) -> bool {
247    let cp = u32::from(ch);
248    matches!(
249        cp,
250        0x1100..=0x11FF
251            | 0x3040..=0x31FF
252            | 0xA960..=0xA97F
253            | 0xAC00..=0xD7FF
254            | 0xFF65..=0xFF9F
255            | 0x1B000..=0x1B16F
256    ) || super::super::cjk::is_local_cjk_codepoint(ch)
257}

259fn load_existing_font_files(db: &mut fontdb::Database, paths: &[&str]) {
260    for raw in paths {
261        let p = Path::new(raw);
262        if p.is_file() {
263            let _ = db.load_font_file(p);
264        }
265    }
266}

383fn preferred_system_fallbacks(ch: char) -> &'static [&'static str] {
384    let cp = u32::from(ch);
385    if matches!(cp, 0x3040..=0x30FF | 0x31F0..=0x31FF | 0xFF65..=0xFF9F | 0x1B000..=0x1B16F) {
386        &[
387            "Hiragino Sans",
388            "Hiragino Kaku Gothic ProN",
389            "PingFang SC",
390            "Noto Sans CJK JP",
391            "Yu Gothic",
392            "Arial Unicode MS",
393        ]
394    } else if matches!(cp, 0x1100..=0x11FF | 0x3130..=0x318F | 0xA960..=0xA97F | 0xAC00..=0xD7FF) {
395        &[
396            "Apple SD Gothic Neo",
397            "Noto Sans CJK KR",
398            "Malgun Gothic",
399            "PingFang SC",
400            "Arial Unicode MS",
401        ]
402    } else if super::super::cjk::is_local_cjk_codepoint(ch) {
403        &[
404            "PingFang SC",
405            "Hiragino Sans",
406            "Hiragino Sans GB",
407            "Apple SD Gothic Neo",
408            "Heiti SC",
409            "Songti SC",
410            "STSong",
411            "Kaiti SC",
412            "Noto Sans CJK SC",
413            "Noto Sans CJK JP",
414            "Noto Sans CJK KR",
415            "WenQuanYi Micro Hei",
416            "Microsoft YaHei",
417            "Yu Gothic",
418            "Malgun Gothic",
419            "SimSun-ExtB",
420            "Arial Unicode MS",
421        ]
422    } else {
423        &[
424            "Arial",
425            "Helvetica Neue",
426            "SF Pro",
427            "Kohinoor Devanagari",
428            "Devanagari Sangam MN",
429            "Geeza Pro",
430            "Thonburi",
431            "Noto Sans",
432            "Noto Sans Georgian",
433            "Noto Sans Devanagari",
434            "Noto Sans Arabic",
435            "DejaVu Sans",
436            "Segoe UI",
437            "Nirmala UI",
438            "Arial Unicode MS",
439        ]
440    }

443#[cfg(target_os = "macos")]
444fn load_macos_mobile_asset_fonts(db: &mut fontdb::Database) {
445    let Ok(entries) = std::fs::read_dir("/System/Library/AssetsV2") else {
446        return;
447    };
448    for entry in entries.flatten() {
449        let path = entry.path();
450        if path
451            .file_name()
452            .and_then(|name| name.to_str())
453            .is_some_and(|name| name.starts_with("com_apple_MobileAsset_Font"))
454        {
455            db.load_fonts_dir(path);
456        }
457    }
