1//! Magic-header detection and OpenType `sfnt` table packing for `TTF`, `OTF`, `TTC`, `WOFF`, and `WOFF2`.
2
3use std::borrow::Cow;
4
5use super::FontError;
6
7/// Maximum decompressed font size accepted when unpacking `WOFF` or `WOFF2` containers (128 MiB).
8pub(super) const MAX_DECOMPRESSED_FONT_BYTES: usize = 128 * 1024 * 1024;
9
10/// Binary container format of a font file.
11#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
12pub enum FontContainerFormat {
13    /// Raw TrueType outlines (`0x00010000`, `true`, or `typ1`).
14    Ttf,
15    /// Raw OpenType CFF outlines (`OTTO`).
16    Otf,
17    /// TrueType/OpenType Font Collection (`ttcf`).
18    Ttc,
19    /// Web Open Font Format 1.0 (`wOFF`, zlib-compressed tables).
20    Woff,
21    /// Web Open Font Format 2.0 (`wOF2`, Brotli-compressed stream + transformed `glyf`/`loca`).
22    Woff2,
23}
24
25/// Inspects the 4-byte magic signature at the start of `bytes`.
26pub fn detect_container_format(bytes: &[u8]) -> Option<FontContainerFormat> {
27    let magic: [u8; 4] = bytes.get(..4)?.try_into().ok()?;
28    match &magic {
29        [0x00, 0x01, 0x00, 0x00] | b"true" | b"typ1" => Some(FontContainerFormat::Ttf),
30        b"OTTO" => Some(FontContainerFormat::Otf),
31        b"ttcf" => Some(FontContainerFormat::Ttc),
32        b"wOFF" => Some(FontContainerFormat::Woff),
33        b"wOF2" => Some(FontContainerFormat::Woff2),
34        _ => None,
35    }
36}
37
38/// Unpacks `bytes` into an OpenType/TrueType `sfnt` or `ttcf` byte buffer readable by
39/// `read_fonts`, `skrifa`, and `harfrust`.
40///
41/// Raw `TTF`, `OTF`, and `TTC` slices are returned borrowed without copying; `WOFF` and `WOFF2`
42/// containers are decompressed and reconstructed into valid `sfnt` buffers.
43pub fn unpack_font_container(bytes: &[u8]) -> Result<Cow<'_, [u8]>, FontError> {
44    let format = detect_container_format(bytes).ok_or(FontError::UnrecognizedFormat)?;
45    match format {
46        FontContainerFormat::Ttf | FontContainerFormat::Otf | FontContainerFormat::Ttc => {
47            if bytes.len() < 12 {
48                return Err(FontError::InvalidContainer("sfnt header truncated"));
49            }
50            Ok(Cow::Borrowed(bytes))
51        }
52        FontContainerFormat::Woff => super::woff1::unpack_woff1(bytes).map(Cow::Owned),
53        FontContainerFormat::Woff2 => super::woff2::unpack_woff2(bytes).map(Cow::Owned),
54    }
55}

57/// One decoded OpenType table ready to be written into a reconstructed `sfnt` buffer.
58pub(super) struct SfntTable {
59    pub tag: [u8; 4],
60    pub data: Vec<u8>,
61}
62
63/// Computes the OpenType 32-bit big-endian table checksum over `data` (zero-padded to 4 bytes).
64pub(super) fn sfnt_checksum(data: &[u8]) -> u32 {
65    let mut sum = 0_u32;
66    let (chunks, rem) = data.as_chunks::<4>();
67    for &chunk in chunks {
68        sum = sum.wrapping_add(u32::from_be_bytes(chunk));
69    }
70    if !rem.is_empty() {
71        let mut tail = [0_u8; 4];
72        tail[..rem.len()].copy_from_slice(rem);
73        sum = sum.wrapping_add(u32::from_be_bytes(tail));
74    }
75    sum
76}

78/// Assembles `tables` with `flavor` into a canonical OpenType `sfnt` byte vector, sorting the
79/// table directory by tag and updating `head.checkSumAdjustment` (`0xB1B0AFBA - total_checksum`).
80pub(super) fn assemble_sfnt(flavor: u32, mut tables: Vec<SfntTable>) -> Result<Vec<u8>, FontError> {
81    if tables.is_empty() || tables.len() > u16::MAX as usize {
82        return Err(FontError::InvalidContainer("invalid sfnt table count"));
83    }
84    tables.sort_by_key(|table| table.tag);
85
86    let num_tables = tables.len() as u16;
87    let max_pow2 = 1_u16 << num_tables.ilog2();
88    let search_range = max_pow2.saturating_mul(16);
89    let entry_selector = num_tables.ilog2() as u16;
90    let range_shift = num_tables.saturating_mul(16).saturating_sub(search_range);
91
92    let dir_size = 12usize + usize::from(num_tables) * 16;
93    let body_size: usize = tables
94        .iter()
95        .map(|t| (t.data.len() + 3) & !3)
96        .try_fold(0usize, usize::checked_add)
97        .ok_or(FontError::InvalidContainer("sfnt size overflow"))?;
98    let total_size = dir_size
99        .checked_add(body_size)
100        .filter(|&size| size <= MAX_DECOMPRESSED_FONT_BYTES)
101        .ok_or(FontError::InvalidContainer("sfnt exceeds maximum size"))?;
102
103    let mut out = vec![0_u8; total_size];
104    out[0..4].copy_from_slice(&flavor.to_be_bytes());
105    out[4..6].copy_from_slice(&num_tables.to_be_bytes());
106    out[6..8].copy_from_slice(&search_range.to_be_bytes());
107    out[8..10].copy_from_slice(&entry_selector.to_be_bytes());
108    out[10..12].copy_from_slice(&range_shift.to_be_bytes());
109
110    let mut head_offset: Option<usize> = None;
111    let mut data_cursor = dir_size;
112    for (idx, mut table) in tables.into_iter().enumerate() {
113        if &table.tag == b"head" && table.data.len() >= 12 {
114            // Zero `checkSumAdjustment` before computing `head`'s table checksum.
115            table.data[8..12].copy_from_slice(&[0, 0, 0, 0]);
116            head_offset = Some(data_cursor);
117        }
118        let checksum = sfnt_checksum(&table.data);
119        let offset_u32 = u32::try_from(data_cursor)
120            .map_err(|_| FontError::InvalidContainer("table offset overflow"))?;
121        let length_u32 = u32::try_from(table.data.len())
122            .map_err(|_| FontError::InvalidContainer("table length overflow"))?;
123
124        let rec = 12 + idx * 16;
125        out[rec..rec + 4].copy_from_slice(&table.tag);
126        out[rec + 4..rec + 8].copy_from_slice(&checksum.to_be_bytes());
127        out[rec + 8..rec + 12].copy_from_slice(&offset_u32.to_be_bytes());
128        out[rec + 12..rec + 16].copy_from_slice(&length_u32.to_be_bytes());
129
130        out[data_cursor..data_cursor + table.data.len()].copy_from_slice(&table.data);
131        data_cursor += (table.data.len() + 3) & !3;
132    }
133
134    if let Some(head_pos) = head_offset {
135        let adjustment = 0xB1B0_AFBA_u32.wrapping_sub(sfnt_checksum(&out));
136        out[head_pos + 8..head_pos + 12].copy_from_slice(&adjustment.to_be_bytes());
137    }
138
139    Ok(out)
140}
