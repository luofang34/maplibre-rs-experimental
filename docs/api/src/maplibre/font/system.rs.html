29/// One parsed OpenType/TrueType face stored inside [`FontSystem`].
30#[derive(Clone)]
31pub struct LoadedFace {
32    /// Unique identifier of this face within its [`FontSystem`].
33    pub id: u64,
34    /// Shared decompressed `sfnt` or `ttcf` container bytes.
35    pub bytes: Arc<Vec<u8>>,
36    /// Collection index (`0` for single-face fonts).
37    pub face_index: u32,
38    /// Cached `harfrust::Font` handle sharing `bytes`.
39    #[cfg(feature = "font-shaping")]
40    pub hb_font: HbFont,
41    /// Normalized lowercase family names matched by `"text-font"` entries.
42    pub families: Vec<String>,
43    /// Weight class (`100..=900`).
44    pub weight: FontWeight,
45    /// Posture (`normal`, `italic`, `oblique`).
46    pub style: FontStyle,
47    /// Optional CSS `unicode-range` filter (`empty` = all codepoints in `cmap`).
48    pub unicode_ranges: Vec<UnicodeRange>,
49}

63impl LoadedFace {
64    /// Returns `true` when `ch` passes `unicode_ranges` and maps to a non-`.notdef` glyph in `cmap`.
65    pub fn covers_char(&self, ch: char) -> bool {
66        #[cfg(feature = "font-shaping")]
67        {
68            let cp = u32::from(ch);
69            let in_range = self.unicode_ranges.is_empty()
70                || self.unicode_ranges.iter().any(|range| range.contains(cp));
71            in_range && self.hb_font.charmap().map_unicode(ch).is_some()
72        }
73        #[cfg(not(feature = "font-shaping"))]
74        {
75            let _ = ch;
76            false
77        }
78    }
79
80    /// Rasterises the nominal `cmap` glyph for `ch` into a 24px SDF bitmap.
81    pub fn rasterize_char(&self, ch: char) -> Option<RasterizedGlyph> {
82        rasterize_char_sdf(&self.bytes, self.face_index, ch)
83    }
84
85    /// Rasterises `glyph_id` (including unencoded `GSUB` glyphs) into a 24px SDF bitmap.
86    pub fn rasterize_glyph(&self, glyph_id: u32) -> Option<RasterizedGlyph> {
87        rasterize_glyph_sdf(&self.bytes, self.face_index, glyph_id)
88    }
89}

91/// Registry of in-memory OpenType/WOFF/WOFF2 fonts, style `font-faces`, and lazy native system

113fn rule_dedup_key(rule: &FontFaceRule, url: &str) -> String {
114    use std::fmt::Write;
115    let style = match rule.style {
116        FontStyle::Normal => "n",
117        FontStyle::Italic => "i",
118        FontStyle::Oblique => "o",
119    };
120    let mut key = format!(
121        "{url}|{}|{}|{style}",
122        rule.family.to_ascii_lowercase(),
123        rule.weight.0
124    );
125    for r in &rule.unicode_ranges {
126        let _ = write!(key, "|{:x}-{:x}", r.start, r.end);
127    }
128    key
129}

147    /// Unpacks a `TTF`, `OTF`, `TTC`, `WOFF`, or `WOFF2` font buffer and registers every face in
148    /// memory across all targets (including `wasm32` when `font-shaping` is enabled).
149    pub fn register_font_bytes(
150        &mut self,
151        family_override: Option<&str>,
152        data: Vec<u8>,
153    ) -> Result<Vec<u64>, FontError> {
154        #[cfg(feature = "font-shaping")]
155        {
156            let unpacked = unpack_font_container(&data)?.into_owned();
157            self.insert_sfnt_faces(
158                Arc::new(unpacked),
159                family_override,
160                None,
161                None,
162                Vec::new(),
163                true,
164            )
165        }
166        #[cfg(not(feature = "font-shaping"))]
167        {
168            let _ = (family_override, data);
169            Err(FontError::InvalidContainer(
170                "in-process OpenType registration requires the `font-shaping` feature",
171            ))
172        }
173    }
174
175    /// Returns `true` when `rule` at `url` has already been unpacked and registered.
176    pub fn has_registered_rule(&self, rule: &FontFaceRule, url: &str) -> bool {
177        self.rule_keys.contains_key(&rule_dedup_key(rule, url))
178    }
179
180    /// Unpacks and registers a font fetched for a [`FontFaceRule`], deduplicating by URL and
181    /// descriptor metadata so repeated tile loads reuse the same face IDs.
182    pub fn register_font_face_rule(
183        &mut self,
184        rule: &FontFaceRule,
185        url: &str,
186        data: Vec<u8>,
187    ) -> Result<Vec<u64>, FontError> {
188        let dedup_key = rule_dedup_key(rule, url);
189        if let Some(existing) = self.rule_keys.get(&dedup_key) {
190            return Ok(existing.clone());
191        }
192        #[cfg(feature = "font-shaping")]
193        {
194            let unpacked = unpack_font_container(&data)?.into_owned();
195            let ids = self.insert_sfnt_faces(
196                Arc::new(unpacked),
197                Some(&rule.family),
198                Some(rule.weight),
199                Some(rule.style),
200                rule.unicode_ranges.clone(),
201                true,
202            )?;
203            self.rule_keys.insert(dedup_key, ids.clone());
204            Ok(ids)
205        }
206        #[cfg(not(feature = "font-shaping"))]
207        {
208            let _ = data;
209            Err(FontError::InvalidContainer(
210                "in-process OpenType registration requires the `font-shaping` feature",
211            ))
212        }
213    }

220    /// Returns `true` when at least one in-memory / `font-faces` face matches `font_stack`.
221    pub fn has_registered_stack(&self, font_stack: &str) -> bool {
222        for raw_entry in font_stack.split(',') {
223            let (family, _, _) = parse_font_stack_entry(raw_entry);
224            let norm = family.to_ascii_lowercase();
225            let raw_norm = raw_entry
226                .trim()
227                .trim_matches(['"', '\''])
228                .to_ascii_lowercase();
229            if self.registered_order.iter().any(|id| {
230                self.faces
231                    .get(id)
232                    .is_some_and(|face| face.families.iter().any(|f| f == &norm || f == &raw_norm))
233            }) {
234                return true;
235            }
236        }
237        false
238    }

240    /// Resolves the best [`LoadedFace`] for `ch` given a comma-separated MapLibre `font_stack`.
241    pub fn resolve_face_for_char(
242        &mut self,
243        font_stack: &str,
244        ch: char,
245        allow_system_fallback: bool,
246    ) -> Option<(u64, Arc<LoadedFace>)> {
247        for raw_entry in font_stack.split(',') {
248            let (family, weight, style) = parse_font_stack_entry(raw_entry);
249            let raw_norm = raw_entry
250                .trim()
251                .trim_matches(['"', '\''])
252                .to_ascii_lowercase();
253            if let Some(hit) = self.best_registered_face(&family, &raw_norm, weight, style, ch) {
254                return Some(hit);
255            }
256        }
257
258        if allow_system_fallback {
259            #[cfg(all(not(target_arch = "wasm32"), feature = "font-shaping"))]
260            if let Some(hit) = self.resolve_native_face(font_stack, ch) {
261                return Some(hit);
262            }
263
264            for &id in &self.registered_order {
265                if let Some(face) = self.faces.get(&id) {
266                    if face.covers_char(ch) {
267                        return Some((id, Arc::clone(face)));
268                    }
269                }
270            }
271        }
272
273        None
274    }

276    /// Rasterises `ch` locally for `localIdeographFontFamily` or missing-glyph fallback.
277    pub fn rasterize_local_char(
278        &mut self,
279        ideograph_family: Option<&str>,
280        font_stack: &str,
281        ch: char,
282    ) -> Option<RasterizedGlyph> {
283        if ch.is_control() || matches!(ch, '\u{200B}'..='\u{200D}' | '\u{FEFF}') {
284            return None;
285        }
286        if let Some((_, face)) = self.resolve_face_for_char(font_stack, ch, false) {
287            if let Some(glyph) = face.rasterize_char(ch) {
288                return Some(glyph);
289            }
290        }
291        let style_suffix = stack_style_suffix(font_stack);
292        let combined = match ideograph_family {
293            Some(family) if !family.is_empty() => {
294                let styled: Vec<String> = family
295                    .split(',')
296                    .map(|f| format!("{}{}", f.trim(), style_suffix))
297                    .collect();
298                if font_stack.is_empty() {
299                    styled.join(",")
300                } else {
301                    format!("{},{}", styled.join(","), font_stack)
302                }
303            }
304            _ => font_stack.to_owned(),
305        };
306        if let Some((_, face)) = self.resolve_face_for_char(&combined, ch, true) {
307            if let Some(glyph) = face.rasterize_char(ch) {
308                return Some(glyph);
309            }
310        }
311        #[cfg(target_arch = "wasm32")]
312        {
313            super::sdf::canvas2d::rasterize_canvas2d_char_sdf(ideograph_family, font_stack, ch)
314        }
315        #[cfg(not(target_arch = "wasm32"))]
316        {
317            None
318        }
319    }

321    /// Shapes `text` with `harfrust` using OpenType faces resolved from `font_stack`.
322    pub fn shape_text(
323        &mut self,
324        font_stack: &str,
325        text: &str,
326        allow_system_fallback: bool,
327    ) -> Option<Vec<ShapedGlyph>> {
328        #[cfg(feature = "font-shaping")]
329        {
330            shape_text_runs(text, |ch| {
331                if ch.is_whitespace() {
332                    if let Some((id, face)) =
333                        self.resolve_face_for_char(font_stack, ' ', allow_system_fallback)
334                    {
335                        if ch == ' ' || face.covers_char(ch) {
336                            return Some((id, face.hb_font.clone()));
337                        }
338                    }
339                }
340                self.resolve_face_for_char(font_stack, ch, allow_system_fallback)
341                    .map(|(id, face)| (id, face.hb_font.clone()))
342            })
343        }
344        #[cfg(not(feature = "font-shaping"))]
345        {
346            let _ = (font_stack, text, allow_system_fallback);
347            None
348        }
349    }

351    fn best_registered_face(
352        &self,
353        family: &str,
354        raw_norm: &str,
355        target_weight: FontWeight,
356        target_style: FontStyle,
357        ch: char,
358    ) -> Option<(u64, Arc<LoadedFace>)> {
359        let norm = family.to_ascii_lowercase();
360        let generic_any = matches!(
361            norm.as_str(),
362            "sans-serif" | "serif" | "monospace" | "system-ui"
363        );
364        let mut best: Option<(u32, u64, Arc<LoadedFace>)> = None;
365        for &id in &self.registered_order {
366            let Some(face) = self.faces.get(&id) else {
367                continue;
368            };
369            let name_match = face.families.iter().any(|f| f == &norm || f == raw_norm);
370            if !name_match && !generic_any {
371                continue;
372            }
373            if !face.covers_char(ch) {
374                continue;
375            }
376            let style_penalty = if face.style == target_style { 0 } else { 500 };
377            let generic_penalty = if name_match { 0 } else { 2000 };
378            let weight_diff = u32::from(face.weight.0.abs_diff(target_weight.0));
379            let score = generic_penalty + style_penalty + weight_diff;
380            if best
381                .as_ref()
382                .is_none_or(|(best_score, _, _)| score < *best_score)
383            {
384                best = Some((score, id, Arc::clone(face)));
385            }
386        }
387        best.map(|(_, id, face)| (id, face))
388    }

390    #[cfg(feature = "font-shaping")]
391    fn insert_sfnt_faces(
392        &mut self,
393        bytes: Arc<Vec<u8>>,
394        family_override: Option<&str>,
395        weight_override: Option<FontWeight>,
396        style_override: Option<FontStyle>,
397        unicode_ranges: Vec<UnicodeRange>,
398        record_registered: bool,
399    ) -> Result<Vec<u64>, FontError> {
400        let file = FileRef::new(bytes.as_slice())
401            .map_err(|_| FontError::InvalidContainer("unreadable OpenType sfnt/ttcf header"))?;
402        let face_count = match &file {
403            FileRef::Font(_) => 1_u32,
404            FileRef::Collection(col) => col.len(),
405        };
406
407        let blob: Arc<dyn AsRef<[u8]> + Send + Sync> = Arc::clone(&bytes) as _;
408        let mut added = Vec::new();
409        for index in 0..face_count {
410            let Ok(font_ref) = FontRef::from_index(bytes.as_slice(), index) else {
411                continue;
412            };
413            let Some(hb_font) = HbFont::new(Arc::clone(&blob), index) else {
414                continue;
415            };
416            let (meta_families, meta_weight, meta_style) = inspect_face_metadata(&font_ref);
417            let mut families = meta_families;
418            if let Some(custom) = family_override {
419                let clean = custom.trim().trim_matches(['"', '\'']).to_ascii_lowercase();
420                if !clean.is_empty() && !families.contains(&clean) {
421                    families.insert(0, clean);
422                }
423            }
424            if families.is_empty() {
425                families.push("sans-serif".to_owned());
426            }
427
428            self.next_id += 1;
429            let id = self.next_id;
430            let loaded = Arc::new(LoadedFace {
431                id,
432                bytes: Arc::clone(&bytes),
433                face_index: index,
434                hb_font,
435                families,
436                weight: weight_override.unwrap_or(meta_weight),
437                style: style_override.unwrap_or(meta_style),
438                unicode_ranges: unicode_ranges.clone(),
439            });
440            self.faces.insert(id, loaded);
441            if record_registered {
442                self.registered_order.push(id);
443            }
444            added.push(id);
445        }
446
447        if added.is_empty() {
448            return Err(FontError::InvalidContainer(
449                "no valid OpenType faces in container",
450            ));
451        }
452        Ok(added)
453    }
