1//! OpenType `GSUB` + `GPOS` complex text shaping powered by `harfrust`.
2//!
3//! Supports unencoded `GlyphId` substitutions (Indic half-forms/conjuncts, Devanagari/Bengali/
4//! Tamil/Thai/Khmer/Myanmar reordering, Latin `fi`/`fl` ligatures, Arabic contextual forms),
5//! `GPOS` kerning, cursive attachment, and combining-mark positioning (`mark`/`mkmk`).

15/// Atlas lookup key for either a nominal Unicode scalar or an OpenType glyph index.
16#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash, Serialize, Deserialize)]
17pub enum GlyphKey {
18    /// Direct Unicode scalar (`char as u32`), used for `.pbf` ranges and unshaped CJK glyphs.
19    Codepoint(u32),
20    /// Font-specific OpenType glyph index produced by `harfrust` shaping.
21    GlyphId {
22        /// Identifier of the owning face inside [`super::FontSystem`].
23        face_id: u64,
24        /// OpenType glyph index (`0` = `.notdef`).
25        glyph_id: u32,
26    },
27}

29/// Formats `(face_id, glyph_id)` as a deterministic string key for JSON-compatible atlas maps.
30pub fn face_glyph_atlas_key(face_id: u64, glyph_id: u32) -> String {
31    format!("{face_id}:{glyph_id}")
32}

34/// One positioned glyph emitted by OpenType shaping, scaled to MapLibre's `24.0` px em box.
35#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
36pub struct ShapedGlyph {
37    /// Identifier of the face in [`super::FontSystem`] that owns `glyph_id`.
38    pub face_id: u64,
39    /// OpenType glyph index (`harfrust::GlyphInfo::glyph_id`).
40    pub glyph_id: u32,
41    /// UTF-8 byte offset of the originating cluster within the input string.
42    pub cluster: u32,
43    /// Horizontal pen advance in 24px-em pixels.
44    pub x_advance: f32,
45    /// Vertical pen advance in 24px-em pixels.
46    pub y_advance: f32,
47    /// Horizontal offset from the current pen position in 24px-em pixels (`+x` rightward).
48    pub x_offset: f32,
49    /// Vertical offset from the baseline in 24px-em pixels (`+y` upward in font design space).
50    pub y_offset: f32,
51    /// `true` when this glyph belongs to a right-to-left BiDi run.
52    #[serde(default)]
53    pub rtl: bool,
54}
55
56impl ShapedGlyph {
57    /// String key used in `SymbolAtlas::face_glyphs`.
58    pub fn atlas_key(&self) -> String {
59        face_glyph_atlas_key(self.face_id, self.glyph_id)
60    }
61}

63/// Shapes a single unidirectional slice `full_text[byte_range]` with `font` via `harfrust`.
64///
65/// Cluster offsets in the returned [`ShapedGlyph`]s are UTF-8 byte indices into `full_text` plus
66/// `cluster_base`.
67pub fn shape_slice_with_font(
68    face_id: u64,
69    font: &HbFont,
70    full_text: &str,
71    byte_range: Range<usize>,
72    cluster_base: usize,
73    rtl: bool,
74) -> Vec<ShapedGlyph> {
75    let Some(slice) = full_text.get(byte_range.clone()) else {
76        return Vec::new();
77    };
78    if slice.is_empty() {
79        return Vec::new();
80    }
81
82    let shaper = ShaperFont::new(font);
83    let upem = f32::from(font.units_per_em().max(1));
84    let scale = SDF_EM_SIZE / upem;
85
86    let mut buffer = Buffer::new();
87    buffer.set_flags(BufferFlags::BEGINNING_OF_TEXT | BufferFlags::END_OF_TEXT);
88    if let Some(pre) = full_text.get(..byte_range.start) {
89        buffer.set_pre_context(pre);
90    }
91    if let Some(post) = full_text.get(byte_range.end..) {
92        buffer.set_post_context(post);
93    }
94    for (rel_idx, ch) in slice.char_indices() {
95        if is_bidi_isolate_or_override(ch) {
96            continue;
97        }
98        let cluster = (cluster_base + byte_range.start + rel_idx) as u32;
99        buffer.push(u32::from(ch), cluster);
100    }
101    if buffer.glyph_infos().is_empty() {
102        return Vec::new();
103    }
104
105    buffer.set_direction(if rtl {
106        Direction::RightToLeft
107    } else {
108        Direction::LeftToRight
109    });
110    buffer.guess_segment_properties();
111
112    if harfrust::shape(&shaper, &mut buffer, ShapeOptions::new()).is_err() {
113        return Vec::new();
114    }
115
116    let infos = buffer.glyph_infos();
117    let positions = buffer.glyph_positions();
118    infos
119        .iter()
120        .zip(positions.iter())
121        .map(|(info, pos)| ShapedGlyph {
122            face_id,
123            glyph_id: info.glyph_id,
124            cluster: info.cluster,
125            x_advance: pos.x_advance as f32 * scale,
126            y_advance: pos.y_advance as f32 * scale,
127            x_offset: pos.x_offset as f32 * scale,
128            y_offset: pos.y_offset as f32 * scale,
129            rtl,
130        })
131        .collect()
132}

134/// Shapes `text` across newline-separated paragraphs, BiDi visual runs, and per-character font
135/// fallback resolved by `resolve_face`.
136///
137/// Returns `None` if `text` has no renderable characters or if any non-ignorable character fails
138/// to resolve an OpenType [`HbFont`], allowing callers to fall back to `.pbf` glyph layout.
139pub fn shape_text_runs<F>(text: &str, mut resolve_face: F) -> Option<Vec<ShapedGlyph>>
140where
141    F: FnMut(char) -> Option<(u64, HbFont)>,
142{
143    if text.is_empty() {
144        return None;
145    }
146
147    let mut all_glyphs = Vec::new();
148    let mut byte_cursor = 0usize;
149    for line in text.split('\n') {
150        if !line.is_empty() {
151            let line_glyphs = shape_single_line(line, byte_cursor, &mut resolve_face)?;
152            all_glyphs.extend(line_glyphs);
153        }
154        byte_cursor += line.len() + 1;
155    }
156
157    (!all_glyphs.is_empty()).then_some(all_glyphs)
158}

160fn shape_single_line<F>(
161    line: &str,
162    cluster_base: usize,
163    resolve_face: &mut F,
164) -> Option<Vec<ShapedGlyph>>
165where
166    F: FnMut(char) -> Option<(u64, HbFont)>,
167{
168    let bidi = BidiInfo::new(line, None);
169    let mut line_glyphs = Vec::new();
170
171    for para in &bidi.paragraphs {
172        let (levels, visual_runs) = bidi.visual_runs(para, para.range.clone());
173        for run in visual_runs {
174            let rtl = levels.get(run.start).is_some_and(|lvl| lvl.is_rtl());
175            let sub_runs = split_by_face(line, run, resolve_face)?;
176            let ordered: Box<dyn Iterator<Item = (Range<usize>, u64, HbFont)>> = if rtl {
177                Box::new(sub_runs.into_iter().rev())
178            } else {
179                Box::new(sub_runs.into_iter())
180            };
181            for (sub_range, face_id, font) in ordered {
182                line_glyphs.extend(shape_slice_with_font(
183                    face_id,
184                    &font,
185                    line,
186                    sub_range,
187                    cluster_base,
188                    rtl,
189                ));
190            }
191        }
192    }
193
194    Some(line_glyphs)
195}

197/// Partitions `line[run]` into contiguous `(byte_range, face_id, HbFont)` spans while keeping
198/// combining marks, joiners (`ZWJ`/`ZWNJ`), and variation selectors attached to the preceding
199/// base character's font face so grapheme clusters never split mid-syllable.
200fn split_by_face<F>(
201    line: &str,
202    run: Range<usize>,
203    resolve_face: &mut F,
204) -> Option<Vec<(Range<usize>, u64, HbFont)>>
205where
206    F: FnMut(char) -> Option<(u64, HbFont)>,
207{
208    let slice = line.get(run.clone())?;
209    let mut spans: Vec<(Range<usize>, u64, HbFont)> = Vec::new();
210
211    for (rel_idx, ch) in slice.char_indices() {
212        let start = run.start + rel_idx;
213        let end = start + ch.len_utf8();
214
215        if is_cluster_continuation(ch) || is_bidi_isolate_or_override(ch) {
216            if let Some((last_range, _, _)) = spans.last_mut() {
217                last_range.end = end;
218                continue;
219            }
220            if is_bidi_isolate_or_override(ch) {
221                continue;
222            }
223        }
224
225        let (face_id, font) = resolve_face(ch)?;
226        match spans.last_mut() {
227            Some((last_range, last_id, _)) if *last_id == face_id && last_range.end == start => {
228                last_range.end = end;
229            }
230            _ => spans.push((start..end, face_id, font)),
231        }
232    }
233
234    Some(spans)
235}

335fn is_bidi_isolate_or_override(ch: char) -> bool {
336    matches!(ch, '\u{200E}' | '\u{200F}' | '\u{202A}'..='\u{202E}' | '\u{2066}'..='\u{2069}')
