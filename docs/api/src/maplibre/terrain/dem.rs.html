8/// Failure while decoding a DEM tile or filling its border.
9#[derive(Clone, Copy, Debug, Error, PartialEq)]
10pub enum DemError {
11    /// Elevation tiles must be square.
12    #[error("DEM tiles must be square, got {width}x{height} pixels")]
13    NotSquare {
14        /// Image width in pixels.
15        width: u32,
16        /// Image height in pixels.
17        height: u32,
18    },
19    /// Elevation tiles must contain at least one sample.
20    #[error("DEM tiles must not be empty")]
21    Empty,
22    /// Neighbouring tiles must have the same number of samples to share borders.
23    #[error("DEM neighbour has {actual} samples per edge, expected {expected}")]
24    DimensionMismatch {
25        /// Samples per edge of this tile.
26        expected: u32,
27        /// Samples per edge of the neighbour.
28        actual: u32,
29    },
30    /// A border must face one of the eight adjacent tiles.
31    #[error("invalid DEM neighbour offset ({dx}, {dy})")]
32    InvalidNeighbour {
33        /// Horizontal offset.
34        dx: i32,
35        /// Vertical offset.
36        dy: i32,
37    },
38    /// Encoded samples cannot be copied between differently encoded tiles.
39    #[error("DEM neighbour unpack vector {actual:?} differs from {expected:?}")]
40    EncodingMismatch {
41        /// This tile's channel factors and base shift.
42        expected: [f64; 4],
43        /// The neighbour's channel factors and base shift.
44        actual: [f64; 4],
45    },
46    /// The border must be supplied in full to avoid partially updated samples.
47    #[error("DEM border ({dx}, {dy}) requires {expected} bytes, got {actual}")]
48    BorderLength {
49        /// Horizontal offset.
50        dx: i32,
51        /// Vertical offset.
52        dy: i32,
53        /// Required byte count.
54        expected: usize,
55        /// Supplied byte count.
56        actual: usize,
57    },
58}

60/// Elevation samples of one tile with a two-pixel border, following GL JS `DEMData`.
61///
62/// Pixels keep their RGB encoding so the same bytes upload to the GPU unchanged; the unpack
63/// vector turns a channel triple into metres on both sides. The border replicates the nearest
64/// edge sample so interpolation stays continuous until neighbours can fill it.
65#[derive(Clone, Debug, PartialEq)]
66pub struct DemTile {
67    dim: u32,
68    stride: u32,
69    pixels: Vec<u8>,
70    unpack: [f64; 4],
71    min: f64,
72    max: f64,
73}
74
75impl DemTile {
76    /// Builds a tile from a decoded image and the source's unpack vector.
77    pub fn from_image(image: &RgbaImage, unpack: [f64; 4]) -> Result<Self, DemError> {
78        Self::decode(image, unpack, None)
79    }
80
81    /// Builds a tile like [`from_image`](Self::from_image), lowering the samples that rise more
82    /// steeply than terrain on both flanks for a sample `sample_metres` across.
83    pub fn from_image_despiked(
84        image: &RgbaImage,
85        unpack: [f64; 4],
86        sample_metres: f64,
87    ) -> Result<Self, DemError> {
88        Self::decode(image, unpack, Some(sample_metres))
89    }
90
91    fn decode(
92        image: &RgbaImage,
93        unpack: [f64; 4],
94        sample_metres: Option<f64>,
95    ) -> Result<Self, DemError> {
96        let (width, height) = image.dimensions();
97        if width != height {
98            return Err(DemError::NotSquare { width, height });
99        }
100        if width == 0 {
101            return Err(DemError::Empty);
102        }
103        let dim = width;
104        let stride = dim + 4;
105        let mut tile = Self {
106            dim,
107            stride,
108            pixels: vec![0; (stride * stride * 4) as usize],
109            unpack,
110            min: f64::INFINITY,
111            max: f64::NEG_INFINITY,
112        };
113        let source = image.as_raw();
114        for y in 0..dim {
115            let source_row = (y * dim * 4) as usize;
116            let target_row = tile.byte_index(0, i64::from(y));
117            tile.pixels[target_row..target_row + (dim * 4) as usize]
118                .copy_from_slice(&source[source_row..source_row + (dim * 4) as usize]);
119        }
120        if let Some(sample_metres) = sample_metres {
121            let replaced = tile.remove_spikes(sample_metres);
122            if replaced > 0 {
123                tracing::debug!(replaced, sample_metres, "removed DEM spike samples");
124            }
125        }
126        tile.replicate_border();
127        for y in 0..dim {
128            for x in 0..dim {
129                let elevation = tile.get(i64::from(x), i64::from(y));
130                tile.min = tile.min.min(elevation);
131                tile.max = tile.max.max(elevation);
132            }
133        }
134        Ok(tile)
135    }
136
137    fn replicate_border(&mut self) {
138        let dim = i64::from(self.dim);
139        for y in -2..dim + 2 {
140            if y < 0 || y >= dim {
141                let from = self.byte_index(0, y.clamp(0, dim - 1));
142                let to = self.byte_index(0, y);
143                self.pixels
144                    .copy_within(from..from + self.dim as usize * 4, to);
145            }
146            for x in [-2, -1, dim, dim + 1] {
147                self.copy_pixel((x.clamp(0, dim - 1), y), (x, y));
148            }
149        }
150    }
151
152    fn copy_pixel(&mut self, from: (i64, i64), to: (i64, i64)) {
153        let from = self.byte_index(from.0, from.1);
154        let to = self.byte_index(to.0, to.1);
155        self.pixels.copy_within(from..from + 4, to);
156    }
157
158    /// Byte offset in the two-pixel padded image.
159    fn byte_index(&self, x: i64, y: i64) -> usize {
160        let stride = i64::from(self.stride);
161        (((y + 2) * stride + (x + 2)) * 4) as usize
162    }
163
164    /// Elevation in metres, clamped to the padded range `-2..=dim + 1`.
165    pub fn get(&self, x: i64, y: i64) -> f64 {
166        let x = x.clamp(-2, i64::from(self.dim) + 1);
167        let y = y.clamp(-2, i64::from(self.dim) + 1);
168        let index = self.byte_index(x, y);
169        let [red, green, blue, base] = self.unpack;
170        f64::from(self.pixels[index]) * red
171            + f64::from(self.pixels[index + 1]) * green
172            + f64::from(self.pixels[index + 2]) * blue
173            - base
174    }
175
176    /// Bilinear elevation in sample space, clamped to `-1..=dim`.
177    ///
178    /// Integer coordinates address pixel centres; negative fractions interpolate the west or
179    /// north border with the first interior sample.
180    pub fn sample_bilinear(&self, x: f64, y: f64) -> f64 {
181        let x = x.clamp(-1.0, f64::from(self.dim));
182        let y = y.clamp(-1.0, f64::from(self.dim));
183        let cx = x.floor();
184        let cy = y.floor();
185        let (tx, ty) = (x - cx, y - cy);
186        let (cx, cy) = (cx as i64, cy as i64);
187        let top = self.get(cx, cy) * (1.0 - tx) + self.get(cx + 1, cy) * tx;
188        let bottom = self.get(cx, cy + 1) * (1.0 - tx) + self.get(cx + 1, cy + 1) * tx;
189        top * (1.0 - ty) + bottom * ty
190    }
191
192    /// Elevation at tile coordinates in `0..=EXTENT`. As in GL JS, a sample sits at the integer
193    /// coordinate of its texel, not at the texel's centre.
194    pub fn elevation_at_tile_coords(&self, x: f64, y: f64) -> f64 {
195        let scale = f64::from(self.dim) / EXTENT;
196        self.sample_bilinear(x.clamp(0.0, EXTENT) * scale, y.clamp(0.0, EXTENT) * scale)
197    }

199    /// Replaces the border facing a neighbour at `(dx, dy)` with that neighbour's edge samples.
200    pub fn backfill_border(
201        &mut self,
202        neighbour: &DemTile,
203        dx: i32,
204        dy: i32,
205    ) -> Result<(), DemError> {
206        self.validate_neighbour(neighbour.dim, neighbour.unpack)?;
207        self.fill_border(dx, dy, &neighbour.edge_samples(dx, dy)?)?;
208        Ok(())
209    }
210
211    pub(super) fn validate_neighbour(&self, dim: u32, unpack: [f64; 4]) -> Result<(), DemError> {
212        if dim != self.dim {
213            return Err(DemError::DimensionMismatch {
214                expected: self.dim,
215                actual: dim,
216            });
217        }
218        if unpack != self.unpack {
219            return Err(DemError::EncodingMismatch {
220                expected: self.unpack,
221                actual: unpack,
222            });
223        }
224        Ok(())
225    }

227    /// Samples of this tile that a neighbour at `(-dx, -dy)` stores in its border facing us.
228    ///
229    /// Row-major over two columns or rows, or a 2x2 corner when both offsets are set.
230    pub fn edge_samples(&self, dx: i32, dy: i32) -> Result<Vec<u8>, DemError> {
231        let dim = i64::from(self.dim);
232        let (x_range, y_range) = Self::border_region(dim, dx, dy)?;
233        let count = (x_range.end - x_range.start) * (y_range.end - y_range.start);
234        let mut samples = Vec::with_capacity(count.max(0) as usize * 4);
235        for y in y_range {
236            for x in x_range.clone() {
237                let index = self.byte_index(x - i64::from(dx) * dim, y - i64::from(dy) * dim);
238                samples.extend_from_slice(&self.pixels[index..index + 4]);
239            }
240        }
241        Ok(samples)
242    }

244    /// Writes samples produced by a neighbour's [`edge_samples`](Self::edge_samples) into the
245    /// border facing that neighbour at `(dx, dy)`.
246    /// Returns whether any pixels changed.
247    pub fn fill_border(&mut self, dx: i32, dy: i32, samples: &[u8]) -> Result<bool, DemError> {
248        let dim = i64::from(self.dim);
249        let (x_range, y_range) = Self::border_region(dim, dx, dy)?;
250        let expected =
251            (x_range.end - x_range.start) as usize * (y_range.end - y_range.start) as usize * 4;
252        if samples.len() != expected {
253            return Err(DemError::BorderLength {
254                dx,
255                dy,
256                expected,
257                actual: samples.len(),
258            });
259        }
260        let mut changed = false;
261        let positions = y_range.flat_map(|y| x_range.clone().map(move |x| (x, y)));
262        for ((x, y), pixel) in positions.zip(samples.as_chunks::<4>().0.iter()) {
263            let index = self.byte_index(x, y);
264            let target = &mut self.pixels[index..index + 4];
265            changed |= target != pixel;
266            target.copy_from_slice(pixel);
267        }
268        Ok(changed)
269    }
270
271    fn border_region(
272        dim: i64,
273        dx: i32,
274        dy: i32,
275    ) -> Result<(std::ops::Range<i64>, std::ops::Range<i64>), DemError> {
276        if !(-1..=1).contains(&dx) || !(-1..=1).contains(&dy) || (dx == 0 && dy == 0) {
277            return Err(DemError::InvalidNeighbour { dx, dy });
278        }
279        let axis = |delta: i32| match delta {
280            -1 => -2..0,
281            1 => dim..dim + 2,
282            _ => 0..dim,
283        };
284        Ok((axis(dx), axis(dy)))
285    }
