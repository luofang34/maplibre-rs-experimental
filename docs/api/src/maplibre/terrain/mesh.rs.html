1//! Regular grid mesh with skirts, shared by every terrain tile.
2
3use crate::coords::{EXTENT_SINT, EXTENT_UINT};
4
5/// Grid cells per tile edge, as in GL JS `meshSize`.
6pub const TERRAIN_MESH_SIZE: u32 = 128;
7
8/// Vertex of the terrain grid in tile units with a skirt marker.
9#[repr(C)]
10#[derive(Clone, Copy, Debug, Eq, PartialEq, bytemuck_derive::Pod, bytemuck_derive::Zeroable)]
11pub struct TerrainVertex {
12    /// Horizontal tile coordinate.
13    pub x: i16,
14    /// Vertical tile coordinate.
15    pub y: i16,
16    /// One for skirt vertices, which the shader drops by the skirt length.
17    pub skirt: u16,
18    padding: u16,
19}
20
21impl TerrainVertex {
22    const fn new(x: i32, y: i32, skirt: u16) -> Self {
23        Self {
24            x: x as i16,
25            y: y as i16,
26            skirt,
27            padding: 0,
28        }
29    }
30}
31
32/// CPU-side terrain mesh.
33#[derive(Clone, Debug, Default, PartialEq, Eq)]
34pub struct TerrainMesh {
35    /// Grid and skirt vertices.
36    pub vertices: Vec<TerrainVertex>,
37    /// Triangle list indices.
38    pub indices: Vec<u32>,
39}
40
41/// Builds the grid with a skirt along each edge, following GL JS `getTerrainMesh`.
42///
43/// Skirts hang below the tile edges so neighbouring tiles at different zoom levels do not
44/// show cracks between their independently sampled edges. A zero resolution uses one cell.
45pub fn create_terrain_mesh(mesh_size: u32) -> TerrainMesh {
46    let n = mesh_size.max(1);
47    let mut mesh = TerrainMesh::default();
48    for y in 0..=n {
49        for x in 0..=n {
50            mesh.vertices.push(TerrainVertex::new(
51                grid_coordinate(x, n),
52                grid_coordinate(y, n),
53                0,
54            ));
55        }
56    }
57    let row = n + 1;
58    for y in 0..n {
59        for x in 0..n {
60            let index = y * row + x;
61            mesh.indices.extend([
62                index,
63                index + row,
64                index + row + 1,
65                index,
66                index + row + 1,
67                index + 1,
68            ]);
69        }
70    }
71    build_skirts(&mut mesh, n);
72    build_polar_caps(&mut mesh, n);
73    mesh
74}
75
76fn grid_coordinate(index: u32, cells: u32) -> i32 {
77    (u64::from(index) * u64::from(EXTENT_UINT) / u64::from(cells)) as i32
78}
79
80fn build_skirts(mesh: &mut TerrainMesh, n: u32) {
81    let row = n + 1;
82    let top = mesh.vertices.len() as u32;
83    let top_edge = 0;
84    let bottom = top + row;
85    let bottom_edge = row * n;
86    for x in 0..=n {
87        mesh.vertices
88            .push(TerrainVertex::new(grid_coordinate(x, n), 0, 1));
89    }
90    for x in 0..=n {
91        mesh.vertices
92            .push(TerrainVertex::new(grid_coordinate(x, n), EXTENT_SINT, 1));
93    }
94    for x in 0..n {
95        mesh.indices.extend([
96            bottom_edge + x,
97            bottom + x,
98            bottom + x + 1,
99            bottom_edge + x,
100            bottom + x + 1,
101            bottom_edge + x + 1,
102            top_edge + x,
103            top + x + 1,
104            top + x,
105            top_edge + x,
106            top_edge + x + 1,
107            top + x + 1,
108        ]);
109    }
110    let left = mesh.vertices.len() as u32;
111    let right = left + row * 2;
112    for x in [0, EXTENT_SINT] {
113        for y in 0..=n {
114            for skirt in [0, 1] {
115                mesh.vertices
116                    .push(TerrainVertex::new(x, grid_coordinate(y, n), skirt));
117            }
118        }
119    }
120    for y in (0..n * 2).step_by(2) {
121        mesh.indices.extend([
122            left + y,
123            left + y + 1,
124            left + y + 3,
125            left + y,
126            left + y + 3,
127            left + y + 2,
128            right + y,
129            right + y + 3,
130            right + y + 1,
131            right + y,
132            right + y + 2,
133            right + y + 3,
134        ]);
135    }
136}

138fn build_polar_caps(mesh: &mut TerrainMesh, n: u32) {
139    for (edge, marker) in [(0, i16::MIN), (n * (n + 1), i16::MAX)] {
140        let start = mesh.vertices.len() as u32;
141        for x in 0..=n {
142            mesh.vertices.push(TerrainVertex::new(
143                grid_coordinate(x, n),
144                i32::from(marker),
145                0,
146            ));
147        }
148        for x in 0..n {
149            // Pole vertices coincide only at a fully spherical projection.
150            for mut triangle in [
151                [edge + x, edge + x + 1, start + x],
152                [edge + x + 1, start + x + 1, start + x],
153            ] {
154                if marker == i16::MAX {
155                    triangle.swap(0, 1);
156                }
157                mesh.indices.extend(triangle);
158            }
159        }
160    }
161}
