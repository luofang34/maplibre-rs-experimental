1//! Resolves the `raster-dem` source a style's terrain refers to.
2
3use crate::{
4    io::source_type::{RasterSource, SourceType},
5    style::{
6        source::{Source, TileAddressingScheme},
7        Style,
8    },
9};
10
11/// Highest zoom the style spec allows for a `raster-dem` source without an explicit maximum.
12const DEFAULT_MAX_ZOOM: u8 = 22;
13
14/// Fetchable DEM source together with the properties the terrain pipeline needs.
15#[derive(Clone, Debug)]
16pub struct DemSource {
17    /// Name of the source in the style.
18    pub name: String,
19    /// Tile URL template resolved into a fetchable source.
20    pub source: SourceType,
21    /// Edge length of one DEM tile in pixels.
22    pub tile_size: u32,
23    /// Channel factors and base shift decoding pixels to metres.
24    pub unpack: [f64; 4],
25    /// Lowest zoom with tiles.
26    pub minzoom: u8,
27    /// Highest zoom with tiles.
28    pub maxzoom: u8,
29    /// Elevation multiplier from the terrain property.
30    pub exaggeration: f32,
31}
32
33/// Returns the DEM source behind the style's `terrain`, or `None` when terrain is off or the
34/// named source has no tile URLs yet.
35pub fn dem_source(style: &Style) -> Option<DemSource> {
36    let terrain = style.terrain.as_ref()?;
37    let Some(Source::RasterDem(dem)) = style.sources.get(&terrain.source) else {
38        tracing::warn!(source = %terrain.source, "terrain names a source that is not raster-dem");
39        return None;
40    };
41    let template = dem.tiles.as_ref()?.first()?;
42    Some(DemSource {
43        name: terrain.source.clone(),
44        source: SourceType::Raster(RasterSource::from_template(
45            template,
46            TileAddressingScheme::XYZ,
47        )),
48        tile_size: dem.tile_size,
49        unpack: dem.unpack_vector(),
50        minzoom: dem.minzoom.unwrap_or(0),
51        maxzoom: dem.maxzoom.unwrap_or(DEFAULT_MAX_ZOOM),
52        exaggeration: terrain.exaggeration,
53    })
54}
