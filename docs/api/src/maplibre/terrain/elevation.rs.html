13/// Elevation in metres, including exaggeration, at a world position of the current zoom.
14///
15/// Samples the DEM tile the position's view tile drapes from, falling back to loaded ancestors
16/// while the tile is in flight. Returns `None` when no DEM covers the position yet.
17pub fn elevation_at_world(
18    tiles: &Tiles,
19    dem: &DemSource,
20    zoom: Zoom,
21    position: WorldCoords,
22) -> Option<f64> {
23    let view_tile = position.into_world_tile(zoom.zoom_level(TILE_SIZE), zoom);
24    let mut coords = dem_tile_coords(view_tile, dem.minzoom, dem.maxzoom)?;
25    loop {
26        if let Some(DemTileComponent::Loaded(dem_tile)) = tiles.query::<&DemTileComponent>(coords) {
27            let (x, y) = tile_local(position, coords, zoom);
28            return Some(
29                dem_tile.tile.elevation_at_tile_coords(x, y) * f64::from(dem.exaggeration),
30            );
31        }
32        coords = coords.get_parent()?;
33    }
34}
35
36/// Position inside `coords` in `0..=EXTENT` tile units.
37fn tile_local(position: WorldCoords, coords: WorldTileCoords, zoom: Zoom) -> (f64, f64) {
38    let tile_pixels = TILE_SIZE * Zoom::from(coords.z).scale_delta(&zoom);
39    let x = (position.x / tile_pixels - f64::from(coords.x)) * EXTENT;
40    let y = (position.y / tile_pixels - f64::from(coords.y)) * EXTENT;
41    (x.clamp(0.0, EXTENT), y.clamp(0.0, EXTENT))
42}

44/// Whether the globe camera of `style` orbits the center's elevation.
45///
46/// Pure vertical perspective has no flat camera to fall back to, so its globe camera orbits
47/// the point the camera looks at, on the terrain or at the center altitude, as the flat
48/// camera does; the globe preset keeps GL JS's sea-level orbit.
49pub(crate) fn globe_orbits_center(style: &crate::style::Style) -> bool {
50    style.projection.as_ref().is_some_and(|projection| {
51        projection.projection_type == crate::projection::ProjectionType::VerticalPerspective
52    })
53}

55/// Decides what the camera orbits at the center every frame, terrain or not: whether the globe
56/// camera orbits the center's elevation, and, without terrain or with the center off the
57/// ground, rests the center at its altitude. A gesture holds it where it is.
58pub fn center_target_system(
59    MapContext {
60        style, view_state, ..
61    }: &mut MapContext,
62) -> SystemResult {
63    view_state.set_globe_orbits_center(globe_orbits_center(style));
64    view_state.set_center_over_terrain(style.terrain.is_some());
65    let projection = style
66        .projection
67        .as_ref()
68        .map_or_else(Default::default, |projection| {
69            projection.projection_type.clone()
70        });
71    view_state.enforce_navigation(&projection);
72    let off_ground = style.terrain.is_none() || !view_state.center_clamped_to_ground();
73    if off_ground && !view_state.center_elevation_frozen() {
74        view_state.set_center_elevation(view_state.center_altitude());
75    }
76    if style.terrain.is_none() {
77        view_state.set_min_elevation(0.0);
78    }
79    Ok(())
80}

82/// Lifts the camera's orbit point onto the terrain under the map center every frame, and
83/// records the lowest elevation of the tile under it for the far plane.
84pub fn center_elevation_system(
85    MapContext {
86        style,
87        view_state,
88        world,
89        ..
90    }: &mut MapContext,
91) -> SystemResult {
92    // Off the ground the center rests at its altitude, which center_target_system applies.
93    if style.terrain.is_none() {
94        return Ok(());
95    }
96    let Some(index) = world.resources.get::<TerrainCoverageIndex>() else {
97        return Ok(());
98    };
99    let zoom = view_state.zoom();
100    let world_size = TILE_SIZE * 2_f64.powf(zoom.value());
101    // A free-globe camera may look past the last row of tiles, where the cap closes the
102    // globe at sea level; the flat center stops at that row.
103    if let Some(pose) = view_state.pose_view() {
104        if pose.center.latitude.abs() > crate::projection::globe::scale::MERCATOR_LATITUDE_LIMIT {
105            let follows_ground =
106                view_state.center_clamped_to_ground() && !view_state.center_elevation_frozen();
107            if follows_ground {
108                view_state.set_center_elevation(0.0);
109            }
110            return Ok(());
111        }
112    }
113    let center = view_state.camera().position();
114    // GL JS lifts the centre by the terrain tile at the integer zoom, not by the tile drawn
115    // under it, which a coarser level may stand in for at a distance.
116    let tile_zoom = zoom.value().floor().clamp(0.0, f64::from(u8::MAX)) as u8;
117    let elevation = index.elevation_at_zoom(
118        &world.tiles,
119        center.x / world_size,
120        center.y / world_size,
121        tile_zoom,
122    );
123    let follows_ground =
124        view_state.center_clamped_to_ground() && !view_state.center_elevation_frozen();
125    if let Some(elevation) = elevation.filter(|_| follows_ground) {
126        view_state.set_center_elevation(elevation);
127    }
128    let center_tile = WorldCoords::at_ground(center.x, center.y)
129        .into_world_tile(zoom.zoom_level(TILE_SIZE), zoom);
130    if let Some(range) = index.tile_elevation_range(center_tile) {
131        view_state.set_min_elevation(range.min_meters);
132    }
133    Ok(())
134}
