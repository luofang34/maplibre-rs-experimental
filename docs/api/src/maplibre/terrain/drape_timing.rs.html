1//! What redrawing a drape costs on the GPU, so a moving view draws only what fits its frame.
2//!
3//! Zooming can invalidate a whole batch of drapes at once; redrawing all of them in one frame
4//! stalls that frame and every host waiting on it. The drape passes are timed together with
5//! timestamp queries where the device has them, and the cost of one drape is a running
6//! average of those measurements. While the camera moves, a frame draws as many drapes as its
7//! time budget holds; the rest keep their previous texture or show an ancestor's, and are
8//! drawn once the view settles.
9
10use std::{
11    sync::atomic::{AtomicU32, Ordering},
12    time::Duration,
13};
14
15use crate::render::gpu_timer::GpuTimer;
16
17/// What a drape is assumed to cost before one has been measured, or on a device without
18/// timestamps: a 2048² texture with its mip chain on a mobile GPU.
19pub const ASSUMED_DRAPE_COST: Duration = Duration::from_millis(2);
20
21/// Weight of the newest measurement in the running average.
22const SMOOTHING: f64 = 0.25;
23
24/// The timer around a frame's drape passes and how many drapes it timed.
25#[derive(Default)]
26pub struct DrapeTimerSlot {
27    pub(crate) timer: Option<GpuTimer>,
28    drapes: AtomicU32,
29}
30
31impl DrapeTimerSlot {
32    /// A slot timing drapes on `device`, which times nothing without timestamp queries.
33    pub fn new(device: &wgpu::Device) -> Self {
34        Self {
35            timer: GpuTimer::new(device),
36            drapes: AtomicU32::new(0),
37        }
38    }
39
40    /// Records how many drapes the timed passes draw.
41    pub(crate) fn timing(&self, drapes: usize) {
42        self.drapes
43            .store(u32::try_from(drapes).unwrap_or(u32::MAX), Ordering::Release);
44    }
45
46    /// The GPU time of a frame's drape passes and how many drapes they drew, once a
47    /// measurement is back.
48    pub(crate) fn take(&self, device: &wgpu::Device, period: f32) -> Option<(Duration, u32)> {
49        let spent = self.timer.as_ref()?.take(device, period)?;
50        let drapes = self.drapes.load(Ordering::Acquire);
51        (drapes > 0).then_some((spent, drapes))
52    }
53}
54
55/// The running average cost of one drape.
56#[derive(Clone, Copy, Debug, Default)]
57pub struct DrapeCost {
58    per_drape: Option<Duration>,
59    last: Option<(Duration, u32)>,
60}
61
62impl DrapeCost {
63    /// Adds a frame's measurement: `spent` on the GPU drawing `drapes` drapes.
64    pub fn measured_frame(&mut self, spent: Duration, drapes: u32) {
65        self.last = Some((spent, drapes));
66        if drapes > 0 {
67            self.measured(spent / drapes);
68        }
69    }
70
71    /// The last frame measured: its drape passes' GPU time and how many drapes they drew.
72    pub fn last_frame(&self) -> Option<(Duration, u32)> {
73        self.last
74    }
75
76    /// Adds a measurement of one drape's cost.
77    pub fn measured(&mut self, per_drape: Duration) {
78        self.per_drape = Some(match self.per_drape {
79            Some(average) => average.mul_f64(1.0 - SMOOTHING) + per_drape.mul_f64(SMOOTHING),
80            None => per_drape,
81        });
82    }
83
84    /// One drape's cost, assumed until measured.
85    pub fn per_drape(&self) -> Duration {
86        self.per_drape.unwrap_or(ASSUMED_DRAPE_COST)
87    }
88
89    /// How many drapes fit in `time`: at least one, so a moving view still refines, and at most
90    /// `limit`.
91    pub fn drapes_within(&self, time: Duration, limit: usize) -> usize {
92        let per_drape = self.per_drape().max(Duration::from_micros(1));
93        let fitting = (time.as_secs_f64() / per_drape.as_secs_f64()).floor() as usize;
94        fitting.clamp(1, limit.max(1))
95    }
96}
