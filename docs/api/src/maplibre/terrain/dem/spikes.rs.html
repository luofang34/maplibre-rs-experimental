1//! Removal of the narrow elevation spikes that some DEM mosaics carry along source seams.
2//!
3//! Resampling across a seam leaves runs of one or two samples hundreds of metres above both
4//! neighbours. Bare-earth terrain is not that steep on both flanks, so a sample that rises above
5//! the median of its row or column window by more than [`SPIKE_SLOPE`] ground sample distances
6//! takes that median. Coarse tiles, where one sample can hold a real isolated peak, have a
7//! threshold of kilometres and keep their peaks. Narrow pits are kept: a gorge one sample wide is
8//! real terrain, and the seams' shallow dips do not stand out against the horizon.
9
10use std::f64::consts::PI;
11
12use super::DemTile;
13use crate::coords::WorldTileCoords;
14
15/// Policy: how many ground sample distances a sample may rise above its window median before it
16/// counts as a spike. Seam ringing is one or two samples wide and stands hundreds of metres proud.
17/// For terrain to rise three sample distances above the median of its five-sample window, it must
18/// climb steeper than atan(3), about 72°, on both flanks. Bare-earth DEMs average the ground over
19/// each sample and hold no such slopes, so steep real relief stays untouched.
20const SPIKE_SLOPE: f64 = 3.0;
21/// Samples on each side of a sample in its row and column windows. Two lets a spike two samples
22/// wide stay a minority of its window.
23const WINDOW_RADIUS: i64 = 2;
24/// Encoded value range of the three 8-bit channels.
25const MAX_ENCODED: f64 = 16_777_215.0;
26
27/// Ground size in metres of one sample of a `dim`-sample DEM tile, at the tile's centre latitude.
28pub(crate) fn sample_metres(coords: WorldTileCoords, dim: u32, circumference_meters: f64) -> f64 {
29    let n = 2_f64.powi(i32::from(u8::from(coords.z)));
30    let latitude = (PI * (1.0 - 2.0 * (f64::from(coords.y) + 0.5) / n))
31        .sinh()
32        .atan();
33    circumference_meters * latitude.cos() / (n * f64::from(dim.max(1)))
34}

36/// Re-encodes elevations for the Mapbox and Terrarium layouts, whose channels weigh 256 times
37/// the next one. Other unpack vectors cannot be inverted channel by channel.
38struct Encoding {
39    step: f64,
40    base: f64,
41}
42
43impl Encoding {
44    fn new([red, green, blue, base]: [f64; 4]) -> Option<Self> {
45        let close = |a: f64, b: f64| (a - b).abs() <= 1e-9 * b.abs();
46        (blue > 0.0 && close(green, blue * 256.0) && close(red, blue * 65_536.0))
47            .then_some(Self { step: blue, base })
48    }
49
50    fn channels(&self, elevation: f64) -> [u8; 3] {
51        let value = ((elevation + self.base) / self.step)
52            .round()
53            .clamp(0.0, MAX_ENCODED) as u32;
54        [(value >> 16) as u8, (value >> 8) as u8, value as u8]
55    }
56}

58/// Median of up to `2 * WINDOW_RADIUS + 1` samples, kept on the stack.
59fn median(samples: impl Iterator<Item = f64>) -> f64 {
60    let mut window = [0.0; 2 * WINDOW_RADIUS as usize + 1];
61    let mut len = 0;
62    for (slot, sample) in window.iter_mut().zip(samples) {
63        *slot = sample;
64        len += 1;
65    }
66    let window = &mut window[..len];
67    window.sort_unstable_by(f64::total_cmp);
68    let middle = len / 2;
69    match len % 2 {
70        0 if middle > 0 => (window[middle - 1] + window[middle]) / 2.0,
71        _ => window.get(middle).copied().unwrap_or(f64::NAN),
72    }
73}
74
75/// The median of a window when it lies below `ceiling`. A median can only lie below it when at
76/// least half the window does, so counting first skips the sort for almost every sample: the filter
77/// runs on every sample of every decoded tile.
78fn median_below(samples: impl Iterator<Item = f64> + Clone, ceiling: f64) -> Option<f64> {
79    let (below, len) = samples
80        .clone()
81        .fold((0_usize, 0_usize), |(below, len), sample| {
82            (below + usize::from(sample < ceiling), len + 1)
83        });
84    (below >= len.div_ceil(2))
85        .then(|| median(samples))
86        .filter(|median| *median < ceiling)
87}

89impl DemTile {
90    /// Replaces the interior samples that are spikes for a ground sample distance of
91    /// `sample_metres` and returns how many changed. Borders are left to the caller.
92    pub(super) fn remove_spikes(&mut self, sample_metres: f64) -> usize {
93        let Some(encoding) = Encoding::new(self.unpack) else {
94            return 0;
95        };
96        if !(sample_metres.is_finite() && sample_metres > 0.0) {
97            return 0;
98        }
99        let threshold = SPIKE_SLOPE * sample_metres;
100        let dim = i64::from(self.dim);
101        let heights: Vec<f64> = (0..dim)
102            .flat_map(|y| (0..dim).map(move |x| (x, y)))
103            .map(|(x, y)| self.get(x, y))
104            .collect();
105        let at = |x: i64, y: i64| heights[(y * dim + x) as usize];
106        let window =
107            |centre: i64| (centre - WINDOW_RADIUS).max(0)..=(centre + WINDOW_RADIUS).min(dim - 1);
108        let mut replaced = 0;
109        for y in 0..dim {
110            for x in 0..dim {
111                let ceiling = at(x, y) - threshold;
112                let row = median_below(window(x).map(|i| at(i, y)), ceiling);
113                let column = median_below(window(y).map(|i| at(x, i)), ceiling);
114                if let Some(median) = row.into_iter().chain(column).reduce(f64::min) {
115                    let index = self.byte_index(x, y);
116                    self.pixels[index..index + 3].copy_from_slice(&encoding.channels(median));
117                    replaced += 1;
118                }
119            }
120        }
121        replaced
122    }
123}
