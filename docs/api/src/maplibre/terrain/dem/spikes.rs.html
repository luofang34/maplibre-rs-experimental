1//! Removal of the narrow elevation spikes that some DEM mosaics carry along source seams.
2//!
3//! Resampling across a seam leaves runs of one or two samples above both neighbours, from tens to
4//! hundreds of metres, down north–south columns of the tile. Two rules lower such a sample to the
5//! median of its window:
6//!
7//! - A peak rule for the tallest spikes: a sample that rises more than [`SPIKE_SLOPE`] ground
8//!   sample distances above its row or column median.
9//! - A seam rule for the lower rest of a seam: a sample that rises more than [`SEAM_FLOOR_SAMPLES`]
10//!   ground sample distances, and at least [`SEAM_FLOOR_METRES`], above its east–west median,
11//!   where such rises continue down the same column for at least [`SEAM_ROWS`] rows.
12//!
13//! Both rules only lower samples. Coarse tiles, where one sample can hold a real isolated peak,
14//! have thresholds of kilometres and keep their peaks. Narrow pits are kept: a gorge one sample
15//! wide is real terrain, and a north–south gorge has the same column signature as the seams'
16//! shallow ringing dips, which do not stand out against the horizon.
17

23/// Policy: how many ground sample distances a sample may rise above its window median before it
24/// counts as a spike. Seam ringing is one or two samples wide and stands hundreds of metres proud.
25/// For terrain to rise three sample distances above the median of its five-sample window, it must
26/// climb steeper than atan(3), about 72°, on both flanks. Bare-earth DEMs average the ground over
27/// each sample and hold no such slopes, so steep real relief stays untouched.
28const SPIKE_SLOPE: f64 = 3.0;
29/// Policy: the rise above the east–west median, in ground sample distances, from which a sample
30/// can belong to a seam: a 45° climb on both flanks. Lower rises are left to real terrain.
31const SEAM_FLOOR_SAMPLES: f64 = 1.0;
32/// Policy: the smallest rise in metres that can belong to a seam. Fine tiles have samples of a
33/// few metres, where steep real ridges reach the sample-based floor; the seams rise far above it.
34const SEAM_FLOOR_METRES: f64 = 20.0;
35/// Policy: the rows a seam's rise must run down one column, give or take [`SEAM_SLACK`]. A ridge
36/// one or two samples wide does not stay within three columns for sixteen rows unless it runs
37/// within 7° of due north, while the seams follow one column for a hundred rows or more.
38const SEAM_ROWS: usize = 16;
39/// Columns either side of a column that continue its seam: ringing wanders by one sample.
40const SEAM_SLACK: usize = 1;
41/// Samples on each side of a sample in its row and column windows. Two lets a spike two samples
42/// wide stay a minority of its window.
43const WINDOW_RADIUS: i64 = 2;
44/// Encoded value range of the three 8-bit channels.
45const MAX_ENCODED: f64 = 16_777_215.0;
46
47/// Ground size in metres of one sample of a `dim`-sample DEM tile, at the tile's centre latitude.
48pub(crate) fn sample_metres(coords: WorldTileCoords, dim: u32, circumference_meters: f64) -> f64 {
49    let n = 2_f64.powi(i32::from(u8::from(coords.z)));
50    let latitude = (PI * (1.0 - 2.0 * (f64::from(coords.y) + 0.5) / n))
51        .sinh()
52        .atan();
53    circumference_meters * latitude.cos() / (n * f64::from(dim.max(1)))
54}

56/// Re-encodes elevations for the Mapbox and Terrarium layouts, whose channels weigh 256 times
57/// the next one. Other unpack vectors cannot be inverted channel by channel.
58struct Encoding {
59    step: f64,
60    base: f64,
61}
62
63impl Encoding {
64    fn new([red, green, blue, base]: [f64; 4]) -> Option<Self> {
65        let close = |a: f64, b: f64| (a - b).abs() <= 1e-9 * b.abs();
66        (blue > 0.0 && close(green, blue * 256.0) && close(red, blue * 65_536.0))
67            .then_some(Self { step: blue, base })
68    }
69
70    fn channels(&self, elevation: f64) -> [u8; 3] {
71        let value = ((elevation + self.base) / self.step)
72            .round()
73            .clamp(0.0, MAX_ENCODED) as u32;
74        [(value >> 16) as u8, (value >> 8) as u8, value as u8]
75    }
76}

78/// Median of up to `2 * WINDOW_RADIUS + 1` samples, kept on the stack.
79fn median(samples: impl Iterator<Item = f64>) -> f64 {
80    let mut window = [0.0; 2 * WINDOW_RADIUS as usize + 1];
81    let mut len = 0;
82    for (slot, sample) in window.iter_mut().zip(samples) {
83        *slot = sample;
84        len += 1;
85    }
86    let window = &mut window[..len];
87    window.sort_unstable_by(f64::total_cmp);
88    let middle = len / 2;
89    match len % 2 {
90        0 if middle > 0 => (window[middle - 1] + window[middle]) / 2.0,
91        _ => window.get(middle).copied().unwrap_or(f64::NAN),
92    }
93}

95/// The median of a window when it lies below `ceiling`. A median can only lie below it when at
96/// least half the window does, so counting first skips the sort for almost every sample: the filter
97/// runs on every sample of every decoded tile.
98fn median_below(samples: impl Iterator<Item = f64> + Clone, ceiling: f64) -> Option<f64> {
99    let (below, len) = samples
100        .clone()
101        .fold((0_usize, 0_usize), |(below, len), sample| {
102            (below + usize::from(sample < ceiling), len + 1)
103        });
104    (below >= len.div_ceil(2))
105        .then(|| median(samples))
106        .filter(|median| *median < ceiling)
107}

109/// Decoded interior samples of a tile, row by row.
110struct Heights {
111    dim: usize,
112    samples: Vec<f64>,
113}
114
115impl Heights {
116    fn at(&self, x: usize, y: usize) -> f64 {
117        self.samples[y * self.dim + x]
118    }
119
120    fn window(&self, centre: usize) -> std::ops::RangeInclusive<usize> {
121        let radius = WINDOW_RADIUS as usize;
122        centre.saturating_sub(radius)..=(centre + radius).min(self.dim - 1)
123    }
124
125    /// The row median of `(x, y)` when the sample rises more than `rise` above it.
126    fn row_median_below(&self, x: usize, y: usize, rise: f64) -> Option<f64> {
127        median_below(self.window(x).map(|i| self.at(i, y)), self.at(x, y) - rise)
128    }
129
130    /// The column median of `(x, y)` when the sample rises more than `rise` above it.
131    fn column_median_below(&self, x: usize, y: usize, rise: f64) -> Option<f64> {
132        median_below(self.window(y).map(|i| self.at(x, i)), self.at(x, y) - rise)
133    }
134}

136/// For every sample, the east–west median it rises above by more than the seam floor, if any.
137fn seam_rises(heights: &Heights, floor: f64) -> Vec<Option<f64>> {
138    let dim = heights.dim;
139    (0..dim * dim)
140        .map(|index| heights.row_median_below(index % dim, index / dim, floor))
141        .collect()
142}
143
144/// Whether each risen sample lies on a run of risen rows at least [`SEAM_ROWS`] long within
145/// [`SEAM_SLACK`] columns of it.
146fn seam_columns(rises: &[Option<f64>], dim: usize) -> Vec<bool> {
147    let near = |x: usize, y: usize| {
148        (x.saturating_sub(SEAM_SLACK)..=(x + SEAM_SLACK).min(dim - 1))
149            .any(|column| rises[y * dim + column].is_some())
150    };
151    let mut seam = vec![false; dim * dim];
152    for x in 0..dim {
153        let mut start = 0;
154        for y in 0..=dim {
155            if y < dim && near(x, y) {
156                continue;
157            }
158            if y - start >= SEAM_ROWS {
159                for row in start..y {
160                    seam[row * dim + x] = rises[row * dim + x].is_some();
161                }
162            }
163            start = y + 1;
164        }
165    }
166    seam
167}

169impl DemTile {
170    /// Lowers the interior samples that are spikes for a ground sample distance of
171    /// `sample_metres` and returns how many changed. Borders are left to the caller.
172    pub(super) fn remove_spikes(&mut self, sample_metres: f64) -> usize {
173        let Some(encoding) = Encoding::new(self.unpack) else {
174            return 0;
175        };
176        if !(sample_metres.is_finite() && sample_metres > 0.0) || self.dim == 0 {
177            return 0;
178        }
179        let dim = self.dim as usize;
180        let heights = Heights {
181            dim,
182            samples: (0..dim as i64)
183                .flat_map(|y| (0..dim as i64).map(move |x| (x, y)))
184                .map(|(x, y)| self.get(x, y))
185                .collect(),
186        };
187        let peak_rise = SPIKE_SLOPE * sample_metres;
188        let seam_floor = (SEAM_FLOOR_SAMPLES * sample_metres).max(SEAM_FLOOR_METRES);
189        let rises = seam_rises(&heights, seam_floor);
190        let seams = seam_columns(&rises, dim);
191        let mut replaced = 0;
192        for y in 0..dim {
193            for x in 0..dim {
194                let peak = heights
195                    .row_median_below(x, y, peak_rise)
196                    .into_iter()
197                    .chain(heights.column_median_below(x, y, peak_rise))
198                    .reduce(f64::min);
199                let seam = rises[y * dim + x].filter(|_| seams[y * dim + x]);
200                if let Some(median) = peak.into_iter().chain(seam).reduce(f64::min) {
201                    let index = self.byte_index(x as i64, y as i64);
202                    self.pixels[index..index + 3].copy_from_slice(&encoding.channels(median));
203                    replaced += 1;
204                }
205            }
206        }
207        replaced
208    }
209}
