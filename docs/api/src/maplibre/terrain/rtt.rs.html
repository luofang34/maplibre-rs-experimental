1//! Transforms that place a source tile's geometry inside a drape target's texture.
2
3use cgmath::{Matrix4, SquareMatrix, Vector3};
4
5use crate::coords::{WorldTileCoords, EXTENT};
6
7/// Homogeneous scale keeping layer z-indices well inside the clip depth range when `w` is one.
8const CLIP_SCALE: f64 = 65536.0;
9/// Fraction of a raster tile's extent by which it overlaps its neighbours in a drape texture.
10pub(crate) const EDGE_OVERLAP: f64 = 0.01;
11
12/// Maps tile-local coordinates of `source` to clip space of the drape texture of `target`.
13///
14/// `source` must be `target` itself, an ancestor, or a descendant; the texture covers exactly
15/// `target`, with tile row zero at the top of the texture.
16pub fn drape_transform(target: WorldTileCoords, source: WorldTileCoords) -> Option<Matrix4<f64>> {
17    let source_to_target = source_to_target(target, source)?;
18    #[rustfmt::skip]
19    let ortho = Matrix4::new(
20        2.0 / EXTENT, 0.0, 0.0, 0.0,
21        0.0, -2.0 / EXTENT, 0.0, 0.0,
22        0.0, 0.0, 1.0, 0.0,
23        -1.0, 1.0, 0.0, 1.0,
24    );
25    #[rustfmt::skip]
26    let clip_scale = Matrix4::new(
27        CLIP_SCALE, 0.0, 0.0, 0.0,
28        0.0, CLIP_SCALE, 0.0, 0.0,
29        0.0, 0.0, CLIP_SCALE, 0.0,
30        0.0, 0.0, 0.0, CLIP_SCALE,
31    );
32    Some(clip_scale * ortho * source_to_target)
33}
34
35/// Maps `source` tile-local coordinates in `0..EXTENT` to `target` tile-local coordinates.
36fn source_to_target(target: WorldTileCoords, source: WorldTileCoords) -> Option<Matrix4<f64>> {
37    let target_zoom = i32::from(u8::from(target.z));
38    let source_zoom = i32::from(u8::from(source.z));
39    let delta = target_zoom - source_zoom;
40    if delta == 0 {
41        return (target.x == source.x && target.y == source.y).then(Matrix4::identity);
42    }
43    if delta > 0 {
44        // The target is a descendant: cut its square out of the source and blow it up.
45        let scale = 2_f64.powi(delta);
46        if target.x >> delta != source.x || target.y >> delta != source.y {
47            return None;
48        }
49        let origin_x = f64::from(target.x - (source.x << delta)) * EXTENT / scale;
50        let origin_y = f64::from(target.y - (source.y << delta)) * EXTENT / scale;
51        return Some(
52            Matrix4::from_scale(scale)
53                * Matrix4::from_translation(Vector3::new(-origin_x, -origin_y, 0.0)),
54        );
55    }
56    // The source is a descendant: shrink it into its quadrant of the target.
57    let delta = -delta;
58    let scale = 2_f64.powi(delta);
59    if source.x >> delta != target.x || source.y >> delta != target.y {
60        return None;
61    }
62    let origin_x = f64::from(source.x - (target.x << delta)) * EXTENT / scale;
63    let origin_y = f64::from(source.y - (target.y << delta)) * EXTENT / scale;
64    Some(
65        Matrix4::from_translation(Vector3::new(origin_x, origin_y, 0.0))
66            * Matrix4::from_scale(1.0 / scale),
67    )
68}
