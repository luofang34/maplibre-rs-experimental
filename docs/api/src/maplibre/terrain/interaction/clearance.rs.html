1//! Enforces camera eye and perspective near-plane clearance above 3D terrain.
2//!
3//! Sampling a single point at `(eye.x, eye.y)` with zero vertical margin allows the perspective
4//! near clip quad (`near_z = height / 50`) to slice triangular holes into steep `60°–75°` Alpine
5//! ridges (such as Nordkette above Innsbruck or the Matterhorn) and traps constant-center orbit
6//! cameras when `min_zoom` or `max_pitch` clamps a single-step lift. Here a 5-point footprint
7//! spanning the eye and the 4 horizontal corners of the near clip quad is checked against both
8//! the finest loaded DEM and the stitched rendered surface, with a positive clearance scaled to
9//! the near-plane world radius.
10

22/// Minimum vertical eye clearance above terrain even at street-level zooms.
23const MIN_TERRAIN_CLEARANCE_METERS: f64 = 12.0;
24/// Multiplier on the 3D world radius of the perspective near clip quad.
25const NEAR_PLANE_SAFETY_FACTOR: f64 = 0.85;
26/// Extra margin above the vertical reach of the lowest near-plane corner.
27const NEAR_CORNER_DROP_FACTOR: f64 = 1.15;
28/// Small cushion added when lifting so an immediate re-check is strictly clear without chatter.
29const LIFT_EPSILON_METERS: f64 = 0.05;
30/// Maximum fixed-point reconciliation passes across coupled `(zoom, pitch)` changes.
31const MAX_CLEARANCE_PASSES: usize = 6;
32
33/// Minimum vertical clearance in metres required between [`camera_ground_position`] and the
34/// highest terrain sample across the eye's near-clip footprint.
35pub fn terrain_eye_clearance_meters(view_state: &ViewState) -> f64 {
36    let (width, height) = view_state.viewport_size();
37    let safe_h = height.max(1.0);
38    let near_z_px = safe_h / 50.0;
39    let ppm = view_state.pixels_per_meter().max(1e-9);
40    let near_z_m = near_z_px / ppm;
41
42    let tan_half_y = (view_state.field_of_view().0 * 0.5).tan().abs();
43    let tan_half_x = tan_half_y * (width.max(1.0) / safe_h);
44    let quad_radius_m = near_z_m * tan_half_x.hypot(tan_half_y);
45    let near_plane_world_radius = near_z_m.hypot(quad_radius_m);
46
47    let pitch = view_state.camera().get_pitch().0.abs();
48    let near_corner_drop = near_z_m * pitch.cos() + quad_radius_m * pitch.sin();
49
50    MIN_TERRAIN_CLEARANCE_METERS
51        .max(near_plane_world_radius * NEAR_PLANE_SAFETY_FACTOR)
52        .max(near_corner_drop * NEAR_CORNER_DROP_FACTOR)
53}

55/// Samples the highest terrain elevation across the 5-point camera + near-plane quad footprint.
56pub(crate) fn footprint_max_ground(
57    view_state: &ViewState,
58    index: &TerrainCoverageIndex,
59    tiles: &Tiles,
60) -> Option<f64> {
61    if index.is_empty() {
62        return None;
63    }
64    let world_size = TILE_SIZE * 2_f64.powf(view_state.zoom().value());
65    if !world_size.is_finite() || world_size <= 0.0 {
66        return None;
67    }
68    let (camera, _) = camera_ground_position(view_state);
69    let (width, height) = view_state.viewport_size();
70    let safe_h = height.max(1.0);
71    let near_z_px = safe_h / 50.0;
72    let tan_half_y = (view_state.field_of_view().0 * 0.5).tan().abs();
73    let half_h_px = near_z_px * tan_half_y;
74    let half_w_px = half_h_px * (width.max(1.0) / safe_h);
75
76    let pitch = view_state.camera().get_pitch().0;
77    let bearing = view_state.camera().get_bearing().0;
78    // Horizontal forward unit vector from camera toward map center, and right unit vector.
79    let fwd = Vector2::new(bearing.sin(), -bearing.cos());
80    let right = Vector2::new(bearing.cos(), bearing.sin());
81
82    let near_fwd_bottom = near_z_px * pitch.sin() + half_h_px * pitch.cos();
83    let near_fwd_top = near_z_px * pitch.sin() - half_h_px * pitch.cos();
84
85    let points = [
86        camera,
87        camera + fwd * near_fwd_bottom - right * half_w_px,
88        camera + fwd * near_fwd_bottom + right * half_w_px,
89        camera + fwd * near_fwd_top - right * half_w_px,
90        camera + fwd * near_fwd_top + right * half_w_px,
91    ];
92
93    let mut max_ground: Option<f64> = None;
94    for pt in points {
95        let mx = pt.x / world_size;
96        let my = pt.y / world_size;
97        if let Some(h) = sample_best_ground(index, tiles, mx, my) {
98            max_ground = Some(max_ground.map_or(h, |prev| prev.max(h)));
99        }
100    }
101    max_ground
102}

104fn sample_best_ground(
105    index: &TerrainCoverageIndex,
106    tiles: &Tiles,
107    mercator_x: f64,
108    mercator_y: f64,
109) -> Option<f64> {
110    if !(0.0..1.0).contains(&mercator_y) || !mercator_x.is_finite() {
111        return None;
112    }
113    let finest = index.loaded_ground_cached(tiles, mercator_x, mercator_y, DemDetail::Finest);
114    let stitched = index.elevation_at(tiles, mercator_x, mercator_y);
115    match (finest, stitched) {
116        (Some(a), Some(b)) => Some(a.max(b)),
117        (Some(a), None) => Some(a),
118        (None, Some(b)) => Some(b),
119        (None, None) => None,
120    }
121}

123fn footprint_deficit(
124    view_state: &ViewState,
125    index: &TerrainCoverageIndex,
126    tiles: &Tiles,
127    extra_margin: f64,
128) -> Option<(f64, f64, f64)> {
129    let (_, altitude) = camera_ground_position(view_state);
130    let max_ground = footprint_max_ground(view_state, index, tiles)?;
131    let required = max_ground + terrain_eye_clearance_meters(view_state) + extra_margin;
132    Some((altitude, max_ground, required))
133}

135/// Lowers camera pitch along the constant-zoom orbit sphere when the current pitch places the
136/// camera or its near-clip quad below the loaded terrain, keeping `zoom` and `center` unchanged.
137/// Returns whether the pitch was adjusted.
138pub fn clamp_pitch_above_terrain_keeping_zoom(
139    style: &Style,
140    view_state: &mut ViewState,
141    world: &World,
142) -> bool {
143    if style.terrain.is_none() || globe::globe_pose(view_state).is_some() {
144        return false;
145    }
146    let Some(index) = world.resources.get::<TerrainCoverageIndex>() else {
147        return false;
148    };
149    let current_pitch = view_state.camera().get_pitch().0;
150    if current_pitch <= 1e-4 {
151        return false;
152    }
153    let is_clear = |view: &ViewState, extra: f64| -> Option<bool> {
154        let (altitude, _, required) = footprint_deficit(view, index, &world.tiles, extra)?;
155        Some(altitude >= required)
156    };
157    let Some(false) = is_clear(view_state, 0.0) else {
158        return false;
159    };
160    view_state.camera_mut().set_pitch(Rad(0.0));
161    if is_clear(view_state, LIFT_EPSILON_METERS) != Some(true) {
162        view_state.camera_mut().set_pitch(Rad(current_pitch));
163        return false;
164    }
165    let mut lo = 0.0_f64;
166    let mut hi = current_pitch;
167    for _ in 0..12 {
168        let mid = 0.5 * (lo + hi);
169        view_state.camera_mut().set_pitch(Rad(mid));
170        if is_clear(view_state, LIFT_EPSILON_METERS).unwrap_or(true) {
171            lo = mid;
172        } else {
173            hi = mid;
174        }
175    }
176    view_state.camera_mut().set_pitch(Rad(lo));
177    true
178}

180/// Lifts the camera above the terrain under its 5-point near-clip footprint by reconciling pitch
181/// and zoom while keeping the orbit center in place. Returns whether the camera moved.
182pub fn keep_camera_above_terrain(style: &Style, view_state: &mut ViewState, world: &World) -> bool {
183    if style.terrain.is_none() {
184        return false;
185    }
186    if globe::globe_pose(view_state).is_some() {
187        return globe::keep_globe_camera_above_terrain(view_state, world);
188    }
189    let Some(index) = world.resources.get::<TerrainCoverageIndex>() else {
190        return false;
191    };
192    let mut moved = false;
193    for _ in 0..MAX_CLEARANCE_PASSES {
194        let Some((altitude, max_ground, required)) =
195            footprint_deficit(view_state, index, &world.tiles, 0.0)
196        else {
197            break;
198        };
199        if altitude >= required {
200            break;
201        }
202        let prev_zoom = view_state.zoom().value();
203        let prev_pitch = view_state.camera().get_pitch().0;
204        let target_altitude = required + LIFT_EPSILON_METERS;
205
206        tracing::debug!(
207            altitude,
208            max_ground,
209            required,
210            zoom = prev_zoom,
211            pitch = prev_pitch.to_degrees(),
212            center_elevation = view_state.center_elevation(),
213            "camera inside terrain clearance envelope; lifting it"
214        );
215
216        lift_eye_at_constant_xy(view_state, target_altitude);
217
218        // If `min_zoom` or `max_pitch` prevented the constant-XY lift from clearing the ridge,
219        // tilt toward nadir and/or back zoom out along the view ray so the eye never stays trapped.
220        if footprint_deficit(view_state, index, &world.tiles, 0.0)
221            .is_some_and(|(alt, _, req)| alt < req)
222        {
223            if !clamp_pitch_above_terrain_keeping_zoom(style, view_state, world) {
224                back_off_zoom_and_pitch(view_state, index, &world.tiles);
225            }
226        }
227
228        let changed = (view_state.zoom().value() - prev_zoom).abs() > 1e-9
229            || (view_state.camera().get_pitch().0 - prev_pitch).abs() > 1e-9;
230        if !changed {
231            break;
232        }
233        moved = true;
234    }
235    moved
236}

238fn lift_eye_at_constant_xy(view_state: &mut ViewState, target_altitude: f64) {
239    let world_size = TILE_SIZE * 2_f64.powf(view_state.zoom().value());
240    let (camera, _) = camera_ground_position(view_state);
241    let center = view_state.camera().position().to_vec();
242    let center_elevation = view_state.center_elevation();
243    let circumference = circumference_at_latitude(
244        view_state.body(),
245        mercator_world_to_lat_lon(center.x, center.y, world_size).latitude,
246    );
247    let from = Vector3::new(
248        camera.x / world_size,
249        camera.y / world_size,
250        target_altitude / circumference,
251    );
252    let to = Vector3::new(
253        center.x / world_size,
254        center.y / world_size,
255        center_elevation / circumference,
256    );
257    let delta = to - from;
258    let distance_3d = delta.magnitude();
259    if distance_3d <= f64::EPSILON {
260        return;
261    }
262    let ground = delta.x.hypot(delta.y);
263    let zoom = (view_state.camera_to_center_distance() / distance_3d / TILE_SIZE).log2();
264    let mut pitch = (ground / distance_3d).clamp(-1.0, 1.0).acos();
265    pitch = if delta.z < 0.0 {
266        std::f64::consts::FRAC_PI_2 - pitch
267    } else {
268        std::f64::consts::FRAC_PI_2 + pitch
269    };
270    view_state.camera_mut().set_pitch(Rad(pitch));
271    view_state.zoom_to(Zoom::new(zoom));
272}

274fn back_off_zoom_and_pitch(
275    view_state: &mut ViewState,
276    index: &TerrainCoverageIndex,
277    tiles: &Tiles,
278) {
279    // Reduce pitch toward nadir first if pitch is steep, then step zoom out so orbit radius grows.
280    let reduced_pitch = (view_state.camera().get_pitch().0 * 0.5).min(0.6);
281    view_state.camera_mut().set_pitch(Rad(reduced_pitch));
282    let Some((_, _, required)) = footprint_deficit(view_state, index, tiles, LIFT_EPSILON_METERS)
283    else {
284        return;
285    };
286    let cos_p = view_state.camera().get_pitch().0.cos().max(0.25);
287    let needed_vertical_m = (required - view_state.center_elevation()).max(10.0);
288    let needed_dist_m = needed_vertical_m / cos_p;
289    let current_dist_m =
290        view_state.camera_to_center_distance() / view_state.pixels_per_meter().max(1e-9);
291    if needed_dist_m > current_dist_m && current_dist_m > 0.0 {
292        let zoom_shift = (needed_dist_m / current_dist_m).log2();
293        let target_zoom = view_state.zoom().value() - zoom_shift;
294        view_state.zoom_to(Zoom::new(target_zoom));
295    }
296    if footprint_deficit(view_state, index, tiles, 0.0).is_some_and(|(alt, _, req)| alt < req) {
297        view_state.camera_mut().set_pitch(Rad(0.0));
298    }
