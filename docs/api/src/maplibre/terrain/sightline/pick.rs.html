1//! Where a screen ray first meets the drawn globe terrain.
2
3use cgmath::Point2;
4
5use super::{lat_lon_to_mercator, DrawnTerrain};
6use crate::{
7    projection::globe::{camera::GlobeCameraState, ray_sphere_intersection},
8    terrain::interaction::TerrainHit,
9};
10
11/// What the ray through a pixel meets first.
12#[derive(Clone, Copy, Debug, PartialEq)]
13pub enum TerrainPick {
14    /// The ground of a terrain tile.
15    Ground(TerrainHit),
16    /// A polar cap, which closes the globe at sea level beyond the last row of tiles where no
17    /// DEM reaches.
18    PolarCap(TerrainHit),
19    /// Nothing: the ray passes every ground there is.
20    Sky,
21    /// Ground no DEM describes yet, which the ray passes below the body's highest ground,
22    /// before any drawn ground.
23    Unknown,
24}
25
26/// What the ray from `camera` through `pixel` meets first on the drawn terrain.
27///
28/// This is the ground under the pointer, not a point snapped to the horizon for a gesture:
29/// a ray into the sky is [`TerrainPick::Sky`].
30pub fn pick_globe_terrain(
31    camera: &GlobeCameraState,
32    terrain: DrawnTerrain<'_>,
33    pixel: Point2<f64>,
34) -> TerrainPick {
35    let Some(direction) = camera.ray_direction_from_pixel(pixel) else {
36        return TerrainPick::Sky;
37    };
38    let eye = camera.camera_position();
39    let ceiling = terrain.body.unit_radius_at(terrain.ceiling());
40    let Some(span) =
41        ray_sphere_intersection(eye, direction, ceiling).filter(|span| span.t_max > 0.0)
42    else {
43        return TerrainPick::Sky;
44    };
45    let crossing = terrain.follow(eye, direction, (span.t_min.max(0.0), span.t_max), |_| false);
46    match (crossing.ground, crossing.unknown) {
47        // Unknown ground is only ever recorded before the ground the line stops at.
48        (_, Some(_)) => TerrainPick::Unknown,
49        (Some(ground), None) => {
50            let hit = TerrainHit {
51                mercator: lat_lon_to_mercator(ground.location),
52                location: ground.location,
53                elevation: ground.elevation,
54            };
55            if ground.polar {
56                TerrainPick::PolarCap(hit)
57            } else {
58                TerrainPick::Ground(hit)
59            }
60        }
61        (None, None) => TerrainPick::Sky,
62    }
63}
