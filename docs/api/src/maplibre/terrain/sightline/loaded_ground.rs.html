1//! Drawn terrain built from DEM tiles a host holds, so lines of sight and the drawn ground can
2//! be asked without a map drawing a frame.
3//!
4//! The host names the terrain tiles a frame would draw and hands over the DEM tiles it has
5//! decoded; which DEM tile each drawn tile samples, one zoom coarser and at most the source's
6//! maxzoom, or the nearest loaded ancestor, is decided here as the map decides it.
7
8use std::collections::BTreeSet;
9
10use super::DrawnTerrain;
11use crate::{
12    coords::WorldTileCoords,
13    projection::body::Body,
14    tcs::tiles::Tiles,
15    terrain::{
16        coverage::TerrainCoverageIndex, dem::DemTile, request_system::dem_tile_coords,
17        source::DemSource, DemTileComponent, LoadedDem,
18    },
19};
20
21/// Why [`LoadedGround::new`] cannot build the ground.
22#[derive(Debug, thiserror::Error)]
23pub enum LoadedGroundError {
24    /// A tile's coordinates lie outside the tile pyramid at its zoom.
25    #[error("tile {coords:?} lies outside the tile pyramid")]
26    InvalidTile {
27        /// The tile's coordinates.
28        coords: WorldTileCoords,
29    },
30}
31
32/// Terrain tiles and the DEM behind them, as a frame would draw them.
33pub struct LoadedGround {
34    pub(crate) tiles: Tiles,
35    pub(crate) index: TerrainCoverageIndex,
36}
37
38impl LoadedGround {
39    /// The DEM tiles the terrain tiles `rendered` sample, in tile order: the tiles to decode
40    /// for [`new`](Self::new). A tile coarser than the source's minzoom samples none.
41    pub fn dem_tiles_for(
42        rendered: impl IntoIterator<Item = WorldTileCoords>,
43        dem: &DemSource,
44    ) -> Vec<WorldTileCoords> {
45        rendered
46            .into_iter()
47            .filter_map(|coords| dem_tile_coords(coords, dem.minzoom, dem.maxzoom))
48            .collect::<BTreeSet<_>>()
49            .into_iter()
50            .collect()
51    }
52
53    /// The terrain tiles `rendered` drawn from the decoded `dem_tiles` of `dem`, the source
54    /// [`dem_source`](crate::terrain::source::dem_source) finds in the style. A drawn tile
55    /// whose DEM tile, and every ancestor of it, is missing from `dem_tiles` is unknown
56    /// ground, as a frame draws it before its DEM arrives.
57    pub fn new(
58        rendered: impl IntoIterator<Item = WorldTileCoords>,
59        dem_tiles: impl IntoIterator<Item = (WorldTileCoords, DemTile)>,
60        dem: &DemSource,
61    ) -> Result<Self, LoadedGroundError> {
62        let mut tiles = Tiles::default();
63        for (coords, tile) in dem_tiles {
64            tiles
65                .spawn_mut(coords)
66                .ok_or(LoadedGroundError::InvalidTile { coords })?
67                .insert(DemTileComponent::Loaded(LoadedDem::new(tile)));
68        }
69        let rendered = rendered
70            .into_iter()
71            .map(|coords| {
72                coords
73                    .build_quad_key()
74                    .map(|_| coords)
75                    .ok_or(LoadedGroundError::InvalidTile { coords })
76            })
77            .collect::<Result<Vec<_>, _>>()?;
78        let index = TerrainCoverageIndex::build(rendered, &tiles, dem);
79        Ok(Self { tiles, index })
80    }
81
82    /// The ground on `body`, to follow lines of sight across and ask the drawn ground of.
83    pub fn terrain(&self, body: Body) -> DrawnTerrain<'_> {
84        DrawnTerrain {
85            index: &self.index,
86            tiles: &self.tiles,
87            body,
88        }
89    }
90
91    /// The index of the drawn tiles, which also answers the best loaded DEM.
92    pub fn index(&self) -> &TerrainCoverageIndex {
93        &self.index
94    }
95
96    /// The decoded DEM tiles, as [`TerrainCoverageIndex::loaded_ground_cached`] reads them.
97    pub fn tiles(&self) -> &Tiles {
98        &self.tiles
99    }
100}
