13/// How high a target is. Heights are drawn metres, exaggeration included, as the terrain and
14/// the camera's orbit target are; a caller with a height above mean sea level multiplies it
15/// by the terrain's exaggeration first if the target is to keep its height over the drawn
16/// ground.
17#[derive(Clone, Copy, Debug, PartialEq)]
18pub enum TargetAltitude {
19    /// At this height above the mean radius.
20    Drawn {
21        /// Height in drawn metres.
22        meters: f64,
23    },
24    /// On the drawn ground.
25    OnGround,
26    /// This far above the drawn ground; the offset itself is not exaggerated.
27    AboveGround {
28        /// Height above the ground in metres.
29        meters: f64,
30    },
31}
32
33/// Whether a target can be seen.
34#[derive(Clone, Copy, Debug, PartialEq, Eq)]
35pub enum Visibility {
36    /// In the view, with nothing known in the way.
37    Visible,
38    /// Behind the camera or outside its frustum.
39    OutsideFrustum,
40    /// The line of sight passes through the body below its lowest drawn ground.
41    BehindGlobe,
42    /// Drawn ground nearer than the target lies on the line of sight.
43    BehindTerrain,
44    /// The line of sight, or the ground a height is measured from, crosses ground no DEM
45    /// describes yet.
46    TerrainUnknown,
47}

49/// Where a target shows and whether it can be seen.
50#[derive(Clone, Copy, Debug, PartialEq)]
51pub struct TargetView {
52    /// Whether the target can be seen.
53    pub visibility: Visibility,
54    /// Viewport pixel, where the target lies in front of the camera.
55    pub pixel: Option<Point2<f64>>,
56    /// Depth the frame would store for the target, in the GPU's reversed-Z convention: one at
57    /// the near plane, zero at the far plane.
58    pub depth: Option<f64>,
59    /// The target's drawn height above the mean radius in metres, once known.
60    pub elevation: Option<f64>,
61}
62
63/// Where the target at `location` and `altitude` shows from `camera`, and whether it can be
64/// seen past the body and, where `terrain` is drawn, the ground.
65pub fn target_view(
66    camera: &GlobeCameraState,
67    terrain: Option<DrawnTerrain<'_>>,
68    location: LatLon,
69    altitude: TargetAltitude,
70) -> TargetView {
71    // Ground outside the rendered tiles is as unknown as ground whose DEM has not loaded.
72    let ground = || {
73        terrain.map_or(Some(0.0), |terrain| {
74            terrain.ground_at(location).ok().flatten()
75        })
76    };
77    let elevation = match altitude {
78        TargetAltitude::Drawn { meters } => Some(meters),
79        TargetAltitude::OnGround => ground(),
80        TargetAltitude::AboveGround { meters } => ground().map(|ground| ground + meters),
81    };
82    let unseen = |visibility| TargetView {
83        visibility,
84        pixel: None,
85        depth: None,
86        elevation,
87    };
88    let Some(height) = elevation else {
89        return unseen(Visibility::TerrainUnknown);
90    };
91    let body = camera.body();
92    let point = lat_lon_to_unit_sphere(location) * body.unit_radius_at(height);
93    let clip = camera.wgpu_view_projection() * point.extend(1.0);
94    if ![clip.x, clip.y, clip.z, clip.w]
95        .iter()
96        .all(|v| v.is_finite())
97        || clip.w <= 0.0
98    {
99        return unseen(Visibility::OutsideFrustum);
100    }
101    let ndc = clip.truncate() / clip.w;
102    let mut view = TargetView {
103        visibility: Visibility::Visible,
104        pixel: Some(camera.ndc_to_pixel(Point2::new(ndc.x, ndc.y))),
105        depth: Some(ndc.z),
106        elevation,
107    };
108    view.visibility = if ndc.x.abs() > 1.0 || ndc.y.abs() > 1.0 || !(0.0..=1.0).contains(&ndc.z) {
109        Visibility::OutsideFrustum
110    } else {
111        line_of_sight(camera, terrain, point)
112    };
113    view
114}

116/// Whether the body or the drawn ground hides `point` from the camera.
117fn line_of_sight(
118    camera: &GlobeCameraState,
119    terrain: Option<DrawnTerrain<'_>>,
120    point: Vector3<f64>,
121) -> Visibility {
122    let eye = camera.camera_position();
123    let to_target = point - eye;
124    let length = to_target.magnitude();
125    let direction = to_target / length;
126    // Below the lowest drawn ground the body is solid; above it only the terrain can hide.
127    let core = terrain.map_or(1.0, |terrain| {
128        terrain
129            .body
130            .unit_radius_at(terrain.index.min_elevation().min(0.0))
131    });
132    let core_hit = ray_sphere_intersection(eye, direction, core)
133        .filter(|hit| hit.t_min > 0.0 && hit.t_min < length * (1.0 - 1e-9));
134    if let Some(hit) = core_hit {
135        // Over a rendered tile without its DEM the ground may lie lower than any drawn.
136        let undescribed = terrain.is_some_and(|terrain| {
137            matches!(
138                terrain.probe(eye + direction * hit.t_min).tile,
139                Some((_, false))
140            )
141        });
142        return if undescribed {
143            Visibility::TerrainUnknown
144        } else {
145            Visibility::BehindGlobe
146        };
147    }
148    let Some(terrain) = terrain else {
149        return Visibility::Visible;
150    };
151    if terrain.index.is_empty() {
152        // Terrain is on but nothing is drawn yet: below the highest ground anything may stand.
153        let nearest = (-eye.dot(direction)).clamp(0.0, length);
154        let lowest = (eye + direction * nearest).magnitude();
155        return if lowest < terrain.body.unit_radius_at(terrain.ceiling()) {
156            Visibility::TerrainUnknown
157        } else {
158            Visibility::Visible
159        };
160    }
161    // The target's own ground, within one DEM sample of it, is where it stands, not in the way.
162    let own_ground = point.normalize();
163    let mercator =
164        lat_lon_to_mercator(crate::projection::globe::unit_sphere_to_lat_lon(own_ground));
165    let reach = terrain
166        .index
167        .rendered_tile_at(mercator.x, mercator.y)
168        .map_or(0.0, |tile| {
169            TileFootprint::of(tile).width / f64::from(terrain.index.dem_tile_size())
170        });
171    let crossing = terrain.follow(eye, direction, (0.0, length), |location| {
172        lat_lon_to_unit_sphere(location).dot(own_ground) >= reach.cos()
173    });
174    match (crossing.ground, crossing.unknown) {
175        (Some(_), _) => Visibility::BehindTerrain,
176        (None, Some(_)) => Visibility::TerrainUnknown,
177        (None, None) => Visibility::Visible,
178    }
179}
