1//! Replaces replicated DEM borders with the true edge samples of loaded neighbours.
2//!
3//! A tile decodes with its border copied from its own edge so interpolation is continuous
4//! straight away; once a neighbour loads, both tiles take each other's edge samples, as GL JS
5//! `backfillDEM` does, so the meshes meet without a step along the shared edge.
6
7use crate::{
8    coords::{WorldTileCoords, ZoomLevel},
9    tcs::tiles::Tiles,
10    terrain::DemTileComponent,
11};
12
13const NEIGHBOUR_OFFSETS: [(i32, i32); 8] = [
14    (-1, -1),
15    (0, -1),
16    (1, -1),
17    (-1, 0),
18    (1, 0),
19    (-1, 1),
20    (0, 1),
21    (1, 1),
22];
23
24/// Neighbour coordinates of a tile, wrapping across the antimeridian and skipping the poles.
25pub fn neighbours(coords: WorldTileCoords) -> Vec<(WorldTileCoords, (i32, i32))> {
26    let zoom = u8::from(coords.z);
27    let tiles_per_axis = 1_i64 << zoom;
28    NEIGHBOUR_OFFSETS
29        .iter()
30        .filter_map(|&(dx, dy)| {
31            let y = i64::from(coords.y) + i64::from(dy);
32            if y < 0 || y >= tiles_per_axis {
33                return None;
34            }
35            let x = (i64::from(coords.x) + i64::from(dx)).rem_euclid(tiles_per_axis);
36            let neighbour = WorldTileCoords {
37                x: x as i32,
38                y: y as i32,
39                z: ZoomLevel::new(zoom),
40            };
41            Some((neighbour, (dx, dy)))
42        })
43        .collect()
44}
45
46/// Fills the borders between a freshly loaded tile and every loaded neighbour, both ways.
47pub fn backfill_neighbours(tiles: &mut Tiles, coords: WorldTileCoords) {
48    for (neighbour, (dx, dy)) in neighbours(coords) {
49        fill_border(tiles, coords, neighbour, dx, dy);
50        fill_border(tiles, neighbour, coords, -dx, -dy);
51    }
52}
53
54/// Copies `source`'s edge into `target`'s border, where `source` sits at `(dx, dy)` from it.
55fn fill_border(
56    tiles: &mut Tiles,
57    target: WorldTileCoords,
58    source: WorldTileCoords,
59    dx: i32,
60    dy: i32,
61) {
62    let Some(DemTileComponent::Loaded(source_dem)) = tiles.query::<&DemTileComponent>(source)
63    else {
64        return;
65    };
66    let samples = match source_dem.tile.edge_samples(dx, dy) {
67        Ok(samples) => samples,
68        Err(error) => {
69            tracing::warn!(%target, %source, %error, "cannot read DEM neighbour edge");
70            return;
71        }
72    };
73    let source_dim = source_dem.tile.dim();
74    let source_unpack = source_dem.tile.unpack();
75    let Some(DemTileComponent::Loaded(target_dem)) =
76        tiles.query_mut::<&mut DemTileComponent>(target)
77    else {
78        return;
79    };
80    let result = target_dem
81        .tile
82        .validate_neighbour(source_dim, source_unpack)
83        .and_then(|()| target_dem.tile.fill_border(dx, dy, &samples));
84    match result {
85        Ok(true) => target_dem.revision = target_dem.revision.wrapping_add(1),
86        Ok(false) => {}
87        Err(error) => tracing::warn!(%target, %source, %error, "cannot backfill DEM border"),
88    }
89}
