1//! Uploads decoded DEM tiles to the shared GPU elevation pool and binds them for terrain meshes.
2
3use crate::{
4    context::MapContext,
5    raster::RasterSourceId,
6    render::{
7        eventually::{Eventually, Eventually::Initialized},
8        memory_budget::{FrameUploads, UploadKind},
9        Renderer,
10    },
11    tcs::system::SystemResult,
12    terrain::{
13        gpu_dem_pool::GpuDemPool, resources::TerrainResources, source::dem_source, DemTileComponent,
14    },
15};
16
17pub fn upload_system(
18    MapContext {
19        style,
20        world,
21        renderer: Renderer { device, queue, .. },
22        ..
23    }: &mut MapContext,
24) -> SystemResult {
25    if crate::render::eye_covering::EyeInFrame::reuses_content(world) {
26        return Ok(());
27    }
28    let Some(dem_spec) = dem_source(style) else {
29        return Ok(());
30    };
31    let source_id = RasterSourceId::from(dem_spec.name.as_str());
32    world.resources.get_or_init_mut::<FrameUploads>();
33    world.resources.get_or_init_mut::<GpuDemPool>();
34    let Some(Initialized(terrain_resources)) =
35        world.resources.get::<Eventually<TerrainResources>>()
36    else {
37        return Ok(());
38    };
39    let to_upload: Vec<_> = world
40        .tiles
41        .pyramid_of::<DemTileComponent>()
42        .coords_coarse_to_fine()
43        .filter_map(|coords| {
44            let Some(DemTileComponent::Loaded(dem)) =
45                world.tiles.query::<&DemTileComponent>(coords)
46            else {
47                return None;
48            };
49            terrain_resources
50                .needs_dem_upload(coords, dem)
51                .then(|| (coords, dem.clone()))
52        })
53        .collect();
54    if to_upload.is_empty() {
55        return Ok(());
56    }
57
58    let mut deferred = false;
59    for (coords, dem) in to_upload {
60        let needs_gpu_write = world
61            .resources
62            .get::<GpuDemPool>()
63            .is_none_or(|pool| pool.needs_upload(&source_id, coords, &dem));
64        if needs_gpu_write {
65            let Some(budget) = world.resources.get_mut::<FrameUploads>() else {
66                break;
67            };
68            if !budget.take(UploadKind::Image, dem.tile.pixels().len()) {
69                deferred = true;
70                break;
71            }
72        }
73        let shared = {
74            let pool = world.resources.get_or_init_mut::<GpuDemPool>();
75            let entry = if needs_gpu_write {
76                pool.upload(device, queue, &source_id, coords, &dem)
77            } else if let Some(existing) = pool.get(&source_id, coords) {
78                existing
79            } else {
80                pool.upload(device, queue, &source_id, coords, &dem)
81            };
82            entry
83                .texture
84                .clone_handle()
85                .map(|texture| (texture, entry.revision))
86        };
87        if let Some((Initialized(terrain_resources), (texture, revision))) = world
88            .resources
89            .get_mut::<Eventually<TerrainResources>>()
90            .zip(shared)
91        {
92            terrain_resources.bind_shared_dem(coords, texture, revision, dem.revision_key());
93        } else if let Some(Initialized(terrain_resources)) =
94            world.resources.get_mut::<Eventually<TerrainResources>>()
95        {
96            terrain_resources.upload_loaded_dem(device, queue, coords, &dem);
97        }
98    }
99    if deferred {
100        crate::render::frame_signals::keep_animating(world);
101    }
102    Ok(())
103}
