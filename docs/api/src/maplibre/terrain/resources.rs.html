29/// Edge length in pixels of one drape texture; twice the tile size, as GL JS `qualityFactor`.
30pub const DRAPE_SIZE: u32 = 2048;
31/// Smallest drape edge accepted; below it the ground is too coarse to read.
32pub const MIN_DRAPE_SIZE: u32 = 256;
33/// Byte stride between per-tile uniform blocks, the WebGPU dynamic offset alignment.
34pub const UNIFORM_STRIDE: u64 = 512;
35// A block that outgrows its stride would overwrite the next tile's; the stride is a
36// multiple of the 256-byte offset alignment every backend accepts.
37const _: () = assert!(std::mem::size_of::<TerrainTileUniforms>() as u64 <= UNIFORM_STRIDE);
38const _: () = assert!(std::mem::size_of::<TerrainTileUniforms>() == 512);
39/// Largest number of terrain tiles drawn in one frame.
40const UNIFORM_CAPACITY: u64 = 1024;
41
42/// Per-tile inputs of the terrain shaders, one block per drawn tile.
43#[repr(C)]
44#[derive(Clone, Copy, Debug, Pod, Zeroable)]
45pub struct TerrainTileUniforms {
46    /// GPU view projection times the tile transform.
47    pub transform: [[f32; 4]; 4],
48    /// Tile-centred globe projection, with the large translation resolved in f64.
49    pub globe_transform: [[f32; 4]; 4],
50    /// Centre latitude sine/cosine, radians per tile unit, and radius; zero radius disables it.
51    pub globe_origin: [f32; 4],
52    /// Maps tile coordinates to unit coordinates of the sampled DEM tile.
53    pub dem_matrix: [[f32; 4]; 4],
54    /// Maps the tile's unit coordinates into the drape texture drawn for it, which is an
55    /// ancestor's while the tile's own is not drawn yet.
56    pub drape_matrix: [[f32; 4]; 4],
57    /// Mercator offset and scale of the tile for the globe path.
58    pub tile_mercator_coords: [f32; 4],
59    /// Channel factors and base shift decoding DEM pixels to metres.
60    pub dem_unpack: [f32; 4],
61    /// Samples per DEM tile edge without the border.
62    pub dem_dim: f32,
63    /// Elevation multiplier.
64    pub exaggeration: f32,
65    /// Distance in metres the skirt vertices drop below the surface.
66    pub skirt_length: f32,
67    /// Strength of directional relief lighting, zero for unlit map styles.
68    pub relief_strength: f32,
69    /// Premultiplied fog colour.
70    pub fog_color: [f32; 4],
71    /// Premultiplied horizon colour the fog blends into.
72    pub horizon_color: [f32; 4],
73    /// Fog depth range (metres for external eyes, otherwise pixels), then blend thresholds.
74    pub fog_range: [f32; 4],
75    /// Fog opacity, globe flag, fallback surface flag, and radial fog flag.
76    pub fog_opacity: [f32; 4],
77    /// Background color while a tile has no valid drape texture.
78    pub surface_color: [f32; 4],
79    /// Tile origin relative to the eye in metres, and metres per tile coordinate.
80    pub fog_position: [f32; 4],
81    /// Direction towards the sun in globe world axes, with w 1 while a host has set one; all
82    /// zero leaves every tile lit as day.
83    pub daylight: [f32; 4],
84    /// Sines of the solar elevations ending night and starting day, then the night brightness.
85    pub twilight: [f32; 4],
86    /// Direction towards the Moon in globe world axes, with w the light it adds overhead;
87    /// all zero without a host Moon.
88    pub moonlight: [f32; 4],
89    /// Procedural DEM relief controls: `[enabled, hypsometric, slope_shading, reference_alt_m]`.
90    pub procedural_relief: [f32; 4],
91    /// Procedural DEM grid & TAWS controls: `[grid_spacing_m, grid_opacity, taws_enabled, tile_span_m]`.
92    pub procedural_grid: [f32; 4],
93    /// Index into the per-frame edge storage buffer in `edge_index[0]`.
94    pub edge_index: [f32; 4],
95}

187    /// Creates terrain resources configured for `drape_mode`, omitting `MipmapGenerator` and
188    /// drape allocations when [`TerrainDrapeMode::ProceduralRelief`] is active.
189    pub fn new_with_mode(
190        device: &wgpu::Device,
191        queue: &crate::render::upload_queue::UploadQueue,
192        pipeline: wgpu::RenderPipeline,
193        color_format: wgpu::TextureFormat,
194        depth_format: wgpu::TextureFormat,
195        msaa: Msaa,
196        drape_mode: TerrainDrapeMode,
197    ) -> Self {
198        let max_dim = device.limits().max_texture_dimension_2d.max(MIN_DRAPE_SIZE);
199        let (drape_size, drape_mode, mipmaps) = match drape_mode {
200            TerrainDrapeMode::Draped { edge_px } => {
201                let clamped = edge_px.clamp(MIN_DRAPE_SIZE, max_dim);
202                (
203                    clamped,
204                    TerrainDrapeMode::Draped { edge_px: clamped },
205                    Some(MipmapGenerator::new(device, color_format)),
206                )
207            }
208            TerrainDrapeMode::ProceduralRelief(style) => {
209                (0, TerrainDrapeMode::ProceduralRelief(style), None)
210            }
211        };
212        let mesh = create_terrain_mesh(TERRAIN_MESH_SIZE);
213        let vertex_buffer = queue.create_buffer_init(
214            device,
215            &wgpu::util::BufferInitDescriptor {
216                label: Some("terrain mesh vertices"),
217                contents: bytemuck::cast_slice(&mesh.vertices),
218                usage: wgpu::BufferUsages::VERTEX,
219            },
220        );
221        let index_buffer = queue.create_buffer_init(
222            device,
223            &wgpu::util::BufferInitDescriptor {
224                label: Some("terrain mesh indices"),
225                contents: bytemuck::cast_slice(&mesh.indices),
226                usage: wgpu::BufferUsages::INDEX,
227            },
228        );
229        let uniform_buffer = device.create_buffer(&wgpu::BufferDescriptor {
230            label: Some("terrain tile uniforms"),
231            size: UNIFORM_STRIDE * UNIFORM_CAPACITY,
232            usage: wgpu::BufferUsages::UNIFORM | wgpu::BufferUsages::COPY_DST,
233            mapped_at_creation: false,
234        });
235        let edge_buffer = device.create_buffer(&wgpu::BufferDescriptor {
236            label: Some("terrain edge storage"),
237            size: (std::mem::size_of::<super::queue_system::edges::EdgeHeights>() as u64)
238                * UNIFORM_CAPACITY,
239            usage: wgpu::BufferUsages::STORAGE | wgpu::BufferUsages::COPY_DST,
240            mapped_at_creation: false,
241        });
242        let sampler = device.create_sampler(&wgpu::SamplerDescriptor {
243            address_mode_u: wgpu::AddressMode::ClampToEdge,
244            address_mode_v: wgpu::AddressMode::ClampToEdge,
245            address_mode_w: wgpu::AddressMode::ClampToEdge,
246            mag_filter: wgpu::FilterMode::Linear,
247            min_filter: wgpu::FilterMode::Linear,
248            mipmap_filter: wgpu::MipmapFilterMode::Linear,
249            // Drapes are seen at grazing angles at the horizon; anisotropy keeps their
250            // detail along the view direction while the mip levels stop the shimmer.
251            anisotropy_clamp: 16,
252            ..Default::default()
253        });
254        // A fresh texture reads as zero, which decodes to sea level with a zero unpack vector.
255        let empty_dem = Texture::new(
256            Some("empty DEM"),
257            device,
258            wgpu::TextureFormat::Rgba8Unorm,
259            1,
260            1,
261            Msaa { samples: 1 },
262            wgpu::TextureUsages::TEXTURE_BINDING,
263        );
264        Self {
265            pipeline,
266            sampler,
267            mipmaps,
268            vertex_buffer,
269            index_buffer,
270            index_count: mesh.indices.len() as u32,
271            uniform_buffer,
272            dem_textures: HashMap::new(),
273            empty_dem,
274            drapes: DrapeCache::default(),
275            drape_scratch: None,
276            draws: Vec::new(),
277            bind_groups: HashMap::new(),
278            edge_buffer,
279            msaa,
280            color_format,
281            depth_format,
282            drape_size,
283            drape_mode,
284        }
285    }

292    /// Updates the active [`TerrainDrapeMode`], immediately releasing `DrapeScratch` and
293    /// `DrapeCache` textures when switching to [`TerrainDrapeMode::ProceduralRelief`] or resizing.
294    pub fn set_drape_mode(&mut self, device: &wgpu::Device, mode: TerrainDrapeMode) {
295        let max_dim = device.limits().max_texture_dimension_2d.max(MIN_DRAPE_SIZE);
296        match mode {
297            TerrainDrapeMode::ProceduralRelief(style) => {
298                self.drape_mode = TerrainDrapeMode::ProceduralRelief(style);
299                self.drape_size = 0;
300                self.drape_scratch = None;
301                self.drapes = DrapeCache::default();
302            }
303            TerrainDrapeMode::Draped { edge_px } => {
304                let clamped = edge_px.clamp(MIN_DRAPE_SIZE, max_dim);
305                if self.drape_size != clamped {
306                    self.drape_size = clamped;
307                    self.drape_scratch = None;
308                    self.drapes = DrapeCache::default();
309                }
310                self.drape_mode = TerrainDrapeMode::Draped { edge_px: clamped };
311                if self.mipmaps.is_none() {
312                    self.mipmaps = Some(MipmapGenerator::new(device, self.color_format));
313                }
314            }
315        }
316    }

348    /// Bytes of drape textures, held and free, and of DEM textures.
349    pub fn texture_bytes(&self) -> (usize, usize) {
350        let (held, free) = self.drape_counts();
351        // Four bytes a texel, and a mip chain adds a third.
352        let drape = (self.drape_size as usize).pow(2) * 4 * 4 / 3;
353        let dem = self
354            .dem_textures
355            .values()
356            .map(|dem| (dem.texture.size.width * dem.texture.size.height * 4) as usize)
357            .sum();
358        ((held + free) * drape, dem)
359    }

361    /// Creates the scratch attachments used by every drape pass.
362    pub fn ensure_scratch(&mut self, device: &wgpu::Device) {
363        if self.drape_scratch.is_some()
364            || self.drape_mode.is_procedural_relief()
365            || self.drape_size == 0
366        {
367            return;
368        }
369        let color = self.msaa.is_multisampling().then(|| {
370            Texture::new(
371                Some("drape multisampled color"),
372                device,
373                self.color_format,
374                self.drape_size,
375                self.drape_size,
376                self.msaa,
377                wgpu::TextureUsages::RENDER_ATTACHMENT,
378            )
379        });
380        let depth_stencil = Texture::new(
381            Some("drape depth stencil"),
382            device,
383            self.depth_format,
384            self.drape_size,
385            self.drape_size,
386            self.msaa,
387            wgpu::TextureUsages::RENDER_ATTACHMENT,
388        );
389        self.drape_scratch = Some(DrapeScratch {
390            color,
391            depth_stencil,
392        });
393    }

400    /// Writes the per-tile uniform blocks and returns how many fit.
401    pub fn write_uniforms(
402        &self,
403        queue: &crate::render::upload_queue::UploadQueue,
404        uniforms: &[TerrainTileUniforms],
405    ) -> usize {
406        let count = uniforms.len().min(UNIFORM_CAPACITY as usize);
407        if count < uniforms.len() {
408            tracing::warn!(
409                requested = uniforms.len(),
410                capacity = UNIFORM_CAPACITY,
411                "more terrain tiles than uniform blocks; distant tiles are skipped"
412            );
413        }
414        let mut bytes = vec![0_u8; count * UNIFORM_STRIDE as usize];
415        for (index, block) in uniforms.iter().take(count).enumerate() {
416            let start = index * UNIFORM_STRIDE as usize;
417            let raw = bytemuck::bytes_of(block);
418            bytes[start..start + raw.len()].copy_from_slice(raw);
419        }
420        queue.write_buffer(&self.uniform_buffer, 0, &bytes);
421        count
422    }

424    /// Writes the per-tile edge height blocks.
425    pub(crate) fn write_edges(
426        &self,
427        queue: &crate::render::upload_queue::UploadQueue,
428        edges: &[super::queue_system::edges::EdgeHeights],
429    ) {
430        let count = edges.len().min(UNIFORM_CAPACITY as usize);
431        if count == 0 {
432            return;
433        }
434        let bytes = bytemuck::cast_slice(&edges[..count]);
435        queue.write_buffer(&self.edge_buffer, 0, bytes);
436    }

438    /// Replaces this frame's terrain draws.
439    pub fn set_draws(&mut self, draws: Vec<TerrainDraw>) {
440        self.draws = draws;
441        // A bind group keeps its textures alive; only the ones this frame draws with stay.
442        #[allow(
443            clippy::mutable_key_type,
444            reason = "wgpu hashes immutable resource identity, independent of internal synchronization"
445        )]
446        let used: HashSet<wgpu::BindGroup> = self
447            .draws
448            .iter()
449            .map(|draw| draw.bind_group.as_ref().clone())
450            .collect();
451        self.bind_groups
452            .retain(|_, group| used.contains(group.as_ref()));
453    }
