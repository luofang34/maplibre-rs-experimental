29/// Edge length in pixels of one drape texture; twice the tile size, as GL JS `qualityFactor`.
30pub const DRAPE_SIZE: u32 = 2048;
31/// Smallest drape edge accepted; below it the ground is too coarse to read.
32pub const MIN_DRAPE_SIZE: u32 = 256;
33/// Byte stride between per-tile uniform blocks, the WebGPU dynamic offset alignment.
34pub const UNIFORM_STRIDE: u64 = 2560;
35// A block that outgrows its stride would overwrite the next tile's; the stride is a
36// multiple of the 256-byte offset alignment every backend accepts.
37const _: () = assert!(std::mem::size_of::<TerrainTileUniforms>() as u64 <= UNIFORM_STRIDE);
38/// Largest number of terrain tiles drawn in one frame.
39const UNIFORM_CAPACITY: u64 = 1024;
40
41/// Per-tile inputs of the terrain shaders, one block per drawn tile.
42#[repr(C)]
43#[derive(Clone, Copy, Debug, Pod, Zeroable)]
44pub struct TerrainTileUniforms {
45    /// GPU view projection times the tile transform.
46    pub transform: [[f32; 4]; 4],
47    /// Tile-centred globe projection, with the large translation resolved in f64.
48    pub globe_transform: [[f32; 4]; 4],
49    /// Centre latitude sine/cosine, radians per tile unit, and radius; zero radius disables it.
50    pub globe_origin: [f32; 4],
51    /// Maps tile coordinates to unit coordinates of the sampled DEM tile.
52    pub dem_matrix: [[f32; 4]; 4],
53    /// Maps the tile's unit coordinates into the drape texture drawn for it, which is an
54    /// ancestor's while the tile's own is not drawn yet.
55    pub drape_matrix: [[f32; 4]; 4],
56    /// Mercator offset and scale of the tile for the globe path.
57    pub tile_mercator_coords: [f32; 4],
58    /// Channel factors and base shift decoding DEM pixels to metres.
59    pub dem_unpack: [f32; 4],
60    /// Samples per DEM tile edge without the border.
61    pub dem_dim: f32,
62    /// Elevation multiplier.
63    pub exaggeration: f32,
64    /// Distance in metres the skirt vertices drop below the surface.
65    pub skirt_length: f32,
66    /// Strength of directional relief lighting, zero for unlit map styles.
67    pub relief_strength: f32,
68    /// Premultiplied fog colour.
69    pub fog_color: [f32; 4],
70    /// Premultiplied horizon colour the fog blends into.
71    pub horizon_color: [f32; 4],
72    /// Fog depth range (metres for external eyes, otherwise pixels), then blend thresholds.
73    pub fog_range: [f32; 4],
74    /// Fog opacity, globe flag, fallback surface flag, and radial fog flag.
75    pub fog_opacity: [f32; 4],
76    /// Background color while a tile has no valid drape texture.
77    pub surface_color: [f32; 4],
78    /// Tile origin relative to the eye in metres, and metres per tile coordinate.
79    pub fog_position: [f32; 4],
80    pub(crate) edge_heights: super::queue_system::edges::EdgeHeights,
81}

192    /// Creates the shared mesh, sampler and uniform storage for an initialized pipeline.
193    pub fn new(
194        device: &wgpu::Device,
195        queue: &crate::render::upload_queue::UploadQueue,
196        pipeline: wgpu::RenderPipeline,
197        color_format: wgpu::TextureFormat,
198        depth_format: wgpu::TextureFormat,
199        msaa: Msaa,
200        drape_size: u32,
201    ) -> Self {
202        let drape_size = drape_size.clamp(
203            MIN_DRAPE_SIZE,
204            device.limits().max_texture_dimension_2d.max(MIN_DRAPE_SIZE),
205        );
206        let mesh = create_terrain_mesh(TERRAIN_MESH_SIZE);
207        let vertex_buffer = queue.create_buffer_init(
208            device,
209            &wgpu::util::BufferInitDescriptor {
210                label: Some("terrain mesh vertices"),
211                contents: bytemuck::cast_slice(&mesh.vertices),
212                usage: wgpu::BufferUsages::VERTEX,
213            },
214        );
215        let index_buffer = queue.create_buffer_init(
216            device,
217            &wgpu::util::BufferInitDescriptor {
218                label: Some("terrain mesh indices"),
219                contents: bytemuck::cast_slice(&mesh.indices),
220                usage: wgpu::BufferUsages::INDEX,
221            },
222        );
223        let uniform_buffer = device.create_buffer(&wgpu::BufferDescriptor {
224            label: Some("terrain tile uniforms"),
225            size: UNIFORM_STRIDE * UNIFORM_CAPACITY,
226            usage: wgpu::BufferUsages::UNIFORM | wgpu::BufferUsages::COPY_DST,
227            mapped_at_creation: false,
228        });
229        let sampler = device.create_sampler(&wgpu::SamplerDescriptor {
230            address_mode_u: wgpu::AddressMode::ClampToEdge,
231            address_mode_v: wgpu::AddressMode::ClampToEdge,
232            address_mode_w: wgpu::AddressMode::ClampToEdge,
233            mag_filter: wgpu::FilterMode::Linear,
234            min_filter: wgpu::FilterMode::Linear,
235            mipmap_filter: wgpu::MipmapFilterMode::Linear,
236            // Drapes are seen at grazing angles at the horizon; anisotropy keeps their
237            // detail along the view direction while the mip levels stop the shimmer.
238            anisotropy_clamp: 16,
239            ..Default::default()
240        });
241        let mipmaps = MipmapGenerator::new(device, color_format);
242        // A fresh texture reads as zero, which decodes to sea level with a zero unpack vector.
243        let empty_dem = Texture::new(
244            Some("empty DEM"),
245            device,
246            wgpu::TextureFormat::Rgba8Unorm,
247            1,
248            1,
249            Msaa { samples: 1 },
250            wgpu::TextureUsages::TEXTURE_BINDING,
251        );
252        Self {
253            pipeline,
254            sampler,
255            mipmaps,
256            vertex_buffer,
257            index_buffer,
258            index_count: mesh.indices.len() as u32,
259            uniform_buffer,
260            dem_textures: HashMap::new(),
261            empty_dem,
262            drapes: DrapeCache::default(),
263            drape_scratch: None,
264            draws: Vec::new(),
265            bind_groups: HashMap::new(),
266            msaa,
267            color_format,
268            depth_format,
269            drape_size,
270        }
271    }

303    /// Bytes of drape textures, held and free, and of DEM textures.
304    pub fn texture_bytes(&self) -> (usize, usize) {
305        let (held, free) = self.drape_counts();
306        // Four bytes a texel, and a mip chain adds a third.
307        let drape = (self.drape_size as usize).pow(2) * 4 * 4 / 3;
308        let dem = self
309            .dem_textures
310            .values()
311            .map(|dem| (dem.texture.size.width * dem.texture.size.height * 4) as usize)
312            .sum();
313        ((held + free) * drape, dem)
314    }

316    /// Creates the scratch attachments used by every drape pass.
317    pub fn ensure_scratch(&mut self, device: &wgpu::Device) {
318        if self.drape_scratch.is_some() {
319            return;
320        }
321        let color = self.msaa.is_multisampling().then(|| {
322            Texture::new(
323                Some("drape multisampled color"),
324                device,
325                self.color_format,
326                self.drape_size,
327                self.drape_size,
328                self.msaa,
329                wgpu::TextureUsages::RENDER_ATTACHMENT,
330            )
331        });
332        let depth_stencil = Texture::new(
333            Some("drape depth stencil"),
334            device,
335            self.depth_format,
336            self.drape_size,
337            self.drape_size,
338            self.msaa,
339            wgpu::TextureUsages::RENDER_ATTACHMENT,
340        );
341        self.drape_scratch = Some(DrapeScratch {
342            color,
343            depth_stencil,
344        });
345    }

352    /// Writes the per-tile uniform blocks and returns how many fit.
353    pub fn write_uniforms(
354        &self,
355        queue: &crate::render::upload_queue::UploadQueue,
356        uniforms: &[TerrainTileUniforms],
357    ) -> usize {
358        let count = uniforms.len().min(UNIFORM_CAPACITY as usize);
359        if count < uniforms.len() {
360            tracing::warn!(
361                requested = uniforms.len(),
362                capacity = UNIFORM_CAPACITY,
363                "more terrain tiles than uniform blocks; distant tiles are skipped"
364            );
365        }
366        let mut bytes = vec![0_u8; count * UNIFORM_STRIDE as usize];
367        for (index, block) in uniforms.iter().take(count).enumerate() {
368            let start = index * UNIFORM_STRIDE as usize;
369            let raw = bytemuck::bytes_of(block);
370            bytes[start..start + raw.len()].copy_from_slice(raw);
371        }
372        queue.write_buffer(&self.uniform_buffer, 0, &bytes);
373        count
374    }

376    /// Replaces this frame's terrain draws.
377    pub fn set_draws(&mut self, draws: Vec<TerrainDraw>) {
378        self.draws = draws;
379        // A bind group keeps its textures alive; only the ones this frame draws with stay.
380        #[allow(
381            clippy::mutable_key_type,
382            reason = "wgpu hashes immutable resource identity, independent of internal synchronization"
383        )]
384        let used: HashSet<wgpu::BindGroup> = self
385            .draws
386            .iter()
387            .map(|draw| draw.bind_group.as_ref().clone())
388            .collect();
389        self.bind_groups
390            .retain(|_, group| used.contains(group.as_ref()));
391    }
