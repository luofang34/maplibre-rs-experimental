7/// The terrain surface tiles the last frame's own selection drew.

15/// How many worlds over each surface tile is seen again: its copies beside the main world.
16#[derive(Default)]
17pub(super) struct SurfaceCopies(pub std::collections::HashMap<WorldTileCoords, Vec<i32>>);
18
19impl SurfaceCopies {
20    /// The copies of the world the view sees its tiles in, keyed by the tile they repeat.
21    pub(super) fn of(tiles: impl Iterator<Item = WorldTileCoords>) -> Self {
22        let mut copies: std::collections::HashMap<WorldTileCoords, Vec<i32>> = Default::default();
23        for (tile, wrap) in tiles.filter_map(WorldTileCoords::wrapped) {
24            if wrap != 0 {
25                let wraps = copies.entry(tile).or_default();
26                if !wraps.contains(&wrap) {
27                    wraps.push(wrap);
28                }
29            }
30        }
31        Self(copies)
32    }
33}

35pub(super) fn for_frame(
36    world: &mut World,
37    drapes: &[TargetSpec],
38    sources: &[Option<WorldTileCoords>],
39) -> (Vec<TargetSpec>, Vec<Option<WorldTileCoords>>) {
40    let surfaces = world
41        .resources
42        .get::<SurfaceTiles>()
43        .map(|surface| surface.0.clone())
44        .unwrap_or_default();
45    let coordinates: Vec<_> = drapes.iter().map(|drape| drape.coords).collect();
46    let (surfaces, mapped) =
47        if let Some(active) = super::cohort::active_sources(world, &coordinates, sources) {
48            super::cohort::surface_pieces(&surfaces, &active)
49        } else {
50            let mapped = map_sources(&surfaces, drapes, sources);
51            (surfaces, mapped)
52        };
53    let specs = surfaces
54        .into_iter()
55        .map(|coords| TargetSpec {
56            absent_sources: Vec::new(),
57            coords,
58            shapes: Vec::new(),
59        })
60        .collect();
61    (specs, mapped)
62}

64fn map_sources(
65    surfaces: &[WorldTileCoords],
66    drapes: &[TargetSpec],
67    sources: &[Option<WorldTileCoords>],
68) -> Vec<Option<WorldTileCoords>> {
69    surfaces
70        .iter()
71        .map(|coords| {
72            drapes
73                .iter()
74                .zip(sources)
75                .find(|(drape, _)| crate::projection::tile_covering::covers(drape.coords, *coords))
76                .and_then(|(_, source)| *source)
77        })
78        .collect()
79}

81pub(super) fn fit_metadata(specs: &[TargetSpec], ready: &mut [bool], mut remaining: usize) {
82    for (spec, ready) in specs.iter().zip(ready) {
83        if !*ready {
84            continue;
85        }
86        if spec.shapes.len() > remaining {
87            *ready = false;
88        } else {
89            remaining -= spec.shapes.len();
90        }
91    }
92}
