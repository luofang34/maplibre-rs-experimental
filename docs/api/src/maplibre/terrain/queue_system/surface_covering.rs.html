1//! Texture budgets do not change the mesh or DEM resolution under a surface.
2use crate::{coords::WorldTileCoords, tcs::world::World, terrain::drape_targets::TargetSpec};

4#[derive(Default)]
5pub(super) struct SurfaceTiles(pub Vec<WorldTileCoords>);

7/// How many worlds over each surface tile is seen again: its copies beside the main world.
8#[derive(Default)]
9pub(super) struct SurfaceCopies(pub std::collections::HashMap<WorldTileCoords, Vec<i32>>);
10
11impl SurfaceCopies {
12    /// The copies of the world the view sees its tiles in, keyed by the tile they repeat.
13    pub(super) fn of(tiles: impl Iterator<Item = WorldTileCoords>) -> Self {
14        let mut copies: std::collections::HashMap<WorldTileCoords, Vec<i32>> = Default::default();
15        for (tile, wrap) in tiles.filter_map(WorldTileCoords::wrapped) {
16            if wrap != 0 {
17                let wraps = copies.entry(tile).or_default();
18                if !wraps.contains(&wrap) {
19                    wraps.push(wrap);
20                }
21            }
22        }
23        Self(copies)
24    }
25}

27pub(super) fn for_frame(
28    world: &mut World,
29    drapes: &[TargetSpec],
30    sources: &[Option<WorldTileCoords>],
31) -> (Vec<TargetSpec>, Vec<Option<WorldTileCoords>>) {
32    let surfaces = world
33        .resources
34        .get::<SurfaceTiles>()
35        .map(|surface| surface.0.clone())
36        .unwrap_or_default();
37    let coordinates: Vec<_> = drapes.iter().map(|drape| drape.coords).collect();
38    let (surfaces, mapped) =
39        if let Some(active) = super::cohort::active_sources(world, &coordinates, sources) {
40            super::cohort::surface_pieces(&surfaces, &active)
41        } else {
42            let mapped = map_sources(&surfaces, drapes, sources);
43            (surfaces, mapped)
44        };
45    let specs = surfaces
46        .into_iter()
47        .map(|coords| TargetSpec {
48            absent_sources: Vec::new(),
49            coords,
50            shapes: Vec::new(),
51        })
52        .collect();
53    (specs, mapped)
54}

56fn map_sources(
57    surfaces: &[WorldTileCoords],
58    drapes: &[TargetSpec],
59    sources: &[Option<WorldTileCoords>],
60) -> Vec<Option<WorldTileCoords>> {
61    surfaces
62        .iter()
63        .map(|coords| {
64            drapes
65                .iter()
66                .zip(sources)
67                .find(|(drape, _)| crate::projection::tile_covering::covers(drape.coords, *coords))
68                .and_then(|(_, source)| *source)
69        })
70        .collect()
71}

73pub(super) fn fit_metadata(specs: &[TargetSpec], ready: &mut [bool], mut remaining: usize) {
74    for (spec, ready) in specs.iter().zip(ready) {
75        if !*ready {
76            continue;
77        }
78        if spec.shapes.len() > remaining {
79            *ready = false;
80        } else {
81            remaining -= spec.shapes.len();
82        }
83    }
84}
