13/// Which of the acquired drapes to draw this frame, at most `budget`: tiles never drawn
14/// first, since a reused texture shows another tile's content until they are, then tiles
15/// whose content changed.
16pub(super) fn budget_redraws(states: &[DrapeState], budget: usize) -> Vec<bool> {
17    let mut redraw = vec![false; states.len()];
18    let mut drawn = 0;
19    for wanted in [DrapeState::New, DrapeState::Changed] {
20        for (index, state) in states.iter().enumerate() {
21            if drawn >= budget {
22                return redraw;
23            }
24            if *state == wanted {
25                redraw[index] = true;
26                drawn += 1;
27            }
28        }
29    }
30    redraw
31}

33/// Tiles whose texture holds no content of their own yet. A new drape the budget deferred
34/// would show whatever tile last used the texture, so its surface uses an ancestor or the
35/// background color until its own texture is drawn.
36pub(super) fn awaiting_first_draw(states: &[DrapeState], redraw: &[bool]) -> Vec<bool> {
37    states
38        .iter()
39        .zip(redraw)
40        .map(|(state, drawn)| *state == DrapeState::New && !drawn)
41        .collect()
42}

44pub(super) fn acquire_drapes(
45    specs: &[TargetSpec],
46    (prints, ready, prefetch): (&[u64], &[bool], &[bool]),
47    (memory, view_textures): (MemoryBudget, usize),
48    (budget, limit): (usize, usize),
49    terrain: &mut TerrainResources,
50    device: &wgpu::Device,
51) -> (
52    Vec<bool>,
53    Vec<Option<WorldTileCoords>>,
54    bool,
55    Vec<DrapeState>,
56) {
57    let keep = covering::retained_textures(specs.iter().map(|spec| spec.coords), |coords| {
58        terrain.drape_texture(coords).is_some()
59    });
60    terrain.retain_drapes(&keep);
61    // Unready tiles keep their last valid texture or a background surface.
62    if memory.is_tight() {
63        terrain.shed_spare_drapes();
64    }
65    // Targets come nearest first, so when textures run out it is the far tiles that
66    // draw with an ancestor's drape.
67    let states: Vec<DrapeState> = specs
68        .iter()
69        .zip(prints)
70        .zip(ready)
71        .zip(prefetch)
72        .map(|(((spec, print), ready), prefetch)| {
73            if !*ready {
74                return DrapeState::Unchanged;
75            }
76            // Past its share of the textures a view goes without, as it does when the map's
77            // allowance is spent, rather than take what another view's share covers.
78            if !terrain.within_drape_allowance(spec.coords, view_textures) {
79                return DrapeState::Withheld;
80            }
81            let may_create = memory.allows_drape_texture(terrain.drape_texture_total());
82            if *prefetch {
83                terrain.acquire_ahead_drape(device, spec.coords, *print, may_create)
84            } else {
85                terrain.acquire_drape(device, spec.coords, *print, may_create)
86            }
87        })
88        .collect();
89    let redraw = choose_redraws(specs, &states, prefetch, (budget, limit), terrain);
90    let mut deferred = false;
91    for ((spec, state), drawn) in specs.iter().zip(&states).zip(&redraw) {
92        if !matches!(state, DrapeState::Unchanged | DrapeState::Withheld) && !drawn {
93            terrain.defer_drape(spec.coords, *state);
94            deferred = true;
95        }
96    }
97    let sources = drape_sources(specs, &states, &redraw, ready, terrain);
98    (redraw, sources, deferred, states)
99}

101/// The drapes to draw this frame. Those the view shows take the budget first; a new one with
102/// no drawn ancestor would leave its tile blank, so it is drawn past the budget, up to the count
103/// a frame's command buffer holds. Prefetched drapes, which no surface shows yet, draw only
104/// with the budget the view leaves and never past it.
105fn choose_redraws(
106    specs: &[TargetSpec],
107    states: &[DrapeState],
108    prefetch: &[bool],
109    (budget, limit): (usize, usize),
110    terrain: &TerrainResources,
111) -> Vec<bool> {
112    let only = |ahead: bool| -> Vec<DrapeState> {
113        states
114            .iter()
115            .zip(prefetch)
116            .map(|(state, prefetched)| {
117                if *prefetched == ahead {
118                    *state
119                } else {
120                    DrapeState::Unchanged
121                }
122            })
123            .collect()
124    };
125    let mut redraw = budget_redraws(&only(false), budget);
126    let mut drawn = redraw.iter().filter(|drawn| **drawn).count();
127    for (index, (spec, state)) in specs.iter().zip(states).enumerate() {
128        if drawn >= limit {
129            break;
130        }
131        if !prefetch[index]
132            && *state == DrapeState::New
133            && !redraw[index]
134            && present_ancestors(spec.coords, terrain).is_empty()
135        {
136            redraw[index] = true;
137            drawn += 1;
138        }
139    }
140    let spare = budget.saturating_sub(drawn);
141    for (index, draw) in budget_redraws(&only(true), spare).into_iter().enumerate() {
142        redraw[index] |= draw;
143    }
144    redraw
145}

147pub(super) fn drape_sources(
148    specs: &[TargetSpec],
149    states: &[DrapeState],
150    redraw: &[bool],
151    ready: &[bool],
152    terrain: &TerrainResources,
153) -> Vec<Option<WorldTileCoords>> {
154    let hidden: Vec<bool> = awaiting_first_draw(states, redraw)
155        .into_iter()
156        .zip(ready)
157        .zip(states)
158        .map(|((hidden, ready), state)| hidden || !ready || *state == DrapeState::Withheld)
159        .collect();
160    let undrawn: HashSet<WorldTileCoords> = specs
161        .iter()
162        .zip(&hidden)
163        .filter(|(_, hidden)| **hidden)
164        .map(|(spec, _)| spec.coords)
165        .collect();
166    specs
167        .iter()
168        .zip(&hidden)
169        .map(|(spec, hidden)| {
170            if *hidden && terrain.drape_texture(spec.coords).is_none() {
171                present_ancestors(spec.coords, terrain)
172                    .into_iter()
173                    .find(|ancestor| !undrawn.contains(ancestor))
174            } else {
175                Some(spec.coords)
176            }
177        })
178        .collect()
179}
