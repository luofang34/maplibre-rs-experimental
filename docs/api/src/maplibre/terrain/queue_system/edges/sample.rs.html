4pub(super) fn scale_of(z: u8) -> f64 {
5    if z < 31 {
6        f64::from(1_u32 << z)
7    } else {
8        2_f64.powi(i32::from(z))
9    }
10}

12pub(super) struct Sources {
13    pub(super) tiles: FastTileMap<Option<WorldTileCoords>>,
14    zooms: Vec<u8>,
15}

17impl Sources {
18    pub(super) fn new(tiles: FastTileMap<Option<WorldTileCoords>>) -> Self {
19        let mut zooms: Vec<_> = tiles.keys().map(|tile| u8::from(tile.z)).collect();
20        zooms.sort_unstable();
21        zooms.dedup();
22        Self { tiles, zooms }
23    }
24}

26/// The loaded DEM of every source a selection samples, looked up once per rebuild so the
27/// per-vertex sampling does no tile query.
28pub(super) struct Dems<'a>(FastTileMap<&'a DemTile>);
29
30impl<'a> Dems<'a> {
31    pub(super) fn new(sources: &Sources, tiles: &'a Tiles) -> Self {
32        let mut map = FastTileMap::with_capacity_and_hasher(
33            sources.tiles.len(),
34            BuildHasherDefault::default(),
35        );
36        for source in sources.tiles.values().flatten() {
37            if !map.contains_key(source) {
38                if let Some(DemTileComponent::Loaded(dem)) =
39                    tiles.query::<&DemTileComponent>(*source)
40                {
41                    map.insert(*source, dem.tile.as_ref());
42                }
43            }
44        }
45        Self(map)
46    }
47}

49#[derive(Clone, Copy)]
50struct ResolvedDem<'a> {
51    dem: &'a DemTile,
52    coords: WorldTileCoords,
53    scale: f64,
54}
55
56impl<'a> ResolvedDem<'a> {
57    #[inline]
58    fn lookup(tile: WorldTileCoords, sources: &Sources, dems: &'a Dems<'a>) -> Option<Self> {
59        let source = sources.tiles.get(&tile).copied().flatten()?;
60        let dem = dems.0.get(&source).copied()?;
61        Some(Self {
62            dem,
63            coords: source,
64            scale: scale_of(u8::from(source.z)),
65        })
66    }
67
68    #[inline]
69    fn sample(self, point: [f64; 2]) -> f64 {
70        let x = (point[0] * self.scale - f64::from(self.coords.x))
71            .rem_euclid(self.scale)
72            .min(1.0);
73        let y = (point[1] * self.scale - f64::from(self.coords.y)).clamp(0.0, 1.0);
74        self.dem.elevation_at_tile_coords(x * EXTENT, y * EXTENT)
75    }
76}

78pub(super) fn build_edges(tile: WorldTileCoords, sources: &Sources, dems: &Dems) -> EdgeHeights {
79    let mut result = EdgeHeights::default();
80    let scale = scale_of(u8::from(tile.z));
81    let tx = f64::from(tile.x);
82    let ty = f64::from(tile.y);
83
84    // Each of the four corners is shared by two incident edges.
85    let nw = shared_height([tx / scale, ty / scale], tile, sources, dems) as f32;
86    let ne = shared_height([(tx + 1.0) / scale, ty / scale], tile, sources, dems) as f32;
87    let sw = shared_height([tx / scale, (ty + 1.0) / scale], tile, sources, dems) as f32;
88    let se = shared_height(
89        [(tx + 1.0) / scale, (ty + 1.0) / scale],
90        tile,
91        sources,
92        dems,
93    ) as f32;
94    result.samples[0] = [nw, sw, nw, ne];
95    result.last = [ne, se, sw, se];
96
97    let can_share_side_owner = sources
98        .zooms
99        .first()
100        .is_none_or(|&min_z| u8::from(tile.z).saturating_sub(min_z) <= 15);
101    if !can_share_side_owner {
102        for (i, slot) in result.samples.iter_mut().enumerate().skip(1) {
103            let t = i as f64 / N as f64;
104            *slot = [[t, 0.0], [t, 1.0], [0.0, t], [1.0, t]].map(|uv| {
105                let point = [(tx + uv[0]) / scale, (ty + uv[1]) / scale];
106                shared_height(point, tile, sources, dems) as f32
107            });
108        }
109        return result;
110    }
111
112    // Horizontal edges: side 0 (north, v = 0) and side 1 (south, v = 1).
113    for (side, v_off) in [(0_usize, 0.0_f64), (1_usize, 1.0_f64)] {
114        let py = (ty + v_off) / scale;
115        let side_owner = owner([(tx + 0.5) / scale, py], tile, sources);
116        fill_horizontal_side(
117            &mut result,
118            side,
119            (tx, py, scale),
120            side_owner,
121            sources,
122            dems,
123        );
124    }
125    // Vertical edges: side 2 (west, u = 0) and side 3 (east, u = 1).
126    for (side, u_off) in [(2_usize, 0.0_f64), (3_usize, 1.0_f64)] {
127        let px = (tx + u_off) / scale;
128        let side_owner = owner([px, (ty + 0.5) / scale], tile, sources);
129        fill_vertical_side(
130            &mut result,
131            side,
132            (px, ty, scale),
133            side_owner,
134            sources,
135            dems,
136        );
137    }
138    result
139}

141fn fill_horizontal_side(
142    result: &mut EdgeHeights,
143    side: usize,
144    (tx, py, tile_scale): (f64, f64, f64),
145    side_owner: WorldTileCoords,
146    sources: &Sources,
147    dems: &Dems,
148) {
149    let owner_scale = scale_of(u8::from(side_owner.z));
150    let owner_x = f64::from(side_owner.x);
151    let owner_y = f64::from(side_owner.y);
152    let owner_dem = ResolvedDem::lookup(side_owner, sources, dems);
153    let y = (py * owner_scale - owner_y).clamp(0.0, 1.0) * N as f64;
154    let gy0 = y.floor();
155    let gy1 = y.ceil();
156    let fy = y.fract();
157    let single_row = fy == 0.0;
158
159    let mut cached_gx = -1_i32;
160    let mut cached_top = 0.0_f64;
161    let mut cached_bot = 0.0_f64;
162
163    let eval_col = |gx: f64| -> (f64, f64) {
164        let px = (owner_x + gx / N as f64) / owner_scale;
165        let interior = single_row && gx > 0.0 && gx < N as f64;
166        let top = if interior {
167            owner_dem.map_or(0.0, |d| {
168                d.sample([px, (owner_y + gy0 / N as f64) / owner_scale])
169            })
170        } else {
171            let p = [px, (owner_y + gy0 / N as f64) / owner_scale];
172            let ep = owner(p, side_owner, sources);
173            sample_source(p, sources.tiles.get(&ep).copied().flatten(), dems)
174        };
175        let bot = if single_row {
176            top
177        } else {
178            let p = [px, (owner_y + gy1 / N as f64) / owner_scale];
179            let ep = owner(p, side_owner, sources);
180            sample_source(p, sources.tiles.get(&ep).copied().flatten(), dems)
181        };
182        (top, bot)
183    };
184
185    for (i, slot) in result.samples.iter_mut().enumerate().skip(1) {
186        let px = (tx + i as f64 / N as f64) / tile_scale;
187        let x = (px * owner_scale - owner_x)
188            .rem_euclid(owner_scale)
189            .min(1.0)
190            * N as f64;
191        let gx0 = x.floor();
192        let fx = x.fract();
193        let gx0_i = gx0 as i32;
194        let (t0, b0) = if gx0_i == cached_gx {
195            (cached_top, cached_bot)
196        } else {
197            eval_col(gx0)
198        };
199        let val = if fx == 0.0 {
200            cached_gx = gx0_i;
201            cached_top = t0;
202            cached_bot = b0;
203            if single_row {
204                t0
205            } else {
206                t0 * (1.0 - fy) + b0 * fy
207            }
208        } else {
209            let gx1 = x.ceil();
210            let (t1, b1) = eval_col(gx1);
211            cached_gx = gx1 as i32;
212            cached_top = t1;
213            cached_bot = b1;
214            let top = t0 * (1.0 - fx) + t1 * fx;
215            if single_row {
216                top
217            } else {
218                let bot = b0 * (1.0 - fx) + b1 * fx;
219                top * (1.0 - fy) + bot * fy
220            }
221        };
222        slot[side] = val as f32;
223    }
224}

226fn fill_vertical_side(
227    result: &mut EdgeHeights,
228    side: usize,
229    (px, ty, tile_scale): (f64, f64, f64),
230    side_owner: WorldTileCoords,
231    sources: &Sources,
232    dems: &Dems,
233) {
234    let owner_scale = scale_of(u8::from(side_owner.z));
235    let owner_x = f64::from(side_owner.x);
236    let owner_y = f64::from(side_owner.y);
237    let owner_dem = ResolvedDem::lookup(side_owner, sources, dems);
238    let x = (px * owner_scale - owner_x)
239        .rem_euclid(owner_scale)
240        .min(1.0)
241        * N as f64;
242    let gx0 = x.floor();
243    let gx1 = x.ceil();
244    let fx = x.fract();
245    let single_col = fx == 0.0;
246
247    let mut cached_gy = -1_i32;
248    let mut cached_left = 0.0_f64;
249    let mut cached_right = 0.0_f64;
250
251    let eval_row = |gy: f64| -> (f64, f64) {
252        let py = (owner_y + gy / N as f64) / owner_scale;
253        let interior = single_col && gy > 0.0 && gy < N as f64;
254        let left = if interior {
255            owner_dem.map_or(0.0, |d| {
256                d.sample([(owner_x + gx0 / N as f64) / owner_scale, py])
257            })
258        } else {
259            let p = [(owner_x + gx0 / N as f64) / owner_scale, py];
260            let ep = owner(p, side_owner, sources);
261            sample_source(p, sources.tiles.get(&ep).copied().flatten(), dems)
262        };
263        let right = if single_col {
264            left
265        } else {
266            let p = [(owner_x + gx1 / N as f64) / owner_scale, py];
267            let ep = owner(p, side_owner, sources);
268            sample_source(p, sources.tiles.get(&ep).copied().flatten(), dems)
269        };
270        (left, right)
271    };
272
273    for (i, slot) in result.samples.iter_mut().enumerate().skip(1) {
274        let py = (ty + i as f64 / N as f64) / tile_scale;
275        let y = (py * owner_scale - owner_y).clamp(0.0, 1.0) * N as f64;
276        let gy0 = y.floor();
277        let fy = y.fract();
278        let gy0_i = gy0 as i32;
279        let (l0, r0) = if gy0_i == cached_gy {
280            (cached_left, cached_right)
281        } else {
282            eval_row(gy0)
283        };
284        let val = if fy == 0.0 {
285            cached_gy = gy0_i;
286            cached_left = l0;
287            cached_right = r0;
288            if single_col {
289                l0
290            } else {
291                l0 * (1.0 - fx) + r0 * fx
292            }
293        } else {
294            let gy1 = y.ceil();
295            let (l1, r1) = eval_row(gy1);
296            cached_gy = gy1 as i32;
297            cached_left = l1;
298            cached_right = r1;
299            let top = if single_col {
300                l0
301            } else {
302                l0 * (1.0 - fx) + r0 * fx
303            };
304            let bot = if single_col {
305                l1
306            } else {
307                l1 * (1.0 - fx) + r1 * fx
308            };
309            top * (1.0 - fy) + bot * fy
310        };
311        slot[side] = val as f32;
312    }
313}

315fn owner(point: [f64; 2], fallback: WorldTileCoords, sources: &Sources) -> WorldTileCoords {
316    // The coarsest touching mesh owns a shared vertex, including four-way corners.
317    // Both sides use this ordering, independent of draw order or tile arrival order.
318    let max_z = u8::from(fallback.z);
319    for &z in &sources.zooms {
320        if z > max_z {
321            break;
322        }
323        let scale = scale_of(z);
324        let x = point[0] * scale;
325        let y = point[1] * scale;
326        let y0 = (y - 1e-7).floor() as i32;
327        let y1 = (y + 1e-7).floor() as i32;
328        let x0 = (x - 1e-7).floor() as i32;
329        let x1 = (x + 1e-7).floor() as i32;
330        let wrap = 1_i32 << z;
331        let probe = |xx: i32, yy: i32| -> Option<WorldTileCoords> {
332            let tile = WorldTileCoords {
333                x: xx.rem_euclid(wrap),
334                y: yy,
335                z: z.into(),
336            };
337            sources.tiles.contains_key(&tile).then_some(tile)
338        };
339        if let Some(tile) = probe(x0, y0) {
340            return tile;
341        }
342        if x1 != x0 {
343            if let Some(tile) = probe(x1, y0) {
344                return tile;
345            }
346        }
347        if y1 != y0 {
348            if let Some(tile) = probe(x0, y1) {
349                return tile;
350            }
351            if x1 != x0 {
352                if let Some(tile) = probe(x1, y1) {
353                    return tile;
354                }
355            }
356        }
357    }
358    fallback
359}

361pub(super) fn shared_height(
362    point: [f64; 2],
363    fallback: WorldTileCoords,
364    sources: &Sources,
365    dems: &Dems,
366) -> f64 {
367    let tile = owner(point, fallback, sources);
368    let scale = scale_of(u8::from(tile.z));
369    let x = (point[0] * scale - f64::from(tile.x))
370        .rem_euclid(scale)
371        .min(1.0)
372        * N as f64;
373    let y = (point[1] * scale - f64::from(tile.y)).clamp(0.0, 1.0) * N as f64;
374    let sample = |gx: f64, gy: f64| {
375        // Corners can touch an even coarser mesh. The common owner provides the
376        // endpoint from which every intermediate fine-edge vertex interpolates.
377        let p = [
378            (f64::from(tile.x) + gx / N as f64) / scale,
379            (f64::from(tile.y) + gy / N as f64) / scale,
380        ];
381        let endpoint_owner = owner(p, tile, sources);
382        sample_source(
383            p,
384            sources.tiles.get(&endpoint_owner).copied().flatten(),
385            dems,
386        )
387    };
388    let fx = x.fract();
389    let fy = y.fract();
390    let top = if fx == 0.0 {
391        sample(x.floor(), y.floor())
392    } else {
393        sample(x.floor(), y.floor()) * (1.0 - fx) + sample(x.ceil(), y.floor()) * fx
394    };
395    if fy == 0.0 {
396        top
397    } else {
398        let bottom = if fx == 0.0 {
399            sample(x.floor(), y.ceil())
400        } else {
401            sample(x.floor(), y.ceil()) * (1.0 - fx) + sample(x.ceil(), y.ceil()) * fx
402        };
403        top * (1.0 - fy) + bottom * fy
404    }
405}

407fn sample_source(point: [f64; 2], source: Option<WorldTileCoords>, dems: &Dems) -> f64 {
408    let Some(source) = source else {
409        return 0.0;
410    };
411    let Some(dem) = dems.0.get(&source) else {
412        return 0.0;
413    };
414    let scale = scale_of(u8::from(source.z));
415    let x = (point[0] * scale - f64::from(source.x))
416        .rem_euclid(scale)
417        .min(1.0);
418    let y = (point[1] * scale - f64::from(source.y)).clamp(0.0, 1.0);
419    dem.elevation_at_tile_coords(x * EXTENT, y * EXTENT)
420}
