1//! A cached drape is replaced only by a complete, uploaded set of visible layers.
2
3use crate::{
4    io::tile_sources::RASTER_LAYER_TYPES,
5    raster::resource::RasterResources,
6    render::{
7        eventually::{Eventually, Eventually::Initialized},
8        tile_view_pattern::coverage::covers_target,
9    },
10    style::Style,
11    tcs::world::World,
12    terrain::drape_targets::TargetSpec,
13    vector::geometry_uploaded,
14};
15
16pub(super) fn ready(
17    spec: &TargetSpec,
18    style: &Style,
19    zoom: f64,
20    world: &World,
21    strict: bool,
22) -> bool {
23    !spec.shapes.is_empty()
24        && spec.shapes.iter().all(|shape| {
25            !shape.raster_layers.is_empty()
26                || ((!strict || shape.source == spec.coords)
27                    && geometry_uploaded(shape.source, world))
28        })
29        && style
30            .layers
31            .iter()
32            .filter(|layer| {
33                RASTER_LAYER_TYPES.contains(&layer.type_.as_str()) && layer.is_visible_at(zoom)
34            })
35            .all(|layer| {
36                let shapes: Vec<_> = spec
37                    .shapes
38                    .iter()
39                    .filter(|shape| shape.raster_layers.iter().any(|(id, _, _)| id == &layer.id))
40                    .collect();
41                let sources: Vec<_> = shapes.iter().map(|shape| shape.source).collect();
42                let absent = matches!(
43                    world.resources.get::<Eventually<RasterResources>>(),
44                    Some(Initialized(resources))
45                        if resources
46                            .layer_source(&layer.id)
47                            .is_some_and(|source| spec.absent_sources.contains(source))
48                );
49                absent
50                    || covers_target(spec.coords, &sources)
51                    || (!shapes.is_empty() && shapes.iter().all(|shape| shape.view_complete))
52            })
53}
