4pub(crate) fn uses_uniform_texture_covering(view_state: &ViewState) -> bool {
5    use cgmath::InnerSpace;
6    view_state.external_globe_eye().is_some_and(|eye| {
7        (eye.position.magnitude() - 1.0) * view_state.body().radius_meters >= 60_000.0
8    })
9}

11pub(crate) fn target_specs(
12    style: &Style,
13    view_state: &ViewState,
14    world: &mut World,
15) -> Result<Vec<TargetSpec>, SystemError> {
16    let zoom = view_state.zoom();
17    let (_, raster_coverings) =
18        drawn_covering(style, view_state, world, zoom.zoom_level(DEFAULT_TILE_SIZE)).map_err(
19            |error| {
20                tracing::error!(%error, "unable to select terrain tiles");
21                SystemError::Setup
22            },
23        )?;
24    let Some(dem) = crate::terrain::source::dem_source(style) else {
25        return Ok(Vec::new());
26    };
27    let view_region = crate::terrain::coverage::take_or_compute_tight_terrain_region(
28        style,
29        view_state,
30        world,
31        dem.tile_size,
32    )
33    .map_err(|error| {
34        tracing::error!(%error, "unable to select terrain tiles");
35        SystemError::Setup
36    })?;
37    let Some(view_region) = view_region else {
38        return Ok(Vec::new());
39    };
40    let memory = world
41        .resources
42        .get::<MemoryBudget>()
43        .copied()
44        .unwrap_or_default();
45    world
46        .resources
47        .insert(surface_covering::SurfaceTiles(view_region.iter().collect()));
48    world
49        .resources
50        .insert(surface_covering::SurfaceCopies::of(view_region.copies()));
51    let uniform_globe = uses_uniform_texture_covering(view_state);
52    world
53        .resources
54        .get_or_init_mut::<cohort::TextureCohort>()
55        .enabled = uniform_globe;
56    let tiles: Vec<_> = if uniform_globe {
57        cohort::targets(
58            world,
59            &view_region.iter().collect::<Vec<_>>(),
60            memory.drape_textures_allowed(),
61        )
62    } else if view_state.has_external_view() || view_state.pose_view().is_some() {
63        // An eye's screen-resolution covering reaches its horizon in more tiles than the
64        // textures allow; the surfaces keep every tile, so only the drapes coarsen.
65        let focus = covering::focus_for_view(style, view_state);
66        covering::for_frame(
67            world,
68            view_region.iter().collect(),
69            // Reserve room for two sibling groups while their parents remain visible.
70            memory.drape_textures_allowed().saturating_sub(8).max(1),
71            focus,
72        )
73    } else {
74        view_region.iter().collect()
75    };
76    super::prefetch::prepare(style, view_state, world, &tiles, memory);
77    world
78        .resources
79        .insert(crate::terrain::request_system::DrapeRequests(tiles.clone()));
80    let mut targets = select_targets(tiles.into_iter(), world, &raster_coverings);
81    if !uniform_globe && !view_state.has_external_view() && view_state.pose_view().is_none() {
82        let prefetch_tiles = world
83            .resources
84            .get::<crate::terrain::request_system::DrapePrefetchRequests>()
85            .map(|requests| requests.0.clone())
86            .unwrap_or_default();
87        if !prefetch_tiles.is_empty() {
88            if let Ok(loose_raster) = crate::render::projection::raster_source_regions(
89                style,
90                view_state,
91                world,
92                crate::render::view_state::ViewStatePadding::Loose,
93            ) {
94                targets.extend(select_targets(
95                    prefetch_tiles.into_iter(),
96                    world,
97                    &loose_raster,
98                ));
99            }
100        }
101    }
102    Ok(collect_layer_specs(
103        targets,
104        style,
105        world,
106        view_state.style_zoom().value(),
107    ))
108}

110/// Selects pure 3D DEM surface tiles for [`crate::render::settings::TerrainDrapeMode::ProceduralRelief`]
111/// without computing 2D drape targets or prefetching draped source tiles.
112pub(crate) fn procedural_specs(
113    style: &Style,
114    view_state: &ViewState,
115    world: &mut World,
116) -> Result<Vec<TargetSpec>, SystemError> {
117    let Some(dem) = crate::terrain::source::dem_source(style) else {
118        return Ok(Vec::new());
119    };
120    let view_region = crate::terrain::coverage::take_or_compute_tight_terrain_region(
121        style,
122        view_state,
123        world,
124        dem.tile_size,
125    )
126    .map_err(|error| {
127        tracing::error!(%error, "unable to select procedural terrain tiles");
128        SystemError::Setup
129    })?;
130    let Some(view_region) = view_region else {
131        return Ok(Vec::new());
132    };
133    let tiles: Vec<_> = view_region.iter().collect();
134    world
135        .resources
136        .insert(surface_covering::SurfaceTiles(tiles.clone()));
137    world
138        .resources
139        .insert(surface_covering::SurfaceCopies::of(view_region.copies()));
140    world
141        .resources
142        .get_or_init_mut::<cohort::TextureCohort>()
143        .enabled = false;
144    world
145        .resources
146        .insert(crate::terrain::request_system::DrapePrefetchRequests::default());
147    world
148        .resources
149        .insert(crate::terrain::request_system::DrapeRequests::default());
150    Ok(tiles
151        .into_iter()
152        .map(|coords| TargetSpec {
153            coords,
154            shapes: Vec::new(),
155            absent_sources: Vec::new(),
156        })
157        .collect())
158}

160/// The key terrain tiles are drawn by, smallest first: `1.0 - c_hat.dot(focus_unit)` on `S^2`
161/// under Globe rendering, and GL JS's planar `(dx, dy)` squared distance at the nominal level
162/// under explicit 2D Mercator rendering. With translucent drapes the order decides which skirts
163/// show through.
164pub(super) fn draw_order_key(
165    style: &Style,
166    view_state: &ViewState,
167    dem_tile_size: f64,
168) -> impl Fn(&WorldTileCoords) -> f64 {
169    enum OrderMode {
170        Spherical(cgmath::Vector3<f64>),
171        Planar([f64; 2]),
172    }
173    let uses_globe = style.projection.as_ref().is_some_and(|p| {
174        p.projection_type
175            .uses_globe_rendering(view_state.zoom().value())
176    });
177    let mode = if uses_globe {
178        crate::render::projection::globe_camera_for_view(view_state)
179            .ok()
180            .map(|globe| OrderMode::Spherical(globe.focus_unit_sphere()))
181    } else {
182        None
183    }
184    .unwrap_or_else(|| {
185        let request = crate::render::projection::CoveringRequest::terrain(
186            view_state.zoom().value(),
187            dem_tile_size,
188        );
189        let tiles_across = 2_f64.powi(i32::from(u8::from(request.level)));
190        let world_size = crate::coords::TILE_SIZE * 2_f64.powf(view_state.zoom().value());
191        let center = view_state.camera().position();
192        OrderMode::Planar([
193            center.x / world_size * tiles_across - 0.5,
194            center.y / world_size * tiles_across - 0.5,
195        ])
196    });
197    move |tile: &WorldTileCoords| match mode {
198        OrderMode::Spherical(focus_unit) => {
199            crate::projection::tile_covering::spherical_tile_distance(*tile, focus_unit)
200        }
201        OrderMode::Planar(center) => {
202            let dx = center[0] - f64::from(tile.x);
203            let dy = center[1] - f64::from(tile.y);
204            dx * dx + dy * dy
205        }
206    }
207}
