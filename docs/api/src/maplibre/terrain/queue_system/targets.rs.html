10pub(crate) fn uses_uniform_texture_covering(view_state: &ViewState) -> bool {
11    use cgmath::InnerSpace;
12    view_state.external_globe_eye().is_some_and(|eye| {
13        (eye.position.magnitude() - 1.0) * view_state.body().radius_meters >= 60_000.0
14    })
15}

26pub(crate) fn target_specs_arc(
27    style: &Style,
28    view_state: &ViewState,
29    world: &mut World,
30) -> Result<std::sync::Arc<Vec<TargetSpec>>, SystemError> {
31    use super::spec_cache;
32    let zoom = view_state.zoom();
33    let (_, raster_coverings) =
34        drawn_covering(style, view_state, world, zoom.zoom_level(DEFAULT_TILE_SIZE)).map_err(
35            |error| {
36                tracing::error!(%error, "unable to select terrain tiles");
37                SystemError::Setup
38            },
39        )?;
40    let Some(dem) = crate::terrain::source::dem_source(style) else {
41        return Ok(std::sync::Arc::new(Vec::new()));
42    };
43    let view_region = crate::terrain::coverage::take_or_compute_tight_terrain_region(
44        style,
45        view_state,
46        world,
47        dem.tile_size,
48    )
49    .map_err(|error| {
50        tracing::error!(%error, "unable to select terrain tiles");
51        SystemError::Setup
52    })?;
53    let Some(view_region) = view_region else {
54        return Ok(std::sync::Arc::new(Vec::new()));
55    };
56    let memory = world
57        .resources
58        .get::<MemoryBudget>()
59        .copied()
60        .unwrap_or_default();
61    world
62        .resources
63        .insert(surface_covering::SurfaceTiles(view_region.iter().collect()));
64    world
65        .resources
66        .insert(surface_covering::SurfaceCopies::of(view_region.copies()));
67    let uniform_globe = uses_uniform_texture_covering(view_state);
68    world
69        .resources
70        .get_or_init_mut::<cohort::TextureCohort>()
71        .enabled = uniform_globe;
72    let allowance =
73        crate::terrain::drape_share::allowance(world, memory, view_region.iter().count());
74    let tiles: Vec<_> = if uniform_globe {
75        cohort::targets(
76            world,
77            &view_region.iter().collect::<Vec<_>>(),
78            allowance.textures,
79        )
80    } else if view_state.has_external_view() || view_state.pose_view().is_some() {
81        // An eye's screen-resolution covering reaches its horizon in more tiles than the
82        // textures allow; the surfaces keep every tile, so only the drapes coarsen.
83        let focus = covering::focus_for_view(style, view_state);
84        covering::for_frame(
85            world,
86            view_region.iter().collect(),
87            // The view's share of the textures, less those it regroups with.
88            allowance.targets,
89            focus,
90        )
91    } else {
92        view_region.iter().collect()
93    };
94    super::prefetch::prepare(
95        style,
96        view_state,
97        world,
98        &tiles,
99        (memory, allowance.targets),
100    );
101    world
102        .resources
103        .insert(crate::terrain::request_system::DrapeRequests(tiles.clone()));
104    world.resources.insert(PrefetchDrapes::default());
105    let arrivals_over_standin = view_state.has_external_view() || view_state.pose_view().is_some();
106    let arrivals = if arrivals_over_standin {
107        Arrivals::OverStandIn
108    } else {
109        Arrivals::Together
110    };
111    let mut prefetch_plan: Option<(
112        Vec<WorldTileCoords>,
113        crate::render::eye_covering::RasterCoverings,
114    )> = None;
115    let mut fresh_prefetch = Vec::new();
116    if !uniform_globe && !arrivals_over_standin {
117        let prefetch_tiles = world
118            .resources
119            .get::<crate::terrain::request_system::DrapePrefetchRequests>()
120            .map(|requests| requests.0.clone())
121            .unwrap_or_default();
122        if !prefetch_tiles.is_empty() {
123            if let Ok(loose_raster) = crate::render::projection::raster_source_regions(
124                style,
125                view_state,
126                world,
127                crate::render::view_state::ViewStatePadding::Loose,
128            ) {
129                let draped: rustc_hash::FxHashSet<WorldTileCoords> =
130                    tiles.iter().copied().collect();
131                let held = |coords: &WorldTileCoords| match world
132                    .resources
133                    .get::<Eventually<TerrainResources>>()
134                {
135                    Some(Eventually::Initialized(terrain)) => terrain.holds_shown_drape(*coords),
136                    _ => false,
137                };
138                fresh_prefetch = prefetch_tiles
139                    .into_iter()
140                    .filter(|coords| !draped.contains(coords) && !held(coords))
141                    .collect();
142                prefetch_plan = Some((fresh_prefetch.clone(), loose_raster));
143            }
144        }
145    }
146
147    let style_zoom = view_state.style_zoom().value();
148    let loose_raster_digest = prefetch_plan
149        .as_ref()
150        .map_or(0, |(_, loose)| spec_cache::raster_coverings_digest(loose));
151    let target_tiles_digest = spec_cache::target_coords_vector_digest(
152        tiles.iter().chain(fresh_prefetch.iter()).copied(),
153        world,
154    );
155    let specs_key = spec_cache::TargetSpecsKey {
156        tiles: tiles.clone(),
157        prefetch_fresh: fresh_prefetch,
158        arrivals_over_standin,
159        uniform_globe,
160        raster_coverings_digest: spec_cache::raster_coverings_digest(&raster_coverings),
161        loose_raster_digest,
162        vector_pool_state: spec_cache::vector_pool_state(world),
163        vector_arrival_epoch: crate::vector::arrival_epoch(world),
164        target_tiles_digest,
165        tiles_len: world.tiles.tiles.len(),
166        raster_gen: spec_cache::raster_mutation_gen(world),
167        style_rev: style.revision(),
168        style_vis_digest: spec_cache::style_visibility_digest(style, style_zoom),
169    };
170    if let Some((cached_specs, cached_prefetch)) = world
171        .resources
172        .get::<spec_cache::DrapePrepareCache>()
173        .and_then(|cache| cache.get_specs(&specs_key))
174    {
175        if !cached_prefetch.is_empty() {
176            world
177                .resources
178                .insert(PrefetchDrapes(cached_prefetch.clone()));
179        }
180        return Ok(cached_specs);
181    }
182
183    let mut targets = select_targets_as(tiles.into_iter(), world, &raster_coverings, arrivals);
184    let mut prefetch_drapes = rustc_hash::FxHashSet::default();
185    if let Some((fresh, loose_raster)) = prefetch_plan {
186        if !fresh.is_empty() {
187            prefetch_drapes = fresh.iter().copied().collect();
188            world
189                .resources
190                .insert(PrefetchDrapes(prefetch_drapes.clone()));
191            targets.extend(select_targets(fresh.into_iter(), world, &loose_raster));
192        }
193    }
194    let specs = std::sync::Arc::new(collect_layer_specs(targets, style, world, style_zoom));
195    world
196        .resources
197        .get_or_init_mut::<spec_cache::DrapePrepareCache>()
198        .store_specs(specs_key, std::sync::Arc::clone(&specs), prefetch_drapes);
199    Ok(specs)
200}

202/// Selects pure 3D DEM surface tiles for [`crate::render::settings::TerrainDrapeMode::ProceduralRelief`]
203/// without computing 2D drape targets or prefetching draped source tiles.
204pub(crate) fn procedural_specs(
205    style: &Style,
206    view_state: &ViewState,
207    world: &mut World,
208) -> Result<Vec<TargetSpec>, SystemError> {
209    let Some(dem) = crate::terrain::source::dem_source(style) else {
210        return Ok(Vec::new());
211    };
212    let view_region = crate::terrain::coverage::take_or_compute_tight_terrain_region(
213        style,
214        view_state,
215        world,
216        dem.tile_size,
217    )
218    .map_err(|error| {
219        tracing::error!(%error, "unable to select procedural terrain tiles");
220        SystemError::Setup
221    })?;
222    let Some(view_region) = view_region else {
223        return Ok(Vec::new());
224    };
225    let tiles: Vec<_> = view_region.iter().collect();
226    world
227        .resources
228        .insert(surface_covering::SurfaceTiles(tiles.clone()));
229    world
230        .resources
231        .insert(surface_covering::SurfaceCopies::of(view_region.copies()));
232    world
233        .resources
234        .get_or_init_mut::<cohort::TextureCohort>()
235        .enabled = false;
236    world
237        .resources
238        .insert(crate::terrain::request_system::DrapePrefetchRequests::default());
239    world
240        .resources
241        .insert(crate::terrain::request_system::DrapeRequests::default());
242    Ok(tiles
243        .into_iter()
244        .map(|coords| TargetSpec {
245            coords,
246            shapes: Vec::new(),
247            absent_sources: Vec::new(),
248        })
249        .collect())
250}

252/// The key terrain tiles are drawn by, smallest first: `1.0 - c_hat.dot(focus_unit)` on `S^2`
253/// under Globe rendering, and GL JS's planar `(dx, dy)` squared distance at the nominal level
254/// under explicit 2D Mercator rendering. With translucent drapes the order decides which skirts
255/// show through.
256pub(crate) fn draw_order_key(
257    style: &Style,
258    view_state: &ViewState,
259    dem_tile_size: f64,
260) -> impl Fn(&WorldTileCoords) -> f64 + use<> {
261    enum OrderMode {
262        Spherical(cgmath::Vector3<f64>),
263        Planar([f64; 2]),
264    }
265    let uses_globe = style.projection.as_ref().is_some_and(|p| {
266        p.projection_type
267            .uses_globe_rendering(view_state.zoom().value())
268    });
269    let mode = if uses_globe {
270        crate::render::projection::globe_camera_for_view(view_state)
271            .ok()
272            .map(|globe| OrderMode::Spherical(globe.focus_unit_sphere()))
273    } else {
274        None
275    }
276    .unwrap_or_else(|| {
277        let request = crate::render::projection::CoveringRequest::terrain(
278            view_state.zoom().value(),
279            dem_tile_size,
280        );
281        let tiles_across = 2_f64.powi(i32::from(u8::from(request.level)));
282        let world_size = crate::coords::TILE_SIZE * 2_f64.powf(view_state.zoom().value());
283        let center = view_state.camera().position();
284        OrderMode::Planar([
285            center.x / world_size * tiles_across - 0.5,
286            center.y / world_size * tiles_across - 0.5,
287        ])
288    });
289    move |tile: &WorldTileCoords| match mode {
290        OrderMode::Spherical(focus_unit) => {
291            crate::projection::tile_covering::spherical_tile_distance(*tile, focus_unit)
292        }
293        OrderMode::Planar(center) => {
294            let dx = center[0] - f64::from(tile.x);
295            let dy = center[1] - f64::from(tile.y);
296            dx * dx + dy * dy
297        }
298    }
