1//! Tile-relative sphere coordinates preserve sub-metre imagery at low altitude.
2
3use cgmath::{Matrix4, Vector3};
4
5use crate::{
6    coords::{WorldTileCoords, EXTENT},
7    render::{projection::globe_camera_for_view, view_state::ViewState},
8    terrain::resources::TerrainTileUniforms,
9};
10
11pub(super) fn set_uniforms(
12    block: &mut TerrainTileUniforms,
13    tile: WorldTileCoords,
14    view: &ViewState,
15) {
16    block.globe_origin = [0.0; 4];
17    // Coarse tiles and polar fans use the global sphere path. Their local angular span is large.
18    if u8::from(tile.z) < 8 {
19        return;
20    }
21    let Ok(camera) = globe_camera_for_view(view) else {
22        return;
23    };
24    let n = 2_f64.powi(i32::from(u8::from(tile.z)));
25    let longitude = ((f64::from(tile.x) + 0.5) / n * 2.0 - 1.0) * std::f64::consts::PI;
26    let latitude = (std::f64::consts::PI * (1.0 - 2.0 * (f64::from(tile.y) + 0.5) / n))
27        .sinh()
28        .atan();
29    let (s, c) = latitude.sin_cos();
30    let (sl, cl) = longitude.sin_cos();
31    let radius = view.body().radius_meters;
32    let east = Vector3::new(cl, 0.0, -sl);
33    let north = Vector3::new(-sl * s, c, -cl * s);
34    let up = Vector3::new(sl * c, s, cl * c);
35    let basis = Matrix4::from_cols(
36        (east / radius).extend(0.0),
37        (north / radius).extend(0.0),
38        (up / radius).extend(0.0),
39        up.extend(1.0),
40    );
41    if let Some(matrix) = (camera.wgpu_view_projection() * basis).cast::<f32>() {
42        block.globe_transform = matrix.into();
43        block.globe_origin = [
44            s as f32,
45            c as f32,
46            (std::f64::consts::TAU / (n * EXTENT)) as f32,
47            radius as f32,
48        ];
49    }
50}
