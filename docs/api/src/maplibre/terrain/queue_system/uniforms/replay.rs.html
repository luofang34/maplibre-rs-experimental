19pub(in crate::terrain::queue_system) fn replay(
20    world: &mut World,
21    style: &Style,
22    view: &ViewState,
23    queue: &crate::render::upload_queue::UploadQueue,
24) -> SystemResult {
25    let fog = terrain_fog(style, view);
26    let lighting = world
27        .resources
28        .get::<crate::render::SceneLighting>()
29        .copied()
30        .unwrap_or_default();
31    let drape_mode = world
32        .resources
33        .get::<crate::render::settings::TerrainDrapeMode>()
34        .copied()
35        .or_else(
36            || match world.resources.get::<Eventually<TerrainResources>>() {
37                Some(Initialized(terrain)) => Some(terrain.drape_mode()),
38                _ => None,
39            },
40        )
41        .unwrap_or_default();
42    let projection = view.gpu_view_projection();
43    let clear = super::super::drape_phase::background_clear_color(style);
44    let clear_color = [
45        clear.r as f32,
46        clear.g as f32,
47        clear.b as f32,
48        clear.a as f32,
49    ];
50    let globe_camera = frame_globe_camera(style, view);
51    if let Some(frame) = world.resources.get_mut::<TerrainEyeFrame>() {
52        for (coords, uniform) in &mut frame.0 {
53            uniform.transform = projection
54                .to_model_view_projection(coords.transform_for_zoom(view.zoom()))
55                .downcast()
56                .into();
57            uniform.fog_color = fog.fog_color;
58            uniform.horizon_color = fog.horizon_color;
59            uniform.fog_range = [fog.near, fog.far, fog.ground_blend, fog.horizon_blend];
60            uniform.fog_opacity[0] = fog.opacity;
61            uniform.fog_opacity[1] = f32::from(fog.globe);
62            surface_uniforms(
63                uniform,
64                *coords,
65                (clear_color, lighting, drape_mode),
66                view,
67                globe_camera.as_ref(),
68            );
69        }
70    }
71    if let (Some(frame), Some(Initialized(terrain))) = (
72        world.resources.get::<TerrainEyeFrame>(),
73        world.resources.get::<Eventually<TerrainResources>>(),
74    ) {
75        let uniforms: Vec<_> = frame.0.iter().map(|(_, uniform)| *uniform).collect();
76        terrain.write_uniforms(queue, &uniforms);
77    }
78    world.resources.insert(DrapePhase::default());
79    super::super::hide_draped_layers(world, style);
80    Ok(())
81}
