1//! Keeps a small prepared tile margin without competing with visible terrain.
2use std::{collections::HashSet, time::Duration};
3
4use crate::{
5    coords::WorldTileCoords,
6    render::{
7        frame_input::FrameInput,
8        memory_budget::{MemoryBudget, MemoryPressure},
9        tile_memory::tile_bytes,
10        view_state::{ViewState, ViewStatePadding},
11        xr::PrefetchView,
12    },
13    style::Style,
14    tcs::world::World,
15    terrain::request_system::DrapePrefetchRequests,
16};
17
18const PREFETCH_BYTES: usize = 32 << 20;
19
20#[derive(Default)]
21struct PrefetchCadence(Option<Duration>);
22
23pub(super) fn prepare(
24    style: &Style,
25    view: &ViewState,
26    world: &mut World,
27    visible: &[WorldTileCoords],
28    memory: MemoryBudget,
29) {
30    let limit = match memory.pressure() {
31        MemoryPressure::Comfortable => 8,
32        MemoryPressure::Low => 4,
33        MemoryPressure::Critical => 0,
34    };
35    if limit == 0 || !view.has_external_view() {
36        world.resources.insert(DrapePrefetchRequests::default());
37        return;
38    }
39    let time = world
40        .resources
41        .get::<FrameInput>()
42        .map_or(Duration::ZERO, |input| input.timestamp);
43    let cadence = world.resources.get_or_init_mut::<PrefetchCadence>();
44    if cadence
45        .0
46        .is_some_and(|previous| time >= previous && time - previous < Duration::from_millis(100))
47    {
48        return;
49    }
50    cadence.0 = Some(time);
51    let ahead = world
52        .resources
53        .get::<PrefetchView>()
54        .and_then(|prefetch| prefetch.view_state.as_ref());
55    let request_view = ahead.unwrap_or(view);
56    let padding = if ahead.is_some() {
57        ViewStatePadding::Tight
58    } else {
59        ViewStatePadding::Loose
60    };
61    let candidates = {
62        let region = crate::terrain::source::dem_source(style)
63            .map(|dem| {
64                crate::render::projection::terrain_region(
65                    style,
66                    request_view,
67                    world,
68                    dem.tile_size,
69                    padding,
70                )
71            })
72            .transpose()
73            .map(Option::flatten)
74            .map_err(|error| {
75                tracing::warn!(%error, "cannot prepare terrain margin");
76            })
77            .ok()
78            .flatten();
79        let tiles = region.map_or_else(Vec::new, |region| region.iter().collect());
80        super::covering::bounded_covering(
81            tiles.into_iter(),
82            memory.drape_textures_allowed().saturating_sub(8).max(1),
83        )
84    };
85    let candidates = select(candidates, visible, limit, |tile| {
86        tile_bytes(&world.tiles, tile)
87    });
88    world.resources.insert(DrapePrefetchRequests(candidates));
89}
90
91fn select(
92    candidates: Vec<WorldTileCoords>,
93    visible: &[WorldTileCoords],
94    limit: usize,
95    bytes: impl Fn(WorldTileCoords) -> usize,
96) -> Vec<WorldTileCoords> {
97    let visible: HashSet<_> = visible.iter().copied().collect();
98    let mut used = 0usize;
99    candidates
100        .into_iter()
101        .filter(|tile| !visible.contains(tile))
102        .take(limit)
103        .take_while(|tile| {
104            // Reserve space for a not-yet-decoded tile instead of treating it as free.
105            used = used.saturating_add(bytes(*tile).max(8 << 20));
106            used <= PREFETCH_BYTES
107        })
108        .collect()
109}
