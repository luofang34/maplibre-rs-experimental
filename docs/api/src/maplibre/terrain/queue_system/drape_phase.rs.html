26type TargetSlots = Vec<Option<usize>>;
27/// Instance metadata placing each redrawn target's source shapes inside its drape texture.
28///
29/// Shapes past `capacity` get no slot: a view falling back to many small child tiles can ask
30/// for more than the metadata buffer holds, and those shapes wait for their own tiles.
31#[cfg(test)]
32pub(super) fn drape_metadata(
33    specs: &[TargetSpec],
34    redraw: &[bool],
35    capacity: usize,
36    view_zoom: Zoom,
37    drape_size: u32,
38) -> (Vec<ShaderTileMetadata>, Vec<TargetSlots>) {
39    let target_sizes = vec![drape_size; specs.len()];
40    drape_metadata_sized(specs, redraw, &target_sizes, capacity, view_zoom)
41}

43pub(super) fn drape_metadata_sized(
44    specs: &[TargetSpec],
45    redraw: &[bool],
46    target_sizes: &[u32],
47    capacity: usize,
48    view_zoom: Zoom,
49) -> (Vec<ShaderTileMetadata>, Vec<TargetSlots>) {
50    let mut metadata = Vec::new();
51    let mut slots = Vec::with_capacity(specs.len());
52    let mut skipped = 0_usize;
53    for ((spec, redraw), &drape_size) in specs.iter().zip(redraw).zip(target_sizes) {
54        let texture_zoom = Zoom::new(
55            f64::from(u8::from(spec.coords.z)) + (f64::from(drape_size) / TILE_SIZE).log2(),
56        );
57        let mut target_slots = Vec::with_capacity(spec.shapes.len());
58        for shape in &spec.shapes {
59            let transform = redraw
60                .then(|| drape_transform(spec.coords, shape.source))
61                .flatten()
62                .and_then(|transform| transform.cast::<f32>());
63            let Some(transform) = transform else {
64                target_slots.push(None);
65                continue;
66            };
67            if metadata.len() >= capacity {
68                target_slots.push(None);
69                skipped += 1;
70                continue;
71            }
72            target_slots.push(Some(metadata.len()));
73            metadata.push(ShaderTileMetadata {
74                transform: transform.into(),
75                zoom_factor: texture_zoom.scale_to_tile(&shape.source) as f32,
76                viewport_width: drape_size as f32,
77                viewport_height: drape_size as f32,
78                tile_mercator_coords: tile_mercator_coordinates(
79                    shape.source.into_tile(TileAddressingScheme::XYZ),
80                )
81                .into(),
82                clip_antimeridian: 0,
83                line_units_per_pixel: 8.0 * texture_zoom.scale_to_tile(&shape.source) as f32,
84                // A raster tile reaches a hair past its edges; a vector tile's lines scale with the
85                // texture's zoom.
86                line_width_scale: if shape.raster_layers.is_empty() {
87                    2.0_f64.powf(texture_zoom.value() - view_zoom.value()) as f32
88                } else {
89                    (1.0 + EDGE_OVERLAP) as f32
90                },
91                clip_region: crate::render::shaders::WHOLE_TILE,
92                _pad: [0; 2],
93            });
94        }
95        slots.push(target_slots);
96    }
97    if skipped > 0 {
98        tracing::warn!(
99            skipped,
100            capacity,
101            "drape shapes exceed the metadata buffer; some tiles drape without them this frame"
102        );
103    }
104    (metadata, slots)
105}

107pub(super) fn build_drape_phase(
108    specs: &[TargetSpec],
109    redraw: &[bool],
110    slots: &[TargetSlots],
111    ranges: &[std::ops::Range<wgpu::BufferAddress>],
112    zoom: Zoom,
113    clear_color: wgpu::Color,
114) -> DrapePhase {
115    let mut phase = DrapePhase::default();
116    for ((spec, redraw), target_slots) in specs.iter().zip(redraw).zip(slots) {
117        if !redraw {
118            continue;
119        }
120        let mut target = DrapeTarget {
121            coords: spec.coords,
122            clear_color,
123            masks: Vec::new(),
124            layers: Vec::new(),
125        };
126        let mut stencil_ids: std::collections::HashMap<crate::coords::WorldTileCoords, u8> =
127            std::collections::HashMap::with_capacity(spec.shapes.len());
128        let mut next_stencil = 0_u8;
129        for (shape, slot) in spec.shapes.iter().zip(target_slots) {
130            let Some(range) = slot.and_then(|index| ranges.get(index)) else {
131                continue;
132            };
133            let mut source_shape = TileShape::with_buffer_range(shape.source, zoom, range.clone());
134            let stencil_ref = *stencil_ids.entry(shape.source).or_insert_with(|| {
135                if next_stencil < 124 {
136                    let assigned = next_stencil;
137                    next_stencil += 1;
138                    assigned
139                } else {
140                    shape.source.stencil_reference_value_3d()
141                }
142            });
143            source_shape.set_stencil_reference(stencil_ref);
144            let shape_data = source_shape.draw_data();
145            target.masks.push(TileMaskItem {
146                draw_function: &DrawState::<TileMaskItem, DrawMasks>::INSTANCE,
147                source_shape: shape_data,
148                generate_borders: false,
149                projection: ProjectionBinding::Flat,
150                is_vector: shape.is_vector,
151            });
152            for layer in &shape.vector_layers {
153                let draw_function: &'static dyn Draw<LayerItem> = if layer.is_line {
154                    &DrawState::<LayerItem, DrawLineTiles>::INSTANCE
155                } else {
156                    &DrawState::<LayerItem, DrawVectorTiles>::INSTANCE
157                };
158                target.layers.push(LayerItem {
159                    draw_function,
160                    index: layer.index,
161                    generate_borders: false,
162                    style_layer: layer.id,
163                    entry_slot: crate::render::render_phase::UNSET_ENTRY_SLOT,
164                    tile: Tile {
165                        coords: layer.coords,
166                    },
167                    source_shape: shape_data,
168                    projection: ProjectionBinding::Flat,
169                    mask_group: MaskGroup::Vector,
170                    run: None,
171                });
172            }
173            #[cfg(feature = "raster")]
174            for (id, index, dem) in &shape.raster_layers {
175                let draw_function: &'static dyn Draw<LayerItem> = if *dem {
176                    &DrawState::<LayerItem, DrawDemTiles>::INSTANCE
177                } else {
178                    &DrawState::<LayerItem, DrawRasterTiles>::INSTANCE
179                };
180                target.layers.push(LayerItem {
181                    draw_function,
182                    index: *index,
183                    generate_borders: false,
184                    style_layer: crate::style::StyleLayerId::intern(id),
185                    entry_slot: crate::render::render_phase::UNSET_ENTRY_SLOT,
186                    tile: Tile {
187                        coords: shape.source,
188                    },
189                    source_shape: shape_data,
190                    projection: ProjectionBinding::Flat,
191                    mask_group: MaskGroup::Mutating,
192                    run: None,
193                });
194            }
195        }
196        target.masks.sort_by_key(PhaseItem::sort_key);
197        target.layers.sort_by_key(|item| item.index);
198        phase.targets.push(target);
199    }
200    phase
201}

203/// Color the drape textures start from: the constant background paint, a neutral cartographic
204/// terrain tone when terrain is active without a background layer, or transparent.
205pub(super) fn background_clear_color(style: &Style) -> wgpu::Color {
206    style
207        .layers
208        .iter()
209        .find(|layer| layer.type_ == "background" && !layer.is_hidden())
210        .and_then(|layer| layer.paint.as_ref()?.get_color())
211        .map(|color| wgpu::Color {
212            r: f64::from(color.color.r),
213            g: f64::from(color.color.g),
214            b: f64::from(color.color.b),
215            a: f64::from(color.alpha),
216        })
217        .unwrap_or_else(|| {
218            if style.terrain.is_some() {
219                wgpu::Color {
220                    r: 0.91,
221                    g: 0.89,
222                    b: 0.85,
223                    a: 1.0,
224                }
225            } else {
226                wgpu::Color::TRANSPARENT
227            }
228        })
229}

231pub(super) fn encode_drapes(
232    specs: &[TargetSpec],
233    redraw: &[bool],
234    clear_color: wgpu::Color,
235    world: &mut World,
236    zoom: Zoom,
237    queue: &crate::render::upload_queue::UploadQueue,
238) -> Result<DrapePhase, SystemError> {
239    let capacity = match world.resources.get::<Eventually<WgpuTileViewPattern>>() {
240        Some(Initialized(pattern)) => pattern.remaining_metadata_capacity(),
241        _ => return Err(SystemError::Dependencies),
242    };
243
244    if !redraw.iter().any(|&r| r) {
245        if !matches!(
246            world.resources.get::<Eventually<TerrainResources>>(),
247            Some(Initialized(_))
248        ) {
249            return Err(SystemError::Dependencies);
250        }
251        return Ok(DrapePhase::default());
252    }
253
254    let target_sizes: Vec<u32> = match world.resources.get::<Eventually<TerrainResources>>() {
255        Some(Initialized(terrain)) => {
256            let fallback = terrain.drape_size();
257            specs
258                .iter()
259                .map(|spec| terrain.drape_edge_px(spec.coords).unwrap_or(fallback))
260                .collect()
261        }
262        _ => return Err(SystemError::Dependencies),
263    };
264    let (metadata, slots) = drape_metadata_sized(specs, redraw, &target_sizes, capacity, zoom);
265    let ranges = {
266        let Some(Initialized(pattern)) =
267            world.resources.get_mut::<Eventually<WgpuTileViewPattern>>()
268        else {
269            return Err(SystemError::Dependencies);
270        };
271        pattern
272            .upload_extra_metadata(queue, &metadata)
273            .map_err(|error| {
274                tracing::error!(%error, "unable to upload drape metadata");
275                SystemError::Setup
276            })?
277    };
278    Ok(build_drape_phase(
279        specs,
280        redraw,
281        &slots,
282        &ranges,
283        zoom,
284        clear_color,
285    ))
286}
