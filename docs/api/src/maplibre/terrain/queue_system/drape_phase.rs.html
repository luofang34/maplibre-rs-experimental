24type TargetSlots = Vec<Option<usize>>;
25/// Instance metadata placing each redrawn target's source shapes inside its drape texture.
26///
27/// Shapes past `capacity` get no slot: a view falling back to many small child tiles can ask
28/// for more than the metadata buffer holds, and those shapes wait for their own tiles.
29pub(super) fn drape_metadata(
30    specs: &[TargetSpec],
31    redraw: &[bool],
32    capacity: usize,
33    view_zoom: Zoom,
34    drape_size: u32,
35) -> (Vec<ShaderTileMetadata>, Vec<TargetSlots>) {
36    let mut metadata = Vec::new();
37    let mut slots = Vec::with_capacity(specs.len());
38    let mut skipped = 0_usize;
39    for (spec, redraw) in specs.iter().zip(redraw) {
40        let texture_zoom = Zoom::new(
41            f64::from(u8::from(spec.coords.z)) + (f64::from(drape_size) / TILE_SIZE).log2(),
42        );
43        let mut target_slots = Vec::with_capacity(spec.shapes.len());
44        for shape in &spec.shapes {
45            let transform = redraw
46                .then(|| drape_transform(spec.coords, shape.source))
47                .flatten()
48                .and_then(|transform| transform.cast::<f32>());
49            let Some(transform) = transform else {
50                target_slots.push(None);
51                continue;
52            };
53            if metadata.len() >= capacity {
54                target_slots.push(None);
55                skipped += 1;
56                continue;
57            }
58            target_slots.push(Some(metadata.len()));
59            metadata.push(ShaderTileMetadata {
60                transform: transform.into(),
61                zoom_factor: texture_zoom.scale_to_tile(&shape.source) as f32,
62                viewport_width: drape_size as f32,
63                viewport_height: drape_size as f32,
64                tile_mercator_coords: tile_mercator_coordinates(
65                    shape.source.into_tile(TileAddressingScheme::XYZ),
66                )
67                .into(),
68                clip_antimeridian: 0,
69                line_units_per_pixel: 8.0 * texture_zoom.scale_to_tile(&shape.source) as f32,
70                // A raster tile reaches a hair past its edges; a vector tile's lines scale with the
71                // texture's zoom.
72                line_width_scale: if shape.raster_layers.is_empty() {
73                    2.0_f64.powf(texture_zoom.value() - view_zoom.value()) as f32
74                } else {
75                    (1.0 + EDGE_OVERLAP) as f32
76                },
77                clip_region: crate::render::shaders::WHOLE_TILE,
78            });
79        }
80        slots.push(target_slots);
81    }
82    if skipped > 0 {
83        tracing::warn!(
84            skipped,
85            capacity,
86            "drape shapes exceed the metadata buffer; some tiles drape without them this frame"
87        );
88    }
89    (metadata, slots)
90}

92pub(super) fn build_drape_phase(
93    specs: &[TargetSpec],
94    redraw: &[bool],
95    slots: &[TargetSlots],
96    ranges: &[std::ops::Range<wgpu::BufferAddress>],
97    zoom: Zoom,
98    clear_color: wgpu::Color,
99) -> DrapePhase {
100    let mut phase = DrapePhase::default();
101    for ((spec, redraw), target_slots) in specs.iter().zip(redraw).zip(slots) {
102        if !redraw {
103            continue;
104        }
105        let mut target = DrapeTarget {
106            coords: spec.coords,
107            clear_color,
108            masks: Vec::new(),
109            layers: Vec::new(),
110        };
111        for (shape, slot) in spec.shapes.iter().zip(target_slots) {
112            let Some(range) = slot.and_then(|index| ranges.get(index)) else {
113                continue;
114            };
115            let source_shape = TileShape::with_buffer_range(shape.source, zoom, range.clone());
116            target.masks.push(TileMaskItem {
117                draw_function: Box::new(DrawState::<TileMaskItem, DrawMasks>::new()),
118                source_shape: source_shape.clone(),
119                generate_borders: false,
120                projection: ProjectionBinding::Flat,
121            });
122            for layer in &shape.vector_layers {
123                let draw_function: Box<dyn Draw<LayerItem>> = if layer.is_line {
124                    Box::new(DrawState::<LayerItem, DrawLineTiles>::new())
125                } else {
126                    Box::new(DrawState::<LayerItem, DrawVectorTiles>::new())
127                };
128                target.layers.push(LayerItem {
129                    draw_function,
130                    index: layer.index,
131                    generate_borders: false,
132                    style_layer: layer.id.clone(),
133                    tile: Tile {
134                        coords: layer.coords,
135                    },
136                    source_shape: source_shape.clone(),
137                    projection: ProjectionBinding::Flat,
138                    run: None,
139                });
140            }
141            #[cfg(feature = "raster")]
142            for (id, index, dem) in &shape.raster_layers {
143                let draw_function: Box<dyn Draw<LayerItem>> = if *dem {
144                    Box::new(DrawState::<LayerItem, DrawDemTiles>::new())
145                } else {
146                    Box::new(DrawState::<LayerItem, DrawRasterTiles>::new())
147                };
148                target.layers.push(LayerItem {
149                    draw_function,
150                    index: *index,
151                    generate_borders: false,
152                    style_layer: id.clone(),
153                    tile: Tile {
154                        coords: shape.source,
155                    },
156                    source_shape: source_shape.clone(),
157                    projection: ProjectionBinding::Flat,
158                    run: None,
159                });
160            }
161        }
162        target.layers.sort_by_key(|item| item.index);
163        phase.targets.push(target);
164    }
165    phase
166}

168/// Color the drape textures start from: the constant background paint, a neutral cartographic
169/// terrain tone when terrain is active without a background layer, or transparent.
170pub(super) fn background_clear_color(style: &Style) -> wgpu::Color {
171    style
172        .layers
173        .iter()
174        .find(|layer| layer.type_ == "background" && !layer.is_hidden())
175        .and_then(|layer| layer.paint.as_ref()?.get_color())
176        .map(|color| wgpu::Color {
177            r: f64::from(color.color.r),
178            g: f64::from(color.color.g),
179            b: f64::from(color.color.b),
180            a: f64::from(color.alpha),
181        })
182        .unwrap_or_else(|| {
183            if style.terrain.is_some() {
184                wgpu::Color {
185                    r: 0.91,
186                    g: 0.89,
187                    b: 0.85,
188                    a: 1.0,
189                }
190            } else {
191                wgpu::Color::TRANSPARENT
192            }
193        })
194}

196pub(super) fn encode_drapes(
197    specs: &[TargetSpec],
198    redraw: &[bool],
199    clear_color: wgpu::Color,
200    world: &mut World,
201    zoom: Zoom,
202    queue: &crate::render::upload_queue::UploadQueue,
203) -> Result<DrapePhase, SystemError> {
204    let capacity = match world.resources.get::<Eventually<WgpuTileViewPattern>>() {
205        Some(Initialized(pattern)) => pattern.remaining_metadata_capacity(),
206        _ => return Err(SystemError::Dependencies),
207    };
208
209    let drape_size = match world.resources.get::<Eventually<TerrainResources>>() {
210        Some(Initialized(terrain)) => terrain.drape_size(),
211        _ => return Err(SystemError::Dependencies),
212    };
213    let (metadata, slots) = drape_metadata(specs, redraw, capacity, zoom, drape_size);
214    let ranges = {
215        let Some(Initialized(pattern)) =
216            world.resources.get_mut::<Eventually<WgpuTileViewPattern>>()
217        else {
218            return Err(SystemError::Dependencies);
219        };
220        pattern
221            .upload_extra_metadata(queue, &metadata)
222            .map_err(|error| {
223                tracing::error!(%error, "unable to upload drape metadata");
224                SystemError::Setup
225            })?
226    };
227    Ok(build_drape_phase(
228        specs,
229        redraw,
230        &slots,
231        &ranges,
232        zoom,
233        clear_color,
234    ))
235}
