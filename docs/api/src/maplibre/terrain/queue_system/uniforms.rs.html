25pub(super) fn replay(
26    world: &mut World,
27    style: &Style,
28    view: &ViewState,
29    queue: &crate::render::upload_queue::UploadQueue,
30) -> SystemResult {
31    let fog = terrain_fog(style, view);
32    let lighting = crate::celestial::lighting(world);
33    let projection = view.gpu_view_projection();
34    if let Some(frame) = world.resources.get_mut::<TerrainEyeFrame>() {
35        for (coords, uniform) in &mut frame.0 {
36            uniform.transform = projection
37                .to_model_view_projection(coords.transform_for_zoom(view.zoom()))
38                .downcast()
39                .into();
40            uniform.fog_color = fog.fog_color;
41            uniform.horizon_color = fog.horizon_color;
42            uniform.fog_range = [fog.near, fog.far, fog.ground_blend, fog.horizon_blend];
43            uniform.fog_opacity[0] = fog.opacity;
44            uniform.fog_opacity[1] = f32::from(fog.globe);
45            surface_uniforms(uniform, *coords, (style, lighting), view);
46        }
47    }
48    if let (Some(frame), Some(Initialized(terrain))) = (
49        world.resources.get::<TerrainEyeFrame>(),
50        world.resources.get::<Eventually<TerrainResources>>(),
51    ) {
52        let uniforms: Vec<_> = frame.0.iter().map(|(_, uniform)| *uniform).collect();
53        terrain.write_uniforms(queue, &uniforms);
54    }
55    world.resources.insert(DrapePhase::default());
56    super::hide_draped_layers(world, style);
57    Ok(())
58}
59/// The nearest uploaded DEM tile at or above the DEM zoom of a view tile.
60pub(super) fn loaded_dem_tile(
61    coords: WorldTileCoords,
62    dem: &DemSource,
63    terrain: &TerrainResources,
64) -> Option<WorldTileCoords> {
65    let mut current = dem_tile_coords(coords, dem.minzoom, dem.maxzoom)?;
66    loop {
67        if terrain.has_dem_texture(current) {
68            return Some(current);
69        }
70        current = current.get_parent()?;
71    }
72}

74/// Maps tile coordinates in `0..EXTENT` of `coords` to unit coordinates inside its DEM tile.
75fn dem_matrix(coords: WorldTileCoords, dem: WorldTileCoords) -> Matrix4<f64> {
76    let delta = i32::from(u8::from(coords.z)) - i32::from(u8::from(dem.z));
77    let scale = 2_f64.powi(delta);
78    let origin_x = f64::from(coords.x - (dem.x << delta)) / scale;
79    let origin_y = f64::from(coords.y - (dem.y << delta)) / scale;
80    Matrix4::from_translation(Vector3::new(origin_x, origin_y, 0.0))
81        * Matrix4::from_nonuniform_scale(1.0 / (EXTENT * scale), 1.0 / (EXTENT * scale), 1.0)
82}

84/// The ancestors of a tile, nearest first, that hold a drape texture.
85pub(super) fn present_ancestors(
86    coords: WorldTileCoords,
87    terrain: &TerrainResources,
88) -> Vec<WorldTileCoords> {
89    let mut ancestors = Vec::new();
90    let mut current = coords;
91    while let Some(parent) = current.get_parent() {
92        if terrain.drape_texture(parent).is_some() {
93            ancestors.push(parent);
94        }
95        current = parent;
96    }
97    ancestors
98}

100/// Maps unit coordinates of `coords` into the unit coordinates of its ancestor `source`.
101fn drape_matrix(coords: WorldTileCoords, source: WorldTileCoords) -> Matrix4<f64> {
102    let delta = i32::from(u8::from(coords.z)) - i32::from(u8::from(source.z));
103    let scale = 2_f64.powi(delta);
104    let origin_x = f64::from(coords.x - (source.x << delta)) / scale;
105    let origin_y = f64::from(coords.y - (source.y << delta)) / scale;
106    Matrix4::from_translation(Vector3::new(origin_x, origin_y, 0.0))
107        * Matrix4::from_nonuniform_scale(1.0 / scale, 1.0 / scale, 1.0)
108}

118pub(super) fn tile_uniforms(
119    coords: WorldTileCoords,
120    textures: TileTextures,
121    terrain: &TerrainResources,
122    dem: &DemSource,
123    transform: &Matrix4<f32>,
124    skirt_length: f32,
125    fog: &TerrainFog,
126) -> Option<TerrainTileUniforms> {
127    let (dem_matrix, dem_unpack, dem_dim) = match textures.dem {
128        Some(dem_coords) => (
129            dem_matrix(coords, dem_coords).cast::<f32>()?,
130            dem.unpack.map(|value| value as f32),
131            (terrain
132                .dem_texture(Some(dem_coords))
133                .size
134                .width
135                .saturating_sub(4))
136            .max(1) as f32,
137        ),
138        None => (Matrix4::identity(), [0.0; 4], 1.0),
139    };
140    Some(TerrainTileUniforms {
141        transform: (*transform).into(),
142        globe_transform: Matrix4::identity().into(),
143        globe_origin: [0.0; 4],
144        dem_matrix: dem_matrix.into(),
145        drape_matrix: textures
146            .drape
147            .map_or_else(Matrix4::identity, |source| drape_matrix(coords, source))
148            .cast::<f32>()?
149            .into(),
150        tile_mercator_coords: tile_mercator_coordinates(
151            coords.into_tile(TileAddressingScheme::XYZ),
152        )
153        .into(),
154        dem_unpack,
155        dem_dim,
156        exaggeration: dem.exaggeration,
157        skirt_length,
158        relief_strength: 0.0,
159        fog_color: fog.fog_color,
160        horizon_color: fog.horizon_color,
161        fog_range: [fog.near, fog.far, fog.ground_blend, fog.horizon_blend],
162        fog_opacity: [
163            fog.opacity,
164            f32::from(fog.globe),
165            f32::from(textures.drape.is_none()),
166            0.0,
167        ],
168        edge_heights: Default::default(),
169        surface_color: [0.0; 4],
170        fog_position: [0.0; 4],
171        daylight: [0.0; 4],
172        twilight: [0.0; 4],
173    })
174}

176/// The fog of the frame from the style's sky and the view, as GL JS `terrainUniformValues`.
177pub(super) fn terrain_fog(style: &Style, view_state: &ViewState) -> TerrainFog {
178    let Some(sky) = &style.sky else {
179        return TerrainFog::default();
180    };
181    let colors = sky.colors_at(view_state.style_zoom().value());
182    let (near, far) = view_state.fog_depth_range();
183    let meters = view_state.eye_fog_meters_per_pixel().unwrap_or(1.0);
184    let (near, far) = (near * meters, far * meters);
185    let globe = style.projection.as_ref().is_some_and(|specification| {
186        specification
187            .projection_type
188            .uses_globe_rendering(view_state.zoom().value())
189    });
190    TerrainFog {
191        fog_color: colors.fog,
192        horizon_color: colors.horizon,
193        near: near as f32,
194        far: far as f32,
195        ground_blend: colors.fog_ground_blend,
196        horizon_blend: colors.horizon_fog_blend,
197        opacity: view_state.fog_opacity(),
198        globe,
199    }
200}

227pub(super) fn prepare_tiles(
228    targets: (&[TargetSpec], &[Option<WorldTileCoords>]),
229    style: &Style,
230    view_state: &ViewState,
231    terrain: &TerrainResources,
232    (dem, skirts, lighting): (
233        &DemSource,
234        crate::terrain::TerrainSkirts,
235        crate::celestial::CelestialLighting,
236    ),
237) -> PreparedTerrain {
238    let (specs, drape_sources) = targets;
239    let zoom = view_state.zoom();
240    let gpu_view_projection = view_state.gpu_view_projection();
241    let fog = terrain_fog(style, view_state);
242    let skirt_length = match skirts {
243        crate::terrain::TerrainSkirts::Auto => {
244            view_state.body().circumference_meters()
245                / 2_f64.powf(zoom.value().max(0.0))
246                / super::SKIRT_DIVISOR
247        }
248        // A skirt of no length folds onto the tile edge and covers no pixel.
249        crate::terrain::TerrainSkirts::None => 0.0,
250    };
251    let mut uniforms = Vec::with_capacity(specs.len());
252    let mut sources = Vec::with_capacity(specs.len());
253    for (spec, drape_source) in specs.iter().zip(drape_sources) {
254        let drape_source = *drape_source;
255        let dem_coords = loaded_dem_tile(spec.coords, dem, terrain);
256        let Some(mut block) = tile_uniforms(
257            spec.coords,
258            TileTextures {
259                dem: dem_coords,
260                drape: drape_source,
261            },
262            terrain,
263            dem,
264            &gpu_view_projection
265                .to_model_view_projection(spec.coords.transform_for_zoom(zoom))
266                .downcast(),
267            skirt_length as f32,
268            &fog,
269        ) else {
270            continue;
271        };
272        surface_uniforms(&mut block, spec.coords, (style, lighting), view_state);
273        uniforms.push(block);
274        sources.push((dem_coords, spec.coords, drape_source));
275    }
276    PreparedTerrain { uniforms, sources }
277}

279/// Adds a draw of each surface tile for every copy of the world that shows it, shifted over by
280/// the copy's whole worlds. Returns the draws' sources and the coordinates, beyond the grid for a
281/// copy, each is placed at.
282pub(super) fn with_world_copies(
283    uniforms: &mut Vec<TerrainTileUniforms>,
284    sources: Vec<TileSources>,
285    copies: Option<&std::collections::HashMap<WorldTileCoords, Vec<i32>>>,
286    view: &ViewState,
287) -> (Vec<TileSources>, Vec<WorldTileCoords>) {
288    let mut placed: Vec<WorldTileCoords> = sources.iter().map(|(_, coords, _)| *coords).collect();
289    let mut all = sources;
290    let Some(copies) = copies else {
291        return (all, placed);
292    };
293    let projection = view.gpu_view_projection();
294    let zoom = view.zoom();
295    for index in 0..all.len() {
296        let coords = all[index].1;
297        let Some(wraps) = copies.get(&coords) else {
298            continue;
299        };
300        let bounds = crate::coords::ZOOM_BOUNDS[usize::from(u8::from(coords.z))] as i32;
301        for wrap in wraps {
302            let shifted = WorldTileCoords {
303                x: coords.x + wrap * bounds,
304                ..coords
305            };
306            let mut block = uniforms[index];
307            block.transform = projection
308                .to_model_view_projection(shifted.transform_for_zoom(zoom))
309                .downcast()
310                .into();
311            block.tile_mercator_coords[0] += *wrap as f32;
312            uniforms.push(block);
313            all.push(all[index]);
314            placed.push(shifted);
315        }
316    }
317    (all, placed)
318}

320fn surface_uniforms(
321    block: &mut TerrainTileUniforms,
322    coords: WorldTileCoords,
323    (style, lighting): (&Style, crate::celestial::CelestialLighting),
324    view: &ViewState,
325) {
326    globe::set_uniforms(block, coords, view);
327    [block.daylight, block.twilight] = lighting.daylight_uniforms();
328    let color = super::background_clear_color(style);
329    block.surface_color = [
330        color.r as f32,
331        color.g as f32,
332        color.b as f32,
333        color.a as f32,
334    ];
335    block.relief_strength = if view.has_external_view() { 0.45 } else { 0.0 };
336    block.fog_opacity[3] = f32::from(view.has_external_view());
337    block.fog_position = view.eye_fog_position(coords).unwrap_or([0.0; 4]);
338}
