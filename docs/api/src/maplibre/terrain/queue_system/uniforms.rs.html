26pub(super) fn replay(
27    world: &mut World,
28    style: &Style,
29    view: &ViewState,
30    queue: &crate::render::upload_queue::UploadQueue,
31) -> SystemResult {
32    let fog = terrain_fog(style, view);
33    let lighting = world
34        .resources
35        .get::<crate::render::SceneLighting>()
36        .copied()
37        .unwrap_or_default();
38    let drape_mode = world
39        .resources
40        .get::<crate::render::settings::TerrainDrapeMode>()
41        .copied()
42        .or_else(
43            || match world.resources.get::<Eventually<TerrainResources>>() {
44                Some(Initialized(terrain)) => Some(terrain.drape_mode()),
45                _ => None,
46            },
47        )
48        .unwrap_or_default();
49    let projection = view.gpu_view_projection();
50    if let Some(frame) = world.resources.get_mut::<TerrainEyeFrame>() {
51        for (coords, uniform) in &mut frame.0 {
52            uniform.transform = projection
53                .to_model_view_projection(coords.transform_for_zoom(view.zoom()))
54                .downcast()
55                .into();
56            uniform.fog_color = fog.fog_color;
57            uniform.horizon_color = fog.horizon_color;
58            uniform.fog_range = [fog.near, fog.far, fog.ground_blend, fog.horizon_blend];
59            uniform.fog_opacity[0] = fog.opacity;
60            uniform.fog_opacity[1] = f32::from(fog.globe);
61            surface_uniforms(uniform, *coords, (style, lighting, drape_mode), view);
62        }
63    }
64    if let (Some(frame), Some(Initialized(terrain))) = (
65        world.resources.get::<TerrainEyeFrame>(),
66        world.resources.get::<Eventually<TerrainResources>>(),
67    ) {
68        let uniforms: Vec<_> = frame.0.iter().map(|(_, uniform)| *uniform).collect();
69        terrain.write_uniforms(queue, &uniforms);
70    }
71    world.resources.insert(DrapePhase::default());
72    super::hide_draped_layers(world, style);
73    Ok(())
74}
75/// The nearest uploaded DEM tile at or above the DEM zoom of a view tile.
76pub(super) fn loaded_dem_tile(
77    coords: WorldTileCoords,
78    dem: &DemSource,
79    terrain: &TerrainResources,
80) -> Option<WorldTileCoords> {
81    let mut current = dem_tile_coords(coords, dem.minzoom, dem.maxzoom)?;
82    loop {
83        if terrain.has_dem_texture(current) {
84            return Some(current);
85        }
86        current = current.get_parent()?;
87    }
88}

90/// Maps tile coordinates in `0..EXTENT` of `coords` to unit coordinates inside its DEM tile.
91fn dem_matrix(coords: WorldTileCoords, dem: WorldTileCoords) -> Matrix4<f64> {
92    let delta = i32::from(u8::from(coords.z)) - i32::from(u8::from(dem.z));
93    let scale = 2_f64.powi(delta);
94    let origin_x = f64::from(coords.x - (dem.x << delta)) / scale;
95    let origin_y = f64::from(coords.y - (dem.y << delta)) / scale;
96    Matrix4::from_translation(Vector3::new(origin_x, origin_y, 0.0))
97        * Matrix4::from_nonuniform_scale(1.0 / (EXTENT * scale), 1.0 / (EXTENT * scale), 1.0)
98}

100/// The ancestors of a tile, nearest first, that hold a drape texture.
101pub(super) fn present_ancestors(
102    coords: WorldTileCoords,
103    terrain: &TerrainResources,
104) -> Vec<WorldTileCoords> {
105    let mut ancestors = Vec::new();
106    let mut current = coords;
107    while let Some(parent) = current.get_parent() {
108        if terrain.drape_texture(parent).is_some() {
109            ancestors.push(parent);
110        }
111        current = parent;
112    }
113    ancestors
114}

116/// Maps unit coordinates of `coords` into the unit coordinates of its ancestor `source`.
117fn drape_matrix(coords: WorldTileCoords, source: WorldTileCoords) -> Matrix4<f64> {
118    let delta = i32::from(u8::from(coords.z)) - i32::from(u8::from(source.z));
119    let scale = 2_f64.powi(delta);
120    let origin_x = f64::from(coords.x - (source.x << delta)) / scale;
121    let origin_y = f64::from(coords.y - (source.y << delta)) / scale;
122    Matrix4::from_translation(Vector3::new(origin_x, origin_y, 0.0))
123        * Matrix4::from_nonuniform_scale(1.0 / scale, 1.0 / scale, 1.0)
124}

134pub(super) fn tile_uniforms(
135    coords: WorldTileCoords,
136    textures: TileTextures,
137    terrain: &TerrainResources,
138    dem: &DemSource,
139    transform: &Matrix4<f32>,
140    skirt_length: f32,
141    fog: &TerrainFog,
142) -> Option<TerrainTileUniforms> {
143    let (dem_matrix, dem_unpack, dem_dim) = match textures.dem {
144        Some(dem_coords) => (
145            dem_matrix(coords, dem_coords).cast::<f32>()?,
146            dem.unpack.map(|value| value as f32),
147            (terrain
148                .dem_texture(Some(dem_coords))
149                .size
150                .width
151                .saturating_sub(4))
152            .max(1) as f32,
153        ),
154        None => (Matrix4::identity(), [0.0; 4], 1.0),
155    };
156    Some(TerrainTileUniforms {
157        transform: (*transform).into(),
158        globe_transform: Matrix4::identity().into(),
159        globe_origin: [0.0; 4],
160        dem_matrix: dem_matrix.into(),
161        drape_matrix: textures
162            .drape
163            .map_or_else(Matrix4::identity, |source| drape_matrix(coords, source))
164            .cast::<f32>()?
165            .into(),
166        tile_mercator_coords: tile_mercator_coordinates(
167            coords.into_tile(TileAddressingScheme::XYZ),
168        )
169        .into(),
170        dem_unpack,
171        dem_dim,
172        exaggeration: dem.exaggeration,
173        skirt_length,
174        relief_strength: 0.0,
175        fog_color: fog.fog_color,
176        horizon_color: fog.horizon_color,
177        fog_range: [fog.near, fog.far, fog.ground_blend, fog.horizon_blend],
178        fog_opacity: [
179            fog.opacity,
180            f32::from(fog.globe),
181            f32::from(textures.drape.is_none()),
182            0.0,
183        ],
184        edge_index: [0; 4],
185        surface_color: [0.0; 4],
186        fog_position: [0.0; 4],
187        daylight: [0.0; 4],
188        twilight: [0.0; 4],
189        moonlight: [0.0; 4],
190        procedural_relief: [0.0; 4],
191        procedural_grid: [0.0; 4],
192    })
193}

195/// The fog of the frame from the style's sky and the view, as GL JS `terrainUniformValues`,
196/// smoothly blended across the Mercator-to-globe transition and faded with altitude and
197/// `atmosphere-blend` on the globe.
198pub(crate) fn terrain_fog(style: &Style, view_state: &ViewState) -> TerrainFog {
199    use cgmath::InnerSpace;
200
201    let Some(sky) = &style.sky else {
202        return TerrainFog::default();
203    };
204    let zoom = view_state.zoom().value();
205    let colors = sky.colors_at(view_state.style_zoom().value());
206    let (near, far) = view_state.fog_depth_range();
207    let meters = view_state.eye_fog_meters_per_pixel().unwrap_or(1.0);
208    let (near, mut far) = (near * meters, far * meters);
209    let transition = style.projection.as_ref().map_or(0.0, |specification| {
210        specification.projection_type.globe_transition(zoom)
211    });
212    let external_or_free = view_state.has_external_view()
213        || view_state.navigation_mode() == crate::render::view_state::NavigationMode::FreeGlobe;
214    let mut opacity = view_state.fog_opacity();
215    let globe = if external_or_free {
216        transition > 0.0
217    } else if transition > 0.0 {
218        let globe_fade = crate::render::projection::globe_camera_for_view(view_state)
219            .ok()
220            .map_or(0.0, |camera| {
221                let distance_radii = camera.camera_position().magnitude();
222                let radius_px = camera.globe_radius_pixels();
223                let horizon_dist_px =
224                    (distance_radii * distance_radii - 1.0).max(0.0).sqrt() * radius_px;
225                if horizon_dist_px > near {
226                    let t = f64::from(transition);
227                    far = far.min(far * (1.0 - t) + horizon_dist_px * t);
228                }
229                let atm_h = 100_000.0 / camera.body().radius_meters;
230                let alt = distance_radii - 1.0;
231                let u = ((alt - atm_h) / (9.0 * atm_h)).clamp(0.0, 1.0) as f32;
232                let inside_atmosphere = 1.0 - u * u * (3.0 - 2.0 * u);
233                inside_atmosphere * (1.0 - sky.atmosphere_blend_at_zoom(zoom))
234            });
235        opacity *= 1.0 - transition + transition * globe_fade;
236        opacity <= 0.0
237    } else {
238        false
239    };
240    TerrainFog {
241        fog_color: colors.fog,
242        horizon_color: colors.horizon,
243        near: near as f32,
244        far: far as f32,
245        ground_blend: colors.fog_ground_blend,
246        horizon_blend: colors.horizon_fog_blend,
247        opacity,
248        globe,
249    }
250}

273/// Captures style, view, DEM and host lighting inputs for preparing terrain tile uniforms.
274pub(super) fn surface_context<'a>(
275    world: &World,
276    style: &'a Style,
277    view_state: &'a ViewState,
278    dem: &'a DemSource,
279) -> SurfaceContext<'a> {
280    let skirts = world.resources.get::<crate::terrain::TerrainSkirts>();
281    let poles = world
282        .resources
283        .get::<crate::terrain::PolarPoleElevations>()
284        .copied()
285        .unwrap_or_default();
286    let lighting = world
287        .resources
288        .get::<crate::render::SceneLighting>()
289        .copied()
290        .unwrap_or_default();
291    let paint_zoom =
292        crate::vector::upload_system::paint::current_zoom(world, view_state.style_zoom().value());
293    SurfaceContext {
294        style,
295        view_state,
296        dem,
297        skirts: skirts.copied().unwrap_or_default(),
298        poles,
299        lighting,
300        paint_zoom,
301    }

304pub(super) fn prepare_tiles(
305    targets: (&[TargetSpec], &[Option<WorldTileCoords>]),
306    terrain: &TerrainResources,
307    tiles: &crate::tcs::tiles::Tiles,
308    ctx: SurfaceContext<'_>,
309) -> PreparedTerrain {
310    let SurfaceContext {
311        style,
312        view_state,
313        dem,
314        skirts,
315        poles,
316        lighting,
317        paint_zoom,
318    } = ctx;
319    let (specs, drape_sources) = targets;
320    let zoom = view_state.zoom();
321    let gpu_view_projection = view_state.gpu_view_projection();
322    let fog = terrain_fog(style, view_state);
323    let skirt_length = match skirts {
324        crate::terrain::TerrainSkirts::Auto => {
325            view_state.body().circumference_meters()
326                / 2_f64.powf(zoom.value().max(0.0))
327                / super::SKIRT_DIVISOR
328        }
329        // A skirt of no length folds onto the tile edge and covers no pixel.
330        crate::terrain::TerrainSkirts::None => 0.0,
331    };
332    let mut uniforms = Vec::with_capacity(specs.len());
333    let mut sources = Vec::with_capacity(specs.len());
334    for (spec, drape_source) in specs.iter().zip(drape_sources) {
335        let drape_source = *drape_source;
336        let dem_coords = loaded_dem_tile(spec.coords, dem, terrain);
337        let Some(mut block) = tile_uniforms(
338            spec.coords,
339            TileTextures {
340                dem: dem_coords,
341                drape: drape_source,
342            },
343            terrain,
344            dem,
345            &gpu_view_projection
346                .to_model_view_projection(spec.coords.transform_for_zoom(zoom))
347                .downcast(),
348            skirt_length as f32,
349            &fog,
350        ) else {
351            continue;
352        };
353        let [north_cap, south_cap] =
354            polar::cap_colors(spec.coords, drape_source, style, tiles, paint_zoom);
355        let z = u8::from(spec.coords.z);
356        let max_row = 1_i32.checked_shl(u32::from(z)).unwrap_or(1) - 1;
357        let pole_elev = if spec.coords.y == 0 {
358            poles.north_meters * dem.exaggeration
359        } else if spec.coords.y == max_row {
360            poles.south_meters * dem.exaggeration
361        } else {
362            0.0
363        };
364        block.edge_index = [0, north_cap, south_cap, pole_elev.to_bits()];
365        surface_uniforms(
366            &mut block,
367            spec.coords,
368            (style, lighting, terrain.drape_mode()),
369            view_state,
370        );
371        uniforms.push(block);
372        sources.push((dem_coords, spec.coords, drape_source));
373    }
374    PreparedTerrain { uniforms, sources }
375}

377/// Adds a draw of each surface tile for every copy of the world that shows it, shifted over by
378/// the copy's whole worlds. Returns the draws' sources and the coordinates, beyond the grid for a
379/// copy, each is placed at.
380pub(super) fn with_world_copies(
381    uniforms: &mut Vec<TerrainTileUniforms>,
382    sources: Vec<TileSources>,
383    copies: Option<&std::collections::HashMap<WorldTileCoords, Vec<i32>>>,
384    view: &ViewState,
385) -> (Vec<TileSources>, Vec<WorldTileCoords>) {
386    let mut placed: Vec<WorldTileCoords> = sources.iter().map(|(_, coords, _)| *coords).collect();
387    let mut all = sources;
388    let Some(copies) = copies else {
389        return (all, placed);
390    };
391    let projection = view.gpu_view_projection();
392    let zoom = view.zoom();
393    for index in 0..all.len() {
394        let coords = all[index].1;
395        let Some(wraps) = copies.get(&coords) else {
396            continue;
397        };
398        let bounds = crate::coords::ZOOM_BOUNDS[usize::from(u8::from(coords.z))] as i32;
399        for wrap in wraps {
400            let shifted = WorldTileCoords {
401                x: coords.x + wrap * bounds,
402                ..coords
403            };
404            let mut block = uniforms[index];
405            block.transform = projection
406                .to_model_view_projection(shifted.transform_for_zoom(zoom))
407                .downcast()
408                .into();
409            block.tile_mercator_coords[0] += *wrap as f32;
410            uniforms.push(block);
411            all.push(all[index]);
412            placed.push(shifted);
413        }
414    }
415    (all, placed)
416}

418fn surface_uniforms(
419    block: &mut TerrainTileUniforms,
420    coords: WorldTileCoords,
421    (style, lighting, drape_mode): (
422        &Style,
423        crate::render::SceneLighting,
424        crate::render::settings::TerrainDrapeMode,
425    ),
426    view: &ViewState,
427) {
428    globe::set_uniforms(block, coords, view);
429    block.daylight = lighting.daylight_uniform();
430    block.twilight = lighting.twilight;
431    block.moonlight = lighting.moonlight;
432    let color = super::background_clear_color(style);
433    block.surface_color = [
434        color.r as f32,
435        color.g as f32,
436        color.b as f32,
437        color.a as f32,
438    ];
439    block.relief_strength = if view.has_external_view() { 0.45 } else { 0.0 };
440    block.fog_opacity[3] = f32::from(view.has_external_view());
441    block.fog_position = view.eye_fog_position(coords).unwrap_or([0.0; 4]);
442    match drape_mode {
443        crate::render::settings::TerrainDrapeMode::ProceduralRelief(relief) => {
444            let tiles_across = 2_f64.powi(i32::from(u8::from(coords.z)));
445            let merc_u =
446                std::f64::consts::PI * (1.0 - 2.0 * ((f64::from(coords.y) + 0.5) / tiles_across));
447            let tile_span_m =
448                (view.body().circumference_meters() / tiles_across / merc_u.cosh()).max(1.0) as f32;
449            block.procedural_relief = [
450                1.0,
451                f32::from(relief.hypsometric),
452                relief.slope_shading,
453                relief.reference_altitude_meters.unwrap_or(0.0),
454            ];
455            block.procedural_grid = [
456                relief.grid_spacing_meters,
457                relief.grid_opacity,
458                f32::from(relief.reference_altitude_meters.is_some()),
459                tile_span_m,
460            ];
461        }
462        crate::render::settings::TerrainDrapeMode::Draped { .. } => {
463            block.procedural_relief = [0.0; 4];
464            block.procedural_grid = [0.0; 4];
465        }
466    }
