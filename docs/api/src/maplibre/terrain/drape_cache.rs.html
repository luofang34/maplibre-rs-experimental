1//! Keeps drape textures across frames and knows when a tile's content changed.
2//!
3//! GL JS renders a terrain tile's layers to texture once and reuses the texture until the
4//! source tiles behind it change, identified by a fingerprint of the source tile keys and the
5//! source's revision. The cache below does the same: a tile whose fingerprint is unchanged
6//! keeps its texture untouched, and textures of tiles that left the view wait in a free list
7//! for the next tile that needs one.
8
9use std::{
10    collections::{HashMap, VecDeque},
11    hash::{Hash, Hasher},
12};
13
14use crate::{coords::WorldTileCoords, terrain::drape_targets::TargetSpec};
15
16/// Largest number of released textures kept for reuse.
17const MAX_FREE_TEXTURES: usize = 8;
18/// Hard ceiling on total drape textures (`entries + parked + free`) held in [`DrapeCache`].
19pub const HARD_MAX_DRAPE_TEXTURES: usize = crate::render::memory_budget::DRAPE_TEXTURE_LIMIT;
20
21/// What of a source tile is on the GPU right now, so a drape redraws when its own tile's
22/// content arrives or leaves and not when any other tile's does.
23pub(crate) trait SourceContent {
24    /// Identity of the uploaded style-layer geometry, or `None` when it has no allocation.
25    fn vector_layer_revision(&self, coords: WorldTileCoords, layer_id: &str) -> Option<u64>;
26    /// Revision of the raster binding, or `None` while no texture is uploaded.
27    fn raster_revision(&self, coords: WorldTileCoords, layer: &str) -> Option<u64>;
28    /// Identity of the evaluated uniforms used by a DEM-shaded layer.
29    fn dem_layer_fingerprint(&self, layer: &str) -> Option<u64>;
30}
31
32/// Fingerprint of everything that decides a drape texture's content.
33pub(crate) fn fingerprint(
34    spec: &TargetSpec,
35    content: &impl SourceContent,
36    clear: wgpu::Color,
37    paint_zoom: f64,
38) -> u64 {
39    let mut hasher = std::hash::DefaultHasher::new();
40    spec.coords.hash(&mut hasher);
41    crate::vector::upload_system::drape_paint_zoom(paint_zoom)
42        .to_bits()
43        .hash(&mut hasher);
44    for component in [clear.r, clear.g, clear.b, clear.a] {
45        component.to_bits().hash(&mut hasher);
46    }
47    let mut shapes: Vec<&crate::terrain::drape_targets::ShapeSpec> = spec.shapes.iter().collect();
48    shapes.sort_by_key(|shape| shape.source);
49    for shape in shapes {
50        shape.source.hash(&mut hasher);
51        for layer in &shape.vector_layers {
52            layer.id.hash(&mut hasher);
53            layer.index.hash(&mut hasher);
54            layer.coords.hash(&mut hasher);
55            content
56                .vector_layer_revision(layer.coords, &layer.id)
57                .hash(&mut hasher);
58        }
59        for (id, index, dem) in &shape.raster_layers {
60            id.hash(&mut hasher);
61            index.hash(&mut hasher);
62            dem.hash(&mut hasher);
63            content.raster_revision(shape.source, id).hash(&mut hasher);
64            if *dem {
65                content.dem_layer_fingerprint(id).hash(&mut hasher);
66            }
67        }
68    }
69    hasher.finish()
70}

72struct Entry<T> {
73    texture: T,
74    fingerprint: Option<u64>,
75    has_content: bool,
76    /// A view has drawn it, so it is parked with its content when its tile leaves the view;
77    /// a drape prepared ahead of the view that none has shown gives its texture back instead.
78    shown: bool,
79}
80
81impl<T> Entry<T> {
82    fn acquire(&mut self, fingerprint: u64) -> DrapeState {
83        let state = if !self.has_content {
84            DrapeState::New
85        } else if self.fingerprint == Some(fingerprint) {
86            DrapeState::Unchanged
87        } else {
88            DrapeState::Changed
89        };
90        self.fingerprint = Some(fingerprint);
91        self.has_content = true;
92        state
93    }
94}
95
96/// Released textures kept with their content, newest first, so a tile that leaves the view
97/// and returns, as tiles do when the head turns, is not drawn again.
98const PARKED_TEXTURES: usize = 8;
99
100/// Drape textures by key, a view tile or a view tile of one of several views, with the
101/// textures of tiles that left their view parked with their content and a free list of blank
102/// ones, which every key shares.
103pub struct DrapeCache<T, K = WorldTileCoords> {
104    entries: HashMap<K, Entry<T>>,
105    parked: VecDeque<(K, Entry<T>)>,
106    free: Vec<T>,
107}
108
109impl<T, K> Default for DrapeCache<T, K> {
110    fn default() -> Self {
111        Self {
112            entries: HashMap::new(),
113            parked: VecDeque::new(),
114            free: Vec::new(),
115        }
116    }
117}
118
119/// What a tile's texture holds when it is acquired for a frame.
120#[derive(Clone, Copy, Debug, PartialEq, Eq)]
121pub enum DrapeState {
122    /// The content of the same sources; nothing to draw.
123    Unchanged,
124    /// Older content of the tile, to be drawn again.
125    Changed,
126    /// Whatever the texture held before, another tile's content or nothing; to be drawn
127    /// before it shows.
128    New,
129    /// No texture: none is spare and the memory budget allows no new one. The tile draws
130    /// with an ancestor's drape until one frees up.
131    Withheld,
132}
133
134impl<T, K: Copy + Eq + std::hash::Hash> DrapeCache<T, K> {
135    /// Ensures `coords` has a texture and reports what it holds: a tile with an unchanged
136    /// fingerprint keeps its content.
137    pub fn acquire(
138        &mut self,
139        coords: K,
140        fingerprint: u64,
141        may_create: bool,
142        create: impl FnOnce() -> T,
143    ) -> DrapeState {
144        self.acquire_as(coords, fingerprint, (may_create, true), create)
145    }
146
147    /// Like [`acquire`](Self::acquire), for a drape prepared ahead of the view, which a view
148    /// has not shown until one acquires it.
149    pub fn acquire_ahead(
150        &mut self,
151        coords: K,
152        fingerprint: u64,
153        may_create: bool,
154        create: impl FnOnce() -> T,
155    ) -> DrapeState {
156        self.acquire_as(coords, fingerprint, (may_create, false), create)
157    }
158
159    fn acquire_as(
160        &mut self,
161        coords: K,
162        fingerprint: u64,
163        (may_create, shown): (bool, bool),
164        create: impl FnOnce() -> T,
165    ) -> DrapeState {
166        match self.entries.get_mut(&coords) {
167            // A drape prepared ahead keeps what it was prepared with while it is out of view:
168            // the looser covering it follows shifts as the view moves, and redrawing it for
169            // that would draw what no one sees. The view checks it once it shows the tile.
170            Some(entry) if !shown && entry.has_content => DrapeState::Unchanged,
171            Some(entry) => {
172                entry.shown |= shown;
173                entry.acquire(fingerprint)
174            }
175            None => {
176                if let Some(index) = self.parked.iter().position(|(parked, _)| *parked == coords) {
177                    if let Some((_, mut entry)) = self.parked.remove(index) {
178                        entry.shown |= shown;
179                        let state = entry.acquire(fingerprint);
180                        self.entries.insert(coords, entry);
181                        return state;
182                    }
183                }
184                // A spare texture first, then a new one while the budget allows. Past the budget
185                // a view takes the texture of a drape prepared ahead of it that none has shown,
186                // then the one parked longest ago; a drape prepared ahead takes neither, and a
187                // tile that finds none goes without.
188                let can_create = may_create && self.total_textures() < HARD_MAX_DRAPE_TEXTURES;
189                let texture = match self.free.pop() {
190                    Some(texture) => texture,
191                    None if can_create => create(),
192                    None if !shown => return DrapeState::Withheld,
193                    None => match self
194                        .take_unshown()
195                        .or_else(|| self.parked.pop_back().map(|(_, entry)| entry.texture))
196                    {
197                        Some(texture) => texture,
198                        None => return DrapeState::Withheld,
199                    },
200                };
201                self.entries.insert(
202                    coords,
203                    Entry {
204                        texture,
205                        fingerprint: Some(fingerprint),
206                        has_content: true,
207                        shown,
208                    },
209                );
210                DrapeState::New
211            }
212        }
213    }
214
215    /// Removes a drape prepared ahead of the view that none has shown, for its texture.
216    fn take_unshown(&mut self) -> Option<T> {
217        let coords = self
218            .entries
219            .iter()
220            .find(|(_, entry)| !entry.shown)
221            .map(|(coords, _)| *coords)?;
222        self.entries.remove(&coords).map(|entry| entry.texture)
223    }
224
225    /// Textures held, parked and spare together.
226    pub fn total_textures(&self) -> usize {
227        self.entries.len() + self.parked.len() + self.free.len()
228    }
229
230    /// Drops every parked and spare texture, for a host short of memory.
231    pub fn shed_spares(&mut self) {
232        self.parked.clear();
233        self.free.clear();
234    }
235
236    /// Cancels an acquired draw, preserving whether the texture has valid tile content.
237    pub fn defer(&mut self, coords: K, state: DrapeState) {
238        if let Some(entry) = self.entries.get_mut(&coords) {
239            entry.fingerprint = None;
240            entry.has_content = state != DrapeState::New;
241        }
242    }
243
244    /// Parks the textures of the tiles `keep` refuses with their content; the oldest parked
245    /// ones go to the free list, and beyond that are dropped. A test of the key, so one view's
246    /// tiles can be released without touching another's.
247    pub fn retain(&mut self, keep: impl Fn(&K) -> bool) {
248        let dropped: Vec<K> = self
249            .entries
250            .keys()
251            .filter(|coords| !keep(coords))
252            .copied()
253            .collect();
254        for coords in dropped {
255            match self.entries.remove(&coords) {
256                Some(entry) if entry.shown => self.parked.push_front((coords, entry)),
257                Some(entry) if self.free.len() < MAX_FREE_TEXTURES => self.free.push(entry.texture),
258                _ => {}
259            }
260        }
261        while self.parked.len() > PARKED_TEXTURES {
262            if let Some((_, entry)) = self.parked.pop_back() {
263                if self.free.len() < MAX_FREE_TEXTURES {
264                    self.free.push(entry.texture);
265                }
266            }
267        }
268    }
269
270    /// Whether a view has drawn the tile's drape, in use or parked after it left the view.
271    pub fn holds_shown(&self, coords: K) -> bool {
272        self.entries.get(&coords).is_some_and(|entry| entry.shown)
273            || self.parked.iter().any(|(parked, _)| *parked == coords)
274    }
275
276    /// Texture of a view tile, if it has one.
277    pub fn get(&self, coords: K) -> Option<&T> {
278        self.entries
279            .get(&coords)
280            .filter(|entry| entry.has_content)
281            .map(|entry| &entry.texture)
282    }
283
284    /// Whether the key holds a texture, drawn or still to be.
285    pub fn holds(&self, coords: K) -> bool {
286        self.entries.contains_key(&coords)
287    }
288
289    /// Number of keys `counted` accepts that hold a texture.
290    pub fn count(&self, counted: impl Fn(&K) -> bool) -> usize {
291        self.entries.keys().filter(|key| counted(key)).count()
292    }
293
294    /// Number of tiles holding a texture.
295    pub fn len(&self) -> usize {
296        self.entries.len()
297    }
298
299    /// Number of released textures waiting for reuse.
300    pub fn free_len(&self) -> usize {
301        self.free.len()
302    }
303
304    /// Number of textures parked with the content of tiles that left the view.
305    pub fn parked_len(&self) -> usize {
306        self.parked.len()
307    }
308}
