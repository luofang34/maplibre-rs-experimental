1//! Moves decoded DEM tiles from worker messages into the world.
2
3use std::{borrow::Cow, marker::PhantomData, rc::Rc};
4
5use crate::{
6    context::MapContext,
7    environment::Environment,
8    io::{
9        apc::{apply_worker_messages, AsyncProcedureCall},
10        tile_retry::{self, RequestKind, TileRequestOutcome},
11    },
12    kernel::Kernel,
13    tcs::system::{System, SystemResult},
14    terrain::{
15        backfill_neighbours,
16        dem::DemTile,
17        source::dem_source,
18        transferables::{DemTransferables, LayerDem, LayerDemMissing},
19        DemTileComponent, LoadedDem,
20    },
21};
22
23pub struct PopulateWorldSystem<E: Environment, T> {
24    kernel: Rc<Kernel<E>>,
25    phantom_t: PhantomData<T>,
26}
27
28impl<E: Environment, T> PopulateWorldSystem<E, T> {
29    pub fn new(kernel: &Rc<Kernel<E>>) -> Self {
30        Self {
31            kernel: kernel.clone(),
32            phantom_t: PhantomData,
33        }
34    }
35}
36
37impl<E: Environment, T: DemTransferables> System for PopulateWorldSystem<E, T> {
38    fn name(&self) -> Cow<'static, str> {
39        "dem_populate_world".into()
40    }
41
42    fn run(&mut self, MapContext { style, world, .. }: &mut MapContext) -> SystemResult {
43        let unpack = dem_source(style).map(|dem| dem.unpack);
44        let messages = self.kernel.apc().receive(|message| {
45            message.has_tag(RequestKind::Dem.message_tag())
46                || message.has_tag(T::LayerDem::message_tag())
47                || message.has_tag(T::LayerDemMissing::message_tag())
48        });
49        apply_worker_messages(messages, |message| {
50            if message.has_tag(RequestKind::Dem.message_tag()) {
51                tile_retry::completed(world, *message.into_transferable::<TileRequestOutcome>()?);
52                return Ok(());
53            }
54            let attempt = message.attempt();
55            let (coords, state) = if message.has_tag(T::LayerDem::message_tag()) {
56                let message = message.into_transferable::<T::LayerDem>()?;
57                let coords = message.coords();
58                if !tile_retry::accepts(world, coords, RequestKind::Dem, attempt) {
59                    return Ok(());
60                }
61                let state =
62                    match unpack.map(|unpack| DemTile::from_image(&message.into_image(), unpack)) {
63                        Some(Ok(tile)) => DemTileComponent::Loaded(LoadedDem::new(tile)),
64                        Some(Err(error)) => {
65                            tracing::warn!(%coords, %error, "DEM tile image is unusable");
66                            DemTileComponent::Missing
67                        }
68                        None => DemTileComponent::Missing,
69                    };
70                (coords, state)
71            } else {
72                let message = message.into_transferable::<T::LayerDemMissing>()?;
73                let coords = message.coords();
74                if !tile_retry::accepts(world, coords, RequestKind::Dem, attempt) {
75                    return Ok(());
76                }
77                (coords, DemTileComponent::Missing)
78            };
79            let loaded = matches!(state, DemTileComponent::Loaded(_));
80            if let Some(component) = world.tiles.query_mut::<&mut DemTileComponent>(coords) {
81                if loaded || !matches!(component, DemTileComponent::Loaded(_)) {
82                    *component = state;
83                }
84            }
85            if loaded {
86                backfill_neighbours(&mut world.tiles, coords);
87            }
88            Ok(())
89        })?;
90
91        Ok(())
92    }
93}
