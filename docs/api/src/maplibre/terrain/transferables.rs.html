16/// Tags of the messages the DEM pipeline sends back from workers.
17#[derive(Clone, Copy, Debug, Eq, PartialEq, Hash)]
18pub enum DemMessageTag {
19    /// A decoded DEM image.
20    LayerDem,
21    /// A DEM tile that could not be fetched or decoded.
22    LayerDemMissing,
23}
24
25impl MessageTag for DemMessageTag {
26    fn dyn_clone(&self) -> Box<dyn MessageTag> {
27        Box::new(*self)
28    }
29}

31/// A decoded DEM tile image on its way to the map thread.
32pub trait LayerDem: IntoMessage + Debug + Send {
33    /// Tag identifying this message kind.
34    fn message_tag() -> &'static dyn MessageTag;
35
36    /// Wraps a decoded image.
37    fn build_from(coords: WorldTileCoords, image: RgbaImage) -> Self;
38
39    /// Wraps a pre-decoded shared [`DemTile`] when the transport runs in-process, falling back
40    /// to `fallback_image` for cross-worker transports.
41    fn build_from_shared_dem(
42        coords: WorldTileCoords,
43        tile: Arc<DemTile>,
44        fallback_image: &RgbaImage,
45    ) -> Self
46    where
47        Self: Sized,
48    {
49        let _ = tile;
50        Self::build_from(coords, fallback_image.clone())
51    }
52
53    /// Coordinates of the DEM tile.
54    fn coords(&self) -> WorldTileCoords;
55
56    /// Unwraps the decoded image.
57    fn into_image(self) -> RgbaImage;
58
59    /// Extracts a shared pre-decoded [`DemTile`] if this transport carries one, or falls back to
60    /// [`Self::into_image`].
61    fn into_shared_dem(self) -> Result<(Arc<DemTile>, f64, Arc<RgbaImage>), RgbaImage>
62    where
63        Self: Sized,
64    {
65        Err(self.into_image())
66    }
67
68    /// Wraps a pre-decoded shared [`DemTile`] together with the body circumference (in metres)
69    /// used when despiking it.
70    fn build_from_cached_dem(
71        coords: WorldTileCoords,
72        tile: Arc<DemTile>,
73        circumference_meters: f64,
74        fallback_image: Arc<RgbaImage>,
75    ) -> Self
76    where
77        Self: Sized,
78    {
79        let _ = circumference_meters;
80        Self::build_from_shared_dem(coords, tile, &fallback_image)
81    }
82}

96enum DefaultDemPayload {
97    Image(RgbaImage),
98    Shared {
99        tile: Arc<DemTile>,
100        circumference_meters: f64,
101        fallback: Arc<RgbaImage>,
102    },
103}
104
105/// In-process DEM message.
106pub struct DefaultLayerDem {
107    coords: WorldTileCoords,
108    payload: DefaultDemPayload,
109}
110
111impl Debug for DefaultLayerDem {
112    fn fmt(&self, f: &mut Formatter<'_>) -> std::fmt::Result {
113        write!(f, "DefaultLayerDem({})", self.coords)
114    }
115}
116
117impl IntoMessage for DefaultLayerDem {
118    fn into(self) -> Message {
119        Message::new(Self::message_tag(), Box::new(self))
120    }
121}
122
123impl LayerDem for DefaultLayerDem {
124    fn message_tag() -> &'static dyn MessageTag {
125        &DemMessageTag::LayerDem
126    }
127
128    fn build_from(coords: WorldTileCoords, image: RgbaImage) -> Self {
129        Self {
130            coords,
131            payload: DefaultDemPayload::Image(image),
132        }
133    }
134
135    fn build_from_cached_dem(
136        coords: WorldTileCoords,
137        tile: Arc<DemTile>,
138        circumference_meters: f64,
139        fallback_image: Arc<RgbaImage>,
140    ) -> Self {
141        Self {
142            coords,
143            payload: DefaultDemPayload::Shared {
144                tile,
145                circumference_meters,
146                fallback: fallback_image,
147            },
148        }
149    }
150
151    fn coords(&self) -> WorldTileCoords {
152        self.coords
153    }
154
155    fn into_image(self) -> RgbaImage {
156        match self.payload {
157            DefaultDemPayload::Image(image) => image,
158            DefaultDemPayload::Shared { fallback, .. } => {
159                Arc::try_unwrap(fallback).unwrap_or_else(|shared| (*shared).clone())
160            }
161        }
162    }
163
164    fn into_shared_dem(self) -> Result<(Arc<DemTile>, f64, Arc<RgbaImage>), RgbaImage> {
165        match self.payload {
166            DefaultDemPayload::Shared {
167                tile,
168                circumference_meters,
169                fallback,
170            } => Ok((tile, circumference_meters, fallback)),
171            DefaultDemPayload::Image(image) => Err(image),
172        }
173    }
174}

207/// Message types a platform uses to transport DEM tiles.
208pub trait DemTransferables: Copy + Clone + 'static {
209    /// Decoded DEM image message.
210    type LayerDem: LayerDem;
211    /// Missing DEM tile message.
212    type LayerDemMissing: LayerDemMissing;
213}
