47/// Divisor of the tile circumference giving the skirt drop, as in GL JS `getSkirtLength`.
48const SKIRT_DIVISOR: f64 = 5.0;
49/// Drape textures drawn in one frame. Every level of a drape's mip chain is a render pass,
50/// and the Metal backend closes a command buffer per pass that stays open until the frame is
51/// submitted, so a burst of arriving tiles must not open more than the queue holds; it also
52/// keeps the frame's GPU time bounded while the rest wait a frame.
53const MAX_DRAPES_PER_FRAME: usize = 24;
54/// Drape textures drawn per frame while a host's eye drives the map. A flight lands hundreds
55/// of tiles within a few frames, and a display's frame budget holds only a few drapes.
56const EYE_DRAPES_PER_FRAME: usize = 8;
57
58/// How many drapes a frame draws, the rest waiting for later frames.
59#[derive(Clone, Copy, Debug)]
60pub struct DrapeBudget {
61    /// Drapes per frame of the map's own camera.
62    pub per_frame: usize,
63    /// Drapes per frame while a host's eye drives the map.
64    pub per_eye_frame: usize,
65    /// GPU time a frame spends on drapes while the camera moves; the drapes past it keep their
66    /// previous or an ancestor's texture until a later frame.
67    pub moving_time: std::time::Duration,
68}
69
70impl Default for DrapeBudget {
71    fn default() -> Self {
72        Self {
73            per_frame: MAX_DRAPES_PER_FRAME,
74            per_eye_frame: EYE_DRAPES_PER_FRAME,
75            moving_time: std::time::Duration::from_millis(4),
76        }
77    }
78}

80pub fn queue_system(
81    MapContext {
82        style,
83        view_state,
84        world,
85        renderer:
86            Renderer {
87                device,
88                queue,
89                settings,
90                ..
91            },
92        ..
93    }: &mut MapContext,
94) -> SystemResult {
95    let drape_mode = world
96        .resources
97        .get::<crate::render::settings::TerrainDrapeMode>()
98        .copied()
99        .unwrap_or_else(|| settings.effective_terrain_drape_mode());
100    if let Some(Initialized(terrain)) = world.resources.get_mut::<Eventually<TerrainResources>>() {
101        if terrain.drape_mode() != drape_mode {
102            terrain.set_drape_mode(device, drape_mode);
103        }
104    }
105    if EyeInFrame::reuses_content(world) {
106        return uniforms::replay(world, style, view_state, queue);
107    }
108    world.resources.insert(TerrainEyeFrame::default());
109    world
110        .resources
111        .insert(surface_covering::SurfaceTiles::default());
112    let Some(dem) = dem_source(style) else {
113        world.resources.insert(DrapePhase::default());
114        set_frame(world, TerrainFrame::default());
115        return Ok(());
116    };
117    let (phase, surface_specs, surface_sources) = if drape_mode.is_procedural_relief() {
118        let specs = procedural_specs(style, view_state, world)?;
119        let sources = vec![None; specs.len()];
120        (DrapePhase::default(), specs, sources)
121    } else {
122        let specs = target_specs(style, view_state, world)?;
123        let PreparedDrapes {
124            redraw,
125            sources,
126            clear_color,
127        } = prepare_drapes(&specs, style, view_state, world, device)?;
128        let paint_zoom = crate::vector::upload_system::paint::current_zoom(
129            world,
130            view_state.style_zoom().value(),
131        );
132        let phase = encode_drapes(
133            &specs,
134            &redraw,
135            clear_color,
136            world,
137            crate::coords::Zoom::new(paint_zoom),
138            queue,
139        )?;
140        let (surface_specs, surface_sources) = surface_covering::for_frame(world, &specs, &sources);
141        (phase, surface_specs, surface_sources)
142    };
143    queue_tiles(
144        world,
145        style,
146        view_state,
147        device,
148        queue,
149        &dem,
150        (&surface_specs, &surface_sources),
151    )?;
152    hide_draped_layers(world, style);
153    let draw_after_layer_index = style
154        .layers
155        .iter()
156        .filter(|layer| layer.type_ == "background")
157        .map(|layer| layer.index)
158        .max()
159        .unwrap_or(0);
160    world.resources.insert(phase);
161    set_frame(
162        world,
163        TerrainFrame {
164            active: true,
165            draw_after_layer_index,
166        },
167    );
168    Ok(())
169}

171/// The GPU content of source tiles, read from the vector pool and the raster textures.
172struct LoadedContent<'a> {
173    pool: Option<&'a VectorBufferPool>,
174    raster: Option<&'a RasterResources>,
175    dem: Option<&'a crate::hillshade::HillshadeResources>,
176}
177
178impl SourceContent for LoadedContent<'_> {
179    fn vector_layer_revision(&self, coords: WorldTileCoords, layer_id: &str) -> Option<u64> {
180        self.pool?
181            .index()
182            .get_layers(coords)?
183            .iter()
184            .find(|entry| entry.style_layer.id == layer_id)
185            .map(|entry| entry.allocation_id())
186    }
187
188    fn raster_revision(&self, coords: WorldTileCoords, layer: &str) -> Option<u64> {
189        self.raster
190            .and_then(|raster| raster.texture_revision(layer, coords))
191    }
192
193    fn dem_layer_fingerprint(&self, layer: &str) -> Option<u64> {
194        self.dem.and_then(|dem| dem.layer_fingerprint(layer))
195    }
196}
197
198fn loaded_content(world: &World) -> LoadedContent<'_> {
199    LoadedContent {
200        pool: match world.resources.get::<Eventually<VectorBufferPool>>() {
201            Some(Initialized(pool)) => Some(pool),
202            _ => None,
203        },
204        raster: match world.resources.get::<Eventually<RasterResources>>() {
205            Some(Initialized(raster)) => Some(raster),
206            _ => None,
207        },
208        dem: match world
209            .resources
210            .get::<Eventually<crate::hillshade::HillshadeResources>>()
211        {
212            Some(Initialized(dem)) => Some(dem),
213            _ => None,
214        },
215    }
216}

221fn queue_tiles(
222    world: &mut World,
223    style: &Style,
224    view_state: &ViewState,
225    device: &wgpu::Device,
226    queue: &crate::render::upload_queue::UploadQueue,
227    dem: &DemSource,
228    targets: (&[TargetSpec], &[Option<WorldTileCoords>]),
229) -> SystemResult {
230    let mut edge_cache = std::mem::take(world.resources.get_or_init_mut::<edges::EdgeCache>());
231    let copies = world
232        .resources
233        .get::<surface_covering::SurfaceCopies>()
234        .map(|copies| copies.0.clone());
235    let ctx = uniforms::surface_context(world, style, view_state, dem);
236    {
237        let Some(Initialized(terrain)) = world.resources.get_mut::<Eventually<TerrainResources>>()
238        else {
239            return Err(SystemError::Dependencies);
240        };
241        let uniforms::PreparedTerrain {
242            mut uniforms,
243            sources,
244        } = uniforms::prepare_tiles(targets, terrain, &world.tiles, ctx);
245        let edges = edge_cache.apply(&sources, &mut uniforms, &world.tiles);
246        let (sources, placed) =
247            uniforms::with_world_copies(&mut uniforms, sources, copies.as_ref(), view_state);
248        terrain.write_edges(queue, &edges);
249        let written = terrain.write_uniforms(queue, &uniforms);
250        tracing::debug!(
251            targets = targets.0.len(),
252            dem_hits = sources.iter().filter(|(dem, _, _)| dem.is_some()).count(),
253            written,
254            "terrain frame queued"
255        );
256        let draws = sources
257            .iter()
258            .take(written)
259            .enumerate()
260            .filter_map(|(index, (dem_coords, coords, drape_source))| {
261                Some(TerrainDraw {
262                    coords: *coords,
263                    bind_group: terrain.tile_bind_group(device, *dem_coords, *drape_source)?,
264                    uniform_offset: (index as u64 * UNIFORM_STRIDE) as u32,
265                })
266            })
267            .collect::<Vec<_>>();
268        let key = targets::draw_order_key(style, view_state, f64::from(dem.tile_size));
269        let mut draws = draws;
270        draws.sort_by(|left, right| key(&left.coords).total_cmp(&key(&right.coords)));
271        terrain.set_draws(draws);
272        if let Some(index) = world
273            .resources
274            .get_mut::<crate::terrain::TerrainCoverageIndex>()
275        {
276            index.set_surface_edges(&sources, std::sync::Arc::clone(&edge_cache.samples));
277        }
278        let kept = placed.into_iter().zip(uniforms).take(written).collect();
279        world.resources.insert(TerrainEyeFrame(kept));
280    }
281    world.resources.insert(edge_cache);
282
283    Ok(())
284}

286struct PreparedDrapes {
287    redraw: Vec<bool>,
288    sources: Vec<Option<WorldTileCoords>>,
289    clear_color: wgpu::Color,
290}

292fn prepare_drapes(
293    specs: &[TargetSpec],
294    style: &Style,
295    view_state: &ViewState,
296    world: &mut World,
297    device: &wgpu::Device,
298) -> Result<PreparedDrapes, SystemError> {
299    let memory = world
300        .resources
301        .get::<MemoryBudget>()
302        .copied()
303        .unwrap_or_default();
304    let clear_color = background_clear_color(style);
305    let paint_zoom =
306        crate::vector::upload_system::paint::current_zoom(world, view_state.style_zoom().value());
307    let prints: Vec<u64> = {
308        let content = loaded_content(world);
309        specs
310            .iter()
311            .map(|spec| fingerprint(spec, &content, clear_color, paint_zoom))
312            .collect()
313    };
314    // A tile whose vector sources are finished but not yet in the buffer pool would drape
315    // blank; it shows an ancestor's drape until they are.
316    let strict = world
317        .resources
318        .get::<cohort::TextureCohort>()
319        .is_some_and(|state| state.enabled);
320    let mut ready: Vec<bool> = specs
321        .iter()
322        .map(|spec| readiness::ready(spec, style, view_state.style_zoom().value(), world, strict))
323        .collect();
324    let capacity = match world.resources.get::<Eventually<WgpuTileViewPattern>>() {
325        Some(Initialized(pattern)) => pattern.remaining_metadata_capacity(),
326        _ => 0,
327    };
328    surface_covering::fit_metadata(specs, &mut ready, capacity);
329    let prefetch: Vec<bool> = {
330        let ahead = world.resources.get::<targets::PrefetchDrapes>();
331        specs
332            .iter()
333            .map(|spec| ahead.is_some_and(|ahead| ahead.0.contains(&spec.coords)))
334            .collect()
335    };
336    let budget = world
337        .resources
338        .get::<DrapeBudget>()
339        .copied()
340        .unwrap_or_default();
341    let limit = if view_state.has_external_view() {
342        budget.per_eye_frame
343    } else {
344        budget.per_frame
345    };
346    // A moving view draws what fits its time; the view at rest draws the rest at full pace.
347    let budget = if crate::render::frame_signals::camera_moved(world, view_state) {
348        world
349            .resources
350            .get::<crate::terrain::drape_timing::DrapeCost>()
351            .copied()
352            .unwrap_or_default()
353            .drapes_within(budget.moving_time, limit)
354    } else {
355        limit
356    };
357    let extra_retained = world
358        .resources
359        .get::<crate::render::eye_covering::SlotResidencyFloor>()
360        .map(|floor| floor.retained_drapes.clone());
361    let (redraw, drape_sources, deferred) = {
362        let Some(Initialized(terrain)) = world.resources.get_mut::<Eventually<TerrainResources>>()
363        else {
364            return Err(SystemError::Dependencies);
365        };
366        terrain.ensure_scratch(device);
367        acquire_drapes(
368            (specs, extra_retained.as_ref()),
369            (&prints, &ready, &prefetch),
370            memory,
371            (budget, limit),
372            terrain,
373            device,
374        )
375    };
376    let drawn = |ahead: bool| {
377        redraw
378            .iter()
379            .zip(&prefetch)
380            .filter(|(drawn, prefetched)| **drawn && **prefetched == ahead)
381            .count()
382    };
383    crate::render::frame_signals::count_drape_redraws(world, drawn(false), drawn(true));
384    // A drape left for a later frame's budget needs that frame.
385    if deferred {
386        crate::render::frame_signals::keep_animating(world);
387    }
388    Ok(PreparedDrapes {
389        redraw,
390        sources: drape_sources,
391        clear_color,
392    })
