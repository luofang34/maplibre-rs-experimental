1//! Single-frame cache for tight terrain [`crate::coords::ViewRegion`] coverings.
2
3use super::TerrainCoverageIndex;
4use crate::render::view_state::ViewStatePadding;
5
6/// Exact view + DEM-bounds key for single-frame reuse of a tight [`crate::coords::ViewRegion`].
7#[derive(Clone, Copy, PartialEq, Eq)]
8pub(crate) struct PreparedTerrainKey {
9    dem_tile_size: u32,
10    width_bits: u64,
11    height_bits: u64,
12    overscan_bits: u64,
13    body_radius_bits: u64,
14    external_view: bool,
15    vp_bits: [[u64; 4]; 4],
16    min_elev_bits: u64,
17    max_elev_bits: u64,
18    dem_bounds_len: usize,
19}
20
21impl PreparedTerrainKey {
22    pub(crate) fn capture(
23        view_state: &crate::render::view_state::ViewState,
24        world: &crate::tcs::world::World,
25        dem_tile_size: u32,
26    ) -> Self {
27        let m: [[f64; 4]; 4] = view_state.view_projection().0.into();
28        let vp_bits = m.map(|col| col.map(f64::to_bits));
29        let (min_elev_bits, max_elev_bits, dem_bounds_len) = world
30            .resources
31            .get::<TerrainCoverageIndex>()
32            .map_or((0, 0, 0), |idx| {
33                (
34                    idx.min_elevation.to_bits(),
35                    idx.max_elevation.to_bits(),
36                    idx.dem_bounds.len(),
37                )
38            });
39        Self {
40            dem_tile_size,
41            width_bits: view_state.width().to_bits(),
42            height_bits: view_state.height().to_bits(),
43            overscan_bits: view_state.request_overscan().to_bits(),
44            body_radius_bits: view_state.body().radius_meters.to_bits(),
45            external_view: view_state.has_external_view(),
46            vp_bits,
47            min_elev_bits,
48            max_elev_bits,
49            dem_bounds_len,
50        }
51    }
52}
53
54/// Cached tight [`crate::coords::ViewRegion`] from `coverage_system` or `multi_view` covering so
55/// `Queue` (`target_specs` / `procedural_specs`) skips a duplicate S2 / frustum covering pass
56/// when neither the camera projection nor DEM elevation bounds changed.
57#[derive(Clone)]
58pub(crate) struct PreparedTerrainRegion {
59    pub(crate) key: PreparedTerrainKey,
60    pub(crate) region: Option<crate::coords::ViewRegion>,
61}
62
63/// Takes the cached tight terrain region if its [`PreparedTerrainKey`] matches `view_state`
64/// and `world`, otherwise computes a fresh `terrain_region`.
65pub(crate) fn take_or_compute_tight_terrain_region(
66    style: &crate::style::Style,
67    view_state: &crate::render::view_state::ViewState,
68    world: &mut crate::tcs::world::World,
69    dem_tile_size: u32,
70) -> Result<Option<crate::coords::ViewRegion>, crate::render::projection::ProjectionStateError> {
71    if let Some(cached) = world.resources.remove::<PreparedTerrainRegion>() {
72        if cached.key == PreparedTerrainKey::capture(view_state, world, dem_tile_size) {
73            return Ok(cached.region);
74        }
75    }
76    crate::render::projection::terrain_region(
77        style,
78        view_state,
79        world,
80        dem_tile_size,
81        ViewStatePadding::Tight,
82    )
83}
