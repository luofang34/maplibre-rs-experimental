12/// Hard LRU cap on terrain elevation textures bound simultaneously in [`TerrainResources`].
13const MAX_TERRAIN_DEM_TEXTURES: usize = crate::terrain::gpu_dem_pool::MAX_GPU_DEM_TEXTURES;
14
15static DEM_TOUCH_TICK: AtomicU64 = AtomicU64::new(0);
16
17fn next_dem_tick() -> u64 {
18    DEM_TOUCH_TICK
19        .fetch_add(1, Ordering::Relaxed)
20        .wrapping_add(1)
21}
22
23pub(super) struct DemTexture {
24    pub(super) id: u64,
25    pub(super) texture: Texture,
26    revision: u32,
27    source: Option<DemRevision>,
28    last_used: AtomicU64,
29}
30
31impl TerrainResources {
32    fn evict_excess_dem_textures(&mut self, keep: WorldTileCoords) {
33        while self.dem_textures.len() >= MAX_TERRAIN_DEM_TEXTURES {
34            let Some(oldest) = self
35                .dem_textures
36                .iter()
37                .filter(|(coords, _)| **coords != keep)
38                .min_by_key(|(_, entry)| entry.last_used.load(Ordering::Relaxed))
39                .map(|(coords, _)| *coords)
40            else {
41                break;
42            };
43            if let Some(removed) = self.dem_textures.remove(&oldest) {
44                self.bind_groups
45                    .retain(|(dem_id, _), _| *dem_id != removed.id);
46            }
47        }
48    }
49
50    pub(crate) fn bind_shared_dem(
51        &mut self,
52        coords: WorldTileCoords,
53        texture: Texture,
54        revision: u32,
55        source: DemRevision,
56    ) {
57        if !self.dem_textures.contains_key(&coords) {
58            self.evict_excess_dem_textures(coords);
59        }
60        let prev = self.dem_textures.get(&coords);
61        let id = match prev {
62            Some(existing) if existing.texture.texture == texture.texture => existing.id,
63            _ => super::next_texture_id(),
64        };
65        if let Some(existing) = prev {
66            if existing.id != id {
67                let stale_id = existing.id;
68                self.bind_groups
69                    .retain(|(dem_id, _), _| *dem_id != stale_id);
70            }
71        }
72        self.dem_textures.insert(
73            coords,
74            DemTexture {
75                id,
76                texture,
77                revision,
78                source: Some(source),
79                last_used: AtomicU64::new(next_dem_tick()),
80            },
81        );
82    }
83
84    pub(crate) fn upload_loaded_dem(
85        &mut self,
86        device: &wgpu::Device,
87        queue: &crate::render::upload_queue::UploadQueue,
88        coords: WorldTileCoords,
89        dem: &LoadedDem,
90    ) {
91        if !self.needs_dem_upload(coords, dem) {
92            return;
93        }
94        let source = dem.revision_key();
95        self.upload_dem(device, queue, coords, &dem.tile, dem.revision);
96        if let Some(entry) = self.dem_textures.get_mut(&coords) {
97            entry.source = Some(source);
98        }
99    }
100
101    /// Whether the uploaded copy of the DEM tile at `coords`, if any, is of another revision.
102    pub(crate) fn needs_dem_upload(&self, coords: WorldTileCoords, dem: &LoadedDem) -> bool {
103        let source = dem.revision_key();
104        !self
105            .dem_textures
106            .get(&coords)
107            .is_some_and(|entry| entry.source.as_ref() == Some(&source))
108    }
109
110    /// Whether a DEM tile has been uploaded.
111    pub fn has_dem_texture(&self, coords: WorldTileCoords) -> bool {
112        self.dem_textures.contains_key(&coords)
113    }
114
115    /// Revision of the uploaded copy of a DEM tile, if any.
116    pub fn dem_revision(&self, coords: WorldTileCoords) -> Option<u32> {
117        self.dem_textures.get(&coords).map(|dem| dem.revision)
118    }
119
120    /// Uploads the bordered pixels of a decoded DEM tile, reusing its texture across revisions.
121    pub fn upload_dem(
122        &mut self,
123        device: &wgpu::Device,
124        queue: &crate::render::upload_queue::UploadQueue,
125        coords: WorldTileCoords,
126        dem: &DemTile,
127        revision: u32,
128    ) {
129        let reusable = self
130            .dem_textures
131            .remove(&coords)
132            .filter(|dem_tex| dem_tex.texture.size.width == dem.stride());
133        self.evict_excess_dem_textures(coords);
134        let (id, texture) = match reusable {
135            Some(dem_tex) => (dem_tex.id, dem_tex.texture),
136            None => (
137                super::next_texture_id(),
138                Texture::new(
139                    Some("DEM tile"),
140                    device,
141                    wgpu::TextureFormat::Rgba8Unorm,
142                    dem.stride(),
143                    dem.stride(),
144                    Msaa { samples: 1 },
145                    wgpu::TextureUsages::TEXTURE_BINDING | wgpu::TextureUsages::COPY_DST,
146                ),
147            ),
148        };
149        queue.write_texture(
150            wgpu::TexelCopyTextureInfo {
151                texture: &texture.texture,
152                mip_level: 0,
153                origin: wgpu::Origin3d::ZERO,
154                aspect: wgpu::TextureAspect::All,
155            },
156            dem.pixels(),
157            wgpu::TexelCopyBufferLayout {
158                offset: 0,
159                bytes_per_row: Some(4 * dem.stride()),
160                rows_per_image: Some(dem.stride()),
161            },
162            texture.size,
163        );
164        self.dem_textures.insert(
165            coords,
166            DemTexture {
167                id,
168                texture,
169                revision,
170                source: None,
171                last_used: AtomicU64::new(next_dem_tick()),
172            },
173        );
174    }
175
176    /// Releases the DEM texture of a tile that left the store.
177    pub fn drop_dem(&mut self, coords: WorldTileCoords) {
178        if let Some(removed) = self.dem_textures.remove(&coords) {
179            self.bind_groups
180                .retain(|(dem_id, _), _| *dem_id != removed.id);
181        }
182    }
183
184    /// DEM texture of a tile, or the flat stand-in while it loads.
185    pub fn dem_texture(&self, coords: Option<WorldTileCoords>) -> &Texture {
186        coords
187            .and_then(|coords| self.dem_textures.get(&coords))
188            .map_or(&self.empty_dem, |dem| {
189                dem.last_used.store(next_dem_tick(), Ordering::Relaxed);
190                &dem.texture
191            })
192    }
193
194    pub(super) fn dem_id(&self, coords: Option<WorldTileCoords>) -> u64 {
195        coords
196            .and_then(|coords| self.dem_textures.get(&coords))
197            .map_or(self.empty_dem_id, |dem| dem.id)
198    }
199
200    /// DEM textures resident on the GPU.
201    pub fn dem_texture_count(&self) -> usize {
202        self.dem_textures.len()
203    }
