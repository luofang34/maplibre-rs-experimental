1//! Cached texture bindings for terrain surfaces and their symbols.
2use super::*;
3use crate::render::resource::share_gpu;
4
5impl TerrainResources {
6    /// Binds the uniform block window, a DEM texture and a drape texture for one tile.
7    pub fn create_bind_group(
8        &self,
9        device: &wgpu::Device,
10        dem: &Texture,
11        drape: &Texture,
12    ) -> wgpu::BindGroup {
13        device.create_bind_group(&wgpu::BindGroupDescriptor {
14            label: Some("terrain tile"),
15            layout: &self.pipeline.get_bind_group_layout(1),
16            entries: &[
17                wgpu::BindGroupEntry {
18                    binding: 0,
19                    resource: wgpu::BindingResource::Buffer(wgpu::BufferBinding {
20                        buffer: &self.uniform_buffer,
21                        offset: 0,
22                        size: NonZeroU64::new(std::mem::size_of::<TerrainTileUniforms>() as u64),
23                    }),
24                },
25                wgpu::BindGroupEntry {
26                    binding: 1,
27                    resource: wgpu::BindingResource::TextureView(&dem.view),
28                },
29                wgpu::BindGroupEntry {
30                    binding: 2,
31                    resource: wgpu::BindingResource::TextureView(&drape.view),
32                },
33                wgpu::BindGroupEntry {
34                    binding: 3,
35                    resource: wgpu::BindingResource::Sampler(&self.sampler),
36                },
37            ],
38        })
39    }
40
41    /// Reuses bindings when both textures are unchanged. No drape source binds an inert
42    /// texture for the shader's background surface; an invalid named source returns `None`.
43    pub fn tile_bind_group(
44        &mut self,
45        device: &wgpu::Device,
46        dem: Option<WorldTileCoords>,
47        drape_source: Option<WorldTileCoords>,
48    ) -> Option<Arc<wgpu::BindGroup>> {
49        let key = {
50            let dem = self.dem_texture(dem);
51            let drape = match drape_source {
52                Some(source) => self.drape_texture(source)?,
53                None => &self.empty_dem,
54            };
55            (dem.texture.clone(), drape.texture.clone())
56        };
57        if let Some(group) = self.bind_groups.get(&key) {
58            return Some(Arc::clone(group));
59        }
60        let group = {
61            let dem = self.dem_texture(dem);
62            let drape = match drape_source {
63                Some(source) => self.drape_texture(source)?,
64                None => &self.empty_dem,
65            };
66            share_gpu(self.create_bind_group(device, dem, drape))
67        };
68        self.bind_groups.insert(key, Arc::clone(&group));
69        Some(group)
70    }
71}
