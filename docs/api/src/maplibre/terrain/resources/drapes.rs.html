1//! Allocation and deferred content of terrain drape textures.
2//!
3//! A drape belongs to a terrain tile of one view. Each view selects the source tiles of a
4//! drape from its own covering, so two views draping the same terrain tile want different
5//! content in it; with one texture between them each would find the other's content changed
6//! and draw it again every frame, and what a view shows would depend on which other views
7//! share its frame. The lookups by tile answer for the view chosen with
8//! [`TerrainResources::show_drapes_of`], and the spare textures are common to all.
9use std::collections::HashSet;
10
11use super::TerrainResources;
12use crate::{
13    coords::WorldTileCoords,
14    render::{eye_covering::ViewSlotId, resource::Texture},
15    terrain::drape_cache::DrapeState,
16};
17
18/// A view, `None` for a map drawn without view slots, and a terrain tile it drapes.
19pub(crate) type DrapeKey = (Option<ViewSlotId>, WorldTileCoords);
20
21/// A drape texture with the view of its first level, which the drape pass draws into every
22/// time the drape is redrawn; the texture's own view spans every level, which an attachment
23/// may not.
24pub(crate) struct DrapeTexture {
25    top_level: wgpu::TextureView,
26    texture: Texture,
27}
28
29impl DrapeTexture {
30    fn new(texture: Texture) -> Self {
31        let top_level = texture.texture.create_view(&wgpu::TextureViewDescriptor {
32            label: Some("drape top level"),
33            base_mip_level: 0,
34            mip_level_count: Some(1),
35            ..Default::default()
36        });
37        Self { top_level, texture }
38    }
39}
40
41impl TerrainResources {
42    /// Chooses the view whose drapes the tile lookups answer with from here on.
43    pub(crate) fn show_drapes_of(&mut self, view: Option<ViewSlotId>) {
44        self.drape_view = view;
45    }
46
47    /// Releases the drapes of a view that draws no more; their textures become spare.
48    #[cfg(feature = "headless")]
49    pub(crate) fn release_drapes_of(&mut self, view: Option<ViewSlotId>) {
50        self.drapes.retain(|(owner, _)| *owner != view);
51    }
52
53    fn drape_key(&self, coords: WorldTileCoords) -> DrapeKey {
54        (self.drape_view, coords)
55    }
56
57    /// Gives a view tile a drape texture and reports what it holds.
58    pub fn acquire_drape(
59        &mut self,
60        device: &wgpu::Device,
61        coords: WorldTileCoords,
62        fingerprint: u64,
63        may_create: bool,
64    ) -> DrapeState {
65        let format = self.color_format;
66        let size = self.drape_size;
67        let key = self.drape_key(coords);
68        self.drapes.acquire(key, fingerprint, may_create, || {
69            DrapeTexture::new(Texture::new_mipmapped(
70                Some("drape texture"),
71                device,
72                format,
73                size,
74                size,
75                wgpu::TextureUsages::RENDER_ATTACHMENT | wgpu::TextureUsages::TEXTURE_BINDING,
76            ))
77        })
78    }
79
80    /// Like [`acquire_drape`](Self::acquire_drape), for a drape prepared ahead of the view.
81    pub fn acquire_ahead_drape(
82        &mut self,
83        device: &wgpu::Device,
84        coords: WorldTileCoords,
85        fingerprint: u64,
86        may_create: bool,
87    ) -> DrapeState {
88        let format = self.color_format;
89        let size = self.drape_size;
90        let key = self.drape_key(coords);
91        self.drapes.acquire_ahead(key, fingerprint, may_create, || {
92            DrapeTexture::new(Texture::new_mipmapped(
93                Some("drape texture"),
94                device,
95                format,
96                size,
97                size,
98                wgpu::TextureUsages::RENDER_ATTACHMENT | wgpu::TextureUsages::TEXTURE_BINDING,
99            ))
100        })
101    }
102
103    /// Whether the view may take a texture for the tile while holding at most `allowed`: a
104    /// tile that has one keeps it.
105    pub(crate) fn within_drape_allowance(&self, coords: WorldTileCoords, allowed: usize) -> bool {
106        let view = self.drape_view;
107        self.drapes.holds(self.drape_key(coords))
108            || self.drapes.count(|(owner, _)| *owner == view) < allowed
109    }
110
111    /// Leaves a drape acquired this frame undrawn, to be drawn on the next.
112    pub fn defer_drape(&mut self, coords: WorldTileCoords, state: DrapeState) {
113        let key = self.drape_key(coords);
114        self.drapes.defer(key, state);
115    }
116
117    /// Drape textures held, parked and spare together.
118    pub fn drape_texture_total(&self) -> usize {
119        self.drapes.total_textures()
120    }
121
122    /// Drops the parked and spare drape textures.
123    pub fn shed_spare_drapes(&mut self) {
124        self.drapes.shed_spares();
125    }
126
127    /// Fills the mip levels of a drape texture after its layers were drawn into it.
128    pub fn generate_drape_mipmaps(
129        &self,
130        device: &wgpu::Device,
131        encoder: &mut wgpu::CommandEncoder,
132        texture: &Texture,
133    ) {
134        if let Some(mipmaps) = &self.mipmaps {
135            mipmaps.generate(device, encoder, &texture.texture);
136        }
137    }
138
139    /// Keeps the requested tile drapes of the view active and parks its other textures with
140    /// their content; the drapes of other views stay as they are. Older parked textures are
141    /// recycled or dropped according to the cache's spare limits.
142    pub fn retain_drapes(&mut self, keep: &HashSet<WorldTileCoords>) {
143        let view = self.drape_view;
144        self.drapes
145            .retain(|(owner, coords)| *owner != view || keep.contains(coords));
146    }
147
148    /// Whether a view has drawn the tile's drape, in use or parked after it left the view.
149    pub fn holds_shown_drape(&self, coords: WorldTileCoords) -> bool {
150        self.drapes.holds_shown(self.drape_key(coords))
151    }
152
153    /// Drape texture of a view tile.
154    pub fn drape_texture(&self, coords: WorldTileCoords) -> Option<&Texture> {
155        self.drapes
156            .get(self.drape_key(coords))
157            .map(|drape| &drape.texture)
158    }
159
160    /// Drape texture of a view tile with the view of its first level, the one drawn into.
161    pub(crate) fn drape_target(
162        &self,
163        coords: WorldTileCoords,
164    ) -> Option<(&Texture, &wgpu::TextureView)> {
165        self.drapes
166            .get(self.drape_key(coords))
167            .map(|drape| (&drape.texture, &drape.top_level))
168    }
169
170    /// Drape textures held for tiles, and parked or waiting in the free list.
171    pub fn drape_counts(&self) -> (usize, usize) {
172        (
173            self.drapes.len(),
174            self.drapes.parked_len() + self.drapes.free_len(),
175        )
176    }
177}
