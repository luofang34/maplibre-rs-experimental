1//! Per-tile stitched edge height texture for WebGL2-compatible vertex-stage lookup.
2
3use super::{TerrainResources, UNIFORM_CAPACITY};
4
5/// One row of the edge height texture: a tile's [`EdgeHeights`](crate::terrain::queue_system::edges::EdgeHeights).
6pub(super) const EDGE_BYTES_PER_TILE: u32 =
7    std::mem::size_of::<crate::terrain::queue_system::edges::EdgeHeights>() as u32;
8/// Edge height texels per tile, four edges to the texel.
9pub(super) const EDGE_TEXELS_PER_TILE: u32 =
10    EDGE_BYTES_PER_TILE / std::mem::size_of::<[f32; 4]>() as u32;
11
12/// Clamps the per-frame edge texture row capacity to the device's 2D texture dimension limit.
13pub(super) fn edge_row_capacity(max_texture_dimension_2d: u32) -> u32 {
14    (UNIFORM_CAPACITY as u32)
15        .min(max_texture_dimension_2d)
16        .max(1)
17}
18
19/// The edge height texture's extent for `tiles` tiles, one row each.
20pub(super) fn edge_texture_size(tiles: u32) -> wgpu::Extent3d {
21    wgpu::Extent3d {
22        width: EDGE_TEXELS_PER_TILE,
23        height: tiles,
24        depth_or_array_layers: 1,
25    }
26}
27
28pub(crate) struct EdgeHeightTexture {
29    texture: wgpu::Texture,
30    pub(super) view: wgpu::TextureView,
31    capacity: usize,
32}
33
34impl EdgeHeightTexture {
35    pub(super) fn new(device: &wgpu::Device) -> Self {
36        let rows = edge_row_capacity(device.limits().max_texture_dimension_2d);
37        let texture = device.create_texture(&wgpu::TextureDescriptor {
38            label: Some("terrain edge heights"),
39            size: edge_texture_size(rows),
40            mip_level_count: 1,
41            sample_count: 1,
42            dimension: wgpu::TextureDimension::D2,
43            format: wgpu::TextureFormat::Rgba32Float,
44            usage: wgpu::TextureUsages::TEXTURE_BINDING | wgpu::TextureUsages::COPY_DST,
45            view_formats: &[],
46        });
47        let view = texture.create_view(&wgpu::TextureViewDescriptor::default());
48        Self {
49            texture,
50            view,
51            capacity: rows as usize,
52        }
53    }
54
55    pub(super) fn capacity(&self) -> usize {
56        self.capacity
57    }
58}
59
60impl Drop for EdgeHeightTexture {
61    fn drop(&mut self) {
62        self.texture.destroy();
63    }
64}
65
66impl TerrainResources {
67    /// Writes the per-tile edge height blocks.
68    pub(crate) fn write_edges(
69        &self,
70        queue: &crate::render::upload_queue::UploadQueue,
71        edges: &[crate::terrain::queue_system::edges::EdgeHeights],
72    ) {
73        let count = edges.len().min(self.edge_heights.capacity());
74        if count == 0 {
75            return;
76        }
77        queue.write_texture(
78            self.edge_heights.texture.as_image_copy(),
79            bytemuck::cast_slice(&edges[..count]),
80            wgpu::TexelCopyBufferLayout {
81                offset: 0,
82                bytes_per_row: Some(EDGE_BYTES_PER_TILE),
83                rows_per_image: None,
84            },
85            edge_texture_size(count as u32),
86        );
87    }
88}
