1//! One allowance of drape textures, divided between the views that drape.
2//!
3//! Memory allows a map so many drape textures, and every view holds its own. Were the views to
4//! draw from the allowance as they come, the first of a frame would take what it wants and the
5//! far tiles of the next would go without a texture and show an ancestor's. The allowance is
6//! instead granted at the start of a frame, by need and in the order of the views' priority:
7//! the first view has the textures it needs to drape its ground uncoarsened, as it would alone,
8//! and each view after it has what is left, never less than a floor that keeps its ground
9//! draped. A view coarsens its drapes to its grant, so the views together never ask for more
10//! textures than there are, whatever the device allows and however many views it draws, and
11//! what a view does not need is not kept from the others.
12
13use rustc_hash::FxHashMap;
14
15use crate::{
16    render::{
17        eye_covering::{DrapeView, ViewSlotId},
18        memory_budget::MemoryBudget,
19    },
20    tcs::world::World,
21};

23/// Textures a view with the whole allowance keeps spare while it regroups its drapes: two
24/// groups of four siblings whose parents stay visible until they are drawn. A view with part
25/// of the allowance keeps that part of them.
26const REGROUPING_TEXTURES: usize = 8;
27
28/// Textures a view is granted however many the views before it need, so its ground is draped,
29/// coarsely, rather than bare.
30#[cfg(any(feature = "headless", test))]
31const FLOOR_TEXTURES: usize = 4;
32
33/// Needs are granted in steps of this many textures, so a covering that gains or loses a tile
34/// from one frame to the next does not regroup the drapes of the views after it.
35#[cfg(any(feature = "headless", test))]
36const NEED_STEP: usize = 4;
37
38/// What each draping view needs and is granted of the drape texture allowance. A map drawn
39/// without view slots has all of it.
40#[derive(Clone, Debug, Default)]
41pub(crate) struct DrapeShares {
42    /// Terrain tiles each view last draped, before coarsening.
43    needs: FxHashMap<ViewSlotId, usize>,
44    grants: FxHashMap<ViewSlotId, usize>,
45}
46
47impl DrapeShares {
48    /// Grants `allowed` textures to `views`, which come in the order of their priority. A view
49    /// that has not draped yet is taken to need them all.
50    #[cfg(any(feature = "headless", test))]
51    pub(crate) fn grant(&mut self, views: &[ViewSlotId], allowed: usize) {
52        let needs: Vec<usize> = views
53            .iter()
54            .map(|view| {
55                self.needs.get(view).map_or(allowed, |tiles| {
56                    (tiles + REGROUPING_TEXTURES)
57                        .next_multiple_of(NEED_STEP)
58                        .min(allowed)
59                })
60            })
61            .collect();
62        let floors: Vec<usize> = needs.iter().map(|need| FLOOR_TEXTURES.min(*need)).collect();
63        let mut left = allowed;
64        self.grants.clear();
65        for (index, view) in views.iter().enumerate() {
66            let after: usize = floors.iter().skip(index + 1).sum();
67            let granted = needs[index]
68                .min(left.saturating_sub(after))
69                .max(floors[index].min(left));
70            left -= granted;
71            self.grants.insert(*view, granted);
72        }
73        self.needs.retain(|view, _| views.contains(view));
74    }
75
76    /// Records the terrain tiles `view` drapes this frame, for the grants of the next.
77    fn need(&mut self, view: ViewSlotId, tiles: usize) {
78        self.needs.insert(view, tiles);
79    }
80}

82/// What the view being drawn may hold and aim for.
83#[derive(Clone, Copy, Debug, PartialEq, Eq)]
84pub(crate) struct DrapeAllowance {
85    /// Drape textures the view may hold.
86    pub(crate) textures: usize,
87    /// Drapes the view coarsens its covering to, leaving the textures it regroups with.
88    pub(crate) targets: usize,
89}
90
91/// The allowance of the view being drawn, which drapes `tiles` terrain tiles before
92/// coarsening: its grant of what `memory` allows the map.
93pub(crate) fn allowance(world: &mut World, memory: MemoryBudget, tiles: usize) -> DrapeAllowance {
94    let allowed = memory.drape_textures_allowed();
95    let view = world.resources.get::<DrapeView>().and_then(|view| view.0);
96    let Some(view) = view else {
97        return of_grant(allowed, allowed);
98    };
99    let shares = world.resources.get_or_init_mut::<DrapeShares>();
100    shares.need(view, tiles);
101    of_grant(
102        shares.grants.get(&view).copied().unwrap_or(allowed),
103        allowed,
104    )
105}
106
107/// The textures the view being drawn may hold.
108pub(crate) fn textures(world: &World, memory: MemoryBudget) -> usize {
109    let allowed = memory.drape_textures_allowed();
110    world
111        .resources
112        .get::<DrapeView>()
113        .and_then(|view| view.0)
114        .and_then(|view| {
115            world
116                .resources
117                .get::<DrapeShares>()?
118                .grants
119                .get(&view)
120                .copied()
121        })
122        .map_or(allowed, |granted| granted.min(allowed).max(1))
123}
124
125fn of_grant(granted: usize, allowed: usize) -> DrapeAllowance {
126    let textures = granted.min(allowed).max(1);
127    let regrouping = (REGROUPING_TEXTURES * textures).div_ceil(allowed.max(1));
128    DrapeAllowance {
129        textures,
130        targets: textures.saturating_sub(regrouping).max(1),
131    }
132}
