24/// Whether a style layer renders into drape textures rather than straight to the screen.
25pub fn is_drapeable(layer_type: &str) -> bool {
26    DRAPEABLE_LAYER_TYPES.contains(&layer_type)
27}

60/// One source tile drawn into a terrain tile's texture: vector data of the tile itself, or
61/// the raster tiles of one source, which covers the view at its own zoom.
62#[derive(Clone, Debug, PartialEq, Eq)]
63pub(crate) struct ShapeSource {
64    pub(crate) coords: WorldTileCoords,
65    pub(crate) raster_source: Option<RasterSourceId>,
66    /// See [`ShapeSpec::view_complete`].
67    pub(crate) view_complete: bool,
68    /// Marks a raster source with nothing over the terrain tile rather than a tile to draw.
69    pub(crate) absent: bool,
70}

72/// Pairs every view tile with the source tiles that hold its data, without the screen path's
73/// parent de-duplication: each drape texture needs its own copy of an ancestor's content.
74/// Raster tiles follow their source's covering, as GL JS pairs a source's visible tiles with
75/// the terrain tiles they overlap; the pyramid search only stands in while they load.
76pub(crate) fn select_targets(
77    coords: impl Iterator<Item = WorldTileCoords>,
78    world: &World,
79    raster_coverings: &RasterCoverings,
80) -> Vec<(WorldTileCoords, Vec<ShapeSource>)> {
81    let Some(sources) = world.resources.get::<ViewTileSources>() else {
82        return Vec::new();
83    };
84    let vector_sources = sources.of_kind(TileKind::Vector);
85    coords
86        .filter(|coords| coords.build_quad_key().is_some())
87        .map(|coords| {
88            let mut shapes: Vec<ShapeSource> =
89                coverage::loaded_or_partial_cover(&vector_sources, coords, world)
90                    .unwrap_or_default()
91                    .into_iter()
92                    .map(|source| ShapeSource {
93                        coords: source,
94                        raster_source: None,
95                        view_complete: false,
96                        absent: false,
97                    })
98                    .collect();
99            for (source_id, covering) in raster_coverings {
100                shapes.extend(raster_shapes(
101                    coords,
102                    source_id,
103                    covering,
104                    sources.of_kind(TileKind::Raster),
105                    world,
106                ));
107            }
108            (coords, shapes)
109        })
110        .collect()
111}

113/// Whether a raster source's request for a tile completed without an image (not found,
114/// failed, or undecodable), so the tile will not arrive and must not be waited for.
115fn source_tile_missing(source: &RasterSourceId, coords: WorldTileCoords, world: &World) -> bool {
116    world
117        .tiles
118        .query::<&crate::raster::RasterLayersDataComponent>(coords)
119        .is_some_and(|component| component.source_is_missing(source))
120}

122/// The tiles of one raster source drawn into a terrain tile's drape.
123///
124/// A source with nothing over the terrain tile is absent, so its layers do not hold the
125/// drape back: one whose view covering is empty (the view is outside its zoom range or
126/// bounds) as well as one whose covering misses this tile. When every wanted tile has
127/// either loaded or completed without an image, the source has delivered all it will; its
128/// loaded ancestors and tiles are drawn as final, or it is absent when it has none.
129fn raster_shapes<H: HasTile>(
130    coords: WorldTileCoords,
131    source_id: &RasterSourceId,
132    covering: &[WorldTileCoords],
133    availability: H,
134    world: &World,
135) -> Vec<ShapeSource> {
136    let absent = || {
137        vec![ShapeSource {
138            coords,
139            raster_source: Some(source_id.clone()),
140            view_complete: true,
141            absent: true,
142        }]
143    };
144    let raster_sources = SourceTiles {
145        source: source_id,
146        availability,
147    };
148    let wanted = covering_shapes_for(coords, covering);
149    if wanted.is_empty() {
150        return absent();
151    }
152    let loaded: Vec<WorldTileCoords> = wanted
153        .iter()
154        .copied()
155        .filter(|source| raster_sources.has_tile(*source, world))
156        .collect();
157    // A partial child set would erase the uncovered part when the drape is cleared,
158    // so an ancestor goes underneath it; the set itself is what the view needs
159    // over this terrain tile.
160    let all_loaded = loaded.len() == wanted.len();
161    let settled = wanted
162        .iter()
163        .all(|tile| loaded.contains(tile) || source_tile_missing(source_id, *tile, world));
164    let complete = coverage::complete_cover(coords, loaded.clone());
165    let (under, over, view_complete) = match complete {
166        Some(complete) => (Vec::new(), complete, false),
167        None if all_loaded || settled => (
168            coverage::loaded_cover(&raster_sources, coords, world).unwrap_or_default(),
169            loaded,
170            true,
171        ),
172        None => (
173            coverage::loaded_cover(&raster_sources, coords, world).unwrap_or_default(),
174            Vec::new(),
175            false,
176        ),
177    };
178    if settled && !all_loaded && under.is_empty() && over.is_empty() {
179        return absent();
180    }
181    let shape = |coords, view_complete| ShapeSource {
182        coords,
183        raster_source: Some(source_id.clone()),
184        view_complete,
185        absent: false,
186    };
187    under
188        .into_iter()
189        .map(|source| shape(source, false))
190        .chain(over.into_iter().map(|source| shape(source, view_complete)))
191        .collect()
192}

194/// Lists the drapeable layers each source tile currently holds.
195pub(crate) fn collect_layer_specs(
196    targets: Vec<(WorldTileCoords, Vec<ShapeSource>)>,
197    style: &Style,
198    world: &World,
199    zoom: f64,
200) -> Vec<TargetSpec> {
201    let vector = world.resources.get::<Eventually<VectorBufferPool>>();
202    let raster = world.resources.get::<Eventually<RasterResources>>();
203    let raster_layers: Vec<_> = style
204        .layers
205        .iter()
206        .filter(|layer| {
207            RASTER_LAYER_TYPES.contains(&layer.type_.as_str()) && layer.is_visible_at(zoom)
208        })
209        .filter_map(|layer| {
210            let Some(Initialized(resources)) = raster else {
211                return None;
212            };
213            Some((
214                layer.id.clone(),
215                layer.index,
216                resources.layer_source(&layer.id)?.clone(),
217                dem_layer_kind(&layer.type_).is_some(),
218            ))
219        })
220        .collect();
221    targets
222        .into_iter()
223        .map(|(coords, shapes)| TargetSpec {
224            coords,
225            absent_sources: shapes
226                .iter()
227                .filter(|shape| shape.absent)
228                .filter_map(|shape| shape.raster_source.clone())
229                .collect(),
230            shapes: shapes
231                .into_iter()
232                .filter(|shape| !shape.absent)
233                .map(|shape| {
234                    let raster_layers = raster_layers.iter()
235                        .filter(|(_, _, source, _)| shape.raster_source.as_ref() == Some(source))
236                        .filter(|(id, _, _, _)| matches!(raster, Some(Initialized(resources)) if resources.layer_texture(id, &shape.coords).is_some()))
237                        .map(|(id, index, _, dem)| (id.clone(), *index, *dem))
238                        .collect();
239                    ShapeSpec {
240                        view_complete: shape.view_complete,
241                        source: shape.coords,
242                        vector_layers: match (&shape.raster_source, vector) {
243                            (None, Some(Initialized(pool))) => {
244                                vector_layer_specs(pool, shape.coords, zoom)
245                            }
246                            _ => Vec::new(),
247                        },
248                        raster_layers,
249                    }
250                })
251                .collect(),
252        })
253        .collect()
254}

256fn vector_layer_specs(
257    pool: &VectorBufferPool,
258    source: WorldTileCoords,
259    zoom: f64,
260) -> Vec<VectorLayerSpec> {
261    pool.index()
262        .get_layers(source)
263        .into_iter()
264        .flatten()
265        .filter(|entry| {
266            entry.style_layer.is_visible_at(zoom)
267                && is_drapeable(&entry.style_layer.type_)
268                && crate::vector::structures::kind(&entry.style_layer).is_none()
269        })
270        .map(|entry| VectorLayerSpec {
271            id: entry.style_layer.id.clone(),
272            index: entry.style_layer.index,
273            is_line: entry.style_layer.type_ == "line",
274            coords: entry.coords,
275        })
276        .collect()
