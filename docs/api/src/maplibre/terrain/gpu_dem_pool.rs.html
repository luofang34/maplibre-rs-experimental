1//! Shared GPU `(dim + 4) x (dim + 4)` elevation texture pool across terrain, hillshade, and colour relief.
2
3use std::collections::HashMap;
4
5use crate::{
6    coords::WorldTileCoords,
7    raster::RasterSourceId,
8    render::{resource::Texture, settings::Msaa, upload_queue::UploadQueue},
9    terrain::{DemRevision, DemTile, LoadedDem},
10};

12/// One resident GPU elevation texture in [`GpuDemPool`].
13pub struct GpuDemEntry {
14    /// `(dim + 4) x (dim + 4)` single-mip `Rgba8Unorm` texture holding 2-texel bordered DEM pixels.
15    pub texture: Texture,
16    /// Wrapping sample revision matching the uploaded [`LoadedDem::revision`].
17    pub revision: u32,
18    pub(crate) source_key: DemRevision,
19}
20
21/// Canonical GPU elevation texture registry shared by [`TerrainResources`](super::resources::TerrainResources)
22/// and [`RasterResources`](crate::raster::resource::RasterResources).
23///
24/// A `raster-dem` source shared by 3D terrain, `hillshade`, and `color-relief` allocates and
25/// uploads each `(source, coords)` DEM texture once on the GPU, updating it in place via
26/// [`UploadQueue::write_texture`] when neighbour borders backfill.
27#[derive(Default)]
28pub struct GpuDemPool {
29    textures: HashMap<(RasterSourceId, WorldTileCoords), GpuDemEntry>,
30    uploads: u64,
31}
32
33impl GpuDemPool {
34    /// Whether `(source, coords)` is missing from the GPU or holds a different [`LoadedDem`] revision.
35    pub fn needs_upload(
36        &self,
37        source: &RasterSourceId,
38        coords: WorldTileCoords,
39        dem: &LoadedDem,
40    ) -> bool {
41        let expected = dem.revision_key();
42        !self
43            .textures
44            .get(&(source.clone(), coords))
45            .is_some_and(|entry| entry.source_key == expected)
46    }
47
48    /// Borrows the resident GPU entry for `(source, coords)`, if uploaded.
49    pub fn get(&self, source: &RasterSourceId, coords: WorldTileCoords) -> Option<&GpuDemEntry> {
50        self.textures.get(&(source.clone(), coords))
51    }
52
53    /// Uploads `dem` into the shared GPU pool, reusing an existing texture of matching stride in place.
54    pub fn upload(
55        &mut self,
56        device: &wgpu::Device,
57        queue: &UploadQueue,
58        source: &RasterSourceId,
59        coords: WorldTileCoords,
60        dem: &LoadedDem,
61    ) -> &GpuDemEntry {
62        let key = (source.clone(), coords);
63        let source_key = dem.revision_key();
64        let stride = dem.tile.stride();
65        let reusable = self
66            .textures
67            .remove(&key)
68            .map(|entry| entry.texture)
69            .filter(|texture| texture.size.width == stride);
70        let texture = reusable.unwrap_or_else(|| allocate_dem_texture(device, stride));
71        write_dem_pixels(queue, &texture, &dem.tile);
72        self.uploads = self.uploads.wrapping_add(1);
73        self.textures.entry(key).or_insert(GpuDemEntry {
74            texture,
75            revision: dem.revision,
76            source_key,
77        })
78    }
79
80    /// Drops every source's GPU DEM texture at `coords`.
81    pub fn drop_coords(&mut self, coords: WorldTileCoords) {
82        self.textures.retain(|(_, at), _| *at != coords);
83    }
84
85    /// Drops one source's GPU DEM texture at `coords`.
86    pub fn drop_source_coords(&mut self, source: &RasterSourceId, coords: WorldTileCoords) {
87        self.textures.remove(&(source.clone(), coords));
88    }
89
90    /// Drops every GPU DEM texture belonging to `source`.
91    pub fn drop_source(&mut self, source: &RasterSourceId) {
92        self.textures.retain(|(id, _), _| id != source);
93    }
94
95    /// Number of unique `(source, coords)` DEM textures currently resident in GPU memory.
96    pub fn texture_count(&self) -> usize {
97        self.textures.len()
98    }
99
100    /// Total GPU texture memory in bytes held by resident DEM tiles in this pool.
101    pub fn texture_bytes(&self) -> usize {
102        self.textures
103            .values()
104            .map(|entry| (entry.texture.size.width * entry.texture.size.height * 4) as usize)
105            .sum()
106    }
107
108    /// Cumulative `write_texture` uploads performed through this pool.
109    pub fn upload_count(&self) -> u64 {
110        self.uploads
111    }
112}
113
114pub(crate) fn allocate_dem_texture(device: &wgpu::Device, stride: u32) -> Texture {
115    Texture::new(
116        Some("DEM tile"),
117        device,
118        wgpu::TextureFormat::Rgba8Unorm,
119        stride,
120        stride,
121        Msaa { samples: 1 },
122        wgpu::TextureUsages::TEXTURE_BINDING | wgpu::TextureUsages::COPY_DST,
123    )
124}
125
126pub(crate) fn write_dem_pixels(queue: &UploadQueue, texture: &Texture, dem: &DemTile) {
127    queue.write_texture(
128        wgpu::TexelCopyTextureInfo {
129            texture: &texture.texture,
130            mip_level: 0,
131            origin: wgpu::Origin3d::ZERO,
132            aspect: wgpu::TextureAspect::All,
133        },
134        dem.pixels(),
135        wgpu::TexelCopyBufferLayout {
136            offset: 0,
137            bytes_per_row: Some(4 * dem.stride()),
138            rows_per_image: Some(dem.stride()),
139        },
140        texture.size,
141    );
142}
