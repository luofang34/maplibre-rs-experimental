1//! Lines of sight across the drawn globe terrain: whether a point in the air or on the ground
2//! can be seen from the camera, and where a screen ray first meets the ground.
3//!
4//! A line is followed from the eye through the rendered tiles of the coverage index, with the
5//! DEM, exaggeration and ancestor fallback the terrain is drawn with, and meets the ground on
6//! the triangles the GPU draws. Where the line runs above a tile's highest ground it skips
7//! ahead to where it could first reach it; near the ground it advances half a mesh cell at a
8//! time and tests every triangle it passes over, so a ridge is not stepped over however long
9//! and flat the line runs. A rendered tile no DEM describes yet is unknown,
10//! neither sea level nor clear: a line that passes below the body's highest ground there
11//! cannot be called unobstructed. Ground the frame does not draw at all, outside the rendered
12//! tiles, hides nothing, as in the picture.
13//!
14//! These lines of sight are the reference for what the frame hides. The overlay's depth test
15//! and its depth probes stand in for them on the GPU, at no cost per point, and differ from
16//! them only by the depth bias and the pixel the GPU tests a point at, as [`crate::overlay`]
17//! describes.

41/// Latitude beyond which the globe is closed by the polar caps at sea level.
42const MAX_VALID_LATITUDE: f64 = 85.051_128_779_806_59;
43const MAX_MERCATOR_Y: f64 = 1.0 - 1e-9;
44/// Steps per mesh cell near the ground.
45const STEPS_PER_CELL: f64 = 2.0;
46const CELLS: f64 = crate::terrain::mesh::TERRAIN_MESH_SIZE as f64;
47/// Smallest step in metres, so a line converging on a bound still advances.
48const MIN_STEP_METERS: f64 = 0.01;

52pub struct DrawnTerrain<'a> {
53    /// The rendered tiles and the DEM tiles behind them.
54    pub index: &'a TerrainCoverageIndex,
55    /// Tiles holding the loaded DEM data.
56    pub tiles: &'a Tiles,
57    /// The body the terrain lies on.
58    pub body: Body,
59}

61/// Where a line first meets the drawn ground.
62#[derive(Clone, Copy, Debug, PartialEq)]
63pub(crate) struct GroundCrossing {
64    /// Distance along the line, in unit-sphere lengths.
65    pub t: f64,
66    /// The ground point.
67    pub location: LatLon,
68    /// Drawn elevation of the ground there in metres, exaggeration included.
69    pub elevation: f64,
70    /// Whether the ground is a polar cap rather than a tile.
71    pub polar: bool,
72}

74/// What a line of sight passes before its end.
75#[derive(Clone, Copy, Debug, Default, PartialEq)]
76pub(crate) struct Crossing {
77    /// The first drawn ground it meets.
78    pub ground: Option<GroundCrossing>,
79    /// Where it first runs below the body's highest ground over a rendered tile no DEM
80    /// describes, if that is before `ground`.
81    pub unknown: Option<f64>,
82}

84/// The ground under one point of a line.
85struct Probe {
86    mercator: Point2<f64>,
87    /// Height of the point above the mean radius in metres.
88    height: f64,
89    /// Distance of the point from the body's center, in radii.
90    radius: f64,
91    polar: bool,
92    /// The rendered tile under the point, and whether its DEM has loaded.
93    tile: Option<(WorldTileCoords, bool)>,
94}
95
96impl DrawnTerrain<'_> {
97    /// Highest ground anywhere, drawn or not yet described, in metres.
98    pub(crate) fn ceiling(&self) -> f64 {
99        self.index
100            .max_elevation()
101            .max(self.body.highest_ground_meters * self.index.exaggeration().max(1.0))
102    }
103
104    /// Height of the drawn ground at a location: where the line up from the body's center
105    /// meets the triangles drawn there. `None` outside the rendered tiles and where the DEM
106    /// has not loaded.
107    pub(crate) fn ground_at(&self, location: LatLon) -> Option<f64> {
108        let up = lat_lon_to_unit_sphere(location);
109        let mercator = lat_lon_to_mercator(location);
110        let cells = self.cells_at(mercator, location.latitude.abs() > MAX_VALID_LATITUDE);
111        cells
112            .into_iter()
113            .flat_map(|cell| self.triangles(cell))
114            .filter_map(|triangle| mesh::intersect(Vector3::new(0.0, 0.0, 0.0), up, triangle))
115            .reduce(f64::max)
116            .map(|radius| (radius - 1.0) * self.body.radius_meters)
117    }
118
119    /// The drawn cells over `mercator`: the rendered tile's grid cell, or under a polar cap
120    /// the fan of the last tile in its column.
121    fn cells_at(&self, mercator: Point2<f64>, polar: bool) -> Vec<Cell> {
122        if polar {
123            let edge = if mercator.y < 0.5 {
124                0.0
125            } else {
126                MAX_MERCATOR_Y
127            };
128            return self
129                .index
130                .rendered_tile_at(mercator.x, edge)
131                .map(|tile| Cell {
132                    row: None,
133                    ..Cell::of(tile, mercator)
134                })
135                .into_iter()
136                .collect();
137        }
138        self.index
139            .rendered_tile_at(mercator.x, mercator.y)
140            .map(|tile| Cell::of(tile, mercator))
141            .into_iter()
142            .collect()
143    }
144
145    fn probe(&self, point: Vector3<f64>) -> Probe {
146        let radius = point.magnitude();
147        let location = unit_sphere_to_lat_lon(point / radius);
148        let mercator = lat_lon_to_mercator(location);
149        let polar = location.latitude.abs() > MAX_VALID_LATITUDE;
150        let tile = (!polar)
151            .then(|| self.index.rendered_tile_at(mercator.x, mercator.y))
152            .flatten()
153            .map(|tile| {
154                let sample = self
155                    .index
156                    .sample_in(self.tiles, tile, mercator.x, mercator.y);
157                (tile, sample.dem_loaded)
158            });
159        Probe {
160            mercator,
161            height: (radius - 1.0) * self.body.radius_meters,
162            radius,
163            polar,
164            tile,
165        }
166    }
167
168    /// Follows the line `eye + direction * t` for `t` in `start..end`, unit-sphere lengths,
169    /// and reports the first drawn ground it meets and any unknown ground it passes before.
170    /// Ground for which `skip` holds is passed through, as a target's own ground is.
171    pub(crate) fn follow(
172        &self,
173        eye: Vector3<f64>,
174        direction: Vector3<f64>,
175        (start, end): (f64, f64),
176        skip: impl Fn(LatLon) -> bool,
177    ) -> Crossing {
178        let mut crossing = Crossing::default();
179        // The stretch from the previous point, unless it was proven clear of ground.
180        let mut previous: Option<(f64, Probe)> = None;
181        // Neighbouring stretches pass over the same cells.
182        let mut drawn: HashMap<Cell, Vec<[Vector3<f64>; 3]>> = HashMap::new();
183        let mut t = start;
184        loop {
185            let probe = self.probe(eye + direction * t);
186            if let Some((from, last)) = &previous {
187                let ends = [last, &probe];
188                let hit = self.first_triangle(eye, direction, (*from, t), ends, &skip, &mut drawn);
189                if hit.is_some() {
190                    crossing.ground = hit;
191                    return crossing;
192                }
193            }
194            let unknown = matches!(probe.tile, Some((_, false))) && probe.height <= self.ceiling();
195            if unknown && crossing.unknown.is_none() {
196                crossing.unknown = Some(t);
197            }
198            if t >= end {
199                return crossing;
200            }
201            let (step, clear) = self.safe_step(&probe);
202            let next = (t + step).min(end);
203            previous = (!clear).then_some((t, probe));
204            t = next;
205        }
206    }
207
208    /// The first drawn triangle the stretch `from..to` of the line crosses, among the cells
209    /// under its two ends. A stretch near the ground is shorter than half a cell, so the
210    /// cells under its ends and the two between them diagonally hold all it passes over.
211    fn first_triangle(
212        &self,
213        eye: Vector3<f64>,
214        direction: Vector3<f64>,
215        (from, to): (f64, f64),
216        ends: [&Probe; 2],
217        skip: &impl Fn(LatLon) -> bool,
218        drawn: &mut HashMap<Cell, Vec<[Vector3<f64>; 3]>>,
219    ) -> Option<GroundCrossing> {
220        let mut cells: Vec<Cell> = Vec::new();
221        let mut seen: std::collections::HashSet<Cell> = std::collections::HashSet::new();
222        let mut add = |cell: Cell| {
223            if seen.insert(cell) {
224                cells.push(cell);
225            }
226        };
227        for end in ends {
228            for cell in self.cells_at(end.mercator, end.polar) {
229                for other in ends {
230                    let mut corner = Cell::of(cell.tile, other.mercator);
231                    if cell.row.is_none() {
232                        corner.row = None;
233                    }
234                    let crossed = [
235                        cell,
236                        corner,
237                        Cell {
238                            column: corner.column,
239                            ..cell
240                        },
241                        Cell {
242                            column: cell.column,
243                            row: corner.row,
244                            ..cell
245                        },
246                    ];
247                    for crossed in crossed {
248                        add(crossed);
249                    }
250                }
251            }
252        }
253        // Near a pole a short stretch sweeps across many sectors of the cap's fan; every sector
254        // between the longitudes of its ends is tested.
255        if let Some(polar) = ends.iter().find(|end| end.polar) {
256            let edge = if polar.mercator.y < 0.5 {
257                0.0
258            } else {
259                MAX_MERCATOR_Y
260            };
261            let sector =
262                1.0 / (2_f64.powi(i32::from(self.index.finest_zoom().unwrap_or(0))) * CELLS);
263            let from_x = ends[0].mercator.x;
264            let arc = (ends[1].mercator.x - from_x + 0.5).rem_euclid(1.0) - 0.5;
265            let steps = (arc.abs() / (sector * 0.5)).ceil() as usize;
266            for step in 0..=steps {
267                let x = from_x + arc * step as f64 / steps.max(1) as f64;
268                for cell in self.cells_at(Point2::new(x - x.floor(), edge), true) {
269                    add(cell);
270                }
271            }
272        }
273        let slack = 1e-12;
274        cells
275            .into_iter()
276            .flat_map(|cell| {
277                let polar = cell.row.is_none();
278                drawn
279                    .entry(cell)
280                    .or_insert_with(|| self.triangles(cell))
281                    .clone()
282                    .into_iter()
283                    .map(move |triangle| (triangle, polar))
284            })
285            .filter_map(|(triangle, polar)| {
286                let t = mesh::intersect(eye, direction, triangle)?;
287                (t >= from - slack && t <= to + slack).then_some((t, polar))
288            })
289            .filter_map(|(t, polar)| {
290                let point = eye + direction * t;
291                let location = unit_sphere_to_lat_lon(point.normalize());
292                (!skip(location)).then(|| GroundCrossing {
293                    t,
294                    location,
295                    elevation: (point.magnitude() - 1.0) * self.body.radius_meters,
296                    polar,
297                })
298            })
299            .min_by(|a, b| a.t.total_cmp(&b.t))
300    }
301
302    /// How far along a line, in unit-sphere lengths, the point of `probe` can move without
303    /// passing drawn ground it could meet.
304    ///
305    /// A straight line's height above a sphere falls by at most the distance travelled, so a
306    /// point that far above the highest ground of its tile cannot reach that ground before
307    /// it has moved as far, nor leave the tile before it has moved the distance to its edge,
308    /// scaled down when the point lies below the mean radius and its ground track runs faster.
309    /// Near the ground it moves half a mesh cell, so the cells under the ends of each stretch
310    /// hold every triangle the stretch passes over. Returns the step and whether the stretch
311    /// it covers is proven clear of ground.
312    fn safe_step(&self, probe: &Probe) -> (f64, bool) {
313        let radius = self.body.radius_meters;
314        let minimum = MIN_STEP_METERS / radius;
315        let bounded = |bound: f64, fine: f64| (bound.max(fine).max(minimum), bound >= fine);
316        if probe.polar {
317            return bounded(
318                (probe.height - self.ceiling()) / radius,
319                self.finest_step(probe.mercator),
320            );
321        }
322        let Some((tile, _)) = probe.tile else {
323            // Outside the rendered tiles nothing is drawn until the line enters one, and every
324            // rendered tile's edges lie on the grid of the finest zoom.
325            let zoom = self.index.finest_zoom().unwrap_or(0);
326            let scale = 2_f64.powi(i32::from(zoom));
327            let cell = WorldTileCoords {
328                x: ((probe.mercator.x - probe.mercator.x.floor()) * scale).floor() as i32,
329                y: (probe.mercator.y * scale).floor() as i32,
330                z: crate::coords::ZoomLevel::new(zoom),
331            };
332            let inside = track_budget(
333                TileFootprint::of(cell).distance_to_edge(probe.mercator),
334                probe.radius,
335            );
336            let (step, _) = bounded(inside, self.finest_step(probe.mercator));
337            // Nothing is drawn here, but a step past the cell may reach a rendered tile.
338            return (step, inside >= step);
339        };
340        let footprint = TileFootprint::of(tile);
341        let edge = footprint.distance_to_edge(probe.mercator);
342        // Within a cell of the edge the neighbour may be several zooms finer, its cells a
343        // fraction of this tile's; a stretch into it advances half of the finest cell, so the
344        // cells under its ends still hold everything it passes over.
345        let fine = if edge < footprint.width / CELLS {
346            self.finest_step(probe.mercator)
347        } else {
348            footprint.width / (CELLS * STEPS_PER_CELL)
349        };
350        let highest = self
351            .index
352            .tile_elevation_range(tile)
353            .map_or(self.ceiling(), |range| range.max_meters);
354        let above = (probe.height - highest) / radius;
355        bounded(above.min(track_budget(edge, probe.radius)), fine)
356    }
357
358    /// Half the width of the finest rendered cell at `mercator`, measured on the pole-ward
359    /// row of the finest zoom's tile there, the narrowest any rendered cell can be; over a cap,
360    /// the last row's.
361    fn finest_step(&self, mercator: Point2<f64>) -> f64 {
362        let zoom = self.index.finest_zoom().unwrap_or(0);
363        let scale = 2_f64.powi(i32::from(zoom));
364        let tile = WorldTileCoords {
365            x: ((mercator.x - mercator.x.floor()) * scale).floor() as i32,
366            y: (mercator.y.clamp(0.0, MAX_MERCATOR_Y) * scale).floor() as i32,
367            z: crate::coords::ZoomLevel::new(zoom),
368        };
369        TileFootprint::of(tile).width / (CELLS * STEPS_PER_CELL)
370    }
371}

373/// How far a point `radius` radii from the body's center can move along a straight line
374/// while its ground track moves at most `angle` radians: the track turns at most the distance
375/// over the radius, faster than the distance itself below the mean radius.
376fn track_budget(angle: f64, radius: f64) -> f64 {
377    angle * radius.min(1.0)
378}

380/// A tile's extent in Mercator units and the shortest ground widths it spans, in radians.
381struct TileFootprint {
382    west: f64,
383    east: f64,
384    north: f64,
385    south: f64,
386    /// Radians per Mercator unit where the tile is narrowest: its pole-ward row.
387    radians_per_unit: f64,
388    /// Ground width of the tile in radians along its narrowest row.
389    width: f64,
390}
391
392impl TileFootprint {
393    fn of(tile: WorldTileCoords) -> Self {
394        let scale = 2_f64.powi(i32::from(u8::from(tile.z)));
395        let (north, south) = (f64::from(tile.y) / scale, f64::from(tile.y + 1) / scale);
396        let poleward = if north + south < 1.0 { north } else { south };
397        let radians_per_unit = std::f64::consts::TAU * mercator_y_to_latitude(poleward).cos();
398        Self {
399            west: f64::from(tile.x) / scale,
400            east: f64::from(tile.x + 1) / scale,
401            north,
402            south,
403            radians_per_unit,
404            width: radians_per_unit / scale,
405        }
406    }
407
408    /// Ground distance in radians from `mercator` to the nearest edge of the tile, at least.
409    fn distance_to_edge(&self, mercator: Point2<f64>) -> f64 {
410        let x = mercator.x - mercator.x.floor();
411        let across = (x - self.west).min(self.east - x);
412        let along = (mercator.y - self.north).min(self.south - mercator.y);
413        across.min(along).max(0.0) * self.radians_per_unit
414    }
415}

417/// Mercator coordinates in `0..1` of a geographic location.
418pub(crate) fn lat_lon_to_mercator(location: LatLon) -> Point2<f64> {
419    let x = location.longitude / 360.0 + 0.5;
420    let y = (1.0 - location.latitude.to_radians().tan().asinh() / std::f64::consts::PI) * 0.5;
421    Point2::new(x, y.clamp(0.0, MAX_MERCATOR_Y))
422}
423
424/// The location of Mercator coordinates in `0..1`.
425pub(crate) fn mercator_to_lat_lon(mercator: Point2<f64>) -> LatLon {
426    LatLon::new(
427        mercator_y_to_latitude(mercator.y).to_degrees(),
428        mercator.x * 360.0 - 180.0,
429    )
430}
431
432fn mercator_y_to_latitude(y: f64) -> f64 {
433    (std::f64::consts::PI * (1.0 - 2.0 * y)).sinh().atan()
434}
