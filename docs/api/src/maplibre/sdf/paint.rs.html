27/// Whether text along a line is placed glyph by glyph: it turns with the map, whether it lies
28/// on the map plane or stands upright to the viewer, or each glyph stands upright on the screen
29/// along the line (`viewport-glyph`). Text aligned with the viewport keeps the straight layout
30/// at the anchor.
31pub(crate) fn text_follows_line(paint: &SymbolPaint, zoom: f64) -> bool {
32    SymbolUniforms::new(paint, zoom, [1, 1]).text_layout[1] > 0.5 || glyphs_stay_upright(paint)
33}
34
35/// Whether the glyphs of text along a line are placed along it each upright on the screen, as
36/// GL JS's `viewport-glyph` rotation alignment places them: road shields in a row stay level
37/// while the row follows its road.
38pub(crate) fn glyphs_stay_upright(paint: &SymbolPaint) -> bool {
39    let read = |name: &str| paint.properties.get(name).and_then(|value| value.as_str());
40    read("text-rotation-alignment") == Some("viewport-glyph")
41        && read("symbol-placement").is_some_and(|value| matches!(value, "line" | "line-center"))
42}

44impl SymbolUniforms {
45    pub fn new(paint: &SymbolPaint, zoom: f64, size: [u32; 2]) -> Self {
46        let properties = FeatureProperties::new();
47        let number = |name, fallback| paint.number(name, &properties, zoom, fallback);
48        Self {
49            text_color: color(paint, "text-color", zoom, [0.0, 0.0, 0.0, 1.0]),
50            halo_color: color(paint, "text-halo-color", zoom, [0.0; 4]),
51            icon_color: color(paint, "icon-color", zoom, [0.0, 0.0, 0.0, 1.0]),
52            icon_halo_color: color(paint, "icon-halo-color", zoom, [0.0; 4]),
53            text: [
54                paint
55                    .text_size
56                    .as_ref()
57                    .and_then(|value| value.evaluate_at_zoom(zoom))
58                    .unwrap_or(16.0),
59                number("text-halo-width", 0.0),
60                number("text-halo-blur", 0.0),
61                number("text-opacity", 1.0),
62            ],
63            icon: [
64                number("icon-size", 1.0),
65                number("icon-halo-width", 0.0),
66                number("icon-halo-blur", 0.0),
67                number("icon-opacity", 1.0),
68            ],
69            text_layout: layout(paint, "text"),
70            icon_layout: layout(paint, "icon"),
71            atlas: [size[0] as f32, size[1] as f32, 0.0, 0.0],
72            placement: [
73                number("text-padding", 2.0),
74                number("icon-padding", 2.0),
75                keep_upright(paint, "text", true),
76                keep_upright(paint, "icon", false),
77            ],
78            translate: {
79                let [text_x, text_y] = crate::sdf::translation::viewport_shift(paint, "text", zoom);
80                let [icon_x, icon_y] = crate::sdf::translation::viewport_shift(paint, "icon", zoom);
81                [text_x, text_y, icon_x, icon_y]
82            },
83        }
84    }
85}

87impl SymbolUniforms {
88    /// The uniforms with `text-size` and `icon-size` evaluated for one feature's properties:
89    /// a data-driven size has no value for the layer as a whole, and a label collides by the
90    /// size it is drawn at.
91    pub fn with_feature_sizes(
92        mut self,
93        paint: &SymbolPaint,
94        properties: &FeatureProperties,
95        zoom: f64,
96    ) -> Self {
97        self.text[0] = feature_style(paint, "text", properties, zoom)[2][0];
98        self.icon[0] = feature_style(paint, "icon", properties, zoom)[2][0];
99        self
100    }
101}

103/// The fill colour, halo colour and size, halo width, halo blur and opacity of a feature's text
104/// (`prefix` "text") or icon (`prefix` "icon"), each evaluated for its properties.
105pub(crate) fn feature_style(
106    paint: &SymbolPaint,
107    prefix: &str,
108    properties: &FeatureProperties,
109    zoom: f64,
110) -> [[f32; 4]; 3] {
111    let color = |name: &str, fallback: [f32; 4]| {
112        paint
113            .properties
114            .get(&format!("{prefix}-{name}"))
115            .and_then(|value| {
116                StyleProperty::<csscolorparser::Color>::parse(value).evaluate_for(properties, zoom)
117            })
118            .map_or(fallback, |color| {
119                let [r, g, b, a] = color.to_array();
120                [r as f32, g as f32, b as f32, a as f32]
121            })
122    };
123    let number = |name: &str, fallback: f32| {
124        paint.number(&format!("{prefix}-{name}"), properties, zoom, fallback)
125    };
126    let size = if prefix == "text" {
127        paint
128            .text_size
129            .as_ref()
130            .and_then(|value| value.evaluate_for(properties, zoom))
131            .unwrap_or(16.0)
132    } else {
133        number("size", 1.0)
134    };
135    [
136        color("color", [0.0, 0.0, 0.0, 1.0]),
137        color("halo-color", [0.0; 4]),
138        [
139            size,
140            number("halo-width", 0.0),
141            number("halo-blur", 0.0),
142            number("opacity", 1.0),
143        ],
144    ]
145}

147fn color(paint: &SymbolPaint, name: &str, zoom: f64, fallback: [f32; 4]) -> [f32; 4] {
148    let Some(color) = paint.properties.get(name).and_then(|value| {
149        StyleProperty::<csscolorparser::Color>::parse(value).evaluate_at_zoom(zoom)
150    }) else {
151        return fallback;
152    };
153    let [r, g, b, a] = color.to_array();
154    [r as f32, g as f32, b as f32, a as f32]
155}

157fn layout(paint: &SymbolPaint, prefix: &str) -> [f32; 4] {
158    let read = |suffix: &str| {
159        paint
160            .properties
161            .get(&format!("{prefix}-{suffix}"))
162            .and_then(|value| value.as_str())
163    };
164    let rotation = read("rotation-alignment");
165    let map_rotation = rotation == Some("map")
166        || (rotation.unwrap_or("auto") == "auto"
167            && paint
168                .properties
169                .get("symbol-placement")
170                .and_then(|value| value.as_str())
171                .is_some_and(|value| matches!(value, "line" | "line-center")));
172    let pitch = read("pitch-alignment");
173    let map_pitch = pitch == Some("map") || (pitch.unwrap_or("auto") == "auto" && map_rotation);
174    [
175        u32::from(map_pitch) as f32,
176        u32::from(map_rotation) as f32,
177        // The tessellator bakes the rotation into each symbol's corners.
178        0.0,
179        u32::from(paint.uses_shared_height() || paint.height_follows_ground(prefix)) as f32,
180    ]
181}
182
183fn keep_upright(paint: &SymbolPaint, prefix: &str, fallback: bool) -> f32 {
184    let line = paint
185        .properties
186        .get("symbol-placement")
187        .and_then(|v| v.as_str())
188        .is_some_and(|value| matches!(value, "line" | "line-center"));
189    f32::from(
190        line && paint
191            .properties
192            .get(&format!("{prefix}-keep-upright"))
193            .and_then(|v| v.as_bool())
194            .unwrap_or(fallback),
195    )
196}
