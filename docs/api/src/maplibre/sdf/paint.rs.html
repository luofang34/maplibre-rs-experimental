24/// Whether text along a line is placed glyph by glyph: it turns with the map, whether it lies
25/// on the map plane or stands upright to the viewer, or each glyph stands upright on the screen
26/// along the line (`viewport-glyph`). Text aligned with the viewport keeps the straight layout
27/// at the anchor.
28pub(crate) fn text_follows_line(paint: &SymbolPaint, zoom: f64) -> bool {
29    SymbolUniforms::new(paint, zoom, [1, 1]).text_layout[1] > 0.5 || glyphs_stay_upright(paint)
30}
31
32/// Whether the glyphs of text along a line are placed along it each upright on the screen, as
33/// GL JS's `viewport-glyph` rotation alignment places them: road shields in a row stay level
34/// while the row follows its road.
35pub(crate) fn glyphs_stay_upright(paint: &SymbolPaint) -> bool {
36    let read = |name: &str| paint.properties.get(name).and_then(|value| value.as_str());
37    read("text-rotation-alignment") == Some("viewport-glyph")
38        && read("symbol-placement").is_some_and(|value| matches!(value, "line" | "line-center"))
39}

41impl SymbolUniforms {
42    pub fn new(paint: &SymbolPaint, zoom: f64, size: [u32; 2]) -> Self {
43        let properties = FeatureProperties::new();
44        let number = |name, fallback| paint.number(name, &properties, zoom, fallback);
45        Self {
46            text_color: color(paint, "text-color", zoom, [0.0, 0.0, 0.0, 1.0]),
47            halo_color: color(paint, "text-halo-color", zoom, [0.0; 4]),
48            icon_color: color(paint, "icon-color", zoom, [0.0, 0.0, 0.0, 1.0]),
49            icon_halo_color: color(paint, "icon-halo-color", zoom, [0.0; 4]),
50            text: [
51                paint
52                    .text_size
53                    .as_ref()
54                    .and_then(|value| value.evaluate_at_zoom(zoom))
55                    .unwrap_or(16.0),
56                number("text-halo-width", 0.0),
57                number("text-halo-blur", 0.0),
58                number("text-opacity", 1.0),
59            ],
60            icon: [
61                number("icon-size", 1.0),
62                number("icon-halo-width", 0.0),
63                number("icon-halo-blur", 0.0),
64                number("icon-opacity", 1.0),
65            ],
66            text_layout: layout(paint, "text"),
67            icon_layout: layout(paint, "icon"),
68            atlas: [size[0] as f32, size[1] as f32, 0.0, 0.0],
69            placement: [
70                number("text-padding", 2.0),
71                number("icon-padding", 2.0),
72                keep_upright(paint, "text", true),
73                keep_upright(paint, "icon", false),
74            ],
75            translate: {
76                let [text_x, text_y] = crate::sdf::translation::viewport_shift(paint, "text", zoom);
77                let [icon_x, icon_y] = crate::sdf::translation::viewport_shift(paint, "icon", zoom);
78                [text_x, text_y, icon_x, icon_y]
79            },
80        }
81    }
82}

84impl SymbolUniforms {
85    /// The uniforms with `text-size` and `icon-size` evaluated for one feature's properties:
86    /// a data-driven size has no value for the layer as a whole, and a label collides by the
87    /// size it is drawn at.
88    pub fn with_feature_sizes(
89        mut self,
90        paint: &SymbolPaint,
91        properties: &FeatureProperties,
92        zoom: f64,
93    ) -> Self {
94        let text_data_driven = paint
95            .text_size
96            .as_ref()
97            .is_some_and(|s| !s.is_feature_constant());
98        let icon_data_driven = paint
99            .compiled_property::<f32>("icon-size")
100            .is_some_and(|s| !s.is_feature_constant());
101        if !text_data_driven && !icon_data_driven {
102            return self;
103        }
104        if text_data_driven {
105            self.text[0] = feature_style(paint, "text", properties, zoom)[2][0];
106        }
107        if icon_data_driven {
108            self.icon[0] = feature_style(paint, "icon", properties, zoom)[2][0];
109        }
110        self
111    }
112}

114/// The fill colour, halo colour and size, halo width, halo blur and opacity of a feature's text
115/// (`prefix` "text") or icon (`prefix` "icon"), each evaluated for its properties.
116pub(crate) fn feature_style(
117    paint: &SymbolPaint,
118    prefix: &str,
119    properties: &FeatureProperties,
120    zoom: f64,
121) -> [[f32; 4]; 3] {
122    let color = |name: &str, fallback: [f32; 4]| {
123        let full_name = format!("{prefix}-{name}");
124        if let Some(prop) = paint.compiled_property::<csscolorparser::Color>(&full_name) {
125            let res = if prop.is_feature_constant() {
126                prop.evaluate_at_zoom(zoom)
127            } else {
128                prop.evaluate_for(properties, zoom)
129            };
130            return res.map_or(fallback, |color| {
131                let [r, g, b, a] = color.to_array();
132                [r as f32, g as f32, b as f32, a as f32]
133            });
134        }
135        fallback
136    };
137    let number = |name: &str, fallback: f32| {
138        let full_name = format!("{prefix}-{name}");
139        if let Some(prop) = paint.compiled_property::<f32>(&full_name) {
140            let res = if prop.is_feature_constant() {
141                prop.evaluate_at_zoom(zoom)
142            } else {
143                prop.evaluate_for(properties, zoom)
144            };
145            return res.filter(|value| value.is_finite()).unwrap_or(fallback);
146        }
147        fallback
148    };
149    let size = if prefix == "text" {
150        if let Some(prop) = &paint.text_size {
151            if prop.is_feature_constant() {
152                prop.evaluate_at_zoom(zoom).unwrap_or(16.0)
153            } else {
154                prop.evaluate_for(properties, zoom).unwrap_or(16.0)
155            }
156        } else {
157            16.0
158        }
159    } else {
160        number("size", 1.0)
161    };
162    [
163        color("color", [0.0, 0.0, 0.0, 1.0]),
164        color("halo-color", [0.0; 4]),
165        [
166            size,
167            number("halo-width", 0.0),
168            number("halo-blur", 0.0),
169            number("opacity", 1.0),
170        ],
171    ]
172}

174fn color(paint: &SymbolPaint, name: &str, zoom: f64, fallback: [f32; 4]) -> [f32; 4] {
175    let Some(color) = paint
176        .compiled_property::<csscolorparser::Color>(name)
177        .and_then(|prop| prop.evaluate_at_zoom(zoom))
178    else {
179        return fallback;
180    };
181    let [r, g, b, a] = color.to_array();
182    [r as f32, g as f32, b as f32, a as f32]
183}

185fn layout(paint: &SymbolPaint, prefix: &str) -> [f32; 4] {
186    let read = |suffix: &str| {
187        paint
188            .properties
189            .get(&format!("{prefix}-{suffix}"))
190            .and_then(|value| value.as_str())
191    };
192    let rotation = read("rotation-alignment");
193    let map_rotation = rotation == Some("map")
194        || (rotation.unwrap_or("auto") == "auto"
195            && paint
196                .properties
197                .get("symbol-placement")
198                .and_then(|value| value.as_str())
199                .is_some_and(|value| matches!(value, "line" | "line-center")));
200    let pitch = read("pitch-alignment");
201    let map_pitch = pitch == Some("map") || (pitch.unwrap_or("auto") == "auto" && map_rotation);
202    [
203        u32::from(map_pitch) as f32,
204        u32::from(map_rotation) as f32,
205        // The tessellator bakes the rotation into each symbol's corners.
206        0.0,
207        u32::from(paint.uses_shared_height() || paint.height_follows_ground(prefix)) as f32,
208    ]
209}

211fn keep_upright(paint: &SymbolPaint, prefix: &str, fallback: bool) -> f32 {
212    let line = paint
213        .properties
214        .get("symbol-placement")
215        .and_then(|v| v.as_str())
216        .is_some_and(|value| matches!(value, "line" | "line-center"));
217    f32::from(
218        line && paint
219            .properties
220            .get(&format!("{prefix}-keep-upright"))
221            .and_then(|v| v.as_bool())
222            .unwrap_or(fallback),
223    )
224}
