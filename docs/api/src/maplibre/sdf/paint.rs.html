8pub(super) struct SymbolUniforms {

17    pub atlas: [f32; 4],
18    pub placement: [f32; 4],

24/// Whether text along a line is placed glyph by glyph: it turns with the map, whether it lies
25/// on the map plane or stands upright to the viewer, or each glyph stands upright on the screen
26/// along the line (`viewport-glyph`). Text aligned with the viewport keeps the straight layout
27/// at the anchor.
28pub(crate) fn text_follows_line(paint: &SymbolPaint, zoom: f64) -> bool {
29    SymbolUniforms::new(paint, zoom, [1, 1]).text_layout[1] > 0.5
30        || glyphs_stay_upright(paint, zoom)
31}
32
33/// Whether the layer places its symbols along lines at `zoom`.
34fn line_placed(paint: &SymbolPaint, zoom: f64) -> bool {
35    paint
36        .layout_enum("symbol-placement", zoom)
37        .is_some_and(|value| matches!(value.as_str(), "line" | "line-center"))
38}
39
40/// Whether the glyphs of text along a line are placed along it each upright on the screen, as
41/// GL JS's `viewport-glyph` rotation alignment places them: road shields in a row stay level
42/// while the row follows its road.
43pub(crate) fn glyphs_stay_upright(paint: &SymbolPaint, zoom: f64) -> bool {
44    paint
45        .layout_enum("text-rotation-alignment", zoom)
46        .as_deref()
47        == Some("viewport-glyph")
48        && line_placed(paint, zoom)
49}

51impl SymbolUniforms {
52    pub fn new(paint: &SymbolPaint, zoom: f64, size: [u32; 2]) -> Self {
53        let properties = FeatureProperties::new();
54        let number = |name, fallback| paint.number(name, &properties, zoom, fallback);
55        Self {
56            text_color: color(paint, "text-color", zoom, [0.0, 0.0, 0.0, 1.0]),
57            halo_color: color(paint, "text-halo-color", zoom, [0.0; 4]),
58            icon_color: color(paint, "icon-color", zoom, [0.0, 0.0, 0.0, 1.0]),
59            icon_halo_color: color(paint, "icon-halo-color", zoom, [0.0; 4]),
60            text: [
61                paint
62                    .text_size
63                    .as_ref()
64                    .and_then(|value| value.evaluate_at_zoom(zoom))
65                    .unwrap_or(16.0),
66                number("text-halo-width", 0.0),
67                number("text-halo-blur", 0.0),
68                number("text-opacity", 1.0),
69            ],
70            icon: [
71                number("icon-size", 1.0),
72                number("icon-halo-width", 0.0),
73                number("icon-halo-blur", 0.0),
74                number("icon-opacity", 1.0),
75            ],
76            text_layout: layout(paint, "text", zoom),
77            icon_layout: layout(paint, "icon", zoom),
78            atlas: [size[0] as f32, size[1] as f32, 0.0, 0.0],
79            placement: [
80                number("text-padding", 2.0),
81                number("icon-padding", 2.0),
82                keep_upright(paint, "text", true, zoom),
83                keep_upright(paint, "icon", false, zoom),
84            ],
85            translate: {
86                let [text_x, text_y] = crate::sdf::translation::viewport_shift(paint, "text", zoom);
87                let [icon_x, icon_y] = crate::sdf::translation::viewport_shift(paint, "icon", zoom);
88                [text_x, text_y, icon_x, icon_y]
89            },
90        }
91    }
92}

94impl SymbolUniforms {
95    /// The uniforms with `text-size` and `icon-size` evaluated for one feature's properties:
96    /// a data-driven size has no value for the layer as a whole, and a label collides by the
97    /// size it is drawn at.
98    pub fn with_feature_sizes(
99        mut self,
100        paint: &SymbolPaint,
101        properties: &FeatureProperties,
102        zoom: f64,
103    ) -> Self {
104        let text_data_driven = paint
105            .text_size
106            .as_ref()
107            .is_some_and(|s| !s.is_feature_constant());
108        let icon_data_driven = paint
109            .compiled_property::<f32>("icon-size")
110            .is_some_and(|s| !s.is_feature_constant());
111        if !text_data_driven && !icon_data_driven {
112            return self;
113        }
114        if text_data_driven {
115            self.text[0] = feature_style(paint, "text", properties, zoom)[2][0];
116        }
117        if icon_data_driven {
118            self.icon[0] = feature_style(paint, "icon", properties, zoom)[2][0];
119        }
120        self
121    }
122}

124/// The fill colour, halo colour and size, halo width, halo blur and opacity of a feature's text
125/// (`prefix` "text") or icon (`prefix` "icon"), each evaluated for its properties.
126pub(crate) fn feature_style(
127    paint: &SymbolPaint,
128    prefix: &str,
129    properties: &FeatureProperties,
130    zoom: f64,
131) -> [[f32; 4]; 3] {
132    let is_text = prefix == "text";
133    let (color_name, halo_color_name, halo_width_name, halo_blur_name, opacity_name) = if is_text {
134        (
135            "text-color",
136            "text-halo-color",
137            "text-halo-width",
138            "text-halo-blur",
139            "text-opacity",
140        )
141    } else {
142        (
143            "icon-color",
144            "icon-halo-color",
145            "icon-halo-width",
146            "icon-halo-blur",
147            "icon-opacity",
148        )
149    };
150    let color = |full_name: &str, fallback: [f32; 4]| {
151        if let Some(prop) = paint.compiled_property::<csscolorparser::Color>(full_name) {
152            let res = if prop.is_feature_constant() {
153                prop.evaluate_at_zoom(zoom)
154            } else {
155                prop.evaluate_for(properties, zoom)
156            };
157            return res.map_or(fallback, |color| color.to_array());
158        }
159        fallback
160    };
161    let number = |full_name: &str, fallback: f32| {
162        if let Some(prop) = paint.compiled_property::<f32>(full_name) {
163            let res = if prop.is_feature_constant() {
164                prop.evaluate_at_zoom(zoom)
165            } else {
166                prop.evaluate_for(properties, zoom)
167            };
168            return res.filter(|value| value.is_finite()).unwrap_or(fallback);
169        }
170        fallback
171    };
172    let size = if is_text {
173        if let Some(prop) = &paint.text_size {
174            if prop.is_feature_constant() {
175                prop.evaluate_at_zoom(zoom).unwrap_or(16.0)
176            } else {
177                prop.evaluate_for(properties, zoom).unwrap_or(16.0)
178            }
179        } else {
180            16.0
181        }
182    } else {
183        number("icon-size", 1.0)
184    };
185    [
186        color(color_name, [0.0, 0.0, 0.0, 1.0]),
187        color(halo_color_name, [0.0; 4]),
188        [
189            size,
190            number(halo_width_name, 0.0),
191            number(halo_blur_name, 0.0),
192            number(opacity_name, 1.0),
193        ],
194    ]
195}

197fn color(paint: &SymbolPaint, name: &str, zoom: f64, fallback: [f32; 4]) -> [f32; 4] {
198    let Some(color) = paint
199        .compiled_property::<csscolorparser::Color>(name)
200        .and_then(|prop| prop.evaluate_at_zoom(zoom))
201    else {
202        return fallback;
203    };
204    color.to_array()
205}

207fn layout(paint: &SymbolPaint, prefix: &str, zoom: f64) -> [f32; 4] {
208    let read = |suffix: &str| paint.layout_enum(&format!("{prefix}-{suffix}"), zoom);
209    let rotation = read("rotation-alignment");
210    let map_rotation = rotation.as_deref() == Some("map")
211        || (rotation.as_deref().unwrap_or("auto") == "auto" && line_placed(paint, zoom));
212    let pitch = read("pitch-alignment");
213    let map_pitch = pitch.as_deref() == Some("map")
214        || (pitch.as_deref().unwrap_or("auto") == "auto" && map_rotation);
215    [
216        u32::from(map_pitch) as f32,
217        u32::from(map_rotation) as f32,
218        // The tessellator bakes the rotation into each symbol's corners.
219        0.0,
220        u32::from(paint.uses_shared_height() || paint.height_follows_ground(prefix)) as f32,
221    ]
222}
223
224fn keep_upright(paint: &SymbolPaint, prefix: &str, fallback: bool, zoom: f64) -> f32 {
225    let line = line_placed(paint, zoom);
226    f32::from(
227        line && paint
228            .properties
229            .get(&format!("{prefix}-keep-upright"))
230            .and_then(|v| v.as_bool())
231            .unwrap_or(fallback),
232    )
233}
