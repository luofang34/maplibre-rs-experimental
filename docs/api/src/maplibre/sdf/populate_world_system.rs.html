29impl<E: Environment, T: VectorTransferables> System for PopulateWorldSystem<E, T> {
30    fn name(&self) -> Cow<'static, str> {
31        "sdf_populate_world_system".into()
32    }
33
34    fn run(
35        &mut self,
36        MapContext {
37            world,
38            style,
39            renderer,
40            ..
41        }: &mut MapContext,
42    ) -> SystemResult {
43        let mut messages = self
44            .kernel
45            .apc()
46            .receive(|message| {
47                message.has_tag(T::SymbolLayerTessellated::message_tag())
48                    || message.has_tag(super::provided::ProvidedImagesReport::message_tag())
49            })
50            .peekable();
51        if messages.peek().is_none() || renderer.settings.symbol_pools.is_empty() {
52            return Ok(());
53        }
54        let layer_by_id: rustc_hash::FxHashMap<&str, &crate::style::layer::StyleLayer> = style
55            .layers
56            .iter()
57            .map(|layer| (layer.id.as_str(), layer))
58            .collect();
59        let mut prop_interner =
60            crate::vector::feature_paint::intern::TilePropertyInterner::default();
61        let mut state_memo = crate::style::feature_state::FeatureStateArcMemo::default();
62        apply_worker_messages(messages, |message| {
63            if message.has_tag(super::provided::ProvidedImagesReport::message_tag()) {
64                let report =
65                    message.into_transferable::<super::provided::ProvidedImagesReport>()?;
66                super::provided::apply(world, *report);
67            } else if message.has_tag(T::SymbolLayerTessellated::message_tag()) {
68                let attempt = message.attempt();
69                let message = message.into_transferable::<T::SymbolLayerTessellated>()?;
70                let coords = message.coords();
71                let mut bucket = message.to_bucket();
72                if let Some(&style_layer) =
73                    tile_retry::accepts(world, coords, RequestKind::Vector, attempt)
74                        .then(|| layer_by_id.get(bucket.style_layer_id.as_str()))
75                        .flatten()
76                {
77                    prop_interner.intern_symbol_bucket(&world.tiles, &mut bucket);
78                    let retessellate =
79                        crate::vector::arrival_sync::sync_arriving_symbol_bucket_for_layer(
80                            style,
81                            style_layer,
82                            &mut bucket,
83                            &mut state_memo,
84                        );
85                    crate::vector::content::accept_symbols(world, bucket);
86                    if retessellate {
87                        tile_retry::refresh_tiles(world, RequestKind::Vector, &[coords]);
88                    }
89                }
90            }
91            Ok(())
92        })?;
93
94        Ok(())
95    }
96}
