42/// Places visible symbols across tiles on throttled epochs and shares placement between eyes.
43#[derive(Default)]
44pub struct CollisionSystem {
45    runs: u32,
46    controller: PlacementController,
47    active_epoch: Option<sliced::ActiveEpoch>,
48    /// View the last committed placement epoch ran for.
49    placed_view: Option<ViewSignature>,
50    /// View seen on the immediately preceding frame, used to detect single-step XR movement.
51    last_frame_view: Option<ViewSignature>,
52    last_content_signature: Option<u64>,
53    last_terrain_signature: Option<u64>,
54    last_external_view: Option<bool>,
55    history: temporal::PlacementHistory,
56    uploaded:
57        FxHashMap<(WorldTileCoords, crate::style::StyleLayerId), upload_diff::CachedBucketMetadata>,
58    scratch: upload_diff::MetadataScratchPool,
59    draw_sort: draw_sort::DrawSort,
60    /// Persistent scratch set of currently visible covering tile coordinates.
61    seen: FxHashSet<WorldTileCoords>,
62    /// Revision of the host's label obstacles the last placement kept clear.
63    obstacles: u64,
64}
65
66/// World-level counter recording how many times [`CollisionSystem`] has executed.
67#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
68pub struct CollisionStats {
69    /// Total non-short-circuited [`CollisionSystem`] ticks on this world.
70    pub runs: u32,
71}

85impl System for CollisionSystem {
86    fn name(&self) -> Cow<'static, str> {
87        "sdf_collision_system".into()
88    }
89
90    fn run(
91        &mut self,
92        MapContext {
93            world,
94            style,
95            view_state,
96            renderer,
97            ..
98        }: &mut MapContext,
99    ) -> SystemResult {
100        if renderer.settings.symbol_pools.is_empty() {
101            return Ok(());
102        }
103        if !super::style_has_symbol_layers(style)
104            && super::symbol_pool_empty(world)
105            && self.uploaded.is_empty()
106            && self.active_epoch.is_none()
107            && !self.history.is_fading()
108        {
109            return Ok(());
110        }
111        let config = world
112            .resources
113            .get::<PlacementConfig>()
114            .copied()
115            .unwrap_or_default();
116        if crate::render::eye_covering::EyeInFrame::reuses_content(world) {
117            sync_symbol_depth_fade(
118                world,
119                &renderer.queue,
120                &renderer.render_graph,
121                &self.history,
122            );
123            self.prune_inactive_symbol_draws(world);
124            return Ok(());
125        }
126
127        let now = world
128            .resources
129            .get::<crate::render::frame_input::FrameInput>()
130            .map_or(std::time::Duration::ZERO, |input| input.timestamp);
131        self.history.set_fade_duration(config.fade_duration);
132        self.history.advance_clock(now);
133
134        self.runs = self.runs.wrapping_add(1);
135        let stats = world.resources.get_or_init_mut::<CollisionStats>();
136        stats.runs = stats.runs.wrapping_add(1);
137
138        self.seen.clear();
139        if let Some(covering) = world.resources.get::<super::covering::SymbolCovering>() {
140            self.seen.extend(covering.tiles.iter().copied());
141        }
142        let zoom = view_state.style_zoom().value();
143        let mut layers = visible_layers(world, style, zoom, &self.seen);
144
145        let content_sig = content_signature(world, &layers);
146        let terrain_sig = terrain_signature(world);
147        let external_view = view_state.has_external_view();
148        let view = (view_state.view_projection().0, view_state.viewport_size());
149        let view_valid = AsRef::<[[f64; 4]; 4]>::as_ref(&view.0)
150            .iter()
151            .flatten()
152            .all(|value| value.is_finite());
153        let prev_frame_view = self.last_frame_view;
154        let moved = !external_view && view_valid && self.placed_view.as_ref() != Some(&view);
155        let moving_this_frame = !external_view
156            && view_valid
157            && prev_frame_view.is_some()
158            && prev_frame_view.as_ref() != Some(&view);
159        let external_moved = external_view && view_valid && prev_frame_view.as_ref() != Some(&view);
160        if view_valid {
161            self.last_frame_view = Some(view);
162        }
163
164        let obstacles = world
165            .resources
166            .get::<super::obstacles::LabelObstacles>()
167            .map_or(0, super::obstacles::LabelObstacles::revision);
168        let triggers = PlacementTriggers {
169            moved,
170            moving_this_frame,
171            external_view,
172            external_moved,
173            new_content: self.last_content_signature != Some(content_sig),
174            environment_changed: self.last_terrain_signature != Some(terrain_sig)
175                || self.last_external_view != Some(external_view),
176            obstructed: obstacles != self.obstacles,
177        };
178
179        let earlier_stage_spent = world
180            .resources
181            .get::<crate::tcs::system::timings::FrameTimings>()
182            .map_or(
183                std::time::Duration::ZERO,
184                crate::tcs::system::timings::FrameTimings::elapsed_this_frame,
185            );
186        if self.active_epoch.is_some() {
187            if triggers.obstructed || config.fade_duration.is_zero() || !view_valid {
188                if let Some(epoch) = self.active_epoch.take() {
189                    epoch.abort(&mut self.scratch);
190                }
191            } else {
192                self.controller.advance_clock(now);
193                self.controller.note_background_dirty(zoom, triggers);
194                let Some(mut epoch) = self.active_epoch.take() else {
195                    return Ok(());
196                };
197                if triggers.new_content || triggers.environment_changed {
198                    epoch.saw_background_change = true;
199                }
200                // If earlier upload/drape stages already spent >2 ms while the camera is actively
201                // moving, defer this symbol collision slice to the next frame to prevent combined
202                // upload + placement spikes.
203                if moving_this_frame
204                    && !config.fade_duration.is_zero()
205                    && earlier_stage_spent > std::time::Duration::from_micros(2_000)
206                {
207                    self.active_epoch = Some(epoch);
208                    sync_symbol_depth_fade(
209                        world,
210                        &renderer.queue,
211                        &renderer.render_graph,
212                        &self.history,
213                    );
214                    self.prune_inactive_symbol_draws(world);
215                    crate::render::frame_signals::keep_animating(world);
216                    return Ok(());
217                }
218                if epoch.step(
219                    world,
220                    style,
221                    &renderer.queue,
222                    &mut self.history,
223                    &mut self.uploaded,
224                    &mut self.scratch,
225                    true,
226                    moving_this_frame,
227                ) {
228                    let followup =
229                        self.history.repeats.yielded_to_previous() || epoch.saw_background_change;
230                    self.controller
231                        .record_commit(epoch.zoom, epoch.has_symbols, followup);
232                    if !moving_this_frame
233                        && self.draw_sort.needs_bearing_sort(
234                            world,
235                            (view_state.camera().get_bearing().0, zoom),
236                            &layers,
237                        )
238                    {
239                        self.draw_sort.follow_bearing(
240                            world,
241                            &renderer.queue,
242                            (view_state.camera().get_bearing().0, zoom),
243                            &layers,
244                        );
245                    }
246                    world.resources.insert(epoch.take_placed());
247                    world
248                        .resources
249                        .insert(super::query::PlacementBearing(epoch.bearing));
250                } else {
251                    self.active_epoch = Some(epoch);
252                }
253                sync_symbol_depth_fade(
254                    world,
255                    &renderer.queue,
256                    &renderer.render_graph,
257                    &self.history,
258                );
259                self.prune_inactive_symbol_draws(world);
260                if self.active_epoch.is_some()
261                    || self.history.is_fading()
262                    || self.controller.is_pending()
263                {
264                    crate::render::frame_signals::keep_animating(world);
265                }
266                return Ok(());
267            }
268        }
269
270        let bearing = view_state.camera().get_bearing().0;
271        if !moving_this_frame
272            && self
273                .draw_sort
274                .needs_bearing_sort(world, (bearing, zoom), &layers)
275        {
276            self.draw_sort
277                .follow_bearing(world, &renderer.queue, (bearing, zoom), &layers);
278        }
279
280        if !self.controller.should_place(now, zoom, config, triggers)
281            || (moving_this_frame
282                && !config.fade_duration.is_zero()
283                && earlier_stage_spent > std::time::Duration::from_micros(2_200)
284                && self.controller.defer_for_frame_budget(triggers))
285        {
286            sync_symbol_depth_fade(
287                world,
288                &renderer.queue,
289                &renderer.render_graph,
290                &self.history,
291            );
292            self.prune_inactive_symbol_draws(world);
293            if self.history.is_fading() || self.controller.is_pending() {
294                crate::render::frame_signals::keep_animating(world);
295            }
296            return Ok(());
297        }
298
299        self.obstacles = obstacles;
300        if view_valid {
301            self.placed_view = Some(view);
302        }
303        self.last_content_signature = Some(content_sig);
304        self.last_terrain_signature = Some(terrain_sig);
305        self.last_external_view = Some(external_view);
306
307        let use_slicing = self.controller.is_initialized()
308            && self.controller.last_dt_non_zero()
309            && !config.fade_duration.is_zero()
310            && !triggers.obstructed;
311        if !moving_this_frame || !use_slicing {
312            self.draw_sort
313                .follow_bearing(world, &renderer.queue, (bearing, zoom), &layers);
314        }
315        layers.sort_by_key(|(index, layer, _)| {
316            (
317                std::cmp::Reverse(*index),
318                std::cmp::Reverse(u8::from(layer.coords.z)),
319                layer.coords.x,
320                layer.coords.y,
321            )
322        });
323        let projection = projection_data_for_view(style, view_state).map_err(|error| {
324            tracing::error!(%error, "symbol projection failed");
325            SystemError::Setup
326        })?;
327        let has_symbols = !layers.is_empty();
328        self.history.begin_with_duration(now, config.fade_duration);
329        self.uploaded.retain(|(coords, _), cached| {
330            let keep = self.seen.contains(coords);
331            if !keep {
332                self.scratch.recycle(std::mem::take(&mut cached.vertices));
333            }
334            keep
335        });
336        let mut epoch = sliced::ActiveEpoch::begin(world, style, view_state, projection, layers);
337        if epoch.step(
338            world,
339            style,
340            &renderer.queue,
341            &mut self.history,
342            &mut self.uploaded,
343            &mut self.scratch,
344            use_slicing,
345            moving_this_frame,
346        ) {
347            let followup_needed = self.history.repeats.yielded_to_previous();
348            self.controller
349                .record_commit(zoom, has_symbols, followup_needed);
350            world.resources.insert(epoch.take_placed());
351            world
352                .resources
353                .insert(super::query::PlacementBearing(bearing));
354        } else {
355            self.active_epoch = Some(epoch);
356        }
357        sync_symbol_depth_fade(
358            world,
359            &renderer.queue,
360            &renderer.render_graph,
361            &self.history,
362        );
363        self.prune_inactive_symbol_draws(world);
364        if self.active_epoch.is_some() || self.history.is_fading() || self.controller.is_pending() {
365            crate::render::frame_signals::keep_animating(world);
366        }
367        Ok(())
368    }
369}

371fn sync_symbol_depth_fade(
372    world: &mut crate::tcs::world::World,
373    queue: &crate::render::upload_queue::UploadQueue,
374    render_graph: &crate::render::graph::RenderGraph,
375    history: &temporal::PlacementHistory,
376) {
377    let time_ms = history.effective_shader_time_ms();
378    let duration_ms = history.fade_duration_ms();
379    let skip_depth_occlusion = !has_symbol_depth_occluders(world, render_graph);
380    if let Some(Initialized(depth)) = world
381        .resources
382        .get_mut::<Eventually<super::depth::SymbolDepth>>()
383    {
384        depth.write_fade_time(queue, time_ms, duration_ms, skip_depth_occlusion);
385    }
386}

388impl CollisionSystem {
389    fn prune_inactive_symbol_draws(&self, world: &mut crate::tcs::world::World) {
390        use crate::{
391            render::render_phase::{Draw, DrawState, RenderPhase, TranslucentItem},
392            sdf::render_commands::DrawSymbols,
393        };
394        let now_ms = self.history.effective_shader_time_ms();
395        let fade_duration_ms = self.history.fade_duration_ms();
396        let Some((phase, Initialized(pool))) = world.resources.query_mut::<(
397            &mut RenderPhase<TranslucentItem>,
398            &Eventually<SymbolBufferPool>,
399        )>() else {
400            return;
401        };
402        let symbol_draw: &dyn Draw<TranslucentItem> =
403            &DrawState::<TranslucentItem, DrawSymbols>::INSTANCE;
404        phase.retain(|item| {
405            if !std::ptr::addr_eq(item.draw_function, symbol_draw) {
406                return true;
407            }
408            let Some(entry) =
409                pool.index()
410                    .resolve_entry(item.tile.coords, item.style_layer, item.entry_slot)
411            else {
412                return false;
413            };
414            self.uploaded
415                .get(&(item.tile.coords, item.style_layer))
416                .is_some_and(|cached| {
417                    cached.allocation_id == entry.allocation_id()
418                        && cached.is_active_at(now_ms, fade_duration_ms)
419                })
420        });
421    }
