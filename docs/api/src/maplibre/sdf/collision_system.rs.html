26/// Places visible symbols across tiles and shares placement between eyes in the same frame.
27/// Retains fade history and refreshes opacity/elevation uploads when content or allocation changes.
28#[derive(Default)]
29pub struct CollisionSystem {
30    runs: u32,
31    /// View the last placement ran for; a different one makes it stale within a frame.
32    placed_view: Option<(cgmath::Matrix4<f64>, (f64, f64))>,
33    history: temporal::PlacementHistory,
34    uploaded: HashMap<(WorldTileCoords, String), (u64, u64)>,
35    draw_sort: draw_sort::DrawSort,
36}
37
38impl CollisionSystem {
39    /// Starts with no placement history or uploaded metadata fingerprints.
40    pub fn new() -> Self {
41        Self::default()
42    }
43}

45fn opacity_fingerprint(metadata: &[SDFShaderFeatureMetadata]) -> u64 {
46    let mut hash = DefaultHasher::new();
47    for entry in metadata {
48        entry.opacity.to_bits().hash(&mut hash);
49        entry.elevation.to_bits().hash(&mut hash);
50        entry.pose.map(f32::to_bits).hash(&mut hash);
51        entry.color.map(f32::to_bits).hash(&mut hash);
52        entry.halo.map(f32::to_bits).hash(&mut hash);
53        entry.params.map(f32::to_bits).hash(&mut hash);
54    }
55    hash.finish()
56}

58impl System for CollisionSystem {
59    fn name(&self) -> Cow<'static, str> {
60        "sdf_collision_system".into()
61    }
62
63    fn run(
64        &mut self,
65        MapContext {
66            world,
67            style,
68            view_state,
69            renderer,
70            ..
71        }: &mut MapContext,
72    ) -> SystemResult {
73        if crate::render::eye_covering::EyeInFrame::reuses_content(world) {
74            return Ok(());
75        }
76        self.runs = self.runs.wrapping_add(1);
77        let seen: HashSet<_> = world
78            .resources
79            .get::<super::covering::SymbolCovering>()
80            .map(|covering| covering.tiles.iter().copied().collect())
81            .unwrap_or_default();
82        let mut layers = visible_layers(world, style, view_state.style_zoom().value(), &seen);
83        let new_content = layers.iter().any(|(_, layer, _)| {
84            allocation(world, layer.coords, &layer.style_layer_id)
85                != self
86                    .uploaded
87                    .get(&(layer.coords, layer.style_layer_id.clone()))
88                    .map(|entry| entry.0)
89        });
90        let view = (view_state.view_projection().0, view_state.viewport_size());
91        // A head-tracked view changes every frame, so it keeps the periodic cadence instead of
92        // placing every label each frame.
93        let moved = !view_state.has_external_view()
94            && AsRef::<[[f64; 4]; 4]>::as_ref(&view.0)
95                .iter()
96                .flatten()
97                .all(|value| value.is_finite())
98            && self.placed_view.as_ref() != Some(&view);
99        if !new_content && !moved && !self.history.fading && !self.runs.is_multiple_of(8) {
100            return Ok(());
101        }
102        self.placed_view = Some(view);
103        self.draw_sort.follow_bearing(
104            world,
105            &renderer.queue,
106            (
107                view_state.camera().get_bearing().0,
108                view_state.style_zoom().value(),
109            ),
110            &layers,
111        );
112        layers.sort_by_key(|(index, layer, _)| {
113            (
114                std::cmp::Reverse(*index),
115                std::cmp::Reverse(u8::from(layer.coords.z)),
116                layer.coords.x,
117                layer.coords.y,
118            )
119        });
120        let projection = projection_data_for_view(style, view_state).map_err(|error| {
121            tracing::error!(%error, "symbol projection failed");
122            SystemError::Setup
123        })?;
124        self.history.begin(
125            world
126                .resources
127                .get::<crate::render::frame_input::FrameInput>()
128                .map_or(std::time::Duration::ZERO, |input| input.timestamp),
129        );
130        let placed = self.place_layers(
131            world,
132            style,
133            view_state,
134            &projection,
135            &renderer.queue,
136            layers,
137        );
138        self.uploaded.retain(|(coords, _), _| seen.contains(coords));
139        world.resources.insert(placed);
140        world.resources.insert(super::query::PlacementBearing(
141            view_state.camera().get_bearing().0,
142        ));
143        // Labels still fading in or out change the next frame.
144        if self.history.fading {
145            crate::render::frame_signals::keep_animating(world);
146        }
147        Ok(())
148    }
149}

160fn visible_layers<'a>(
161    world: &'a crate::tcs::world::World,
162    style: &'a crate::style::Style,
163    zoom: f64,
164    seen: &HashSet<WorldTileCoords>,
165) -> Vec<(
166    u32,
167    &'a crate::sdf::SymbolLayerData,
168    &'a crate::style::layer::SymbolPaint,
169)> {
170    let mut layers = Vec::new();
171    for coords in seen {
172        if let Some(component) = world.tiles.query::<&SymbolLayersDataComponent>(*coords) {
173            for layer in &component.layers {
174                if let Some(style_layer) = style
175                    .layers
176                    .iter()
177                    .find(|style| style.id == layer.style_layer_id && style.is_visible_at(zoom))
178                {
179                    if let Some(LayerPaint::Symbol(paint)) = &style_layer.paint {
180                        layers.push((style_layer.index, layer, paint));
181                    }
182                }
183            }
184        }
185    }
186    layers
187}

189fn allocation(world: &crate::tcs::world::World, coords: WorldTileCoords, id: &str) -> Option<u64> {
190    let Initialized(pool) = world.resources.get::<Eventually<SymbolBufferPool>>()? else {
191        return None;
192    };
193    pool.index()
194        .get_layers(coords)?
195        .iter()
196        .find(|entry| entry.style_layer.id == id)
197        .map(|entry| entry.allocation_id())
198}

200impl CollisionSystem {
201    fn place_layers(
202        &mut self,
203        world: &crate::tcs::world::World,
204        style: &crate::style::Style,
205        view_state: &crate::render::view_state::ViewState,
206        projection: &crate::render::projection::ShaderProjectionData,
207        queue: &crate::render::upload_queue::UploadQueue,
208        layers: Vec<VisibleLayer<'_>>,
209    ) -> PlacedSymbols {
210        let mut boxes = CollisionGrid::new(view_state.width(), view_state.height());
211        let mut updates = Vec::new();
212        let mut placed = PlacedSymbols::default();
213        let mut groups: Vec<(
214            u32,
215            &crate::style::layer::SymbolPaint,
216            Vec<&crate::sdf::SymbolLayerData>,
217        )> = Vec::new();
218        for (index, layer, paint) in layers {
219            match groups.last_mut() {
220                Some((last, _, members))
221                    if *last == index && members[0].style_layer_id == layer.style_layer_id =>
222                {
223                    members.push(layer);
224                }
225                _ => groups.push((index, paint, vec![layer])),
226            }
227        }
228        // A layer GL JS would group with a lower one is placed by it, after the pass.
229        let leaders = family::leaders(style);
230        let present: HashSet<&str> = groups
231            .iter()
232            .map(|(_, _, members)| members[0].style_layer_id.as_str())
233            .collect();
234        let mut outcomes = HashMap::new();
235        let mut mirrored = Vec::new();
236        let mut placed_layers = Vec::new();
237        for (_, paint, members) in groups {
238            let id = members[0].style_layer_id.clone();
239            if let Some(leader) = leaders
240                .get(&id)
241                .filter(|leader| **leader != id && present.contains(leader.as_str()))
242            {
243                mirrored.push((leader.clone(), paint, members));
244                continue;
245            }
246            let limits = style
247                .layers
248                .iter()
249                .find(|style| style.id == id)
250                .map(|layer| [layer.minzoom.unwrap_or(0.0), layer.maxzoom.unwrap_or(24.0)])
251                .unwrap_or([0.0, 24.0]);
252            let (metadata, outcome) = place_layer(
253                world,
254                view_state,
255                projection,
256                &members,
257                paint,
258                limits,
259                (&mut boxes, &mut placed, &mut self.history),
260            );
261            if leaders.get(&id) == Some(&id) {
262                outcomes.insert(id, (members.clone(), outcome));
263            }
264            placed_layers.push((members, metadata));
265        }
266        for (leader, paint, members) in mirrored {
267            let Some((leader_members, outcome)) = outcomes.get(&leader) else {
268                continue;
269            };
270            let metadata = family::mirror(
271                (leader_members, outcome),
272                &members,
273                (paint, view_state.style_zoom().value()),
274                &mut placed,
275            );
276            placed_layers.push((members, metadata));
277        }
278        for (members, metadata) in placed_layers {
279            for (layer, metadata) in members.into_iter().zip(metadata) {
280                let key = (layer.coords, layer.style_layer_id.clone());
281                let Some(generation) = allocation(world, layer.coords, &layer.style_layer_id)
282                else {
283                    continue;
284                };
285                let fingerprint = opacity_fingerprint(&metadata);
286                if self.uploaded.get(&key) != Some(&(generation, fingerprint)) {
287                    updates.push((key.clone(), metadata));
288                    self.uploaded.insert(key, (generation, fingerprint));
289                }
290            }
291        }
292        if let Some(Initialized(pool)) = world.resources.get::<Eventually<SymbolBufferPool>>() {
293            for ((coords, id), metadata) in updates {
294                if let Some(entry) = pool
295                    .index()
296                    .get_layers(coords)
297                    .and_then(|entries| entries.iter().find(|entry| entry.style_layer.id == id))
298                {
299                    pool.update_feature_metadata(queue, entry, &metadata);
300                }
301            }
302        }
303        placed
304    }
305}
