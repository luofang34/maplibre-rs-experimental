20/// Maps the local geometry slice index to its original MVT/MLT feature index in the source layer.
21pub(crate) fn source_feature_indices(count: usize, slice: Option<&[usize]>) -> Vec<Option<u32>> {
22    match slice {
23        Some(indices) => indices.iter().map(|&idx| u32::try_from(idx).ok()).collect(),
24        None => (0..count).map(|idx| u32::try_from(idx).ok()).collect(),
25    }
26}

28pub(super) fn build_source_feature(
29    tessellator: &mut TextTessellator,
30    slice_idx: Option<usize>,
31    geometry: &geo_types::Geometry<f64>,
32) -> (SourceFeature, Arc<FeatureProperties>) {
33    let id = slice_idx
34        .and_then(|i| tessellator.source_ids.get(i).copied())
35        .flatten();
36    let feature_index = slice_idx
37        .and_then(|i| tessellator.source_indices.get(i).copied())
38        .flatten();
39    let props = tessellator.take_feature_properties(slice_idx, id);
40    let kind = match geometry {
41        geo_types::Geometry::Point(_) | geo_types::Geometry::MultiPoint(_) => GeometryType::Point,
42        geo_types::Geometry::LineString(_) | geo_types::Geometry::MultiLineString(_) => {
43            GeometryType::LineString
44        }
45        geo_types::Geometry::Polygon(_) | geo_types::Geometry::MultiPolygon(_) => {
46            GeometryType::Polygon
47        }
48        _ => GeometryType::Unknown,
49    };
50    (
51        SourceFeature {
52            id,
53            kind,
54            feature_index,
55        },
56        props,
57    )
58}

60/// Drops feature property entries that are only read during tessellation (`text-field`,
61/// `icon-image`, `filter`, layout-only keys) once symbol layout finishes, retaining only runtime
62/// placement/paint properties, `\u{1}*` internal state/identity tags, and `promote_id_key`.
63pub(crate) fn prune_symbol_features(
64    features: &mut [Feature],
65    style: &StyleLayer,
66    promote_id_key: Option<&str>,
67) {
68    let Some(needed) = symbol_runtime_referenced_keys(style, promote_id_key) else {
69        return;
70    };
71    let empty = empty_feature_properties();
72    let mut cache: FxHashMap<usize, Arc<FeatureProperties>> = FxHashMap::default();
73
74    for feature in features {
75        if feature.data.properties.is_empty() {
76            continue;
77        }
78        let needs_prune = feature.data.properties.keys().any(|k| {
79            !k.starts_with('\u{1}') && !k.starts_with(FEATURE_STATE_PREFIX) && !needed.contains(k)
80        });
81        if !needs_prune {
82            continue;
83        }
84        let ptr = Arc::as_ptr(&feature.data.properties) as usize;
85        if let Some(pruned) = cache.get(&ptr) {
86            feature.data.properties = Arc::clone(pruned);
87            continue;
88        }
89        let mut filtered = FeatureProperties::default();
90        for (k, v) in feature.data.properties.iter() {
91            if k.starts_with('\u{1}') || k.starts_with(FEATURE_STATE_PREFIX) || needed.contains(k) {
92                filtered.insert(k.clone(), v.clone());
93            }
94        }
95        let pruned = if filtered.is_empty() {
96            Arc::clone(empty)
97        } else {
98            Arc::new(filtered)
99        };
100        cache.insert(ptr, Arc::clone(&pruned));
101        feature.data.properties = pruned;
102    }
103}

105#[cfg(test)]
106mod tests {
107    use serde_json::json;
108
109    use super::*;
110    use crate::{
111        sdf::SymbolFeatureData,
112        style::{
113            expression::Value,
114            layer::{LayerPaint, SymbolPaint},
115        },
116    };
117
118    #[test]
119    fn prune_symbol_features_drops_tessellation_only_keys_and_shares_empty_arc() {
120        let mut props_json = serde_json::Map::new();
121        props_json.insert("text-color".to_owned(), json!(["get", "tint"]));
122        let style = StyleLayer {
123            id: "places".to_owned(),
124            type_: "symbol".to_owned(),
125            paint: Some(LayerPaint::Symbol(SymbolPaint {
126                text_field: Some(crate::style::layer::StyleProperty::parse(&json!("{name}"))),
127                properties: props_json,
128                ..Default::default()
129            })),
130            ..Default::default()
131        };
132
133        let mut f_only_name = FeatureProperties::default();
134        f_only_name.insert("name".into(), Value::String("Zürich".into()));
135        f_only_name.insert("name:fr".into(), Value::String("Zurich".into()));
136
137        let mut f_with_runtime = FeatureProperties::default();
138        f_with_runtime.insert("name".into(), Value::String("Bern".into()));
139        f_with_runtime.insert("tint".into(), Value::String("#112233".into()));
140        f_with_runtime.insert(format!("{FEATURE_STATE_PREFIX}hover"), Value::Bool(true));
141
142        let mut features = vec![
143            Feature {
144                parts: [None, None, None],
145                data: SymbolFeatureData {
146                    properties: Arc::new(f_only_name),
147                    source_feature_index: Some(0),
148                    ..Default::default()
149                },
150                bbox: crate::euclid::Box2D::zero(),
151                indices: 0..0,
152                text_anchor: crate::euclid::Point2D::new(0.0, 0.0),
153                anchor_shifts: Vec::new(),
154                text_sets: Vec::new(),
155                anchor_sets: Vec::new(),
156                text_colors: Vec::new(),
157                fallback: false,
158                repeat_key: None,
159                str: String::new(),
160                line: None,
161            },
162            Feature {
163                parts: [None, None, None],
164                data: SymbolFeatureData {
165                    properties: Arc::new(f_with_runtime),
166                    source_feature_index: Some(1),
167                    ..Default::default()
168                },
169                bbox: crate::euclid::Box2D::zero(),
170                indices: 0..0,
171                text_anchor: crate::euclid::Point2D::new(0.0, 0.0),
172                anchor_shifts: Vec::new(),
173                text_sets: Vec::new(),
174                anchor_sets: Vec::new(),
175                text_colors: Vec::new(),
176                fallback: false,
177                repeat_key: None,
178                str: String::new(),
179                line: None,
180            },
181        ];
182
183        prune_symbol_features(&mut features, &style, None);
184        assert!(features[0].data.properties.is_empty());
185        assert!(Arc::ptr_eq(
186            &features[0].data.properties,
187            empty_feature_properties()
188        ));
189        assert_eq!(features[1].data.properties.len(), 2);
190        assert_eq!(
191            features[1].data.properties.get("tint"),
192            Some(&Value::String("#112233".into()))
193        );
194        assert!(features[1]
195            .data
196            .properties
197            .contains_key(&format!("{FEATURE_STATE_PREFIX}hover")));
198        assert!(!features[1].data.properties.contains_key("name"));
199    }
