38/// Where the labels of a geometry go when they do not follow a line, as GL JS places them:
39/// at every point, at the first vertex of every line, and at the pole of inaccessibility of
40/// every polygon.
41pub(super) fn anchors(geometry: &Geometry<f64>) -> Vec<Point<f64>> {
42    let polygon = |polygon: &geo_types::Polygon<f64>| {
43        let rings: Vec<Vec<[f64; 2]>> = std::iter::once(polygon.exterior())
44            .chain(polygon.interiors())
45            .map(|ring| ring.coords().map(|c| [c.x, c.y]).collect())
46            .collect();
47        super::pole::pole_of_inaccessibility(&rings, POLE_PRECISION).map(|[x, y]| Point::new(x, y))
48    };
49    let first = |line: &geo_types::LineString<f64>| {
50        line.0.first().map(|point| Point::new(point.x, point.y))
51    };
52    match geometry {
53        Geometry::Point(point) => vec![*point],
54        Geometry::MultiPoint(points) => points.iter().copied().collect(),
55        Geometry::LineString(line) => first(line).into_iter().collect(),
56        Geometry::MultiLineString(lines) => lines.iter().filter_map(first).collect(),
57        Geometry::Polygon(shape) => polygon(shape).into_iter().collect(),
58        Geometry::MultiPolygon(shapes) => shapes.iter().filter_map(polygon).collect(),
59        other => other.centroid().into_iter().collect(),
60    }
61}

66pub(super) fn append(
67    symbol: &CollectedSymbol,
68    paint: &SymbolPaint,
69    zoom: f64,
70    atlas: &SymbolAtlas,
71    buffer: &mut VertexBuffers<ShaderSymbolVertex, u32>,
72    features: &mut Vec<Feature>,
73) {
74    let start = buffer.indices.len();
75    let first_vertex = buffer.vertices.len();
76    let mut icon_padding = 0.0_f32;
77    let text = paint
78        .label_among(&symbol.properties, zoom, Some(&atlas.icons))
79        .unwrap_or_default();
80    if let Some(icon) = paint
81        .text_among_images("icon-image", &symbol.properties, zoom, &atlas.icons)
82        .and_then(|name| atlas.icons.get(&name))
83    {
84        let ratio = icon.metrics[3];
85        let width = icon.rect[2] as f32 / ratio;
86        let height = icon.rect[3] as f32 / ratio;
87        let offset = offset(paint, "icon-offset", 1.0, (&symbol.properties, zoom));
88        let fractions = anchor_fractions(
89            &paint
90                .text("icon-anchor", &symbol.properties, zoom)
91                .unwrap_or_else(|| "center".into()),
92        );
93        let height_offset = if paint.uses_shared_height() {
94            0.0
95        } else {
96            paint.height_offset("icon", &symbol.properties, zoom)
97        };
98        let icon_size = paint.number("icon-size", &symbol.properties, zoom, 1.0);
99        // An image whose anchor is not its centre, such as a shield with a banner above its
100        // body, is moved so that the anchor stands where the centre would.
101        let recentre = [icon.metrics[0] / ratio, icon.metrics[1] / ratio];
102        let placed = [
103            -width * fractions[0] + offset[0] - recentre[0],
104            -height * fractions[1] + offset[1] - recentre[1],
105            width * (1.0 - fractions[0]) + offset[0] - recentre[0],
106            height * (1.0 - fractions[1]) + offset[1] - recentre[1],
107        ];
108        let fitted = fit_to_text(paint, symbol, zoom, atlas, [width, height], offset);
109        let shift = crate::sdf::translation::tile_translation(paint, "icon", zoom);
110        let anchor = Point::new(symbol.anchor.x() + shift[0], symbol.anchor.y() + shift[1]);
111        let rotation = paint
112            .number("icon-rotate", &symbol.properties, zoom, 0.0)
113            .to_radians()
114            + if super::text_layout::is_vertical(symbol, paint, zoom, atlas) {
115                std::f32::consts::FRAC_PI_2
116            } else {
117                0.0
118            };
119        // Layout pixels are drawn scaled by icon-size, which a fitted icon does not take.
120        let undo_size = if fitted.is_some() {
121            1.0 / icon_size
122        } else {
123            1.0
124        };
125        for piece in super::icon_quads::icon_quads(icon, fitted.unwrap_or(placed), fitted.is_some())
126        {
127            icon_padding = icon_padding.max(piece.padding * undo_size);
128            let part = AtlasEntry {
129                rect: piece.rect,
130                ..icon.clone()
131            };
132            quad(
133                buffer,
134                anchor,
135                piece.bounds.map(|edge| edge * undo_size),
136                &part,
137                height_offset,
138                symbol.angle,
139                rotation,
140            );
141        }
142    }
143    let first_glyph_index = buffer.indices.len();
144    let laid = super::text_layout::append(symbol, paint, zoom, atlas, buffer);
145    let glyph_offsets = laid.centres;
146    if start == buffer.indices.len() {
147        return;
148    }
149    // Collision follows the text where there is one, as the icon alone otherwise.
150    let shift = crate::sdf::translation::tile_translation(
151        paint,
152        if text.is_empty() { "icon" } else { "text" },
153        zoom,
154    );
155    let anchor = Point2D::new(
156        (symbol.anchor.x() + shift[0]) as f32,
157        (symbol.anchor.y() + shift[1]) as f32,
158    );
159    let bbox = bounds(
160        &buffer.vertices,
161        &buffer.indices[start..],
162        paint,
163        &symbol.properties,
164        zoom,
165    );
166    let mut parts = crate::sdf::placement_geometry::measure(&mut buffer.vertices[first_vertex..]);
167    // An icon collides by its image, not by the clear border its quad reaches into.
168    for part in parts[1..].iter_mut().flatten() {
169        let pad = f64::from(icon_padding);
170        part.bounds = [
171            part.bounds[0] + pad,
172            part.bounds[1] + pad,
173            part.bounds[2] - pad,
174            part.bounds[3] - pad,
175        ];
176    }
177    // Text collides by the box of its lines, not by the bitmaps of its glyphs, which reach
178    // a few pixels beyond it.
179    // A label of images alone, such as a row of road shields along a road, has no glyphs to
180    // measure, so its box is the layout's.
181    if parts[0].is_none() {
182        let image = buffer.vertices[first_vertex..]
183            .iter()
184            .find(|vertex| vertex.a_data[2] == 3);
185        if let (Some(vertex), Some(extent)) = (
186            image,
187            super::text_layout::extent(symbol, paint, zoom, atlas),
188        ) {
189            parts[0] = Some(crate::sdf::placement_geometry::SymbolBounds {
190                bounds: extent.map(f64::from),
191                height: f64::from(f32::from_bits(vertex.a_pixeloffset[2] as u32)),
192                angle: f64::from(f32::from_bits(vertex.a_pixeloffset[3] as u32)),
193                text: true,
194            });
195        }
196    }
197    if let (Some(text_part), None) = (&mut parts[0], &symbol.line) {
198        if let Some([left, top, right, bottom]) =
199            super::text_layout::extent(symbol, paint, zoom, atlas)
200        {
201            text_part.bounds = [left, top, right, bottom].map(f64::from);
202        }
203    }
204    features.push(Feature {
205        parts,
206        data: crate::sdf::SymbolFeatureData {
207            id: symbol.source.id,
208            geometry_type: symbol.source.kind,
209            properties: symbol.properties.clone(),
210            sort_key: paint.number("symbol-sort-key", &symbol.properties, zoom, 0.0),
211        },
212        bbox,
213        indices: start..buffer.indices.len(),
214        text_anchor: anchor,
215        anchor_shifts: super::text_layout::variable_shifts(symbol, paint, zoom, atlas),
216        text_sets: laid.sets,
217        anchor_sets: laid.anchor_sets,
218        text_colors: laid.colors,
219        fallback: symbol.fallback,
220        str: text,
221        line: symbol
222            .line
223            .as_ref()
224            .filter(|_| !glyph_offsets.is_empty())
225            .map(|(polyline, distance)| crate::sdf::LineLabel {
226                polyline: polyline.clone(),
227                anchor_distance: *distance,
228                glyph_offsets,
229                image_sizes: laid.image_sizes,
230                first_glyph_index,
231            }),
232    });
233}

235/// The icon box `[left, top, right, bottom]` that `icon-text-fit` stretches around the text,
236/// in text-size pixels, or `None` when the icon keeps its own size.
237fn fit_to_text(
238    paint: &SymbolPaint,
239    symbol: &CollectedSymbol,
240    zoom: f64,
241    atlas: &SymbolAtlas,
242    [icon_width, icon_height]: [f32; 2],
243    offset: [f32; 2],
244) -> Option<[f32; 4]> {
245    let fit = paint.text("icon-text-fit", &symbol.properties, zoom)?;
246    if fit == "none" {
247        return None;
248    }
249    let text_size = paint
250        .text_size
251        .as_ref()
252        .and_then(|value| value.evaluate_for(&symbol.properties, zoom))
253        .unwrap_or(16.0);
254    let scale = text_size / 24.0;
255    // A vertical label's icon is fitted before the symbol turns.
256    let [left, top, right, bottom] =
257        super::text_layout::layout_extent(symbol, paint, zoom, atlas)?.map(|edge| edge * scale);
258    let padding = paint
259        .properties
260        .get("icon-text-fit-padding")
261        .and_then(|value| value.as_array())
262        .filter(|values| values.len() == 4)
263        .map_or([0.0; 4], |values| {
264            let read = |index: usize| values[index].as_f64().unwrap_or(0.0) as f32;
265            [read(0), read(1), read(2), read(3)]
266        });
267    let (min_x, max_x) = if matches!(fit.as_str(), "width" | "both") {
268        (
269            offset[0] + left - padding[3],
270            offset[0] + right + padding[1],
271        )
272    } else {
273        let min = offset[0] + (left + right - icon_width) / 2.0;
274        (min, min + icon_width)
275    };
276    let (min_y, max_y) = if matches!(fit.as_str(), "height" | "both") {
277        (
278            offset[1] + top - padding[0],
279            offset[1] + bottom + padding[2],
280        )
281    } else {
282        let min = offset[1] + (top + bottom - icon_height) / 2.0;
283        (min, min + icon_height)
284    };
285    Some([min_x, min_y, max_x, max_y])
286}

288/// The `[x, y]` offset a layout property gives one symbol, times `scale`; zero without one.
289pub(super) fn offset(
290    paint: &SymbolPaint,
291    name: &str,
292    scale: f32,
293    (properties, zoom): (&crate::style::expression::FeatureProperties, f64),
294) -> [f32; 2] {
295    let Some(value) = paint.properties.get(name) else {
296        return [0.0; 2];
297    };
298    // A written pair is the common case and needs no expression.
299    let [x, y] = match crate::style::translation::Pair::from_json(value) {
300        Some(pair) => pair.0,
301        None => {
302            crate::style::property::StyleProperty::<crate::style::translation::Pair>::parse(value)
303                .evaluate_for(properties, zoom)
304                .map_or([0.0; 2], |pair| pair.0)
305        }
306    };
307    [x as f32 * scale, y as f32 * scale]
308}

310pub(super) fn anchor_fractions(anchor: &str) -> [f32; 2] {
311    [
312        if anchor.contains("left") {
313            0.0
314        } else if anchor.contains("right") {
315            1.0
316        } else {
317            0.5
318        },
319        if anchor.contains("top") {
320            0.0
321        } else if anchor.contains("bottom") {
322            1.0
323        } else {
324            0.5
325        },
326    ]
327}

329pub(super) fn quad(
330    buffer: &mut VertexBuffers<ShaderSymbolVertex, u32>,
331    anchor: Point<f64>,
332    bounds: [f32; 4],
333    image: &AtlasEntry,
334    height: f32,
335    angle: f32,
336    rotation: f32,
337) {
338    let base = buffer.vertices.len() as u32;
339    let (sin, cos) = rotation.sin_cos();
340    let [x, y, width, height_pixels] = image.rect;
341    for (index, (u, v)) in [
342        (x, y),
343        (x + width, y),
344        (x + width, y + height_pixels),
345        (x, y + height_pixels),
346    ]
347    .into_iter()
348    .enumerate()
349    {
350        let px = if index == 0 || index == 3 {
351            bounds[0]
352        } else {
353            bounds[2]
354        };
355        let py = if index < 2 { bounds[1] } else { bounds[3] };
356        // The quad turns about the anchor, offsets included, as GL JS rotates its corners.
357        let (px, py) = (px * cos - py * sin, px * sin + py * cos);
358        buffer.vertices.push(ShaderSymbolVertex {
359            a_pos_offset: [
360                anchor.x().round() as i32,
361                anchor.y().round() as i32,
362                (px * 32.0).round() as i32,
363                (py * 32.0).round() as i32,
364            ],
365            a_data: [u, v, image.kind, 0],
366            a_pixeloffset: [0, 0, height.to_bits() as i32, angle.to_bits() as i32],
367        });
368    }
369    buffer
370        .indices
371        .extend([base, base + 1, base + 2, base, base + 2, base + 3]);
372}

374fn bounds(
375    vertices: &[ShaderSymbolVertex],
376    indices: &[u32],
377    paint: &SymbolPaint,
378    properties: &FeatureProperties,
379    zoom: f64,
380) -> Box2D<f32, crate::sdf::TileSpace> {
381    let text_size = paint
382        .text_size
383        .as_ref()
384        .and_then(|value| value.evaluate_for(properties, zoom))
385        .unwrap_or(16.0);
386    let icon_size = paint.number("icon-size", properties, zoom, 1.0);
387    let mut bounds = Box2D::new(
388        Point2D::new(f32::INFINITY, f32::INFINITY),
389        Point2D::new(f32::NEG_INFINITY, f32::NEG_INFINITY),
390    );
391    for index in indices {
392        let vertex = &vertices[*index as usize];
393        let scale = if matches!(vertex.a_data[2], 0 | 3) {
394            text_size / 24.0
395        } else {
396            icon_size
397        };
398        let point: Point2D<f32, crate::sdf::TileSpace> = Point2D::new(
399            vertex.a_pos_offset[2] as f32 / 32.0 * scale,
400            vertex.a_pos_offset[3] as f32 / 32.0 * scale,
401        );
402        bounds.min.x = bounds.min.x.min(point.x);
403        bounds.min.y = bounds.min.y.min(point.y);
404        bounds.max.x = bounds.max.x.max(point.x);
405        bounds.max.y = bounds.max.y.max(point.y);
406    }
407    bounds
