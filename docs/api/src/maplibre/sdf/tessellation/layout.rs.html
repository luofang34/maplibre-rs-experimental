40/// What makes two line labels repeats of each other: their text and the images they show,
41/// both those standing in the text as one placeholder each and the icon behind it, so the
42/// same number on an Interstate shield and on a US-route shield are different routes. A
43/// label without text, such as a one-way arrow, has no repeats to keep apart.
44pub(super) fn repeat_key(
45    paint: &SymbolPaint,
46    properties: &FeatureProperties,
47    zoom: f64,
48    atlas: &SymbolAtlas,
49) -> Option<String> {
50    let images = Some(&atlas.icons as &dyn crate::style::expression::ImageSet);
51    let mut key = paint.label_among(properties, zoom, images)?;
52    for section in paint.label_sections_among(properties, zoom, images) {
53        if let Some(image) = section.image {
54            key.push('\u{0}');
55            key.push_str(&image);
56        }
57    }
58    if let Some(icon) = paint.text_among_images("icon-image", properties, zoom, &atlas.icons) {
59        key.push('\u{1}');
60        key.push_str(&icon);
61    }
62    Some(key)
63}

65/// Where the labels of a geometry go when they do not follow a line, as GL JS places them:
66/// at every point, at the first vertex of every line, and at the pole of inaccessibility of
67/// every polygon.
68pub(super) fn anchors(geometry: &Geometry<f64>) -> Vec<Point<f64>> {
69    let polygon = |polygon: &geo_types::Polygon<f64>| {
70        let rings: Vec<Vec<[f64; 2]>> = std::iter::once(polygon.exterior())
71            .chain(polygon.interiors())
72            .map(|ring| ring.coords().map(|c| [c.x, c.y]).collect())
73            .collect();
74        super::pole::pole_of_inaccessibility(&rings, POLE_PRECISION).map(|[x, y]| Point::new(x, y))
75    };
76    let first = |line: &geo_types::LineString<f64>| {
77        line.0.first().map(|point| Point::new(point.x, point.y))
78    };
79    match geometry {
80        Geometry::Point(point) => vec![*point],
81        Geometry::MultiPoint(points) => points.iter().copied().collect(),
82        Geometry::LineString(line) => first(line).into_iter().collect(),
83        Geometry::MultiLineString(lines) => lines.iter().filter_map(first).collect(),
84        Geometry::Polygon(shape) => polygon(shape).into_iter().collect(),
85        Geometry::MultiPolygon(shapes) => shapes.iter().filter_map(polygon).collect(),
86        other => other.centroid().into_iter().collect(),
87    }
88}

93pub(super) fn append(
94    symbol: &CollectedSymbol,
95    paint: &SymbolPaint,
96    zoom: f64,
97    atlas: &SymbolAtlas,
98    buffer: &mut VertexBuffers<ShaderSymbolVertex, u32>,
99    features: &mut Vec<Feature>,
100) {
101    let start = buffer.indices.len();
102    let first_vertex = buffer.vertices.len();
103    let mut icon_padding = 0.0_f32;
104    let text = paint
105        .label_among(&symbol.properties, zoom, Some(&atlas.icons))
106        .unwrap_or_default();
107    if let Some(icon) = paint
108        .text_among_images("icon-image", &symbol.properties, zoom, &atlas.icons)
109        .and_then(|name| atlas.icons.get(&name))
110    {
111        let ratio = icon.metrics[3];
112        let width = icon.rect[2] as f32 / ratio;
113        let height = icon.rect[3] as f32 / ratio;
114        let offset = offset(paint, "icon-offset", 1.0, (&symbol.properties, zoom));
115        let fractions = anchor_fractions(
116            &paint
117                .text("icon-anchor", &symbol.properties, zoom)
118                .unwrap_or_else(|| "center".into()),
119        );
120        let height_offset = if paint.uses_shared_height() {
121            0.0
122        } else {
123            paint.height_offset("icon", &symbol.properties, zoom)
124        };
125        let icon_size = paint.number("icon-size", &symbol.properties, zoom, 1.0);
126        // An image whose anchor is not its centre, such as a shield with a banner above its
127        // body, is moved so that the anchor stands where the centre would.
128        let recentre = [icon.metrics[0] / ratio, icon.metrics[1] / ratio];
129        let placed = [
130            -width * fractions[0] + offset[0] - recentre[0],
131            -height * fractions[1] + offset[1] - recentre[1],
132            width * (1.0 - fractions[0]) + offset[0] - recentre[0],
133            height * (1.0 - fractions[1]) + offset[1] - recentre[1],
134        ];
135        let fitted = fit_to_text(paint, symbol, zoom, atlas, [width, height], offset);
136        let shift = crate::sdf::translation::tile_translation(paint, "icon", zoom);
137        let anchor = Point::new(symbol.anchor.x() + shift[0], symbol.anchor.y() + shift[1]);
138        let rotation = paint
139            .number("icon-rotate", &symbol.properties, zoom, 0.0)
140            .to_radians()
141            + if super::text_layout::is_vertical(symbol, paint, zoom, atlas) {
142                std::f32::consts::FRAC_PI_2
143            } else {
144                0.0
145            };
146        // Layout pixels are drawn scaled by icon-size, which a fitted icon does not take.
147        let undo_size = if fitted.is_some() {
148            1.0 / icon_size
149        } else {
150            1.0
151        };
152        for piece in super::icon_quads::icon_quads(icon, fitted.unwrap_or(placed), fitted.is_some())
153        {
154            icon_padding = icon_padding.max(piece.padding * undo_size);
155            let part = AtlasEntry {
156                rect: piece.rect,
157                ..icon.clone()
158            };
159            quad(
160                buffer,
161                anchor,
162                piece.bounds.map(|edge| edge * undo_size),
163                &part,
164                height_offset,
165                symbol.angle,
166                rotation,
167            );
168        }
169    }
170    let first_glyph_index = buffer.indices.len();
171    let laid = super::text_layout::append(symbol, paint, zoom, atlas, buffer);
172    let glyph_offsets = laid.centres;
173    if start == buffer.indices.len() {
174        return;
175    }
176    // Collision follows the text where there is one, as the icon alone otherwise.
177    let shift = crate::sdf::translation::tile_translation(
178        paint,
179        if text.is_empty() { "icon" } else { "text" },
180        zoom,
181    );
182    let anchor = Point2D::new(
183        (symbol.anchor.x() + shift[0]) as f32,
184        (symbol.anchor.y() + shift[1]) as f32,
185    );
186    let bbox = bounds(
187        &buffer.vertices,
188        &buffer.indices[start..],
189        paint,
190        &symbol.properties,
191        zoom,
192    );
193    let mut parts = crate::sdf::placement_geometry::measure(&mut buffer.vertices[first_vertex..]);
194    // An icon collides by its image, not by the clear border its quad reaches into.
195    for part in parts[1..].iter_mut().flatten() {
196        let pad = f64::from(icon_padding);
197        part.bounds = [
198            part.bounds[0] + pad,
199            part.bounds[1] + pad,
200            part.bounds[2] - pad,
201            part.bounds[3] - pad,
202        ];
203    }
204    // Text collides by the box of its lines, not by the bitmaps of its glyphs, which reach
205    // a few pixels beyond it.
206    // A label of images alone, such as a row of road shields along a road, has no glyphs to
207    // measure, so its box is the layout's.
208    if parts[0].is_none() {
209        let image = buffer.vertices[first_vertex..]
210            .iter()
211            .find(|vertex| vertex.a_data[2] == 3);
212        if let (Some(vertex), Some(extent)) = (
213            image,
214            super::text_layout::extent(symbol, paint, zoom, atlas),
215        ) {
216            parts[0] = Some(crate::sdf::placement_geometry::SymbolBounds {
217                bounds: extent.map(f64::from),
218                height: f64::from(f32::from_bits(vertex.a_pixeloffset[2] as u32)),
219                angle: f64::from(f32::from_bits(vertex.a_pixeloffset[3] as u32)),
220                text: true,
221            });
222        }
223    }
224    if let (Some(text_part), None) = (&mut parts[0], &symbol.line) {
225        if let Some([left, top, right, bottom]) =
226            super::text_layout::extent(symbol, paint, zoom, atlas)
227        {
228            text_part.bounds = [left, top, right, bottom].map(f64::from);
229        }
230    }
231    features.push(Feature {
232        parts,
233        data: crate::sdf::SymbolFeatureData {
234            id: symbol.source.id,
235            geometry_type: symbol.source.kind,
236            properties: symbol.properties.clone(),
237            sort_key: paint.number("symbol-sort-key", &symbol.properties, zoom, 0.0),
238        },
239        bbox,
240        indices: start..buffer.indices.len(),
241        text_anchor: anchor,
242        anchor_shifts: super::text_layout::variable_shifts(symbol, paint, zoom, atlas),
243        text_sets: laid.sets,
244        anchor_sets: laid.anchor_sets,
245        text_colors: laid.colors,
246        fallback: symbol.fallback,
247        repeat_key: symbol.repeat_key.clone(),
248        str: text,
249        line: symbol
250            .line
251            .as_ref()
252            .filter(|_| !glyph_offsets.is_empty())
253            .map(|(polyline, distance)| crate::sdf::LineLabel {
254                polyline: polyline.clone(),
255                anchor_distance: *distance,
256                glyph_offsets,
257                image_sizes: laid.image_sizes,
258                first_glyph_index,
259            }),
260    });
261}

263/// The icon box `[left, top, right, bottom]` that `icon-text-fit` stretches around the text,
264/// in text-size pixels, or `None` when the icon keeps its own size.
265fn fit_to_text(
266    paint: &SymbolPaint,
267    symbol: &CollectedSymbol,
268    zoom: f64,
269    atlas: &SymbolAtlas,
270    [icon_width, icon_height]: [f32; 2],
271    offset: [f32; 2],
272) -> Option<[f32; 4]> {
273    let fit = paint.text("icon-text-fit", &symbol.properties, zoom)?;
274    if fit == "none" {
275        return None;
276    }
277    let text_size = paint
278        .text_size
279        .as_ref()
280        .and_then(|value| value.evaluate_for(&symbol.properties, zoom))
281        .unwrap_or(16.0);
282    let scale = text_size / 24.0;
283    // A vertical label's icon is fitted before the symbol turns.
284    let [left, top, right, bottom] =
285        super::text_layout::layout_extent(symbol, paint, zoom, atlas)?.map(|edge| edge * scale);
286    let padding = paint
287        .properties
288        .get("icon-text-fit-padding")
289        .and_then(|value| value.as_array())
290        .filter(|values| values.len() == 4)
291        .map_or([0.0; 4], |values| {
292            let read = |index: usize| values[index].as_f64().unwrap_or(0.0) as f32;
293            [read(0), read(1), read(2), read(3)]
294        });
295    let (min_x, max_x) = if matches!(fit.as_str(), "width" | "both") {
296        (
297            offset[0] + left - padding[3],
298            offset[0] + right + padding[1],
299        )
300    } else {
301        let min = offset[0] + (left + right - icon_width) / 2.0;
302        (min, min + icon_width)
303    };
304    let (min_y, max_y) = if matches!(fit.as_str(), "height" | "both") {
305        (
306            offset[1] + top - padding[0],
307            offset[1] + bottom + padding[2],
308        )
309    } else {
310        let min = offset[1] + (top + bottom - icon_height) / 2.0;
311        (min, min + icon_height)
312    };
313    Some([min_x, min_y, max_x, max_y])
314}

316/// The `[x, y]` offset a layout property gives one symbol, times `scale`; zero without one.
317pub(super) fn offset(
318    paint: &SymbolPaint,
319    name: &str,
320    scale: f32,
321    (properties, zoom): (&crate::style::expression::FeatureProperties, f64),
322) -> [f32; 2] {
323    let Some(value) = paint.properties.get(name) else {
324        return [0.0; 2];
325    };
326    // A written pair is the common case and needs no expression.
327    let [x, y] = match crate::style::translation::Pair::from_json(value) {
328        Some(pair) => pair.0,
329        None => {
330            crate::style::property::StyleProperty::<crate::style::translation::Pair>::parse(value)
331                .evaluate_for(properties, zoom)
332                .map_or([0.0; 2], |pair| pair.0)
333        }
334    };
335    [x as f32 * scale, y as f32 * scale]
336}

338pub(super) fn anchor_fractions(anchor: &str) -> [f32; 2] {
339    [
340        if anchor.contains("left") {
341            0.0
342        } else if anchor.contains("right") {
343            1.0
344        } else {
345            0.5
346        },
347        if anchor.contains("top") {
348            0.0
349        } else if anchor.contains("bottom") {
350            1.0
351        } else {
352            0.5
353        },
354    ]
355}

357pub(super) fn quad(
358    buffer: &mut VertexBuffers<ShaderSymbolVertex, u32>,
359    anchor: Point<f64>,
360    bounds: [f32; 4],
361    image: &AtlasEntry,
362    height: f32,
363    angle: f32,
364    rotation: f32,
365) {
366    let base = buffer.vertices.len() as u32;
367    let (sin, cos) = rotation.sin_cos();
368    let [x, y, width, height_pixels] = image.rect;
369    for (index, (u, v)) in [
370        (x, y),
371        (x + width, y),
372        (x + width, y + height_pixels),
373        (x, y + height_pixels),
374    ]
375    .into_iter()
376    .enumerate()
377    {
378        let px = if index == 0 || index == 3 {
379            bounds[0]
380        } else {
381            bounds[2]
382        };
383        let py = if index < 2 { bounds[1] } else { bounds[3] };
384        // The quad turns about the anchor, offsets included, as GL JS rotates its corners.
385        let (px, py) = (px * cos - py * sin, px * sin + py * cos);
386        buffer.vertices.push(ShaderSymbolVertex {
387            a_pos_offset: [
388                anchor.x().round() as i32,
389                anchor.y().round() as i32,
390                (px * 32.0).round() as i32,
391                (py * 32.0).round() as i32,
392            ],
393            a_data: [u, v, image.kind, 0],
394            a_pixeloffset: [0, 0, height.to_bits() as i32, angle.to_bits() as i32],
395        });
396    }
397    buffer
398        .indices
399        .extend([base, base + 1, base + 2, base, base + 2, base + 3]);
400}

402fn bounds(
403    vertices: &[ShaderSymbolVertex],
404    indices: &[u32],
405    paint: &SymbolPaint,
406    properties: &FeatureProperties,
407    zoom: f64,
408) -> Box2D<f32, crate::sdf::TileSpace> {
409    let text_size = paint
410        .text_size
411        .as_ref()
412        .and_then(|value| value.evaluate_for(properties, zoom))
413        .unwrap_or(16.0);
414    let icon_size = paint.number("icon-size", properties, zoom, 1.0);
415    let mut bounds = Box2D::new(
416        Point2D::new(f32::INFINITY, f32::INFINITY),
417        Point2D::new(f32::NEG_INFINITY, f32::NEG_INFINITY),
418    );
419    for index in indices {
420        let vertex = &vertices[*index as usize];
421        let scale = if matches!(vertex.a_data[2], 0 | 3) {
422            text_size / 24.0
423        } else {
424            icon_size
425        };
426        let point: Point2D<f32, crate::sdf::TileSpace> = Point2D::new(
427            vertex.a_pos_offset[2] as f32 / 32.0 * scale,
428            vertex.a_pos_offset[3] as f32 / 32.0 * scale,
429        );
430        bounds.min.x = bounds.min.x.min(point.x);
431        bounds.min.y = bounds.min.y.min(point.y);
432        bounds.max.x = bounds.max.x.max(point.x);
433        bounds.max.y = bounds.max.y.max(point.y);
434    }
435    bounds
