1//! Joins the lines of a layer that end where another begins and carry the same text, as GL JS
2//! `mergeLines` does before it places labels along them, so a road cut into pieces is labelled
3//! as one road.
4
5use std::collections::HashMap;
6
7use crate::style::expression::FeatureProperties;
8
9/// One line to be labelled, with what its label is made from.
10pub(super) struct PendingLine {
11    /// The text of the label; a line without text is never merged.
12    pub text: Option<String>,
13    pub line: Vec<[f64; 2]>,
14    pub source: super::layout::SourceFeature,
15    pub properties: FeatureProperties,
16}
17
18struct Chunk {
19    points: Vec<[f64; 2]>,
20    next: Option<usize>,
21}
22
23struct Merged {
24    feature: usize,
25    first: usize,
26    last: usize,
27    alive: bool,
28}
29
30fn key(text: &str, chunks: &[Chunk], merged: &Merged, on_right: bool) -> String {
31    let points = &chunks[if on_right { merged.last } else { merged.first }].points;
32    let point = if on_right {
33        points[points.len() - 1]
34    } else {
35        points[0]
36    };
37    format!("{text}:{}:{}", point[0], point[1])
38}
39
40struct Lines {
41    chunks: Vec<Chunk>,
42    merged: Vec<Merged>,
43    left: HashMap<String, usize>,
44    right: HashMap<String, usize>,
45}
46
47impl Lines {
48    /// Appends `line` to the merged line that ends where it starts.
49    fn merge_from_right(&mut self, keys: (&str, &str), line: (usize, usize)) -> usize {
50        let (left_key, right_key) = keys;
51        let i = self.right.remove(left_key).unwrap_or_default();
52        self.right.insert(right_key.to_owned(), i);
53        let last = self.merged[i].last;
54        self.chunks[last].next = Some(line.0);
55        self.merged[i].last = line.1;
56        i
57    }
58
59    /// Prepends `line` to the merged line that starts where it ends.
60    fn merge_from_left(&mut self, keys: (&str, &str), line: (usize, usize)) -> usize {
61        let (left_key, right_key) = keys;
62        let i = self.left.remove(right_key).unwrap_or_default();
63        self.left.insert(left_key.to_owned(), i);
64        let first = self.merged[i].first;
65        self.chunks[line.1].next = Some(first);
66        self.merged[i].first = line.0;
67        i
68    }
69
70    fn joined(&self, merged: &Merged) -> Vec<[f64; 2]> {
71        let mut points = self.chunks[merged.first].points.clone();
72        let mut next = self.chunks[merged.first].next;
73        while let Some(index) = next {
74            points.extend_from_slice(&self.chunks[index].points[1..]);
75            next = self.chunks[index].next;
76        }
77        points
78    }
79}

81/// The lines with those that join end to start and share their text made into one.
82pub(super) fn merge_lines(features: Vec<PendingLine>) -> Vec<PendingLine> {
83    let mut lines = Lines {
84        chunks: Vec::new(),
85        merged: Vec::new(),
86        left: HashMap::new(),
87        right: HashMap::new(),
88    };
89    for (index, feature) in features.iter().enumerate() {
90        lines.chunks.push(Chunk {
91            points: feature.line.clone(),
92            next: None,
93        });
94        let chunk = lines.chunks.len() - 1;
95        let line = Merged {
96            feature: index,
97            first: chunk,
98            last: chunk,
99            alive: true,
100        };
101        let text = match &feature.text {
102            Some(text) if feature.line.len() > 1 && !text.is_empty() => text.clone(),
103            _ => {
104                lines.merged.push(line);
105                continue;
106            }
107        };
108        let left_key = key(&text, &lines.chunks, &line, false);
109        let right_key = key(&text, &lines.chunks, &line, true);
110        let keys = (left_key.as_str(), right_key.as_str());
111        let span = (chunk, chunk);
112        let touches_both = lines.right.get(&left_key).zip(lines.left.get(&right_key));
113        if touches_both.is_some_and(|(right, left)| right != left) {
114            // Lines with the same text meet both ends of this one: join all three.
115            let j = lines.merge_from_left(keys, span);
116            let joined = (lines.merged[j].first, lines.merged[j].last);
117            let i = lines.merge_from_right(keys, joined);
118            lines.left.remove(&left_key);
119            lines.right.remove(&right_key);
120            let end = key(&text, &lines.chunks, &lines.merged[i], true);
121            lines.right.insert(end, i);
122            lines.merged[j].alive = false;
123        } else if lines.right.contains_key(&left_key) {
124            lines.merge_from_right(keys, span);
125        } else if lines.left.contains_key(&right_key) {
126            lines.merge_from_left(keys, span);
127        } else {
128            lines.merged.push(line);
129            let i = lines.merged.len() - 1;
130            lines.left.insert(left_key, i);
131            lines.right.insert(right_key, i);
132        }
133    }
134    let mut slots: Vec<Option<PendingLine>> = features.into_iter().map(Some).collect();
135    let mut result = Vec::new();
136    for merged in &lines.merged {
137        if !merged.alive {
138            continue;
139        }
140        let Some(mut feature) = slots[merged.feature].take() else {
141            continue;
142        };
143        feature.line = lines.joined(merged);
144        result.push(feature);
145    }
146    result
147}
