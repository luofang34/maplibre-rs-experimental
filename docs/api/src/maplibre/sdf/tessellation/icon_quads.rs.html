1//! The quads of an icon, split along the stretch zones of its sprite when `icon-text-fit`
2//! fits it to text, as GL JS `getIconQuads` does.
3
4use crate::sdf::assets::{AtlasEntry, IconStretch, TextFit};
5
6/// A piece of an icon: its corners as `[left, top, right, bottom]` in layout pixels and the
7/// atlas rectangle it shows.
8pub(super) struct IconQuad {
9    pub bounds: [f32; 4],
10    pub rect: [u32; 4],
11    /// How far, in layout pixels, the quad reaches beyond the image into its clear border.
12    pub padding: f32,
13}
14
15/// One cut through the image: pixels that keep their size before it, and stretchable
16/// pixels before it.
17#[derive(Clone, Copy)]
18struct Cut {
19    fixed: f32,
20    stretch: f32,
21}
22
23fn total(ranges: &[[f32; 2]]) -> f32 {
24    ranges.iter().map(|range| range[1] - range[0]).sum()
25}
26
27/// How much of the ranges lies between `min` and `max`.
28fn within(ranges: &[[f32; 2]], min: f32, max: f32) -> f32 {
29    ranges
30        .iter()
31        .map(|range| range[1].clamp(min, max) - range[0].clamp(min, max))
32        .sum()
33}
34
35fn cuts(zones: &[[f32; 2]], fixed_size: f32, stretch_size: f32) -> Vec<Cut> {
36    let mut cuts = vec![Cut {
37        fixed: 0.0,
38        stretch: 0.0,
39    }];
40    for [start, end] in zones {
41        let last = cuts[cuts.len() - 1];
42        cuts.push(Cut {
43            fixed: start - last.stretch,
44            stretch: last.stretch,
45        });
46        cuts.push(Cut {
47            fixed: start - last.stretch,
48            stretch: last.stretch + (end - start),
49        });
50    }
51    cuts.push(Cut {
52        fixed: fixed_size,
53        stretch: stretch_size,
54    });
55    cuts
56}

58/// The icon box after `textFitWidth` and `textFitHeight` have pulled it towards the aspect
59/// ratio of the content area.
60fn apply_text_fit(icon: [f32; 4], stretch: &IconStretch, content: [f32; 4]) -> [f32; 4] {
61    let [mut left, mut top, right, bottom] = icon;
62    let (mut width, mut height) = (right - left, bottom - top);
63    let aspect = (content[2] - content[0]) / (content[3] - content[1]);
64    let fit_width = stretch.text_fit_width.unwrap_or(TextFit::StretchOrShrink);
65    let fit_height = stretch.text_fit_height.unwrap_or(TextFit::StretchOrShrink);
66    if fit_height == TextFit::Proportional {
67        if (fit_width == TextFit::StretchOnly && width / height < aspect)
68            || fit_width == TextFit::Proportional
69        {
70            let wider = (height * aspect).ceil();
71            left *= wider / width;
72            width = wider;
73        }
74    } else if fit_width == TextFit::Proportional
75        && fit_height == TextFit::StretchOnly
76        && aspect != 0.0
77        && width / height > aspect
78    {
79        let taller = (width / aspect).ceil();
80        top *= taller / height;
81        height = taller;
82    }
83    [left, top, left + width, top + height]
84}

86/// The pieces of `icon` for a box `[left, top, right, bottom]` in layout pixels. Without
87/// `fitted` the icon is one piece. With it, the stretch zones grow to the box and the rest
88/// keeps its size, and a content area stands for the box itself.
89pub(super) fn icon_quads(icon: &AtlasEntry, boxed: [f32; 4], fitted: bool) -> Vec<IconQuad> {
90    let ratio = icon.metrics[3];
91    let [x, y, width, height] = icon.rect;
92    let (image_width, image_height) = (width as f32, height as f32);
93    let Some(stretch) = icon.stretch.as_deref().filter(|_| fitted) else {
94        if fitted {
95            return vec![IconQuad {
96                bounds: boxed,
97                rect: icon.rect,
98                padding: 0.0,
99            }];
100        }
101        // The quad reaches one texel into the atlas's clear border, so the image fades out at
102        // its edge instead of ending on a hard line.
103        let pad = 1.0 / ratio;
104        let [left, top, right, bottom] = boxed;
105        let [x, y, width, height] = icon.rect;
106        return vec![IconQuad {
107            bounds: [left - pad, top - pad, right + pad, bottom + pad],
108            rect: [
109                x.saturating_sub(1),
110                y.saturating_sub(1),
111                width + 2,
112                height + 2,
113            ],
114            padding: pad,
115        }];
116    };
117    let stretch_x = if stretch.stretch_x.is_empty() {
118        vec![[0.0, image_width]]
119    } else {
120        stretch.stretch_x.clone()
121    };
122    let stretch_y = if stretch.stretch_y.is_empty() {
123        vec![[0.0, image_height]]
124    } else {
125        stretch.stretch_y.clone()
126    };
127    let (stretch_width, stretch_height) = (total(&stretch_x), total(&stretch_y));
128    let (fixed_width, fixed_height) = (image_width - stretch_width, image_height - stretch_height);
129
130    let mut boxed = boxed;
131    let mut offsets = ([0.0, 0.0], [stretch_width, stretch_height]);
132    let mut fixed_offsets = ([0.0, 0.0], [fixed_width, fixed_height]);
133    if let Some(content) = stretch.content {
134        if stretch.text_fit_width.is_some() || stretch.text_fit_height.is_some() {
135            boxed = apply_text_fit(boxed, stretch, content);
136        }
137        let stretch_before = [
138            within(&stretch_x, 0.0, content[0]),
139            within(&stretch_y, 0.0, content[1]),
140        ];
141        let stretch_inside = [
142            within(&stretch_x, content[0], content[2]),
143            within(&stretch_y, content[1], content[3]),
144        ];
145        offsets = (stretch_before, stretch_inside);
146        fixed_offsets = (
147            [
148                content[0] - stretch_before[0],
149                content[1] - stretch_before[1],
150            ],
151            [
152                content[2] - content[0] - stretch_inside[0],
153                content[3] - content[1] - stretch_inside[1],
154            ],
155        );
156    }
157    let [left, top, right, bottom] = boxed;
158    let (icon_width, icon_height) = (right - left, bottom - top);
159
160    // A position along an axis: the stretched share of the box plus the fixed pixels.
161    let place = |cut: Cut, axis: usize| {
162        let (origin, span) = if axis == 0 {
163            (left, icon_width)
164        } else {
165            (top, icon_height)
166        };
167        let stretch_total = if axis == 0 {
168            stretch_width
169        } else {
170            stretch_height
171        };
172        let em = (cut.stretch - offsets.0[axis]) / offsets.1[axis] * span + origin;
173        let px = (cut.fixed - fixed_offsets.0[axis])
174            - fixed_offsets.1[axis] * cut.stretch / stretch_total;
175        // A box smaller than the pixels that cannot stretch grows to hold them: the stretched
176        // share never scales below the fixed content.
177        let at_least = (fixed_offsets.1[axis] / ratio / span).max(1.0);
178        em * at_least + px / ratio
179    };
180    let x_cuts = cuts(&stretch_x, fixed_width, stretch_width);
181    let y_cuts = cuts(&stretch_y, fixed_height, stretch_height);
182    let mut quads = Vec::new();
183    for columns in x_cuts.windows(2) {
184        for rows in y_cuts.windows(2) {
185            let (x1, x2) = (
186                columns[0].stretch + columns[0].fixed,
187                columns[1].stretch + columns[1].fixed,
188            );
189            let (y1, y2) = (
190                rows[0].stretch + rows[0].fixed,
191                rows[1].stretch + rows[1].fixed,
192            );
193            if x2 <= x1 || y2 <= y1 {
194                continue;
195            }
196            quads.push(IconQuad {
197                bounds: [
198                    place(columns[0], 0),
199                    place(rows[0], 1),
200                    place(columns[1], 0),
201                    place(rows[1], 1),
202                ],
203                rect: [
204                    x + x1 as u32,
205                    y + y1 as u32,
206                    (x2 - x1) as u32,
207                    (y2 - y1) as u32,
208                ],
209                padding: 0.0,
210            });
211        }
212    }
213    quads
214}
