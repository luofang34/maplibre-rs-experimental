1//! Right-to-left text in the order it reads on a line.
2use std::ops::Range;
3
4use unicode_bidi::{bidi_class, BidiClass, BidiInfo};
5
6/// Whether `c` reads from right to left.
7fn is_rtl(c: char) -> bool {
8    matches!(
9        bidi_class(c),
10        BidiClass::R | BidiClass::AL | BidiClass::RLE | BidiClass::RLO | BidiClass::RLI
11    )
12}
13
14/// Whether `c` only steers the bidirectional algorithm: marks, embeddings, overrides and
15/// isolates, which GL JS's bidirectional pass leaves out of the text it draws.
16pub(super) fn is_bidi_control(c: char) -> bool {
17    matches!(
18        c,
19        '\u{061C}' | '\u{200E}' | '\u{200F}' | '\u{202A}'..='\u{202E}' | '\u{2066}'..='\u{2069}'
20    )
21}
22
23/// The lines with each also ended after any paragraph separator in it: GL JS's bidirectional
24/// pass takes the separators as paragraph ends on top of the breaks that fit the width.
25pub(super) fn split_paragraphs(chars: &[char], lines: Vec<Range<usize>>) -> Vec<Range<usize>> {
26    let mut split = Vec::with_capacity(lines.len());
27    for line in lines {
28        let mut start = line.start;
29        for index in line.clone() {
30            if index + 1 < line.end && bidi_class(chars[index]) == BidiClass::B {
31                split.push(start..index + 1);
32                start = index + 1;
33            }
34        }
35        split.push(start..line.end);
36    }
37    split
38}

40/// The indices of the characters of `line`, in the order they are drawn from left to right.
41fn visual_order(chars: &[char], line: Range<usize>) -> Vec<usize> {
42    let logical: Vec<usize> = line.clone().collect();
43    if !chars[line.clone()].iter().any(|c| is_rtl(*c)) {
44        return logical;
45    }
46    let text: String = chars[line.clone()].iter().collect();
47    // The byte at which each character starts.
48    let starts: Vec<usize> = text.char_indices().map(|(at, _)| at).collect();
49    let char_at = |byte: usize| starts.partition_point(|start| *start < byte);
50    let info = BidiInfo::new(&text, None);
51    let mut order = Vec::with_capacity(logical.len());
52    for paragraph in &info.paragraphs {
53        let (levels, runs) = info.visual_runs(paragraph, paragraph.range.clone());
54        for run in runs {
55            let mut indices: Vec<usize> = (char_at(run.start)..char_at(run.end)).collect();
56            if levels[run.start].is_rtl() {
57                indices.reverse();
58            }
59            order.extend(indices);
60        }
61    }
62    if order.len() != logical.len() {
63        return logical;
64    }
65    order.into_iter().map(|index| line.start + index).collect()
66}

68/// Puts the characters of each line, and what goes with each, in the order they read on the line.
69pub(super) fn reorder_lines<T: Clone>(
70    chars: &mut [char],
71    styles: &mut [T],
72    lines: &[Range<usize>],
73) {
74    for line in lines {
75        let order = visual_order(chars, line.clone());
76        let (reordered_chars, reordered_styles): (Vec<char>, Vec<T>) = order
77            .iter()
78            .map(|index| (chars[*index], styles[*index].clone()))
79            .unzip();
80        chars[line.clone()].copy_from_slice(&reordered_chars);
81        styles[line.clone()].clone_from_slice(&reordered_styles);
82    }
83}
