1//! Text written top to bottom: `text-writing-mode: vertical`.
2//!
3//! The label is laid out along a horizontal line, as GL JS shapes it, and the whole label then
4//! turns a quarter clockwise about its anchor. Characters that stand upright in vertical text
5//! take a full em of the line and are turned back, so they read upright on the map; the rest
6//! keep their own advance and lie on their side.
7use std::ops::Range;
8
9use super::{styles::CharStyle, Emitter};
10use crate::{
11    sdf::{assets::AtlasEntry, tessellation::layout::anchor_fractions},
12    style::layer::SymbolPaint,
13};

15/// Where a vertical glyph's box sits from its line, which GL JS measures from the em box rather
16/// than from the baseline.
17const UPRIGHT_BASELINE: f32 = 17.0;
18/// The offset along the line that centres an upright glyph on its em box.
19const UPRIGHT_CENTRE: f32 = 7.0;
20
21/// Whether the first entry of `text-writing-mode` is `vertical`, which a label written both ways
22/// takes while it is placeable.
23pub(super) fn prefers_vertical(paint: &SymbolPaint) -> bool {
24    paint
25        .properties
26        .get("text-writing-mode")
27        .and_then(serde_json::Value::as_array)
28        .and_then(|modes| modes.first())
29        .and_then(serde_json::Value::as_str)
30        == Some("vertical")
31}
32
33/// The ways `text-writing-mode` allows, best first, as `[first, second]` with `true` for
34/// vertical; `None` unless it names both.
35pub(super) fn orientations(paint: &SymbolPaint) -> Option<[bool; 2]> {
36    let modes = paint
37        .properties
38        .get("text-writing-mode")?
39        .as_array()?
40        .iter()
41        .filter_map(serde_json::Value::as_str)
42        .collect::<Vec<_>>();
43    let vertical_first = modes.iter().position(|mode| *mode == "vertical")?
44        < modes.iter().position(|mode| *mode == "horizontal")?;
45    Some([vertical_first, !vertical_first])
46}

48/// Whether the text has a character that stands upright in vertical writing, the only text that
49/// is written vertically.
50pub(super) fn allows_vertical_writing(chars: &[char]) -> bool {
51    chars.iter().any(|c| has_upright_orientation(*c))
52}

54/// Whether a character of a vertical label stands upright: all but spaces and Arabic script.
55pub(super) fn is_upright(c: char) -> bool {
56    !(c.is_whitespace() || in_arabic_script(c))
57}

136/// How far a character advances along a vertical label's line: a full em for an upright one.
137pub(super) fn advance(style: &CharStyle<'_>, c: char) -> f32 {
138    if let Some((_, size, _)) = style.image {
139        // An image takes its height along the line.
140        return size[1];
141    }
142    if is_upright(c) {
143        24.0 * style.scale
144    } else {
145        style.advance(c)
146    }
147}

149/// Length in layout pixels of a vertical label's line, without the spacing after the last glyph.
150pub(super) fn line_length(
151    chars: &[char],
152    styles: &[CharStyle<'_>],
153    line: Range<usize>,
154    spacing: f32,
155) -> f32 {
156    let advances: Vec<f32> = line
157        .filter(|index| styles[*index].is_drawn(chars[*index]))
158        .map(|index| advance(&styles[index], chars[index]) + spacing)
159        .collect();
160    if advances.is_empty() {
161        0.0
162    } else {
163        advances.iter().sum::<f32>() - spacing
164    }
165}

167/// Turns a layout box `[left, top, right, bottom]` a quarter clockwise about the anchor.
168pub(super) fn turned_box([left, top, right, bottom]: [f32; 4]) -> [f32; 4] {
169    [-bottom, left, -top, right]
170}

172/// Where a box of `[width, height]` on the screen sits from its anchor when it takes `anchor`
173/// with `offset`, relative to being centred on it.
174pub(super) fn anchor_shift(anchor: &str, [width, height]: [f32; 2], offset: [f32; 2]) -> [f32; 2] {
175    let [x, y] = anchor_fractions(anchor);
176    [
177        -width * (x - 0.5) + offset[0],
178        -height * (y - 0.5) + offset[1],
179    ]
180}

182/// A line of a vertical label as it is placed.
183pub(super) struct VerticalLine {
184    /// Where the line's glyph boxes start, from the top of the label before the label turns.
185    pub baseline: f32,
186    /// The largest scale on the line.
187    pub largest: f32,
188    /// How far the column grows past an em to hold an image wider than that.
189    pub extra: f32,
190    /// The width of the column's content: an em of the largest scale, or its widest image.
191    pub content: f32,
192    /// Where along the line the first glyph starts.
193    pub pen: f32,
194}
195
196/// Adds the quads of the characters `line` holds.
197pub(super) fn place_line(
198    emit: &mut Emitter<'_>,
199    (chars, styles, spacing): (&[char], &[CharStyle<'_>], f32),
200    line: Range<usize>,
201    placed: &VerticalLine,
202) {
203    let mut pen = placed.pen;
204    // How far the line shifts its glyphs sideways, as GL JS keeps the column centred.
205    let column_offset = ((placed.largest - 1.0) * 24.0).max(placed.extra);
206    for index in line {
207        let style = &styles[index];
208        let c = chars[index];
209        if let Some((entry, size, hang)) = style.image {
210            let entry = AtlasEntry {
211                kind: 3,
212                ..entry.clone()
213            };
214            let half_advance = size[1] / 2.0;
215            let left = hang[0] - half_advance;
216            let top = hang[1];
217            let line_offset = column_offset / 2.0 + (24.0 - size[0]) / 2.0;
218            let across = -(placed.baseline + placed.content - size[1] - line_offset
219                + UPRIGHT_BASELINE)
220                + (12.0 - half_advance);
221            let along = pen - UPRIGHT_CENTRE;
222            emit.quad_rotated(
223                &entry,
224                [
225                    left + across,
226                    top + along,
227                    left + size[0] + across,
228                    top + size[1] + along,
229                ],
230                0.0,
231            );
232            pen += size[1] + spacing;
233            continue;
234        }
235        let Some(glyph) = style.glyphs.get(&(c as u32)) else {
236            continue;
237        };
238        let scale = style.scale;
239        let line_offset = column_offset / 2.0 - (scale - 1.0) * 24.0;
240        if glyph.rect[2] > 0 && glyph.rect[3] > 0 {
241            let first_index = emit.buffer.indices.len();
242            let (width, height) = (glyph.rect[2] as f32 * scale, glyph.rect[3] as f32 * scale);
243            if is_upright(c) {
244                emit_upright(
245                    emit,
246                    glyph,
247                    (pen, scale),
248                    (width, height),
249                    placed,
250                    line_offset,
251                );
252            } else {
253                // Lying on its side: placed along the line, then turned with the label.
254                let x = pen + glyph.metrics[0] * scale;
255                let y = placed.baseline - glyph.metrics[1] * scale + placed.content
256                    - scale * 24.0
257                    - line_offset;
258                emit.quad_rotated(
259                    glyph,
260                    [x, y, x + width, y + height],
261                    std::f32::consts::FRAC_PI_2,
262                );
263            }
264            if let Some(color) = style.color {
265                emit.colour(first_index, color);
266            }
267        }
268        pen += advance(style, c) + spacing;
269    }
270}

272/// An upright glyph: turned back, so its quad is the plain box where the label's turn puts it.
273fn emit_upright(
274    emit: &mut Emitter<'_>,
275    glyph: &AtlasEntry,
276    (pen, scale): (f32, f32),
277    (width, height): (f32, f32),
278    placed: &VerticalLine,
279    line_offset: f32,
280) {
281    let half_advance = glyph.metrics[2] * scale / 2.0;
282    let from_baseline = placed.content - scale * 24.0;
283    let left = glyph.metrics[0] * scale
284        - half_advance
285        - (placed.baseline + from_baseline - line_offset + UPRIGHT_BASELINE);
286    let top = -glyph.metrics[1] * scale + pen - UPRIGHT_CENTRE;
287    emit.quad_rotated(glyph, [left, top, left + width, top + height], 0.0);
288}
