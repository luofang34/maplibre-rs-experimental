19/// How far the label moves, in layout pixels, when it takes each of its variable anchors
20/// instead of the first; empty unless the label has several.
21pub(in crate::sdf::tessellation) fn variable_shifts(
22    symbol: &CollectedSymbol,
23    paint: &SymbolPaint,
24    zoom: f64,
25    atlas: &SymbolAtlas,
26) -> Vec<[f32; 2]> {
27    let anchors = variable_anchors(paint);
28    if anchors.len() < 2 {
29        return Vec::new();
30    }
31    let Some(block) = block(symbol, paint, zoom, atlas) else {
32        return Vec::new();
33    };
34    let corner = |anchor: &str| {
35        let offset = anchored_offset(paint, &anchors, anchor, (&symbol.properties, zoom));
36        if block.vertical {
37            // The turned box is as wide as the label is tall.
38            return vertical::anchor_shift(anchor, [block.height, block.max_line], offset);
39        }
40        let fractions = anchor_fractions(anchor);
41        [
42            -block.max_line * fractions[0] + offset[0],
43            -block.height * fractions[1] + offset[1],
44        ]
45    };
46    let first = corner(&anchors[0]);
47    anchors
48        .iter()
49        .map(|anchor| {
50            let corner = corner(anchor);
51            [corner[0] - first[0], corner[1] - first[1]]
52        })
53        .collect()
54}

56/// The two ways a point label whose style allows both is written, as `[first, second]` with
57/// `true` for vertical; `None` for a text with one.
58pub(in crate::sdf::tessellation) fn both_orientations(
59    paint: &SymbolPaint,
60    text: &str,
61) -> Option<[bool; 2]> {
62    let chars: Vec<char> = text.chars().collect();
63    if crate::sdf::tessellation::is_line_placed(paint) || !vertical::allows_vertical_writing(&chars)
64    {
65        return None;
66    }
67    vertical::orientations(paint)
68}

70/// Whether the label is written top to bottom, which turns the whole symbol a quarter clockwise.
71pub(in crate::sdf::tessellation) fn is_vertical(
72    symbol: &CollectedSymbol,
73    paint: &SymbolPaint,
74    zoom: f64,
75    atlas: &SymbolAtlas,
76) -> bool {
77    block(symbol, paint, zoom, atlas).is_some_and(|block| block.vertical)
78}

80/// The label's layout box `[left, top, right, bottom]` before a vertical label turns, in layout
81/// pixels.
82pub(in crate::sdf::tessellation) fn layout_extent(
83    symbol: &CollectedSymbol,
84    paint: &SymbolPaint,
85    zoom: f64,
86    atlas: &SymbolAtlas,
87) -> Option<[f32; 4]> {
88    let block = block(symbol, paint, zoom, atlas)?;
89    if block.max_line <= 0.0 {
90        return None;
91    }
92    let left = -block.max_line * block.fractions[0] + block.offset[0];
93    let top = -block.height * block.fractions[1] + block.offset[1];
94    Some([left, top, left + block.max_line, top + block.height])
95}

97/// The label's layout box `[left, top, right, bottom]` around the anchor, in layout pixels.
98pub(in crate::sdf::tessellation) fn extent(
99    symbol: &CollectedSymbol,
100    paint: &SymbolPaint,
101    zoom: f64,
102    atlas: &SymbolAtlas,
103) -> Option<[f32; 4]> {
104    let block = block(symbol, paint, zoom, atlas)?;
105    if block.max_line <= 0.0 {
106        return None;
107    }
108    let left = -block.max_line * block.fractions[0] + block.offset[0];
109    let top = -block.height * block.fractions[1] + block.offset[1];
110    let laid = [left, top, left + block.max_line, top + block.height];
111    Some(if block.vertical {
112        vertical::turned_box(laid)
113    } else {
114        laid
115    })
116}

118/// Width in layout pixels of the label on one line, without wrapping; zero without glyphs.
119pub(in crate::sdf::tessellation) fn unwrapped_width(
120    paint: &SymbolPaint,
121    symbol: &CollectedSymbol,
122    zoom: f64,
123    atlas: &SymbolAtlas,
124) -> f32 {
125    let Some(text) = paint.label_among(&symbol.properties, zoom, Some(&atlas.icons)) else {
126        return 0.0;
127    };
128    let chars: Vec<char> = text.chars().collect();
129    let Some(styles) = char_styles(paint, symbol, zoom, atlas, chars.len()) else {
130        return 0.0;
131    };
132    let spacing = paint.number("text-letter-spacing", &symbol.properties, zoom, 0.0) * 24.0;
133    line_width(&chars, &styles, 0..chars.len(), spacing)
134}
