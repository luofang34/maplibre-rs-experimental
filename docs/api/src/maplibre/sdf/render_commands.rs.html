13pub struct SetSymbolPipeline;
14impl<P: PhaseItem> RenderCommand<P> for SetSymbolPipeline {
15    fn render<'w>(
16        world: &'w World,
17        _item: &P,
18        pass: &mut crate::render::tracked_pass::TrackedRenderPass<'w>,
19    ) -> RenderCommandResult {
20        let Some((Initialized(symbol_pipeline), Initialized(projection_resources))) =
21            world.resources.query::<(
22                &Eventually<SymbolPipeline>,
23                &Eventually<ProjectionGpuResources>,
24            )>()
25        else {
26            return RenderCommandResult::Failure;
27        };
28        let Some(_) = symbol_pipeline
29            .for_globe(projection_resources.is_globe_active(ProjectionBinding::View))
30        else {
31            return RenderCommandResult::Failure;
32        };
33
34        pass.set_bind_group(0, projection_resources.bind_group(), &[]);
35        let Some(Initialized(depth)) = world
36            .resources
37            .get::<Eventually<super::depth::SymbolDepth>>()
38        else {
39            return RenderCommandResult::Failure;
40        };
41        pass.set_bind_group(2, &depth.binding, &[]);
42        RenderCommandResult::Success
43    }
44}

46pub struct DrawSymbol;
47impl RenderCommand<TranslucentItem> for DrawSymbol {
48    fn render<'w>(
49        world: &'w World,
50        item: &TranslucentItem,
51        pass: &mut crate::render::tracked_pass::TrackedRenderPass<'w>,
52    ) -> RenderCommandResult {
53        let Some((
54            Initialized(symbol_buffer_pool),
55            covering,
56            Initialized(pipeline),
57            Initialized(projection_resources),
58        )) = world.resources.query::<(
59            &Eventually<SymbolBufferPool>,
60            &super::covering::SymbolCovering,
61            &Eventually<SymbolPipeline>,
62            &Eventually<ProjectionGpuResources>,
63        )>()
64        else {
65            return RenderCommandResult::Failure;
66        };
67        let Some(mode) =
68            pipeline.for_globe(projection_resources.is_globe_active(ProjectionBinding::View))
69        else {
70            return RenderCommandResult::Failure;
71        };
72
73        let Some(entry) = symbol_buffer_pool.index().resolve_entry(
74            item.tile.coords,
75            item.style_layer,
76            item.entry_slot,
77        ) else {
78            return RenderCommandResult::Failure;
79        };
80
81        let Some(binding) = world
82            .resources
83            .get::<SymbolTextures>()
84            .and_then(|textures| textures.binding_by_id(item.tile.coords, item.style_layer))
85        else {
86            return RenderCommandResult::Failure;
87        };
88        pass.set_bind_group(1, &binding.group, &[]);
89
90        let index_range = entry.indices_buffer_range();
91
92        if index_range.is_empty() {
93            tracing::error!("Tried to draw a vector tile without any vertices");
94            return RenderCommandResult::Failure;
95        }
96
97        let whole = world
98            .resources
99            .get::<crate::render::tracked_pass::DrawCapabilities>()
100            .is_some_and(|capabilities| capabilities.base_vertex);
101        let (first, base_vertex) = symbol_buffer_pool.bind_layer(pass, entry, (2, whole));
102        let instances = if whole {
103            let Some(tile_instance) = item.source_shape.instance_index() else {
104                return RenderCommandResult::Failure;
105            };
106            pass.set_vertex_buffer(1, covering.buffer.slice(..));
107            tile_instance..tile_instance + 1
108        } else {
109            let Some(tile_view_pattern_buffer) = item.source_shape.buffer_range() else {
110                return RenderCommandResult::Failure;
111            };
112            pass.set_vertex_buffer(1, covering.buffer.slice(tile_view_pattern_buffer));
113            0..1
114        };
115        draw_indices(
116            pass,
117            mode,
118            (
119                first..first + entry.indices_range().len() as u32,
120                base_vertex,
121                instances,
122            ),
123            binding.separate_halo,
124        )
125    }
126}
127
128fn draw_indices<'w>(
129    pass: &mut crate::render::tracked_pass::TrackedRenderPass<'w>,
130    pipeline: &'w SymbolModePipelines,
131    (indices, base_vertex, instances): (std::ops::Range<u32>, i32, std::ops::Range<u32>),
132    separate_halo: bool,
133) -> RenderCommandResult {
134    if separate_halo {
135        let (Some(halo), Some(fill)) = (&pipeline.halo, &pipeline.fill) else {
136            return RenderCommandResult::Failure;
137        };
138        // All halos must precede fills so an adjacent glyph cannot cover a finished stroke.
139        pass.set_pipeline(halo);
140        pass.draw_indexed(indices.clone(), base_vertex, instances.clone());
141        pass.set_pipeline(fill);
142        pass.draw_indexed(indices, base_vertex, instances);
143    } else {
144        let Some(combined) = &pipeline.combined else {
145            return RenderCommandResult::Failure;
146        };
147        pass.set_pipeline(combined);
148        pass.draw_indexed(indices, base_vertex, instances);
149    }
150    RenderCommandResult::Success
151}
