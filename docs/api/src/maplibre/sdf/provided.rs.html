1//! The tiles whose labels name provided images: which names they drew, at what pixel ratio,
2//! and whether their worker still waits for some of them.
3//!
4//! A tile's worker lays its labels out at once with what is ready and keeps waiting for the
5//! rest; the map keeps drawing frames while it does, so the labels it sends again when the
6//! images arrive reach the screen of an otherwise idle map. A tile that leaves the view stops
7//! its waiting worker. A tile is requested again only when its images must be made anew: its
8//! namespace was invalidated, the display's pixel ratio changed, or a provider was
9//! temporarily unavailable.
10

24/// The wait before a tile whose provider was unavailable is requested again, doubling with
25/// each unavailable request in a row up to [`RETRY_LAST`].
26const RETRY_FIRST: Duration = Duration::from_secs(1);
27/// The longest wait between requests of a tile whose provider stays unavailable.
28const RETRY_LAST: Duration = Duration::from_secs(60);
29
30/// Where a tile's worker is with the provided images its labels name.
31#[derive(Clone, Copy, Debug, PartialEq, Eq, Serialize, Deserialize)]
32pub enum ProvidedImagesState {
33    /// Labels were sent with the images that were ready; the worker waits for the rest.
34    Awaiting,
35    /// Every image has an answer, and labels that gained one were sent again.
36    Settled,
37    /// A provider was unavailable for now; the tile is requested again after a back-off.
38    Retry,
39}
40
41/// A worker's report on the provided images one tile's labels name.
42#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
43pub struct ProvidedImagesReport {
44    /// The tile.
45    pub coords: WorldTileCoords,
46    /// The request the report belongs to.
47    pub attempt: Option<u64>,
48    /// The provided names the tile's labels ask for.
49    pub names: Vec<String>,
50    /// The pixel ratio the images were asked for.
51    pub pixel_ratio: f32,
52    /// Where the worker is.
53    pub state: ProvidedImagesState,
54}
55
56/// Routes [`ProvidedImagesReport`]s.
57#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
58pub struct ProvidedImagesTag;
59
60impl MessageTag for ProvidedImagesTag {
61    fn dyn_clone(&self) -> Box<dyn MessageTag> {
62        Box::new(*self)
63    }
64}

66impl ProvidedImagesReport {

79/// Device pixels per layout pixel of the display, as the host reported it. Without it the
80/// ratio is read from the surface and viewport sizes, which rounding makes slightly off.
81#[derive(Clone, Copy, Debug)]
82pub(crate) struct DisplayPixelRatio(pub(crate) f32);

84/// Records the display's pixel ratio, which provided images are made for.
85pub(crate) fn set_pixel_ratio(world: &mut World, ratio: f64) {
86    world.resources.insert(DisplayPixelRatio(ratio as f32));
87}

89/// The provided images of one tile's labels.
90#[derive(Default)]
91pub(crate) struct TileProvidedImages {
92    names: Vec<String>,
93    pixel_ratio: f32,
94    /// The request whose worker still waits for images.
95    awaiting: Option<u64>,
96    /// Requests in a row that found a provider unavailable, which the back-off doubles with.
97    unavailable: u32,
98}
99
100impl TileComponent for TileProvidedImages {}

102/// Applies a worker's report, unless a newer request for the tile replaced the one it is from.
103pub(crate) fn apply(world: &mut World, report: ProvidedImagesReport) {
104    let coords = report.coords;
105    if !tile_retry::accepts(world, coords, RequestKind::Vector, report.attempt) {
106        return;
107    }
108    if world.tiles.query::<&TileProvidedImages>(coords).is_none() {
109        let Some(mut tile) = world.tiles.spawn_mut(coords) else {
110            return;
111        };
112        tile.insert(TileProvidedImages::default());
113    }
114    let mut retry = None;
115    if let Some(tile) = world.tiles.query_mut::<&mut TileProvidedImages>(coords) {
116        tile.names = report.names;
117        tile.pixel_ratio = report.pixel_ratio;
118        tile.awaiting = (report.state == ProvidedImagesState::Awaiting)
119            .then_some(report.attempt)
120            .flatten();
121        match report.state {
122            ProvidedImagesState::Awaiting => {}
123            ProvidedImagesState::Settled => tile.unavailable = 0,
124            ProvidedImagesState::Retry => {
125                retry = Some(RETRY_FIRST.saturating_mul(1 << tile.unavailable.min(6)));
126                tile.unavailable = tile.unavailable.saturating_add(1);
127            }
128        }
129    }
130    if let Some(delay) = retry {
131        tile_retry::retry_later(world, coords, RequestKind::Vector, delay.min(RETRY_LAST));
132    }
133    crate::render::frame_signals::mark_dirty(world);
134}

136/// Whether the tile's labels draw provided images or still wait for some, which a request
137/// that processes only part of the tile's sources could not account for.
138pub(crate) fn uses_images(world: &World, coords: WorldTileCoords) -> bool {
139    world
140        .tiles
141        .query::<&TileProvidedImages>(coords)
142        .is_some_and(|images| !images.names.is_empty() || images.awaiting.is_some())
143}

145/// Whether a tile's worker still waits for images, so frames must keep coming to show them.
146pub(crate) fn awaiting(world: &World) -> bool {
147    world.tiles.tiles.values().any(|tile| {
148        world
149            .tiles
150            .query::<&TileProvidedImages>(tile.coords)
151            .is_some_and(|images| images.awaiting.is_some())
152    })
153}

155/// Whether the tile drew provided images for another pixel ratio than `pixel_ratio`.
156pub(crate) fn drawn_for_another_ratio(
157    world: &World,
158    coords: WorldTileCoords,
159    pixel_ratio: f32,
160) -> bool {
161    world
162        .tiles
163        .query::<&TileProvidedImages>(coords)
164        .is_some_and(|images| !images.names.is_empty() && images.pixel_ratio != pixel_ratio)
165}

167/// Notes that a tile was requested anew for `pixel_ratio`, so it is not requested again for
168/// that ratio before its worker reports, and forgets its wait, returning the attempt whose
169/// worker waited: the new request reports its own images, and the old one's labels would be
170/// refused.
171pub(crate) fn restarted(
172    world: &mut World,
173    coords: WorldTileCoords,
174    pixel_ratio: f32,
175) -> Option<u64> {
176    let images = world.tiles.query_mut::<&mut TileProvidedImages>(coords)?;
177    images.pixel_ratio = pixel_ratio;
178    // The new request reports the names it draws; a style that dropped them leaves none.
179    images.names.clear();
180    images.awaiting.take()
181}

183/// Stops the waiting workers of tiles no longer wanted, returning their attempts.
184pub(crate) fn release_unwanted(world: &mut World, wanted: &HashSet<WorldTileCoords>) -> Vec<u64> {
185    let coords: Vec<WorldTileCoords> = world
186        .tiles
187        .tiles
188        .values()
189        .map(|tile| tile.coords)
190        .filter(|coords| !wanted.contains(coords))
191        .collect();
192    let mut released = Vec::new();
193    for coords in coords {
194        if let Some(images) = world.tiles.query_mut::<&mut TileProvidedImages>(coords) {
195            released.extend(images.awaiting.take());
196        }
197    }
198    released
199}

201/// Forgets the answers of `namespace` in `providers` and requests again the tiles that drew
202/// them, returning how many: what a map does when a host invalidates a namespace.
203pub(crate) fn invalidate_namespace(
204    providers: Option<&crate::sdf::assets::ImageProviders>,
205    world: &mut World,
206    namespace: &str,
207) -> usize {
208    if let Some(providers) = providers {
209        providers.invalidate(namespace);
210    }
211    invalidate(world, namespace)
212}
213
214/// Requests again the tiles whose labels name an image of `namespace`, returning how many.
215pub(crate) fn invalidate(world: &mut World, namespace: &str) -> usize {
216    let prefix = format!("{namespace}:");
217    let tiles: Vec<WorldTileCoords> = world
218        .tiles
219        .tiles
220        .values()
221        .map(|tile| tile.coords)
222        .filter(|coords| {
223            world
224                .tiles
225                .query::<&TileProvidedImages>(*coords)
226                .is_some_and(|images| images.names.iter().any(|name| name.starts_with(&prefix)))
227        })
228        .collect();
229    tile_retry::refresh_tiles(world, RequestKind::Vector, &tiles);
230    tiles.len()
231}
