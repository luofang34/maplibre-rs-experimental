1//! Worker transport for symbol collision geometry and atlases.
2use serde::{Deserialize, Serialize};
3
4use crate::{
5    euclid::{Box2D, Point2D},
6    sdf::Feature,
7};

9/// Serializable collision geometry for one label and its icon.
10#[derive(Serialize, Deserialize)]
11pub struct SymbolFeature {
12    bounds: [f32; 4],
13    indices: [usize; 2],
14    anchor: [f32; 2],
15    text: String,
16    #[serde(default)]
17    parts: [Option<crate::sdf::placement_geometry::SymbolBounds>; 3],
18    #[serde(default)]
19    id: Option<u64>,
20    #[serde(default)]
21    properties: std::collections::BTreeMap<String, serde_json::Value>,
22    #[serde(default)]
23    sort_key: f32,
24    #[serde(default)]
25    geometry_type: crate::style::filter::GeometryType,
26    #[serde(default)]
27    line: Option<LineLabelWire>,
28    #[serde(default)]
29    anchor_shifts: Vec<[f32; 2]>,
30    #[serde(default)]
31    text_sets: Vec<[usize; 2]>,
32    #[serde(default)]
33    anchor_sets: Vec<u8>,
34    #[serde(default)]
35    text_colors: Vec<([usize; 2], [f32; 4])>,
36    #[serde(default)]
37    fallback: bool,
38    #[serde(default)]
39    repeat_key: Option<String>,
40}

42/// The line a label follows, in a form that survives the worker boundary.
43#[derive(Serialize, Deserialize)]
44pub struct LineLabelWire {
45    polyline: Vec<[f32; 2]>,
46    anchor_distance: f32,
47    glyph_offsets: Vec<f32>,
48    #[serde(default)]
49    image_sizes: Vec<[f32; 2]>,
50    first_glyph_index: usize,
51}

53impl From<&Feature> for SymbolFeature {
54    fn from(feature: &Feature) -> Self {
55        Self {
56            bounds: [
57                feature.bbox.min.x,
58                feature.bbox.min.y,
59                feature.bbox.max.x,
60                feature.bbox.max.y,
61            ],
62            indices: [feature.indices.start, feature.indices.end],
63            anchor: [feature.text_anchor.x, feature.text_anchor.y],
64            text: feature.str.clone(),
65            parts: feature.parts,
66            id: feature.data.id,
67            properties: feature
68                .data
69                .properties
70                .iter()
71                .map(|(key, value)| (key.clone(), value.to_json()))
72                .collect(),
73            sort_key: feature.data.sort_key,
74            geometry_type: feature.data.geometry_type,
75            anchor_shifts: feature.anchor_shifts.clone(),
76            text_sets: feature
77                .text_sets
78                .iter()
79                .map(|set| [set.start, set.end])
80                .collect(),
81            anchor_sets: feature.anchor_sets.clone(),
82            text_colors: feature
83                .text_colors
84                .iter()
85                .map(|(range, color)| ([range.start, range.end], *color))
86                .collect(),
87            fallback: feature.fallback,
88            repeat_key: feature.repeat_key.as_deref().map(str::to_owned),
89            line: feature.line.as_ref().map(|line| LineLabelWire {
90                polyline: line.polyline.to_vec(),
91                anchor_distance: line.anchor_distance,
92                glyph_offsets: line.glyph_offsets.clone(),
93                image_sizes: line.image_sizes.clone(),
94                first_glyph_index: line.first_glyph_index,
95            }),
96        }
97    }
98}

100impl From<SymbolFeature> for Feature {
101    fn from(feature: SymbolFeature) -> Self {
102        Self {
103            bbox: Box2D::new(
104                Point2D::new(feature.bounds[0], feature.bounds[1]),
105                Point2D::new(feature.bounds[2], feature.bounds[3]),
106            ),
107            indices: feature.indices[0]..feature.indices[1],
108            text_anchor: Point2D::new(feature.anchor[0], feature.anchor[1]),
109            str: feature.text,
110            parts: feature.parts,
111            data: crate::sdf::SymbolFeatureData {
112                id: feature.id,
113                properties: feature
114                    .properties
115                    .into_iter()
116                    .map(|(key, value)| (key, crate::style::expression::Value::from_json(&value)))
117                    .collect(),
118                sort_key: feature.sort_key,
119                geometry_type: feature.geometry_type,
120            },
121            anchor_shifts: feature.anchor_shifts,
122            text_sets: feature.text_sets.iter().map(|set| set[0]..set[1]).collect(),
123            anchor_sets: feature.anchor_sets,
124            text_colors: feature
125                .text_colors
126                .into_iter()
127                .map(|(range, color)| (range[0]..range[1], color))
128                .collect(),
129            fallback: feature.fallback,
130            repeat_key: feature.repeat_key.map(std::sync::Arc::from),
131            line: feature.line.map(|line| crate::sdf::LineLabel {
132                polyline: line.polyline.into(),
133                anchor_distance: line.anchor_distance,
134                glyph_offsets: line.glyph_offsets,
135                image_sizes: line.image_sizes,
136                first_glyph_index: line.first_glyph_index,
137            }),
138        }
139    }
