18/// Characters wanted from each font stack.
19pub(super) type GlyphRequests = HashMap<String, BTreeSet<u32>>;
20
21/// Answers that no image exists and remembers every name asked about, so evaluating an
22/// expression against it walks each `image` it may reach and records that image's name,
23/// computed names and the fallbacks of a `coalesce` included.
24#[derive(Debug, Default)]
25struct NamesAsked(RefCell<Vec<String>>);
26
27impl ImageSet for NamesAsked {
28    fn contains_image(&self, name: &str) -> bool {
29        self.0.borrow_mut().push(name.to_owned());
30        false
31    }
32}
33
34/// Adds the characters of a feature's label to the fonts that draw them, as the label reads
35/// with `images` held.
36fn glyphs(
37    fonts: &mut GlyphRequests,
38    paint: &crate::style::layer::SymbolPaint,
39    (properties, zoom): (&crate::style::expression::FeatureProperties, f64),
40    images: &dyn ImageSet,
41) {
42    let Some(text) = paint.label_among(properties, zoom, Some(images)) else {
43        return;
44    };
45    fonts.entry(paint.font_stack()).or_default().extend(
46        text.chars()
47            .filter(|c| *c != crate::style::expression::FORMAT_IMAGE)
48            .map(|c| c as u32),
49    );
50    // A section with a font of its own needs that font's glyphs for its characters.
51    let mut characters = text.chars();
52    for section in paint.label_sections_among(properties, zoom, Some(images)) {
53        let own: Vec<char> = characters.by_ref().take(section.length).collect();
54        if let Some(font) = &section.font {
55            fonts
56                .entry(font.clone())
57                .or_default()
58                .extend(own.into_iter().map(|c| c as u32));
59        }
60    }
61}

63pub(super) fn requests<'l>(
64    layers: impl IntoIterator<Item = &'l StyleLayer>,
65    data: &[u8],
66    zoom: f64,
67) -> (GlyphRequests, HashSet<String>) {
68    let mut fonts = GlyphRequests::new();
69    let mut icons = HashSet::new();
70    let Ok(tile) = geozero::mvt::Tile::decode(data) else {
71        return (fonts, icons);
72    };
73    for layer in layers {
74        if layer.is_hidden() {
75            continue;
76        }
77        let Some(LayerPaint::Symbol(paint)) = &layer.paint else {
78            continue;
79        };
80        let Some(source) = tile
81            .layers
82            .iter()
83            .find(|source| Some(&source.name) == layer.source_layer.as_ref())
84        else {
85            continue;
86        };
87        let filter = match layer
88            .filter
89            .as_ref()
90            .map(crate::style::filter::Filter::parse)
91        {
92            Some(Ok(filter)) => Some(filter),
93            Some(Err(_)) => continue,
94            None => None,
95        };
96        for feature in &source.features {
97            let properties = feature_properties(source, feature);
98            if let Some(filter) = &filter {
99                if !filter.evaluate(&crate::style::filter::FeatureContext {
100                    properties: &properties,
101                    geometry_type: crate::style::filter::GeometryType::from_mvt(
102                        feature.r#type.unwrap_or_default(),
103                    ),
104                    id: feature
105                        .id
106                        .map(|id| crate::style::expression::Value::Number(id as f64)),
107                    zoom,
108                }) {
109                    continue;
110                }
111            }
112            // Every image the label or icon can reach is asked about, so each is requested.
113            let asked = NamesAsked::default();
114            glyphs(&mut fonts, paint, (&properties, zoom), &asked);
115            if let Some(icon) = paint
116                .text_among_images("icon-image", &properties, zoom, &asked)
117                .filter(|icon| !icon.is_empty())
118            {
119                icons.insert(icon);
120            }
121            icons.extend(
122                paint
123                    .label_sections_among(&properties, zoom, Some(&asked))
124                    .into_iter()
125                    .filter_map(|section| section.image),
126            );
127            icons.extend(asked.0.into_inner());
128        }
129    }
130    (fonts, icons)
131}
