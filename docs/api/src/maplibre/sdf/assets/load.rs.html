23/// A symbol asset could not be loaded, and a later attempt can succeed.
24#[derive(Debug, thiserror::Error)]
25#[error("symbol asset {url} is temporarily unavailable: {reason}")]
26pub struct SymbolAssetError {
27    /// The URL that failed.
28    pub url: String,
29    /// The transport or server error.
30    pub reason: String,
31}

33/// The style's symbol settings every source shares: where glyphs and sprites come from, and
34/// the images a host added. Which layers want them is a source's own business.
35#[derive(Clone, Copy, Debug)]
36pub struct SymbolAssetConfig<'a> {
37    /// The glyph URL template.
38    pub glyphs: Option<&'a str>,
39    /// The style's `sprite`, one URL or a list of `{id, url}`.
40    pub sprite: Option<&'a serde_json::Value>,
41    /// Images a host added to the style.
42    pub images: &'a HashMap<String, StyleImage>,
43    /// Device pixels per layout pixel of the display, which provided images are made for.
44    pub pixel_ratio: f32,
45}
46
47impl<'a> SymbolAssetConfig<'a> {
48    /// The symbol settings of `style`, for a display of one device pixel per layout pixel.
49    pub fn of(style: &'a Style) -> Self {
50        Self {
51            glyphs: style.glyphs.as_deref(),
52            sprite: style.sprite.as_ref(),
53            images: &style.images,
54            pixel_ratio: 1.0,
55        }
56    }
57}

59/// A tile's atlas, and the provided images its labels name.
60pub(crate) struct LoadedSymbolAssets {
61    /// The glyphs and images that were ready.
62    pub(crate) atlas: Arc<SymbolAtlas>,
63    /// Every provided name the labels ask for, whether it was ready or not.
64    pub(crate) provided: Vec<String>,
65    /// The provided names with no answer yet, which a later load can pack once they have one.
66    pub(crate) awaiting: Vec<String>,
67}

69/// Loads the glyph ranges and sprites that the features of one source's tile use under
70/// `layers` through the client's shared cache.
71///
72/// `layers` are the source's own layers as its tile is cut, so they are the ones its symbols
73/// are laid out with: a GeoJSON layer reads the source layer its tile is encoded under, and a
74/// layer of another source never matches this tile's layers, whatever their names.
75///
76/// Missing or malformed assets are logged once and left out of the atlas. A transport or server
77/// failure is returned so the caller can retry the tile instead of keeping an incomplete atlas.
78pub async fn load_symbol_assets<'l, HC: HttpClient>(
79    client: &SourceClient<HC>,
80    config: SymbolAssetConfig<'_>,
81    layers: impl IntoIterator<Item = &'l StyleLayer>,
82    data: &[u8],
83    zoom: f64,
84) -> Result<Arc<SymbolAtlas>, SymbolAssetError> {
85    Ok(
86        load_symbol_assets_awaiting(client, config, layers, data, zoom)
87            .await?
88            .atlas,
89    )
90}

92/// Like [`load_symbol_assets`], packing the provided images whose answers are known and
93/// listing the ones still to come instead of waiting for them.
94pub(crate) async fn load_symbol_assets_awaiting<'l, HC: HttpClient>(
95    client: &SourceClient<HC>,
96    config: SymbolAssetConfig<'_>,
97    layers: impl IntoIterator<Item = &'l StyleLayer>,
98    data: &[u8],
99    zoom: f64,
100) -> Result<LoadedSymbolAssets, SymbolAssetError> {
101    let (fonts, icons) = requests(layers, data, zoom);
102    let mut builder = AtlasBuilder::new();
103    for (font, characters) in fonts {
104        let ranges: BTreeSet<_> = characters.iter().map(|code| (code / 256) * 256).collect();
105        for range in ranges {
106            if let Some(glyphs) = glyph_range(client, config.glyphs, &font, range).await? {
107                builder.glyph_subset(&font, &glyphs, Some(&characters));
108            }
109        }
110    }
111    if !icons.is_empty() {
112        for (prefix, url) in sprite_sources(config.sprite) {
113            load_sprites(client, &mut builder, &prefix, &url, &icons).await?;
114        }
115    }
116    for (name, image) in config.images {
117        if icons.contains(name) {
118            pack_style_image(&mut builder, name, image, None);
119        }
120    }
121    // The sprite and the style's own images come first; a provider fills only the gaps.
122    let (provided, awaiting) = pack_provided(
123        &mut builder,
124        client.image_providers(),
125        &icons,
126        config.pixel_ratio,
127    );
128    Ok(LoadedSymbolAssets {
129        atlas: builder.finish(),
130        provided,
131        awaiting,
132    })
133}

135/// Packs the provided images among `icons` that nothing else supplied and whose answers are
136/// known, returning every provided name and the ones without an answer yet.
137fn pack_provided(
138    builder: &mut AtlasBuilder,
139    providers: &ImageProviders,
140    icons: &HashSet<String>,
141    pixel_ratio: f32,
142) -> (Vec<String>, Vec<String>) {
143    let mut provided: Vec<String> = icons
144        .iter()
145        .filter(|name| !builder.atlas.icons.contains_key(*name) && providers.provides(name))
146        .cloned()
147        .collect();
148    provided.sort();
149    let mut awaiting = Vec::new();
150    for name in &provided {
151        match providers.known(name, pixel_ratio).as_deref() {
152            Some(Resolved::Image(image)) => {
153                pack_style_image(builder, name, &image.image, image.anchor);
154            }
155            Some(Resolved::None) => {}
156            None => awaiting.push(name.clone()),
157        }
158    }
159    (provided, awaiting)
160}

162/// RGBA bytes with each colour channel scaled by its alpha.
163fn premultiply(data: &[u8]) -> Vec<u8> {
164    let mut out = data.to_vec();
165    for pixel in out.as_chunks_mut::<4>().0 {
166        let alpha = u32::from(pixel[3]);
167        for channel in &mut pixel[..3] {
168            *channel = ((u32::from(*channel) * alpha + 127) / 255) as u8;
169        }
170    }
171    out
172}

174/// Packs an image the host added to the style, replacing a sprite icon of the same name.
175/// `anchor`, in image pixels, is the point placed where the image's centre would be.
176fn pack_style_image(
177    builder: &mut AtlasBuilder,
178    name: &str,
179    image: &StyleImage,
180    anchor: Option<[f32; 2]>,
181) {
182    if image.width == 0
183        || image.height == 0
184        || image.data.len() != image.width as usize * image.height as usize * 4
185    {
186        return;
187    }
188    let premultiplied;
189    let data = if image.sdf {
190        &image.data
191    } else {
192        premultiplied = premultiply(&image.data);
193        &premultiplied
194    };
195    let Some(rect) = builder.pack(image.width, image.height, data) else {
196        return;
197    };
198    let [shift_x, shift_y] = anchor.map_or([0.0, 0.0], |[x, y]| {
199        [x - image.width as f32 / 2.0, y - image.height as f32 / 2.0]
200    });
201    builder.atlas.icons.insert(
202        name.to_owned(),
203        AtlasEntry {
204            rect,
205            metrics: [shift_x, shift_y, 0.0, image.pixel_ratio.max(0.01)],
206            kind: if image.sdf { 2 } else { 1 },
207            ..Default::default()
208        },
209    );
210}

212/// The range from the style's glyph server, or the bundled Latin range when that is all there is.
213async fn glyph_range<HC: HttpClient>(
214    client: &SourceClient<HC>,
215    template: Option<&str>,
216    font: &str,
217    range: u32,
218) -> Result<Option<Arc<Glyphs>>, SymbolAssetError> {
219    let bundled = || client.assets().bundled_glyphs(BUNDLED_LATIN);
220    let Some(template) = template else {
221        return Ok(if range == 0 {
222            bundled().await.ok()
223        } else {
224            None
225        });
226    };
227    let url = glyph_url(template, font, range);
228    match client.assets().glyphs(client, &url).await {
229        Ok(glyphs) => Ok(Some(glyphs)),
230        Err(AssetFailure::Retryable(reason)) => Err(SymbolAssetError { url, reason }),
231        Err(_) if range == 0 => Ok(bundled().await.ok()),
232        Err(_) => Ok(None),
233    }
234}

236fn glyph_url(template: &str, font: &str, range: u32) -> String {
237    let encoded: String = font
238        .bytes()
239        .map(|byte| {
240            if byte.is_ascii_alphanumeric() || b"-_.~".contains(&byte) {
241                char::from(byte).to_string()
242            } else {
243                format!("%{byte:02X}")
244            }
245        })
246        .collect();
247    template
248        .replace("{fontstack}", &encoded.replace('+', "%20"))
249        .replace("{range}", &format!("{range}-{}", range + 255))
250}

252pub(super) fn sprite_sources(sprite: Option<&serde_json::Value>) -> Vec<(String, String)> {
253    match sprite {
254        Some(serde_json::Value::String(url)) => vec![(String::new(), url.clone())],
255        Some(serde_json::Value::Array(sources)) => sources
256            .iter()
257            .filter_map(|source| {
258                let id = source.get("id")?.as_str()?;
259                // The sprite named `default` supplies the plain image names.
260                let prefix = if id == "default" {
261                    String::new()
262                } else {
263                    format!("{id}:")
264                };
265                Some((prefix, source.get("url")?.as_str()?.to_string()))
266            })
267            .collect(),
268        _ => Vec::new(),
269    }
270}

272async fn load_sprites<HC: HttpClient>(
273    client: &SourceClient<HC>,
274    builder: &mut AtlasBuilder,
275    prefix: &str,
276    url: &str,
277    wanted: &HashSet<String>,
278) -> Result<(), SymbolAssetError> {
279    let json_url = sprite_url(url, "json");
280    let png_url = sprite_url(url, "png");
281    match client.assets().sprites(client, &json_url, &png_url).await {
282        Ok(sheet) => pack_sprites(builder, prefix, wanted, &sheet),
283        Err(AssetFailure::Retryable(reason)) => {
284            return Err(SymbolAssetError {
285                url: json_url,
286                reason,
287            })
288        }
289        Err(_) => {}
290    }
291    Ok(())
292}

294fn pack_sprites(
295    builder: &mut AtlasBuilder,
296    prefix: &str,
297    wanted: &HashSet<String>,
298    sheet: &SpriteSheet,
299) {
300    let image = &sheet.image;
301    for (name, value) in &sheet.document {
302        let name = format!("{prefix}{name}");
303        if !wanted.contains(&name) {
304            continue;
305        }
306        let read = |key: &str| {
307            value
308                .get(key)
309                .and_then(|value| value.as_u64())
310                .and_then(|value| u32::try_from(value).ok())
311        };
312        let (Some(x), Some(y), Some(width), Some(height)) =
313            (read("x"), read("y"), read("width"), read("height"))
314        else {
315            continue;
316        };
317        if width == 0
318            || height == 0
319            || x.saturating_add(width) > image.width()
320            || y.saturating_add(height) > image.height()
321        {
322            continue;
323        }
324        let pixels = image::imageops::crop_imm(image, x, y, width, height).to_image();
325        let sdf = value.get("sdf").and_then(|v| v.as_bool()).unwrap_or(false);
326        // Colour icons are stored premultiplied, so filtering them never darkens an edge with
327        // the colour of the clear texels around it.
328        let packed = if sdf {
329            pixels.into_raw()
330        } else {
331            premultiply(pixels.as_raw())
332        };
333        let Some(rect) = builder.pack(width, height, &packed) else {
334            continue;
335        };
336        let ratio = value
337            .get("pixelRatio")
338            .and_then(|value| value.as_f64())
339            .unwrap_or(1.0)
340            .max(0.01) as f32;
341        builder.atlas.icons.insert(
342            name,
343            AtlasEntry {
344                rect,
345                metrics: [0.0, 0.0, 0.0, ratio],
346                kind: if value.get("sdf").and_then(|v| v.as_bool()).unwrap_or(false) {
347                    2
348                } else {
349                    1
350                },
351                stretch: icon_stretch(value),
352            },
353        );
354    }
355}

357fn ranges(value: &serde_json::Value, key: &str) -> Vec<[f32; 2]> {
358    value
359        .get(key)
360        .and_then(|ranges| ranges.as_array())
361        .into_iter()
362        .flatten()
363        .filter_map(|range| {
364            let range = range.as_array()?;
365            Some([
366                range.first()?.as_f64()? as f32,
367                range.get(1)?.as_f64()? as f32,
368            ])
369        })
370        .collect()
371}

373fn text_fit(value: &serde_json::Value, key: &str) -> Option<TextFit> {
374    match value.get(key)?.as_str()? {
375        "stretchOrShrink" => Some(TextFit::StretchOrShrink),
376        "stretchOnly" => Some(TextFit::StretchOnly),
377        "proportional" => Some(TextFit::Proportional),
378        _ => None,
379    }
380}

382/// The stretch fields of a sprite entry, or `None` when it has none.
383fn icon_stretch(value: &serde_json::Value) -> Option<Box<IconStretch>> {
384    let stretch = IconStretch {
385        stretch_x: ranges(value, "stretchX"),
386        stretch_y: ranges(value, "stretchY"),
387        content: value.get("content").and_then(|content| {
388            let content = content.as_array()?;
389            let read = |index: usize| content.get(index)?.as_f64().map(|edge| edge as f32);
390            Some([read(0)?, read(1)?, read(2)?, read(3)?])
391        }),
392        text_fit_width: text_fit(value, "textFitWidth"),
393        text_fit_height: text_fit(value, "textFitHeight"),
394    };
395    (stretch != IconStretch::default()).then(|| Box::new(stretch))
396}

398fn sprite_url(url: &str, extension: &str) -> String {
399    let (path, query) = url
400        .split_once('?')
401        .map_or((url, None), |(path, query)| (path, Some(query)));
402    format!(
403        "{path}.{extension}{}",
404        query.map_or_else(String::new, |query| format!("?{query}"))
405    )
406}
