1//! Images made on request for the names a style asks for, such as road shields drawn from
2//! a route's network and number, so that no host has to list every image in advance.
3//!
4//! A host registers a [`StyleImageProvider`] under a namespace. A name of the form
5//! `namespace:id` that neither the sprite nor the style's own images supply is then asked of
6//! that provider, from the tile worker that lays the label out, never from the render thread.
7//! The label is drawn at once with whatever the style falls back to and laid out again when
8//! the image arrives. The style itself only names images, so it still serializes; the
9//! provider lives in the runtime, beside the worker's other services.
10

28/// What a provider is asked for.
29#[derive(Clone, Debug, PartialEq)]
30pub struct ImageRequest {
31    /// The whole name the style asked for, namespace included.
32    pub name: String,
33    /// The part of the name after the namespace and its colon.
34    pub id: String,
35    /// Device pixels per layout pixel of the display the image is drawn on.
36    pub pixel_ratio: f32,
37}
38
39/// A generated image and where it sits on its anchor.
40#[derive(Clone, Debug, PartialEq)]
41pub struct ProvidedImage {
42    /// Straight-alpha RGBA pixels; `pixel_ratio` says how many of them cover a layout pixel.
43    /// A colour image is never an SDF, so `sdf` stays false for it.
44    pub image: StyleImage,
45    /// The point, in image pixels from the top-left corner, that `icon-anchor` places as the
46    /// image's centre; `None` uses the centre. A shield with a banner above it names the
47    /// centre of the shield body here, so the body, not the whole picture, sits on the road.
48    pub anchor: Option<[f32; 2]>,
49}
50
51/// A provider's answer for one name.
52#[derive(Clone, Debug, PartialEq)]
53pub enum ImageResolution {
54    /// The image to draw.
55    Image(ProvidedImage),
56    /// The name is understood but has no image, such as a route the rules draw no shield
57    /// for. The style's fallback is drawn instead; the answer is kept like an image.
58    Absent,
59}
60
61/// Why a provider has no answer.
62#[derive(Clone, Debug, PartialEq, Eq, thiserror::Error)]
63pub enum ImageProviderError {
64    /// A later request may succeed, such as when a resource is still downloading. The name
65    /// is asked again when its tile is next requested.
66    #[error("temporarily unavailable: {0}")]
67    Unavailable(String),
68    /// The name cannot be drawn, such as a malformed request. The failure is kept like an
69    /// answer until the namespace is invalidated.
70    #[error("cannot be provided: {0}")]
71    Failed(String),
72}

74/// A provider's answer to come. Native workers run it on any thread, so with
75/// `thread-safe-futures` it must be `Send`; a `Send` future fits either way.
76#[cfg(feature = "thread-safe-futures")]
77pub type ProvideFuture<'a> =
78    Pin<Box<dyn Future<Output = Result<ImageResolution, ImageProviderError>> + Send + 'a>>;
79/// A provider's answer to come.
80#[cfg(not(feature = "thread-safe-futures"))]
81pub type ProvideFuture<'a> =
82    Pin<Box<dyn Future<Output = Result<ImageResolution, ImageProviderError>> + 'a>>;

84/// Makes images for the names of one namespace.
85///
86/// Calls run on tile workers, concurrently, and may take as long as generation takes; the
87/// registry runs each distinct request once at a time and bounds how many run together.
88pub trait StyleImageProvider: Send + Sync + 'static {
89    /// Identifies everything the images depend on besides the request, such as the hash of a
90    /// rule pack. Answers made under another generation are not reused.
91    fn generation(&self) -> String;
92
93    /// Makes the image for `request`.
94    fn provide(&self, request: ImageRequest) -> ProvideFuture<'_>;
95}
96
97/// How much generation the registry allows.
98#[derive(Clone, Copy, Debug, PartialEq, Eq)]
99pub struct ProviderLimits {
100    /// Generators running at once across every worker.
101    pub max_running: usize,
102    /// Requests waiting for a generator; more are refused as temporarily unavailable.
103    pub max_waiting: usize,
104    /// Bytes of answers kept, least recently used first out.
105    pub cache_bytes: usize,
106}
107
108impl Default for ProviderLimits {
109    fn default() -> Self {
110        Self {
111            max_running: 4,
112            max_waiting: 1024,
113            cache_bytes: 32 * 1024 * 1024,
114        }
115    }
116}

118/// Counts of what the registry did, for diagnostics and tests.
119#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
120pub struct ImageProviderStats {
121    /// Lookups of a provided name by a tile.
122    pub requested: u64,
123    /// Lookups answered from the cache without waiting.
124    pub cache_hits: u64,
125    /// Provider calls made.
126    pub calls: u64,
127    /// Calls that returned an image.
128    pub images: u64,
129    /// Calls that returned [`ImageResolution::Absent`].
130    pub absent: u64,
131    /// Calls that failed for good.
132    pub failed: u64,
133    /// Calls that failed for now.
134    pub unavailable: u64,
135    /// Requests refused because the waiting queue was full.
136    pub refused: u64,
137    /// Calls dropped unfinished, because every tile waiting for them went away.
138    pub cancelled: u64,
139    /// Sources of tiles whose labels were laid out again because images arrived.
140    pub relaid: u64,
141    /// Generators running now.
142    pub running: usize,
143    /// Requests waiting for a generator now.
144    pub waiting: usize,
145    /// Bytes of answers held.
146    pub cached_bytes: usize,
147}

198impl fmt::Debug for ImageProviders {
199    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
200        let namespaces: Vec<String> = self
201            .namespaces()
202            .iter()
203            .map(|namespace| namespace.prefix.clone())
204            .collect();
205        formatter
206            .debug_struct("ImageProviders")
207            .field("namespaces", &namespaces)
208            .field("limits", &self.limits())
209            .finish_non_exhaustive()
210    }
211}

219/// A provider call in progress; dropping it unfinished counts as a cancellation.
220struct Call<'a> {
221    counters: &'a Counters,
222    finished: bool,
223}
224
225impl Drop for Call<'_> {
226    fn drop(&mut self) {
227        if !self.finished {
228            Counters::add(&self.counters.cancelled);
229        }
230    }
231}

233impl ImageProviders {
234    /// A registry without providers that generates within `limits`.
235    pub fn with_limits(limits: ProviderLimits) -> Self {
236        Self(Arc::new(Registry {
237            namespaces: Mutex::default(),
238            epochs: AtomicU64::new(0),
239            limits: Mutex::new(limits),
240            cache: AssetCache::with_budget(limits.cache_bytes),
241            gate: Arc::default(),
242            counters: Counters::default(),
243            missing: Mutex::default(),
244        }))

247    /// Records a requested icon or pattern identifier that neither the sprite, the style's
248    /// images, nor any registered provider supplied.
249    pub(crate) fn record_missing(&self, name: &str) {
250        if !name.is_empty() {
251            self.0
252                .missing
253                .lock()
254                .unwrap_or_else(PoisonError::into_inner)
255                .insert(name.to_owned());
256        }
257    }
258
259    /// Drains missing image identifiers recorded since the last call.
260    pub(crate) fn take_missing(&self) -> Vec<String> {
261        let mut guard = self
262            .0
263            .missing
264            .lock()
265            .unwrap_or_else(PoisonError::into_inner);
266        std::mem::take(&mut *guard).into_iter().collect()
267    }
268
269    fn namespaces(&self) -> MutexGuard<'_, Vec<Namespace>> {
270        self.0
271            .namespaces
272            .lock()
273            .unwrap_or_else(PoisonError::into_inner)
274    }
275
276    fn limits(&self) -> ProviderLimits {
277        *self.0.limits.lock().unwrap_or_else(PoisonError::into_inner)
278    }
279
280    /// Changes how many generators run and wait from now on; the cache keeps the budget it
281    /// was made with.
282    pub fn set_limits(&self, limits: ProviderLimits) {
283        *self.0.limits.lock().unwrap_or_else(PoisonError::into_inner) = limits;
284    }

286    /// Counts a source whose labels were laid out again with arrived images.
287    pub(crate) fn count_relaid(&self) {
288        Counters::add(&self.0.counters.relaid);
289    }

291    fn next_epoch(&self) -> u64 {
292        self.0
293            .epochs
294            .fetch_add(1, Ordering::Relaxed)
295            .wrapping_add(1)
296    }

298    /// Asks `provider` for the names that start with `namespace` and a colon, replacing any
299    /// provider registered for it before.
300    pub fn register(&self, namespace: &str, provider: Arc<dyn StyleImageProvider>) {
301        let prefix = format!("{namespace}:");
302        let epoch = self.next_epoch();
303        let mut namespaces = self.namespaces();
304        namespaces.retain(|existing| existing.prefix != prefix);
305        namespaces.push(Namespace {
306            prefix,
307            provider,
308            epoch,
309        });
310    }
311
312    /// Stops asking for the names of `namespace`; returns whether a provider served it.
313    pub fn unregister(&self, namespace: &str) -> bool {
314        let prefix = format!("{namespace}:");
315        let mut namespaces = self.namespaces();
316        let before = namespaces.len();
317        namespaces.retain(|existing| existing.prefix != prefix);
318        namespaces.len() != before
319    }
320
321    /// Forgets every answer of `namespace`, so that the names are asked again; returns whether
322    /// a provider serves it. A map re-lays the tiles that drew them (see
323    /// `HeadlessMap::invalidate_provided_images`).
324    pub fn invalidate(&self, namespace: &str) -> bool {
325        let prefix = format!("{namespace}:");
326        let epoch = self.next_epoch();
327        let mut namespaces = self.namespaces();
328        let Some(found) = namespaces
329            .iter_mut()
330            .find(|existing| existing.prefix == prefix)
331        else {
332            return false;
333        };
334        found.epoch = epoch;
335        true
336    }

338    /// What the registry has done so far.
339    pub fn stats(&self) -> ImageProviderStats {
340        let counters = &self.0.counters;
341        let read = |counter: &AtomicU64| counter.load(Ordering::Relaxed);
342        let (running, waiting) = self.0.gate.load();
343        ImageProviderStats {
344            requested: read(&counters.requested),
345            cache_hits: read(&counters.cache_hits),
346            calls: read(&counters.calls),
347            images: read(&counters.images),
348            absent: read(&counters.absent),
349            failed: read(&counters.failed),
350            unavailable: read(&counters.unavailable),
351            refused: read(&counters.refused),
352            cancelled: read(&counters.cancelled),
353            relaid: read(&counters.relaid),
354            running,
355            waiting,
356            cached_bytes: self.0.cache.approximate_bytes(),
357        }
358    }

361    pub fn provides(&self, name: &str) -> bool {

364            .any(|namespace| name.starts_with(&namespace.prefix))

367    /// The provider for `name`, its request and the cache key of its answer.
368    fn route(
369        &self,
370        name: &str,
371        pixel_ratio: f32,
372    ) -> Option<(Arc<dyn StyleImageProvider>, ImageRequest, String)> {
373        let (provider, prefix, epoch) = self
374            .namespaces()
375            .iter()
376            .find(|namespace| name.starts_with(&namespace.prefix))
377            .map(|namespace| {
378                (
379                    namespace.provider.clone(),
380                    namespace.prefix.len(),
381                    namespace.epoch,
382                )
383            })?;
384        let key = format!("{epoch}|{}|{pixel_ratio:.3}|{name}", provider.generation());
385        let request = ImageRequest {
386            name: name.to_owned(),
387            id: name[prefix..].to_owned(),
388            pixel_ratio,
389        };
390        Some((provider, request, key))
391    }

393    /// The answer for `name` if one is already known, without asking for it.
394    pub(crate) fn known(&self, name: &str, pixel_ratio: f32) -> Option<Arc<Resolved>> {
395        let (_, _, key) = self.route(name, pixel_ratio)?;
396        Counters::add(&self.0.counters.requested);
397        let known = self.0.cache.peek::<Resolved>(&key);
398        if known.is_some() {
399            Counters::add(&self.0.counters.cache_hits);
400        }
401        known
402    }

404    /// The answer for `name`, asking its provider once however many tiles wait for it.
405    /// `Err` carries why it is unavailable for now.
406    pub(crate) async fn resolve(
407        &self,
408        name: &str,
409        pixel_ratio: f32,
410    ) -> Result<Arc<Resolved>, String> {
411        let Some((provider, request, key)) = self.route(name, pixel_ratio) else {
412            return Ok(Arc::new(Resolved::None));
413        };
414        let registry = &self.0;
415        let limits = self.limits();
416        let answer = registry
417            .cache
418            .load(key, || async move {
419                let Ok(_permit) =
420                    Gate::enter(&registry.gate, limits.max_running, limits.max_waiting).await
421                else {
422                    Counters::add(&registry.counters.refused);
423                    return Err(AssetFailure::Retryable(
424                        "the image generation queue is full".to_owned(),
425                    ));
426                };
427                Counters::add(&registry.counters.calls);
428                let mut call = Call {
429                    counters: &registry.counters,
430                    finished: false,
431                };
432                let name = request.name.clone();
433                let answer = provider.provide(request).await;
434                call.finished = true;
435                Self::keep(&registry.counters, &name, answer)
436            })
437            .await;
438        answer.map_err(|failure| match failure {
439            AssetFailure::Retryable(reason) | AssetFailure::Terminal(reason) => reason,
440            AssetFailure::NotFound => "not found".to_owned(),
441        })
442    }

444    /// What to keep of a provider's answer, and its size.
445    fn keep(
446        counters: &Counters,
447        name: &str,
448        answer: Result<ImageResolution, ImageProviderError>,
449    ) -> Result<(Resolved, usize), AssetFailure> {
450        // A remembered absence costs a key and a little bookkeeping.
451        const ANSWER_BYTES: usize = 64;
452        match answer {
453            Ok(ImageResolution::Image(provided)) => {
454                let image = &provided.image;
455                let expected = image.width as usize * image.height as usize * 4;
456                if image.width == 0
457                    || image.height == 0
458                    || image.data.len() != expected
459                    || !(image.pixel_ratio.is_finite() && image.pixel_ratio > 0.0)
460                {
461                    Counters::add(&counters.failed);
462                    tracing::warn!(%name, width = image.width, height = image.height, bytes = image.data.len(), "provided image is malformed");
463                    return Ok((Resolved::None, ANSWER_BYTES));
464                }
465                Counters::add(&counters.images);
466                let bytes = image.data.len() + name.len() + ANSWER_BYTES;
467                Ok((Resolved::Image(provided), bytes))
468            }
469            Ok(ImageResolution::Absent) => {
470                Counters::add(&counters.absent);
471                Ok((Resolved::None, name.len() + ANSWER_BYTES))
472            }
473            Err(ImageProviderError::Failed(reason)) => {
474                Counters::add(&counters.failed);
475                tracing::warn!(%name, %reason, "image provider failed");
476                Ok((Resolved::None, name.len() + ANSWER_BYTES))
477            }
478            Err(ImageProviderError::Unavailable(reason)) => {
479                Counters::add(&counters.unavailable);
480                tracing::warn!(%name, %reason, "provided image unavailable for now");
481                Err(AssetFailure::Retryable(reason))
482            }
483        }
484    }
