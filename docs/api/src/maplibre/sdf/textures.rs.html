40#[derive(Default)]
41pub(super) struct SymbolTextures {
42    atlases: rustc_hash::FxHashMap<usize, Weak<TileAtlas>>,
43    tile_atlases: rustc_hash::FxHashMap<(WorldTileCoords, usize), Arc<TileAtlas>>,
44    bindings: rustc_hash::FxHashMap<(WorldTileCoords, crate::style::StyleLayerId), DrawBinding>,
45    /// Tiles whose labels gained an atlas since the last drain.
46    bound: Vec<WorldTileCoords>,
47}
48
49impl SymbolTextures {
50    pub fn prepare(
51        &mut self,
52        gpu: &TextureContext<'_>,
53        key: (WorldTileCoords, crate::style::StyleLayerId),
54        atlas: &Arc<SymbolAtlas>,
55        paint: &SymbolPaint,
56        zoom: f64,
57    ) {
58        let uniforms = SymbolUniforms::new(paint, zoom, atlas.size);
59        let separate_halo = uniforms.text[1] > 1.0
60            || uniforms.icon[1] > 1.0
61            || (uniforms.text[1] > 0.0
62                && paint.number("text-letter-spacing", &Default::default(), zoom, 0.0) < 0.0);
63        if let Some(binding) = self.bindings.get_mut(&key) {
64            if binding.atlas.source.same_gpu_atlas(atlas) {
65                binding.separate_halo = separate_halo;
66                if binding.uniforms != uniforms {
67                    gpu.queue
68                        .write_buffer(&binding.buffer, 0, bytemuck::bytes_of(&uniforms));
69                    binding.uniforms = uniforms;
70                }
71                return;
72            }
73        }
74        // Source atlases can share tile coordinates without sharing sprite or font pixels.
75        let identity = atlas.gpu_identity();
76        let texture = self
77            .tile_atlases
78            .get(&(key.0, identity))
79            .filter(|cached| cached.source.same_gpu_atlas(atlas))
80            .cloned()
81            .or_else(|| {
82                self.atlases
83                    .get(&identity)
84                    .and_then(Weak::upgrade)
85                    .filter(|cached| cached.source.same_gpu_atlas(atlas))
86            })
87            .unwrap_or_else(|| {
88                let texture = share_gpu(TileAtlas::new(gpu, atlas));
89                self.atlases.insert(identity, Arc::downgrade(&texture));
90                texture
91            });
92        self.tile_atlases
93            .insert((key.0, identity), Arc::clone(&texture));
94        if !self.bound.contains(&key.0) {
95            self.bound.push(key.0);
96        }
97        if let Some(old) = self
98            .bindings
99            .insert(key, DrawBinding::new(gpu, texture, uniforms, separate_halo))
100        {
101            gpu.queue.cancel_staged_buffer(&old.buffer);
102        }
103    }
104
105    /// Tiles whose glyphs and icons became drawable since the last call.
106    pub fn take_bound(&mut self) -> Vec<WorldTileCoords> {
107        std::mem::take(&mut self.bound)
108    }
109
110    #[cfg(all(test, feature = "headless"))]
111    pub fn binding(&self, coords: WorldTileCoords, layer: &str) -> Option<&DrawBinding> {
112        self.binding_by_id(coords, crate::style::StyleLayerId::intern(layer))
113    }
114
115    #[inline]
116    pub fn binding_by_id(
117        &self,
118        coords: WorldTileCoords,
119        layer: crate::style::StyleLayerId,
120    ) -> Option<&DrawBinding> {
121        self.bindings.get(&(coords, layer))
122    }
123
124    pub fn retain(&mut self, tiles: &crate::tcs::tiles::Tiles) {
125        self.bindings.retain(|(coords, _), _| {
126            tiles
127                .query::<&super::SymbolLayersDataComponent>(*coords)
128                .is_some()
129        });
130        self.tile_atlases.retain(|(coords, _), _| {
131            tiles
132                .query::<&super::SymbolLayersDataComponent>(*coords)
133                .is_some()
134        });
135        self.atlases.retain(|_, atlas| atlas.strong_count() > 0);
136    }

139impl TileAtlas {
140    fn new(gpu: &TextureContext<'_>, atlas: &Arc<SymbolAtlas>) -> Self {
141        let pixel_bytes = atlas.take_pixels_for_gpu();
142        let expected_len = (atlas.size[0] as usize)
143            .saturating_mul(atlas.size[1] as usize)
144            .saturating_mul(4);
145        let fallback_zeroes;
146        let upload_slice: &[u8] = if pixel_bytes.len() == expected_len {
147            &pixel_bytes
148        } else {
149            fallback_zeroes = vec![0_u8; expected_len];
150            &fallback_zeroes
151        };
152        let texture = gpu.queue.create_texture_with_data(
153            gpu.device,
154            &wgpu::TextureDescriptor {
155                label: Some("symbol atlas"),
156                size: wgpu::Extent3d {
157                    width: atlas.size[0],
158                    height: atlas.size[1],
159                    depth_or_array_layers: 1,
160                },
161                mip_level_count: 1,
162                sample_count: 1,
163                dimension: wgpu::TextureDimension::D2,
164                format: wgpu::TextureFormat::Rgba8Unorm,
165                usage: wgpu::TextureUsages::TEXTURE_BINDING | wgpu::TextureUsages::COPY_DST,
166                view_formats: &[],
167            },
168            TextureDataOrder::LayerMajor,
169            upload_slice,
170        );
171        Self {
172            source: atlas.clone(),
173            texture,
174        }
175    }
176}

178impl DrawBinding {
179    fn new(
180        gpu: &TextureContext<'_>,
181        atlas: Arc<TileAtlas>,
182        uniforms: SymbolUniforms,
183        separate_halo: bool,
184    ) -> Self {
185        let buffer = gpu.queue.create_buffer_init(
186            gpu.device,
187            &wgpu::util::BufferInitDescriptor {
188                label: Some("symbol paint"),
189                contents: bytemuck::bytes_of(&uniforms),
190                usage: wgpu::BufferUsages::UNIFORM | wgpu::BufferUsages::COPY_DST,
191            },
192        );
193        let sampler = gpu.device.create_sampler(&wgpu::SamplerDescriptor {
194            mag_filter: wgpu::FilterMode::Linear,
195            min_filter: wgpu::FilterMode::Linear,
196            ..Default::default()
197        });
198        let view = atlas.texture.create_view(&Default::default());
199        let group = gpu.device.create_bind_group(&wgpu::BindGroupDescriptor {
200            label: Some("symbol atlas and paint"),
201            layout: &gpu.pipeline.get_bind_group_layout(1),
202            entries: &[
203                wgpu::BindGroupEntry {
204                    binding: 0,
205                    resource: wgpu::BindingResource::TextureView(&view),
206                },
207                wgpu::BindGroupEntry {
208                    binding: 1,
209                    resource: wgpu::BindingResource::Sampler(&sampler),
210                },
211                wgpu::BindGroupEntry {
212                    binding: 2,
213                    resource: buffer.as_entire_binding(),
214                },
215            ],
216        });
217        Self {
218            atlas,
219            group,
220            separate_halo,
221            buffer,
222            uniforms,
223        }
224    }
