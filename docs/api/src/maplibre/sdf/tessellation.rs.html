38/// Worker-side geometry for screen-sized symbols anchored to map features.
39pub struct TextTessellator {
40    geo_writer: GeoWriter,
41    paint: SymbolPaint,
42    zoom: f64,
43    atlas: Arc<SymbolAtlas>,
44    collected: Vec<CollectedSymbol>,
45    /// Lines to be labelled along, held until they can be merged.
46    pending_lines: Vec<line_merge::PendingLine>,
47    /// Where each text already has a label along a line, so a repeat of it keeps its distance.
48    line_anchors_by_text: std::collections::HashMap<String, Vec<[f64; 2]>>,
49    properties: FeatureProperties,
50    /// Symbol triangles.
51    pub quad_buffer: VertexBuffers<ShaderSymbolVertex, IndexDataType>,
52    /// Collision bounds and exact triangle ranges per symbol.
53    pub features: Vec<Feature>,
54    /// Source layer extent conversion.
55    pub coordinate_scale: f64,
56    /// How many times the tile is magnified past the zoom its source stops at; zero or one when
57    /// it is not.
58    pub overscaling: f64,
59    /// Source IDs indexed by the feature order supplied to the geometry processor.
60    pub source_ids: Vec<Option<u64>>,
61}

63impl TextTessellator {
64    /// A text collector with the offline fallback font.
65    pub fn new(text_field: StyleProperty<FormattedText>, zoom: f64) -> Self {
66        Self {
67            paint: SymbolPaint {
68                text_field: Some(text_field),
69                ..Default::default()
70            },
71            zoom,
72            ..Default::default()
73        }
74    }
75
76    /// Collects features using an existing atlas without decoding fallback glyphs again.
77    pub fn with_assets(paint: SymbolPaint, zoom: f64, atlas: Arc<SymbolAtlas>) -> Self {
78        Self {
79            geo_writer: GeoWriter::default(),
80            paint,
81            zoom,
82            atlas,
83            collected: Vec::new(),
84            pending_lines: Vec::new(),
85            line_anchors_by_text: std::collections::HashMap::new(),
86            properties: FeatureProperties::new(),
87            quad_buffer: VertexBuffers::new(),
88            features: Vec::new(),
89            coordinate_scale: 1.0,
90            overscaling: 1.0,
91            source_ids: Vec::new(),
92        }
93    }
94
95    /// Sets the style and the assets used by this tile's symbols.
96    pub fn configure(&mut self, paint: SymbolPaint, atlas: Arc<SymbolAtlas>) {
97        self.paint = paint;
98        self.atlas = atlas;
99    }
100
101    /// Builds quads and collision ranges from the collected map features.
102    pub fn finish(&mut self) {
103        let pending = std::mem::take(&mut self.pending_lines);
104        for line in line_merge::merge_lines(pending) {
105            self.properties = line.properties;
106            self.collect_along_lines(vec![line.line], LinePlacement::Line, line.source);
107        }
108        self.properties.clear();
109        self.collected.sort_by(|a, b| {
110            let key = |symbol: &CollectedSymbol| {
111                self.paint
112                    .number("symbol-sort-key", &symbol.properties, self.zoom, 0.0)
113            };
114            key(a).total_cmp(&key(b))
115        });
116        for symbol in &self.collected {
117            layout::append(
118                symbol,
119                &self.paint,
120                self.zoom,
121                &self.atlas,
122                &mut self.quad_buffer,
123                &mut self.features,
124            );
125        }
126        if draw_order::sorts_by_height(&self.paint, self.zoom) {
127            draw_order::reorder(&mut self.quad_buffer, &mut self.features);
128        }
129    }
130}

190/// How a symbol follows a line, from `symbol-placement`.
191#[derive(Clone, Copy, PartialEq, Eq)]
192enum LinePlacement {
193    /// Repeated along the line.
194    Line,
195    /// Once, at the middle of the line.
196    Center,
197}
198
199fn line_placement(paint: &SymbolPaint) -> Option<LinePlacement> {
200    match paint
201        .properties
202        .get("symbol-placement")
203        .and_then(|value| value.as_str())
204    {
205        Some("line") => Some(LinePlacement::Line),
206        Some("line-center") => Some(LinePlacement::Center),
207        _ => None,
208    }
209}

211/// Whether the layer places its text along lines.
212pub(crate) fn is_line_placed(paint: &SymbolPaint) -> bool {
213    line_placement(paint).is_some()
214}

216/// Tile units in one pixel of a tile drawn at its own zoom: 4096 units span 512 pixels.
217const TILE_UNITS_PER_PIXEL: f64 = 8.0;
218
219impl TextTessellator {
220    fn collect(
221        &mut self,
222        anchor: Point<f64>,
223        angle: f32,
224        source: layout::SourceFeature,
225        (line, repeat_key): (Option<layout::LineContext>, Option<&Arc<str>>),
226    ) {
227        let ways = line
228            .is_none()
229            .then(|| {
230                self.paint
231                    .label_among(&self.properties, self.zoom, Some(&self.atlas.icons))
232            })
233            .flatten()
234            .and_then(|text| text_layout::both_orientations(&self.paint, &text));
235        // A text that may be written both ways is laid out twice; the second shows when the
236        // first has no place.
237        for (index, vertical) in ways
238            .map_or_else(
239                || vec![None],
240                |[first, second]| vec![Some(first), Some(second)],
241            )
242            .into_iter()
243            .enumerate()
244        {
245            self.collected.push(CollectedSymbol {
246                source,
247                line: line.clone(),
248                anchor,
249                angle,
250                properties: self.properties.clone(),
251                vertical,
252                fallback: index == 1,
253                repeat_key: repeat_key.cloned(),
254            });
255        }
256    }
257
258    /// Whether a label with the same repeat key already sits within `repeat_distance` of `point`; a
259    /// label that does not is remembered.
260    fn anchor_is_too_close(
261        &mut self,
262        key: Option<&str>,
263        point: [f64; 2],
264        repeat_distance: f64,
265    ) -> bool {
266        let Some(key) = key else {
267            return false;
268        };
269        let others = self
270            .line_anchors_by_text
271            .entry(key.to_string())
272            .or_default();
273        let too_close = others
274            .iter()
275            .any(|other| (other[0] - point[0]).hypot(other[1] - point[1]) < repeat_distance);
276        if !too_close {
277            others.push(point);
278        }
279        too_close
280    }
281
282    /// Anchors along the lines of a feature, as many as the spacing allows.
283    fn collect_along_lines(
284        &mut self,
285        lines: Vec<Vec<[f64; 2]>>,
286        placement: LinePlacement,
287        source: layout::SourceFeature,
288    ) {
289        let probe = CollectedSymbol {
290            source,
291            line: None,
292            anchor: Point::new(0.0, 0.0),
293            angle: 0.0,
294            properties: self.properties.clone(),
295            vertical: None,
296            fallback: false,
297            repeat_key: None,
298        };
299        let text_size = self
300            .paint
301            .text_size
302            .as_ref()
303            // GL JS lays a label out at the size of the zoom above the tile's, which the shader then scales down.
304            .and_then(|value| value.evaluate_at_zoom(self.zoom + 1.0))
305            .map_or(16.0, f64::from);
306        let text_width = f64::from(text_layout::unwrapped_width(
307            &self.paint,
308            &probe,
309            self.zoom,
310            &self.atlas,
311        ));
312        // GL JS spaces labels by the longer of the text and the icon, and scales both by the
313        // text size; an icon-only label has no text to bend, so its bends are not checked.
314        let icon_width = self
315            .paint
316            .text_among_images("icon-image", &self.properties, self.zoom, &self.atlas.icons)
317            .and_then(|name| self.atlas.icons.get(&name))
318            .map_or(0.0, |icon| {
319                f64::from(icon.rect[2]) / f64::from(icon.metrics[3])
320                    * f64::from(
321                        self.paint
322                            .number("icon-size", &self.properties, self.zoom, 1.0),
323                    )
324            });
325        let label_pixels = text_width.max(icon_width) * text_size / 24.0;
326        let number = |name, fallback| {
327            self.paint
328                .number(name, &self.properties, self.zoom, fallback)
329        };
330        let overscaling = self.overscaling.max(1.0);
331        // A magnified tile has fewer tile units under a pixel.
332        let units_per_pixel = TILE_UNITS_PER_PIXEL / overscaling;
333        let params = line_anchors::AnchorSpacing {
334            spacing: f64::from(number("symbol-spacing", 250.0)) * units_per_pixel,
335            max_angle: f64::from(number("text-max-angle", 45.0)).to_radians(),
336            label_length: label_pixels * units_per_pixel,
337            text_size: text_size * units_per_pixel,
338            checks_bends: text_width > 0.0,
339            overscaling,
340        };
341        let parts: Vec<Vec<[f64; 2]>> = match placement {
342            LinePlacement::Line => line_anchors::clip_to_tile(&lines),
343            LinePlacement::Center => lines.into_iter().filter(|line| line.len() > 1).collect(),
344        };
345        let repeat_key: Option<Arc<str>> = match placement {
346            LinePlacement::Line => {
347                layout::repeat_key(&self.paint, &self.properties, self.zoom, &self.atlas)
348                    .map(Arc::from)
349            }
350            LinePlacement::Center => None,
351        };
352        for part in parts {
353            let anchors = match placement {
354                LinePlacement::Line => line_anchors::line_anchors(&part, params),
355                LinePlacement::Center => line_anchors::center_anchor(&part, params)
356                    .into_iter()
357                    .collect(),
358            };
359            if anchors.is_empty() {
360                continue;
361            }
362            let polyline: Arc<[[f32; 2]]> = part
363                .iter()
364                .map(|point| [point[0] as f32, point[1] as f32])
365                .collect();
366            for anchor in anchors {
367                if self.anchor_is_too_close(
368                    repeat_key.as_deref(),
369                    anchor.point,
370                    params.spacing / 2.0,
371                ) {
372                    continue;
373                }
374                let distance = line_anchors::distance_to(&part, anchor);
375                self.collect(
376                    Point::new(anchor.point[0], anchor.point[1]),
377                    anchor.angle as f32,
378                    source,
379                    (
380                        Some((polyline.clone(), distance as f32)),
381                        repeat_key.as_ref(),
382                    ),
383                );
384            }
385        }
386    }
387}

389/// The rings of a polygon wound as GL JS winds them before it places labels: the outer ring
390/// runs clockwise on screen and holes run counter-clockwise, so labels start at the same end.
391fn polygon_rings(polygon: &geo_types::Polygon<f64>) -> Vec<Vec<[f64; 2]>> {
392    let wound = |ring: &geo_types::LineString<f64>, outer: bool| {
393        let mut points: Vec<[f64; 2]> = ring.coords().map(|c| [c.x, c.y]).collect();
394        let area: f64 = points
395            .iter()
396            .zip(points.iter().cycle().skip(1))
397            .map(|(a, b)| a[0] * b[1] - b[0] * a[1])
398            .sum();
399        // Tile coordinates grow downwards, so positive area is clockwise on screen.
400        if (area > 0.0) != outer {
401            points.reverse();
402        }
403        points
404    };
405    std::iter::once(wound(polygon.exterior(), true))
406        .chain(polygon.interiors().iter().map(|ring| wound(ring, false)))
407        .collect()
408}

410impl FeatureProcessor for TextTessellator {
411    fn feature_end(&mut self, idx: u64) -> GeoResult<()> {
412        if let Some(geometry) = self.geo_writer.take_geometry() {
413            let id = usize::try_from(idx)
414                .ok()
415                .and_then(|idx| self.source_ids.get(idx).copied())
416                .flatten();
417            let kind = match &geometry {
418                Geometry::Point(_) | Geometry::MultiPoint(_) => GeometryType::Point,
419                Geometry::LineString(_) | Geometry::MultiLineString(_) => GeometryType::LineString,
420                Geometry::Polygon(_) | Geometry::MultiPolygon(_) => GeometryType::Polygon,
421                _ => GeometryType::Unknown,
422            };
423            let source = layout::SourceFeature { id, kind };
424            let lines: Option<Vec<Vec<[f64; 2]>>> = match &geometry {
425                Geometry::LineString(line) => {
426                    Some(vec![line.coords().map(|c| [c.x, c.y]).collect()])
427                }
428                Geometry::MultiLineString(lines) => Some(
429                    lines
430                        .iter()
431                        .map(|line| line.coords().map(|c| [c.x, c.y]).collect())
432                        .collect(),
433                ),
434                // The rings of a polygon are lines to follow, as in GL JS.
435                Geometry::Polygon(polygon) => Some(polygon_rings(polygon)),
436                Geometry::MultiPolygon(polygons) => {
437                    Some(polygons.iter().flat_map(polygon_rings).collect())
438                }
439                _ => None,
440            };
441            match (line_placement(&self.paint), lines) {
442                // Lines that continue one another are merged once all of them are known.
443                (Some(LinePlacement::Line), Some(lines)) => {
444                    let text = self.paint.label_among(
445                        &self.properties,
446                        self.zoom,
447                        Some(&self.atlas.icons),
448                    );
449                    for line in lines.into_iter().filter(|line| line.len() > 1) {
450                        self.pending_lines.push(line_merge::PendingLine {
451                            text: text.clone(),
452                            line,
453                            source,
454                            properties: self.properties.clone(),
455                        });
456                    }
457                }
458                (Some(placement), Some(lines)) => {
459                    self.collect_along_lines(lines, placement, source)
460                }
461                // A point has no line to follow.
462                (Some(_), None) => {}
463                (None, _) => {
464                    // The buffer around a tile repeats the labels of its neighbours, which draw
465                    // them from their own tiles.
466                    let tile = 0.0..crate::coords::EXTENT;
467                    for anchor in layout::anchors(&geometry)
468                        .into_iter()
469                        .filter(|anchor| tile.contains(&anchor.x()) && tile.contains(&anchor.y()))
470                    {
471                        self.collect(anchor, 0.0, source, (None, None));
472                    }
473                }
474            }
475        }
476        self.properties.clear();
477        Ok(())
478    }
