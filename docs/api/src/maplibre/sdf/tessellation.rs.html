39/// Worker-side geometry for screen-sized symbols anchored to map features.
40pub struct TextTessellator {
41    geo_writer: GeoWriter,
42    paint: SymbolPaint,
43    zoom: f64,
44    atlas: Arc<SymbolAtlas>,
45    collected: Vec<CollectedSymbol>,
46    /// Lines to be labelled along, held until they can be merged.
47    pending_lines: Vec<line_merge::PendingLine>,
48    /// Where each text already has a label along a line, so a repeat of it keeps its distance.
49    line_anchors_by_text: rustc_hash::FxHashMap<String, Vec<[f64; 2]>>,
50    properties: FeatureProperties,
51    /// Pre-decoded shared properties indexed by feature order when supplied by MVT processing.
52    pub(crate) shared_properties: Vec<Arc<FeatureProperties>>,
53    /// Symbol triangles.
54    pub quad_buffer: VertexBuffers<ShaderSymbolVertex, IndexDataType>,
55    /// Collision bounds and exact triangle ranges per symbol.
56    pub features: Vec<Feature>,
57    /// Source layer extent conversion.
58    pub coordinate_scale: f64,
59    /// How many times the tile is magnified past the zoom its source stops at; zero or one when
60    /// it is not.
61    pub overscaling: f64,
62    /// Source IDs indexed by the feature order supplied to the geometry processor.
63    pub source_ids: Vec<Option<u64>>,
64    /// Original MVT/MLT feature indices indexed by slice feature order.
65    pub source_indices: Vec<Option<u32>>,
66    /// Source `promoteId` and active runtime `feature-state` snapshot for this layer.
67    pub(crate) feature_state_ctx:
68        Option<crate::vector::tessellation::feature_paint::FeatureStateSyncContext>,
69    /// When `Some`, point symbols stream straight into `quad_buffer`/`features` and prune
70    /// feature properties immediately instead of buffering `CollectedSymbol`s.
71    stream_point_runtime_keys: Option<Option<rustc_hash::FxHashSet<String>>>,
72}

74impl TextTessellator {
75    /// A text collector with the offline fallback font.
76    pub fn new(text_field: StyleProperty<FormattedText>, zoom: f64) -> Self {
77        Self {
78            paint: SymbolPaint {
79                text_field: Some(text_field),
80                ..Default::default()
81            },
82            zoom,
83            ..Default::default()
84        }
85    }
86
87    /// Collects features using an existing atlas without decoding fallback glyphs again.
88    pub fn with_assets(mut paint: SymbolPaint, zoom: f64, atlas: Arc<SymbolAtlas>) -> Self {
89        paint.resolve_zoom_enums(zoom);
90        Self {
91            geo_writer: GeoWriter::default(),
92            paint,
93            zoom,
94            atlas,
95            collected: Vec::new(),
96            pending_lines: Vec::new(),
97            line_anchors_by_text: rustc_hash::FxHashMap::default(),
98            properties: FeatureProperties::default(),
99            shared_properties: Vec::new(),
100            quad_buffer: VertexBuffers::new(),
101            features: Vec::new(),
102            coordinate_scale: 1.0,
103            overscaling: 1.0,
104            source_ids: Vec::new(),
105            source_indices: Vec::new(),
106            feature_state_ctx: None,
107            stream_point_runtime_keys: None,
108        }
109    }
110
111    /// Streams point-placed labels straight into vertex/feature buffers and prunes properties
112    /// per feature instead of retaining every `CollectedSymbol` until [`Self::finish`].
113    pub(crate) fn enable_streaming_point_layout(
114        &mut self,
115        runtime_keys: Option<rustc_hash::FxHashSet<String>>,
116    ) {
117        self.stream_point_runtime_keys = Some(runtime_keys);
118    }
119
120    /// Whether point-placed symbols streamed and pruned properties during collection.
121    pub(crate) fn streamed_point_layout(&self) -> bool {
122        self.stream_point_runtime_keys.is_some()
123    }
124
125    /// Sets the style and the assets used by this tile's symbols.
126    pub fn configure(&mut self, mut paint: SymbolPaint, atlas: Arc<SymbolAtlas>) {
127        paint.resolve_zoom_enums(self.zoom);
128        self.paint = paint;
129        self.atlas = atlas;
130    }
131
132    /// Builds quads and collision ranges from the collected map features.
133    pub fn finish(&mut self) {
134        let pending = std::mem::take(&mut self.pending_lines);
135        for line in line_merge::merge_lines(pending) {
136            self.collect_along_lines(
137                vec![line.line],
138                LinePlacement::Line,
139                line.source,
140                &line.properties,
141            );
142        }
143        self.properties.clear();
144        let collected = std::mem::take(&mut self.collected);
145        self.shared_properties.clear();
146        prune::finish_collected(
147            collected,
148            (&self.paint, self.zoom, self.overscaling),
149            &self.atlas,
150            &mut self.quad_buffer,
151            &mut self.features,
152        );
153        if draw_order::sorts_by_height(&self.paint, self.zoom) {
154            draw_order::reorder(&mut self.quad_buffer, &mut self.features);
155        }
156    }
157}

159impl Default for TextTessellator {
160    fn default() -> Self {
161        Self::with_assets(SymbolPaint::default(), 0.0, fallback_atlas())
162    }

208impl PropertyProcessor for TextTessellator {

210        if let Some(value) = property_value(value) {

217#[derive(Clone, Copy, PartialEq, Eq)]
218enum LinePlacement {
219    Line,
220    Center,
221}
222
223fn line_placement(paint: &SymbolPaint) -> Option<LinePlacement> {
224    match paint
225        .properties
226        .get("symbol-placement")
227        .and_then(|value| value.as_str())
228    {
229        Some("line") => Some(LinePlacement::Line),
230        Some("line-center") => Some(LinePlacement::Center),
231        _ => None,
232    }
233}

235/// Whether the layer places its text along lines.
236pub(crate) fn is_line_placed(paint: &SymbolPaint) -> bool {
237    line_placement(paint).is_some()

240const TILE_UNITS_PER_PIXEL: f64 = 8.0;
241
242impl TextTessellator {
243    pub(super) fn take_feature_properties(
244        &mut self,
245        idx: Option<usize>,
246        id: Option<u64>,
247    ) -> Arc<FeatureProperties> {
248        if let Some(shared) = idx.and_then(|i| self.shared_properties.get(i)) {
249            self.properties.clear();
250            return Arc::clone(shared);
251        }
252        let mut props = std::mem::take(&mut self.properties);
253        if let Some(ctx) = &self.feature_state_ctx {
254            crate::style::feature_state::sync_properties_feature_state(
255                &mut props,
256                id,
257                ctx.promote_property.as_deref(),
258                ctx.source_state.as_deref(),
259                ctx.source_layer.as_deref(),
260            );
261        }
262        if props.is_empty() {
263            Arc::clone(empty_feature_properties())
264        } else {
265            Arc::new(props)
266        }
267    }
268
269    fn collect(
270        &mut self,
271        anchor: Point<f64>,
272        angle: f32,
273        source: layout::SourceFeature,
274        props: &Arc<FeatureProperties>,
275        line_context: (Option<layout::LineContext>, Option<&Arc<str>>),
276    ) {
277        prune::collect_symbol(self, anchor, angle, source, props, line_context);
278    }
279
280    fn anchor_is_too_close(
281        &mut self,
282        key: Option<&str>,
283        point: [f64; 2],
284        repeat_distance: f64,
285    ) -> bool {
286        let Some(key) = key else {
287            return false;
288        };
289        let others = self
290            .line_anchors_by_text
291            .entry(key.to_string())
292            .or_default();
293        let too_close = others
294            .iter()
295            .any(|other| (other[0] - point[0]).hypot(other[1] - point[1]) < repeat_distance);
296        if !too_close {
297            others.push(point);
298        }
299        too_close
300    }
301
302    fn collect_along_lines(
303        &mut self,
304        lines: Vec<Vec<[f64; 2]>>,
305        placement: LinePlacement,
306        source: layout::SourceFeature,
307        props: &Arc<FeatureProperties>,
308    ) {
309        let probe = CollectedSymbol {
310            source,
311            line: None,
312            anchor: Point::new(0.0, 0.0),
313            angle: 0.0,
314            properties: Arc::clone(props),
315            vertical: None,
316            fallback: false,
317            repeat_key: None,
318        };
319        let text_size = self
320            .paint
321            .text_size
322            .as_ref()
323            .and_then(|value| value.evaluate_at_zoom(self.zoom + 1.0))
324            .map_or(16.0, f64::from);
325        let text_width = f64::from(text_layout::unwrapped_width(
326            &self.paint,
327            &probe,
328            self.zoom,
329            &self.atlas,
330        ));
331        let icon_width = self
332            .paint
333            .text_among_images("icon-image", props, self.zoom, &self.atlas.icons)
334            .and_then(|name| self.atlas.icons.get(&name))
335            .map_or(0.0, |icon| {
336                f64::from(icon.rect[2]) / f64::from(icon.metrics[3])
337                    * f64::from(self.paint.number("icon-size", props, self.zoom, 1.0))
338            });
339        let label_pixels = text_width.max(icon_width) * text_size / 24.0;
340        let number = |name, fallback| self.paint.number(name, props, self.zoom, fallback);
341        let overscaling = self.overscaling.max(1.0);
342        let units_per_pixel = TILE_UNITS_PER_PIXEL / overscaling;
343        let params = line_anchors::AnchorSpacing {
344            spacing: f64::from(number("symbol-spacing", 250.0)) * units_per_pixel,
345            max_angle: f64::from(number("text-max-angle", 45.0)).to_radians(),
346            label_length: label_pixels * units_per_pixel,
347            text_size: text_size * units_per_pixel,
348            checks_bends: text_width > 0.0,
349            overscaling,
350        };
351        let parts: Vec<Vec<[f64; 2]>> = match placement {
352            LinePlacement::Line => line_anchors::clip_to_tile(&lines),
353            LinePlacement::Center => lines.into_iter().filter(|line| line.len() > 1).collect(),
354        };
355        let repeat_key: Option<Arc<str>> = match placement {
356            LinePlacement::Line => {
357                layout::repeat_key(&self.paint, props, self.zoom, &self.atlas).map(Arc::from)
358            }
359            LinePlacement::Center => None,
360        };
361        for part in parts {
362            let anchors = match placement {
363                LinePlacement::Line => line_anchors::line_anchors(&part, params),
364                LinePlacement::Center => line_anchors::center_anchor(&part, params)
365                    .into_iter()
366                    .collect(),
367            };
368            if anchors.is_empty() {
369                continue;
370            }
371            let polyline: Arc<[[f32; 2]]> = part
372                .iter()
373                .map(|point| [point[0] as f32, point[1] as f32])
374                .collect();
375            for anchor in anchors {
376                if self.anchor_is_too_close(
377                    repeat_key.as_deref(),
378                    anchor.point,
379                    params.spacing / 2.0,
380                ) {
381                    continue;
382                }
383                let distance = line_anchors::distance_to(&part, anchor);
384                self.collect(
385                    Point::new(anchor.point[0], anchor.point[1]),
386                    anchor.angle as f32,
387                    source,
388                    props,
389                    (
390                        Some((polyline.clone(), distance as f32)),
391                        repeat_key.as_ref(),
392                    ),
393                );
394            }
395        }
396    }
397}

399impl FeatureProcessor for TextTessellator {
400    fn feature_end(&mut self, idx: u64) -> GeoResult<()> {
401        if let Some(geometry) = self.geo_writer.take_geometry() {
402            let slice_idx = usize::try_from(idx).ok();
403            let (source, props) = prune::build_source_feature(self, slice_idx, &geometry);
404            let placement = line_placement(&self.paint);
405            let lines: Option<Vec<Vec<[f64; 2]>>> = if placement.is_some() {
406                match &geometry {
407                    Geometry::LineString(line) => {
408                        Some(vec![line.coords().map(|c| [c.x, c.y]).collect()])
409                    }
410                    Geometry::MultiLineString(lines) => Some(
411                        lines
412                            .iter()
413                            .map(|line| line.coords().map(|c| [c.x, c.y]).collect())
414                            .collect(),
415                    ),
416                    Geometry::Polygon(polygon) => Some(line_anchors::polygon_rings(polygon)),
417                    Geometry::MultiPolygon(polygons) => Some(
418                        polygons
419                            .iter()
420                            .flat_map(line_anchors::polygon_rings)
421                            .collect(),
422                    ),
423                    _ => None,
424                }
425            } else {
426                None
427            };
428            match (placement, lines) {
429                (Some(LinePlacement::Line), Some(lines)) => {
430                    let text = self
431                        .paint
432                        .label_among(&props, self.zoom, Some(&self.atlas.icons));
433                    for line in lines.into_iter().filter(|line| line.len() > 1) {
434                        self.pending_lines.push(line_merge::PendingLine {
435                            text: text.clone(),
436                            line,
437                            source,
438                            properties: Arc::clone(&props),
439                        });
440                    }
441                }
442                (Some(placement), Some(lines)) => {
443                    self.collect_along_lines(lines, placement, source, &props)
444                }
445                (Some(_), None) => {}
446                (None, _) => {
447                    let tile = 0.0..crate::coords::EXTENT;
448                    for anchor in layout::anchors(&geometry)
449                        .into_iter()
450                        .filter(|anchor| tile.contains(&anchor.x()) && tile.contains(&anchor.y()))
451                    {
452                        self.collect(anchor, 0.0, source, &props, (None, None));
453                    }
454                    if self.stream_point_runtime_keys.is_some() {
455                        if let Ok(mut recycled) = Arc::try_unwrap(props) {
456                            recycled.clear();
457                            self.properties = recycled;
458                        }
459                    }
460                }
461            }
462        } else {
463            self.properties.clear();
464        }
465        Ok(())
466    }
