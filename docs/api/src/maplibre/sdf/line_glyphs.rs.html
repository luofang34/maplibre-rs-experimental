1//! Glyph positions along a polyline, for labels whose text follows the line.
2//!
3//! Glyphs sit at fixed distances from the label's anchor along the line, so a bend turns
4//! them with it instead of leaving them on one straight baseline. Distances and points are
5//! tile units, the space of the label's map-aligned plane.
6
7/// Where one glyph's centre lies and which way it points.
8#[derive(Clone, Copy, Debug, PartialEq)]
9pub(crate) struct GlyphPose {
10    /// Centre of the glyph in tile units.
11    pub point: [f32; 2],
12    /// Direction of the line under the glyph in radians, turned half way around when flipped.
13    pub angle: f32,
14    /// How far the line under the glyph stands above the ground under the label's anchor, in
15    /// metres, for a label that follows the ground: the ground under the ends of the glyph's
16    /// segment of the line, interpolated along it. Zero otherwise.
17    pub rise: f32,
18}
19
20/// Finds points by arc length along a polyline, resuming where the last one was found so that
21/// a run of ascending distances walks the line once.
22struct LineWalker<'a> {
23    polyline: &'a [[f32; 2]],
24    segment: usize,
25    travelled: f32,
26}
27
28impl<'a> LineWalker<'a> {
29    fn new(polyline: &'a [[f32; 2]]) -> Self {
30        Self {
31            polyline,
32            segment: 0,
33            travelled: 0.0,
34        }
35    }
36
37    /// The point at arc length `distance` from the start of the polyline and the direction of
38    /// the segment it lies on; `None` when the line is too short to reach it. Distances must
39    /// not decrease between calls.
40    fn point_at(&mut self, distance: f32) -> Option<([f32; 2], f32)> {
41        if distance < 0.0 {
42            return None;
43        }
44        while let [from, to] = self.polyline.get(self.segment..self.segment + 2)? {
45            let (dx, dy) = (to[0] - from[0], to[1] - from[1]);
46            let length = dx.hypot(dy);
47            if length > 0.0 && distance <= self.travelled + length {
48                let t = (distance - self.travelled) / length;
49                return Some(([from[0] + dx * t, from[1] + dy * t], dy.atan2(dx)));
50            }
51            self.travelled += length;
52            self.segment += 1;
53        }
54        None
55    }
56}

58/// Poses of glyphs whose centres are `offsets` (tile units, signed) from the anchor at
59/// `anchor_distance` along `polyline`. `flip` reads the text against the line direction: each
60/// glyph takes the mirrored distance and turns half way around. `None` when any glyph would
61/// lie beyond either end of the line, in which case the label is not drawn.
62pub(crate) fn place_glyphs(
63    polyline: &[[f32; 2]],
64    anchor_distance: f32,
65    offsets: &[f32],
66    flip: bool,
67) -> Option<Vec<GlyphPose>> {
68    let distances: Vec<f32> = offsets
69        .iter()
70        .map(|offset| {
71            if flip {
72                anchor_distance - offset
73            } else {
74                anchor_distance + offset
75            }
76        })
77        .collect();
78    // Glyphs are found in order of distance with one walk along the line.
79    let mut order: Vec<usize> = (0..distances.len()).collect();
80    order.sort_by(|a, b| distances[*a].total_cmp(&distances[*b]));
81    let mut walker = LineWalker::new(polyline);
82    let mut poses = vec![None; distances.len()];
83    for index in order {
84        let (point, angle) = walker.point_at(distances[index])?;
85        poses[index] = Some(GlyphPose {
86            point,
87            angle: if flip {
88                angle + std::f32::consts::PI
89            } else {
90                angle
91            },
92            rise: 0.0,
93        });
94    }
95    poses.into_iter().collect()
96}

98/// Whether text that reads along the line would end up upside down: its first glyph lies to
99/// the right of its last on the screen, as GL JS decides for horizontal text.
100pub(crate) fn reads_backwards(first: [f64; 2], last: [f64; 2]) -> bool {
101    first[0] > last[0]
102}
