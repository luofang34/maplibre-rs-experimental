23/// Coordinates and metrics of a glyph or sprite in the atlas.
24#[derive(Clone, Debug, Default, Serialize, Deserialize)]
25pub struct AtlasEntry {
26    /// Pixel rectangle x, y, width, height.
27    pub rect: [u32; 4],
28    /// Left bearing, top bearing, advance, pixel ratio. An icon has no bearings or advance;
29    /// its first two are instead the offset, in image pixels, from its centre to the point
30    /// placed where its centre would be.
31    pub metrics: [f32; 4],
32    /// Zero for text SDF, one for RGBA icons, two for SDF icons.
33    pub kind: u32,
34    /// How an icon stretches around its text, from the sprite's `stretchX`, `stretchY`,
35    /// `content` and `textFit*` fields; `None` for glyphs and plain icons.
36    #[serde(default)]
37    pub stretch: Option<Box<IconStretch>>,
38}
39
40/// The sprite fields that let an icon stretch to fit its text, in image pixels.
41#[derive(Clone, Debug, Default, PartialEq, Serialize, Deserialize)]
42pub struct IconStretch {
43    /// Column ranges that grow; empty when the icon does not stretch horizontally.
44    pub stretch_x: Vec<[f32; 2]>,
45    /// Row ranges that grow; empty when the icon does not stretch vertically.
46    pub stretch_y: Vec<[f32; 2]>,
47    /// The area, `[left, top, right, bottom]`, that the text is fitted to.
48    pub content: Option<[f32; 4]>,
49    /// How the content area follows the text horizontally.
50    pub text_fit_width: Option<TextFit>,
51    /// How the content area follows the text vertically.
52    pub text_fit_height: Option<TextFit>,
53}
54
55/// The `textFitWidth` and `textFitHeight` sprite values.
56#[derive(Clone, Copy, Debug, PartialEq, Eq, Serialize, Deserialize)]
57pub enum TextFit {
58    /// The content area may stretch or shrink to the text.
59    StretchOrShrink,
60    /// The content area may only stretch.
61    StretchOnly,
62    /// The content area keeps its aspect ratio.
63    Proportional,
64}

66/// Immutable atlas shared by all symbol layers of a tile.
67#[derive(Clone, Default, Debug, Serialize, Deserialize)]
68pub struct SymbolAtlas {
69    /// RGBA pixels.
70    pub pixels: Vec<u8>,
71    /// Texture dimensions.
72    pub size: [u32; 2],
73    /// Font stack and Unicode scalar to atlas entry.
74    pub glyphs: HashMap<String, HashMap<u32, AtlasEntry>>,
75    /// Sprite names to atlas entries.
76    pub icons: HashMap<String, AtlasEntry>,
77}
78
79impl SymbolAtlas {
80    /// Estimated CPU storage, counting allocated pixel capacity and glyph metadata.
81    pub fn approximate_bytes(&self) -> usize {
82        self.pixels.capacity()
83            + self
84                .glyphs
85                .iter()
86                .map(|(font, glyphs)| {
87                    font.capacity()
88                        + glyphs.capacity()
89                            * (std::mem::size_of::<u32>() + std::mem::size_of::<AtlasEntry>())
90                })
91                .sum::<usize>()
92            + self.icons.keys().map(|name| name.capacity()).sum::<usize>()
93            + self.icons.capacity()
94                * (std::mem::size_of::<String>() + std::mem::size_of::<AtlasEntry>())
95    }
96}

98/// Incremental shelf packer with a transparent border around every image.
99pub(super) struct AtlasBuilder {
100    pub(super) atlas: SymbolAtlas,
101    cursor: [u32; 2],
102    row_height: u32,
103}
104
105impl AtlasBuilder {
106    pub(super) fn new() -> Self {
107        Self {
108            atlas: SymbolAtlas {
109                size: [1024, 1],
110                ..Default::default()
111            },
112            cursor: [1, 1],
113            row_height: 0,
114        }
115    }
116
117    pub(super) fn pack(&mut self, width: u32, height: u32, pixels: &[u8]) -> Option<[u32; 4]> {
118        let [atlas_width, _] = self.atlas.size;
119        if width > atlas_width - 2 || height > 4094 || pixels.len() != (width * height * 4) as usize
120        {
121            return None;
122        }
123        if self.cursor[0] + width + 1 > atlas_width {
124            self.cursor = [1, self.cursor[1] + self.row_height + 2];
125            self.row_height = 0;
126        }
127        let [x, y] = self.cursor;
128        if y + height + 1 > 4096 {
129            return None;
130        }
131        let atlas_height = self.atlas.size[1].max(y + height + 1);
132        self.atlas
133            .pixels
134            .resize((atlas_height * atlas_width * 4) as usize, 0);
135        for row in 0..height {
136            let start = (((y + row) * atlas_width + x) * 4) as usize;
137            let source = (row * width * 4) as usize;
138            self.atlas.pixels[start..start + (width * 4) as usize]
139                .copy_from_slice(&pixels[source..source + (width * 4) as usize]);
140        }
141        self.cursor[0] += width + 2;
142        self.row_height = self.row_height.max(height);
143        self.atlas.size[1] = atlas_height;
144        Some([x, y, width, height])
145    }
146
147    pub(super) fn glyph_range(
148        &mut self,
149        font: &str,
150        bytes: &[u8],
151    ) -> Result<(), prost::DecodeError> {
152        self.glyph_subset(font, &glyphs::Glyphs::decode(bytes)?, None);
153        Ok(())
154    }
155
156    pub(super) fn glyph_subset(
157        &mut self,
158        font: &str,
159        data: &glyphs::Glyphs,
160        wanted: Option<&std::collections::BTreeSet<u32>>,
161    ) {
162        for glyph in data.stacks.iter().flat_map(|stack| &stack.glyphs) {
163            if wanted.is_some_and(|wanted| !wanted.contains(&glyph.id)) {
164                continue;
165            }
166            if glyph.width > 1016 || glyph.height > 4088 {
167                continue;
168            }
169            let rect =
170                if let Some(bitmap) = glyph.bitmap.as_ref().filter(|bitmap| !bitmap.is_empty()) {
171                    let pixels: Vec<u8> = bitmap
172                        .iter()
173                        .flat_map(|&value| [value, value, value, 255])
174                        .collect();
175                    let Some(rect) = self.pack(glyph.width + 6, glyph.height + 6, &pixels) else {
176                        continue;
177                    };
178                    rect
179                } else {
180                    [0; 4]
181                };
182            self.atlas
183                .glyphs
184                .entry(font.to_string())
185                .or_default()
186                .insert(
187                    glyph.id,
188                    AtlasEntry {
189                        rect,
190                        metrics: [
191                            glyph.left as f32 - 3.0,
192                            glyph.top as f32 + 3.0,
193                            glyph.advance as f32,
194                            1.0,
195                        ],
196                        kind: 0,
197                        ..Default::default()
198                    },
199                );
200        }
201    }
202
203    pub(super) fn finish(mut self) -> Arc<SymbolAtlas> {
204        self.atlas.size[1] = self.atlas.size[1].max(1);
205        self.atlas
206            .pixels
207            .resize((self.atlas.size[0] * self.atlas.size[1] * 4) as usize, 0);
208        Arc::new(self.atlas)
209    }
210}

212/// Bundled glyphs keep offline styles usable when no glyph URL is supplied.
213pub fn fallback_atlas() -> Arc<SymbolAtlas> {
214    let mut builder = AtlasBuilder::new();
215    if let Err(error) = builder.glyph_range(
216        "Open Sans Regular",
217        include_bytes!("../../../data/0-255.pbf"),
218    ) {
219        tracing::error!(%error, "bundled symbol glyphs are invalid");
220    }
221    builder.finish()
222}
