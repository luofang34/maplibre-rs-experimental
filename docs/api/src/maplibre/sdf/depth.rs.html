1//! A snapshot of opaque depth for symbol-anchor visibility and per-frame fade clock uniform.
2
3use crate::render::depth_copy::DepthCopyPipeline;
4
5pub(crate) struct SymbolDepth {
6    pub texture: wgpu::Texture,
7    samples: u32,
8    pub view: wgpu::TextureView,
9    fade_buffer: wgpu::Buffer,
10    last_fade_bits: Option<[u32; 2]>,
11    pub binding: wgpu::BindGroup,
12    pub copy: DepthCopyPipeline,
13}
14
15impl SymbolDepth {
16    pub fn layout() -> Vec<wgpu::BindGroupLayoutEntry> {
17        vec![
18            wgpu::BindGroupLayoutEntry {
19                binding: 0,
20                visibility: wgpu::ShaderStages::VERTEX,
21                ty: wgpu::BindingType::Texture {
22                    // An unfiltered float binding preserves raw depth reads on GLSL backends.
23                    sample_type: wgpu::TextureSampleType::Float { filterable: false },
24                    view_dimension: wgpu::TextureViewDimension::D2,
25                    multisampled: false,
26                },
27                count: None,
28            },
29            wgpu::BindGroupLayoutEntry {
30                binding: 1,
31                visibility: wgpu::ShaderStages::VERTEX,
32                ty: wgpu::BindingType::Buffer {
33                    ty: wgpu::BufferBindingType::Uniform,
34                    has_dynamic_offset: false,
35                    min_binding_size: std::num::NonZeroU64::new(16),
36                },
37                count: None,
38            },
39        ]
40    }
41
42    pub fn new(
43        device: &wgpu::Device,
44        size: wgpu::Extent3d,
45        samples: u32,
46        pipeline: &wgpu::RenderPipeline,
47    ) -> Self {
48        let texture = device.create_texture(&wgpu::TextureDescriptor {
49            label: Some("symbol anchor depth"),
50            size,
51            mip_level_count: 1,
52            sample_count: 1,
53            dimension: wgpu::TextureDimension::D2,
54            format: wgpu::TextureFormat::Depth32Float,
55            usage: wgpu::TextureUsages::RENDER_ATTACHMENT | wgpu::TextureUsages::TEXTURE_BINDING,
56            view_formats: &[],
57        });
58        let view = texture.create_view(&Default::default());
59        let fade_buffer = device.create_buffer(&wgpu::BufferDescriptor {
60            label: Some("symbol fade clock"),
61            size: 16,
62            usage: wgpu::BufferUsages::UNIFORM | wgpu::BufferUsages::COPY_DST,
63            mapped_at_creation: false,
64        });
65        let binding = device.create_bind_group(&wgpu::BindGroupDescriptor {
66            label: Some("symbol anchor depth and fade"),
67            layout: &pipeline.get_bind_group_layout(2),
68            entries: &[
69                wgpu::BindGroupEntry {
70                    binding: 0,
71                    resource: wgpu::BindingResource::TextureView(&view),
72                },
73                wgpu::BindGroupEntry {
74                    binding: 1,
75                    resource: fade_buffer.as_entire_binding(),
76                },
77            ],
78        });
79        Self {
80            texture,
81            view,
82            fade_buffer,
83            last_fade_bits: None,
84            binding,
85            samples,
86            copy: DepthCopyPipeline::new(device, samples),
87        }
88    }
89
90    pub fn write_fade_time(
91        &mut self,
92        queue: &crate::render::upload_queue::UploadQueue,
93        fade_time_ms: f32,
94        fade_duration_ms: f32,
95    ) {
96        let bits = [fade_time_ms.to_bits(), fade_duration_ms.to_bits()];
97        if self.last_fade_bits != Some(bits) {
98            let payload = [fade_time_ms, fade_duration_ms, 0.0_f32, 0.0_f32];
99            queue.write_buffer(&self.fade_buffer, 0, bytemuck::bytes_of(&payload));
100            self.last_fade_bits = Some(bits);
101        }
102    }
103}
