5pub(crate) struct SymbolDepth {

15    pub binding: wgpu::BindGroup,
16    pub copy: Option<DepthCopyPipeline>,

18
19impl SymbolDepth {
20    pub fn layout() -> Vec<wgpu::BindGroupLayoutEntry> {
21        vec![
22            wgpu::BindGroupLayoutEntry {
23                binding: 0,
24                visibility: wgpu::ShaderStages::VERTEX,
25                ty: wgpu::BindingType::Texture {
26                    // An unfiltered float binding preserves raw depth reads on GLSL backends.
27                    sample_type: wgpu::TextureSampleType::Float { filterable: false },
28                    view_dimension: wgpu::TextureViewDimension::D2,
29                    multisampled: false,
30                },
31                count: None,
32            },
33            wgpu::BindGroupLayoutEntry {
34                binding: 1,
35                visibility: wgpu::ShaderStages::VERTEX,
36                ty: wgpu::BindingType::Buffer {
37                    ty: wgpu::BufferBindingType::Uniform,
38                    has_dynamic_offset: false,
39                    min_binding_size: std::num::NonZeroU64::new(16),
40                },
41                count: None,
42            },
43        ]
44    }
45
46    pub fn new(
47        device: &wgpu::Device,
48        size: wgpu::Extent3d,
49        samples: u32,
50        pipeline: &wgpu::RenderPipeline,
51    ) -> Self {
52        let placeholder_size = wgpu::Extent3d {
53            width: 1,
54            height: 1,
55            depth_or_array_layers: 1,
56        };
57        let (texture, view) = create_depth_target(device, placeholder_size);
58        let fade_buffer = device.create_buffer(&wgpu::BufferDescriptor {
59            label: Some("symbol fade clock"),
60            size: 16,
61            usage: wgpu::BufferUsages::UNIFORM | wgpu::BufferUsages::COPY_DST,
62            mapped_at_creation: false,
63        });
64        let bind_group_layout = pipeline.get_bind_group_layout(2);
65        let binding = create_bind_group(device, &bind_group_layout, &view, &fade_buffer);
66        Self {
67            texture,
68            target_size: size,
69            samples,
70            allocated_full: false,
71            bind_group_layout,
72            view,
73            fade_buffer,
74            last_fade_bits: None,
75            skip_depth_copy: true,
76            binding,
77            copy: None,
78        }
79    }
80
81    fn ensure_allocated(&mut self, device: &wgpu::Device) {
82        if self.allocated_full {
83            return;
84        }
85        let (new_tex, new_view) = create_depth_target(device, self.target_size);
86        let new_binding = create_bind_group(
87            device,
88            &self.bind_group_layout,
89            &new_view,
90            &self.fade_buffer,
91        );
92        let old_tex = std::mem::replace(&mut self.texture, new_tex);
93        old_tex.destroy();
94        self.view = new_view;
95        self.binding = new_binding;
96        if self.copy.is_none() {
97            self.copy = Some(DepthCopyPipeline::new(device, self.samples));
98        }
99        self.allocated_full = true;
100    }
101
102    pub fn write_fade_time(
103        &mut self,
104        device: &wgpu::Device,
105        queue: &crate::render::upload_queue::UploadQueue,
106        fade_time_ms: f32,
107        fade_duration_ms: f32,
108        skip_depth_occlusion: bool,
109    ) {
110        if !skip_depth_occlusion {
111            self.ensure_allocated(device);
112        }
113        self.skip_depth_copy = skip_depth_occlusion;
114        let skip_flag = if skip_depth_occlusion {
115            1.0_f32
116        } else {
117            0.0_f32
118        };
119        let bits = [
120            fade_time_ms.to_bits(),
121            fade_duration_ms.to_bits(),
122            skip_flag.to_bits(),
123        ];
124        if self.last_fade_bits != Some(bits) {
125            let payload = [fade_time_ms, fade_duration_ms, skip_flag, 0.0_f32];
126            queue.write_buffer(&self.fade_buffer, 0, bytemuck::bytes_of(&payload));
127            self.last_fade_bits = Some(bits);
128        }
129    }
130}

132fn create_depth_target(
133    device: &wgpu::Device,
134    size: wgpu::Extent3d,
135) -> (wgpu::Texture, wgpu::TextureView) {
136    let texture = device.create_texture(&wgpu::TextureDescriptor {
137        label: Some("symbol anchor depth"),
138        size,
139        mip_level_count: 1,
140        sample_count: 1,
141        dimension: wgpu::TextureDimension::D2,
142        format: wgpu::TextureFormat::Depth32Float,
143        usage: wgpu::TextureUsages::RENDER_ATTACHMENT | wgpu::TextureUsages::TEXTURE_BINDING,
144        view_formats: &[],
145    });
146    let view = texture.create_view(&Default::default());
147    (texture, view)
148}
149
150fn create_bind_group(
151    device: &wgpu::Device,
152    layout: &wgpu::BindGroupLayout,
153    view: &wgpu::TextureView,
154    fade_buffer: &wgpu::Buffer,
155) -> wgpu::BindGroup {
156    device.create_bind_group(&wgpu::BindGroupDescriptor {
157        label: Some("symbol anchor depth and fade"),
158        layout,
159        entries: &[
160            wgpu::BindGroupEntry {
161                binding: 0,
162                resource: wgpu::BindingResource::TextureView(view),
163            },
164            wgpu::BindGroupEntry {
165                binding: 1,
166                resource: fade_buffer.as_entire_binding(),
167            },
168        ],
169    })
170}
