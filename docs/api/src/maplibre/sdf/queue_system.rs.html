1//! Draws a stable symbol covering independently of terrain texture readiness.
2use crate::{
3    context::MapContext,
4    render::{
5        eventually::{Eventually, Eventually::Initialized},
6        render_phase::{DrawState, RenderPhase, TranslucentItem},
7        shaders::ShaderTileMetadata,
8        tile_view_pattern::TileShape,
9    },
10    sdf::{covering::SymbolCovering, render_commands::DrawSymbols, SymbolBufferPool},
11    tcs::{
12        system::{SystemError, SystemResult},
13        tiles::Tile,
14    },
15};
16
17pub fn queue_system(
18    MapContext {
19        world,
20        style,
21        view_state,
22        renderer,
23        ..
24    }: &mut MapContext,
25) -> SystemResult {
26    super::covering::update(world, style, view_state, &renderer.device, &renderer.queue);
27    let Some((covering, phase, Initialized(pool))) = world.resources.query_mut::<(
28        &SymbolCovering,
29        &mut RenderPhase<TranslucentItem>,
30        &Eventually<SymbolBufferPool>,
31    )>() else {
32        return Err(SystemError::Dependencies);
33    };
34    let stride = size_of::<ShaderTileMetadata>() as u64;
35    for (index, coords) in covering.tiles.iter().enumerate() {
36        let start = index as u64 * stride;
37        let shape = TileShape::with_buffer_range(*coords, view_state.zoom(), start..start + stride);
38        for layer in pool.index().get_layers(*coords).into_iter().flatten() {
39            if !layer
40                .style_layer
41                .is_visible_at(view_state.style_zoom().value())
42            {
43                continue;
44            }
45            phase.add(TranslucentItem {
46                draw_function: Box::new(DrawState::<TranslucentItem, DrawSymbols>::new()),
47                index: layer.style_layer.index,
48                style_layer: layer.style_layer.id.clone(),
49                tile: Tile { coords: *coords },
50                source_shape: shape.clone(),
51            });
52        }
53    }
54    Ok(())
55}
