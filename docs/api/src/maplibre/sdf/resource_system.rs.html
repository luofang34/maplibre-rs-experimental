15pub fn resource_system(
16    MapContext {
17        world, renderer, ..
18    }: &mut MapContext,
19) -> SystemResult {
20    let Some((pool, pipeline, Initialized(projection))) = world.resources.query_mut::<(
21        &mut Eventually<SymbolBufferPool>,
22        &mut Eventually<SymbolPipeline>,
23        &mut Eventually<ProjectionGpuResources>,
24    )>() else {
25        return Err(SystemError::Dependencies);
26    };
27    pool.initialize(|| {
28        BufferPool::from_device_with_sizes(&renderer.device, renderer.settings.symbol_pools)
29    });
30    pipeline.initialize(|| create_pipeline(renderer, projection));
31    let Some((depth, Initialized(pipeline))) = world.resources.query_mut::<(
32        &mut Eventually<super::depth::SymbolDepth>,
33        &Eventually<SymbolPipeline>,
34    )>() else {
35        return Err(SystemError::Dependencies);
36    };
37    let Initialized(source) = &renderer.resources.depth_texture else {
38        return Err(SystemError::Dependencies);
39    };
40    let size = source.texture.size();
41    let samples = source.texture.sample_count();
42    depth.reinitialize(
43        || super::depth::SymbolDepth::new(&renderer.device, size, samples, &pipeline.combined),
44        &(size.width, size.height, samples),
45    );
46    Ok(())
47}

49fn create_pipeline(
50    renderer: &crate::render::Renderer,
51    projection: &ProjectionGpuResources,
52) -> SymbolPipeline {
53    let surface = &renderer.resources.surface;
54    let shader = SymbolShader {
55        format: surface.surface_format(),
56    };
57    let mut descriptor = TilePipeline::new(
58        "symbol_pipeline".into(),
59        renderer.settings,
60        shader.describe_vertex(),
61        shader.describe_fragment(),
62        crate::render::resource::TilePipelineOptions {
63            depth_stencil_enabled: true,
64            update_stencil: false,
65            debug_stencil: true,
66            wireframe: false,
67            multisampling: surface.is_multisampling_supported(renderer.settings.msaa),
68            textured: true,
69        },
70    )
71    .with_depth_write()
72    .describe_render_pipeline();
73    if let Some(layout) = descriptor
74        .layout
75        .as_mut()
76        .and_then(|groups| groups.first_mut())
77    {
78        layout.push(wgpu::BindGroupLayoutEntry {
79            binding: 2,
80            visibility: wgpu::ShaderStages::VERTEX_FRAGMENT,
81            ty: wgpu::BindingType::Buffer {
82                ty: wgpu::BufferBindingType::Uniform,
83                has_dynamic_offset: false,
84                min_binding_size: None,
85            },
86            count: None,
87        });
88    }
89    descriptor
90        .layout
91        .get_or_insert_with(Vec::new)
92        .push(super::depth::SymbolDepth::layout());
93    if let Some(depth) = descriptor.depth_stencil.as_mut() {
94        depth.depth_compare = Some(wgpu::CompareFunction::Always);
95    }
96    let combined = descriptor
97        .initialize_with_prefix_layouts(&renderer.device, &[projection.bind_group_layout()]);
98    descriptor.fragment.entry_point = "halo";
99    let halo = descriptor
100        .initialize_with_prefix_layouts(&renderer.device, &[projection.bind_group_layout()]);
101    descriptor.fragment.entry_point = "fill";
102    let fill = descriptor
103        .initialize_with_prefix_layouts(&renderer.device, &[projection.bind_group_layout()]);
104    SymbolPipeline {
105        combined,
106        halo,
107        fill,
108    }
109}
