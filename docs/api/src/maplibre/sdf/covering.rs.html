34pub(super) fn targets(world: &World) -> Vec<WorldTileCoords> {
35    let mut seen = HashSet::default();
36    let mut targets = Vec::new();
37    if let Some(Initialized(pattern)) = world.resources.get::<Eventually<WgpuTileViewPattern>>() {
38        for tile in pattern.iter() {
39            tile.render_kind(TileKind::Vector, |shape| {
40                if seen.insert(shape.coords()) {
41                    targets.push(shape.coords());
42                }
43            });
44        }
45    }
46    targets
47}

49pub(super) fn upload_tiles(world: &World) -> Vec<WorldTileCoords> {
50    let mut tiles = world
51        .resources
52        .get::<SymbolCovering>()
53        .map(|covering| covering.tiles.clone())
54        .unwrap_or_default();
55    let mut seen: HashSet<_> = tiles.iter().copied().collect();
56    for target in targets(world) {
57        if seen.insert(target) {
58            tiles.push(target);
59        }
60        let mut current = target;
61        while let Some(parent) = current.get_parent() {
62            current = parent;
63            if world
64                .tiles
65                .query::<&SymbolLayersDataComponent>(parent)
66                .is_some_and(|symbols| !symbols.pending_assets)
67            {
68                if seen.insert(parent) {
69                    tiles.push(parent);
70                }
71                break;
72            }
73        }
74    }
75    tiles
76}

78fn visible_symbol_hash(style: &Style, zoom: f64) -> u64 {
79    use std::hash::{Hash, Hasher};
80    let mut hasher = rustc_hash::FxHasher::default();
81    for (idx, layer) in style.layers.iter().enumerate() {
82        if layer.type_ == "symbol" && layer.is_visible_at(zoom) {
83            idx.hash(&mut hasher);
84        }
85    }
86    hasher.finish()
87}

89fn visible_symbol_ids(style: &Style, zoom: f64) -> Vec<crate::style::StyleLayerId> {
90    style
91        .layers
92        .iter()
93        .filter(|layer| layer.type_ == "symbol" && layer.is_visible_at(zoom))
94        .map(crate::style::layer::StyleLayer::layer_id)
95        .collect()
96}

98fn ready(
99    world: &World,
100    visible_layers: &[crate::style::StyleLayerId],
101    coords: WorldTileCoords,
102) -> bool {
103    let Some(symbols) = world.tiles.query::<&SymbolLayersDataComponent>(coords) else {
104        return false;
105    };
106    if symbols.pending_assets {
107        return false;
108    }
109    let Some(Initialized(pool)) = world.resources.get::<Eventually<SymbolBufferPool>>() else {
110        return false;
111    };
112    let loaded = pool.index().get_layers(coords);
113    symbols.layers.iter().all(|layer| {
114        let style_id = crate::style::StyleLayerId::intern(&layer.style_layer_id);
115        !visible_layers.contains(&style_id)
116            || layer.buffer.usable_indices == 0
117            || layer.buffer.buffer.indices.is_empty()
118            || loaded.is_some_and(|entries| entries.iter().any(|e| e.style_layer_id == style_id))
119    })
120}

122fn select_with_status(
123    targets: &[WorldTileCoords],
124    ready: impl Fn(WorldTileCoords) -> bool,
125) -> (Vec<WorldTileCoords>, bool) {
126    let mut selected = HashSet::default();
127    let mut all_targets_ready = !targets.is_empty();
128    for &target in targets {
129        let mut current = target;
130        let mut first = true;
131        loop {
132            if ready(current) {
133                selected.insert(current);
134                break;
135            }
136            if first {
137                all_targets_ready = false;
138                first = false;
139            }
140            let Some(parent) = current.get_parent() else {
141                break;
142            };
143            current = parent;
144        }
145    }
146    // Fine labels keep their placement while a neighbour still needs a parent. Collision
147    // priority resolves duplicate anchors across levels without hiding an entire region.
148    let mut result: Vec<_> = selected.into_iter().collect();
149    result.sort_by_key(|tile| (tile.z, tile.y, tile.x));
150    (result, all_targets_ready)
151}

154pub(super) fn select(
155    targets: &[WorldTileCoords],
156    ready: impl Fn(WorldTileCoords) -> bool,
157) -> Vec<WorldTileCoords> {
158    select_with_status(targets, ready).0
159}

161pub(super) fn update(
162    world: &mut World,
163    style: &Style,
164    view: &ViewState,
165    device: &wgpu::Device,
166    queue: &crate::render::upload_queue::UploadQueue,
167) {
168    let topology_gen = match world.resources.get::<Eventually<WgpuTileViewPattern>>() {
169        Some(Initialized(pattern)) => pattern.topology_generation(),
170        _ => 0,
171    };
172    let (pool_rev, pool_evict) = match world.resources.get::<Eventually<SymbolBufferPool>>() {
173        Some(Initialized(pool)) => (pool.revision(), pool.eviction_generation()),
174        _ => (0, 0),
175    };
176    let visible_hash = visible_symbol_hash(style, view.style_zoom().value());
177    let key = (
178        topology_gen,
179        pool_rev,
180        pool_evict,
181        style.revision(),
182        visible_hash,
183    );
184    let can_reuse = crate::render::eye_covering::EyeInFrame::reuses_content(world)
185        || world
186            .resources
187            .get::<SymbolCovering>()
188            .is_some_and(|c| c.all_targets_ready && c.selection_key == Some(key));
189    if can_reuse {
190        if let Some(covering) = world.resources.get_mut::<SymbolCovering>() {
191            if covering.tiles.is_empty() {
192                return;
193            }
194            let SymbolCovering {
195                tiles,
196                buffer,
197                metadata_scratch,
198                last_uploaded_bytes,
199                ..
200            } = covering;
201            metadata_scratch.clear();
202            metadata_scratch.extend(tiles.iter().map(|coords| metadata(*coords, view)));
203            let bytes = bytemuck::cast_slice(metadata_scratch.as_slice());
204            if !bytes.is_empty() && last_uploaded_bytes.as_slice() != bytes {
205                queue.write_buffer(buffer, 0, bytes);
206                last_uploaded_bytes.clear();
207                last_uploaded_bytes.extend_from_slice(bytes);
208            }
209        }
210        return;
211    }
212    let visible_layers = visible_symbol_ids(style, view.style_zoom().value());
213    let (tiles, all_targets_ready) = select_with_status(&targets(world), |coords| {
214        ready(world, &visible_layers, coords)
215    });
216    let next_key = Some(key);
217    if tiles.is_empty() {
218        if let Some(covering) = world.resources.get_mut::<SymbolCovering>() {
219            covering.tiles.clear();
220            covering.selection_key = next_key;
221            covering.all_targets_ready = false;
222        }
223        return;
224    }
225    let mut entries = world
226        .resources
227        .get_mut::<SymbolCovering>()
228        .map(|c| std::mem::take(&mut c.metadata_scratch))
229        .unwrap_or_default();
230    entries.clear();
231    entries.extend(tiles.iter().map(|coords| metadata(*coords, view)));
232    let bytes = bytemuck::cast_slice(&entries);
233    let required = (bytes.len() as u64).max(256);
234    let grow = world
235        .resources
236        .get::<SymbolCovering>()
237        .is_none_or(|covering| covering.buffer.size() < required);
238    if grow {
239        if let Some(old) = world.resources.remove::<SymbolCovering>() {
240            queue.cancel_staged_buffer(&old.buffer);
241            old.buffer.destroy();
242        }
243        let buffer = device.create_buffer(&wgpu::BufferDescriptor {
244            label: Some("symbol tile transforms"),
245            size: required.next_power_of_two(),
246            usage: wgpu::BufferUsages::VERTEX | wgpu::BufferUsages::COPY_DST,
247            mapped_at_creation: false,
248        });
249        world.resources.insert(SymbolCovering {
250            tiles: Vec::new(),
251            buffer,
252            selection_key: None,
253            all_targets_ready: false,
254            metadata_scratch: Vec::new(),
255            last_uploaded_bytes: Vec::new(),
256        });
257    }
258    if let Some(covering) = world.resources.get_mut::<SymbolCovering>() {
259        if !bytes.is_empty() && covering.last_uploaded_bytes.as_slice() != bytes {
260            queue.write_buffer(&covering.buffer, 0, bytes);
261            covering.last_uploaded_bytes.clear();
262            covering.last_uploaded_bytes.extend_from_slice(bytes);
263        }
264        covering.tiles = tiles;
265        covering.selection_key = next_key;
266        covering.all_targets_ready = all_targets_ready;
267        covering.metadata_scratch = entries;
268    }
269}

271fn metadata(coords: WorldTileCoords, view: &ViewState) -> ShaderTileMetadata {
272    let zoom = view.zoom();
273    let zoom_factor = view.style_zoom().scale_to_tile(&coords) as f32;
274    ShaderTileMetadata {
275        transform: view
276            .gpu_view_projection()
277            .to_model_view_projection(coords.transform_for_zoom(zoom))
278            .downcast()
279            .into(),
280        zoom_factor,
281        viewport_width: view.width() as f32,
282        viewport_height: view.height() as f32,
283        tile_mercator_coords: tile_mercator_coordinates(
284            coords.into_tile(crate::style::source::TileAddressingScheme::XYZ),
285        )
286        .into(),
287        line_width_scale: 1.0,
288        line_units_per_pixel: 8.0 * zoom_factor,
289        clip_antimeridian: u32::from(u8::from(coords.z) == 0),
290        clip_region: crate::render::shaders::WHOLE_TILE,
291        _pad: [0; 2],
292    }
