30pub(super) fn targets(world: &World) -> Vec<WorldTileCoords> {
31    let mut seen = HashSet::default();
32    let mut targets = Vec::new();
33    if let Some(Initialized(pattern)) = world.resources.get::<Eventually<WgpuTileViewPattern>>() {
34        for tile in pattern.iter() {
35            tile.render_kind(TileKind::Vector, |shape| {
36                if seen.insert(shape.coords()) {
37                    targets.push(shape.coords());
38                }
39            });
40        }
41    }
42    targets
43}

45pub(super) fn upload_tiles(world: &World) -> Vec<WorldTileCoords> {
46    let mut tiles = world
47        .resources
48        .get::<SymbolCovering>()
49        .map(|covering| covering.tiles.clone())
50        .unwrap_or_default();
51    let mut seen: HashSet<_> = tiles.iter().copied().collect();
52    for target in targets(world) {
53        if seen.insert(target) {
54            tiles.push(target);
55        }
56        let mut current = target;
57        while let Some(parent) = current.get_parent() {
58            current = parent;
59            if world
60                .tiles
61                .query::<&SymbolLayersDataComponent>(parent)
62                .is_some_and(|symbols| !symbols.pending_assets)
63            {
64                if seen.insert(parent) {
65                    tiles.push(parent);
66                }
67                break;
68            }
69        }
70    }
71    tiles
72}

74fn ready(world: &World, style: &Style, zoom: f64, coords: WorldTileCoords) -> bool {
75    let Some(symbols) = world.tiles.query::<&SymbolLayersDataComponent>(coords) else {
76        return false;
77    };
78    if symbols.pending_assets {
79        return false;
80    }
81    let Some(Initialized(pool)) = world.resources.get::<Eventually<SymbolBufferPool>>() else {
82        return false;
83    };
84    let loaded = pool.get_loaded_style_layers_at(coords).unwrap_or_default();
85    symbols.layers.iter().all(|layer| {
86        let visible = style
87            .layers
88            .iter()
89            .any(|style| style.id == layer.style_layer_id && style.is_visible_at(zoom));
90        !visible
91            || layer.buffer.usable_indices == 0
92            || layer.buffer.buffer.indices.is_empty()
93            || loaded.contains(layer.style_layer_id.as_str())
94    })
95}

97pub(super) fn select(
98    targets: &[WorldTileCoords],
99    ready: impl Fn(WorldTileCoords) -> bool,
100) -> Vec<WorldTileCoords> {
101    let mut selected = HashSet::default();
102    for &target in targets {
103        let mut current = target;
104        loop {
105            if ready(current) {
106                selected.insert(current);
107                break;
108            }
109            let Some(parent) = current.get_parent() else {
110                break;
111            };
112            current = parent;
113        }
114    }
115    // Fine labels keep their placement while a neighbour still needs a parent. Collision
116    // priority resolves duplicate anchors across levels without hiding an entire region.
117    let mut result: Vec<_> = selected.into_iter().collect();
118    result.sort_by_key(|tile| (tile.z, tile.y, tile.x));
119    result
120}

122pub(super) fn update(
123    world: &mut World,
124    style: &Style,
125    view: &ViewState,
126    device: &wgpu::Device,
127    queue: &crate::render::upload_queue::UploadQueue,
128) {
129    let tiles = if crate::render::eye_covering::EyeInFrame::reuses_content(world) {
130        world
131            .resources
132            .get::<SymbolCovering>()
133            .map(|covering| covering.tiles.clone())
134            .unwrap_or_default()
135    } else {
136        select(&targets(world), |coords| {
137            ready(world, style, view.style_zoom().value(), coords)
138        })
139    };
140    if tiles.is_empty() {
141        if let Some(covering) = world.resources.get_mut::<SymbolCovering>() {
142            covering.tiles.clear();
143        }
144        return;
145    }
146    let entries: Vec<_> = tiles.iter().map(|coords| metadata(*coords, view)).collect();
147    let bytes = bytemuck::cast_slice(&entries);
148    let required = (bytes.len() as u64).max(256);
149    let grow = world
150        .resources
151        .get::<SymbolCovering>()
152        .is_none_or(|covering| covering.buffer.size() < required);
153    if grow {
154        if let Some(old) = world.resources.remove::<SymbolCovering>() {
155            queue.cancel_staged_buffer(&old.buffer);
156            old.buffer.destroy();
157        }
158        let buffer = device.create_buffer(&wgpu::BufferDescriptor {
159            label: Some("symbol tile transforms"),
160            size: required.next_power_of_two(),
161            usage: wgpu::BufferUsages::VERTEX | wgpu::BufferUsages::COPY_DST,
162            mapped_at_creation: false,
163        });
164        world.resources.insert(SymbolCovering {
165            tiles: Vec::new(),
166            buffer,
167        });
168    }
169    if let Some(covering) = world.resources.get_mut::<SymbolCovering>() {
170        if !bytes.is_empty() {
171            queue.write_buffer(&covering.buffer, 0, bytes);
172        }
173        covering.tiles = tiles;
174    }
175}

177fn metadata(coords: WorldTileCoords, view: &ViewState) -> ShaderTileMetadata {
178    let zoom = view.zoom();
179    let zoom_factor = view.style_zoom().scale_to_tile(&coords) as f32;
180    ShaderTileMetadata {
181        transform: view
182            .gpu_view_projection()
183            .to_model_view_projection(coords.transform_for_zoom(zoom))
184            .downcast()
185            .into(),
186        zoom_factor,
187        viewport_width: view.width() as f32,
188        viewport_height: view.height() as f32,
189        tile_mercator_coords: tile_mercator_coordinates(
190            coords.into_tile(crate::style::source::TileAddressingScheme::XYZ),
191        )
192        .into(),
193        line_width_scale: 1.0,
194        line_units_per_pixel: 8.0 * zoom_factor,
195        clip_antimeridian: u32::from(u8::from(coords.z) == 0),
196        clip_region: crate::render::shaders::WHOLE_TILE,
197        _pad: [0; 2],
198    }
199}
