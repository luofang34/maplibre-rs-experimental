24pub(super) fn targets(world: &World) -> Vec<WorldTileCoords> {
25    let mut seen = HashSet::new();
26    let mut targets = Vec::new();
27    if let Some(Initialized(pattern)) = world.resources.get::<Eventually<WgpuTileViewPattern>>() {
28        for tile in pattern.iter() {
29            tile.render_kind(TileKind::Vector, |shape| {
30                if seen.insert(shape.coords()) {
31                    targets.push(shape.coords());
32                }
33            });
34        }
35    }
36    targets
37}

39pub(super) fn upload_tiles(world: &World) -> Vec<WorldTileCoords> {
40    let mut tiles = world
41        .resources
42        .get::<SymbolCovering>()
43        .map(|covering| covering.tiles.clone())
44        .unwrap_or_default();
45    let mut seen: HashSet<_> = tiles.iter().copied().collect();
46    for target in targets(world) {
47        if seen.insert(target) {
48            tiles.push(target);
49        }
50        let mut current = target;
51        while let Some(parent) = current.get_parent() {
52            current = parent;
53            if world
54                .tiles
55                .query::<&SymbolLayersDataComponent>(parent)
56                .is_some_and(|symbols| !symbols.pending_assets)
57            {
58                if seen.insert(parent) {
59                    tiles.push(parent);
60                }
61                break;
62            }
63        }
64    }
65    tiles
66}

68fn ready(world: &World, style: &Style, zoom: f64, coords: WorldTileCoords) -> bool {
69    let Some(symbols) = world.tiles.query::<&SymbolLayersDataComponent>(coords) else {
70        return false;
71    };
72    if symbols.pending_assets {
73        return false;
74    }
75    let Some(Initialized(pool)) = world.resources.get::<Eventually<SymbolBufferPool>>() else {
76        return false;
77    };
78    let loaded = pool.get_loaded_style_layers_at(coords).unwrap_or_default();
79    symbols.layers.iter().all(|layer| {
80        let visible = style
81            .layers
82            .iter()
83            .any(|style| style.id == layer.style_layer_id && style.is_visible_at(zoom));
84        !visible
85            || layer.buffer.buffer.indices.is_empty()
86            || loaded.contains(layer.style_layer_id.as_str())
87    })
88}

90pub(super) fn select(
91    targets: &[WorldTileCoords],
92    ready: impl Fn(WorldTileCoords) -> bool,
93) -> Vec<WorldTileCoords> {
94    let mut selected = HashSet::new();
95    for &target in targets {
96        let mut current = target;
97        loop {
98            if ready(current) {
99                selected.insert(current);
100                break;
101            }
102            let Some(parent) = current.get_parent() else {
103                break;
104            };
105            current = parent;
106        }
107    }
108    // Fine labels keep their placement while a neighbour still needs a parent. Collision
109    // priority resolves duplicate anchors across levels without hiding an entire region.
110    let mut result: Vec<_> = selected.into_iter().collect();
111    result.sort_by_key(|tile| (tile.z, tile.y, tile.x));
112    result
113}

115pub(super) fn update(
116    world: &mut World,
117    style: &Style,
118    view: &ViewState,
119    device: &wgpu::Device,
120    queue: &crate::render::upload_queue::UploadQueue,
121) {
122    let tiles = if crate::render::eye_covering::EyeInFrame::reuses_content(world) {
123        world
124            .resources
125            .get::<SymbolCovering>()
126            .map(|covering| covering.tiles.clone())
127            .unwrap_or_default()
128    } else {
129        select(&targets(world), |coords| {
130            ready(world, style, view.style_zoom().value(), coords)
131        })
132    };
133    let entries: Vec<_> = tiles.iter().map(|coords| metadata(*coords, view)).collect();
134    let bytes = bytemuck::cast_slice(&entries);
135    let required = (bytes.len() as u64).max(256);
136    let grow = world
137        .resources
138        .get::<SymbolCovering>()
139        .is_none_or(|covering| covering.buffer.size() < required);
140    if grow {
141        let buffer = device.create_buffer(&wgpu::BufferDescriptor {
142            label: Some("symbol tile transforms"),
143            size: required.next_power_of_two(),
144            usage: wgpu::BufferUsages::VERTEX | wgpu::BufferUsages::COPY_DST,
145            mapped_at_creation: false,
146        });
147        world.resources.insert(SymbolCovering {
148            tiles: Vec::new(),
149            buffer,
150        });
151    }
152    if let Some(covering) = world.resources.get_mut::<SymbolCovering>() {
153        queue.write_buffer(&covering.buffer, 0, bytes);
154        covering.tiles = tiles;
155    }
156}

158fn metadata(coords: WorldTileCoords, view: &ViewState) -> ShaderTileMetadata {
159    let zoom = view.zoom();
160    let zoom_factor = view.style_zoom().scale_to_tile(&coords) as f32;
161    ShaderTileMetadata {
162        transform: view
163            .gpu_view_projection()
164            .to_model_view_projection(coords.transform_for_zoom(zoom))
165            .downcast()
166            .into(),
167        zoom_factor,
168        viewport_width: view.width() as f32,
169        viewport_height: view.height() as f32,
170        tile_mercator_coords: tile_mercator_coordinates(
171            coords.into_tile(crate::style::source::TileAddressingScheme::XYZ),
172        )
173        .into(),
174        line_width_scale: 1.0,
175        line_units_per_pixel: 8.0 * zoom_factor,
176        clip_antimeridian: u32::from(u8::from(coords.z) == 0),
177    }
178}
