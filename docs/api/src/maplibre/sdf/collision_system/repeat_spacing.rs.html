1//! Keeps the repeats of a label along its lines apart on screen, across tiles.
2//!
3//! `symbol-spacing` spaces a label's repeats along a line in the tile's own pixels, so a tilted
4//! or narrow view that shows a tile magnified near its far edge, or many tiles of one road,
5//! packs the same road shield every hundred pixels or so on screen. This rule measures the
6//! repeats where they are drawn instead.
7use std::{collections::HashMap, sync::Arc};
8
9use crate::sdf::{Feature, SymbolLayerData};
10
11/// The least distance on screen, in logical pixels, between two repeats of one label in one
12/// style layer.
13///
14/// It is the default `symbol-spacing`: the gap a style gets between repeats when a tile is
15/// seen flat at its own zoom, which this keeps at any zoom, tilt or field of view. Collision
16/// boxes are measured in logical pixels, the units labels are sized in, so the gap scales
17/// with the labels on a display of any pixel ratio.
18pub(super) const MIN_REPEAT_DISTANCE: f64 = 250.0;
19
20type Positions = HashMap<String, HashMap<Arc<str>, Vec<[f64; 2]>>>;
21
22/// Where the repeats of each label were placed on screen, by style layer and repeat key, in
23/// this placement and the one before.
24#[derive(Default)]
25pub(super) struct RepeatSpacing {
26    placed: Positions,
27    previous: Positions,
28}
29
30/// A label that repeats along lines, and the centre of its box on screen.
31pub(super) struct Repeat<'a> {
32    layer: &'a str,
33    key: &'a Arc<str>,
34    at: [f64; 2],
35}
36
37impl<'a> Repeat<'a> {
38    /// The repeat a label would place with `rectangles`, its text and icon boxes on screen;
39    /// `None` for a label that does not repeat or whose box is not centred in `viewport`.
40    ///
41    /// A label just past the edge of the view is placed so that it can slide in, but nobody
42    /// sees it there, so it neither keeps a repeat in view from its place nor is kept from its
43    /// own.
44    pub(super) fn of(
45        layer: &'a SymbolLayerData,
46        feature: &'a Feature,
47        rectangles: &[Option<[f64; 4]>; 2],
48        viewport: [f64; 2],
49    ) -> Option<Self> {
50        let key = feature.repeat_key.as_ref()?;
51        let [left, top, right, bottom] = rectangles[0].or(rectangles[1])?;
52        let at = [(left + right) * 0.5, (top + bottom) * 0.5];
53        let in_view = (0.0..=viewport[0]).contains(&at[0]) && (0.0..=viewport[1]).contains(&at[1]);
54        in_view.then_some(Self {
55            layer: &layer.style_layer_id,
56            key,
57            at,
58        })
59    }
60}
61
62impl RepeatSpacing {
63    /// Starts a placement; the one that ends becomes the previous.
64    pub(super) fn begin(&mut self) {
65        self.previous = std::mem::take(&mut self.placed);
66    }
67
68    /// Whether `repeat` would sit closer than [`MIN_REPEAT_DISTANCE`] to a repeat of its label
69    /// placed earlier in this placement or, unless it was shown itself, in the previous one.
70    ///
71    /// Placement goes in a fixed order, so the first of two close repeats keeps its place
72    /// while the view holds still. A repeat that was not shown also yields to those shown
73    /// before, wherever they come in the order, so a repeat that a newly loaded tile brings,
74    /// or that the moving view brings near, does not take the place of one already shown.
75    pub(super) fn too_close(&self, repeat: &Repeat<'_>, was_visible: bool) -> bool {
76        let near = |positions: &Positions| {
77            positions
78                .get(repeat.layer)
79                .and_then(|keys| keys.get(repeat.key))
80                .is_some_and(|placed| {
81                    placed.iter().any(|other| {
82                        (other[0] - repeat.at[0]).hypot(other[1] - repeat.at[1])
83                            < MIN_REPEAT_DISTANCE
84                    })
85                })
86        };
87        near(&self.placed) || (!was_visible && near(&self.previous))
88    }
89
90    /// Records that `repeat` took its place.
91    pub(super) fn remember(&mut self, repeat: Repeat<'_>) {
92        self.placed
93            .entry(repeat.layer.to_owned())
94            .or_default()
95            .entry(repeat.key.clone())
96            .or_default()
97            .push(repeat.at);
98    }
99}
