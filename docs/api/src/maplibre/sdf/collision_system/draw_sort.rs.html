1//! Labels that may overlap are drawn in the order GL JS sorts them for the current bearing.
2//!
3//! Tessellation orders a layer's index buffer by unrotated height. GL JS sorts the labels again
4//! by their height on the rotated screen whenever the bearing changes, so the lower of two
5//! overlapping labels covers the other however the map is turned; the index buffer of each such
6//! layer is rewritten here when the bearing it was sorted for changes.
7
8use std::collections::HashMap;
9
10use crate::{
11    coords::WorldTileCoords,
12    render::eventually::{Eventually, Eventually::Initialized},
13    sdf::{
14        tessellation::{drawn_order, sorts_by_height},
15        SymbolBufferPool, SymbolLayerData,
16    },
17    style::layer::SymbolPaint,
18};
19
20/// Discrete angular steps around `2π` (`96` steps = `3.75°` per bucket, hitting `0°`, `45°`,
21/// `90°`, `180°`, and `270°` exactly while suppressing sub-degree orbit index-buffer rewrites).
22const BEARING_BUCKETS: u16 = 96;
23
24fn bearing_bucket(bearing: f64) -> u16 {
25    if !bearing.is_finite() {
26        return 0;
27    }
28    let normalized = bearing.rem_euclid(std::f64::consts::TAU);
29    ((normalized / std::f64::consts::TAU) * f64::from(BEARING_BUCKETS)).round() as u16
30        % BEARING_BUCKETS
31}
32
33fn bucket_angle_rad(bucket: u16) -> f64 {
34    (f64::from(bucket) / f64::from(BEARING_BUCKETS)) * std::f64::consts::TAU
35}
36
37/// The quantized bearing bucket each layer's indices were last sorted for, by tile and allocation.
38#[derive(Default)]
39pub(super) struct DrawSort {
40    sorted: HashMap<(WorldTileCoords, String), (u64, u16)>,
41}
42
43impl DrawSort {
44    /// Returns `true` when any height-sorted symbol bucket in `layers` has not yet been ordered
45    /// for `bearing`'s discrete angular bucket.
46    pub(super) fn needs_bearing_sort(
47        &self,
48        world: &crate::tcs::world::World,
49        (bearing, zoom): (f64, f64),
50        layers: &[(u32, &SymbolLayerData, &SymbolPaint)],
51    ) -> bool {
52        let Some(Initialized(pool)) = world.resources.get::<Eventually<SymbolBufferPool>>() else {
53            return false;
54        };
55        let bucket = bearing_bucket(bearing);
56        for (_, layer, paint) in layers {
57            if !sorts_by_height(paint, zoom) {
58                continue;
59            }
60            let Some(entry) = pool.index().get_layers(layer.coords).and_then(|entries| {
61                entries
62                    .iter()
63                    .find(|entry| entry.style_layer.id == layer.style_layer_id)
64            }) else {
65                continue;
66            };
67            let sorted_for = self
68                .sorted
69                .get(&(layer.coords, layer.style_layer_id.clone()))
70                .filter(|(allocation, _)| *allocation == entry.allocation_id())
71                .map_or(0_u16, |(_, b)| *b);
72            if sorted_for != bucket {
73                return true;
74            }
75        }
76        false
77    }
78
79    /// Rewrites the indices of the `layers` that sort by height and were sorted for another
80    /// bearing bucket; an upload that has not been sorted yet holds the unrotated (`0`) order.
81    pub(super) fn follow_bearing(
82        &mut self,
83        world: &crate::tcs::world::World,
84        queue: &crate::render::upload_queue::UploadQueue,
85        (bearing, zoom): (f64, f64),
86        layers: &[(u32, &SymbolLayerData, &SymbolPaint)],
87    ) {
88        let Some(Initialized(pool)) = world.resources.get::<Eventually<SymbolBufferPool>>() else {
89            return;
90        };
91        let bucket = bearing_bucket(bearing);
92        let sort_bearing = bucket_angle_rad(bucket);
93        // Uploads that left the pool take their entries with them.
94        self.sorted
95            .retain(|(coords, _), _| pool.index().get_layers(*coords).is_some());
96        for (_, layer, paint) in layers {
97            if !sorts_by_height(paint, zoom) {
98                continue;
99            }
100            let Some(entry) = pool.index().get_layers(layer.coords).and_then(|entries| {
101                entries
102                    .iter()
103                    .find(|entry| entry.style_layer.id == layer.style_layer_id)
104            }) else {
105                continue;
106            };
107            let key = (layer.coords, layer.style_layer_id.clone());
108            let sorted_for = self
109                .sorted
110                .get(&key)
111                .filter(|(allocation, _)| *allocation == entry.allocation_id())
112                .map_or(0_u16, |(_, b)| *b);
113            if sorted_for != bucket {
114                if let Some(indices) = indices_in_order(layer, sort_bearing) {
115                    pool.update_indices(queue, entry, &indices);
116                }
117            }
118            self.sorted.insert(key, (entry.allocation_id(), bucket));
119        }
120    }
121}

123/// The layer's indices with each label's run in drawn order; `None` when the runs do not
124/// cover the buffer as tessellation leaves them.
125fn indices_in_order(layer: &SymbolLayerData, bearing: f64) -> Option<Vec<u32>> {
126    let source = &layer.buffer.buffer.indices;
127    let covered: usize = layer
128        .features
129        .iter()
130        .map(|feature| feature.indices.len())
131        .sum();
132    if covered != source.len() {
133        return None;
134    }
135    let mut indices = Vec::with_capacity(source.len());
136    for index in drawn_order(&layer.features, bearing) {
137        indices.extend_from_slice(source.get(layer.features[index].indices.clone())?);
138    }
139    Some(indices)
140}
