1//! Dirty sub-range diffing and coalesced GPU uploads for `SDFShaderFeatureMetadata`.
2//!
3//! Newly allocated symbol buckets are zero-initialized in GPU memory (`SDFShaderFeatureMetadata::default()`).
4//! During a `PlacementEpoch`, most symbols in a tile either stay clean or remain hidden, so
5//! diffing per-vertex metadata and uploading only coalesced dirty runs eliminates ~90% of
6//! `queue.writeBuffer` bandwidth compared to rewriting entire bucket allocations.
7
8use bytemuck::Pod;
9
10use crate::{
11    render::{resource::Queue, shaders::SDFShaderFeatureMetadata},
12    vector::resource::{BufferPool, IndexEntry},
13};
14
15/// Clean vertices tolerated between two dirty runs before splitting into two `writeBuffer` calls.
16/// `32 * 80 B = 2,560 B`, balancing fewer WebGPU IPC calls against minimal clean-byte over-upload.
17const MAX_CLEAN_GAP_VERTICES: usize = 32;
18
19/// Reusable pool of per-bucket `SDFShaderFeatureMetadata` vectors and zeroed staging slices so
20/// placement epochs do not churn multi-hundred-kilobyte heap allocations on WASM or native.
21#[derive(Default)]
22pub(super) struct MetadataScratchPool {
23    free: Vec<Vec<SDFShaderFeatureMetadata>>,
24    zeros: Vec<SDFShaderFeatureMetadata>,
25}
26
27impl MetadataScratchPool {
28    /// Pops a cleared `Vec<SDFShaderFeatureMetadata>` with retained capacity when available.
29    pub(super) fn acquire(&mut self) -> Vec<SDFShaderFeatureMetadata> {
30        let mut buf = self.free.pop().unwrap_or_default();
31        buf.clear();
32        buf
33    }
34
35    /// Returns a `Vec<SDFShaderFeatureMetadata>` to the scratch pool if capacity is worth keeping.
36    pub(super) fn recycle(&mut self, mut buf: Vec<SDFShaderFeatureMetadata>) {
37        if buf.capacity() > 0 && buf.capacity() <= 65_536 && self.free.len() < 16 {
38            buf.clear();
39            self.free.push(buf);
40        }
41    }
42
43    fn zero_slice(&mut self, len: usize) -> &[SDFShaderFeatureMetadata] {
44        if self.zeros.len() < len {
45            self.zeros.resize(len, SDFShaderFeatureMetadata::default());
46        }
47        &self.zeros[..len]
48    }
49}

51/// Last GPU-resident `SDFShaderFeatureMetadata` state for one `(WorldTileCoords, style_layer_id)` bucket.
52#[derive(Clone, Debug, Default)]
53pub(super) struct CachedBucketMetadata {
54    /// `IndexEntry::allocation_id()` of the backing GPU buffer slice.
55    pub allocation_id: u64,
56    /// Whether at least one vertex in `vertices` has target `opacity > 0.0`.
57    pub has_target_visible: bool,
58    /// Latest `start_time_ms` among vertices with `start_opacity > 0.0`, or `-1.0` when none.
59    pub latest_fade_start_ms: f32,
60    /// Empty when all GPU vertices in this allocation are `SDFShaderFeatureMetadata::default()`.
61    pub vertices: Vec<SDFShaderFeatureMetadata>,
62}
63
64impl CachedBucketMetadata {
65    fn empty(allocation_id: u64) -> Self {
66        Self {
67            allocation_id,
68            has_target_visible: false,
69            latest_fade_start_ms: -1.0,
70            vertices: Vec::new(),
71        }
72    }
73
74    /// Whether any vertex in this bucket evaluates to non-zero opacity at `now_ms`.
75    pub(super) fn is_active_at(&self, now_ms: f32, fade_duration_ms: f32) -> bool {
76        self.has_target_visible
77            || (fade_duration_ms > 0.0
78                && self.latest_fade_start_ms >= 0.0
79                && now_ms < self.latest_fade_start_ms + fade_duration_ms)
80    }
81}

83/// Diffs `next` against `prev` (treating a fresh `allocation_id` as all-zero in GPU memory) and
84/// writes only coalesced dirty vertex ranges to `pool`.
85#[cfg(test)]
86pub(super) fn diff_and_upload_bucket<
87    Q: Queue<B>,
88    B,
89    V: crate::render::shaders::PoolVertex,
90    I: Pod,
91    TM: Pod,
92>(
93    pool: &BufferPool<Q, B, V, I, TM, SDFShaderFeatureMetadata>,
94    queue: &Q,
95    entry: &IndexEntry,
96    prev: Option<&CachedBucketMetadata>,
97    next: Vec<SDFShaderFeatureMetadata>,
98) -> CachedBucketMetadata {
99    let mut scratch = MetadataScratchPool::default();
100    diff_and_upload_bucket_with_scratch(pool, queue, entry, prev, next, &mut scratch)
101}

103/// Diffs `next` against `prev` and recycles unused `SDFShaderFeatureMetadata` buffers into `scratch`.
104pub(super) fn diff_and_upload_bucket_with_scratch<
105    Q: Queue<B>,
106    B,
107    V: crate::render::shaders::PoolVertex,
108    I: Pod,
109    TM: Pod,
110>(
111    pool: &BufferPool<Q, B, V, I, TM, SDFShaderFeatureMetadata>,
112    queue: &Q,
113    entry: &IndexEntry,
114    prev: Option<&CachedBucketMetadata>,
115    next: Vec<SDFShaderFeatureMetadata>,
116    scratch: &mut MetadataScratchPool,
117) -> CachedBucketMetadata {
118    let allocation_id = entry.allocation_id();
119    let stride = size_of::<SDFShaderFeatureMetadata>() as u64;
120    let range = entry.feature_metadata_buffer_range();
121    let allocated_vertices = ((range.end - range.start) / stride) as usize;
122    if allocated_vertices == 0 {
123        scratch.recycle(next);
124        return CachedBucketMetadata::empty(allocation_id);
125    }
126
127    let prev_slice = prev
128        .filter(|cached| cached.allocation_id == allocation_id)
129        .map_or(&[][..], |cached| cached.vertices.as_slice());
130
131    // Fast path 1: all symbols hidden on a newly zeroed allocation or already-zeroed bucket.
132    if next.is_empty() && prev_slice.is_empty() {
133        scratch.recycle(next);
134        return CachedBucketMetadata::empty(allocation_id);
135    }
136
137    // Fast path 2: bucket became completely hidden after having visible symbols.
138    if next.is_empty() {
139        scratch.recycle(next);
140        let zero = SDFShaderFeatureMetadata::default();
141        let zero_bytes = bytemuck::bytes_of(&zero);
142        let limit = prev_slice.len().min(allocated_vertices);
143        for (start, end) in dirty_runs(limit, |i| bytemuck::bytes_of(&prev_slice[i]) != zero_bytes)
144        {
145            let zeros = scratch.zero_slice(end - start);
146            pool.update_feature_metadata_range(queue, entry, start, zeros);
147        }
148        return CachedBucketMetadata::empty(allocation_id);
149    }
150
151    let mut next = next;
152    if next.len() > allocated_vertices {
153        next.truncate(allocated_vertices);
154    }
155    let zero = SDFShaderFeatureMetadata::default();
156    let limit = next.len();
157    for (start, end) in dirty_runs(limit, |i| {
158        let old = prev_slice.get(i).unwrap_or(&zero);
159        bytemuck::bytes_of(old) != bytemuck::bytes_of(&next[i])
160    }) {
161        pool.update_feature_metadata_range(queue, entry, start, &next[start..end]);
162    }
163    if prev_slice.len() > limit {
164        let tail_end = prev_slice.len().min(allocated_vertices);
165        let zero_bytes = bytemuck::bytes_of(&zero);
166        for (rel_start, rel_end) in dirty_runs(tail_end - limit, |i| {
167            bytemuck::bytes_of(&prev_slice[limit + i]) != zero_bytes
168        }) {
169            let zeros = scratch.zero_slice(rel_end - rel_start);
170            pool.update_feature_metadata_range(queue, entry, limit + rel_start, zeros);
171        }
172    }
173
174    let mut has_target_visible = false;
175    let mut latest_fade_start_ms = -1.0_f32;
176    for vertex in &next {
177        if vertex.has_target_opacity() {
178            has_target_visible = true;
179        }
180        if vertex.has_start_opacity() {
181            latest_fade_start_ms = latest_fade_start_ms.max(vertex.start_time_ms());
182        }
183    }
184
185    CachedBucketMetadata {
186        allocation_id,
187        has_target_visible,
188        latest_fade_start_ms,
189        vertices: next,
190    }
191}

193fn dirty_runs(len: usize, mut is_dirty: impl FnMut(usize) -> bool) -> Vec<(usize, usize)> {
194    let mut runs: Vec<(usize, usize)> = Vec::new();
195    let mut i = 0;
196    while i < len {
197        if !is_dirty(i) {
198            i += 1;
199            continue;
200        }
201        let start = i;
202        i += 1;
203        while i < len && is_dirty(i) {
204            i += 1;
205        }
206        let end = i;
207        if let Some(last) = runs.last_mut() {
208            if start - last.1 <= MAX_CLEAN_GAP_VERTICES {
209                last.1 = end;
210                continue;
211            }
212        }
213        runs.push((start, end));
214    }
215    runs
216}

244    fn test_pool_with_bucket(
245        vertex_count: usize,
246    ) -> (
247        BufferPool<
248            RecordingQueue,
249            u32,
250            ShaderSymbolVertex,
251            u32,
252            ShaderLayerMetadata,
253            SDFShaderFeatureMetadata,
254        >,
255        RecordingQueue,
256        WorldTileCoords,
257    ) {
258        let mut pool = BufferPool::new(
259            BackingBufferDescriptor::new(1_u32, 64 * 1024),
260            BackingBufferDescriptor::new(2_u32, 64 * 1024),
261            BackingBufferDescriptor::new(3_u32, 4 * 1024),
262            BackingBufferDescriptor::new(4_u32, 64 * 1024),
263        );
264        let queue = RecordingQueue::default();
265        let coords = WorldTileCoords {
266            x: 0,
267            y: 0,
268            z: ZoomLevel::from(10),
269        };
270        let mut buffer = OverAlignedVertexBuffer::empty();
271        buffer.buffer.vertices = vec![ShaderSymbolVertex::default(); vertex_count];
272        buffer.buffer.indices = vec![0, 1, 2, 0, 2, 3];
273        buffer.usable_indices = 6;

291        (pool, queue, coords)
292    }

294    #[test]
295    fn initial_sparse_placement_uploads_only_placed_quad_subrange() {
296        let (pool, queue, coords) = test_pool_with_bucket(200);
297        let entry = &pool.index().get_layers(coords).expect("layers")[0];
298        let stride = size_of::<SDFShaderFeatureMetadata>();
299
300        // Only 1 quad (vertices 100..104) is placed out of 200 vertices.
301        let mut next = vec![SDFShaderFeatureMetadata::default(); 200];
302        for v in &mut next[100..104] {
303            v.set_target_opacity(1.0);
304        }
305        let cached = diff_and_upload_bucket(&pool, &queue, entry, None, next.clone());
306        assert_eq!(
307            *queue.writes.borrow(),
308            vec![((100 * stride) as u64, 4 * stride)]
309        );
310
311        // Identical follow-up epoch issues 0 writes.
312        queue.writes.borrow_mut().clear();
313        let cached = diff_and_upload_bucket(&pool, &queue, entry, Some(&cached), next.clone());
314        assert!(queue.writes.borrow().is_empty());
315
316        // Hiding that single quad zeroes only those 4 vertices.
317        let cached = diff_and_upload_bucket(&pool, &queue, entry, Some(&cached), Vec::new());
318        assert_eq!(
319            *queue.writes.borrow(),
320            vec![((100 * stride) as u64, 4 * stride)]
321        );
322        assert!(cached.vertices.is_empty());
323    }

325    #[test]
326    fn nearby_dirty_quads_coalesce_while_distant_quads_split() {
327        let (pool, queue, coords) = test_pool_with_bucket(200);
328        let entry = &pool.index().get_layers(coords).expect("layers")[0];
329        let stride = size_of::<SDFShaderFeatureMetadata>();
330
331        let mut next = vec![SDFShaderFeatureMetadata::default(); 200];
332        // Quads at 10..14 and 20..24 are separated by 6 clean vertices (<= 32) -> coalesced!
333        // Quad at 120..124 is 96 clean vertices away (> 32) -> separate sub-range!
334        for idx in (10..14).chain(20..24).chain(120..124) {
335            next[idx].set_target_opacity(1.0);
336        }
337        let _ = diff_and_upload_bucket(&pool, &queue, entry, None, next);
338        assert_eq!(
339            *queue.writes.borrow(),
340            vec![
341                ((10 * stride) as u64, 14 * stride),
342                ((120 * stride) as u64, 4 * stride),
343            ]
344        );
345    }
