24/// What one frame's placement of a layer shares between its features.
25struct LayerFrame<'a> {
26    world: &'a crate::tcs::world::World,
27    view_state: &'a crate::render::view_state::ViewState,
28    projection: &'a crate::render::projection::ShaderProjectionData,
29    paint: &'a crate::style::layer::SymbolPaint,
30    uniforms: SymbolUniforms,
31    zoom_limits: [f64; 2],
32    glyphs_stay_upright: bool,
33    has_data_driven_sizes: bool,
34    has_terrain_or_height: bool,
35    has_icon_text_fit: bool,
36    constant_rules: Option<rules::PlacementRules>,
37    visibility_policy: Option<&'a crate::sdf::visibility::SymbolVisibility>,
38}

40/// The GPU-visible outcome of placing one feature.
41struct FeaturePlacement {
42    /// Where the text sits relative to its laid-out anchor, in layout pixels, when a later
43    /// `text-variable-anchor` was the first to fit.
44    text_shift: [f32; 2],
45    /// The `text-variable-anchor` entry the text took.
46    anchor: usize,
47    opacity: [f32; 2],
48    ground: f32,
49    line: Option<LinePoses>,
50    /// Whether the label found a place this frame, which its fade may not show yet.
51    placed: bool,
52}

54/// Places the features of one style layer across every visible tile together, in ascending
55/// `symbol-sort-key` order, and returns each tile's metadata and placement in the order of
56/// `layers`.
57pub(super) fn place_layer(
58    world: &crate::tcs::world::World,
59    view_state: &crate::render::view_state::ViewState,
60    projection: &crate::render::projection::ShaderProjectionData,
61    layers: &[&crate::sdf::SymbolLayerData],
62    paint: &crate::style::layer::SymbolPaint,
63    zoom_limits: [f64; 2],
64    placement: (
65        &mut CollisionGrid,
66        &mut PlacedSymbols,
67        &mut temporal::PlacementHistory,
68    ),
69) -> (Vec<Vec<SDFShaderFeatureMetadata>>, super::family::Outcomes) {
70    let (boxes, placed, history) = placement;
71    let mut metadata: Vec<_> = layers.iter().map(|layer| empty_metadata(layer)).collect();
72    let mut outcomes: super::family::Outcomes = (0..layers.len()).map(|_| Vec::new()).collect();
73    let zoom = view_state.style_zoom().value();
74    let has_data_driven_sizes = paint
75        .text_size
76        .as_ref()
77        .is_some_and(|s| !s.is_feature_constant())
78        || paint
79            .compiled_property::<f32>("icon-size")
80            .is_some_and(|s| !s.is_feature_constant());
81    let has_terrain_or_height = world
82        .resources
83        .get::<crate::terrain::coverage::TerrainCoverageIndex>()
84        .is_some()
85        || paint.uses_shared_height();
86    let visibility_policy = world
87        .resources
88        .get::<crate::sdf::visibility::SymbolVisibility>();
89    let frame = LayerFrame {
90        world,
91        view_state,
92        projection,
93        paint,
94        uniforms: SymbolUniforms::new(paint, zoom, [1, 1]),
95        zoom_limits,
96        glyphs_stay_upright: crate::sdf::paint::glyphs_stay_upright(paint, zoom),
97        has_data_driven_sizes,
98        has_terrain_or_height,
99        has_icon_text_fit: paint.properties.contains_key("icon-text-fit"),
100        constant_rules: rules::PlacementRules::constant_for_layer(paint, zoom),
101        visibility_policy,
102    };
103    for (position, feature_index, feature, was_visible) in ordered_features(layers, history) {
104        // A second way of writing a label is placed with the first, and only when that finds
105        // no place.
106        if feature.fallback {
107            continue;
108        }
109        let layer = layers[position];
110        let mut next = Some((feature_index, feature, was_visible, false));
111        while let Some((index, feature, was_visible, suppressed)) = next.take() {
112            let frame = LayerFrame {
113                uniforms: if frame.has_data_driven_sizes {
114                    frame
115                        .uniforms
116                        .with_feature_sizes(paint, &feature.data.properties, zoom)
117                } else {
118                    frame.uniforms
119                },
120                ..frame
121            };
122            let outcome = frame.place_feature(
123                (layer, feature, index, was_visible, suppressed),
124                (&mut *boxes, &mut *placed, &mut *history),
125            );
126            let poses = match &outcome.line {
127                Some(LinePoses::Poses(poses)) => Some(poses.clone()),
128                _ => None,
129            };
130            write_feature_metadata(
131                layer,
132                feature,
133                (outcome.opacity, outcome.text_shift, outcome.anchor),
134                outcome.ground,
135                poses.as_deref(),
136                (
137                    paint,
138                    view_state.style_zoom().value(),
139                    &mut metadata[position],
140                ),
141            );
142            if outcome.opacity != [0.0, 0.0] {
143                if outcomes[position].is_empty() {
144                    outcomes[position].resize_with(layer.features.len(), || None);
145                }
146                if let Some(slot) = outcomes[position].get_mut(index) {
147                    *slot = Some(super::family::Outcome {
148                        opacity: outcome.opacity,
149                        text_shift: outcome.text_shift,
150                        anchor: outcome.anchor,
151                        ground: outcome.ground,
152                        poses,
153                    });
154                }
155            }
156            next = layer
157                .features
158                .get(index + 1)
159                .filter(|other| other.fallback)
160                .map(|other| {
161                    (
162                        index + 1,
163                        other,
164                        history.was_visible(layer, other),
165                        suppressed || outcome.placed,
166                    )
167                });
168        }
169    }
170    (metadata, outcomes)
171}

173impl LayerFrame<'_> {
174    /// Zoom range a label may show in; one that was visible keeps a margin so it does not
175    /// blink at the edge.
176    fn limits_for(&self, was_visible: bool) -> [f64; 2] {
177        let margin = if was_visible { 0.15 } else { 0.0 };
178        [self.zoom_limits[0] - margin, self.zoom_limits[1] + margin]
179    }
180
181    /// The rectangles with the text moved to the first anchor of `text-variable-anchor` that
182    /// fits, and that move in layout pixels; the first anchor when none does.
183    fn place_feature(
184        &self,
185        (layer, feature, feature_index, was_visible, suppressed): (
186            &crate::sdf::SymbolLayerData,
187            &crate::sdf::Feature,
188            usize,
189            bool,
190            bool,
191        ),
192        (boxes, placed, history): (
193            &mut CollisionGrid,
194            &mut PlacedSymbols,
195            &mut temporal::PlacementHistory,
196        ),
197    ) -> FeaturePlacement {
198        let (view_state, projection) = (self.view_state, self.projection);
199        let zoom = view_state.style_zoom().value();
200        let label_ground = if self.has_terrain_or_height {
201            LabelGround::new(self.world, layer, feature, self.paint, zoom)
202        } else {
203            LabelGround::flat(self.world, layer.coords)
204        };
205        let ground = label_ground.elevation;
206        let relevance = self.visibility_policy.map_or(1.0, |policy| {
207            policy.opacity(layer, feature, ground, view_state)
208        });
209        let limits = self.limits_for(was_visible);
210        let anchor_clip = (!suppressed && relevance > 0.0)
211            .then(|| {
212                crate::sdf::placement::project(
213                    layer.coords,
214                    [
215                        f64::from(feature.text_anchor.x),
216                        f64::from(feature.text_anchor.y),
217                    ],
218                    f64::from(ground),
219                    view_state,
220                    projection,
221                )
222            })
223            .flatten();
224        let pre_shown = !suppressed
225            && relevance > 0.0
226            && (feature.line.is_some()
227                || self.uniforms.text_layout[0] > 0.5
228                || self.uniforms.icon_layout[0] > 0.5
229                || anchor_clip.is_some_and(|clip| {
230                    if clip.w <= 0.0 {
231                        return false;
232                    }
233                    let sx = (clip.x / clip.w + 1.0) * 0.5 * view_state.width();
234                    let sy = (1.0 - clip.y / clip.w) * 0.5 * view_state.height();
235                    sx >= -500.0
236                        && sx <= view_state.width() + 500.0
237                        && sy >= -500.0
238                        && sy <= view_state.height() + 500.0
239                }))
240            && local_zoom_visible_from_clip(anchor_clip, view_state, limits)
241            && !beyond_perspective_cutoff_with_clip(
242                layer,
243                feature,
244                ground,
245                view_state,
246                projection,
247                &self.uniforms,
248                anchor_clip.map(|clip| (f64::from(ground), clip)),
249            )
250            && (!self.has_terrain_or_height
251                || !crate::sdf::placement::buried_in_terrain(self.world, layer, feature, ground));
252        let line = if pre_shown || was_visible {
253            feature.line.as_ref().map(|_| {
254                line_glyph_poses(
255                    layer,
256                    feature,
257                    &label_ground,
258                    view_state,
259                    projection,
260                    (&self.uniforms, self.glyphs_stay_upright),
261                )
262            })
263        } else {
264            None
265        };
266        let shown = pre_shown && !matches!(line, Some(LinePoses::DoesNotFit));
267        let (rectangles, glyph_boxes) = if shown {
268            label_boxes(
269                (layer, feature, ground),
270                (&line, self.glyphs_stay_upright),
271                (view_state, projection, &self.uniforms),
272                anchor_clip.map(|clip| (f64::from(ground), clip)),
273            )
274        } else {
275            ([None, None], Vec::new())
276        };
277        let (rectangles, glyph_boxes, text_shift, anchor, visible, duplicate) = if rectangles
278            == [None, None]
279        {
280            ([None, None], Vec::new(), [0.0; 2], 0, [false, false], false)
281        } else {
282            let rules = self.constant_rules.unwrap_or_else(|| {
283                rules::PlacementRules::new(self.paint, &feature.data.properties, zoom)
284            });
285            let viewport = [view_state.width(), view_state.height()];
286            let (rectangles, text_shift, anchor) =
287                if let (Some(text), true) = (rectangles[0], feature.anchor_shifts.len() > 1) {
288                    if self.uniforms.text_layout[0] <= 0.5
289                        && (text[2] < -400.0
290                            || text[3] < -400.0
291                            || text[0] > viewport[0] + 400.0
292                            || text[1] > viewport[1] + 400.0)
293                    {
294                        (rectangles, [0.0; 2], 0)
295                    } else {
296                        let clip_with_ground = anchor_clip.map(|clip| (f64::from(ground), clip));
297                        self.first_fitting_anchor(
298                            (
299                                layer,
300                                feature,
301                                ground,
302                                history.previous_anchor(layer, feature),
303                            ),
304                            (rectangles, &glyph_boxes),
305                            (&rules, &*boxes, viewport),
306                            (
307                                text_perspective_scale(
308                                    layer,
309                                    feature,
310                                    ground,
311                                    view_state,
312                                    projection,
313                                    &self.uniforms,
314                                    clip_with_ground,
315                                ),
316                                text_rotation(
317                                    layer,
318                                    feature,
319                                    ground,
320                                    view_state,
321                                    projection,
322                                    &self.uniforms,
323                                    clip_with_ground,
324                                ),
325                            ),
326                        )
327                    }
328                } else {
329                    (rectangles, [0.0; 2], 0)
330                };
331            let repeat = Repeat::of(layer, feature, &rectangles, viewport);
332            // A repeat too close to one already placed takes no room from labels after it.
333            let (rectangles, glyph_boxes) = if repeat
334                .as_ref()
335                .is_some_and(|repeat| history.repeats.too_close(repeat, was_visible))
336            {
337                ([None, None], Vec::new())
338            } else {
339                (rectangles, glyph_boxes)
340            };
341            let visible = rules.place_along_line(rectangles, &glyph_boxes, boxes, viewport);
342            let any_visible = visible[0] || visible[1];
343            let duplicate = any_visible && history.shown_elsewhere(layer, feature);
344            if let Some(repeat) = repeat.filter(|_| any_visible && !duplicate) {
345                history.repeats.remember(repeat);
346            }
347            (
348                rectangles,
349                glyph_boxes,
350                text_shift,
351                anchor,
352                visible,
353                duplicate,
354            )
355        };
356        let opacity = history.opacity(layer, feature, visible);
357        if visible[0] || was_visible {
358            history.remember_anchor(layer, feature, visible[0].then_some(anchor));
359        }
360        // A placed label is found from its first frame, while it starts fading in, as GL JS
361        // queries its collision index; a copy drawn by another tile is found once.
362        if visible.iter().any(|v| *v) && !duplicate {
363            placed.0.push(PlacedSymbol {
364                coords: layer.coords,
365                layer: layer.style_layer_id.clone(),
366                feature: feature_index,
367                rectangles: [0, 1].map(|i| if visible[i] { rectangles[i] } else { None }),
368                glyph_boxes: if visible[0] { glyph_boxes } else { Vec::new() },
369            });
370        }
371        FeaturePlacement {
372            placed: visible.iter().any(|v| *v),
373            text_shift,
374            anchor,
375            opacity: {
376                let mut opacity = opacity.map(|value| value * relevance);
377                // Text that follows its line has no layout off it: while the line cannot hold
378                // it, its glyphs are hidden at once, as GL JS hides them, rather than fading out
379                // in the straight layout they were cut from.
380                if matches!(line, Some(LinePoses::DoesNotFit)) {
381                    opacity[0] = 0.0;
382                }
383                opacity
384            },
385            ground,
386            line,
387        }
388    }
389}

409        ShaderSymbolVertex {
410            a_pos_offset: [anchor[0], anchor[1], 0, 0],

416    #[test]
417    fn each_glyph_writes_its_own_pose_to_its_six_indices() {
418        // Two glyph quads after one icon quad: vertices 4..8 and 8..12.
419        let mut buffer = OverAlignedVertexBuffer::empty();
420        buffer.buffer.vertices = (0..12).map(|_| vertex([100, 200])).collect();
421        for quad in 0..3_u32 {
422            let base = quad * 4;
423            buffer
424                .buffer
425                .indices
426                .extend([base, base + 1, base + 2, base, base + 2, base + 3]);
427        }
428        let layer = crate::sdf::SymbolLayerData {
429            atlas: None,
430            coords: crate::coords::WorldTileCoords {
431                x: 0,
432                y: 0,
433                z: ZoomLevel::from(0),
434            },
435            source_layer: "roads".into(),
436            style_layer_id: "label".into(),
437            buffer,
438            features: Vec::<Feature>::new(),
439        };
440        let line = LineLabel {
441            polyline: [[0.0, 0.0], [500.0, 0.0]].into(),
442            anchor_distance: 250.0,
443            glyph_offsets: vec![-10.0, 10.0],
444            image_sizes: vec![[0.0; 2]; 2],
445            first_glyph_index: 6,
446        };
447        let poses = [
448            GlyphPose {
449                point: [90.0, 200.0],
450                angle: 0.5,
451                rise: 0.0,
452            },
453            GlyphPose {
454                point: [110.0, 205.0],
455                angle: 0.75,
456                rise: 12.5,
457            },
458        ];
459        let mut metadata = vec![SDFShaderFeatureMetadata::default(); 12];
460        write_glyph_poses(&layer, &line, &poses, &mut metadata);
461        assert!(
462            metadata[..4].iter().all(|entry| entry.pose[3] == 0.0),
463            "the icon has no pose"
464        );
465        assert!(metadata[4..8]
466            .iter()
467            .all(|entry| entry.pose == [-10.0, 0.0, 0.5, POSED_GLYPH]));
468        assert!(metadata[8..]
469            .iter()
470            .all(|entry| entry.pose == [10.0, 5.0, 0.75, POSED_GLYPH + 12.5]));
471    }
