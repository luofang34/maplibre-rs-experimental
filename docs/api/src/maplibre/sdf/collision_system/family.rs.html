1//! Layers GL JS groups into one symbol bucket share one placement.
2//!
3//! GL JS builds one bucket for style layers alike in type, source, source layer, zoom range,
4//! filter and layout, and only the lowest of them places it. The others draw that bucket with
5//! their own paint and the same placement, so they neither collide with one another nor
6//! with the lowest, and a query finds a placed label in each of them.
7
8use std::collections::HashMap;
9
10use super::layer_pass::{empty_metadata, write_feature_metadata};
11use crate::{
12    render::shaders::SDFShaderFeatureMetadata,
13    sdf::{
14        line_glyphs::GlyphPose,
15        query::{PlacedSymbol, PlacedSymbols},
16        SymbolLayerData,
17    },
18    style::Style,
19};

21/// How one feature was placed, for the layers that share its placement.
22pub(super) struct Outcome {
23    pub(super) opacity: [f32; 2],
24    pub(super) text_shift: [f32; 2],
25    pub(super) anchor: usize,
26    pub(super) ground: f32,
27    pub(super) poses: Option<Vec<GlyphPose>>,
28}
29
30/// Each tile's features as placed, in the order of the layer's tiles.
31pub(super) type Outcomes = Vec<Vec<Option<Outcome>>>;
32
33/// The layer that places for each symbol layer GL JS would group with others: the lowest of
34/// its group in style order. Layers that are alone are absent.
35pub(super) fn leaders(style: &Style) -> HashMap<String, String> {
36    let mut groups: HashMap<String, Vec<(u32, &str)>> = HashMap::new();
37    for layer in style.layers.iter().filter(|layer| layer.type_ == "symbol") {
38        if let Some(key) = layer.layout_group_key() {
39            groups
40                .entry(key)
41                .or_default()
42                .push((layer.index, &layer.id));
43        }
44    }
45    let mut leaders = HashMap::new();
46    for members in groups.values().filter(|members| members.len() > 1) {
47        let Some((_, lowest)) = members.iter().min_by_key(|(index, _)| *index) else {
48            continue;
49        };
50        for (_, id) in members {
51            leaders.insert((*id).to_owned(), (*lowest).to_owned());
52        }
53    }
54    leaders
55}

57/// The metadata of `members` drawn with their own paint and the placement of `leader`'s
58/// tiles, and their labels added to the queryable ones under their own layer.
59pub(super) fn mirror(
60    (leader, outcomes): (&[&SymbolLayerData], &Outcomes),
61    members: &[&SymbolLayerData],
62    (paint, zoom): (&crate::style::layer::SymbolPaint, f64),
63    placed: &mut PlacedSymbols,
64) -> Vec<Vec<SDFShaderFeatureMetadata>> {
65    let leader_id = leader.first().map(|layer| layer.style_layer_id.clone());
66    let copies: Vec<PlacedSymbol> = placed
67        .0
68        .iter()
69        .filter(|symbol| Some(&symbol.layer) == leader_id.as_ref())
70        .flat_map(|symbol| {
71            members
72                .iter()
73                .filter(|member| member.coords == symbol.coords)
74                .map(|member| PlacedSymbol {
75                    layer: member.style_layer_id.clone(),
76                    ..symbol.clone()
77                })
78        })
79        .collect();
80    placed.0.extend(copies);
81    members
82        .iter()
83        .map(|member| {
84            let mut metadata = empty_metadata(member);
85            let Some(tile) = leader
86                .iter()
87                .position(|layer| layer.coords == member.coords)
88            else {
89                return metadata;
90            };
91            for (index, outcome) in outcomes[tile].iter().enumerate() {
92                let (Some(outcome), Some(feature)) = (outcome, member.features.get(index)) else {
93                    continue;
94                };
95                write_feature_metadata(
96                    member,
97                    feature,
98                    (outcome.opacity, outcome.text_shift, outcome.anchor),
99                    outcome.ground,
100                    outcome.poses.as_deref(),
101                    (paint, zoom, &mut metadata),
102                );
103            }
104            metadata
105        })
106        .collect()
107}
