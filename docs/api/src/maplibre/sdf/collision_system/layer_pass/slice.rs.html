15const FEATURE_BATCH_CHECK: usize = 32;

17/// In-progress placement state across a style layer's visible tiles.
18pub(in crate::sdf::collision_system) struct LayerSliceProgress {
19    ordered: Vec<(usize, usize, bool)>,
20    cursor: usize,
21    pub(in crate::sdf::collision_system) metadata: Vec<Vec<SDFShaderFeatureMetadata>>,
22    pub(in crate::sdf::collision_system) outcomes: super::super::family::Outcomes,
23}

25impl LayerSliceProgress {
26    pub(in crate::sdf::collision_system) fn start(
27        world: &crate::tcs::world::World,
28        view_state: &crate::render::view_state::ViewState,
29        projection: &crate::render::projection::ShaderProjectionData,
30        layers: &[&crate::sdf::SymbolLayerData],
31        paint: &crate::style::layer::SymbolPaint,
32        history: &temporal::PlacementHistory,
33    ) -> Self {
34        let metadata: Vec<_> = layers.iter().map(|layer| empty_metadata(layer)).collect();
35        let outcomes: super::super::family::Outcomes =
36            (0..layers.len()).map(|_| Vec::new()).collect();
37        let zoom = view_state.style_zoom().value();
38        let uniforms = SymbolUniforms::new(paint, zoom, [1, 1]);
39        let has_terrain_or_height = world
40            .resources
41            .get::<crate::terrain::coverage::TerrainCoverageIndex>()
42            .is_some()
43            || paint.uses_shared_height();
44        let culled: Vec<bool> = layers
45            .iter()
46            .map(|layer| {
47                tile_culled_on_flat_mercator(
48                    layer,
49                    view_state,
50                    projection,
51                    &uniforms,
52                    has_terrain_or_height,
53                    history,
54                )
55            })
56            .collect();
57        if culled.iter().all(|&c| c) {
58            return Self {
59                ordered: Vec::new(),
60                cursor: 0,
61                metadata,
62                outcomes,
63            };
64        }
65        let ordered = ordered_features(layers, &culled, history)
66            .into_iter()
67            .filter_map(|(pos, feat_idx, feature, was_visible)| {
68                (!feature.fallback).then_some((pos, feat_idx, was_visible))
69            })
70            .collect();
71        Self {
72            ordered,
73            cursor: 0,
74            metadata,
75            outcomes,
76        }
77    }
78
79    /// Advances feature placement until complete (`true`) or `should_yield()` triggers (`false`).
80    #[allow(clippy::too_many_arguments)]
81    pub(in crate::sdf::collision_system) fn step(
82        &mut self,
83        world: &crate::tcs::world::World,
84        view_state: &crate::render::view_state::ViewState,
85        projection: &crate::render::projection::ShaderProjectionData,
86        layers: &[Option<&crate::sdf::SymbolLayerData>],
87        paint: &crate::style::layer::SymbolPaint,
88        zoom_limits: [f64; 2],
89        placement: (
90            &mut CollisionGrid,
91            &mut PlacedSymbols,
92            &mut temporal::PlacementHistory,
93            &mut super::super::upload_diff::MetadataScratchPool,
94        ),
95        mut should_yield: impl FnMut() -> bool,
96    ) -> bool {
97        if self.cursor >= self.ordered.len() {
98            return true;
99        }
100        let (boxes, placed, history, scratch) = placement;
101        let zoom = view_state.style_zoom().value();
102        let has_data_driven_sizes = paint
103            .text_size
104            .as_ref()
105            .is_some_and(|s| !s.is_feature_constant())
106            || paint
107                .compiled_property::<f32>("icon-size")
108                .is_some_and(|s| !s.is_feature_constant());
109        let has_terrain_or_height = world
110            .resources
111            .get::<crate::terrain::coverage::TerrainCoverageIndex>()
112            .is_some()
113            || paint.uses_shared_height();
114        let repeat_distance = f64::from(paint.number(
115            "symbol-spacing",
116            &crate::style::expression::FeatureProperties::default(),
117            zoom,
118            super::super::repeat_spacing::MIN_REPEAT_DISTANCE as f32,
119        ))
120        .clamp(1.0, super::super::repeat_spacing::MIN_REPEAT_DISTANCE);
121        let frame = LayerFrame {
122            world,
123            view_state,
124            projection,
125            paint,
126            uniforms: SymbolUniforms::new(paint, zoom, [1, 1]),
127            zoom_limits,
128            glyphs_stay_upright: crate::sdf::paint::glyphs_stay_upright(paint, zoom),
129            has_data_driven_sizes,
130            has_terrain_or_height,
131            has_icon_text_fit: paint.properties.contains_key("icon-text-fit"),
132            constant_rules: rules::PlacementRules::constant_for_layer(paint, zoom),
133            visibility_policy: world
134                .resources
135                .get::<crate::sdf::visibility::SymbolVisibility>(),
136            repeat_distance,
137        };
138
139        let mut processed_in_step = 0_usize;
140        while self.cursor < self.ordered.len() {
141            if processed_in_step > 0
142                && processed_in_step.is_multiple_of(FEATURE_BATCH_CHECK)
143                && should_yield()
144            {
145                return false;
146            }
147            let (position, feature_index, was_visible) = self.ordered[self.cursor];
148            self.cursor += 1;
149            processed_in_step += 1;
150
151            let Some(layer) = layers.get(position).copied().flatten() else {
152                continue;
153            };
154            let Some(initial_feature) = layer.features.get(feature_index) else {
155                continue;
156            };
157            let mut next = Some((feature_index, initial_feature, was_visible, false));
158            while let Some((index, feature, was_visible, suppressed)) = next.take() {
159                let feat_frame = LayerFrame {
160                    uniforms: if frame.has_data_driven_sizes {
161                        frame
162                            .uniforms
163                            .with_feature_sizes(paint, &feature.data.properties, zoom)
164                    } else {
165                        frame.uniforms
166                    },
167                    ..frame
168                };
169                let outcome = feat_frame.place_feature(
170                    (layer, feature, index, was_visible, suppressed),
171                    (&mut *boxes, &mut *placed, &mut *history),
172                );
173                let poses = match &outcome.line {
174                    Some(LinePoses::Poses(poses)) => Some(poses.clone()),
175                    _ => None,
176                };
177                if (!outcome.ramps[0].is_hidden() || !outcome.ramps[1].is_hidden())
178                    && self.metadata[position].capacity() == 0
179                {
180                    self.metadata[position] = scratch.acquire();
181                }
182                write_feature_metadata(
183                    layer,
184                    feature,
185                    (outcome.ramps, outcome.text_shift, outcome.anchor),
186                    outcome.ground,
187                    poses.as_deref(),
188                    (paint, zoom, &mut self.metadata[position]),
189                );
190                if !outcome.ramps[0].is_hidden() || !outcome.ramps[1].is_hidden() {
191                    if self.outcomes[position].is_empty() {
192                        self.outcomes[position].resize_with(layer.features.len(), || None);
193                    }
194                    if let Some(slot) = self.outcomes[position].get_mut(index) {
195                        *slot = Some(super::super::family::Outcome {
196                            ramps: outcome.ramps,
197                            text_shift: outcome.text_shift,
198                            anchor: outcome.anchor,
199                            ground: outcome.ground,
200                            poses,
201                        });
202                    }
203                }
204                next = layer
205                    .features
206                    .get(index + 1)
207                    .filter(|other| other.fallback)
208                    .map(|other| {
209                        (
210                            index + 1,
211                            other,
212                            history.was_visible(layer, other),
213                            suppressed || outcome.placed,
214                        )
215                    });
216            }
217        }
218        true
219    }

221    pub(in crate::sdf::collision_system) fn recycle(
222        self,
223        scratch: &mut super::super::upload_diff::MetadataScratchPool,
224    ) {
225        for buf in self.metadata {
226            scratch.recycle(buf);
227        }
