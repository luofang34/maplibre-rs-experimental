1//! Text and icon overlap, dependency, and collision insertion rules.
2use super::super::collision_grid::CollisionGrid;
3use crate::style::{
4    expression::FeatureProperties,
5    layer::{StyleProperty, SymbolPaint},
6};
7
8/// How a symbol treats overlap, from `text-overlap` and `icon-overlap`.
9#[derive(Clone, Copy, PartialEq, Eq)]
10enum Overlap {
11    /// Collides with every placed symbol.
12    Never,
13    /// Never collides.
14    Always,
15    /// Collides only with symbols that are not cooperative.
16    Cooperative,
17}
18
19pub(super) struct PlacementRules {
20    overlap: [Overlap; 2],
21    ignore: [bool; 2],
22    optional: [bool; 2],
23}
24
25impl PlacementRules {
26    pub(super) fn new(paint: &SymbolPaint, properties: &FeatureProperties, zoom: f64) -> Self {
27        let read = |suffix: &str| {
28            ["text", "icon"].map(|prefix| {
29                paint
30                    .properties
31                    .get(&format!("{prefix}-{suffix}"))
32                    .and_then(|value| {
33                        StyleProperty::<bool>::parse(value).evaluate_for(properties, zoom)
34                    })
35                    .unwrap_or(false)
36            })
37        };
38        let allow = read("allow-overlap");
39        // The newer property replaces the boolean when a style sets both.
40        let overlap = [0, 1].map(|i| {
41            let name = format!("{}-overlap", ["text", "icon"][i]);
42            match paint.properties.get(&name).and_then(|value| value.as_str()) {
43                Some("always") => Overlap::Always,
44                Some("cooperative") => Overlap::Cooperative,
45                Some(_) => Overlap::Never,
46                None if allow[i] => Overlap::Always,
47                None => Overlap::Never,
48            }
49        });
50        Self {
51            overlap,
52            ignore: read("ignore-placement"),
53            optional: read("optional"),
54        }
55    }
56
57    #[cfg(test)]
58    pub(super) fn place(
59        &self,
60        rectangles: [Option<[f64; 4]>; 2],
61        grid: &mut CollisionGrid,
62        viewport: [f64; 2],
63    ) -> [bool; 2] {
64        self.place_along_line(rectangles, &[], grid, viewport)
65    }
66
67    /// Places a label whose text collides through `glyph_boxes`, one box per glyph along its
68    /// line, instead of through the single rectangle around all of it. The rectangle still
69    /// decides whether the label is on screen.
70    pub(super) fn place_along_line(
71        &self,
72        rectangles: [Option<[f64; 4]>; 2],
73        glyph_boxes: &[[f64; 4]],
74        grid: &mut CollisionGrid,
75        viewport: [f64; 2],
76    ) -> [bool; 2] {
77        let visible = self.visible(rectangles, glyph_boxes, grid, viewport);
78        for (i, shown) in visible.into_iter().enumerate() {
79            if shown && !self.ignore[i] {
80                for part in Self::collision_boxes(i, rectangles, glyph_boxes) {
81                    grid.insert_as(part, self.overlap[i] == Overlap::Cooperative);
82                }
83            }
84        }
85        visible
86    }
87
88    fn collision_boxes(
89        i: usize,
90        rectangles: [Option<[f64; 4]>; 2],
91        glyph_boxes: &[[f64; 4]],
92    ) -> Vec<[f64; 4]> {
93        if i == 0 && !glyph_boxes.is_empty() {
94            glyph_boxes.to_vec()
95        } else {
96            rectangles[i].into_iter().collect()
97        }
98    }
99
100    /// Which parts of a label would be shown, without taking their place in the grid.
101    pub(super) fn visible(
102        &self,
103        rectangles: [Option<[f64; 4]>; 2],
104        glyph_boxes: &[[f64; 4]],
105        grid: &CollisionGrid,
106        viewport: [f64; 2],
107    ) -> [bool; 2] {
108        self.visible_as(self.overlap, rectangles, glyph_boxes, grid, viewport)
109    }
110
111    /// Like `visible`, with the text treated as one that may not overlap: the first attempt of
112    /// a label with variable anchors, before it accepts overlap at its first anchor.
113    pub(super) fn visible_without_text_overlap(
114        &self,
115        rectangles: [Option<[f64; 4]>; 2],
116        glyph_boxes: &[[f64; 4]],
117        grid: &CollisionGrid,
118        viewport: [f64; 2],
119    ) -> [bool; 2] {
120        let overlap = [Overlap::Never, self.overlap[1]];
121        self.visible_as(overlap, rectangles, glyph_boxes, grid, viewport)
122    }
123
124    fn visible_as(
125        &self,
126        overlap: [Overlap; 2],
127        rectangles: [Option<[f64; 4]>; 2],
128        glyph_boxes: &[[f64; 4]],
129        grid: &CollisionGrid,
130        viewport: [f64; 2],
131    ) -> [bool; 2] {
132        let accepted = [0, 1].map(|i| {
133            rectangles[i].is_some_and(|rect| {
134                rect.iter().all(|v| v.is_finite())
135                    && rect[2] >= -VIEWPORT_PADDING
136                    && rect[3] >= -VIEWPORT_PADDING
137                    && rect[0] < viewport[0] + VIEWPORT_PADDING
138                    && rect[1] < viewport[1] + VIEWPORT_PADDING
139                    && Self::collision_boxes(i, rectangles, glyph_boxes)
140                        .iter()
141                        .all(|part| match overlap[i] {
142                            Overlap::Always => true,
143                            Overlap::Never => !grid.overlaps(*part),
144                            Overlap::Cooperative => !grid.overlaps_non_cooperative(*part),
145                        })
146            })
147        });
148        [0, 1].map(|i| {
149            accepted[i] && (rectangles[1 - i].is_none() || accepted[1 - i] || self.optional[1 - i])
150        })
151    }
152}
153
154/// How far past the viewport edge a box still takes part in placement, as in GL JS, so a label
155/// whose icon is just off screen keeps its text and labels slide in without popping.
156const VIEWPORT_PADDING: f64 = 100.0;
