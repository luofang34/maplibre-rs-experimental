1//! Text and icon overlap, dependency, and collision insertion rules.
2use super::super::collision_grid::CollisionGrid;
3use crate::style::{expression::FeatureProperties, layer::SymbolPaint};
4
5/// How a symbol treats overlap, from `text-overlap` and `icon-overlap`.
6#[derive(Clone, Copy, PartialEq, Eq)]
7enum Overlap {
8    /// Collides with every placed symbol.
9    Never,
10    /// Never collides.
11    Always,
12    /// Collides only with symbols that are not cooperative.
13    Cooperative,
14}
15
16#[derive(Clone, Copy)]
17pub(super) struct PlacementRules {
18    overlap: [Overlap; 2],
19    ignore: [bool; 2],
20    optional: [bool; 2],
21}
22
23impl PlacementRules {
24    pub(super) fn constant_for_layer(paint: &SymbolPaint, zoom: f64) -> Option<Self> {
25        let is_const = |name: &str| {
26            paint
27                .compiled_property::<bool>(name)
28                .is_none_or(|prop| prop.is_feature_constant())
29        };
30        if ![
31            "text-allow-overlap",
32            "icon-allow-overlap",
33            "text-ignore-placement",
34            "icon-ignore-placement",
35            "text-optional",
36            "icon-optional",
37        ]
38        .into_iter()
39        .all(is_const)
40        {
41            return None;
42        }
43        Some(Self::new(paint, &FeatureProperties::default(), zoom))
44    }
45
46    pub(super) fn new(paint: &SymbolPaint, properties: &FeatureProperties, zoom: f64) -> Self {
47        let read = |names: [&str; 2]| {
48            names.map(|name| {
49                paint
50                    .compiled_property::<bool>(name)
51                    .and_then(|prop| {
52                        if prop.is_feature_constant() {
53                            prop.evaluate_at_zoom(zoom)
54                        } else {
55                            prop.evaluate_for(properties, zoom)
56                        }
57                    })
58                    .unwrap_or(false)
59            })
60        };
61        let allow = read(["text-allow-overlap", "icon-allow-overlap"]);
62        // The newer property replaces the boolean when a style sets both.
63        let overlap_names = ["text-overlap", "icon-overlap"];
64        // Layout follows the integer zoom a tile is laid out at, not the view's fraction.
65        let overlap =
66            [0, 1].map(
67                |i| match paint.layout_enum(overlap_names[i], zoom.floor()).as_deref() {
68                    Some("always") => Overlap::Always,
69                    Some("cooperative") => Overlap::Cooperative,
70                    Some(_) => Overlap::Never,
71                    None if allow[i] => Overlap::Always,
72                    None => Overlap::Never,
73                },
74            );
75        Self {
76            overlap,
77            ignore: read(["text-ignore-placement", "icon-ignore-placement"]),
78            optional: read(["text-optional", "icon-optional"]),
79        }
80    }

82    #[cfg(test)]
83    pub(super) fn place(
84        &self,
85        rectangles: [Option<[f64; 4]>; 2],
86        grid: &mut CollisionGrid,
87        viewport: [f64; 2],
88    ) -> [bool; 2] {
89        self.place_along_line(rectangles, &[], grid, viewport)
90    }
91
92    /// Places a label whose text collides through `glyph_boxes`, one box per glyph along its
93    /// line, instead of through the single rectangle around all of it. The rectangle still
94    /// decides whether the label is on screen.
95    pub(super) fn place_along_line(
96        &self,
97        rectangles: [Option<[f64; 4]>; 2],
98        glyph_boxes: &[[f64; 4]],
99        grid: &mut CollisionGrid,
100        viewport: [f64; 2],
101    ) -> [bool; 2] {
102        let visible = self.visible(rectangles, glyph_boxes, grid, viewport);
103        for (i, shown) in visible.into_iter().enumerate() {
104            if shown && !self.ignore[i] {
105                for part in Self::collision_boxes(i, rectangles, glyph_boxes) {
106                    grid.insert_as(part, self.overlap[i] == Overlap::Cooperative);
107                }
108            }
109        }
110        visible
111    }
112
113    fn collision_boxes(
114        i: usize,
115        rectangles: [Option<[f64; 4]>; 2],
116        glyph_boxes: &[[f64; 4]],
117    ) -> Vec<[f64; 4]> {
118        if i == 0 && !glyph_boxes.is_empty() {
119            glyph_boxes.to_vec()
120        } else {
121            rectangles[i].into_iter().collect()
122        }
123    }
124
125    /// Which parts of a label would be shown, without taking their place in the grid.
126    pub(super) fn visible(
127        &self,
128        rectangles: [Option<[f64; 4]>; 2],
129        glyph_boxes: &[[f64; 4]],
130        grid: &CollisionGrid,
131        viewport: [f64; 2],
132    ) -> [bool; 2] {
133        self.visible_as(self.overlap, rectangles, glyph_boxes, grid, viewport)
134    }
135
136    /// Like `visible`, with the text treated as one that may not overlap: the first attempt of
137    /// a label with variable anchors, before it accepts overlap at its first anchor.
138    pub(super) fn visible_without_text_overlap(
139        &self,
140        rectangles: [Option<[f64; 4]>; 2],
141        glyph_boxes: &[[f64; 4]],
142        grid: &CollisionGrid,
143        viewport: [f64; 2],
144    ) -> [bool; 2] {
145        let overlap = [Overlap::Never, self.overlap[1]];
146        self.visible_as(overlap, rectangles, glyph_boxes, grid, viewport)
147    }
148
149    fn visible_as(
150        &self,
151        overlap: [Overlap; 2],
152        rectangles: [Option<[f64; 4]>; 2],
153        glyph_boxes: &[[f64; 4]],
154        grid: &CollisionGrid,
155        viewport: [f64; 2],
156    ) -> [bool; 2] {
157        let accepted = [0, 1].map(|i| {
158            rectangles[i].is_some_and(|rect| {
159                rect.iter().all(|v| v.is_finite())
160                    && rect[2] >= -VIEWPORT_PADDING
161                    && rect[3] >= -VIEWPORT_PADDING
162                    && rect[0] < viewport[0] + VIEWPORT_PADDING
163                    && rect[1] < viewport[1] + VIEWPORT_PADDING
164                    && Self::collision_boxes(i, rectangles, glyph_boxes)
165                        .iter()
166                        .all(|part| match overlap[i] {
167                            Overlap::Always => true,
168                            Overlap::Never => !grid.overlaps(*part),
169                            Overlap::Cooperative => !grid.overlaps_non_cooperative(*part),
170                        })
171            })
172        });
173        [0, 1].map(|i| {
174            accepted[i] && (rectangles[1 - i].is_none() || accepted[1 - i] || self.optional[1 - i])
175        })
176    }
177}
178
179/// How far past the viewport edge a box still takes part in placement, as in GL JS, so a label
180/// whose icon is just off screen keeps its text and labels slide in without popping.
181const VIEWPORT_PADDING: f64 = 100.0;
