1//! Placement history follows a symbol across tile replacement, with bounded fades.
2use std::{collections::HashMap, time::Duration};
3
4use crate::{
5    coords::WorldTileCoords,
6    sdf::{Feature, SymbolLayerData},
7};
8
9#[derive(Clone, Hash, PartialEq, Eq)]
10struct Key {
11    layer: String,
12    text: String,
13    id: Option<u64>,
14}
15#[derive(Clone)]
16struct State {
17    position: [f64; 2],
18    zoom: u8,
19    opacity: [f32; 2],
20    target: [bool; 2],
21    last_seen: Duration,
22    claimed: u64,
23    claimed_by: WorldTileCoords,
24    /// The variable anchor the text was last placed with.
25    anchor: Option<usize>,
26}
27#[derive(Default)]
28pub(super) struct PlacementHistory {
29    states: HashMap<Key, Vec<State>>,
30    now: Duration,
31    frame: u64,
32    pub(super) fading: bool,
33    /// Where the repeats of labels along lines were placed on screen.
34    pub(super) repeats: super::repeat_spacing::RepeatSpacing,
35}
36impl PlacementHistory {
37    pub(super) fn begin(&mut self, now: Duration) {
38        self.now = now;
39        self.frame = self.frame.wrapping_add(1);
40        self.fading = false;
41        self.repeats.begin();
42        self.states.retain(|_, states| {
43            states.retain(|state| now.saturating_sub(state.last_seen) < Duration::from_secs(1));
44            !states.is_empty()
45        });
46    }
47    pub(super) fn was_visible(&self, layer: &SymbolLayerData, feature: &Feature) -> bool {
48        self.states
49            .get(&key(layer, feature))
50            .and_then(|states| {
51                states
52                    .iter()
53                    .find(|state| matches(state, layer.coords, feature))
54            })
55            .is_some_and(|state| state.target.iter().any(|v| *v))
56    }
57    /// Whether another tile's copy of the label already took it this frame, so this copy is
58    /// neither drawn nor found by a query. Asked before [`Self::opacity`] takes it.
59    pub(super) fn shown_elsewhere(&self, layer: &SymbolLayerData, feature: &Feature) -> bool {
60        let frame = self.frame;
61        self.states.get(&key(layer, feature)).is_some_and(|states| {
62            states
63                .iter()
64                .find(|state| {
65                    matches(state, layer.coords, feature)
66                        && !(state.claimed == frame && state.claimed_by == layer.coords)
67                })
68                .is_some_and(|state| state.claimed == frame)
69        })
70    }
71
72    pub(super) fn opacity(
73        &mut self,
74        layer: &SymbolLayerData,
75        feature: &Feature,
76        target: [bool; 2],
77    ) -> [f32; 2] {
78        let frame = self.frame;
79        let states = self.states.entry(key(layer, feature)).or_default();
80        // A state a same-tile feature already took this frame belongs to that feature, so a
81        // second feature of the tile with the same key is tracked separately.
82        let existing = states.iter().position(|state| {
83            matches(state, layer.coords, feature)
84                && !(state.claimed == frame && state.claimed_by == layer.coords)
85        });
86        let index = existing.unwrap_or_else(|| {
87            states.push(State {
88                position: position(layer.coords, feature),
89                zoom: u8::from(layer.coords.z),
90                opacity: [0.0; 2],
91                target,
92                last_seen: self.now,
93                claimed: frame.wrapping_sub(1),
94                claimed_by: layer.coords,
95                anchor: None,
96            });
97            states.len() - 1
98        });
99        let state = &mut states[index];
100        // A label repeated across tiles is drawn once, so a parent and child that are both
101        // visible do not double up.
102        if state.claimed == frame {
103            return [0.0; 2];
104        }
105        let step = (self.now.saturating_sub(state.last_seen).as_secs_f32() / 0.16).min(1.0);
106        for (i, visible) in target.iter().enumerate() {
107            state.opacity[i] =
108                (state.opacity[i] + if state.target[i] { step } else { -step }).clamp(0.0, 1.0);
109            self.fading |= state.opacity[i] != if *visible { 1.0 } else { 0.0 };
110        }
111        state.target = target;
112        state.last_seen = self.now;
113        state.claimed = frame;
114        state.claimed_by = layer.coords;
115        state.position = position(layer.coords, feature);
116        state.zoom = u8::from(layer.coords.z);
117        state.opacity
118    }
119}
120impl PlacementHistory {
121    /// The anchor the label's text was placed with while it was last shown or fading out, which
122    /// GL JS tries first so a label that still fits does not jump between anchors.
123    pub(super) fn previous_anchor(
124        &self,
125        layer: &SymbolLayerData,
126        feature: &Feature,
127    ) -> Option<usize> {
128        self.states
129            .get(&key(layer, feature))?
130            .iter()
131            .find(|state| matches(state, layer.coords, feature))?
132            .anchor
133    }
134
135    /// Records the anchor this frame placed the label's text with, on the state
136    /// [`Self::opacity`] took for it. A label not placed keeps its anchor while it fades out,
137    /// as GL JS keeps the offsets of symbols that are not yet hidden.
138    pub(super) fn remember_anchor(
139        &mut self,
140        layer: &SymbolLayerData,
141        feature: &Feature,
142        anchor: Option<usize>,
143    ) {
144        let frame = self.frame;
145        if let Some(state) = self
146            .states
147            .get_mut(&key(layer, feature))
148            .and_then(|states| {
149                states.iter_mut().find(|state| {
150                    state.claimed == frame
151                        && state.claimed_by == layer.coords
152                        && matches(state, layer.coords, feature)
153                })
154            })
155        {
156            if anchor.is_some() || state.opacity[0] == 0.0 {
157                state.anchor = anchor;
158            }
159        }
160    }
161}

163fn key(layer: &SymbolLayerData, feature: &Feature) -> Key {
164    Key {
165        layer: layer.style_layer_id.clone(),
166        text: feature.str.clone(),
167        id: feature.data.id,
168    }
169}
170fn position(tile: WorldTileCoords, feature: &Feature) -> [f64; 2] {
171    let scale = 2_f64.powi(i32::from(u8::from(tile.z)));
172    [
173        (f64::from(tile.x) + f64::from(feature.text_anchor.x) / 4096.0) / scale,
174        (f64::from(tile.y) + f64::from(feature.text_anchor.y) / 4096.0) / scale,
175    ]
176}
177fn matches(state: &State, tile: WorldTileCoords, feature: &Feature) -> bool {
178    let point = position(tile, feature);
179    let tolerance = 8.0 / (4096.0 * 2_f64.powi(i32::from(state.zoom.min(u8::from(tile.z)))));
180    let dx = (point[0] - state.position[0] + 0.5).rem_euclid(1.0) - 0.5;
181    dx.abs() <= tolerance && (point[1] - state.position[1]).abs() <= tolerance
182}
