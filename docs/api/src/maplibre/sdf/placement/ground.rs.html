15pub(in crate::sdf) fn canonical_tile(coords: WorldTileCoords) -> Option<TileCoords> {
16    let count = i32::try_from(ZOOM_BOUNDS[usize::from(u8::from(coords.z))]).ok()?;
17    (coords.y >= 0 && coords.y < count).then_some(TileCoords {
18        x: coords.x.rem_euclid(count) as u32,
19        y: coords.y as u32,
20        z: coords.z,
21    })
22}

24pub(in crate::sdf) fn project(
25    coords: WorldTileCoords,
26    anchor: [f64; 2],
27    elevation: f64,
28    view: &ViewState,
29    projection: &ShaderProjectionData,
30) -> Option<Vector4<f64>> {
31    let tile = canonical_tile(coords)?;
32    let transition = f64::from(projection.transition);
33    if transition <= 0.0 {
34        return Some(
35            view.gpu_view_projection()
36                .to_model_view_projection(coords.transform_for_zoom(view.zoom()))
37                .get()
38                * Vector4::new(anchor[0], anchor[1], elevation, 1.0),
39        );
40    }
41    let surface = crate::projection::globe::project_tile_coordinates_to_unit_sphere(
42        tile.x,
43        tile.y,
44        u8::from(tile.z),
45        anchor[0],
46        anchor[1],
47    );
48    let plane = Vector4::<f32>::from(projection.clipping_plane).cast::<f64>()?;
49    if projection.transition >= 0.98 && plane.dot(surface.extend(1.0)) < 0.0 {
50        return None;
51    }
52    let globe = Matrix4::<f32>::from(projection.main_matrix).cast::<f64>()?
53        * (surface * (1.0 + elevation / f64::from(projection.radius_meters))).extend(1.0);
54    if transition >= 1.0 {
55        return Some(globe);
56    }
57    let flat = view
58        .gpu_view_projection()
59        .to_model_view_projection(coords.transform_for_zoom(view.zoom()))
60        .get()
61        * Vector4::new(anchor[0], anchor[1], elevation, 1.0);
62    Some(flat * (1.0 - transition) + globe * transition)
63}

65/// Heights within this of the ground count as on it, so DEM sampling noise never hides a label.
66const BURIAL_TOLERANCE_METERS: f32 = 1.0;
67
68/// The terrain elevation in metres at `point`, in tile units of the tile at `coords`; sea level
69/// where no DEM covers it.
70pub(in crate::sdf) fn elevation_at(world: &World, coords: WorldTileCoords, point: [f64; 2]) -> f32 {
71    let scale = 2_f64.powi(i32::from(u8::from(coords.z)));
72    let x = (f64::from(coords.x) + point[0] / 4096.0) / scale;
73    let y = (f64::from(coords.y) + point[1] / 4096.0) / scale;
74    world
75        .resources
76        .get::<TerrainCoverageIndex>()
77        .and_then(|index| {
78            index.elevation_at(&world.tiles, x, y).or_else(|| {
79                index.loaded_ground_cached(&world.tiles, x, y, crate::terrain::DemDetail::Finest)
80            })
81        })
82        .unwrap_or(0.0) as f32
83}

85/// The terrain elevation under the feature's text anchor.
86pub(in crate::sdf) fn elevation(world: &World, layer: &SymbolLayerData, feature: &Feature) -> f32 {
87    elevation_at(world, layer.coords, anchor_point(feature))
88}
89
90fn anchor_point(feature: &Feature) -> [f64; 2] {
91    [
92        f64::from(feature.text_anchor.x),
93        f64::from(feature.text_anchor.y),
94    ]
95}

97/// The height of the label's anchor: the terrain under it unless the style places the symbol
98/// at an absolute height, plus the style's height offset.
99pub(in crate::sdf) fn symbol_elevation(
100    world: &World,
101    layer: &SymbolLayerData,
102    feature: &Feature,
103    paint: &SymbolPaint,
104    zoom: f64,
105) -> f32 {
106    let terrain = elevation(world, layer, feature);
107    if !paint.uses_shared_height() {
108        return terrain;
109    }
110    let base = if paint.height_follows_ground("text") {
111        terrain
112    } else {
113        0.0
114    };
115    base + paint.height_offset("text", &feature.data.properties, zoom)
116}

118/// Whether the anchor at `height` is inside the terrain surface below it, where the depth test
119/// hides it. Such a label must not hold collision space or answer queries.
120pub(in crate::sdf) fn buried_in_terrain(
121    world: &World,
122    layer: &SymbolLayerData,
123    feature: &Feature,
124    height: f32,
125) -> bool {
126    elevation(world, layer, feature) - height > BURIAL_TOLERANCE_METERS
127}

129/// The ground a label along a line stands on. Its anchor is drawn at [`Self::elevation`], and
130/// each glyph that follows the ground rises with the terrain under it, so a label climbing a
131/// slope lies on the slope rather than through it.
132pub(in crate::sdf) struct LabelGround<'a> {
133    world: &'a World,
134    coords: WorldTileCoords,
135    /// The height of the label's anchor, as [`symbol_elevation`] gives it.
136    pub(in crate::sdf) elevation: f32,
137    /// The terrain under the anchor, which glyphs rise from; `None` when the label's height
138    /// does not follow the ground.
139    anchor_terrain: Option<f32>,
140}
141
142impl<'a> LabelGround<'a> {
143    pub(in crate::sdf) fn flat(world: &'a World, coords: WorldTileCoords) -> Self {
144        Self {
145            world,
146            coords,
147            elevation: 0.0,
148            anchor_terrain: None,
149        }
150    }
151
152    pub(in crate::sdf) fn new(
153        world: &'a World,
154        layer: &SymbolLayerData,
155        feature: &Feature,
156        paint: &SymbolPaint,
157        zoom: f64,
158    ) -> Self {
159        let follows = paint.height_follows_ground("text")
160            && world.resources.get::<TerrainCoverageIndex>().is_some();
161        Self {
162            world,
163            coords: layer.coords,
164            elevation: symbol_elevation(world, layer, feature, paint, zoom),
165            anchor_terrain: follows.then(|| elevation(world, layer, feature)),
166        }
167    }
168
169    /// How far the terrain at `point`, in tile units, stands above the terrain under the
170    /// anchor, in metres: what a glyph there rises by.
171    pub(in crate::sdf) fn rise_at(&self, point: [f64; 2]) -> f32 {
172        self.anchor_terrain.map_or(0.0, |anchor| {
173            elevation_at(self.world, self.coords, point) - anchor
174        })
175    }
