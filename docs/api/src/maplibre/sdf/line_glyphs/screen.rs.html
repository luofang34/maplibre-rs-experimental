1//! Glyph positions along a line as the screen shows it, for labels that turn with their line
2//! but stand upright to the viewer: the viewport label plane of GL JS's `placeGlyphAlongLine`.
3//!
4//! Glyphs are spaced in screen pixels along the projected line, walking outward from the
5//! anchor, so a label keeps its pixel size however the line is foreshortened. Each glyph's
6//! centre is mapped back onto the line in tile units, and its direction is the line's own in
7//! tile units, which the vertex shader turns to the screen every frame. A head-tracked view
8//! places labels less often than it draws, and a glyph so posed stays on its line and turned
9//! along it between placements.
10
11use super::GlyphPose;
12
13/// Clip w below which a line vertex counts as behind the eye; the line is cut there.
14const NEAR_W: f64 = 1e-3;
15
16/// A point of the line on the screen.
17#[derive(Clone, Copy, Debug, PartialEq)]
18pub(crate) struct OnScreen {
19    /// Pixels, y down.
20    pub screen: [f64; 2],
21    /// Clip-space w, the point's depth in front of the eye.
22    pub w: f64,
23}
24
25/// A line vertex with where the screen shows it.
26#[derive(Clone, Copy)]
27struct Vertex {
28    tile: [f64; 2],
29    at: OnScreen,
30}
31
32/// The vertex just in front of the eye where the line from `from` toward `tile`, which lies
33/// behind it, crosses [`NEAR_W`]. Clip w is not affine along a line on the globe, so the
34/// crossing is bisected on the projection itself; a vertex that still ends up behind the eye
35/// is refused rather than placed.
36fn cut_at_eye(
37    from: Vertex,
38    tile: [f64; 2],
39    project: &impl Fn([f64; 2]) -> Option<OnScreen>,
40) -> Option<Vertex> {
41    let at = |t: f64| {
42        [
43            from.tile[0] + (tile[0] - from.tile[0]) * t,
44            from.tile[1] + (tile[1] - from.tile[1]) * t,
45        ]
46    };
47    let (mut near, mut far) = (0.0_f64, 1.0_f64);
48    for _ in 0..60 {
49        let middle = 0.5 * (near + far);
50        if project(at(middle)).is_some_and(|point| point.w > NEAR_W) {
51            near = middle;
52        } else {
53            far = middle;
54        }
55    }
56    let point = at(near);
57    let on_screen = project(point).filter(|point| point.w > NEAR_W)?;
58    Some(Vertex {
59        tile: point,
60        at: on_screen,
61    })
62}
63
64/// The segment the anchor lies on and the anchor's own point.
65fn anchor_segment(polyline: &[[f32; 2]], anchor_distance: f32) -> Option<(usize, [f64; 2])> {
66    let mut travelled = 0.0_f64;
67    for (index, pair) in polyline.windows(2).enumerate() {
68        let (from, to) = (pair[0].map(f64::from), pair[1].map(f64::from));
69        let length = (to[0] - from[0]).hypot(to[1] - from[1]);
70        let distance = f64::from(anchor_distance);
71        if length > 0.0 && distance <= travelled + length {
72            let t = ((distance - travelled) / length).max(0.0);
73            return Some((
74                index,
75                [
76                    from[0] + (to[0] - from[0]) * t,
77                    from[1] + (to[1] - from[1]) * t,
78                ],
79            ));
80        }
81        travelled += length;
82    }
83    None
84}

86/// Poses of glyphs whose centres lie `offsets` screen pixels (signed, along the text) from the
87/// anchor at `anchor_distance` along `polyline`, as `project` shows the line. `flip` reads the
88/// text against the line, each glyph walking the other way and turning half way around. A
89/// pose's point and angle are the tile-unit point on the line and the tile-unit direction of
90/// the segment it lies on, the way the text reads. `None` when a glyph would lie beyond
91/// either end of the visible line.
92pub(crate) fn place_glyphs_on_screen(
93    polyline: &[[f32; 2]],
94    anchor_distance: f32,
95    offsets: &[f64],
96    flip: bool,
97    project: impl Fn([f64; 2]) -> Option<OnScreen>,
98) -> Option<Vec<GlyphPose>> {
99    let (segment, anchor_tile) = anchor_segment(polyline, anchor_distance)?;
100    let anchor = Vertex {
101        tile: anchor_tile,
102        at: project(anchor_tile).filter(|at| at.w > NEAR_W)?,
103    };
104    offsets
105        .iter()
106        .map(|offset| place_one(polyline, segment, anchor, *offset, flip, &project))
107        .collect()
108}
109
110/// One glyph of [`place_glyphs_on_screen`], walking outward from the anchor.
111fn place_one(
112    polyline: &[[f32; 2]],
113    segment: usize,
114    anchor: Vertex,
115    offset: f64,
116    flip: bool,
117    project: &impl Fn([f64; 2]) -> Option<OnScreen>,
118) -> Option<GlyphPose> {
119    let mut direction: isize = if offset > 0.0 { 1 } else { -1 };
120    let mut angle = 0.0;
121    if flip {
122        direction = -direction;
123        angle = std::f64::consts::PI;
124    }
125    if direction < 0 {
126        angle += std::f64::consts::PI;
127    }
128    let mut index = if direction > 0 {
129        segment as isize
130    } else {
131        segment as isize + 1
132    };
133    let wanted = offset.abs();
134    let (mut previous, mut current) = (anchor, anchor);
135    let (mut walked, mut length) = (0.0, 0.0);
136    let mut cut = false;
137    while walked + length <= wanted {
138        index += direction;
139        // The line ends, or was cut where it passes behind the eye, before the glyph.
140        if cut || index < 0 || index as usize >= polyline.len() {
141            return None;
142        }
143        walked += length;
144        previous = current;
145        let tile = polyline[index as usize].map(f64::from);
146        current = match project(tile).filter(|at| at.w > NEAR_W) {
147            Some(at) => Vertex { tile, at },
148            None => {
149                cut = true;
150                cut_at_eye(previous, tile, project)?
151            }
152        };
153        let (dx, dy) = (
154            current.at.screen[0] - previous.at.screen[0],
155            current.at.screen[1] - previous.at.screen[1],
156        );
157        length = dx.hypot(dy);
158    }
159    let s = (wanted - walked) / length;
160    // A point a fraction `s` along the segment on the screen lies a perspective-corrected
161    // fraction along it in tile space.
162    let (w0, w1) = (previous.at.w, current.at.w);
163    let t = s * w0 / (s * w0 + (1.0 - s) * w1);
164    let point = [
165        previous.tile[0] + (current.tile[0] - previous.tile[0]) * t,
166        previous.tile[1] + (current.tile[1] - previous.tile[1]) * t,
167    ];
168    let along = (current.tile[1] - previous.tile[1]).atan2(current.tile[0] - previous.tile[0]);
169    // Within half a turn of zero, the range the line's own direction takes.
170    let angle = (angle + along + std::f64::consts::PI).rem_euclid(std::f64::consts::TAU)
171        - std::f64::consts::PI;
172    Some(GlyphPose {
173        point: point.map(|value| value as f32),
174        angle: angle as f32,
175        rise: 0.0,
176    })
177}
