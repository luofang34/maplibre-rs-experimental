1//! Stars behind the globe, drawn from the directions, magnitudes and colours a host sets with
2//! [`crate::context::MapContext::set_stars`].
3//!
4//! Stars lie at infinity: the globe's view-projection carries their directions with w = 0, so
5//! they turn with the camera and never move with it. They are drawn first in the main pass at
6//! the far plane and test depth without writing it, so the globe, its terrain and every layer
7//! cover them. A globe whose style paints nothing on it, with no background and no terrain,
8//! shows the stars through its disc. Where the atmosphere is drawn, each star fades with the
9//! brightness of the sky in its direction, so a lit sky hides the stars a night sky shows.
10

31/// Draws the host's stars, sun and moon behind a globe. Build it after
32/// [`crate::render::RenderPlugin`].
33#[derive(Default)]
34pub struct StarfieldPlugin;
35
36impl<E: Environment> Plugin<E> for StarfieldPlugin {
37    fn build(
38        &self,
39        schedule: &mut crate::schedule::Schedule,
40        _kernel: std::rc::Rc<crate::kernel::Kernel<E>>,
41        world: &mut World,
42        graph: &mut crate::render::graph::RenderGraph,
43    ) {
44        world
45            .resources
46            .insert(Eventually::<StarfieldPipeline>::Uninitialized);
47        world
48            .resources
49            .insert(Eventually::<sun::SunPipeline>::Uninitialized);
50        world
51            .resources
52            .insert(Eventually::<sun::depth::SunDepth>::Uninitialized);
53        world
54            .resources
55            .insert(Eventually::<moon::MoonPipeline>::Uninitialized);
56        schedule.add_system_to_stage(RenderStageLabel::Prepare, pipeline::resource_system);
57        schedule.add_system_to_stage(RenderStageLabel::Prepare, sun::resource_system);
58        schedule.add_system_to_stage(RenderStageLabel::Prepare, moon::resource_system);
59        schedule.add_system_to_stage(RenderStageLabel::Queue, upload::queue_system);
60        schedule.add_system_to_stage(RenderStageLabel::Queue, sun::queue_system);
61        schedule.add_system_to_stage(RenderStageLabel::Queue, moon::queue_system);
62        add_sun_depth_copy(graph);
63    }
64}

66/// Puts the sun's depth copy between the main and translucent passes, after everything the
67/// main pass draws and before the sun is drawn.
68fn add_sun_depth_copy(graph: &mut crate::render::graph::RenderGraph) {
69    use crate::render::draw_graph;
70    let Some(draw_graph) = graph.get_sub_graph_mut(draw_graph::NAME) else {
71        tracing::error!("draw graph is missing; the sun's glare will not see the ground");
72        return;
73    };
74    draw_graph.add_node(sun::depth::SUN_DEPTH_COPY, sun::depth::SunDepthCopyNode);
75    let edges = [
76        draw_graph.add_node_edge(draw_graph::node::MAIN_PASS, sun::depth::SUN_DEPTH_COPY),
77        draw_graph.add_node_edge(
78            sun::depth::SUN_DEPTH_COPY,
79            draw_graph::node::TRANSLUCENT_PASS,
80        ),
81    ];
82    for edge in edges {
83        if let Err(error) = edge {
84            tracing::error!(?error, "the sun's depth copy cannot be ordered");
85        }
86    }
87}

89/// Draws the stars queued for this frame, if any; the main pass calls it before any layer.
90pub fn draw<'w>(pass: &mut TrackedRenderPass<'w>, world: &'w World) {
91    let Some((Initialized(StarfieldPipeline(pipeline)), Initialized(projection), gpu)) =
92        world.resources.query::<(
93            &Eventually<StarfieldPipeline>,
94            &Eventually<ProjectionGpuResources>,
95            &StarfieldGpu,
96        )>()
97    else {
98        return;
99    };
100    let Some((instances, frame)) = gpu.drawable() else {
101        return;
102    };
103    pass.set_pipeline(pipeline);
104    pass.set_bind_group(0, projection.bind_group(), &[]);
105    pass.set_bind_group(1, frame, &[]);
106    pass.set_vertex_buffer(0, instances.slice(..));
107    pass.draw(0..6, 0..gpu.count());
108}
