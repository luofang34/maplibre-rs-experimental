1//! Hillshade and colour relief layers: DEM tiles shaded on the GPU.
2//!
3//! Both layer types read their raster-dem source through the raster tile path: the tiles are
4//! fetched, decoded and uploaded like imagery, and the shaders decode elevations from the
5//! texels. Hillshade takes the slope from the eight neighbours of a texel and shades it with
6//! the layer's lights; colour relief maps the elevation through the layer's colour ramp. With
7//! terrain both still draw to the screen rather than into the drape textures.
8

28/// Registers the DEM-shaded layer types; the raster plugin must be registered as well, since
29/// it fetches and uploads the tiles.
30#[derive(Default)]
31pub struct HillshadePlugin;
32
33impl<E: Environment> Plugin<E> for HillshadePlugin {
34    fn build(
35        &self,
36        schedule: &mut Schedule,
37        _kernel: Rc<Kernel<E>>,
38        world: &mut World,
39        _graph: &mut RenderGraph,
40    ) {
41        world
42            .resources
43            .insert(Eventually::<HillshadeResources>::Uninitialized);
44        schedule.add_system_to_stage(RenderStageLabel::Prepare, resource_system::resource_system);
45        schedule.add_system_to_stage(RenderStageLabel::Prepare, prepare_system::prepare_system);
46        schedule.add_system_to_stage(RenderStageLabel::Queue, queue_system::queue_system);
47    }
48}
49
50/// Whether a style layer is one of the DEM-shaded kinds.
51pub fn dem_layer_kind(layer_type: &str) -> Option<DemLayerKind> {
52    match layer_type {
53        "hillshade" => Some(DemLayerKind::Hillshade),
54        "color-relief" => Some(DemLayerKind::ColorRelief),
55        _ => None,
56    }
