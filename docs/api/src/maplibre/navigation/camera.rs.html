11/// Sets the camera pitch while keeping the ground point at the viewport center pinned.
12pub fn set_pitch_keeping_center(view_state: &mut ViewState, pitch: impl Into<Rad<f64>>) {
13    let pitch: Rad<f64> = pitch.into();
14    if view_state.navigation_mode() == crate::render::view_state::NavigationMode::FreeGlobe {
15        if let Some(pose) = view_state.globe_pose() {
16            let current = view_state.camera().get_pitch();
17            let _ = view_state.orbit_globe_pose(pose, Rad(0.0), pitch - current);
18            return;
19        }
20    }
21    view_state.camera_mut().set_pitch(pitch);
22}

24/// Sets the camera bearing while keeping the ground point at the viewport center pinned.
25pub fn set_bearing_keeping_center(view_state: &mut ViewState, bearing: impl Into<Rad<f64>>) {
26    let bearing: Rad<f64> = bearing.into();
27    if view_state.navigation_mode() == crate::render::view_state::NavigationMode::FreeGlobe {
28        if let Some(pose) = view_state.globe_pose() {
29            let current = view_state.camera().get_bearing();
30            if let Err(error) = view_state.orbit_globe_pose(pose, bearing - current, Rad(0.0)) {
31                tracing::warn!(%error, "free globe bearing rejected");
32            }
33            return;
34        }
35    }
36    view_state.camera_mut().set_bearing(bearing);
37}

39/// Rotates camera bearing by `delta` around the ground point under `cursor` (falling back to
40/// the viewport center when the cursor is off the surface).
41pub fn rotate_bearing_around_cursor(
42    style: &crate::style::Style,
43    view_state: &mut ViewState,
44    cursor: Option<Vector2<f64>>,
45    delta: Rad<f64>,
46) {
47    if view_state.navigation_mode() == crate::render::view_state::NavigationMode::FreeGlobe {
48        if let Some(pose) = view_state.globe_pose() {
49            let _ = view_state.orbit_globe_pose(pose, delta, Rad(0.0));
50        }
51        return;
52    }
53    let center = center_pixel(view_state);
54    let pivot = cursor.map_or(center, |c| Point2::new(c.x, c.y));
55
56    if let Some(before_globe) = active_globe_camera(style, view_state) {
57        let anchor = before_globe.screen_point_to_location(pivot);
58        let current = view_state.camera().get_bearing();
59        view_state.camera_mut().set_bearing(current + delta);
60        if let (Some(anchor), Some(after_globe)) = (anchor, active_globe_camera(style, view_state))
61        {
62            if let Some(under) = after_globe.screen_point_to_location(pivot) {
63                let update = crate::projection::globe::interaction::pan_center_to_anchor(
64                    after_globe.center(),
65                    after_globe.bearing_degrees(),
66                    anchor,
67                    under,
68                );
69                let zoom = view_state.zoom();
70                let world_center = crate::coords::WorldCoords::from_lat_lon(update.center, zoom);
71                view_state
72                    .camera_mut()
73                    .move_to(Point2::new(world_center.x, world_center.y));
74            }
75        }
76        return;
77    }
78
79    let plane = view_state.center_elevation();
80    let before = view_state
81        .inverted_view_projection()
82        .ok()
83        .and_then(|inv| view_state.window_to_world_at_elevation(&pivot.to_vec(), &inv, plane));
84    let current = view_state.camera().get_bearing();
85    view_state.camera_mut().set_bearing(current + delta);
86    if let Some(before) = before {
87        set_location_at_pixel(view_state, before, plane, pivot);
88    }
89}

91/// Base Auto-Tilt curve in degrees as a function of continuous zoom.
92///
93/// Smoothly transitions from `0°` (overhead) at `zoom <= 7.5` through regional (`25°` at `z9.0`),
94/// mountain range (`50°` at `z10.5`), alpine overview (`68°` at `z12.0`, matching the 3D terrain
95/// demo), valley (`73°` at `z14.0`), up to `76°` at `zoom >= 16.0`.
96pub(super) fn base_auto_tilt_degrees(zoom: f64) -> f64 {
97    const STOPS: &[(f64, f64)] = &[
98        (7.5, 0.0),
99        (9.0, 25.0),
100        (10.5, 50.0),
101        (12.0, 68.0),
102        (14.0, 73.0),
103        (16.0, 76.0),
104    ];
105    if zoom <= STOPS[0].0 {
106        return STOPS[0].1;
107    }
108    for window in STOPS.windows(2) {
109        let (z0, p0) = window[0];
110        let (z1, p1) = window[1];
111        if zoom <= z1 {
112            let t = ((zoom - z0) / (z1 - z0)).clamp(0.0, 1.0);
113            let s = t * t * (3.0 - 2.0 * t);
114            return p0 + (p1 - p0) * s;
115        }
116    }
117    STOPS[STOPS.len() - 1].1
118}

120/// Normalizes an angle in radians to `[-PI, PI]`.
121pub fn normalize_angle_signed(angle: f64) -> f64 {
122    let two_pi = std::f64::consts::TAU;
123    let mut a = (angle + std::f64::consts::PI).rem_euclid(two_pi) - std::f64::consts::PI;
124    if a <= -std::f64::consts::PI {
125        a += two_pi;
126    }
127    a
