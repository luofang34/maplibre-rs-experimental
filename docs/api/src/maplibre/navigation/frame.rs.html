26/// Width of the band along each viewport edge that scrolls the map, in logical pixels.
27const EDGE_ZONE_PX: f64 = 24.0;

29impl NavigationState {
30    /// Captures the home camera on the first frame, advances the camera with
31    /// [`advance`](Self::advance), and syncs `NavigationHudState` into `world.resources` for
32    /// the GPU HUD overlay.
33    pub fn update_frame(&mut self, map_context: &mut MapContext, dt: Duration) {
34        self.ensure_initialized(map_context);
35        self.advance(&map_context.style, &mut map_context.view_state, dt);
36        self.sync_hud(map_context);
37    }

39    /// Moves the camera by `dt` of HUD compass drag, trackpad rotation, North-Lock, smooth
40    /// settle animations, and edge scrolling. A host that draws its own controls calls this
41    /// each frame while [`wants_frame`](Self::wants_frame) holds.
42    pub fn advance(&mut self, style: &Style, view: &mut ViewState, dt: Duration) {
43        self.update_hover(view);
44        self.apply_compass_drag(view);
45        if self.pending_trackpad_rotation.0.abs() > 1e-6 {
46            let delta = std::mem::replace(&mut self.pending_trackpad_rotation, Rad(0.0));
47            if !self.north_locked {
48                rotate_bearing_around_cursor(style, view, self.cursor_position, delta);
49            }
50        }
51        if self.north_locked && normalize_angle_signed(view.camera().get_bearing().0).abs() > 1e-4 {
52            set_bearing_keeping_center(view, Rad(0.0));
53        }
54        let alpha = (1.0 - (-dt.as_secs_f64() * 14.0).exp()).clamp(0.05, 1.0);
55        self.advance_fly_home(style, view, alpha);
56        self.advance_bearing(view, alpha);
57        self.advance_pitch(view, alpha);
58        self.edge_scroll(style, view, dt);
59    }

61    /// Hit-tests the cursor against the HUD with the viewport's actual size and edge insets.
62    fn update_hover(&mut self, view: &ViewState) {
63        let Some(cursor) = self.cursor_position else {
64            return;
65        };
66        self.hovered_button = if self.show_hud {
67            NavigationHudState::hit_test(cursor, view.width(), view.height(), view.edge_insets())
68        } else {
69            None
70        };
71    }

73    /// Applies the pointer movement of a HUD compass drag.
74    fn apply_compass_drag(&mut self, view: &mut ViewState) {
75        if self.pressed_button != Some(HudButton::Compass) || !self.compass_dragged {
76            return;
77        }
78        let (Some(cursor), Some(last)) = (self.cursor_position, self.compass_drag_last) else {
79            return;
80        };
81        self.compass_drag_last = Some(cursor);
82        self.drag_compass(view, cursor - last);
83    }

85    /// Turns the bearing with the horizontal and the pitch with the vertical component of a
86    /// compass drag of `delta` logical pixels, for hosts that draw their own compass.
87    pub fn drag_compass(&mut self, view: &mut ViewState, delta: Vector2<f64>) {
88        if delta.magnitude2() <= f64::EPSILON {
89            return;
90        }
91        if !self.north_locked && delta.x.abs() > 0.0 {
92            let bearing = view.camera().get_bearing();
93            set_bearing_keeping_center(view, bearing - Rad::from(Deg(delta.x * 0.65)));
94            self.on_manual_bearing();
95        }
96        if delta.y.abs() > 0.0 {
97            let max_pitch = view.camera().max_pitch().0.to_degrees();
98            let pitch = view.camera().get_pitch().0.to_degrees();
99            self.on_manual_tilt();
100            set_pitch_keeping_center(view, Deg((pitch - delta.y * 0.5).clamp(0.0, max_pitch)));
101        }
102    }

104    /// Eases zoom and center toward the home camera.
105    fn advance_fly_home(&mut self, style: &Style, view: &mut ViewState, alpha: f64) {
106        if !self.fly_home {
107            return;
108        }
109        let Some(home) = self.home else {
110            self.fly_home = false;
111            return;
112        };
113        let current_zoom = view.zoom().value();
114        let next_zoom = current_zoom + (home.zoom - current_zoom) * alpha;
115        let center = center_pixel(view);
116        // On Mercator views there is no globe to zoom around; `zoom_to` below covers both.
117        zoom_globe_around_pixel_with_pitch(style, view, center, Zoom::new(next_zoom), None);
118        view.zoom_to(Zoom::new(next_zoom));
119        let world_size = TILE_SIZE * 2.0_f64.powf(view.zoom().value());
120        let current = view.camera().position();
121        let target = home.normalized_center * world_size;
122        let next = current + (target - current) * alpha;
123        view.camera_mut().move_to(next);
124        if (home.zoom - next_zoom).abs() < 0.01 && (target - next).magnitude() < 0.5 {
125            view.zoom_to(Zoom::new(home.zoom));
126            let world_size = TILE_SIZE * 2.0_f64.powf(home.zoom);
127            view.camera_mut()
128                .move_to(home.normalized_center * world_size);
129            self.fly_home = false;
130        }
131    }

133    /// Eases the bearing toward its target the short way round.
134    fn advance_bearing(&mut self, view: &mut ViewState, alpha: f64) {
135        let Some(target) = self.target_bearing_rad else {
136            return;
137        };
138        let current = view.camera().get_bearing().0;
139        let diff = normalize_angle_signed(target - current);
140        if diff.abs() <= 0.0015 {
141            set_bearing_keeping_center(view, Rad(target));
142            self.target_bearing_rad = None;
143        } else {
144            set_bearing_keeping_center(view, Rad(current + diff * alpha));
145        }
146    }
147
148    /// Eases the pitch toward its target, or back onto the Auto-Tilt curve once terrain
149    /// clearance allows, such as after panning off a steep ridge over a valley.
150    fn advance_pitch(&mut self, view: &mut ViewState, alpha: f64) {
151        let max_pitch = view.camera().max_pitch().0.to_degrees();
152        let current = view.camera().get_pitch().0.to_degrees();
153        if let Some(target) = self.target_pitch_deg {
154            let target = target.clamp(0.0, max_pitch);
155            let diff = target - current;
156            if diff.abs() <= 0.08 {
157                set_pitch_keeping_center(view, Deg(target));
158                self.target_pitch_deg = None;
159            } else {
160                set_pitch_keeping_center(view, Deg(current + diff * alpha));
161            }
162        } else if self.three_dimensional && self.auto_tilt && !self.fly_home {
163            let desired = self
164                .pitch_for_zoom(view.zoom().value())
165                .clamp(0.0, max_pitch);
166            let diff = desired - current;
167            if diff.abs() > super::modes::AUTO_TILT_TOLERANCE_DEG {
168                set_pitch_keeping_center(view, Deg(current + diff * alpha));
169            }
170        }
171    }

173    /// Scrolls the map while the cursor rests near a viewport edge in the RTS profile, or
174    /// whenever edge scrolling is enabled.
175    fn edge_scroll(&mut self, style: &Style, view: &mut ViewState, dt: Duration) {
176        if !(self.pointer_profile == PointerProfile::Rts || self.edge_scroll_override)
177            || self.pressed_button.is_some()
178            || self.selection_drag.is_some()
179        {
180            return;
181        }
182        let Some(cursor) = self.cursor_position else {
183            return;
184        };
185        let (w, h) = (view.width(), view.height());
186        if w <= 100.0
187            || h <= 100.0
188            || !(0.0..=w).contains(&cursor.x)
189            || !(0.0..=h).contains(&cursor.y)
190            || self.is_over_hud(view, cursor)
191        {
192            return;
193        }
194        let velocity = Vector2::new(edge_velocity(cursor.x, w), edge_velocity(cursor.y, h));
195        if velocity.x == 0.0 && velocity.y == 0.0 {
196            return;
197        }
198        let boost = if self.shift_held { 1.8 } else { 1.0 };
199        let step =
200            velocity * (self.edge_scroll_speed_px_per_sec.max(0.0) * boost * dt.as_secs_f64());
201        let center = center_pixel(view);
202        let plane = view.center_elevation();
203        if !pan_globe_by_pixels(style, view, center, -step) {
204            pan_plane_by_pixels(view, center, -step, plane);
205        }
206    }

208    /// Publishes the state the GPU HUD overlay draws.
209    fn sync_hud(&self, map_context: &mut MapContext) {
210        let hud = map_context
211            .world
212            .resources
213            .get_or_init_mut::<NavigationHudState>();
214        hud.visible = self.show_hud;
215        hud.three_dimensional = self.three_dimensional;
216        hud.auto_tilt = self.auto_tilt;
217        hud.north_locked = self.north_locked;
218        hud.rts_mode = self.pointer_profile == PointerProfile::Rts;
219        hud.hovered = self.hovered_button;
220        hud.pressed = self.pressed_button;
221        hud.selection_box = self.selection_drag.and_then(|(start, end)| {
222            ((end - start).magnitude() >= 3.0).then_some([
223                start.x.min(end.x) as f32,
224                start.y.min(end.y) as f32,
225                start.x.max(end.x) as f32,
226                start.y.max(end.y) as f32,
227            ])
228        });
229    }

232/// Edge-scroll velocity along one axis in `-1..=1`, easing in quadratically across the band.
233fn edge_velocity(position: f64, extent: f64) -> f64 {
234    if position < EDGE_ZONE_PX {
235        let t = ((EDGE_ZONE_PX - position) / EDGE_ZONE_PX).clamp(0.0, 1.0);
236        -t * t
237    } else if position > extent - EDGE_ZONE_PX {
238        let t = ((position - (extent - EDGE_ZONE_PX)) / EDGE_ZONE_PX).clamp(0.0, 1.0);
239        t * t
240    } else {
241        0.0
242    }
