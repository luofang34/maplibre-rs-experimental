14/// Pitch difference in degrees below which Auto-Tilt leaves the camera alone.
15pub(super) const AUTO_TILT_TOLERANCE_DEG: f64 = 0.12;
16
17impl NavigationState {
18    /// Captures the style's initial camera state on the first frame and enables Auto-Tilt by
19    /// default on 3D terrain and Globe styles.
20    pub fn ensure_initialized(&mut self, map_context: &mut MapContext) {
21        if self.initialized {
22            return;
23        }
24        self.initialized = true;
25        let has_terrain = map_context.style.terrain.is_some();
26        let has_globe = map_context
27            .style
28            .projection
29            .as_ref()
30            .is_some_and(|p| p.projection_type != ProjectionType::Mercator);
31        let style_pitch = map_context.style.pitch.unwrap_or(0.0);
32        let has_3d_or_globe = has_terrain || has_globe || style_pitch > 0.5;
33        let auto_tilt_by_default = has_terrain || (has_globe && style_pitch <= 0.5);
34
35        let zoom = map_context.view_state.zoom().value();
36        let world_size = TILE_SIZE * 2.0_f64.powf(zoom);
37        let pos = map_context.view_state.camera().position();
38        let normalized_center = Point2::new(pos.x / world_size, pos.y / world_size);
39        let bearing = map_context.view_state.camera().get_bearing();
40        let initial_pitch_deg = if auto_tilt_by_default {
41            base_auto_tilt_degrees(zoom)
42        } else {
43            map_context.view_state.camera().get_pitch().0.to_degrees()
44        };
45        let pitch = Rad::from(Deg(initial_pitch_deg));
46
47        self.home = Some(HomeCamera {
48            normalized_center,
49            zoom,
50            bearing,
51            pitch,
52            has_3d_or_globe,
53        });
54        self.three_dimensional = true;
55        self.auto_tilt = auto_tilt_by_default;
56
57        if let Some(hud) = map_context.world.resources.get::<NavigationHudState>() {
58            self.show_hud = hud.visible;
59            self.edge_scroll_speed_px_per_sec = hud.edge_scroll_speed_px_per_sec;
60            self.edge_scroll_override = hud.edge_scroll_enabled;
61        }
62
63        // Sync initial camera pitch to the Auto-Tilt curve when Auto-Tilt is active so zooming
64        // both in and out from the starting zoom has immediate, symmetric tilt response.
65        if self.auto_tilt {
66            let max_pitch = map_context.view_state.camera().max_pitch().0.to_degrees();
67            map_context
68                .view_state
69                .camera_mut()
70                .set_pitch(Deg(initial_pitch_deg.min(max_pitch)));
71        }
72        tracing::info!(
73            zoom,
74            initial_pitch_deg,
75            max_pitch_deg = map_context.view_state.camera().max_pitch().0.to_degrees(),
76            three_d = self.three_dimensional,
77            auto_tilt = self.auto_tilt,
78            "NavigationState initialized"
79        );
80    }

82    /// Computes the Auto-Tilt pitch in degrees for a given `zoom`.
83    ///
84    /// Uses C1-continuous Hermite smoothstep interpolation across zoom stops:
85    /// `0°` at `zoom <= 8.5` (overhead globe/regional view), ramping through `22°` (`z10`),
86    /// `48°` (`z12`), `64°` (`z13.5`), and `74°` (`z15`) up to `78°` (`z >= 16.5`).
87    pub fn pitch_for_zoom(&self, zoom: f64) -> f64 {
88        if !self.three_dimensional {
89            return 0.0;
90        }
91        base_auto_tilt_degrees(zoom)
92    }
93
94    /// Updates the smooth Auto-Tilt target pitch whenever zoom changes.
95    pub fn on_zoom_changed(&mut self, next_zoom: f64) {
96        self.fly_home = false;
97        if self.three_dimensional && self.auto_tilt {
98            self.target_pitch_deg = Some(self.pitch_for_zoom(next_zoom));
99        }
100    }

102    /// Accumulates a trackpad two-finger rotation gesture (`WindowEvent::RotationGesture`,
103    /// where `delta_degrees` is positive for counter-clockwise finger rotation).
104    pub fn process_trackpad_rotate(&mut self, delta_degrees: f64) {
105        if !self.north_locked && delta_degrees.is_finite() && delta_degrees.abs() > 1e-4 {
106            self.on_manual_bearing();
107            self.pending_trackpad_rotation += Rad::from(Deg(delta_degrees));
108        }
109    }

111    /// Called when the user manually tilts the camera (e.g. Right-drag, `Alt`+drag, two-finger
112    /// vertical drag, or Compass vertical drag). Disengages `Auto-Tilt` into `3D·MAN` mode.
113    pub fn on_manual_tilt(&mut self) {
114        self.three_dimensional = true;
115        self.auto_tilt = false;
116        self.target_pitch_deg = None;
117        self.fly_home = false;
118    }

123        self.fly_home = false;
124    }

131    /// Cycles the single `3D/2D/Auto` control button:
132    /// - `3D·AUTO` -> `2D` (smoothly flattens pitch to `0°`)
133    /// - `2D` -> `3D·AUTO` (restores Auto-Tilt pitch, or at least `55°` at low zoom so the
134    ///   button always gives visible 3D feedback)
135    /// - `3D·MAN` -> `3D·AUTO` (re-engages Auto-Tilt along the zoom curve)
136    pub fn cycle_3d_mode(&mut self, current_zoom: f64) {
137        self.fly_home = false;
138        let curve_pitch = base_auto_tilt_degrees(current_zoom);
139        if self.three_dimensional && self.auto_tilt {
140            if curve_pitch < 15.0 {
141                // At low zoom where Auto-Tilt is flat (0°), clicking `3D` tilts immediately
142                // into manual 3D (55°) so the user always sees 3D tilt when pressing the button.
143                self.three_dimensional = true;
144                self.auto_tilt = false;
145                self.target_pitch_deg = Some(55.0);
146            } else {
147                // 3D·AUTO -> 2D
148                self.three_dimensional = false;
149                self.auto_tilt = false;
150                self.target_pitch_deg = Some(0.0);
151            }
152        } else if !self.three_dimensional {
153            // 2D -> 3D·AUTO
154            self.three_dimensional = true;
155            self.auto_tilt = true;
156            self.target_pitch_deg = Some(curve_pitch.max(55.0));
157        } else {
158            // 3D·MAN -> 3D·AUTO
159            self.three_dimensional = true;
160            self.auto_tilt = true;
161            self.target_pitch_deg = Some(curve_pitch);
162        }
163        tracing::info!(
164            current_zoom,
165            three_d = self.three_dimensional,
166            auto_tilt = self.auto_tilt,
167            target_pitch = ?self.target_pitch_deg,
168            "cycled 3D mode"
169        );
170    }

172    /// Handles a click on the Compass button: resets bearing to North (`0°`) if rotated,
173    /// or toggles `north_locked` if already facing North.
174    pub fn trigger_compass(&mut self, current_bearing: Rad<f64>) {
175        self.fly_home = false;
176        let normalized = normalize_angle_signed(current_bearing.0);
177        if normalized.abs() > 0.01 {
178            self.target_bearing_rad = Some(0.0);
179        } else {
180            self.north_locked = !self.north_locked;
181            self.target_bearing_rad = Some(0.0);
182        }
183    }

185    /// Triggers a smooth reset back to the initial `HomeCamera` view and restores default Auto-Tilt.
186    pub fn trigger_home(&mut self) {
187        let Some(home) = self.home else {
188            return;
189        };
190        self.three_dimensional = true;
191        self.auto_tilt = home.has_3d_or_globe;
192        self.north_locked = false;
193        self.target_bearing_rad = Some(normalize_angle_signed(home.bearing.0));
194        self.target_pitch_deg = Some(home.pitch.0.to_degrees());
195        self.fly_home = true;
196    }

198    /// Toggles between `PointerProfile::Map` and `PointerProfile::Rts`.
199    pub fn toggle_pointer_profile(&mut self) {
200        self.pointer_profile = match self.pointer_profile {
201            PointerProfile::Map => PointerProfile::Rts,
202            PointerProfile::Rts => PointerProfile::Map,
203        };
204        self.selection_drag = None;
205    }

207    /// Takes any pending smooth zoom delta queued by HUD `+`/`-` buttons or double-click.
208    pub fn take_pending_smooth_zoom(&mut self) -> Option<f64> {
209        if self.pending_smooth_zoom.abs() > f64::EPSILON {
210            let delta = self.pending_smooth_zoom;
211            self.pending_smooth_zoom = 0.0;
212            Some(delta)
213        } else {
214            None
215        }
216    }
217    /// Cancels any pending smooth pitch settle animation (for instance when terrain clearance
218    /// clamps the camera pitch on a steep slope).
219    pub fn cancel_pitch_animation(&mut self) {
220        self.target_pitch_deg = None;
221    }

223    /// Whether any smooth camera animation (`target_pitch`, `target_bearing`, `fly_home`) is active.
224    pub fn is_animating(&self) -> bool {
225        self.target_pitch_deg.is_some() || self.target_bearing_rad.is_some() || self.fly_home
226    }
227
228    /// Whether [`update_frame`](Self::update_frame) still moves the camera: an animation runs,
229    /// or Auto-Tilt is easing the pitch back onto its curve. A host that redraws on demand keeps
230    /// drawing frames while this holds.
231    pub fn wants_frame(&self, view: &ViewState) -> bool {
232        if self.is_animating() {
233            return true;
234        }
235        if !(self.three_dimensional && self.auto_tilt) {
236            return false;
237        }
238        let max_pitch = view.camera().max_pitch().0.to_degrees();
239        let desired = self
240            .pitch_for_zoom(view.zoom().value())
241            .clamp(0.0, max_pitch);
242        (desired - view.camera().get_pitch().0.to_degrees()).abs() > AUTO_TILT_TOLERANCE_DEG
243    }
244
245    /// Stops every running camera animation, leaving the camera where it is, for instance when
246    /// the host hands the camera to another source.
247    pub fn stop_animations(&mut self) {
248        self.target_pitch_deg = None;
249        self.target_bearing_rad = None;
250        self.fly_home = false;
251        self.pending_smooth_zoom = 0.0;
252    }
253
254    /// Couples the pitch to zoom along the Auto-Tilt curve, easing onto it, or keeps the
255    /// current pitch as a manual tilt.
256    pub fn set_auto_tilt(&mut self, enabled: bool, current_zoom: f64) {
257        self.fly_home = false;
258        if enabled {
259            self.three_dimensional = true;
260            self.auto_tilt = true;
261            self.target_pitch_deg = Some(self.pitch_for_zoom(current_zoom));
262        } else {
263            self.on_manual_tilt();
264        }
265    }
266
267    /// Eases to a manually chosen pitch in degrees, leaving `Auto-Tilt` for `3D·MAN`.
268    pub fn set_manual_pitch(&mut self, degrees: f64) {
269        self.on_manual_tilt();
270        self.target_pitch_deg = Some(degrees.max(0.0));
271    }
