16const SHADER: &str = include_str!("../render/shaders/overlay.wgsl");

18/// Share by which a vertex's or probe's reversed-Z depth is raised before it is tested against
19/// the frame's depth. That brings it nearer by about `DEPTH_BIAS * (1 - distance / far)` of
20/// its distance from the camera: the full share close to the camera, less towards the far
21/// plane. Overlay lines are straight chords between points on the ground, a few pixels apart,
22/// so over a convex slope they run a little under the drawn surface; this keeps them from
23/// hiding what lies on the ground, while a ridge hides what lies behind it by far more.
24pub const DEPTH_BIAS: f32 = 0.01;
25
26/// The mesh last uploaded: its reusable buffers, capacities, index count, and revision.
27pub(super) struct MeshBuffers {
28    pub(super) vertices: wgpu::Buffer,
29    vertex_capacity: u64,
30    pub(super) indices: wgpu::Buffer,
31    index_capacity: u64,
32    pub(super) count: u32,
33    revision: u64,
34}

36const MIN_OVERLAY_BUFFER_BYTES: u64 = 4096;

38/// What the overlay pass draws with.
39pub(crate) struct OverlayGpu {
40    pub(super) pipeline: wgpu::RenderPipeline,
41    pub(super) bind_group: wgpu::BindGroup,
42    viewport: wgpu::Buffer,
43    size: Option<(u32, u32)>,
44    pub(super) mesh: Option<MeshBuffers>,
45}
46
47impl OverlayGpu {
48    fn new(
49        device: &wgpu::Device,
50        queue: &UploadQueue,
51        format: wgpu::TextureFormat,
52        depth_format: wgpu::TextureFormat,
53        samples: u32,
54    ) -> Self {
55        let viewport = queue.create_buffer_init(
56            device,
57            &BufferInitDescriptor {
58                label: Some("overlay viewport"),
59                contents: bytemuck::cast_slice(&[1.0_f32, 1.0, DEPTH_BIAS, 0.0]),
60                usage: wgpu::BufferUsages::UNIFORM | wgpu::BufferUsages::COPY_DST,
61            },
62        );
63        let layout = device.create_bind_group_layout(&wgpu::BindGroupLayoutDescriptor {
64            label: Some("overlay"),
65            entries: &[wgpu::BindGroupLayoutEntry {
66                binding: 0,
67                visibility: wgpu::ShaderStages::VERTEX,
68                ty: wgpu::BindingType::Buffer {
69                    ty: wgpu::BufferBindingType::Uniform,
70                    has_dynamic_offset: false,
71                    min_binding_size: None,
72                },
73                count: None,
74            }],
75        });
76        let bind_group = device.create_bind_group(&wgpu::BindGroupDescriptor {
77            label: Some("overlay"),
78            layout: &layout,
79            entries: &[wgpu::BindGroupEntry {
80                binding: 0,
81                resource: viewport.as_entire_binding(),
82            }],
83        });
84        let pipeline_layout = device.create_pipeline_layout(&wgpu::PipelineLayoutDescriptor {
85            label: Some("overlay"),
86            bind_group_layouts: &[Some(&layout)],
87            immediate_size: 0,
88        });
89        let module = device.create_shader_module(wgpu::ShaderModuleDescriptor {
90            label: Some("overlay"),
91            source: wgpu::ShaderSource::Wgsl(SHADER.into()),
92        });
93        let pipeline = device.create_render_pipeline(&wgpu::RenderPipelineDescriptor {
94            label: Some("overlay"),
95            layout: Some(&pipeline_layout),
96            vertex: wgpu::VertexState {
97                module: &module,
98                entry_point: Some("vs"),
99                compilation_options: Default::default(),
100                buffers: &[Some(wgpu::VertexBufferLayout {
101                    array_stride: std::mem::size_of::<OverlayVertex>() as wgpu::BufferAddress,
102                    step_mode: wgpu::VertexStepMode::Vertex,
103                    attributes: &wgpu::vertex_attr_array![0 => Float32x2, 1 => Float32, 2 => Float32x4],
104                })],
105            },
106            fragment: Some(wgpu::FragmentState {
107                module: &module,
108                entry_point: Some("fs"),
109                compilation_options: Default::default(),
110                targets: &[Some(wgpu::ColorTargetState {
111                    format,
112                    blend: Some(wgpu::BlendState::PREMULTIPLIED_ALPHA_BLENDING),
113                    write_mask: wgpu::ColorWrites::ALL,
114                })],
115            }),
116            primitive: wgpu::PrimitiveState::default(),
117            // Tested as the map's 3D geometry is (reversed Z), never written: content of the
118            // overlay does not hide other content of it.
119            depth_stencil: Some(wgpu::DepthStencilState {
120                format: depth_format,
121                depth_write_enabled: Some(false),
122                depth_compare: Some(wgpu::CompareFunction::GreaterEqual),
123                stencil: wgpu::StencilState::default(),
124                bias: wgpu::DepthBiasState::default(),
125            }),
126            multisample: wgpu::MultisampleState {
127                count: samples,
128                ..Default::default()
129            },
130            multiview_mask: None,
131            cache: None,
132        });
133        Self {
134            pipeline,
135            bind_group,
136            viewport,
137            size: None,
138            mesh: None,
139        }
140    }
141
142    /// Tells the shader the render target's size when it changes.
143    fn set_size(&mut self, queue: &UploadQueue, size: &PhysicalSize) {
144        let wanted = (size.width(), size.height());
145        if self.size == Some(wanted) {
146            return;
147        }
148        let data = [wanted.0 as f32, wanted.1 as f32, DEPTH_BIAS, 0.0];
149        queue.write_buffer(&self.viewport, 0, bytemuck::cast_slice(&data));
150        self.size = Some(wanted);
151    }
152
153    /// Uploads the host's mesh when it changed since the last upload, reusing GPU buffers
154    /// whenever their capacity already fits the new geometry.
155    fn upload(&mut self, device: &wgpu::Device, queue: &UploadQueue, overlay: &HostOverlay) {
156        if self.mesh.as_ref().map(|m| m.revision) == Some(overlay.revision()) {
157            return;
158        }
159        let mesh = overlay.mesh();
160        // Whole triangles only: a trailing partial triangle would read past the mesh.
161        let count = mesh.indices.len() - mesh.indices.len() % 3;
162        if mesh.vertices.is_empty() || count == 0 {
163            if let Some(existing) = self.mesh.as_mut() {
164                existing.count = 0;
165                existing.revision = overlay.revision();
166            }
167            return;
168        }
169        let v_bytes: &[u8] = bytemuck::cast_slice(&mesh.vertices);
170        let i_bytes: &[u8] = bytemuck::cast_slice(mesh.indices.get(..count).unwrap_or_default());
171        let grow = |needed: usize| -> u64 {
172            (needed as u64)
173                .max(MIN_OVERLAY_BUFFER_BYTES)
174                .checked_next_power_of_two()
175                .unwrap_or(needed as u64)
176        };
177        let alloc = |label, capacity: u64, usage: wgpu::BufferUsages| {
178            device.create_buffer(&wgpu::BufferDescriptor {
179                label: Some(label),
180                size: capacity,
181                usage: usage | wgpu::BufferUsages::COPY_DST,
182                mapped_at_creation: false,
183            })
184        };
185        let buffers = match self.mesh.as_mut() {
186            Some(existing) => {
187                if v_bytes.len() as u64 > existing.vertex_capacity {
188                    existing.vertex_capacity = grow(v_bytes.len());
189                    existing.vertices = alloc(
190                        "overlay vertices",
191                        existing.vertex_capacity,
192                        wgpu::BufferUsages::VERTEX,
193                    );
194                }
195                if i_bytes.len() as u64 > existing.index_capacity {
196                    existing.index_capacity = grow(i_bytes.len());
197                    existing.indices = alloc(
198                        "overlay indices",
199                        existing.index_capacity,
200                        wgpu::BufferUsages::INDEX,
201                    );
202                }
203                existing
204            }
205            None => {
206                let vertex_capacity = grow(v_bytes.len());
207                let index_capacity = grow(i_bytes.len());
208                self.mesh.insert(MeshBuffers {
209                    vertices: alloc(
210                        "overlay vertices",
211                        vertex_capacity,
212                        wgpu::BufferUsages::VERTEX,
213                    ),
214                    vertex_capacity,
215                    indices: alloc("overlay indices", index_capacity, wgpu::BufferUsages::INDEX),
216                    index_capacity,
217                    count: 0,
218                    revision: 0,
219                })
220            }
221        };
222        queue.write_buffer(&buffers.vertices, 0, v_bytes);
223        queue.write_buffer(&buffers.indices, 0, i_bytes);
224        buffers.count = count as u32;
225        buffers.revision = overlay.revision();
226    }
227}

229/// Builds the pipeline once the render target exists, then keeps the size and mesh current.
230pub(super) fn prepare_system(
231    MapContext {
232        world,
233        renderer:
234            Renderer {
235                device,
236                queue,
237                resources: RenderResources { surface, .. },
238                settings,
239                ..
240            },
241        ..
242    }: &mut MapContext,
243) -> SystemResult {
244    let Some((gpu, overlay)) = world
245        .resources
246        .query_mut::<(&mut Eventually<OverlayGpu>, &HostOverlay)>()
247    else {
248        return Err(SystemError::Dependencies);
249    };
250    let samples = if surface.is_multisampling_supported(settings.msaa) {
251        settings.msaa
252    } else {
253        Msaa { samples: 1 }
254    };
255    let gpu = gpu.initialize(|| {
256        OverlayGpu::new(
257            device,
258            queue,
259            surface.surface_format(),
260            settings.depth_texture_format,
261            samples.samples,
262        )
263    });
264    gpu.set_size(queue, &surface.size());
265    gpu.upload(device, queue, overlay);
266    Ok(())
267}
