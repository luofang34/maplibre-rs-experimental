1//! The one loader every request a map makes goes through, whichever thread makes it.
2//!
3//! A host injects a loader once; the map thread fetches TileJSON with it and native tile
4//! workers fetch tiles, glyphs and sprites with clones of the same handle. Any
5//! [`HttpClient`] is a loader, so a host can pass its own transport or an in-process fake;
6//! `pmtiles://` URLs are read out of their archive with range requests through it.
7

9    collections::HashMap,

12};

22/// Fetches whole resources and byte ranges of them.
23///
24/// Implemented by every [`HttpClient`]; a loader that is not `Clone` implements it directly.
25#[cfg_attr(not(feature = "thread-safe-futures"), async_trait(?Send))]
26#[cfg_attr(feature = "thread-safe-futures", async_trait)]
27pub trait ResourceLoader: Send + Sync + 'static {
28    /// The resource's body.
29    async fn load(&self, url: &str) -> Result<Vec<u8>, SourceFetchError>;
30    /// `range` of the resource's body.
31    async fn load_range(&self, url: &str, range: ByteRange) -> Result<Vec<u8>, SourceFetchError>;
32}
33
34#[cfg_attr(not(feature = "thread-safe-futures"), async_trait(?Send))]
35#[cfg_attr(feature = "thread-safe-futures", async_trait)]
36impl<T: HttpClient> ResourceLoader for T {
37    async fn load(&self, url: &str) -> Result<Vec<u8>, SourceFetchError> {
38        self.fetch(url).await
39    }
40
41    async fn load_range(&self, url: &str, range: ByteRange) -> Result<Vec<u8>, SourceFetchError> {
42        self.fetch_range(url, range).await
43    }
44}

46/// A loader shared by the map thread and its workers; clones share one loader.
47///
48/// Requests for the same bytes while one is in flight wait for it instead of fetching again.
49/// If the request they wait for is cancelled, the next of them fetches in its place.
50#[derive(Clone)]
51pub struct SharedLoader(Arc<Inner>);
52
53struct Inner {
54    loader: Box<dyn ResourceLoader>,
55    in_flight: Mutex<HashMap<Key, Vec<oneshot::Sender<Outcome>>>>,
56    pmtiles: PmtilesReader,
57}
58
59type Key = (String, Option<ByteRange>);
60
61/// What the request in flight delivers to the requests waiting for it.
62#[derive(Clone)]
63enum Outcome {
64    Body(Arc<[u8]>),
65    Failed(Arc<Failure>),
66}
67
68/// A failure another request met, for a request that waited on it.
69#[derive(Debug, thiserror::Error)]
70#[error("{description}")]
71struct Failure {
72    description: String,
73    not_found: bool,
74    retryable: bool,
75}
76
77impl Outcome {
78    fn new(result: &Result<Vec<u8>, SourceFetchError>) -> Self {
79        match result {
80            Ok(body) => Self::Body(Arc::from(body.as_slice())),
81            Err(error) => Self::Failed(Arc::new(Failure {
82                description: error.describe(),
83                not_found: error.is_not_found(),
84                retryable: error.is_retryable(),
85            })),
86        }
87    }
88
89    fn into_result(self, url: &str) -> Result<Vec<u8>, SourceFetchError> {
90        match self {
91            Self::Body(body) => Ok(body.to_vec()),
92            Self::Failed(failure) if failure.not_found => Err(SourceFetchError::not_found(url)),
93            Self::Failed(failure) if failure.retryable => {
94                Err(SourceFetchError::temporary(SharedFailure(failure)))
95            }
96            Self::Failed(failure) => Err(SourceFetchError(Box::new(SharedFailure(failure)))),
97        }
98    }
99}
100
101#[derive(Debug, thiserror::Error)]
102#[error(transparent)]
103struct SharedFailure(Arc<Failure>);
104
105/// Removes the request from the in-flight table however its fetch ends; dropped by a
106/// cancellation, it drops the waiters' senders, which wakes them to fetch for themselves.
107struct Leader<'a> {
108    inner: &'a Inner,
109    key: Key,
110    /// Set once the waiters are answered, after which a new request may own the key.
111    finished: bool,
112}
113
114impl Leader<'_> {
115    fn finish(mut self, outcome: Outcome) {
116        self.finished = true;
117        for waiter in self.inner.take(&self.key) {
118            waiter.send(outcome.clone()).ok();
119        }
120    }
121}
122
123impl Drop for Leader<'_> {
124    fn drop(&mut self) {
125        if !self.finished {
126            self.inner.take(&self.key);
127        }
128    }
129}
130
131impl Inner {
132    fn table(&self) -> std::sync::MutexGuard<'_, HashMap<Key, Vec<oneshot::Sender<Outcome>>>> {
133        self.in_flight
134            .lock()
135            .unwrap_or_else(PoisonError::into_inner)
136    }
137
138    fn take(&self, key: &Key) -> Vec<oneshot::Sender<Outcome>> {
139        self.table().remove(key).unwrap_or_default()
140    }
141
142    async fn load(&self, url: &str, range: Option<ByteRange>) -> Result<Vec<u8>, SourceFetchError> {
143        let key = (url.to_owned(), range);
144        loop {
145            let waiting = {
146                let mut table = self.table();
147                match table.get_mut(&key) {
148                    Some(waiters) => {
149                        let (sender, receiver) = oneshot::channel();
150                        waiters.push(sender);
151                        Some(receiver)
152                    }
153                    None => {
154                        table.insert(key.clone(), Vec::new());
155                        None
156                    }
157                }
158            };
159            match waiting {
160                Some(receiver) => match receiver.await {
161                    Ok(outcome) => return outcome.into_result(url),
162                    // The request waited on was cancelled; the next fetches in its place.
163                    Err(oneshot::Canceled) => continue,
164                },
165                None => {
166                    let leader = Leader {
167                        inner: self,
168                        key: key.clone(),
169                        finished: false,
170                    };
171                    let result = match range {
172                        Some(range) => self.loader.load_range(url, range).await,
173                        None => self.loader.load(url).await,
174                    };
175                    leader.finish(Outcome::new(&result));
176                    return result;
177                }
178            }
179        }
180    }
181}

183impl SharedLoader {
184    /// Shares `loader` with everything that receives a clone.
185    pub fn new(loader: impl ResourceLoader) -> Self {
186        Self(Arc::new(Inner {
187            loader: Box::new(loader),
188            in_flight: Mutex::default(),
189            pmtiles: PmtilesReader::default(),
190        }))
191    }
192}

194impl fmt::Debug for SharedLoader {
195    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
196        f.write_str("SharedLoader")
197    }
198}

200#[cfg_attr(not(feature = "thread-safe-futures"), async_trait(?Send))]
201#[cfg_attr(feature = "thread-safe-futures", async_trait)]
202impl HttpClient for SharedLoader {
203    async fn fetch(&self, url: &str) -> Result<Vec<u8>, SourceFetchError> {
204        // An archive's range requests come back through this loader, so concurrent readers of
205        // one archive share them.
206        match PmtilesUrl::parse(url) {
207            Some(request) => self.0.pmtiles.fetch(self, url, request).await,
208            None => self.0.load(url, None).await,
209        }
210    }
211
212    async fn fetch_range(&self, url: &str, range: ByteRange) -> Result<Vec<u8>, SourceFetchError> {
213        self.0.load(url, Some(range)).await
214    }
215}
