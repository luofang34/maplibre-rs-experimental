18/// The subset of a TileJSON document needed to address tiles.
19#[derive(Debug, PartialEq)]
20#[cfg_attr(any(test, not(target_arch = "wasm32")), derive(Deserialize))]
21pub struct TileJson {
22    /// Tile URL templates.
23    pub tiles: Vec<String>,
24    /// Lowest zoom level with tiles.
25    #[cfg_attr(any(test, not(target_arch = "wasm32")), serde(default))]
26    pub minzoom: Option<u8>,
27    /// Highest zoom level with tiles.
28    #[cfg_attr(any(test, not(target_arch = "wasm32")), serde(default))]
29    pub maxzoom: Option<u8>,
30    /// Bounds in which tiles are available.
31    #[cfg_attr(any(test, not(target_arch = "wasm32")), serde(default))]
32    pub bounds: Option<(f64, f64, f64, f64)>,
33    /// Credit the tiles' provider asks for.
34    #[cfg_attr(any(test, not(target_arch = "wasm32")), serde(default))]
35    pub attribution: Option<String>,
36    /// Y-axis convention of the tile URLs.
37    #[cfg_attr(any(test, not(target_arch = "wasm32")), serde(default))]
38    pub scheme: Option<TileAddressingScheme>,
39    /// Edge length of a tile in pixels.
40    #[cfg_attr(
41        any(test, not(target_arch = "wasm32")),
42        serde(default, rename = "tileSize")
43    )]
44    pub tile_size: Option<u32>,
45    /// Elevation packing of DEM tiles.
46    #[cfg_attr(any(test, not(target_arch = "wasm32")), serde(default))]
47    pub encoding: Option<DemEncoding>,
48}
49
50impl TileJson {
51    /// Parses a [`TileJson`] subset from UTF-8 JSON bytes without `serde::Deserialize` bloat.
52    pub fn from_json_slice(bytes: &[u8]) -> Result<Self, serde_json::Error> {
53        let value: serde_json::Value = serde_json::from_slice(bytes)?;
54        Self::from_json_value(&value)
55    }
56
57    /// Parses a [`TileJson`] subset from a parsed JSON [`serde_json::Value`].
58    pub fn from_json_value(value: &serde_json::Value) -> Result<Self, serde_json::Error> {
59        let Some(map) = value.as_object() else {
60            return Err(field_err("tilejson", "expected object"));
61        };
62        let tiles = json_opt_string_vec(map, "tiles")?.ok_or_else(|| missing_field_err("tiles"))?;
63        let scheme = match json_opt_str(map, "scheme")? {
64            None => None,
65            Some("xyz") => Some(TileAddressingScheme::XYZ),
66            Some("tms") => Some(TileAddressingScheme::TMS),
67            Some(_) => return Err(field_err("scheme", "expected `xyz` or `tms`")),
68        };
69        let encoding = match json_opt_str(map, "encoding")? {
70            None => None,
71            Some("mapbox") => Some(DemEncoding::Mapbox),
72            Some("terrarium") => Some(DemEncoding::Terrarium),
73            Some("custom") => Some(DemEncoding::Custom),
74            Some(_) => {
75                return Err(field_err(
76                    "encoding",
77                    "expected `mapbox`, `terrarium`, or `custom`",
78                ))
79            }
80        };
81        Ok(Self {
82            tiles,
83            minzoom: json_opt_u8(map, "minzoom")?,
84            maxzoom: json_opt_u8(map, "maxzoom")?,
85            bounds: json_opt_bounds4(map, "bounds")?,
86            attribution: json_opt_string(map, "attribution")?,
87            scheme,
88            tile_size: json_opt_u32(map, "tileSize")?,
89            encoding,
90        })
91    }
92}

94/// The addressing fields a TileJSON document can fill in, shared by tile and DEM sources.
95struct TileJsonFields<'a> {
96    tiles: &'a mut Option<Vec<String>>,
97    minzoom: &'a mut Option<u8>,
98    maxzoom: &'a mut Option<u8>,
99    bounds: &'a mut Option<(f64, f64, f64, f64)>,
100    attribution: &'a mut Option<String>,
101}
102
103impl TileJsonFields<'_> {
104    fn apply(self, tile_json: TileJson) {
105        if self.tiles.is_none() {
106            *self.tiles = Some(tile_json.tiles);
107        }
108        if self.minzoom.is_none() {
109            *self.minzoom = tile_json.minzoom;
110        }
111        if self.maxzoom.is_none() {
112            *self.maxzoom = tile_json.maxzoom;
113        }
114        if self.bounds.is_none() {
115            *self.bounds = tile_json.bounds;
116        }
117        if self.attribution.is_none() {
118            *self.attribution = tile_json.attribution;
119        }
120    }
121}

123/// Fills the tile URLs, zoom range, attribution, scheme and tile size a source leaves
124/// unspecified from its TileJSON document, as GL JS's `loadTileJson` does.
125pub fn apply_tile_json(source: &mut VectorSource, tile_json: TileJson) {
126    if source.scheme.is_none() {
127        source.scheme = tile_json.scheme;
128    }
129    if source.tile_size.is_none() {
130        source.tile_size = tile_json.tile_size;
131    }
132    TileJsonFields {
133        tiles: &mut source.tiles,
134        minzoom: &mut source.minzoom,
135        maxzoom: &mut source.maxzoom,
136        bounds: &mut source.bounds,
137        attribution: &mut source.attribution,
138    }
139    .apply(tile_json);
140}

142/// Fills the tile URLs, zoom range, tile size and encoding of a DEM source from its TileJSON
143/// document; a document such as Mapterhorn's declares `"encoding": "terrarium"`.
144pub fn apply_tile_json_to_dem(source: &mut RasterDemSource, tile_json: TileJson) {
145    if source.tile_size.is_none() {
146        source.tile_size = tile_json.tile_size;
147    }
148    if source.encoding.is_none() {
149        source.encoding = tile_json.encoding;
150    }
151    TileJsonFields {
152        tiles: &mut source.tiles,
153        minzoom: &mut source.minzoom,
154        maxzoom: &mut source.maxzoom,
155        bounds: &mut source.bounds,
156        attribution: &mut source.attribution,
157    }
158    .apply(tile_json);
159}

161/// Returns the TileJSON URL of a source that declares `url` but no `tiles`.
162fn pending_tile_json_url(source: &Source) -> Option<String> {
163    let (tiles, url) = match source {
164        Source::Vector(vector) | Source::Raster(vector) => (&vector.tiles, &vector.url),
165        Source::RasterDem(dem) => (&dem.tiles, &dem.url),
166        Source::GeoJson(_) | Source::Image(_) | Source::Video(_) | Source::Canvas(_) => {
167            return None;
168        }
169    };
170    tiles.is_none().then(|| url.clone()).flatten()
171}
172
173fn apply_tile_json_to_source(source: &mut Source, tile_json: TileJson) {
174    match source {
175        Source::Vector(vector) | Source::Raster(vector) => apply_tile_json(vector, tile_json),
176        Source::RasterDem(dem) => apply_tile_json_to_dem(dem, tile_json),
177        Source::GeoJson(_) | Source::Image(_) | Source::Video(_) | Source::Canvas(_) => {}
178    }
179}

181/// Fetches the TileJSON of every tile source that declares a `url` but no `tiles`.
182///
183/// A source whose document cannot be fetched or parsed is left unchanged so its layers fall back
184/// to the crate default source instead of failing the whole style.
185///
186/// A windowed map does this when its renderer initializes; a headless map's host awaits it
187/// before `HeadlessMap::new`, passing the kernel's
188/// source client. Nothing in it blocks, so it runs on a browser's event loop.
189pub async fn resolve_tile_json_sources<HC: HttpClient>(
190    style: &mut Style,
191    client: &SourceClient<HC>,
192) {
193    let pending: Vec<(String, String, bool)> = style
194        .sources
195        .iter()
196        .filter_map(|(name, source)| {
197            pending_tile_json_url(source).map(|url| (name.clone(), url, source.is_volatile()))
198        })
199        .collect();
200    if pending.is_empty() {
201        return;
202    }
203    let results = crate::util::async_join::join_all_concurrent(
204        pending
205            .iter()
206            .map(|(_, url, volatile)| client.fetch_source_url(url, *volatile)),
207    )
208    .await;
209    for ((name, url, _), fetched) in pending.into_iter().zip(results) {
210        let Some(source) = style.sources.get_mut(&name) else {
211            continue;
212        };
213        match fetched {
214            Ok(bytes) => match TileJson::from_json_slice(&bytes) {
215                Ok(tile_json) => {
216                    tracing::info!(source = %name, %url, "resolved TileJSON source");
217                    apply_tile_json_to_source(source, tile_json);
218                }
219                Err(error) => {
220                    tracing::warn!(source = %name, %url, %error, "TileJSON document is invalid");
221                }
222            },
223            Err(error) => {
224                tracing::warn!(
225                    source = %name,
226                    %url,
227                    error = %error.describe(),
228                    "TileJSON document is unreachable"
229                );
230            }
231        }
232    }
233}
