13/// Cache-first [`HttpClient`] decorator that serves stored offline resources even when
14/// [`NetworkStatus::Offline`] and optionally records online responses into the ambient offline cache.
15pub struct OfflineHttpClient<HC: HttpClient, S: OfflineStore> {
16    inner: HC,
17    store: Arc<S>,
18    ambient_caching: bool,
19}
20
21impl<HC: HttpClient, S: OfflineStore> Clone for OfflineHttpClient<HC, S> {
22    fn clone(&self) -> Self {
23        Self {
24            inner: self.inner.clone(),
25            store: Arc::clone(&self.store),
26            ambient_caching: self.ambient_caching,
27        }
28    }
29}
30
31impl<HC: HttpClient, S: OfflineStore> OfflineHttpClient<HC, S> {
32    /// Wraps `inner` with cache-first lookups against `store` and ambient caching enabled.
33    pub fn new(inner: HC, store: Arc<S>) -> Self {
34        Self {
35            inner,
36            store,
37            ambient_caching: true,
38        }
39    }
40
41    /// Configures whether online `200 OK` responses are opportunistically written to ambient storage.
42    pub fn with_ambient_caching(mut self, enabled: bool) -> Self {
43        self.ambient_caching = enabled;
44        self
45    }
46
47    /// Borrows the wrapped inner [`HttpClient`].
48    pub fn inner(&self) -> &HC {
49        &self.inner
50    }
51
52    /// Borrows the backing [`OfflineStore`].
53    pub fn store(&self) -> &Arc<S> {
54        &self.store
55    }
56}

58#[cfg_attr(not(feature = "thread-safe-futures"), async_trait(?Send))]
59#[cfg_attr(feature = "thread-safe-futures", async_trait)]
60impl<HC: HttpClient, S: OfflineStore> HttpClient for OfflineHttpClient<HC, S> {
61    async fn fetch(&self, url: &str) -> Result<Vec<u8>, SourceFetchError> {
62        let req = HttpRequest::new(url, ResourceKind::infer_from_url(url));
63        self.fetch_request_with_status(&req, NetworkStatus::Online)
64            .await
65    }
66
67    async fn fetch_range(&self, url: &str, range: ByteRange) -> Result<Vec<u8>, SourceFetchError> {
68        let req = HttpRequest::new(url, ResourceKind::infer_from_url(url)).with_range(range);
69        self.fetch_request_with_status(&req, NetworkStatus::Online)
70            .await
71    }
72
73    async fn fetch_request(&self, req: &HttpRequest) -> Result<Vec<u8>, SourceFetchError> {
74        self.fetch_request_with_status(req, NetworkStatus::Online)
75            .await
76    }
77
78    async fn fetch_request_with_status(
79        &self,
80        req: &HttpRequest,
81        status: NetworkStatus,
82    ) -> Result<Vec<u8>, SourceFetchError> {
83        if !req.cache_policy.is_volatile() {
84            if let Ok(Some(cached)) = self.store.get_resource(&req.url).await {
85                return match req.range {
86                    Some(range) => range.slice(&req.url, &cached).map(<[u8]>::to_vec),
87                    None => Ok(cached),
88                };
89            }
90        }
91        let body = self.inner.fetch_request_with_status(req, status).await?;
92        if self.ambient_caching
93            && !req.cache_policy.is_volatile()
94            && req.range.is_none()
95            && is_remote_http_url(&req.url)
96        {
97            let _ = self
98                .store
99                .put_ambient_resource(&req.url, req.resource_kind, &body)
100                .await;
101        }
102        Ok(body)
103    }
104}
