65    pub(super) fn next_stamp(&self) -> i64 {
66        self.clock.fetch_add(1, Ordering::Relaxed) + 1
67    }

74pub(super) fn kind_to_i32(kind: ResourceKind) -> i32 {
75    match kind {
76        ResourceKind::Style => 1,
77        ResourceKind::Source => 2,
78        ResourceKind::Tile => 3,
79        ResourceKind::Glyphs => 4,
80        ResourceKind::SpriteImage => 5,
81        ResourceKind::SpriteJson => 6,
82        ResourceKind::Image => 7,
83        ResourceKind::Unknown => 0,
84    }

87pub(super) fn encode_resource_blob(raw: &[u8]) -> (Vec<u8>, bool) {
88    let already_compressed_image = raw.starts_with(b"\x89PNG\r\n\x1a\n")
89        || raw.starts_with(&[0xff, 0xd8, 0xff])
90        || (raw.len() >= 12 && &raw[0..4] == b"RIFF" && &raw[8..12] == b"WEBP");
91    if raw.len() < 128 || already_compressed_image {
92        return (raw.to_vec(), false);
93    }
94    let mut encoder = ZlibEncoder::new(Vec::new(), Compression::fast());
95    if encoder.write_all(raw).is_ok() {
96        if let Ok(compressed) = encoder.finish() {
97            if compressed.len() < raw.len() {
98                return (compressed, true);
99            }
100        }
101    }
102    (raw.to_vec(), false)
103}

105pub(super) fn decode_resource_blob(
106    stored: Vec<u8>,
107    compressed: bool,
108) -> Result<Vec<u8>, OfflineStoreError> {
109    if !compressed {
110        return Ok(stored);
111    }
112    let mut decoder = ZlibDecoder::new(stored.as_slice());
113    let mut out = Vec::new();
114    decoder.read_to_end(&mut out)?;
115    Ok(out)
116}

118pub(super) fn read_tile_limit(conn: &Connection) -> Result<Option<u64>, OfflineStoreError> {
119    let val: Option<i64> = conn
120        .query_row(
121            "SELECT value FROM offline_config WHERE key = ?1",
122            params![CFG_TILE_COUNT_LIMIT],
123            |row| row.get(0),
124        )
125        .optional()
126        .map_err(sql_err)?;
127    Ok(val.and_then(|v| (v >= 0).then_some(v as u64)))
128}

130pub(super) fn read_max_ambient_bytes(conn: &Connection) -> Result<u64, OfflineStoreError> {
131    let val: Option<i64> = conn
132        .query_row(
133            "SELECT value FROM offline_config WHERE key = ?1",
134            params![CFG_MAX_AMBIENT_BYTES],
135            |row| row.get(0),
136        )
137        .optional()
138        .map_err(sql_err)?;
139    Ok(val.unwrap_or(DEFAULT_AMBIENT_BYTES).max(0) as u64)
140}

142pub(super) fn pinned_tile_count(conn: &Connection) -> Result<u64, OfflineStoreError> {
143    let count: i64 = conn
144        .query_row(
145            "SELECT COUNT(*) FROM resources r
146             WHERE r.kind = ?1
147               AND EXISTS (SELECT 1 FROM region_resources rr WHERE rr.url = r.url)",
148            params![kind_to_i32(ResourceKind::Tile)],
149            |row| row.get(0),
150        )
151        .map_err(sql_err)?;
152    Ok(count.max(0) as u64)
153}

155pub(super) fn is_url_pinned(conn: &Connection, url: &str) -> Result<bool, OfflineStoreError> {
156    let exists: Option<i32> = conn
157        .query_row(
158            "SELECT 1 FROM region_resources WHERE url = ?1 LIMIT 1",
159            params![url],
160            |row| row.get(0),
161        )
162        .optional()
163        .map_err(sql_err)?;
164    Ok(exists.is_some())
165}

167pub(super) fn gc_orphaned_non_ambient(conn: &Connection) -> Result<(), OfflineStoreError> {
168    conn.execute(

178pub(super) fn enforce_ambient_budget(conn: &Connection) -> Result<(), OfflineStoreError> {
179    let budget = read_max_ambient_bytes(conn)?;
180    let total: i64 = conn
181        .query_row(
182            "SELECT COALESCE(SUM(stored_size), 0) FROM resources r
183             WHERE r.ambient = 1
184               AND NOT EXISTS (SELECT 1 FROM region_resources rr WHERE rr.url = r.url)",
185            [],
186            |row| row.get(0),
187        )
188        .map_err(sql_err)?;
189    let mut current = total.max(0) as u64;
190    if current <= budget {
191        return Ok(());
192    }
193    let mut stmt = conn
194        .prepare(
195            "SELECT r.url, r.stored_size FROM resources r
196             WHERE r.ambient = 1
197               AND NOT EXISTS (SELECT 1 FROM region_resources rr WHERE rr.url = r.url)
198             ORDER BY r.accessed_at ASC",
199        )
200        .map_err(sql_err)?;
201    let candidates: Vec<(String, i64)> = stmt
202        .query_map([], |row| Ok((row.get(0)?, row.get(1)?)))
203        .map_err(sql_err)?
204        .collect::<Result<Vec<_>, _>>()
205        .map_err(sql_err)?;
206    drop(stmt);
207
208    for (url, sz) in candidates {
209        if current <= budget {
210            break;
211        }
212        conn.execute("DELETE FROM resources WHERE url = ?1", params![url])
213            .map_err(sql_err)?;
214        current = current.saturating_sub(sz.max(0) as u64);
215    }
216    Ok(())
217}

219pub(super) fn query_region_status(
220    conn: &Connection,
221    id: OfflineRegionId,
222) -> Result<OfflineRegionStatus, OfflineStoreError> {
223    let (complete, required): (i64, i64) = conn
224        .query_row(
225            "SELECT complete, required_count FROM regions WHERE id = ?1",
226            params![id.0],
227            |row| Ok((row.get(0)?, row.get(1)?)),
228        )
229        .optional()
230        .map_err(sql_err)?
231        .ok_or(OfflineStoreError::RegionNotFound(id))?;
232
233    let (res_count, res_bytes, tile_count): (i64, i64, i64) = conn
234        .query_row(
235            "SELECT
236                COUNT(r.url),
237                COALESCE(SUM(r.raw_size), 0),
238                COALESCE(SUM(CASE WHEN r.kind = ?2 THEN 1 ELSE 0 END), 0)
239             FROM region_resources rr
240             JOIN resources r ON r.url = rr.url
241             WHERE rr.region_id = ?1",
242            params![id.0, kind_to_i32(ResourceKind::Tile)],
243            |row| Ok((row.get(0)?, row.get(1)?, row.get(2)?)),
244        )
245        .map_err(sql_err)?;
246
247    Ok(OfflineRegionStatus {
248        complete: complete != 0,
249        completed_resource_count: res_count.max(0) as u64,
250        completed_resource_bytes: res_bytes.max(0) as u64,
251        completed_tile_count: tile_count.max(0) as u64,
252        required_resource_count: required.max(0) as u64,
253    })
254}

256pub(super) fn merge_database_sync(
257    inner: &SqliteInner,
258    sideloaded_path: &Path,
259) -> Result<Vec<OfflineRegionId>, OfflineStoreError> {
260    let src = Connection::open_with_flags(
261        sideloaded_path,
262        OpenFlags::SQLITE_OPEN_READ_ONLY | OpenFlags::SQLITE_OPEN_NO_MUTEX,
263    )
264    .map_err(sql_err)?;
265
266    type SrcRegionRow = (i64, String, Vec<u8>, i64, i64);
267    let src_regions: Vec<SrcRegionRow> = src
268        .prepare("SELECT id, definition_json, metadata, complete, required_count FROM regions ORDER BY id ASC")
269        .map_err(sql_err)?
270        .query_map([], |r| Ok((r.get(0)?, r.get(1)?, r.get(2)?, r.get(3)?, r.get(4)?)))
271        .map_err(sql_err)?
272        .collect::<Result<Vec<_>, _>>()
273        .map_err(sql_err)?;
274
275    type SrcResourceRow = (String, i32, Vec<u8>, i64, i64, i64, i64);
276    let src_resources: Vec<SrcResourceRow> = src
277        .prepare(
278            "SELECT url, kind, data, compressed, raw_size, stored_size, ambient FROM resources",
279        )
280        .map_err(sql_err)?
281        .query_map([], |r| {
282            Ok((
283                r.get(0)?,
284                r.get(1)?,
285                r.get(2)?,
286                r.get(3)?,
287                r.get(4)?,
288                r.get(5)?,
289                r.get(6)?,
290            ))
291        })
292        .map_err(sql_err)?
293        .collect::<Result<Vec<_>, _>>()
294        .map_err(sql_err)?;
295
296    let src_links: Vec<(i64, String)> = src
297        .prepare("SELECT region_id, url FROM region_resources")
298        .map_err(sql_err)?
299        .query_map([], |r| Ok((r.get(0)?, r.get(1)?)))
300        .map_err(sql_err)?
301        .collect::<Result<Vec<_>, _>>()
302        .map_err(sql_err)?;
303
304    let mut dst = inner.lock();
305    let tx = dst.transaction().map_err(sql_err)?;
306
307    if let Some(limit) = read_tile_limit(&tx)? {
308        let current_tiles = pinned_tile_count(&tx)?;
309        let tile_kind = kind_to_i32(ResourceKind::Tile);
310        let mut newly_pinned_tiles = 0_u64;
311        for (url, kind, ..) in &src_resources {
312            let linked_in_src = src_links.iter().any(|(_, u)| u == url);
313            if *kind == tile_kind && linked_in_src && !is_url_pinned(&tx, url)? {
314                newly_pinned_tiles += 1;
315            }
316        }
317        if current_tiles.saturating_add(newly_pinned_tiles) > limit {
318            return Err(OfflineStoreError::TileLimitExceeded { limit });
319        }
320    }
321
322    for (url, kind, data, compressed, raw_size, stored_size, ambient) in src_resources {
323        let stamp = inner.next_stamp();
324        tx.execute(
325            "INSERT INTO resources (url, kind, data, compressed, raw_size, stored_size, ambient, accessed_at)
326             VALUES (?1, ?2, ?3, ?4, ?5, ?6, ?7, ?8)
327             ON CONFLICT(url) DO UPDATE SET
328                kind = excluded.kind,
329                data = excluded.data,
330                compressed = excluded.compressed,
331                raw_size = excluded.raw_size,
332                stored_size = excluded.stored_size,
333                accessed_at = excluded.accessed_at",
334            params![url, kind, data, compressed, raw_size, stored_size, ambient, stamp],
335        )
336        .map_err(sql_err)?;
337    }
338
339    let mut id_map: HashMap<i64, OfflineRegionId> = HashMap::new();
340    let mut created_ids = Vec::with_capacity(src_regions.len());
341    for (old_id, def_json, metadata, complete, req_count) in src_regions {
342        tx.execute(
343            "INSERT INTO regions (definition_json, metadata, complete, required_count)
344             VALUES (?1, ?2, ?3, ?4)",
345            params![def_json, metadata, complete, req_count],
346        )
347        .map_err(sql_err)?;
348        let new_id = OfflineRegionId(tx.last_insert_rowid());
349        id_map.insert(old_id, new_id);
350        created_ids.push(new_id);
351    }
352
353    for (old_reg_id, url) in src_links {
354        if let Some(new_id) = id_map.get(&old_reg_id) {
355            tx.execute(
356                "INSERT OR IGNORE INTO region_resources (region_id, url) VALUES (?1, ?2)",
357                params![new_id.0, url],
358            )
359            .map_err(sql_err)?;
360        }
361    }
362
363    enforce_ambient_budget(&tx)?;
364    tx.commit().map_err(sql_err)?;
365    Ok(created_ids)
366}
