12const MAX_MERCATOR_LAT: f64 = 85.051_128_779_806_6;
13const MAX_ENUMERABLE_ZOOM: u8 = 22;
14
15/// Geographic bounding box, zoom range, display scale, and style reference defining an offline pack.
16#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
17pub struct OfflineRegionDefinition {
18    /// Remote or local style JSON URL, when the region is backed by an external style document.
19    #[serde(default, skip_serializing_if = "Option::is_none")]
20    pub style_url: Option<String>,
21    /// Inline MapLibre style JSON string, when supplied directly without a style URL.
22    #[serde(default, skip_serializing_if = "Option::is_none")]
23    pub style_json: Option<String>,
24    /// Geographic bounding box `[west, south, east, north]` in degrees (`west > east` crosses the antimeridian).
25    pub bounds: [f64; 4],
26    /// Minimum discrete tile zoom level to download (`0..=22`).
27    pub min_zoom: u8,
28    /// Maximum discrete tile zoom level to download (`min_zoom..=22`).
29    pub max_zoom: u8,
30    /// Display device pixel ratio (e.g. `1.0` or `2.0`).
31    pub pixel_ratio: f32,
32    /// Opaque application-defined metadata bytes persisted alongside the region.
33    #[serde(default)]
34    pub metadata: Vec<u8>,
35}

86    /// Deterministically enumerates all `WorldTileCoords` intersecting `self.bounds` across
87    /// `self.min_zoom..=self.max_zoom` in ascending `(z, x, y)` order.
88    pub fn tile_pyramid(&self) -> Vec<WorldTileCoords> {
89        self.tile_pyramid_for_source(None, None, None)
90    }
91
92    /// Deterministically enumerates all `WorldTileCoords` intersecting both `self.bounds` and an
93    /// optional `source_bounds`, clamped to `source_min_zoom..=source_max_zoom`.
94    pub fn tile_pyramid_for_source(
95        &self,
96        source_min_zoom: Option<u8>,
97        source_max_zoom: Option<u8>,
98        source_bounds: Option<(f64, f64, f64, f64)>,
99    ) -> Vec<WorldTileCoords> {
100        let z_start = self
101            .min_zoom
102            .max(source_min_zoom.unwrap_or(0))
103            .min(MAX_ENUMERABLE_ZOOM);
104        let z_end = self
105            .max_zoom
106            .min(source_max_zoom.unwrap_or(MAX_ENUMERABLE_ZOOM))
107            .min(MAX_ENUMERABLE_ZOOM);
108        if z_start > z_end {
109            return Vec::new();
110        }
111
112        let [west, south, east, north] = self.bounds;
113        let clamped_south = south.min(north).clamp(-MAX_MERCATOR_LAT, MAX_MERCATOR_LAT);
114        let clamped_north = north.max(south).clamp(-MAX_MERCATOR_LAT, MAX_MERCATOR_LAT);
115
116        let src_lat_range = source_bounds.map(|(_, s, _, n)| {
117            (
118                s.min(n).clamp(-MAX_MERCATOR_LAT, MAX_MERCATOR_LAT),
119                n.max(s).clamp(-MAX_MERCATOR_LAT, MAX_MERCATOR_LAT),
120            )
121        });
122        let (eff_south, eff_north) = match src_lat_range {
123            Some((s, n)) => {
124                let low = clamped_south.max(s);
125                let high = clamped_north.min(n);
126                if low > high {
127                    return Vec::new();
128                }
129                (low, high)
130            }
131            None => (clamped_south, clamped_north),
132        };
133
134        let lon_spans = intersect_lon_spans(lon_intervals(west, east), source_bounds);
135        if lon_spans.is_empty() {
136            return Vec::new();
137        }
138
139        let mut coords_set: BTreeSet<(u8, i32, i32)> = BTreeSet::new();
140        for z in z_start..=z_end {
141            let grid = 1_u32 << u32::from(z);
142            let (y_min, y_max) = lat_tile_range(eff_south, eff_north, grid);
143            for (lon_w, lon_e) in &lon_spans {
144                let (x_min, x_max) = lon_tile_range(*lon_w, *lon_e, grid);
145                for x in x_min..=x_max {
146                    for y in y_min..=y_max {
147                        coords_set.insert((z, x, y));
148                    }
149                }
150            }
151        }
152
153        coords_set
154            .into_iter()
155            .map(|(z, x, y)| WorldTileCoords::from((x, y, ZoomLevel::new(z))))
156            .collect()
157    }
158}

160fn lon_intervals(west: f64, east: f64) -> Vec<(f64, f64)> {
161    if (east - west) >= 360.0 || (west <= -180.0 && east >= 180.0) {
162        return vec![(-180.0, 180.0)];
163    }
164    let w = west.clamp(-180.0, 180.0);
165    let e = east.clamp(-180.0, 180.0);
166    if w <= e {
167        vec![(w, e)]
168    } else {
169        vec![(w, 180.0), (-180.0, e)]
170    }
171}

173fn intersect_lon_spans(
174    region_spans: Vec<(f64, f64)>,
175    source_bounds: Option<(f64, f64, f64, f64)>,
176) -> Vec<(f64, f64)> {
177    let Some((sw, _, se, _)) = source_bounds else {
178        return region_spans;
179    };
180    let src_spans = lon_intervals(sw, se);
181    let mut out = Vec::new();
182    for (rw, re) in &region_spans {
183        for (xw, xe) in &src_spans {
184            let low = rw.max(*xw);
185            let high = re.min(*xe);
186            if low <= high {
187                out.push((low, high));
188            }
189        }
190    }
191    out
192}

194fn lon_tile_range(west: f64, east: f64, grid: u32) -> (i32, i32) {
195    let n = f64::from(grid);
196    let max_idx = (grid - 1) as i32;
197    let frac_w = ((west + 180.0) / 360.0) * n;
198    let x_min = (frac_w.floor() as i32).clamp(0, max_idx);
199    if east <= west {
200        return (x_min, x_min);
201    }
202    let frac_e = ((east + 180.0) / 360.0) * n;
203    let adjusted_e = if frac_e > frac_w {
204        frac_e - 1e-9
205    } else {
206        frac_e
207    };
208    let x_max = (adjusted_e.floor() as i32).clamp(x_min, max_idx);
209    (x_min, x_max)
210}

212fn lat_to_mercator_frac(lat_deg: f64) -> f64 {
213    let lat_rad = lat_deg
214        .clamp(-MAX_MERCATOR_LAT, MAX_MERCATOR_LAT)
215        .to_radians();
216    (1.0 - (lat_rad.tan() + 1.0 / lat_rad.cos()).ln() / PI) * 0.5
217}

219fn lat_tile_range(south: f64, north: f64, grid: u32) -> (i32, i32) {
220    let n = f64::from(grid);
221    let max_idx = (grid - 1) as i32;
222    let frac_n = lat_to_mercator_frac(north) * n;
223    let y_min = (frac_n.floor() as i32).clamp(0, max_idx);
224    if north <= south {
225        return (y_min, y_min);
226    }
227    let frac_s = lat_to_mercator_frac(south) * n;
228    let adjusted_s = if frac_s > frac_n {
229        frac_s - 1e-9
230    } else {
231        frac_s
232    };
233    let y_max = (adjusted_s.floor() as i32).clamp(y_min, max_idx);
234    (y_min, y_max)
