42/// A worker payload has a different Rust type from the one its consumer requested.
43#[derive(Error, Debug)]
44#[error("message {tag:?} carries payload type {actual:?}, expected {expected}")]
45pub struct MessageError {
46    /// Routing tag attached to the rejected result.
47    pub tag: &'static dyn MessageTag,
48    /// Rust type name requested by the consumer.
49    pub expected: &'static str,
50    /// Runtime type identifier of the rejected payload.
51    pub actual: TypeId,
52}
53
54/// A tagged worker result with an owned, type-erased payload.
55///
56/// Tags route messages to consumers; a tag does not enforce the payload's Rust type.
57#[derive(Debug)]
58pub struct Message {
59    tag: &'static dyn MessageTag,
60    transferable: Box<dyn Any + Send>,
61    attempt: Option<u64>,
62}
63
64impl Message {
65    /// Associates a routing tag with an owned payload; the caller must keep their types consistent.
66    pub fn new(tag: &'static dyn MessageTag, transferable: Box<dyn Any + Send>) -> Self {
67        Self {
68            tag,
69            transferable,
70            attempt: None,
71        }
72    }
73
74    /// Associates this result with the request that produced it, including partial payloads.
75    pub fn with_attempt(mut self, attempt: u64) -> Self {
76        self.attempt = Some(attempt);
77        self
78    }
79
80    /// Request identity retained independently of the concrete worker payload.
81    pub fn attempt(&self) -> Option<u64> {
82        self.attempt
83    }
84
85    /// Takes the payload as its concrete Rust type.
86    /// A mismatch drops the payload and returns its tag and type context.
87    pub fn into_transferable<T: 'static>(self) -> Result<Box<T>, MessageError> {
88        let actual = self.transferable.as_ref().type_id();
89        self.transferable.downcast::<T>().map_err(|_| MessageError {
90            tag: self.tag,
91            expected: type_name::<T>(),
92            actual,
93        })
94    }
95
96    /// Tests routing-tag equality, including the tag's concrete type.
97    pub fn has_tag(&self, tag: &'static dyn MessageTag) -> bool {
98        self.tag == tag
99    }
100
101    /// Borrows the routing tag for this result.
102    pub fn tag(&self) -> &'static dyn MessageTag {
103        self.tag
104    }
105}

119/// Inputs for an [`AsyncProcedure`]
120#[derive(Clone, Serialize, Deserialize)]
121pub enum Input {
122    /// Requests source data and processing for a tile using the supplied style snapshot.
123    TileRequest {
124        /// Tile grid location requested from the worker.
125        coords: WorldTileCoords,
126        /// Style snapshot used to choose and process the tile's layers.
127        style: Style,
128    },
129    /// Requests a tile with a map-owned attempt token for final completion matching.
130    TrackedTileRequest {
131        /// Tile grid location requested from the worker.
132        coords: WorldTileCoords,
133        /// Style snapshot used to choose and process the tile's layers.
134        style: Style,
135        /// Identifier distinguishing requests when coordinates are evicted and reused.
136        attempt: u64,
137        /// The zoom a vector tile is drawn at when that is past its source's last zoom; zero
138        /// when it is drawn at its own zoom.
139        #[serde(default)]
140        overscaled_zoom: u8,
141        /// Device pixels per layout pixel of the display, which generated images are made for.
142        #[serde(default = "one")]
143        pixel_ratio: f32,
144    },
145}
146
147fn one() -> f32 {
148    1.0
149}
150
151impl Input {
152    /// Consumes a tile request and its optional completion token.
153    pub fn into_tile_request(self) -> (WorldTileCoords, Style, Option<u64>) {
154        match self {
155            Self::TileRequest { coords, style } => (coords, style, None),
156            Self::TrackedTileRequest {
157                coords,
158                style,
159                attempt,
160                ..
161            } => (coords, style, Some(attempt)),
162        }
163    }
164
165    /// The zoom the requested tile is magnified to, or zero when it is not magnified.
166    pub fn overscaled_zoom(&self) -> u8 {
167        match self {
168            Self::TileRequest { .. } => 0,
169            Self::TrackedTileRequest {
170                overscaled_zoom, ..
171            } => *overscaled_zoom,
172        }
173    }
174
175    /// Device pixels per layout pixel of the display the tile is drawn on.
176    pub fn pixel_ratio(&self) -> f32 {
177        match self {
178            Self::TileRequest { .. } => 1.0,
179            Self::TrackedTileRequest { pixel_ratio, .. } => *pixel_ratio,
180        }
181    }
182}

184/// A worker result could not be delivered to its caller.
185#[derive(Error, Debug)]
186pub enum SendError {
187    /// The transport rejected the message or its receiving endpoint is gone.
188    #[error("worker transport failed while {operation}")]
189    Transmission {
190        /// Transport operation that rejected the result.
191        operation: &'static str,
192        /// Underlying channel or platform error.
193        #[source]
194        source: Box<dyn std::error::Error>,
195    },
196    /// The result has a routing tag unsupported by this transport.
197    #[error("worker transport does not support message tag {tag:?}")]
198    UnsupportedTag {
199        /// Rejected routing tag.
200        tag: &'static dyn MessageTag,
201    },
202    /// The result's payload is incompatible with the transport.
203    #[error("worker result has an invalid payload")]
204    Payload(#[from] MessageError),
205}

232/// A procedure call or incoming worker message could not be prepared or admitted.
233#[derive(Error, Debug)]
234pub enum CallError {
235    /// The executor rejected the call before execution.
236    #[error("scheduling work failed")]
237    Schedule(#[from] ScheduleError),
238    /// The call input could not be encoded.
239    #[error("serializing data failed")]
240    Serialize(#[source] Box<dyn std::error::Error>),
241    /// A worker message could not be decoded.
242    #[error("deserializing failed")]
243    Deserialize(#[source] Box<dyn std::error::Error>),
244    /// A worker could not decode its call input.
245    #[error("deserializing input failed")]
246    DeserializeInput(#[source] Box<dyn std::error::Error>),
247}

249/// A statically addressable worker procedure taking owned input, a reply context, and a kernel.
250/// Function pointers allow worker transports to identify procedures without serializing closures.
251pub type AsyncProcedure<K, C> = fn(input: Input, context: C, kernel: K) -> AsyncProcedureFuture;
252
253/// Schedules tile work and receives its tagged results without blocking the render loop.
254/// Native implementations can transfer owned Rust values; browser workers can serialize them.
255pub trait AsyncProcedureCall<K: OffscreenKernel>: 'static {
256    /// Cloneable reply endpoint handed to each procedure.
257    type Context: Context + Send + Clone;
258
259    /// Owned matching results drained from the transport.
260    type ReceiveIterator<F: FnMut(&Message) -> bool>: Iterator<Item = Message>;
261
262    /// Drains currently available matching results, retaining nonmatching messages for other consumers.
263    fn receive<F: FnMut(&Message) -> bool>(&self, filter: F) -> Self::ReceiveIterator<F>;
264
265    /// Admits a call without waiting for completion; input or scheduling failures are returned.
266    /// Procedure failures occur asynchronously and are handled by the selected implementation.
267    fn call(
268        &self,
269        input: Input,
270        procedure: AsyncProcedure<K, Self::Context>,
271    ) -> Result<(), CallError>;
272
273    /// Stops the tracked call with `attempt` if it is still running. A transport that cannot
274    /// reach its worker lets the call finish; the map ignores a result it gave up on.
275    fn cancel(&self, _attempt: u64) {}
276
277    /// Whether results arrived that no [`Self::receive`] has offered to its consumers yet, so
278    /// the next frame has something to apply. A transport that cannot tell answers `false`.
279    ///
280    /// Any receive counts as offering, which holds because every consumer receives in every
281    /// frame; ask between frames.
282    fn has_arrivals(&self) -> bool {
283        false
284    }
285
286    /// The image providers the workers ask, when they share this transport's memory. A
287    /// transport whose workers build their own configuration has none to offer here.
288    fn image_providers(&self) -> Option<&crate::sdf::assets::ImageProviders> {
289        None
290    }
291}

293/// Reply sender whose clones share the scheduler-backed APC's receiving channel.
294#[derive(Clone)]
295pub struct SchedulerContext {
296    sender: Sender<Message>,
297}
298
299impl Context for SchedulerContext {
300    fn send_back<T: IntoMessage>(&self, message: T) -> Result<(), SendError> {
301        self.sender
302            .send(message.into())
303            .map_err(|source| SendError::Transmission {
304                operation: "sending a result to the caller",
305                source: Box::new(source),
306            })
307    }
308}

310/// An APC using a scheduler for execution and an in-process channel for replies.
311///
312/// Receiving drains available matches in delivery order. Procedure failures are logged;
313/// only preparation and admission failures are returned by [`AsyncProcedureCall::call`].
314pub struct SchedulerAsyncProcedureCall<K: OffscreenKernel, S: Scheduler> {
315    channel: (Sender<Message>, Receiver<Message>),
316    buffer: RefCell<Vec<Message>>,
317    /// Messages [`AsyncProcedureCall::has_arrivals`] moved into `buffer` that no receive has
318    /// offered to a consumer since.
319    unoffered: Cell<bool>,
320    scheduler: S,
321    phantom_k: PhantomData<K>,
322    offscreen_kernel_config: OffscreenKernelConfig,
323    /// Tracked calls still running, by attempt, so a cancelled one can be stopped.
324    running: Arc<Mutex<HashMap<u64, AbortHandle>>>,
325}
326
327impl<K: OffscreenKernel, S: Scheduler> SchedulerAsyncProcedureCall<K, S> {
328    /// Creates an empty reply channel and keeps the worker kernel configuration for future calls.
329    pub fn new(scheduler: S, offscreen_kernel_config: OffscreenKernelConfig) -> Self {
330        Self {
331            channel: mpsc::channel(),
332            buffer: RefCell::new(Vec::new()),
333            unoffered: Cell::new(false),
334            phantom_k: PhantomData,
335            scheduler,
336            offscreen_kernel_config,
337            running: Arc::default(),
338        }
339    }
340
341    fn running(&self) -> std::sync::MutexGuard<'_, HashMap<u64, AbortHandle>> {
342        self.running.lock().unwrap_or_else(PoisonError::into_inner)
343    }
344}
345
346impl<K: OffscreenKernel, S: Scheduler> AsyncProcedureCall<K> for SchedulerAsyncProcedureCall<K, S> {
347    type Context = SchedulerContext;
348    type ReceiveIterator<F: FnMut(&Message) -> bool> = IntoIter<Message>;
349
350    fn receive<F: FnMut(&Message) -> bool>(&self, mut filter: F) -> Self::ReceiveIterator<F> {
351        self.unoffered.set(false);
352        let mut buffer = self.buffer.borrow_mut();
353        // Partial and final tile completions must retain the worker's delivery order.
354        let mut ret: Vec<_> = buffer.extract_if(.., |message| filter(message)).collect();
355
356        while let Ok(message) = self.channel.1.try_recv() {
357            tracing::debug!("Data reached main thread: {message:?}");
358
359            if filter(&message) {
360                ret.push(message);
361            } else {
362                buffer.push(message)
363            }
364        }
365
366        ret.into_iter()
367    }
368
369    fn call(
370        &self,
371        input: Input,
372        procedure: AsyncProcedure<K, Self::Context>,
373    ) -> Result<(), CallError> {
374        let sender = self.channel.0.clone();
375        let offscreen_kernel_config = self.offscreen_kernel_config.clone();
376        let attempt = match &input {
377            Input::TrackedTileRequest { attempt, .. } => Some(*attempt),
378            Input::TileRequest { .. } => None,
379        };
380        let (abort, registration) = AbortHandle::new_pair();
381        if let Some(attempt) = attempt {
382            self.running().insert(attempt, abort);
383        }
384        let running = self.running.clone();
385
386        self.scheduler
387            .schedule(move || async move {
388                tracing::debug!(thread = ?std::thread::current().name(), "processing worker call");
389
390                let kernel = K::create(offscreen_kernel_config);
391                // A result that cannot be delivered, because the map already shut down, must
392                // not take the worker thread down with it.
393                let call = procedure(input, SchedulerContext { sender }, kernel);
394                match Abortable::new(call, registration).await {
395                    Ok(Err(error)) => tracing::warn!(?error, "procedure failed"),
396                    Ok(Ok(())) => {}
397                    Err(Aborted) => tracing::debug!(?attempt, "cancelled worker call stopped"),
398                }
399                if let Some(attempt) = attempt {
400                    running
401                        .lock()
402                        .unwrap_or_else(PoisonError::into_inner)
403                        .remove(&attempt);
404                }
405            })
406            .map_err(CallError::Schedule)
407    }
408
409    fn has_arrivals(&self) -> bool {
410        // The receiver cannot be peeked; buffered messages keep their order ahead of the channel.
411        let mut buffer = self.buffer.borrow_mut();
412        while let Ok(message) = self.channel.1.try_recv() {
413            buffer.push(message);
414            self.unoffered.set(true);
415        }
416        self.unoffered.get()
417    }
418
419    fn cancel(&self, attempt: u64) {
420        if let Some(abort) = self.running().remove(&attempt) {
421            abort.abort();
422        }
423    }
424
425    fn image_providers(&self) -> Option<&crate::sdf::assets::ImageProviders> {
426        Some(&self.offscreen_kernel_config.image_providers)
427    }
428}

430/// Applies valid results in the entire drained batch before reporting its first type error.
431pub(crate) fn apply_worker_messages(
432    messages: impl IntoIterator<Item = Message>,
433    mut apply: impl FnMut(Message) -> Result<(), MessageError>,
434) -> Result<(), MessageError> {
435    let mut first_error = None;
436    for message in messages {
437        if let Err(error) = apply(message) {
438            tracing::error!(%error, "worker result rejected");
439            first_error.get_or_insert(error);
440        }
441    }
442    first_error.map_or(Ok(()), Err)
443}
