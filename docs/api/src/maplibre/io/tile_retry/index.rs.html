10/// Tracks only tiles with an in-flight attempt or an active retry deadline per [`RequestKind`],
11/// avoiding full-store scans across settled cached tiles on every frame.
12#[derive(Default)]
13pub(super) struct RetryIndex {
14    /// Tiles whose `RetryState::pending` is `true` for each [`RequestKind::index`].
15    pub(super) pending: [FxHashSet<WorldTileCoords>; 3],
16    /// Tiles whose `RetryState::deadline` is `Some(_)` for each [`RequestKind::index`].
17    pub(super) scheduled: [FxHashSet<WorldTileCoords>; 3],
18    /// Last wanted set recorded by [`want`] for each [`RequestKind::index`].
19    pub(super) last_wanted: [Option<FxHashSet<WorldTileCoords>>; 3],
20}
21
22/// The tiles each request family considered this frame; a family that did not run leaves
23/// its requests alone.
24#[derive(Default)]
25struct WantedRequests([Option<FxHashSet<WorldTileCoords>>; 3]);
26
27/// Attempts given up on, for the next request system to stop in its worker.
28#[derive(Default)]
29struct CancelledAttempts(Vec<u64>);
30
31impl RetryIndex {
32    /// Whether `coords` is currently outside the most recent `want()` set for `kind`.
33    pub(super) fn is_out_of_view(&self, kind: RequestKind, coords: WorldTileCoords) -> bool {
34        self.last_wanted[kind.index()]
35            .as_ref()
36            .is_some_and(|wanted| !wanted.contains(&coords))
37    }
38
39    pub(super) fn mark_pending(&mut self, kind: RequestKind, coords: WorldTileCoords) {
40        let idx = kind.index();
41        self.scheduled[idx].remove(&coords);
42        self.pending[idx].insert(coords);
43    }
44
45    pub(super) fn finish_pending(
46        &mut self,
47        kind: RequestKind,
48        coords: WorldTileCoords,
49        has_deadline: bool,
50    ) {
51        let idx = kind.index();
52        self.pending[idx].remove(&coords);
53        if has_deadline {
54            self.scheduled[idx].insert(coords);
55        } else {
56            self.scheduled[idx].remove(&coords);
57        }
58    }
59
60    pub(super) fn mark_scheduled(&mut self, kind: RequestKind, coords: WorldTileCoords) {
61        self.scheduled[kind.index()].insert(coords);
62    }
63
64    pub(super) fn clear_kind(&mut self, kind: RequestKind, coords: WorldTileCoords) {
65        let idx = kind.index();
66        self.pending[idx].remove(&coords);
67        self.scheduled[idx].remove(&coords);
68    }
69}

71pub(super) fn needs_frame_indexed(world: &World) -> bool {
72    let Some(index) = world.resources.get::<RetryIndex>() else {
73        return false;
74    };
75    for (kind_idx, pending) in index.pending.iter().enumerate() {
76        for &coords in pending {
77            if world
78                .tiles
79                .query::<&TileRequestRetries>(coords)
80                .is_some_and(|retries| retries.0[kind_idx].pending)
81            {
82                return true;
83            }
84        }
85    }
86    if index.scheduled.iter().all(FxHashSet::is_empty) {
87        return false;
88    }
89    let now = observed_now(world);
90    for (kind_idx, scheduled) in index.scheduled.iter().enumerate() {
91        for &coords in scheduled {
92            if world
93                .tiles
94                .query::<&TileRequestRetries>(coords)
95                .is_some_and(|retries| retries.0[kind_idx].retry_due(now))
96            {
97                return true;
98            }
99        }
100    }
101    false
102}

104pub(super) fn next_retry_delay_indexed(world: &World) -> Option<Duration> {
105    let index = world.resources.get::<RetryIndex>()?;
106    if index.scheduled.iter().all(FxHashSet::is_empty) {
107        return None;
108    }
109    let now = observed_now(world);
110    let mut shortest: Option<Duration> = None;
111    for (kind_idx, scheduled) in index.scheduled.iter().enumerate() {
112        for &coords in scheduled {
113            let Some(retries) = world.tiles.query::<&TileRequestRetries>(coords) else {
114                continue;
115            };
116            let state = &retries.0[kind_idx];
117            if !state.out_of_view {
118                if let Some(due) = state.deadline {
119                    let wait = due.saturating_sub(now);
120                    shortest = Some(shortest.map_or(wait, |prev| prev.min(wait)));
121                }
122            }
123        }
124    }
125    shortest
126}

128/// Records the tiles a request family still wants, whether or not it requested them now.
129pub(crate) fn want<S: std::hash::BuildHasher>(
130    world: &mut World,
131    kind: RequestKind,
132    coords: &HashSet<WorldTileCoords, S>,
133) {
134    let idx = kind.index();
135    let wanted_set: FxHashSet<WorldTileCoords> = coords.iter().copied().collect();
136    let active: Vec<WorldTileCoords> = world
137        .resources
138        .get::<RetryIndex>()
139        .map(|retry_idx| {
140            retry_idx.pending[idx]
141                .iter()
142                .chain(retry_idx.scheduled[idx].iter())
143                .copied()
144                .collect()
145        })
146        .unwrap_or_default();
147    let mut missing = Vec::new();
148    for held in active {
149        if let Some(retries) = world.tiles.query_mut::<&mut TileRequestRetries>(held) {
150            retries.0[idx].out_of_view = !wanted_set.contains(&held);
151        } else {
152            missing.push(held);
153        }
154    }
155    let retry_idx = world.resources.get_or_init_mut::<RetryIndex>();
156    for gone in missing {
157        retry_idx.clear_kind(kind, gone);
158    }
159    retry_idx.last_wanted[idx] = Some(wanted_set.clone());
160    world.resources.get_or_init_mut::<WantedRequests>().0[idx] = Some(wanted_set);
161}

163/// Gives up the requests in flight that no request family wanted this frame, so they stop
164/// holding a slot, and returns the tiles they were for. Tiles in `in_use` keep theirs: the
165/// drape requests a family follows lag a frame behind what the view uses.
166pub(crate) fn cancel_unwanted<S: std::hash::BuildHasher>(
167    world: &mut World,
168    in_use: &HashSet<WorldTileCoords, S>,
169) -> Vec<WorldTileCoords> {
170    let Some(wanted) = world
171        .resources
172        .get_mut::<WantedRequests>()
173        .map(std::mem::take)
174    else {
175        return Vec::new();
176    };
177    let mut candidate_set = FxHashSet::default();
178    if let Some(retry_idx) = world.resources.get::<RetryIndex>() {
179        for pending in &retry_idx.pending {
180            for &coords in pending {
181                if !in_use.contains(&coords) {
182                    candidate_set.insert(coords);
183                }
184            }
185        }
186    }
187    if candidate_set.is_empty() {
188        return Vec::new();
189    }
190    let mut coords_list: Vec<WorldTileCoords> = candidate_set.into_iter().collect();
191    coords_list.sort_by_key(|c| c.build_quad_key());
192    let mut cancelled = Vec::new();
193    let mut tiles = Vec::new();
194    let mut given_up = Vec::new();
195    let mut missing = Vec::new();
196    for coords in coords_list {
197        let Some(retries) = world.tiles.query_mut::<&mut TileRequestRetries>(coords) else {
198            missing.push(coords);
199            continue;
200        };
201        for ((state, wanted), kind) in retries.0.iter_mut().zip(&wanted.0).zip([
202            RequestKind::Raster,
203            RequestKind::Dem,
204            RequestKind::Vector,
205        ]) {
206            let unwanted = wanted.as_ref().is_some_and(|set| !set.contains(&coords));
207            if state.pending && unwanted {
208                cancelled.extend(state.attempt);
209                *state = RetryState::default();
210                given_up.push((coords, kind));
211                if !tiles.contains(&coords) {
212                    tiles.push(coords);
213                }
214            }
215        }
216    }
217    if !given_up.is_empty() || !missing.is_empty() {
218        let retry_idx = world.resources.get_or_init_mut::<RetryIndex>();
219        for &(coords, kind) in &given_up {
220            retry_idx.clear_kind(kind, coords);
221        }
222        for gone in missing {
223            for kind in [RequestKind::Raster, RequestKind::Dem, RequestKind::Vector] {
224                retry_idx.clear_kind(kind, gone);
225            }
226        }
227    }
228    for (coords, kind) in given_up {
229        crate::io::tile_load::abandoned(world, coords, kind);
230    }
231    world
232        .resources
233        .get_or_init_mut::<CancelledAttempts>()
234        .0
235        .extend(cancelled);
236    tiles
237}

239/// Stops the worker calls of the attempts cancelled since the last call.
240pub(crate) fn stop_cancelled<K: crate::environment::OffscreenKernel>(
241    world: &mut World,
242    apc: &impl crate::io::apc::AsyncProcedureCall<K>,
243) {
244    let cancelled = world
245        .resources
246        .get_mut::<CancelledAttempts>()
247        .map(|cancelled| std::mem::take(&mut cancelled.0))
248        .unwrap_or_default();
249    for attempt in cancelled {
250        apc.cancel(attempt);
251    }
