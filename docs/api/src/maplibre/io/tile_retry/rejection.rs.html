1//! Requests the worker transport refused to take, which wait out a backoff like failed ones
2//! instead of being offered again every frame.
3
4use super::{
5    index::RetryIndex, mark_refresh_sources, now, observed_now, RequestKind, TileRequestRetries,
6};
7use crate::{coords::WorldTileCoords, tcs::world::World};
8
9/// Records that the transport refused the request for the tile, so it is offered again once
10/// a backoff passes; a host drawing on demand wakes for it after
11/// [`crate::context::MapContext::next_retry_delay`]. `issued_for` is what the request was
12/// for, so the retry is for it again.
13pub(crate) fn rejected(
14    world: &mut World,
15    coords: WorldTileCoords,
16    kind: RequestKind,
17    issued_for: Option<&[String]>,
18) {
19    let now = now(world);
20    let out_of_view = world
21        .resources
22        .get::<RetryIndex>()
23        .is_some_and(|index| index.is_out_of_view(kind, coords));
24    if world.tiles.query::<&TileRequestRetries>(coords).is_none() {
25        if let Some(mut tile) = world.tiles.spawn_mut(coords) {
26            tile.insert(TileRequestRetries::default());
27        }
28    }
29    let Some(retries) = world.tiles.query_mut::<&mut TileRequestRetries>(coords) else {
30        return;
31    };
32    let state = &mut retries.0[kind.index()];
33    state.pending = false;
34    state.stale = false;
35    state.out_of_view = out_of_view;
36    state.back_off(now);
37    world
38        .resources
39        .get_or_init_mut::<RetryIndex>()
40        .finish_pending(kind, coords, true);
41    mark_refresh_sources(world, coords, issued_for);
42}
43
44/// Whether the tile waits out a backoff before its next request, with none in flight.
45pub(crate) fn backing_off(world: &World, coords: WorldTileCoords, kind: RequestKind) -> bool {
46    let now = observed_now(world);
47    world
48        .tiles
49        .query::<&TileRequestRetries>(coords)
50        .is_some_and(|retries| {
51            let state = &retries.0[kind.index()];
52            !state.pending && state.deadline.is_some_and(|due| due > now)
53        })
54}
