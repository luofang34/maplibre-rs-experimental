1//! Bounds how many tiles are being fetched and processed at once.
2//!
3//! A covering can ask for hundreds of tiles in one frame: a flight's destination joins the
4//! requests, and a warm disk cache answers every one of them within a second. Each worker
5//! then tessellates a tile at a time, and a low-zoom planet tile takes hundreds of
6//! megabytes while it is being built. Processing dozens at once exhausts a headset's memory
7//! limit. The request systems take the nearest tiles first and stop once this many are in
8//! flight, so a burst becomes a stream the frame loop drains.
9
10use crate::{
11    coords::WorldTileCoords,
12    io::tile_load,
13    render::memory_budget::MemoryBudget,
14    tcs::{tiles::Tiles, world::World},
15};
16
17/// Tiles that may be fetched or processed at the same time across every source.
18pub const MAX_TILES_IN_FLIGHT: usize = 16;
19
20/// Records whether each request family had to skip at least one wanted tile in its last run
21/// because the in-flight request budget was exhausted (`budget == 0`).
22#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
23pub(crate) struct BackpressureDeferred {
24    pub vector: bool,
25    pub raster: bool,
26    pub dem: bool,
27}
28
29impl BackpressureDeferred {
30    fn any(self) -> bool {
31        self.vector || self.raster || self.dem
32    }
33}
34
35/// Updates whether the vector request pass deferred wanted tiles under backpressure.
36pub(crate) fn set_vector_deferred(world: &mut World, deferred: bool) {
37    world
38        .resources
39        .get_or_init_mut::<BackpressureDeferred>()
40        .vector = deferred;
41}
42
43/// Updates whether the raster request pass deferred wanted tiles under backpressure.
44#[cfg(feature = "raster")]
45pub(crate) fn set_raster_deferred(world: &mut World, deferred: bool) {
46    world
47        .resources
48        .get_or_init_mut::<BackpressureDeferred>()
49        .raster = deferred;
50}
51
52/// Updates whether the DEM request pass deferred wanted tiles under backpressure.
53pub(crate) fn set_dem_deferred(world: &mut World, deferred: bool) {
54    world
55        .resources
56        .get_or_init_mut::<BackpressureDeferred>()
57        .dem = deferred;
58}
59
60/// Whether wanted tiles remain unrequested while the host allows in-flight tile work.
61pub(crate) fn has_unrequested_visible_tiles(world: &World) -> bool {
62    world
63        .resources
64        .get::<BackpressureDeferred>()
65        .is_some_and(|deferred| deferred.any())
66        && allowed_in_flight(world) > 0
67}
68
69/// Whether deferred wanted tiles can be dispatched immediately on the next frame because
70/// `tiles_in_flight` has dropped below the allowed in-flight ceiling.
71pub(crate) fn has_ready_deferred_requests(world: &World) -> bool {
72    has_unrequested_visible_tiles(world) && request_budget(world) > 0
73}
74
75/// Whether every request for the tile has produced its outcome or was given up.
76pub(crate) fn is_settled(tiles: &Tiles, coords: WorldTileCoords) -> bool {
77    !tile_load::is_loading(tiles, coords)
78}
79
80/// Tiles with a request that has not produced its outcome yet.
81pub fn tiles_in_flight(world: &World) -> usize {
82    tile_load::loading_count(world)
83}
84
85/// Cumulative count of tile requests admitted across all request families in `world`.
86pub fn total_tile_requests(world: &World) -> u64 {
87    tile_load::total_issued(world)
88}
89
90fn allowed_in_flight(world: &World) -> usize {
91    world
92        .resources
93        .get::<MemoryBudget>()
94        .copied()
95        .unwrap_or_default()
96        .tiles_in_flight_allowed(MAX_TILES_IN_FLIGHT)
97}
98
99/// How many more tiles may be requested this frame: up to the in-flight bound, fewer
100/// while the host is short of memory and none while it is nearly out.
101pub fn request_budget(world: &World) -> usize {
102    allowed_in_flight(world).saturating_sub(tiles_in_flight(world))
103}
104
105/// Shares available requests with the DEM pass without starving vector tiles under pressure.
106pub(crate) fn vector_request_budget(world: &World, has_terrain: bool) -> usize {
107    let available = request_budget(world);
108    let reserved = if has_terrain {
109        (available / 2).min(4)
110    } else {
111        0
112    };
113    available - reserved
114}
115
116/// Shares available requests with the raster pass so slow remote DEM fetches never starve
117/// local or draped raster tiles.
118pub(crate) fn dem_request_budget(world: &World, has_raster: bool) -> usize {
119    let available = request_budget(world);
120    let reserved = if has_raster {
121        (available / 3).min(8)
122    } else {
123        0
124    };
125    available - reserved
126}
