1//! Bounds how many tiles are being fetched and processed at once.
2//!
3//! A covering can ask for hundreds of tiles in one frame: a flight's destination joins the
4//! requests, and a warm disk cache answers every one of them within a second. Each worker
5//! then tessellates a tile at a time, and a low-zoom planet tile takes hundreds of
6//! megabytes while it is being built. Processing dozens at once exhausts a headset's memory
7//! limit. The request systems take the nearest tiles first and stop once this many are in
8//! flight, so a burst becomes a stream the frame loop drains.
9
10use crate::{
11    coords::WorldTileCoords,
12    raster::RasterLayersDataComponent,
13    render::memory_budget::MemoryBudget,
14    tcs::{tiles::Tiles, world::World},
15    terrain::DemTileComponent,
16    vector::VectorLayerBucketComponent,
17};
18
19/// Tiles that may be fetched or processed at the same time across every source.
20pub const MAX_TILES_IN_FLIGHT: usize = 12;
21
22/// Whether every request for the tile has produced a result.
23pub(crate) fn is_settled(tiles: &Tiles, coords: WorldTileCoords) -> bool {
24    let retry_loading = tiles
25        .query::<&crate::io::tile_retry::TileRequestRetries>(coords)
26        .is_some_and(crate::io::tile_retry::TileRequestRetries::pending);
27    let vector_loading = tiles
28        .query::<&VectorLayerBucketComponent>(coords)
29        .is_some_and(|component| !component.done);
30    let raster_loading = tiles
31        .query::<&RasterLayersDataComponent>(coords)
32        .is_some_and(|component| component.layers.is_empty());
33    let dem_loading = matches!(
34        tiles.query::<&DemTileComponent>(coords),
35        Some(DemTileComponent::Pending)
36    );
37    let symbol_loading = tiles
38        .query::<&crate::sdf::SymbolLayersDataComponent>(coords)
39        .is_some_and(|component| component.pending_assets);
40    !(retry_loading || vector_loading || raster_loading || dem_loading || symbol_loading)
41}
42
43/// Tiles with a request that has not produced its result yet.
44pub fn tiles_in_flight(tiles: &Tiles) -> usize {
45    tiles
46        .tiles
47        .values()
48        .filter(|tile| !is_settled(tiles, tile.coords))
49        .count()
50}
51
52/// How many more tiles may be requested this frame: up to the in-flight bound, fewer
53/// while the host is short of memory and none while it is nearly out.
54pub fn request_budget(world: &World) -> usize {
55    let allowed = world
56        .resources
57        .get::<MemoryBudget>()
58        .copied()
59        .unwrap_or_default()
60        .tiles_in_flight_allowed(MAX_TILES_IN_FLIGHT);
61    allowed.saturating_sub(tiles_in_flight(&world.tiles))
62}
63
64/// Shares available requests with the DEM pass without starving vector tiles under pressure.
65pub(crate) fn vector_request_budget(world: &World, has_terrain: bool) -> usize {
66    let available = request_budget(world);
67    let reserved = if has_terrain {
68        (available / 2).min(4)
69    } else {
70        0
71    };
72    available - reserved
73}
