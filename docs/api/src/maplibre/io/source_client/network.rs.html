3use std::{
4    fmt,
5    sync::{
6        atomic::{AtomicBool, AtomicU64, Ordering},
7        Arc, Mutex, PoisonError,
8    },
9};
10
11use thiserror::Error;
12
13/// Whether remote HTTP/HTTPS network sockets may be used to satisfy source misses.
14#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Hash)]
15pub enum NetworkStatus {
16    /// Remote HTTP and HTTPS network transports are permitted.
17    #[default]
18    Online,
19    /// Remote HTTP and HTTPS misses fail immediately with [`OfflineError`] while local files,
20    /// `mbtiles://`, `data:`, in-memory/disk caches, and offline stores remain active.
21    Offline,
22}
23
24impl NetworkStatus {
25    /// Returns `true` when [`Self::Online`].
26    pub const fn is_online(self) -> bool {
27        matches!(self, Self::Online)
28    }
29
30    /// Returns `true` when [`Self::Offline`].
31    pub const fn is_offline(self) -> bool {
32        matches!(self, Self::Offline)
33    }
34}
35
36/// Alias for [`NetworkStatus`].
37pub type Connectivity = NetworkStatus;
38
39/// A remote HTTP/HTTPS resource miss was suppressed because [`NetworkState`] is [`NetworkStatus::Offline`].
40#[derive(Error, Debug, Clone, PartialEq, Eq)]
41#[error("network is offline; cannot fetch remote resource {url}")]
42pub struct OfflineError {
43    /// The remote URL whose network socket request was blocked while offline.
44    pub url: String,
45}
46
47type StatusListener = Arc<dyn Fn(NetworkStatus) + Send + Sync>;
48
49struct NetworkStateInner {
50    online: AtomicBool,
51    generation: AtomicU64,
52    listeners: Mutex<Vec<StatusListener>>,
53}
54
55/// Cloneable, thread-safe connectivity handle shared by [`super::HttpSourceClient`] and transports.
56#[derive(Clone)]
57pub struct NetworkState(Arc<NetworkStateInner>);
58
59impl Default for NetworkState {
60    fn default() -> Self {
61        Self::new()
62    }
63}
64
65impl fmt::Debug for NetworkState {
66    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
67        f.debug_struct("NetworkState")
68            .field("status", &self.status())
69            .field("generation", &self.generation())
70            .finish()
71    }
72}
73
74impl NetworkState {
75    /// Creates a connectivity handle initially set to [`NetworkStatus::Online`].
76    pub fn new() -> Self {
77        Self::with_status(NetworkStatus::Online)
78    }
79
80    /// Creates a connectivity handle initially set to [`NetworkStatus::Offline`].
81    pub fn offline() -> Self {
82        Self::with_status(NetworkStatus::Offline)
83    }
84
85    /// Creates a connectivity handle with the given initial `status`.
86    pub fn with_status(status: NetworkStatus) -> Self {
87        Self(Arc::new(NetworkStateInner {
88            online: AtomicBool::new(status.is_online()),
89            generation: AtomicU64::new(0),
90            listeners: Mutex::new(Vec::new()),
91        }))
92    }
93
94    /// Whether remote network transports are currently enabled.
95    pub fn is_online(&self) -> bool {
96        self.0.online.load(Ordering::Acquire)
97    }
98
99    /// Current [`NetworkStatus`] snapshot.
100    pub fn status(&self) -> NetworkStatus {
101        if self.is_online() {
102            NetworkStatus::Online
103        } else {
104            NetworkStatus::Offline
105        }
106    }
107
108    /// Monotonic counter incremented each time connectivity transitions between online and offline.
109    pub fn generation(&self) -> u64 {
110        self.0.generation.load(Ordering::Acquire)
111    }
112
113    /// Sets whether remote network sockets are enabled, bumping [`Self::generation`] and notifying
114    /// registered listeners when the state changes.
115    pub fn set_online(&self, online: bool) {
116        let prev = self.0.online.swap(online, Ordering::AcqRel);
117        if prev != online {
118            self.0.generation.fetch_add(1, Ordering::AcqRel);
119            let status = if online {
120                NetworkStatus::Online
121            } else {
122                NetworkStatus::Offline
123            };
124            let callbacks: Vec<StatusListener> = self
125                .0
126                .listeners
127                .lock()
128                .unwrap_or_else(PoisonError::into_inner)
129                .clone();
130            for callback in callbacks {
131                callback(status);
132            }
133        }
134    }
135
136    /// Updates connectivity to `status`.
137    pub fn set_status(&self, status: NetworkStatus) {
138        self.set_online(status.is_online());
139    }
140
141    /// Registers a listener invoked whenever connectivity changes between [`NetworkStatus::Online`]
142    /// and [`NetworkStatus::Offline`].
143    pub fn on_change(&self, listener: impl Fn(NetworkStatus) + Send + Sync + 'static) {
144        self.0
145            .listeners
146            .lock()
147            .unwrap_or_else(PoisonError::into_inner)
148            .push(Arc::new(listener));
149    }
150}
