7/// Caching policy attached to an [`HttpRequest`].
8#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Hash)]
9pub enum CachePolicy {
10    /// Standard caching across in-memory and persistent disk or browser storage.
11    #[default]
12    Normal,
13    /// Volatile source policy: bypasses persistent disk and browser `CacheStorage` across sessions.
14    Volatile,
15}
16
17impl CachePolicy {
18    /// Returns [`Self::Volatile`] when `volatile` is `true`, otherwise [`Self::Normal`].
19    pub const fn from_volatile(volatile: bool) -> Self {
20        if volatile {
21            Self::Volatile
22        } else {
23            Self::Normal
24        }
25    }
26
27    /// Whether persistent cross-session caching must be skipped for this request.
28    pub const fn is_volatile(self) -> bool {
29        matches!(self, Self::Volatile)
30    }
31}

33/// Classification of the map resource being requested, exposed to [`RequestTransform`] hooks.
34#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, PartialOrd, Ord, Hash)]
35pub enum ResourceKind {
36    /// Root MapLibre style JSON document.
37    Style,
38    /// Source metadata such as TileJSON or an external GeoJSON document.
39    Source,
40    /// Vector, raster, or elevation tile payload.
41    Tile,
42    /// Signed-distance-field font glyph range (`{fontstack}/{range}.pbf`).
43    Glyphs,
44    /// Sprite atlas PNG image.
45    SpriteImage,
46    /// Sprite atlas JSON index document.
47    SpriteJson,
48    /// Standalone image source picture.
49    Image,
50    /// Unclassified URL request.
51    #[default]
52    Unknown,
53}
54
55impl ResourceKind {
56    /// Best-effort classification from URL path conventions when a caller supplies only a URL.
57    pub fn infer_from_url(url: &str) -> Self {
58        let path = url.split('?').next().unwrap_or(url);
59        let lower = path.to_ascii_lowercase();
60        if lower.ends_with("sprite.json") || lower.ends_with("sprite@2x.json") {
61            Self::SpriteJson
62        } else if lower.ends_with("sprite.png") || lower.ends_with("sprite@2x.png") {
63            Self::SpriteImage
64        } else if lower.ends_with(".pbf") && looks_like_glyph_range(&lower) {
65            Self::Glyphs
66        } else if lower.ends_with("style.json") {
67            Self::Style
68        } else if lower.ends_with("tiles.json")
69            || lower.ends_with("tilejson.json")
70            || lower.ends_with(".geojson")
71        {
72            Self::Source
73        } else if lower.ends_with(".pbf") || lower.ends_with(".mvt") || lower.ends_with(".mlt") {
74            Self::Tile
75        } else {
76            Self::Unknown
77        }
78    }
79}

81fn looks_like_glyph_range(lower_path: &str) -> bool {
82    let Some(stem) = lower_path.strip_suffix(".pbf") else {
83        return false;
84    };
85    let file = stem.rsplit('/').next().unwrap_or(stem);
86    let Some((start, end)) = file.split_once('-') else {
87        return false;
88    };
89    !start.is_empty()
90        && !end.is_empty()
91        && start.bytes().all(|b| b.is_ascii_digit())
92        && end.bytes().all(|b| b.is_ascii_digit())
93}

95/// Outgoing HTTP/resource request descriptor passed to [`RequestTransform`] and [`super::HttpClient`].
96#[derive(Clone, Debug, PartialEq, Eq)]
97pub struct HttpRequest {
98    /// Target URL (before or after [`RequestTransform`] rewriting).
99    pub url: String,
100    /// Optional byte range for partial fetches (e.g. PMTiles headers and directories).
101    pub range: Option<ByteRange>,
102    /// Custom HTTP request headers `(name, value)` forwarded by native and web transports.
103    pub headers: Vec<(String, String)>,
104    /// Persistent cache policy (`Normal` vs `Volatile`).
105    pub cache_policy: CachePolicy,
106    /// Kind of map resource this request fetches.
107    pub resource_kind: ResourceKind,
108}
109
110impl HttpRequest {
111    /// Constructs a request for `url` and `resource_kind` with [`CachePolicy::Normal`] and no headers.
112    pub fn new(url: impl Into<String>, resource_kind: ResourceKind) -> Self {
113        Self {
114            url: url.into(),
115            range: None,
116            headers: Vec::new(),
117            cache_policy: CachePolicy::Normal,
118            resource_kind,
119        }
120    }
121
122    /// Attaches a [`ByteRange`] to request a partial response slice.
123    pub fn with_range(mut self, range: ByteRange) -> Self {
124        self.range = Some(range);
125        self
126    }
127
128    /// Sets an optional [`ByteRange`] on the request.
129    pub fn with_optional_range(mut self, range: Option<ByteRange>) -> Self {
130        self.range = range;
131        self
132    }
133
134    /// Appends a custom HTTP header `(name, value)` to the request.
135    pub fn with_header(mut self, name: impl Into<String>, value: impl Into<String>) -> Self {
136        self.headers.push((name.into(), value.into()));
137        self
138    }
139
140    /// Sets the [`CachePolicy`] for the request.
141    pub fn with_cache_policy(mut self, cache_policy: CachePolicy) -> Self {
142        self.cache_policy = cache_policy;
143        self
144    }
145}

147/// Thread-safe callback that rewrites URLs or attaches custom HTTP headers prior to transport dispatch.
148pub type RequestTransform = Arc<dyn Fn(HttpRequest) -> HttpRequest + Send + Sync>;
149
150#[derive(Clone, Default)]
151pub(crate) struct TransformSlot(Arc<RwLock<Option<RequestTransform>>>);
152
153impl TransformSlot {
154    pub(crate) fn set(&self, transform: RequestTransform) {
155        *self.0.write().unwrap_or_else(PoisonError::into_inner) = Some(transform);
156    }
157
158    pub(crate) fn clear(&self) {
159        *self.0.write().unwrap_or_else(PoisonError::into_inner) = None;
160    }
161
162    pub(crate) fn apply(&self, req: HttpRequest) -> HttpRequest {
163        let hook = self
164            .0
165            .read()
166            .unwrap_or_else(PoisonError::into_inner)
167            .clone();
168        match hook {
169            Some(transform) => transform(req),
170            None => req,
171        }
172    }
173}

175/// Whether `url` uses a remote `http://` or `https://` scheme that requires network access on cache miss.
176pub fn is_remote_http_url(url: &str) -> bool {
177    url.get(..7)
178        .is_some_and(|prefix| prefix.eq_ignore_ascii_case("http://"))
179        || url
180            .get(..8)
181            .is_some_and(|prefix| prefix.eq_ignore_ascii_case("https://"))
182}

191/// Decodes a `data:[<mediatype>][;base64],<data>` URI locally without touching any network transport.
192/// Returns `None` when `url` does not start with `data:`.
193pub(crate) fn decode_data_url(
194    url: &str,
195    range: Option<ByteRange>,
196) -> Option<Result<Vec<u8>, SourceFetchError>> {
197    if !url
198        .get(..5)
199        .is_some_and(|prefix| prefix.eq_ignore_ascii_case("data:"))
200    {
201        return None;
202    }
203    let rest = &url[5..];
204    let Some((meta, payload)) = rest.split_once(',') else {
205        return Some(Err(SourceFetchError(Box::new(DataUrlError {
206            reason: "missing comma separator",
207        }))));
208    };
209    let is_base64 = meta
210        .rsplit(';')
211        .next()
212        .is_some_and(|token| token.trim().eq_ignore_ascii_case("base64"));
213    let decoded = if is_base64 {
214        decode_base64(payload).ok_or_else(|| {
215            SourceFetchError(Box::new(DataUrlError {
216                reason: "invalid base64 payload",
217            }))
218        })
219    } else {
220        Ok(percent_decode_bytes(payload))
221    };
222    Some(decoded.and_then(|bytes| match range {
223        Some(range) => range.slice(url, &bytes).map(<[u8]>::to_vec),
224        None => Ok(bytes),
225    }))
226}

228fn percent_decode_bytes(input: &str) -> Vec<u8> {
229    let raw = input.as_bytes();
230    let mut out = Vec::with_capacity(raw.len());
231    let mut idx = 0;
232    while idx < raw.len() {
233        if raw[idx] == b'%' && idx + 2 < raw.len() {
234            if let (Some(hi), Some(lo)) = (
235                (raw[idx + 1] as char).to_digit(16),
236                (raw[idx + 2] as char).to_digit(16),
237            ) {
238                out.push(((hi << 4) | lo) as u8);
239                idx += 3;
240                continue;
241            }
242        }
243        out.push(raw[idx]);
244        idx += 1;
245    }
246    out
247}

249fn decode_base64(input: &str) -> Option<Vec<u8>> {
250    let mut clean = Vec::with_capacity(input.len());
251    for byte in input.bytes() {
252        if !byte.is_ascii_whitespace() {
253            clean.push(byte);
254        }
255    }
256    if clean.is_empty() {
257        return Some(Vec::new());
258    }
259    let mut out = Vec::with_capacity(clean.len() * 3 / 4);
260    let mut buf = 0_u32;
261    let mut bits = 0_u8;
262    let mut seen_pad = false;
263    for byte in clean {
264        if byte == b'=' {
265            seen_pad = true;
266            continue;
267        }
268        if seen_pad {
269            return None;
270        }
271        let val = match byte {
272            b'A'..=b'Z' => byte - b'A',
273            b'a'..=b'z' => byte - b'a' + 26,
274            b'0'..=b'9' => byte - b'0' + 52,
275            b'+' | b'-' => 62,
276            b'/' | b'_' => 63,
277            _ => return None,
278        };
279        buf = (buf << 6) | u32::from(val);
280        bits += 6;
281        if bits >= 8 {
282            bits -= 8;
283            out.push(((buf >> bits) & 0xff) as u8);
284        }
285    }
286    Some(out)
