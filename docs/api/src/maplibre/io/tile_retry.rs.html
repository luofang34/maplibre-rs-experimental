18/// Request families whose completion controls a retry deadline.
19#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash, Serialize, Deserialize)]
20pub enum RequestKind {
21    /// Decoded map imagery.
22    Raster,
23    /// Elevation data used for terrain and shading.
24    Dem,
25    /// Vector geometry and its associated symbol processing.
26    Vector,
27}
28
29impl MessageTag for RequestKind {
30    fn dyn_clone(&self) -> Box<dyn MessageTag> {
31        Box::new(*self)
32    }
33}
34
35impl RequestKind {
36    /// Identifies final request results independently from pixel payloads.
37    pub fn message_tag(self) -> &'static dyn MessageTag {
38        match self {
39            Self::Raster => &Self::Raster,
40            Self::Dem => &Self::Dem,
41            Self::Vector => &Self::Vector,
42        }
43    }
44
45    pub(crate) fn index(self) -> usize {
46        match self {
47            Self::Raster => 0,
48            Self::Dem => 1,
49            Self::Vector => 2,
50        }
51    }
52}

54/// Disposition after all sources in one tile request have completed.
55#[derive(Clone, Copy, Debug, PartialEq, Eq, Serialize, Deserialize)]
56pub enum RequestDisposition {
57    /// Every source completed without a transient failure.
58    Complete,
59    /// At least one source failed temporarily and can be requested again.
60    Retry,
61}

63/// Final worker result; its attempt identifies the admitted request across tile eviction.
64#[derive(Clone, Copy, Debug, Serialize, Deserialize)]
65pub struct TileRequestOutcome {
66    /// Tile coordinates shared by this request's source results.
67    pub coords: WorldTileCoords,
68    /// Request family whose admission this outcome finishes.
69    pub kind: RequestKind,
70    /// Map-owned attempt identifier; direct untracked calls carry `None`.
71    pub attempt: Option<u64>,
72    /// Whether the request needs another attempt.
73    pub disposition: RequestDisposition,
74}
75
76impl IntoMessage for TileRequestOutcome {
77    fn into(self) -> Message {
78        Message::new(self.kind.message_tag(), Box::new(self))
79    }
80}

82#[derive(Default)]
83struct RetryState {
84    attempt: Option<u64>,
85    pending: bool,
86    delay: Duration,
87    deadline: Option<Duration>,
88    /// The style changed while this request was in flight, so its result is refetched.
89    stale: bool,
90    /// The sources the attempt was limited to, so that a retry of it is for them again;
91    /// `None` for an attempt that processed every source.
92    issued_for: Option<Vec<String>>,
93    /// The family left the tile out of what it wanted in its last frame, so a passed retry
94    /// deadline waits for the tile to come back into view instead of asking for frames.
95    out_of_view: bool,
96}
97
98impl RetryState {
99    /// Whether the next frame has a retry of this tile to issue.
100    fn retry_due(&self, now: Duration) -> bool {
101        !self.out_of_view && self.deadline.is_some_and(|due| due <= now)
102    }
103
104    /// Schedules the next attempt after a delay that doubles with each failure.
105    fn back_off(&mut self, now: Duration) {
106        self.delay = if self.delay.is_zero() {
107            Duration::from_secs(1)
108        } else {
109            self.delay.saturating_mul(2).min(Duration::from_secs(30))
110        };
111        self.deadline = Some(now.saturating_add(self.delay));
112    }
113}

116pub(crate) struct TileRequestRetries([RetryState; 3]);

123fn retry_states(world: &World) -> impl Iterator<Item = &RetryState> {
124    world
125        .tiles
126        .tiles
127        .values()
128        .filter_map(|tile| world.tiles.query::<&TileRequestRetries>(tile.coords))
129        .flat_map(|retries| retries.0.iter())
130}

132/// Whether a frame has request work: a request in flight, whose result the next frame applies,
133/// or a retry whose deadline has passed, which the next frame issues.
134pub(crate) fn needs_frame(world: &World) -> bool {
135    let now = observed_now(world);
136    retry_states(world).any(|state| state.pending || state.retry_due(now))
137}
138
139/// Shortest wait until a scheduled retry deadline is due, or `None` when no retry waits.
140pub(crate) fn next_retry_delay(world: &World) -> Option<Duration> {
141    let now = observed_now(world);
142    retry_states(world)
143        .filter(|state| !state.out_of_view)
144        .filter_map(|state| state.deadline.map(|due| due.saturating_sub(now)))
145        .min()
146}

148/// Requests every requested tile of `kind` again, keeping what it shows until the new data lands.
149pub(crate) fn refresh(world: &mut World, kind: RequestKind) {
150    let coords: Vec<WorldTileCoords> = world.tiles.tiles.values().map(|tile| tile.coords).collect();
151    refresh_tiles(world, kind, &coords);
152}

154/// What the pending refresh of a tile is for, so that the request made for it can leave the
155/// tile's other sources as they are.
156#[derive(Default)]
157enum RefreshSources {
158    /// No refresh is pending, or the request for it has been made.
159    #[default]
160    None,
161    /// The data of these sources changed.
162    Some(BTreeSet<String>),
163    /// Anything about the tile may have changed.
164    Every,
165}
166impl TileComponent for RefreshSources {}
167
168/// Like [`refresh_tiles`], for a change to the data of `sources` alone: a request made for
169/// the refresh processes those sources and keeps what the tile shows of the others.
170pub(crate) fn refresh_tiles_for_sources(
171    world: &mut World,
172    kind: RequestKind,
173    tiles: &[WorldTileCoords],
174    sources: &[String],
175) {
176    for coords in tiles {
177        mark_refresh_sources(world, *coords, Some(sources));
178    }
179    refresh_due(world, kind, tiles);
180}
181
182/// Notes which sources a pending refresh of the tile is for: those given add to the ones
183/// noted before, and a refresh for every source (`None`) outranks any for some, whichever
184/// comes first.
185fn mark_refresh_sources(world: &mut World, coords: WorldTileCoords, sources: Option<&[String]>) {
186    let marked = match sources {
187        None => RefreshSources::Every,
188        Some(sources) => RefreshSources::Some(sources.iter().cloned().collect()),
189    };
190    if let Some(existing) = world.tiles.query_mut::<&mut RefreshSources>(coords) {
191        match (existing, marked) {
192            (RefreshSources::Every, _) => {}
193            (RefreshSources::Some(scope), RefreshSources::Some(sources)) => scope.extend(sources),
194            (existing, marked) => *existing = marked,
195        }
196        return;
197    }
198    if let Some(mut tile) = world.tiles.spawn_mut(coords) {
199        tile.insert(marked);
200    }
201}

203/// The sources the pending refresh of the tile is for, cleared for the request that takes
204/// them; `None` when the refresh is for every source, or none is pending.
205pub(crate) fn take_refresh_sources(
206    world: &mut World,
207    coords: WorldTileCoords,
208) -> Option<Vec<String>> {
209    let pending = world.tiles.query_mut::<&mut RefreshSources>(coords)?;
210    match std::mem::take(pending) {
211        RefreshSources::Some(sources) => Some(sources.into_iter().collect()),
212        RefreshSources::None | RefreshSources::Every => None,
213    }
214}

216/// Requests `tiles` again for `kind`, keeping what they show until the new data lands.
217pub(crate) fn refresh_tiles(world: &mut World, kind: RequestKind, tiles: &[WorldTileCoords]) {
218    for coords in tiles {
219        mark_refresh_sources(world, *coords, None);
220    }
221    refresh_due(world, kind, tiles);
222}
223
224/// Makes the next request for each of `tiles` due now, or once the one in flight lands.
225fn refresh_due(world: &mut World, kind: RequestKind, tiles: &[WorldTileCoords]) {
226    crate::render::frame_signals::mark_dirty(world);
227    let now = now(world);
228    for coords in tiles {
229        let Some(retries) = world.tiles.query_mut::<&mut TileRequestRetries>(*coords) else {
230            continue;
231        };
232        let state = &mut retries.0[kind.index()];
233        if state.attempt.is_none() {
234            continue;
235        }
236        if state.pending {
237            state.stale = true;
238        } else if state.deadline.is_none() {
239            // A tile already backing off retries at its deadline with the new data.
240            state.delay = Duration::ZERO;
241            state.deadline = Some(now);
242        }
243    }
244}

246/// Requests a finished tile again after `delay`, because part of what its worker sent later
247/// was unavailable for now.
248pub(crate) fn retry_later(
249    world: &mut World,
250    coords: WorldTileCoords,
251    kind: RequestKind,
252    delay: Duration,
253) {
254    let now = now(world);
255    let Some(retries) = world.tiles.query_mut::<&mut TileRequestRetries>(coords) else {
256        return;
257    };
258    let state = &mut retries.0[kind.index()];
259    if state.pending || state.deadline.is_some() {
260        return;
261    }
262    state.deadline = Some(now.saturating_add(delay));
263    // The retry brings what the whole tile was asked for, not what a later data change was.
264    mark_refresh_sources(world, coords, None);
265}

267pub(crate) fn due(world: &mut World, coords: WorldTileCoords, kind: RequestKind) -> bool {
268    let now = now(world);
269    world
270        .tiles
271        .query::<&TileRequestRetries>(coords)
272        .is_some_and(|retries| {
273            let state = &retries.0[kind.index()];
274            !state.pending && state.deadline.is_some_and(|deadline| now >= deadline)
275        })
276}

288pub(crate) fn next_attempt(world: &mut World) -> u64 {
289    let attempts = world.resources.get_or_init_mut::<RequestAttempts>();
290    let next = attempts.0.get().wrapping_add(1);
291    attempts.0.set(next);
292    next
293}

295/// Records the attempt made for the tile, limited to `only_sources` or for every source.
296pub(crate) fn started(
297    world: &mut World,
298    coords: WorldTileCoords,
299    kind: RequestKind,
300    attempt: u64,
301    only_sources: Option<&[String]>,
302) {
303    if world.tiles.query::<&TileRequestRetries>(coords).is_none() {
304        if let Some(mut tile) = world.tiles.spawn_mut(coords) {
305            tile.insert(TileRequestRetries::default());
306        }
307    }
308    if let Some(retries) = world.tiles.query_mut::<&mut TileRequestRetries>(coords) {
309        let state = &mut retries.0[kind.index()];
310        state.attempt = Some(attempt);
311        state.pending = true;
312        state.deadline = None;
313        state.stale = false;
314        state.issued_for = only_sources.map(<[String]>::to_vec);
315    }
316    crate::io::tile_load::issued(world, coords, kind);
317}

319pub(crate) fn completed(world: &mut World, outcome: TileRequestOutcome) {
320    let now = now(world);
321    let Some(retries) = world
322        .tiles
323        .query_mut::<&mut TileRequestRetries>(outcome.coords)
324    else {
325        return;
326    };
327    let state = &mut retries.0[outcome.kind.index()];
328    // An evicted request can finish after the same coordinates have been requested again.
329    if !state.pending || state.attempt != outcome.attempt {
330        return;
331    }
332    state.pending = false;
333    let stale = std::mem::take(&mut state.stale);
334    let loaded = matches!(outcome.disposition, RequestDisposition::Complete);
335    crate::io::tile_load::landed(world, outcome.coords, outcome.kind, !loaded);
336    let Some(retries) = world
337        .tiles
338        .query_mut::<&mut TileRequestRetries>(outcome.coords)
339    else {
340        return;
341    };
342    let state = &mut retries.0[outcome.kind.index()];
343    let mut retried_for = None;
344    match outcome.disposition {
345        RequestDisposition::Complete => {
346            state.delay = Duration::ZERO;
347            state.deadline = stale.then_some(now);
348        }
349        RequestDisposition::Retry => {
350            state.back_off(now);
351            retried_for = Some(state.issued_for.clone());
352        }
353    }
354    // The retry is for what the attempt was for, so a data change noted since cannot
355    // narrow it to its own sources and lose the ones that failed.
356    if let Some(issued_for) = retried_for {
357        mark_refresh_sources(world, outcome.coords, issued_for.as_deref());
358    }
359    crate::render::frame_signals::resource_ready(
360        world,
361        crate::render::frame_signals::ResourceReady::Tile {
362            coords: outcome.coords,
363            kind: outcome.kind,
364            loaded,
365        },
366    );
367}

369/// Payloads from a finished attempt remain valid until a new attempt replaces its identity.
370pub(crate) fn accepts(
371    world: &World,
372    coords: WorldTileCoords,
373    kind: RequestKind,
374    attempt: Option<u64>,
375) -> bool {
376    world
377        .tiles
378        .query::<&TileRequestRetries>(coords)
379        .and_then(|retries| retries.0[kind.index()].attempt)
380        == attempt
381}

383/// The tiles each request family considered this frame; a family that did not run leaves
384/// its requests alone.
385#[derive(Default)]
386struct WantedRequests([Option<rustc_hash::FxHashSet<WorldTileCoords>>; 3]);
387
388/// Attempts given up on, for the next request system to stop in its worker.
389#[derive(Default)]
390struct CancelledAttempts(Vec<u64>);
391
392/// Records the tiles a request family still wants, whether or not it requested them now.
393pub(crate) fn want<S: std::hash::BuildHasher>(
394    world: &mut World,
395    kind: RequestKind,
396    coords: &HashSet<WorldTileCoords, S>,
397) {
398    let held: Vec<WorldTileCoords> = world.tiles.tiles.values().map(|tile| tile.coords).collect();
399    for held in held {
400        if let Some(retries) = world.tiles.query_mut::<&mut TileRequestRetries>(held) {
401            retries.0[kind.index()].out_of_view = !coords.contains(&held);
402        }
403    }
404    world.resources.get_or_init_mut::<WantedRequests>().0[kind.index()] =
405        Some(coords.iter().copied().collect());
406}

408/// Gives up the requests in flight that no request family wanted this frame, so they stop
409/// holding a slot, and returns the tiles they were for. Tiles in `in_use` keep theirs: the
410/// drape requests a family follows lag a frame behind what the view uses.
411pub(crate) fn cancel_unwanted<S: std::hash::BuildHasher>(
412    world: &mut World,
413    in_use: &HashSet<WorldTileCoords, S>,
414) -> Vec<WorldTileCoords> {
415    let Some(wanted) = world
416        .resources
417        .get_mut::<WantedRequests>()
418        .map(std::mem::take)
419    else {
420        return Vec::new();
421    };
422    let coords: Vec<WorldTileCoords> = world
423        .tiles
424        .tiles
425        .values()
426        .map(|tile| tile.coords)
427        .filter(|coords| !in_use.contains(coords))
428        .collect();
429    let mut cancelled = Vec::new();
430    let mut tiles = Vec::new();
431    let mut given_up = Vec::new();
432    for coords in coords {
433        let Some(retries) = world.tiles.query_mut::<&mut TileRequestRetries>(coords) else {
434            continue;
435        };
436        for ((state, wanted), kind) in retries.0.iter_mut().zip(&wanted.0).zip([
437            RequestKind::Raster,
438            RequestKind::Dem,
439            RequestKind::Vector,
440        ]) {
441            let unwanted = wanted.as_ref().is_some_and(|set| !set.contains(&coords));
442            if state.pending && unwanted {
443                cancelled.extend(state.attempt);
444                *state = RetryState::default();
445                given_up.push((coords, kind));
446                if !tiles.contains(&coords) {
447                    tiles.push(coords);
448                }
449            }
450        }
451    }
452    for (coords, kind) in given_up {
453        crate::io::tile_load::abandoned(world, coords, kind);
454    }
455    world
456        .resources
457        .get_or_init_mut::<CancelledAttempts>()
458        .0
459        .extend(cancelled);
460    tiles
461}

463/// Stops the worker calls of the attempts cancelled since the last call.
464pub(crate) fn stop_cancelled<K: crate::environment::OffscreenKernel>(
465    world: &mut World,
466    apc: &impl crate::io::apc::AsyncProcedureCall<K>,
467) {
468    let cancelled = world
469        .resources
470        .get_mut::<CancelledAttempts>()
471        .map(|cancelled| std::mem::take(&mut cancelled.0))
472        .unwrap_or_default();
473    for attempt in cancelled {
474        apc.cancel(attempt);
475    }
476}
