15/// Request families whose completion controls a retry deadline.
16#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash, Serialize, Deserialize)]
17pub enum RequestKind {
18    /// Decoded map imagery.
19    Raster,
20    /// Elevation data used for terrain and shading.
21    Dem,
22    /// Vector geometry and its associated symbol processing.
23    Vector,
24}
25
26impl MessageTag for RequestKind {
27    fn dyn_clone(&self) -> Box<dyn MessageTag> {
28        Box::new(*self)
29    }
30}
31
32impl RequestKind {
33    /// Identifies final request results independently from pixel payloads.
34    pub fn message_tag(self) -> &'static dyn MessageTag {
35        match self {
36            Self::Raster => &Self::Raster,
37            Self::Dem => &Self::Dem,
38            Self::Vector => &Self::Vector,
39        }
40    }
41
42    fn index(self) -> usize {
43        match self {
44            Self::Raster => 0,
45            Self::Dem => 1,
46            Self::Vector => 2,
47        }
48    }
49}

51/// Disposition after all sources in one tile request have completed.
52#[derive(Clone, Copy, Debug, PartialEq, Eq, Serialize, Deserialize)]
53pub enum RequestDisposition {
54    /// Every source completed without a transient failure.
55    Complete,
56    /// At least one source failed temporarily and can be requested again.
57    Retry,
58}

60/// Final worker result; its attempt identifies the admitted request across tile eviction.
61#[derive(Clone, Copy, Debug, Serialize, Deserialize)]
62pub struct TileRequestOutcome {
63    /// Tile coordinates shared by this request's source results.
64    pub coords: WorldTileCoords,
65    /// Request family whose admission this outcome finishes.
66    pub kind: RequestKind,
67    /// Map-owned attempt identifier; direct untracked calls carry `None`.
68    pub attempt: Option<u64>,
69    /// Whether the request needs another attempt.
70    pub disposition: RequestDisposition,
71}
72
73impl IntoMessage for TileRequestOutcome {
74    fn into(self) -> Message {
75        Message::new(self.kind.message_tag(), Box::new(self))
76    }
77}

79#[derive(Default)]
80struct RetryState {
81    attempt: Option<u64>,
82    pending: bool,
83    delay: Duration,
84    deadline: Option<Duration>,
85    /// The style changed while this request was in flight, so its result is refetched.
86    stale: bool,
87}

89#[derive(Default)]
90pub(crate) struct TileRequestRetries([RetryState; 3]);
91impl TileComponent for TileRequestRetries {}

93impl TileRequestRetries {
94    pub(crate) fn pending(&self) -> bool {
95        self.0.iter().any(|state| state.pending)
96    }
97}

103struct RequestClock {
104    started: Instant,
105    last: Duration,
106}
107
108impl Default for RequestClock {
109    fn default() -> Self {
110        Self {
111            started: Instant::now(),
112            last: Duration::ZERO,
113        }
114    }
115}

117fn now(world: &mut World) -> Duration {
118    let timestamp = world
119        .resources
120        .get::<FrameInput>()
121        .map(|input| input.timestamp);
122    let clock = world.resources.get_or_init_mut::<RequestClock>();
123    clock.last = clock
124        .last
125        .max(timestamp.unwrap_or_else(|| clock.started.elapsed()));
126    clock.last
127}

129/// Whether a frame has request work: a request in flight, whose result the next frame applies,
130/// or a retry whose deadline has passed, which the next frame issues.
131pub(crate) fn needs_frame(world: &World) -> bool {
132    let now = world
133        .resources
134        .get::<RequestClock>()
135        .map_or(Duration::ZERO, |clock| {
136            clock.last.max(clock.started.elapsed())
137        });
138    world.tiles.tiles.values().any(|tile| {
139        world
140            .tiles
141            .query::<&TileRequestRetries>(tile.coords)
142            .is_some_and(|retries| {
143                retries
144                    .0
145                    .iter()
146                    .any(|state| state.pending || state.deadline.is_some_and(|due| due <= now))
147            })
148    })
149}

151/// Requests every requested tile of `kind` again, keeping what it shows until the new data lands.
152pub(crate) fn refresh(world: &mut World, kind: RequestKind) {
153    let coords: Vec<WorldTileCoords> = world.tiles.tiles.values().map(|tile| tile.coords).collect();
154    refresh_tiles(world, kind, &coords);
155}
156
157/// Requests `tiles` again for `kind`, keeping what they show until the new data lands.
158pub(crate) fn refresh_tiles(world: &mut World, kind: RequestKind, tiles: &[WorldTileCoords]) {
159    crate::render::frame_signals::mark_dirty(world);
160    let now = now(world);
161    for coords in tiles {
162        let Some(retries) = world.tiles.query_mut::<&mut TileRequestRetries>(*coords) else {
163            continue;
164        };
165        let state = &mut retries.0[kind.index()];
166        if state.attempt.is_none() {
167            continue;
168        }
169        if state.pending {
170            state.stale = true;
171        } else if state.deadline.is_none() {
172            // A tile already backing off retries at its deadline with the new data.
173            state.delay = Duration::ZERO;
174            state.deadline = Some(now);
175        }
176    }
177}

179/// Requests a finished tile again after `delay`, because part of what its worker sent later
180/// was unavailable for now.
181pub(crate) fn retry_later(
182    world: &mut World,
183    coords: WorldTileCoords,
184    kind: RequestKind,
185    delay: Duration,
186) {
187    let now = now(world);
188    let Some(retries) = world.tiles.query_mut::<&mut TileRequestRetries>(coords) else {
189        return;
190    };
191    let state = &mut retries.0[kind.index()];
192    if state.pending || state.deadline.is_some() {
193        return;
194    }
195    state.deadline = Some(now.saturating_add(delay));
196}

198pub(crate) fn due(world: &mut World, coords: WorldTileCoords, kind: RequestKind) -> bool {
199    let now = now(world);
200    world
201        .tiles
202        .query::<&TileRequestRetries>(coords)
203        .is_some_and(|retries| {
204            let state = &retries.0[kind.index()];
205            !state.pending && state.deadline.is_some_and(|deadline| now >= deadline)
206        })
207}

209pub(crate) fn waiting(world: &World, coords: WorldTileCoords, kind: RequestKind) -> bool {

214            let state = &retries.0[kind.index()];

219pub(crate) fn next_attempt(world: &mut World) -> u64 {
220    let attempts = world.resources.get_or_init_mut::<RequestAttempts>();
221    let next = attempts.0.get().wrapping_add(1);
222    attempts.0.set(next);
223    next
224}

226pub(crate) fn started(world: &mut World, coords: WorldTileCoords, kind: RequestKind, attempt: u64) {
227    if world.tiles.query::<&TileRequestRetries>(coords).is_none() {
228        if let Some(mut tile) = world.tiles.spawn_mut(coords) {
229            tile.insert(TileRequestRetries::default());
230        }
231    }
232    if let Some(retries) = world.tiles.query_mut::<&mut TileRequestRetries>(coords) {
233        let state = &mut retries.0[kind.index()];
234        state.attempt = Some(attempt);
235        state.pending = true;
236        state.deadline = None;
237        state.stale = false;
238    }
239}

241pub(crate) fn completed(world: &mut World, outcome: TileRequestOutcome) {
242    let now = now(world);
243    let Some(retries) = world
244        .tiles
245        .query_mut::<&mut TileRequestRetries>(outcome.coords)
246    else {
247        return;
248    };
249    let state = &mut retries.0[outcome.kind.index()];
250    // An evicted request can finish after the same coordinates have been requested again.
251    if !state.pending || state.attempt != outcome.attempt {
252        return;
253    }
254    state.pending = false;
255    let stale = std::mem::take(&mut state.stale);
256    let loaded = matches!(outcome.disposition, RequestDisposition::Complete);
257    match outcome.disposition {
258        RequestDisposition::Complete => {
259            state.delay = Duration::ZERO;
260            state.deadline = stale.then_some(now);
261        }
262        RequestDisposition::Retry => {
263            state.delay = if state.delay.is_zero() {
264                Duration::from_secs(1)
265            } else {
266                state.delay.saturating_mul(2).min(Duration::from_secs(30))
267            };
268            state.deadline = Some(now.saturating_add(state.delay));
269        }
270    }
271    crate::render::frame_signals::resource_ready(
272        world,
273        crate::render::frame_signals::ResourceReady::Tile {
274            coords: outcome.coords,
275            kind: outcome.kind,
276            loaded,
277        },
278    );
279}

281/// Payloads from a finished attempt remain valid until a new attempt replaces its identity.
282pub(crate) fn accepts(
283    world: &World,
284    coords: WorldTileCoords,
285    kind: RequestKind,
286    attempt: Option<u64>,
287) -> bool {
288    world
289        .tiles
290        .query::<&TileRequestRetries>(coords)
291        .and_then(|retries| retries.0[kind.index()].attempt)
292        == attempt
293}

295/// The tiles each request family considered this frame; a family that did not run leaves
296/// its requests alone.
297#[derive(Default)]
298struct WantedRequests([Option<HashSet<WorldTileCoords>>; 3]);

300/// Attempts given up on, for the next request system to stop in its worker.
301#[derive(Default)]
302struct CancelledAttempts(Vec<u64>);

304/// Records the tiles a request family still wants, whether or not it requested them now.
305pub(crate) fn want(world: &mut World, kind: RequestKind, coords: &HashSet<WorldTileCoords>) {
306    world.resources.get_or_init_mut::<WantedRequests>().0[kind.index()] = Some(coords.clone());
307}

309/// Gives up the requests in flight that no request family wanted this frame, so they stop
310/// holding a slot, and returns the tiles they were for. Tiles in `in_use` keep theirs: the
311/// drape requests a family follows lag a frame behind what the view uses.
312pub(crate) fn cancel_unwanted(
313    world: &mut World,
314    in_use: &HashSet<WorldTileCoords>,
315) -> Vec<WorldTileCoords> {
316    let Some(wanted) = world
317        .resources
318        .get_mut::<WantedRequests>()
319        .map(std::mem::take)
320    else {
321        return Vec::new();
322    };
323    let coords: Vec<WorldTileCoords> = world
324        .tiles
325        .tiles
326        .values()
327        .map(|tile| tile.coords)
328        .filter(|coords| !in_use.contains(coords))
329        .collect();
330    let mut cancelled = Vec::new();
331    let mut tiles = Vec::new();
332    for coords in coords {
333        let Some(retries) = world.tiles.query_mut::<&mut TileRequestRetries>(coords) else {
334            continue;
335        };
336        for (state, wanted) in retries.0.iter_mut().zip(&wanted.0) {
337            let unwanted = wanted.as_ref().is_some_and(|set| !set.contains(&coords));
338            if state.pending && unwanted {
339                cancelled.extend(state.attempt);
340                *state = RetryState::default();
341                if !tiles.contains(&coords) {
342                    tiles.push(coords);
343                }
344            }
345        }
346    }
347    world
348        .resources
349        .get_or_init_mut::<CancelledAttempts>()
350        .0
351        .extend(cancelled);
352    tiles
353}

355/// Stops the worker calls of the attempts cancelled since the last call.
356pub(crate) fn stop_cancelled<K: crate::environment::OffscreenKernel>(
357    world: &mut World,
358    apc: &impl crate::io::apc::AsyncProcedureCall<K>,
359) {
360    let cancelled = world
361        .resources
362        .get_mut::<CancelledAttempts>()
363        .map(|cancelled| std::mem::take(&mut cancelled.0))
364        .unwrap_or_default();
365    for attempt in cancelled {
366        apc.cancel(attempt);
367    }
368}
