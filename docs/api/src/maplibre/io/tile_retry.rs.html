20/// Request families whose completion controls a retry deadline.
21#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash, Serialize, Deserialize)]
22pub enum RequestKind {
23    /// Decoded map imagery.
24    Raster,
25    /// Elevation data used for terrain and shading.
26    Dem,
27    /// Vector geometry and its associated symbol processing.
28    Vector,
29}
30
31impl MessageTag for RequestKind {
32    fn dyn_clone(&self) -> Box<dyn MessageTag> {
33        Box::new(*self)
34    }
35}
36
37impl RequestKind {
38    /// Identifies final request results independently from pixel payloads.
39    pub fn message_tag(self) -> &'static dyn MessageTag {
40        match self {
41            Self::Raster => &Self::Raster,
42            Self::Dem => &Self::Dem,
43            Self::Vector => &Self::Vector,
44        }
45    }
46
47    fn index(self) -> usize {
48        match self {
49            Self::Raster => 0,
50            Self::Dem => 1,
51            Self::Vector => 2,
52        }
53    }
54}

56/// Disposition after all sources in one tile request have completed.
57#[derive(Clone, Copy, Debug, PartialEq, Eq, Serialize, Deserialize)]
58pub enum RequestDisposition {
59    /// Every source completed without a transient failure.
60    Complete,
61    /// At least one source failed temporarily and can be requested again.
62    Retry,
63}
64
65/// Final worker result; its attempt identifies the admitted request across tile eviction.
66#[derive(Clone, Copy, Debug, Serialize, Deserialize)]
67pub struct TileRequestOutcome {
68    /// Tile coordinates shared by this request's source results.
69    pub coords: WorldTileCoords,
70    /// Request family whose admission this outcome finishes.
71    pub kind: RequestKind,
72    /// Map-owned attempt identifier; direct untracked calls carry `None`.
73    pub attempt: Option<u64>,
74    /// Whether the request needs another attempt.
75    pub disposition: RequestDisposition,
76}
77
78impl IntoMessage for TileRequestOutcome {
79    fn into(self) -> Message {
80        Message::new(self.kind.message_tag(), Box::new(self))
81    }
82}

84#[derive(Default)]
85struct RetryState {
86    attempt: Option<u64>,
87    pending: bool,
88    delay: Duration,
89    deadline: Option<Duration>,
90    /// The style changed while this request was in flight, so its result is refetched.
91    stale: bool,
92    /// The sources the attempt was limited to, so that a retry of it is for them again;
93    /// `None` for an attempt that processed every source.
94    issued_for: Option<Vec<String>>,
95}

98pub(crate) struct TileRequestRetries([RetryState; 3]);
99impl TileComponent for TileRequestRetries {}

101impl TileRequestRetries {
102    pub(crate) fn pending(&self) -> bool {
103        self.0.iter().any(|state| state.pending)
104    }

111struct RequestClock {
112    started: Instant,
113    last: Duration,
114}
115
116impl Default for RequestClock {
117    fn default() -> Self {
118        Self {
119            started: Instant::now(),
120            last: Duration::ZERO,
121        }
122    }
123}

125fn now(world: &mut World) -> Duration {
126    let timestamp = world
127        .resources
128        .get::<FrameInput>()
129        .map(|input| input.timestamp);
130    let clock = world.resources.get_or_init_mut::<RequestClock>();
131    clock.last = clock
132        .last
133        .max(timestamp.unwrap_or_else(|| clock.started.elapsed()));
134    clock.last
135}

137/// Whether a frame has request work: a request in flight, whose result the next frame applies,
138/// or a retry whose deadline has passed, which the next frame issues.
139pub(crate) fn needs_frame(world: &World) -> bool {
140    let now = world
141        .resources
142        .get::<RequestClock>()
143        .map_or(Duration::ZERO, |clock| {
144            clock.last.max(clock.started.elapsed())
145        });
146    world.tiles.tiles.values().any(|tile| {
147        world
148            .tiles
149            .query::<&TileRequestRetries>(tile.coords)
150            .is_some_and(|retries| {
151                retries
152                    .0
153                    .iter()
154                    .any(|state| state.pending || state.deadline.is_some_and(|due| due <= now))
155            })
156    })
157}

159/// Requests every requested tile of `kind` again, keeping what it shows until the new data lands.
160pub(crate) fn refresh(world: &mut World, kind: RequestKind) {
161    let coords: Vec<WorldTileCoords> = world.tiles.tiles.values().map(|tile| tile.coords).collect();
162    refresh_tiles(world, kind, &coords);
163}

165/// What the pending refresh of a tile is for, so that the request made for it can leave the
166/// tile's other sources as they are.
167#[derive(Default)]
168enum RefreshSources {
169    /// No refresh is pending, or the request for it has been made.
170    #[default]
171    None,
172    /// The data of these sources changed.
173    Some(BTreeSet<String>),
174    /// Anything about the tile may have changed.
175    Every,
176}
177impl TileComponent for RefreshSources {}
178
179/// Like [`refresh_tiles`], for a change to the data of `sources` alone: a request made for
180/// the refresh processes those sources and keeps what the tile shows of the others.
181pub(crate) fn refresh_tiles_for_sources(
182    world: &mut World,
183    kind: RequestKind,
184    tiles: &[WorldTileCoords],
185    sources: &[String],
186) {
187    for coords in tiles {
188        mark_refresh_sources(world, *coords, Some(sources));
189    }
190    refresh_due(world, kind, tiles);
191}
192
193/// Notes which sources a pending refresh of the tile is for: those given add to the ones
194/// noted before, and a refresh for every source (`None`) outranks any for some, whichever
195/// comes first.
196fn mark_refresh_sources(world: &mut World, coords: WorldTileCoords, sources: Option<&[String]>) {
197    let marked = match sources {
198        None => RefreshSources::Every,
199        Some(sources) => RefreshSources::Some(sources.iter().cloned().collect()),
200    };
201    if let Some(existing) = world.tiles.query_mut::<&mut RefreshSources>(coords) {
202        match (existing, marked) {
203            (RefreshSources::Every, _) => {}
204            (RefreshSources::Some(scope), RefreshSources::Some(sources)) => scope.extend(sources),
205            (existing, marked) => *existing = marked,
206        }
207        return;
208    }
209    if let Some(mut tile) = world.tiles.spawn_mut(coords) {
210        tile.insert(marked);
211    }
212}

214/// The sources the pending refresh of the tile is for, cleared for the request that takes
215/// them; `None` when the refresh is for every source, or none is pending.
216pub(crate) fn take_refresh_sources(
217    world: &mut World,
218    coords: WorldTileCoords,
219) -> Option<Vec<String>> {
220    let pending = world.tiles.query_mut::<&mut RefreshSources>(coords)?;
221    match std::mem::take(pending) {
222        RefreshSources::Some(sources) => Some(sources.into_iter().collect()),
223        RefreshSources::None | RefreshSources::Every => None,
224    }
225}

227/// Requests `tiles` again for `kind`, keeping what they show until the new data lands.
228pub(crate) fn refresh_tiles(world: &mut World, kind: RequestKind, tiles: &[WorldTileCoords]) {
229    for coords in tiles {
230        mark_refresh_sources(world, *coords, None);
231    }
232    refresh_due(world, kind, tiles);
233}
234
235/// Makes the next request for each of `tiles` due now, or once the one in flight lands.
236fn refresh_due(world: &mut World, kind: RequestKind, tiles: &[WorldTileCoords]) {
237    crate::render::frame_signals::mark_dirty(world);
238    let now = now(world);
239    for coords in tiles {
240        let Some(retries) = world.tiles.query_mut::<&mut TileRequestRetries>(*coords) else {
241            continue;
242        };
243        let state = &mut retries.0[kind.index()];
244        if state.attempt.is_none() {
245            continue;
246        }
247        if state.pending {
248            state.stale = true;
249        } else if state.deadline.is_none() {
250            // A tile already backing off retries at its deadline with the new data.
251            state.delay = Duration::ZERO;
252            state.deadline = Some(now);
253        }
254    }
255}

257/// Requests a finished tile again after `delay`, because part of what its worker sent later
258/// was unavailable for now.
259pub(crate) fn retry_later(
260    world: &mut World,
261    coords: WorldTileCoords,
262    kind: RequestKind,
263    delay: Duration,
264) {
265    let now = now(world);
266    let Some(retries) = world.tiles.query_mut::<&mut TileRequestRetries>(coords) else {
267        return;
268    };
269    let state = &mut retries.0[kind.index()];
270    if state.pending || state.deadline.is_some() {
271        return;
272    }
273    state.deadline = Some(now.saturating_add(delay));
274    // The retry brings what the whole tile was asked for, not what a later data change was.
275    mark_refresh_sources(world, coords, None);
276}

278pub(crate) fn due(world: &mut World, coords: WorldTileCoords, kind: RequestKind) -> bool {
279    let now = now(world);
280    world
281        .tiles
282        .query::<&TileRequestRetries>(coords)
283        .is_some_and(|retries| {
284            let state = &retries.0[kind.index()];
285            !state.pending && state.deadline.is_some_and(|deadline| now >= deadline)
286        })
287}

299pub(crate) fn next_attempt(world: &mut World) -> u64 {
300    let attempts = world.resources.get_or_init_mut::<RequestAttempts>();
301    let next = attempts.0.get().wrapping_add(1);
302    attempts.0.set(next);
303    next
304}

306/// Records the attempt made for the tile, limited to `only_sources` or for every source.
307pub(crate) fn started(
308    world: &mut World,
309    coords: WorldTileCoords,
310    kind: RequestKind,
311    attempt: u64,
312    only_sources: Option<&[String]>,
313) {
314    if world.tiles.query::<&TileRequestRetries>(coords).is_none() {
315        if let Some(mut tile) = world.tiles.spawn_mut(coords) {
316            tile.insert(TileRequestRetries::default());
317        }
318    }
319    if let Some(retries) = world.tiles.query_mut::<&mut TileRequestRetries>(coords) {
320        let state = &mut retries.0[kind.index()];
321        state.attempt = Some(attempt);
322        state.pending = true;
323        state.deadline = None;
324        state.stale = false;
325        state.issued_for = only_sources.map(<[String]>::to_vec);
326    }
327}

329pub(crate) fn completed(world: &mut World, outcome: TileRequestOutcome) {
330    let now = now(world);
331    let Some(retries) = world
332        .tiles
333        .query_mut::<&mut TileRequestRetries>(outcome.coords)
334    else {
335        return;
336    };
337    let state = &mut retries.0[outcome.kind.index()];
338    // An evicted request can finish after the same coordinates have been requested again.
339    if !state.pending || state.attempt != outcome.attempt {
340        return;
341    }
342    state.pending = false;
343    let stale = std::mem::take(&mut state.stale);
344    let loaded = matches!(outcome.disposition, RequestDisposition::Complete);
345    let mut retried_for = None;
346    match outcome.disposition {
347        RequestDisposition::Complete => {
348            state.delay = Duration::ZERO;
349            state.deadline = stale.then_some(now);
350        }
351        RequestDisposition::Retry => {
352            state.delay = if state.delay.is_zero() {
353                Duration::from_secs(1)
354            } else {
355                state.delay.saturating_mul(2).min(Duration::from_secs(30))
356            };
357            state.deadline = Some(now.saturating_add(state.delay));
358            retried_for = Some(state.issued_for.clone());
359        }
360    }
361    // The retry is for what the attempt was for, so a data change noted since cannot
362    // narrow it to its own sources and lose the ones that failed.
363    if let Some(issued_for) = retried_for {
364        mark_refresh_sources(world, outcome.coords, issued_for.as_deref());
365    }
366    crate::render::frame_signals::resource_ready(
367        world,
368        crate::render::frame_signals::ResourceReady::Tile {
369            coords: outcome.coords,
370            kind: outcome.kind,
371            loaded,
372        },
373    );
374}

376/// Payloads from a finished attempt remain valid until a new attempt replaces its identity.
377pub(crate) fn accepts(
378    world: &World,
379    coords: WorldTileCoords,
380    kind: RequestKind,
381    attempt: Option<u64>,
382) -> bool {
383    world
384        .tiles
385        .query::<&TileRequestRetries>(coords)
386        .and_then(|retries| retries.0[kind.index()].attempt)
387        == attempt
388}

390/// The tiles each request family considered this frame; a family that did not run leaves
391/// its requests alone.
392#[derive(Default)]
393struct WantedRequests([Option<HashSet<WorldTileCoords>>; 3]);
394
395/// Attempts given up on, for the next request system to stop in its worker.
396#[derive(Default)]
397struct CancelledAttempts(Vec<u64>);
398
399/// Records the tiles a request family still wants, whether or not it requested them now.
400pub(crate) fn want(world: &mut World, kind: RequestKind, coords: &HashSet<WorldTileCoords>) {
401    world.resources.get_or_init_mut::<WantedRequests>().0[kind.index()] = Some(coords.clone());
402}

404/// Gives up the requests in flight that no request family wanted this frame, so they stop
405/// holding a slot, and returns the tiles they were for. Tiles in `in_use` keep theirs: the
406/// drape requests a family follows lag a frame behind what the view uses.
407pub(crate) fn cancel_unwanted(
408    world: &mut World,
409    in_use: &HashSet<WorldTileCoords>,
410) -> Vec<WorldTileCoords> {
411    let Some(wanted) = world
412        .resources
413        .get_mut::<WantedRequests>()
414        .map(std::mem::take)
415    else {
416        return Vec::new();
417    };
418    let coords: Vec<WorldTileCoords> = world
419        .tiles
420        .tiles
421        .values()
422        .map(|tile| tile.coords)
423        .filter(|coords| !in_use.contains(coords))
424        .collect();
425    let mut cancelled = Vec::new();
426    let mut tiles = Vec::new();
427    for coords in coords {
428        let Some(retries) = world.tiles.query_mut::<&mut TileRequestRetries>(coords) else {
429            continue;
430        };
431        for (state, wanted) in retries.0.iter_mut().zip(&wanted.0) {
432            let unwanted = wanted.as_ref().is_some_and(|set| !set.contains(&coords));
433            if state.pending && unwanted {
434                cancelled.extend(state.attempt);
435                *state = RetryState::default();
436                if !tiles.contains(&coords) {
437                    tiles.push(coords);
438                }
439            }
440        }
441    }
442    world
443        .resources
444        .get_or_init_mut::<CancelledAttempts>()
445        .0
446        .extend(cancelled);
447    tiles
448}

450/// Stops the worker calls of the attempts cancelled since the last call.
451pub(crate) fn stop_cancelled<K: crate::environment::OffscreenKernel>(
452    world: &mut World,
453    apc: &impl crate::io::apc::AsyncProcedureCall<K>,
454) {
455    let cancelled = world
456        .resources
457        .get_mut::<CancelledAttempts>()
458        .map(|cancelled| std::mem::take(&mut cancelled.0))
459        .unwrap_or_default();
460    for attempt in cancelled {
461        apc.cancel(attempt);
462    }
463}
