10/// Tile coordinates that cannot be addressed by any tile URL.
11#[derive(Debug, Error)]
12#[error("tile coordinates {coords} are outside the addressable range")]
13pub struct InvalidTileCoords {
14    /// Coordinates that failed to convert into an addressing scheme.
15    pub coords: WorldTileCoords,
16}
17
18/// Half the Web Mercator world's width, metres: the EPSG:3857 extent.
19const MERCATOR_HALF_EXTENT_M: f64 = 20_037_508.342_789_244;
20
21/// The tile's `minx,miny,maxx,maxy` in EPSG:3857 metres, as MapLibre's
22/// `{bbox-epsg-3857}` placeholder states it for WMS-style servers. The box
23/// is the tile's ground, so it does not depend on the row scheme.
24fn bbox_epsg_3857(coords: &WorldTileCoords) -> Option<String> {
25    let tile = coords.into_tile(TileAddressingScheme::XYZ)?;
26    let size = 2.0 * MERCATOR_HALF_EXTENT_M / f64::from(1_u32 << u8::from(tile.z));
27    let min_x = f64::from(tile.x) * size - MERCATOR_HALF_EXTENT_M;
28    let max_y = MERCATOR_HALF_EXTENT_M - f64::from(tile.y) * size;
29    Some(format!("{min_x},{},{},{max_y}", max_y - size, min_x + size))
30}
31
32fn format_template(
33    template: &str,
34    scheme: TileAddressingScheme,
35    coords: &WorldTileCoords,
36) -> Option<String> {
37    let tile = coords.into_tile(scheme)?;
38    let mut url = template
39        .replace("{z}", &u8::from(tile.z).to_string())
40        .replace("{x}", &tile.x.to_string())
41        .replace("{y}", &tile.y.to_string());
42    if url.contains("{bbox-epsg-3857}") {
43        url = url.replace("{bbox-epsg-3857}", &bbox_epsg_3857(coords)?);
44    }
45    Some(url)
46}

48/// Represents a source from which vector tiles are fetched.
49#[derive(Clone, Debug)]
50pub struct TessellateSource {
51    template: String,
52    scheme: TileAddressingScheme,
53}
54
55impl TessellateSource {
56    /// Creates a source addressed as `{url}/{z}/{x}/{y}.{filetype}`.
57    pub fn new(url: &str, filetype: &str) -> Self {
58        Self::from_template(
59            format!("{url}/{{z}}/{{x}}/{{y}}.{filetype}"),
60            TileAddressingScheme::XYZ,
61        )
62    }
63
64    /// Creates a source from a style tile URL template containing `{z}`, `{x}` and `{y}`, or
65    /// `{bbox-epsg-3857}`.
66    pub fn from_template(template: impl Into<String>, scheme: TileAddressingScheme) -> Self {
67        Self {
68            template: template.into(),
69            scheme,
70        }
71    }
72
73    /// Returns the URL template with unexpanded placeholders.
74    pub fn template(&self) -> &str {
75        &self.template
76    }
77
78    /// Returns the tile URL, or `None` when the coordinates are outside the world.
79    pub fn format(&self, coords: &WorldTileCoords) -> Option<String> {
80        format_template(&self.template, self.scheme, coords)
81    }
82}
83
84impl Default for TessellateSource {
85    fn default() -> Self {
86        Self::new("https://maps.tuerantuer.org/europe_germany", "pbf")
87    }
88}

152/// Represents the tiles' different types of source.
153#[derive(Clone, Debug)]
154pub enum SourceType {
155    /// GeoJSON document tiled by the worker instead of fetched per tile. Boxed because a
156    /// declaration holding its data is many times the size of the other variants.
157    GeoJson(Box<GeoJsonTileSource>),
158    /// Raster image tile URL template.
159    Raster(RasterSource),
160    /// One picture resampled into raster tiles.
161    Image(ImageTileSource),
162    /// Vector tile URL template for geometry processing.
163    Tessellate(TessellateSource),
164}
165
166impl SourceType {
167    /// Returns the tile URL, or `None` when the coordinates are outside the world.
168    pub fn format(&self, coords: &WorldTileCoords) -> Option<String> {
169        match self {
170            SourceType::GeoJson(_) | SourceType::Image(_) => None,
171            SourceType::Raster(raster_source) => raster_source.format(coords),
172            SourceType::Tessellate(tessellate_source) => tessellate_source.format(coords),
173        }
174    }
175}
