1//! Tiles read out of a PMTiles archive with HTTP range requests.
2//!
3//! As with GL JS's `pmtiles` protocol, a source names an archive as `pmtiles://<archive URL>`.
4//! That URL answers with a TileJSON document built from the archive's header and metadata,
5//! whose tile template `pmtiles://<archive URL>/{z}/{x}/{y}` then answers with each tile. The
6//! header and directories are read once per archive and kept; a tile costs one range request.
7
8use std::{
9    collections::HashMap,
10    sync::{Arc, Mutex, PoisonError},
11};

26const SCHEME: &str = "pmtiles://";
27/// Root, leaf and leaf-of-leaf: the deepest the format nests directories.
28const MAX_DEPTH: usize = 3;
29
30/// What a `pmtiles://` URL asks for.
31#[derive(Clone, Debug, PartialEq, Eq)]
32pub enum PmtilesUrl {
33    /// The archive's TileJSON.
34    Archive(String),
35    /// One tile of the archive.
36    Tile {
37        /// The archive's own URL.
38        archive: String,
39        /// Zoom.
40        z: u8,
41        /// Column.
42        x: u32,
43        /// Row.
44        y: u32,
45    },
46}
47
48impl PmtilesUrl {
49    /// Reads a `pmtiles://` URL; any other URL is `None`.
50    pub fn parse(url: &str) -> Option<Self> {
51        let rest = url.strip_prefix(SCHEME)?;
52        let mut parts = rest.rsplitn(4, '/');
53        let tile = (|| {
54            let y = parts.next()?.parse().ok()?;
55            let x = parts.next()?.parse().ok()?;
56            let z = parts.next()?.parse().ok()?;
57            Some((parts.next()?.to_owned(), z, x, y))
58        })();
59        Some(match tile {
60            Some((archive, z, x, y)) => Self::Tile { archive, z, x, y },
61            None => Self::Archive(rest.to_owned()),
62        })
63    }
64}

66/// An archive could not be read.
67#[derive(Error, Debug)]
68#[error("cannot read PMTiles archive {archive}")]
69pub struct PmtilesError {
70    /// The archive's URL.
71    pub archive: String,
72    /// Why.
73    #[source]
74    pub source: PmtilesFormatError,
75}
76
77fn failed(archive: &str, source: PmtilesFormatError) -> SourceFetchError {
78    SourceFetchError(Box::new(PmtilesError {
79        archive: archive.to_owned(),
80        source,
81    }))
82}

84struct Archive {

90/// Reads tiles out of PMTiles archives, keeping each archive's header and directories.
91#[derive(Clone, Default)]
92pub struct PmtilesReader {
93    archives: Arc<Mutex<HashMap<String, Arc<Archive>>>>,
94}

96async fn range<C: HttpClient>(
97    client: &C,
98    base: &HttpRequest,
99    status: NetworkStatus,
100    url: &str,
101    offset: u64,
102    length: u64,
103) -> Result<Vec<u8>, SourceFetchError> {
104    let mut sub = base.clone();
105    sub.url = url.to_owned();
106    sub.range = Some(ByteRange { offset, length });
107    client.fetch_request_with_status(&sub, status).await
108}

110impl PmtilesReader {
111    /// Answers a `pmtiles://` request through `client`'s range requests.
112    pub async fn fetch<C: HttpClient>(
113        &self,
114        client: &C,
115        url: &str,
116        request: PmtilesUrl,
117    ) -> Result<Vec<u8>, SourceFetchError> {
118        let base = HttpRequest::new(url, ResourceKind::Unknown);
119        self.fetch_with_context(client, &base, NetworkStatus::Online, request)
120            .await
121    }
122
123    /// Answers a `pmtiles://` request forwarding `base`'s headers, [`CachePolicy`](crate::io::source_client::CachePolicy), and `status`.
124    pub async fn fetch_with_context<C: HttpClient>(
125        &self,
126        client: &C,
127        base: &HttpRequest,
128        status: NetworkStatus,
129        request: PmtilesUrl,
130    ) -> Result<Vec<u8>, SourceFetchError> {
131        match request {
132            PmtilesUrl::Archive(archive) => self.tile_json(client, base, status, &archive).await,
133            PmtilesUrl::Tile { archive, z, x, y } => {
134                let id = tile_id(z, x, y).map_err(|error| failed(&archive, error))?;
135                let opened = self.open(client, base, status, &archive).await?;
136                let Some(entry) = self
137                    .locate(client, base, status, &archive, &opened, id)
138                    .await?
139                else {
140                    return Err(SourceFetchError::not_found(&base.url));
141                };
142                let bytes = range(
143                    client,
144                    base,
145                    status,
146                    &archive,
147                    opened.header.tiles_offset + entry.offset,
148                    entry.length,
149                )
150                .await?;
151                decompress(opened.header.tile_compression, bytes, "tile")
152                    .map_err(|error| failed(&archive, error))
153            }
154        }
155    }
156
157    async fn open<C: HttpClient>(
158        &self,
159        client: &C,
160        base: &HttpRequest,
161        status: NetworkStatus,
162        archive: &str,
163    ) -> Result<Arc<Archive>, SourceFetchError> {
164        if !base.cache_policy.is_volatile() {
165            if let Some(opened) = self.cached(archive) {
166                return Ok(opened);
167            }
168        }
169        let bytes = range(client, base, status, archive, 0, HEADER_BYTES).await?;
170        let header = Header::parse(&bytes).map_err(|error| failed(archive, error))?;
171        let root = self
172            .directory(
173                client,
174                base,
175                status,
176                archive,
177                &header,
178                header.root.0,
179                header.root.1,
180            )
181            .await?;
182        let opened = Arc::new(Archive {
183            header,
184            root,
185            leaves: Mutex::default(),
186        });
187        if !base.cache_policy.is_volatile() {
188            self.archives
189                .lock()
190                .unwrap_or_else(PoisonError::into_inner)
191                .insert(archive.to_owned(), opened.clone());
192        }
193        Ok(opened)
194    }
195
196    fn cached(&self, archive: &str) -> Option<Arc<Archive>> {
197        self.archives
198            .lock()
199            .unwrap_or_else(PoisonError::into_inner)
200            .get(archive)
201            .cloned()
202    }
203
204    #[allow(clippy::too_many_arguments)]
205    async fn directory<C: HttpClient>(
206        &self,
207        client: &C,
208        base: &HttpRequest,
209        status: NetworkStatus,
210        archive: &str,
211        header: &Header,
212        offset: u64,
213        length: u64,
214    ) -> Result<Vec<Entry>, SourceFetchError> {
215        let bytes = range(client, base, status, archive, offset, length).await?;
216        decompress(header.internal_compression, bytes, "directory")
217            .and_then(|bytes| parse_directory(&bytes))
218            .map_err(|error| failed(archive, error))
219    }
220
221    async fn locate<C: HttpClient>(
222        &self,
223        client: &C,
224        base: &HttpRequest,
225        status: NetworkStatus,
226        archive: &str,
227        opened: &Archive,
228        id: u64,
229    ) -> Result<Option<Entry>, SourceFetchError> {
230        let mut directory = Arc::new(opened.root.clone());
231        for _ in 0..MAX_DEPTH {
232            match find(&directory, id) {
233                None => return Ok(None),
234                Some(Found::Tile(entry)) => return Ok(Some(entry)),
235                Some(Found::Leaf(leaf)) => {
236                    let offset = opened.header.leaves_offset + leaf.offset;
237                    let known = opened
238                        .leaves
239                        .lock()
240                        .unwrap_or_else(PoisonError::into_inner)
241                        .get(&offset)
242                        .cloned();
243                    directory = match known {
244                        Some(entries) => entries,
245                        None => {
246                            let entries = Arc::new(
247                                self.directory(
248                                    client,
249                                    base,
250                                    status,
251                                    archive,
252                                    &opened.header,
253                                    offset,
254                                    leaf.length,
255                                )
256                                .await?,
257                            );
258                            opened
259                                .leaves
260                                .lock()
261                                .unwrap_or_else(PoisonError::into_inner)
262                                .insert(offset, entries.clone());
263                            entries
264                        }
265                    };
266                }
267            }
268        }
269        Err(failed(archive, PmtilesFormatError::TooDeep))
270    }

272    /// The TileJSON GL JS's protocol builds: zooms, bounds and centre from the header, the
273    /// rest from the archive's metadata.
274    async fn tile_json<C: HttpClient>(
275        &self,
276        client: &C,
277        base: &HttpRequest,
278        status: NetworkStatus,
279        archive: &str,
280    ) -> Result<Vec<u8>, SourceFetchError> {
281        let opened = self.open(client, base, status, archive).await?;
282        let header = &opened.header;
283        let metadata = range(
284            client,
285            base,
286            status,
287            archive,
288            header.metadata.0,
289            header.metadata.1,
290        )
291        .await?;
292        let metadata = decompress(header.internal_compression, metadata, "metadata")
293            .map_err(|error| failed(archive, error))?;
294        let mut document = match serde_json::from_slice::<serde_json::Value>(&metadata) {
295            Ok(serde_json::Value::Object(fields)) => fields,
296            _ => serde_json::Map::new(),
297        };
298        let [west, south, east, north] = header.bounds;
299        for (key, value) in [
300            ("tilejson", serde_json::json!("3.0.0")),
301            (
302                "tiles",
303                serde_json::json!([format!("{SCHEME}{archive}/{{z}}/{{x}}/{{y}}")]),
304            ),
305            ("minzoom", serde_json::json!(header.zooms.0)),
306            ("maxzoom", serde_json::json!(header.zooms.1)),
307            ("bounds", serde_json::json!([west, south, east, north])),
308            (
309                "center",
310                serde_json::json!([header.center.0, header.center.1, header.center.2]),
311            ),
312        ] {
313            document.insert(key.to_owned(), value);
314        }
315        serde_json::to_vec(&document).map_err(|error| SourceFetchError(Box::new(error)))
316    }
317}

319/// A transport that also answers `pmtiles://` URLs, for hosts whose loader is not a
320/// [`SharedLoader`](crate::io::resource_loader::SharedLoader), which reads archives itself.
321#[derive(Clone)]
322pub struct PmtilesClient<HC: HttpClient> {
323    inner: HC,
324    reader: PmtilesReader,
325}
326
327impl<HC: HttpClient> PmtilesClient<HC> {
328    /// Reads archives through `inner`'s range requests and passes every other URL to it.
329    pub fn new(inner: HC) -> Self {
330        Self {
331            inner,
332            reader: PmtilesReader::default(),
333        }
334    }
335}
336
337impl<HC: HttpClient + Default> Default for PmtilesClient<HC> {
338    fn default() -> Self {
339        Self::new(HC::default())
340    }
341}
342
343#[cfg_attr(not(feature = "thread-safe-futures"), async_trait(?Send))]
344#[cfg_attr(feature = "thread-safe-futures", async_trait)]
345impl<HC: HttpClient> HttpClient for PmtilesClient<HC> {
346    async fn fetch(&self, url: &str) -> Result<Vec<u8>, SourceFetchError> {
347        match PmtilesUrl::parse(url) {
348            Some(request) => self.reader.fetch(&self.inner, url, request).await,
349            None => self.inner.fetch(url).await,
350        }
351    }
352
353    async fn fetch_range(&self, url: &str, range: ByteRange) -> Result<Vec<u8>, SourceFetchError> {
354        self.inner.fetch_range(url, range).await
355    }
356
357    async fn fetch_request(&self, req: &HttpRequest) -> Result<Vec<u8>, SourceFetchError> {
358        self.fetch_request_with_status(req, NetworkStatus::Online)
359            .await
360    }
361
362    async fn fetch_request_with_status(
363        &self,
364        req: &HttpRequest,
365        status: NetworkStatus,
366    ) -> Result<Vec<u8>, SourceFetchError> {
367        if req.range.is_none() {
368            if let Some(pmtiles_req) = PmtilesUrl::parse(&req.url) {
369                return self
370                    .reader
371                    .fetch_with_context(&self.inner, req, status, pmtiles_req)
372                    .await;
373            }
374        }
375        self.inner.fetch_request_with_status(req, status).await
376    }
377}
