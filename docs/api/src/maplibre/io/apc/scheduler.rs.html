28#[cfg(target_arch = "wasm32")]
29fn is_intermediate_vector_layer(message: &Message) -> bool {
30    use crate::vector::transferables::TileTessellated as _;
31    matches!(
32        message
33            .tag()
34            .as_any()
35            .downcast_ref::<crate::vector::transferables::VectorMessageTag>(),
36        Some(
37            crate::vector::transferables::VectorMessageTag::LayerMissing
38                | crate::vector::transferables::VectorMessageTag::LayerTessellated
39                | crate::vector::transferables::VectorMessageTag::SymbolLayerTessellated
40                | crate::vector::transferables::VectorMessageTag::LayerIndexed
41        )
42    ) || message.has_tag(crate::sdf::provided::ProvidedImagesReport::message_tag())
43        || message
44            .downcast_ref::<crate::vector::transferables::DefaultTileTessellated>()
45            .is_some_and(|tile| !tile.pending_symbols())
46}

51/// Reply sender whose clones share the scheduler-backed APC's receiving channel.
52#[derive(Clone)]
53pub struct SchedulerContext {
54    pub(crate) sender: Sender<Message>,
55    pub(crate) waker: Option<ArrivalWaker>,
56    #[cfg(target_arch = "wasm32")]
57    pending: Arc<Mutex<Vec<Message>>>,
58}
59
60impl SchedulerContext {
61    pub(crate) fn new(sender: Sender<Message>, waker: Option<ArrivalWaker>) -> Self {
62        Self {
63            sender,
64            waker,
65            #[cfg(target_arch = "wasm32")]
66            pending: Arc::new(Mutex::new(Vec::new())),
67        }
68    }
69
70    #[cfg(target_arch = "wasm32")]
71    pub(super) fn flush(&self) -> Result<(), SendError> {
72        let buffered = std::mem::take(&mut *crate::platform::lock_unpoisoned(&self.pending));
73        if buffered.is_empty() {
74            return Ok(());
75        }
76        for msg in buffered {
77            self.sender
78                .send(msg)
79                .map_err(|source| SendError::Transmission {
80                    operation: "sending a result to the caller",
81                    source: Box::new(source),
82                })?;
83        }
84        if let Some(wake) = &self.waker {
85            wake();
86        }
87        Ok(())
88    }
89}
90
91impl Context for SchedulerContext {
92    fn send_back<T: IntoMessage>(&self, message: T) -> Result<(), SendError> {
93        let message = message.into();
94        #[cfg(target_arch = "wasm32")]
95        {
96            if is_intermediate_vector_layer(&message) {
97                crate::platform::lock_unpoisoned(&self.pending).push(message);
98                return Ok(());
99            }
100            let buffered = std::mem::take(&mut *crate::platform::lock_unpoisoned(&self.pending));
101            for pending_msg in buffered {
102                self.sender
103                    .send(pending_msg)
104                    .map_err(|source| SendError::Transmission {
105                        operation: "sending a result to the caller",
106                        source: Box::new(source),
107                    })?;
108            }
109        }
110        self.sender
111            .send(message)
112            .map_err(|source| SendError::Transmission {
113                operation: "sending a result to the caller",
114                source: Box::new(source),
115            })?;
116        if let Some(wake) = &self.waker {
117            wake();
118        }
119        Ok(())
120    }
121}

123/// An APC using a scheduler for execution and an in-process channel for replies.
124///
125/// Receiving drains available matches in delivery order. Procedure failures are logged;
126/// only preparation and admission failures are returned by [`AsyncProcedureCall::call`].
127pub struct SchedulerAsyncProcedureCall<K: OffscreenKernel, S: Scheduler> {
128    pub(crate) channel: (Sender<Message>, Receiver<Message>),
129    /// Admitted messages retained for sibling consumers in the current frame.
130    buffer: RefCell<Vec<Message>>,
131    /// Unadmitted arrivals peeked from `channel.1` or deferred to a subsequent frame.
132    queued: RefCell<VecDeque<Message>>,
133    /// Whether `receive` may admit new messages from `channel` during the current frame.
134    batch_open: Cell<bool>,
135    /// When `true` (initial boot), `receive` drains all ready worker results without batch caps.
136    #[cfg_attr(not(target_arch = "wasm32"), allow(dead_code))]
137    boot_unthrottled: Cell<bool>,
138    scheduler: S,
139    phantom_k: PhantomData<K>,
140    offscreen_kernel_config: OffscreenKernelConfig,
141    /// Tracked calls still running, by attempt, so a cancelled one can be stopped.
142    running: Arc<Mutex<HashMap<u64, AbortHandle>>>,
143    pub(crate) waker: SharedArrivalWaker,
144}

146impl<K: OffscreenKernel, S: Scheduler> SchedulerAsyncProcedureCall<K, S> {
147    /// Creates an empty reply channel and keeps the worker kernel configuration for future calls.
148    pub fn new(scheduler: S, offscreen_kernel_config: OffscreenKernelConfig) -> Self {
149        Self {
150            channel: mpsc::channel(),
151            buffer: RefCell::new(Vec::new()),
152            queued: RefCell::new(VecDeque::new()),
153            batch_open: Cell::new(true),
154            boot_unthrottled: Cell::new(true),
155            phantom_k: PhantomData,
156            scheduler,
157            offscreen_kernel_config,
158            running: Arc::default(),
159            waker: Arc::default(),
160        }
161    }
162
163    pub(super) fn running(&self) -> std::sync::MutexGuard<'_, HashMap<u64, AbortHandle>> {
164        crate::platform::lock_unpoisoned(&self.running)
165    }
166}

168impl<K: OffscreenKernel, S: Scheduler> AsyncProcedureCall<K> for SchedulerAsyncProcedureCall<K, S> {
169    type Context = SchedulerContext;
170    type ReceiveIterator<F: FnMut(&Message) -> bool> = IntoIter<Message>;
171
172    fn receive<F: FnMut(&Message) -> bool>(&self, mut filter: F) -> Self::ReceiveIterator<F> {
173        let mut buffer = self.buffer.borrow_mut();
174        // Partial and final tile completions must retain the worker's delivery order.
175        let mut ret: Vec<_> = buffer.extract_if(.., |message| filter(message)).collect();
176
177        if !self.batch_open.get() {
178            return ret.into_iter();
179        }
180
181        #[cfg(target_arch = "wasm32")]
182        {
183            if self.boot_unthrottled.get() {
184                let mut queued = self.queued.borrow_mut();
185                while let Some(message) = queued.pop_front() {
186                    if filter(&message) {
187                        ret.push(message);
188                    } else {
189                        buffer.push(message);
190                    }
191                }
192                while let Ok(message) = self.channel.1.try_recv() {
193                    if filter(&message) {
194                        ret.push(message);
195                    } else {
196                        buffer.push(message);
197                    }
198                }
199                return ret.into_iter();
200            }
201            let mut queued = self.queued.borrow_mut();
202            if queued.is_empty() {
203                if let Ok(message) = self.channel.1.try_recv() {
204                    queued.push_back(message);
205                } else {
206                    return ret.into_iter();
207                }
208            }
209            while let Ok(message) = self.channel.1.try_recv() {
210                queued.push_back(message);
211            }
212            let mut admitted_attempts = rustc_hash::FxHashSet::with_capacity_and_hasher(
213                MAX_ADMITTED_ATTEMPTS_PER_BATCH,
214                Default::default(),
215            );
216            for existing in buffer.iter().chain(ret.iter()) {
217                if let Some(id) = existing.attempt() {
218                    admitted_attempts.insert(id);
219                }
220            }
221            let mut deferred = VecDeque::new();
222            while let Some(message) = queued.pop_front() {
223                let attempt = message.attempt();
224                let admit = attempt.is_none_or(|id| {
225                    admitted_attempts.contains(&id)
226                        || admitted_attempts.len() < MAX_ADMITTED_ATTEMPTS_PER_BATCH
227                });
228                if !admit {
229                    deferred.push_back(message);
230                    continue;
231                }
232                if let Some(id) = attempt {
233                    admitted_attempts.insert(id);
234                }
235                if filter(&message) {
236                    ret.push(message);
237                } else {
238                    buffer.push(message);
239                }
240            }
241            if !deferred.is_empty() {
242                self.batch_open.set(false);
243            }
244            *queued = deferred;
245        }
246
247        #[cfg(not(target_arch = "wasm32"))]
248        {
249            let mut queued = self.queued.borrow_mut();
250            while let Some(message) = queued.pop_front() {
251                if filter(&message) {
252                    ret.push(message);
253                } else {
254                    buffer.push(message);
255                }
256            }
257            while let Ok(message) = self.channel.1.try_recv() {
258                tracing::debug!("Data reached main thread: {message:?}");
259                if filter(&message) {
260                    ret.push(message);
261                } else {
262                    buffer.push(message);
263                }
264            }
265        }
266
267        ret.into_iter()
268    }

270    fn call(
271        &self,
272        input: Input,
273        procedure: AsyncProcedure<K, Self::Context>,
274    ) -> Result<(), CallError> {
275        let sender = self.channel.0.clone();
276        let waker: Option<ArrivalWaker> =
277            if let Some(wake) = crate::platform::lock_unpoisoned(&self.waker).clone() {
278                Some(wake)
279            } else {
280                let waker_slot = self.waker.clone();
281                Some(Arc::new(move || {
282                    let current = crate::platform::lock_unpoisoned(&waker_slot).clone();
283                    if let Some(wake) = current {
284                        wake();
285                    }
286                }))
287            };
288        let offscreen_kernel_config = self.offscreen_kernel_config.clone();
289        let attempt = input.attempt;
290        let (abort, registration) = AbortHandle::new_pair();
291        self.running().insert(attempt, abort);
292        let running = self.running.clone();
293        let finish_waker = waker.clone();
294
295        let scheduled = self.scheduler.schedule(move || async move {
296            tracing::debug!(thread = ?std::thread::current().name(), "processing worker call");
297
298            let kernel = K::create(offscreen_kernel_config);
299            let context = SchedulerContext::new(sender, waker);
300            #[cfg(target_arch = "wasm32")]
301            let flush_context = context.clone();
302            // A result that cannot be delivered, because the map already shut down, must
303            // not take the worker thread down with it.
304            let call = procedure(input, context, kernel);
305            match Abortable::new(call, registration).await {
306                Ok(Err(ProcedureError::Send(_))) => {
307                    tracing::debug!(attempt, "discarding worker result after map shutdown")
308                }
309                Ok(Err(error)) => tracing::warn!(?error, "procedure failed"),
310                Ok(Ok(())) => {
311                    #[cfg(target_arch = "wasm32")]
312                    let _ = flush_context.flush();
313                }
314                Err(Aborted) => tracing::debug!(attempt, "cancelled worker call stopped"),
315            }
316            let remaining = {
317                let mut map = crate::platform::lock_unpoisoned(&running);
318                map.remove(&attempt);
319                map.len()
320            };
321            if remaining == 0 {
322                if let Some(wake) = finish_waker {
323                    wake();
324                }
325            }
326        });
327        if scheduled.is_err() {
328            // A rejected factory never runs, so nothing else removes its entry.
329            self.running().remove(&attempt);
330        }
331        scheduled.map_err(CallError::Schedule)
332    }

334    fn has_arrivals(&self) -> bool {
335        self.batch_open.set(true);
336        if self
337            .buffer
338            .try_borrow()
339            .is_ok_and(|buffer| !buffer.is_empty())
340        {
341            return true;
342        }
343        let mut queued = self.queued.borrow_mut();
344        if !queued.is_empty() {
345            return true;
346        }
347        if let Ok(message) = self.channel.1.try_recv() {
348            queued.push_back(message);
349            return true;
350        }
351        false
352    }
353
354    fn cancel(&self, attempt: u64) {
355        if let Some(abort) = self.running().remove(&attempt) {
356            abort.abort();
357        }
358    }

364    fn prewarm_call(&self, tag: super::ProcedureTag, input: Input) -> Result<(), CallError> {
365        match tag {
366            super::ProcedureTag::FetchVector => self.call(
367                input,
368                crate::vector::fetch_vector_apc::<
369                    K,
370                    crate::vector::DefaultVectorTransferables,
371                    Self::Context,
372                >,
373            ),
374            #[cfg(feature = "raster")]
375            super::ProcedureTag::FetchRaster => self.call(
376                input,
377                crate::raster::fetch_raster_apc::<
378                    K,
379                    crate::raster::DefaultRasterTransferables,
380                    Self::Context,
381                >,
382            ),
383            #[cfg(not(feature = "raster"))]
384            super::ProcedureTag::FetchRaster => Ok(()),
385            super::ProcedureTag::FetchDem => self.call(
386                input,
387                crate::terrain::fetch_dem_apc::<
388                    K,
389                    crate::terrain::DefaultDemTransferables,
390                    Self::Context,
391                >,
392            ),
393        }
394    }

404    fn set_waker(&self, wake: Arc<dyn Fn() + Send + Sync>) {
405        self.scheduler.set_main_thread_waker(wake.clone());
406        let wrapped = self.scheduler.wrap_worker_waker(wake);
407        *crate::platform::lock_unpoisoned(&self.waker) = Some(wrapped);
