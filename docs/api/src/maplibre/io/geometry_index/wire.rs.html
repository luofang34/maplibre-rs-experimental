1//! Struct-of-arrays wire encoding of [`TileIndex`] for single-threaded worker transport.
2//!
3//! Multipart features without an explicit feature `id` share an `Arc<FeatureProperties>` pointer
4//! across their geometry parts so `query_rendered_features` deduplicates them as one feature.
5//! [`WireTileIndex`] deduplicates property maps by `Arc` pointer into `prop_offsets` /
6//! `prop_blobs` and interns `source_layer` strings into `strings`, preserving `Arc` sharing across
7//! FlatBuffer serialization boundaries with a constant number of vector allocations per tile.
8
9#![deny(missing_docs)]
10

22const KIND_POLYGON: u8 = 2;

24const TAG_NULL: u8 = 0;

33/// Packed descriptor for one geometry part in [`WireTileIndex`].
34#[derive(Debug, Clone, Copy, PartialEq, Eq)]
35pub struct WireGeometry {
36    /// Feature identifier when `has_id` is true.
37    pub id: u64,
38    /// Index into [`WireTileIndex::strings`] naming the source layer.
39    pub source_layer_idx: u32,
40    /// Index into [`WireTileIndex::prop_offsets`] naming the shared property map.
41    pub prop_group_idx: u32,
42    /// Feature position in its source layer for top-down tie breaking.
43    pub feature_index: u32,
44    /// Start offset in [`WireTileIndex::ring_ends`] (used by polygons).
45    pub ring_start: u32,
46    /// Number of rings in [`WireTileIndex::ring_ends`] (`0` for points and line strings).
47    pub ring_count: u32,
48    /// Start coordinate-pair index in [`WireTileIndex::xy`] (`xy[2 * coord_start ..]`).
49    pub coord_start: u32,
50    /// Number of `(x, y)` coordinate pairs belonging to this geometry.
51    pub coord_count: u32,
52    /// Geometry kind tag (`0` = point, `1` = line string, `2` = polygon).
53    pub kind: u8,
54    /// Whether `id` carries an explicit feature id.
55    pub has_id: bool,
56}

58/// Flat Struct-of-Arrays wire representation of a [`TileIndex`] for worker transport.
59#[derive(Debug, Clone, Default, PartialEq)]
60pub struct WireTileIndex {
61    /// Whether the index uses an R-tree (`true`) or sequential list (`false`).
62    pub is_spatial: bool,
63    /// Interned `source_layer` names referenced by [`WireGeometry::source_layer_idx`].
64    pub strings: Vec<String>,
65    /// Concatenated binary property payloads for unique `Arc<FeatureProperties>` groups.
66    pub prop_blobs: Vec<u8>,
67    /// Byte offsets into `prop_blobs` (`len == group_count + 1`, starting at `0`).
68    pub prop_offsets: Vec<u32>,
69    /// Packed descriptors for every geometry part.
70    pub geometries: Vec<WireGeometry>,
71    /// Cumulative ring end coordinate counts (relative to `coord_start`) for polygons.
72    pub ring_ends: Vec<u32>,
73    /// Interleaved `[x0, y0, x1, y1, ...]` tile-local coordinates.
74    pub xy: Vec<f64>,
75}
76
77impl WireTileIndex {
78    /// Packs `index` into contiguous flat arrays while preserving shared `Arc<FeatureProperties>`
79    /// groups and interned `source_layer` names.
80    pub fn from_tile_index(index: &TileIndex) -> Self {
81        let count = match index {
82            TileIndex::Spatial { tree } => tree.size(),
83            TileIndex::Linear { list } => list.len(),
84        };
85        let mut wire = Self {
86            is_spatial: matches!(index, TileIndex::Spatial { .. }),
87            strings: Vec::new(),
88            prop_blobs: Vec::new(),
89            prop_offsets: vec![0],
90            geometries: Vec::with_capacity(count),
91            ring_ends: Vec::new(),
92            xy: Vec::with_capacity(count * 4),
93        };
94        let mut layer_map: FxHashMap<Arc<str>, u32> = FxHashMap::default();
95        let mut prop_map: FxHashMap<usize, u32> = FxHashMap::default();
96
97        let mut push = |geom: &IndexedGeometry<f64>| {
98            let source_layer_idx =
99                *layer_map
100                    .entry(geom.source_layer.clone())
101                    .or_insert_with(|| {
102                        let idx = wire.strings.len() as u32;
103                        wire.strings.push(geom.source_layer.as_ref().to_owned());
104                        idx
105                    });
106            let prop_ptr = Arc::as_ptr(&geom.properties) as usize;
107            let prop_group_idx = *prop_map.entry(prop_ptr).or_insert_with(|| {
108                let idx = (wire.prop_offsets.len() - 1) as u32;
109                encode_properties(&geom.properties, &mut wire.prop_blobs);
110                wire.prop_offsets.push(wire.prop_blobs.len() as u32);
111                idx
112            });
113
114            let coord_start = (wire.xy.len() / 2) as u32;
115            let ring_start = wire.ring_ends.len() as u32;
116            let (kind, ring_count, coord_count) = match &geom.exact {
117                ExactGeometry::Point(pt) => {
118                    wire.xy.push(pt.x());
119                    wire.xy.push(pt.y());
120                    (KIND_POINT, 0, 1)
121                }
122                ExactGeometry::LineString(line) => {
123                    for c in &line.0 {
124                        wire.xy.push(c.x);
125                        wire.xy.push(c.y);
126                    }
127                    (KIND_LINESTRING, 0, line.0.len() as u32)
128                }
129                ExactGeometry::Polygon(poly) => {
130                    let mut rel_coords = 0_u32;
131                    for c in &poly.exterior().0 {
132                        wire.xy.push(c.x);
133                        wire.xy.push(c.y);
134                        rel_coords += 1;
135                    }
136                    wire.ring_ends.push(rel_coords);
137                    for hole in poly.interiors() {
138                        for c in &hole.0 {
139                            wire.xy.push(c.x);
140                            wire.xy.push(c.y);
141                            rel_coords += 1;
142                        }
143                        wire.ring_ends.push(rel_coords);
144                    }
145                    (KIND_POLYGON, 1 + poly.interiors().len() as u32, rel_coords)
146                }
147            };
148
149            wire.geometries.push(WireGeometry {
150                id: geom.id.unwrap_or(0),
151                source_layer_idx,
152                prop_group_idx,
153                feature_index: geom.feature_index,
154                ring_start,
155                ring_count,
156                coord_start,
157                coord_count,
158                kind,
159                has_id: geom.id.is_some(),
160            });
161        };
162
163        match index {
164            TileIndex::Spatial { tree } => tree.iter().for_each(&mut push),
165            TileIndex::Linear { list } => list.iter().for_each(&mut push),
166        }
167        wire
168    }

170    /// Reconstructs a [`TileIndex`] from flat wire arrays, bulk-loading an [`RTree`] when
171    /// `is_spatial` is true.
172    pub fn into_tile_index(self) -> TileIndex {
173        let layers: Vec<Arc<str>> = self.strings.into_iter().map(Arc::from).collect();
174        let mut props: Vec<Arc<FeatureProperties>> =
175            Vec::with_capacity(self.prop_offsets.len().saturating_sub(1));
176        for pair in self.prop_offsets.windows(2) {
177            let start = pair[0] as usize;
178            let end = pair[1] as usize;
179            let decoded = self
180                .prop_blobs
181                .get(start..end)
182                .and_then(decode_properties)
183                .unwrap_or_default();
184            props.push(Arc::new(decoded));
185        }
186
187        let total_coords = self.xy.len() / 2;
188        let mut geometries = Vec::with_capacity(self.geometries.len());
189        for g in self.geometries {
190            let Some(source_layer) = layers.get(g.source_layer_idx as usize).cloned() else {
191                continue;
192            };
193            let Some(properties) = props.get(g.prop_group_idx as usize).cloned() else {
194                continue;
195            };
196            let coord_start = g.coord_start as usize;
197            let coord_end = coord_start.saturating_add(g.coord_count as usize);
198            if coord_end > total_coords {
199                continue;
200            }
201            let meta = FeatureMeta {
202                properties,
203                source_layer,
204                id: g.has_id.then_some(g.id),
205                feature_index: g.feature_index,
206            };
207            let indexed = match g.kind {
208                KIND_POINT if g.coord_count == 1 => {
209                    let pt = Point::new(self.xy[2 * coord_start], self.xy[2 * coord_start + 1]);
210                    IndexedGeometry::from_point(pt, meta)
211                }
212                KIND_LINESTRING => {
213                    let line = slice_linestring(&self.xy, coord_start, coord_end);
214                    IndexedGeometry::from_linestring(line, meta)
215                }
216                KIND_POLYGON if g.ring_count >= 1 => {
217                    let ring_start = g.ring_start as usize;
218                    let ring_end = ring_start.saturating_add(g.ring_count as usize);
219                    let Some(ends) = self.ring_ends.get(ring_start..ring_end) else {
220                        continue;
221                    };
222                    let mut prev = 0_usize;
223                    let mut rings = Vec::with_capacity(ends.len());
224                    let mut valid = true;
225                    for &rel_end in ends {
226                        let rel_end = rel_end as usize;
227                        if rel_end < prev || coord_start + rel_end > coord_end {
228                            valid = false;
229                            break;
230                        }
231                        rings.push(slice_linestring(
232                            &self.xy,
233                            coord_start + prev,
234                            coord_start + rel_end,
235                        ));
236                        prev = rel_end;
237                    }
238                    if !valid || rings.is_empty() {
239                        continue;
240                    }
241                    let mut iter = rings.into_iter();
242                    let exterior = iter.next().unwrap();
243                    let interiors: Vec<LineString<f64>> = iter.collect();
244                    IndexedGeometry::from_polygon(Polygon::new(exterior, interiors), meta)
245                }
246                _ => None,
247            };
248            if let Some(indexed) = indexed {
249                geometries.push(indexed);
250            }
251        }
252
253        if self.is_spatial {
254            TileIndex::Spatial {
255                tree: RTree::bulk_load(geometries),
256            }
257        } else {
258            TileIndex::Linear { list: geometries }
259        }
260    }
261}

263fn slice_linestring(xy: &[f64], start_coord: usize, end_coord: usize) -> LineString<f64> {
264    let mut coords = Vec::with_capacity(end_coord.saturating_sub(start_coord));
265    for i in start_coord..end_coord {
266        coords.push(Coord {
267            x: xy[2 * i],
268            y: xy[2 * i + 1],
269        });
270    }
271    LineString::new(coords)
272}

274pub(crate) fn encode_properties(props: &FeatureProperties, out: &mut Vec<u8>) {
275    out.extend_from_slice(&(props.len() as u32).to_le_bytes());
276    for (key, value) in props {
277        encode_str(key, out);
278        encode_value(value, out);
279    }
280}
281
282fn encode_str(s: &str, out: &mut Vec<u8>) {
283    let bytes = s.as_bytes();
284    out.extend_from_slice(&(bytes.len() as u32).to_le_bytes());
285    out.extend_from_slice(bytes);
286}

288fn encode_value(value: &Value, out: &mut Vec<u8>) {
289    match value {
290        Value::Null => out.push(TAG_NULL),
291        Value::Bool(false) => out.push(TAG_FALSE),
292        Value::Bool(true) => out.push(TAG_TRUE),
293        Value::Number(n) => {
294            out.push(TAG_NUMBER);
295            out.extend_from_slice(&n.to_le_bytes());
296        }
297        Value::String(s) => {
298            out.push(TAG_STRING);
299            encode_str(s, out);
300        }
301        Value::Color(c) => {
302            out.push(TAG_COLOR);
303            for ch in [c.r, c.g, c.b, c.a] {
304                out.extend_from_slice(&ch.to_le_bytes());
305            }
306        }
307        Value::Array(items) => {
308            out.push(TAG_ARRAY);
309            out.extend_from_slice(&(items.len() as u32).to_le_bytes());
310            for item in items {
311                encode_value(item, out);
312            }
313        }
314        Value::Object(map) => {
315            out.push(TAG_OBJECT);
316            out.extend_from_slice(&(map.len() as u32).to_le_bytes());
317            for (k, v) in map {
318                encode_str(k, out);
319                encode_value(v, out);
320            }
321        }
322        Value::Collator(_) | Value::Image(_) | Value::Formatted(_) => {
323            out.push(TAG_NULL);
324        }
325    }
326}

328fn decode_properties(mut buf: &[u8]) -> Option<FeatureProperties> {
329    decode_properties_from(&mut buf)

332pub(crate) fn decode_properties_from(buf: &mut &[u8]) -> Option<FeatureProperties> {
333    let count = read_u32(buf)? as usize;
334    let mut props = FeatureProperties::new();
335    for _ in 0..count {
336        let key = read_string(buf)?;
337        let val = decode_value(buf)?;
338        props.insert(key, val);
339    }
340    Some(props)
341}

343pub(crate) fn read_u8(buf: &mut &[u8]) -> Option<u8> {
344    let (&byte, rest) = buf.split_first()?;
345    *buf = rest;
346    Some(byte)
347}

349pub(crate) fn read_u32(buf: &mut &[u8]) -> Option<u32> {
350    if buf.len() < 4 {
351        return None;
352    }
353    let (head, rest) = buf.split_at(4);
354    *buf = rest;
355    Some(u32::from_le_bytes(head.try_into().ok()?))
356}
357
358fn read_f64(buf: &mut &[u8]) -> Option<f64> {
359    if buf.len() < 8 {
360        return None;
361    }
362    let (head, rest) = buf.split_at(8);
363    *buf = rest;
364    Some(f64::from_le_bytes(head.try_into().ok()?))
365}
366
367fn read_string(buf: &mut &[u8]) -> Option<String> {
368    let len = read_u32(buf)? as usize;
369    if buf.len() < len {
370        return None;
371    }
372    let (head, rest) = buf.split_at(len);
373    *buf = rest;
374    std::str::from_utf8(head).ok().map(str::to_owned)
375}

377fn decode_value(buf: &mut &[u8]) -> Option<Value> {
378    match read_u8(buf)? {
379        TAG_NULL => Some(Value::Null),
380        TAG_FALSE => Some(Value::Bool(false)),
381        TAG_TRUE => Some(Value::Bool(true)),
382        TAG_NUMBER => Some(Value::Number(read_f64(buf)?)),
383        TAG_STRING => Some(Value::String(read_string(buf)?)),
384        TAG_COLOR => Some(Value::Color(Color::new(
385            read_f64(buf)?,
386            read_f64(buf)?,
387            read_f64(buf)?,
388            read_f64(buf)?,
389        ))),
390        TAG_ARRAY => {
391            let len = read_u32(buf)? as usize;
392            let mut items = Vec::with_capacity(len.min(1024));
393            for _ in 0..len {
394                items.push(decode_value(buf)?);
395            }
396            Some(Value::Array(items))
397        }
398        TAG_OBJECT => {
399            let len = read_u32(buf)? as usize;
400            let mut map = BTreeMap::new();
401            for _ in 0..len {
402                let k = read_string(buf)?;
403                let v = decode_value(buf)?;
404                map.insert(k, v);
405            }
406            Some(Value::Object(map))
407        }
408        _ => None,
409    }
