1//! Cutting an indexed geometry down to a tile and its buffer, as GL JS's `geojson-vt` cuts a
2//! GeoJSON feature for each tile it reaches: one axis at a time, each ring on its own.
3//!
4//! A feature indexed whole in every tile it reaches would be found once for each of them, and
5//! an extrusion's roof would stand in a neighbour's index where that tile holds none of it.
6
7use geo_types::{Coord, LineString, Polygon, Rect};
8
9use super::{ExactGeometry, FeatureMeta, IndexedGeometry};
10
11/// One side of the clipping rectangle: the axis it bounds and the range kept on it.
12#[derive(Clone, Copy)]
13struct Slab {
14    axis: usize,
15    min: f64,
16    max: f64,
17}
18
19impl Slab {
20    fn of(coord: Coord<f64>, axis: usize) -> f64 {
21        if axis == 0 {
22            coord.x
23        } else {
24            coord.y
25        }
26    }
27
28    /// Where the segment `a`–`b` crosses `at` on this slab's axis.
29    fn crossing(&self, a: Coord<f64>, b: Coord<f64>, at: f64) -> Coord<f64> {
30        let (from, to) = (Self::of(a, self.axis), Self::of(b, self.axis));
31        let t = (at - from) / (to - from);
32        let point = Coord {
33            x: a.x + (b.x - a.x) * t,
34            y: a.y + (b.y - a.y) * t,
35        };
36        // The crossing lies exactly on the edge, whatever rounding the interpolation does.
37        if self.axis == 0 {
38            Coord { x: at, ..point }
39        } else {
40            Coord { y: at, ..point }
41        }
42    }
43
44    /// A closed ring cut to the slab, with the slab's edges closing what it cut off.
45    fn ring(&self, ring: &[Coord<f64>]) -> Vec<Coord<f64>> {
46        let mut kept = Vec::with_capacity(ring.len());
47        for pair in ring.windows(2) {
48            let (a, b) = (pair[0], pair[1]);
49            let (from, to) = (Self::of(a, self.axis), Self::of(b, self.axis));
50            let inside = |value: f64| (self.min..=self.max).contains(&value);
51            if inside(from) {
52                kept.push(a);
53            }
54            // A segment can leave across one edge and come back across the other.
55            let mut crossings = [self.min, self.max]
56                .into_iter()
57                .filter(|edge| (from - edge) * (to - edge) < 0.0)
58                .collect::<Vec<_>>();
59            if from > to {
60                crossings.reverse();
61            }
62            kept.extend(crossings.into_iter().map(|edge| self.crossing(a, b, edge)));
63        }
64        if let Some(&first) = kept.first() {
65            kept.push(first);
66        }
67        kept
68    }
69
70    /// The runs of a line within the slab.
71    fn line(&self, line: &[Coord<f64>]) -> Vec<Vec<Coord<f64>>> {
72        let inside =
73            |coord: Coord<f64>| (self.min..=self.max).contains(&Self::of(coord, self.axis));
74        let mut runs = Vec::new();
75        let mut run: Vec<Coord<f64>> = Vec::new();
76        for (index, &point) in line.iter().enumerate() {
77            if index > 0 {
78                let a = line[index - 1];
79                let (from, to) = (Self::of(a, self.axis), Self::of(point, self.axis));
80                let mut crossings = [self.min, self.max]
81                    .into_iter()
82                    .filter(|edge| (from - edge) * (to - edge) < 0.0)
83                    .collect::<Vec<_>>();
84                if from > to {
85                    crossings.reverse();
86                }
87                for edge in crossings {
88                    let crossing = self.crossing(a, point, edge);
89                    if run.is_empty() {
90                        run.push(crossing);
91                    } else {
92                        run.push(crossing);
93                        runs.push(std::mem::take(&mut run));
94                    }
95                }
96            }
97            if inside(point) {
98                run.push(point);
99            }
100        }
101        runs.push(run);
102        runs.retain(|run| run.len() >= 2);
103        runs
104    }
105}
106
107fn slabs(area: Rect<f64>) -> [Slab; 2] {
108    let (min, max) = (area.min(), area.max());
109    [
110        Slab {
111            axis: 0,
112            min: min.x,
113            max: max.x,
114        },
115        Slab {
116            axis: 1,
117            min: min.y,
118            max: max.y,
119        },
120    ]
121}

123impl IndexedGeometry<f64> {
124    /// The geometry in the units of a child tile `scale` times finer whose top left corner lies
125    /// at `offset` of the child's units, as GL JS moves an overzoomed tile's features into it.
126    pub fn rescaled(&self, scale: f64, offset: [f64; 2]) -> Self {
127        use geo::MapCoords;
128        let map = |coord: geo_types::Coord<f64>| geo_types::Coord {
129            x: coord.x * scale - offset[0],
130            y: coord.y * scale - offset[1],
131        };
132        let exact = match &self.exact {
133            ExactGeometry::Polygon(polygon) => ExactGeometry::Polygon(polygon.map_coords(map)),
134            ExactGeometry::LineString(line) => ExactGeometry::LineString(line.map_coords(map)),
135            ExactGeometry::Point(point) => ExactGeometry::Point(point.map_coords(map)),
136        };
137        let (lower, upper) = (self.bounds.lower(), self.bounds.upper());
138        let (lower, upper) = (map(lower.0), map(upper.0));
139        Self {
140            bounds: rstar::AABB::from_corners(lower.into(), upper.into()),
141            exact,
142            properties: self.properties.clone(),
143            source_layer: self.source_layer.clone(),
144            id: self.id,
145            feature_index: self.feature_index,
146        }
147    }
148
149    /// The parts of the geometry within `area`, each indexed on its own.
150    pub fn clipped_to(self, area: Rect<f64>) -> Vec<Self> {
151        let meta = FeatureMeta {
152            properties: self.properties,
153            source_layer: self.source_layer,
154            id: self.id,
155            feature_index: self.feature_index,
156        };
157        let slabs = slabs(area);
158        match self.exact {
159            ExactGeometry::Polygon(polygon) => {
160                let cut = |ring: &LineString<f64>| {
161                    slabs
162                        .iter()
163                        .fold(ring.0.clone(), |ring, slab| slab.ring(&ring))
164                };
165                let exterior = cut(polygon.exterior());
166                if exterior.len() < 4 {
167                    return Vec::new();
168                }
169                let holes = polygon
170                    .interiors()
171                    .iter()
172                    .map(cut)
173                    .filter(|ring| ring.len() >= 4)
174                    .map(LineString::new)
175                    .collect();
176                Self::from_polygon(Polygon::new(LineString::new(exterior), holes), meta)
177                    .into_iter()
178                    .collect()
179            }
180            ExactGeometry::Point(point) => {
181                let (min, max) = (area.min(), area.max());
182                let inside =
183                    (min.x..=max.x).contains(&point.x()) && (min.y..=max.y).contains(&point.y());
184                inside
185                    .then(|| Self::from_point(point, meta))
186                    .flatten()
187                    .into_iter()
188                    .collect()
189            }
190            ExactGeometry::LineString(line) => slabs
191                .iter()
192                .fold(vec![line.0], |runs, slab| {
193                    runs.iter().flat_map(|run| slab.line(run)).collect()
194                })
195                .into_iter()
196                .filter_map(|run| Self::from_linestring(LineString::new(run), meta.clone()))
197                .collect(),
198        }
199    }
200}
