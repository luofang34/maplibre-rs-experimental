1//! Cutting an indexed geometry down to a tile and its buffer, as GL JS's `geojson-vt` cuts a
2//! GeoJSON feature for each tile it reaches: one axis at a time, each ring on its own.
3//!
4//! A feature indexed whole in every tile it reaches would be found once for each of them, and
5//! an extrusion's roof would stand in a neighbour's index where that tile holds none of it.
6
7use geo_types::{LineString, Polygon, Rect};
8
9use super::{ExactGeometry, FeatureMeta, IndexedGeometry};
10use crate::coords::clip::{line_within, ring_within};
11
12impl IndexedGeometry<f64> {
13    /// The geometry in the units of a child tile `scale` times finer whose top left corner lies
14    /// at `offset` of the child's units, as GL JS moves an overzoomed tile's features into it.
15    pub fn rescaled(&self, scale: f64, offset: [f64; 2]) -> Self {
16        use geo::MapCoords;
17        let map = |coord: geo_types::Coord<f64>| geo_types::Coord {
18            x: coord.x * scale - offset[0],
19            y: coord.y * scale - offset[1],
20        };
21        let exact = match &self.exact {
22            ExactGeometry::Polygon(polygon) => ExactGeometry::Polygon(polygon.map_coords(map)),
23            ExactGeometry::LineString(line) => ExactGeometry::LineString(line.map_coords(map)),
24            ExactGeometry::Point(point) => ExactGeometry::Point(point.map_coords(map)),
25        };
26        let (lower, upper) = (self.bounds.lower(), self.bounds.upper());
27        let (lower, upper) = (map(lower.0), map(upper.0));
28        Self {
29            bounds: rstar::AABB::from_corners(lower.into(), upper.into()),
30            exact,
31            properties: self.properties.clone(),
32            source_layer: self.source_layer.clone(),
33            id: self.id,
34            feature_index: self.feature_index,
35        }
36    }
37
38    /// The parts of the geometry within `area`, each indexed on its own.
39    pub fn clipped_to(self, area: Rect<f64>) -> Vec<Self> {
40        let meta = FeatureMeta {
41            properties: self.properties,
42            source_layer: self.source_layer,
43            id: self.id,
44            feature_index: self.feature_index,
45        };
46        match self.exact {
47            ExactGeometry::Polygon(polygon) => {
48                let cut = |ring: &LineString<f64>| ring_within(&ring.0, area);
49                let exterior = cut(polygon.exterior());
50                if exterior.len() < 4 {
51                    return Vec::new();
52                }
53                let holes = polygon
54                    .interiors()
55                    .iter()
56                    .map(cut)
57                    .filter(|ring| ring.len() >= 4)
58                    .map(LineString::new)
59                    .collect();
60                Self::from_polygon(Polygon::new(LineString::new(exterior), holes), meta)
61                    .into_iter()
62                    .collect()
63            }
64            ExactGeometry::Point(point) => {
65                let (min, max) = (area.min(), area.max());
66                let inside =
67                    (min.x..=max.x).contains(&point.x()) && (min.y..=max.y).contains(&point.y());
68                inside
69                    .then(|| Self::from_point(point, meta))
70                    .flatten()
71                    .into_iter()
72                    .collect()
73            }
74            ExactGeometry::LineString(line) => line_within(line.0, area)
75                .into_iter()
76                .filter_map(|run| Self::from_linestring(LineString::new(run), meta.clone()))
77                .collect(),
78        }
79    }
80}
