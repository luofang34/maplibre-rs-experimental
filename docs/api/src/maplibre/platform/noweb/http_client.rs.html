18/// Cloneable HTTP transport that fetches complete bodies and preserves HTTP or network causes.
19/// A configured cache directory enables the HTTP cache middleware; clones share the client.
20#[derive(Clone)]
21pub struct ReqwestHttpClient {
22    client: ClientWithMiddleware,
23}

25enum TransportClassification {
26    NotFound(String),
27    HttpResponse { url: String, status: u16 },
28    Temporary,
29    Terminal,
30}
31
32impl TransportClassification {
33    fn into_fetch_error(self, source: impl std::error::Error + 'static) -> SourceFetchError {
34        match self {
35            Self::NotFound(url) => SourceFetchError::not_found(&url),
36            Self::HttpResponse { url, status } => {
37                SourceFetchError::http_response(&url, status, source)
38            }
39            Self::Temporary => SourceFetchError::temporary(source),
40            Self::Terminal => SourceFetchError(Box::new(source)),
41        }
42    }
43}
44
45impl SourceFetchError {
46    fn classify_reqwest_ref(err: &reqwest::Error) -> TransportClassification {
47        if let Some(status) = err.status() {
48            let url = err
49                .url()
50                .map_or("<unknown>", reqwest::Url::as_str)
51                .to_owned();
52            if status == StatusCode::NOT_FOUND {
53                return TransportClassification::NotFound(url);
54            }
55            return TransportClassification::HttpResponse {
56                url,
57                status: status.as_u16(),
58            };
59        }
60        if err.is_timeout() || err.is_connect() || err.is_request() || err.is_body() {
61            return TransportClassification::Temporary;
62        }
63        classify_error_chain(std::error::Error::source(err))
64    }
65}
66
67fn is_transient_io_error(err: &std::io::Error) -> bool {
68    use std::io::ErrorKind;
69    match err.kind() {
70        ErrorKind::TimedOut
71        | ErrorKind::ConnectionRefused
72        | ErrorKind::ConnectionReset
73        | ErrorKind::ConnectionAborted
74        | ErrorKind::NotConnected
75        | ErrorKind::AddrNotAvailable
76        | ErrorKind::NetworkUnreachable
77        | ErrorKind::HostUnreachable
78        | ErrorKind::Interrupted
79        | ErrorKind::WouldBlock
80        | ErrorKind::UnexpectedEof => true,
81        ErrorKind::Other => {
82            let msg = err.to_string().to_ascii_lowercase();
83            msg.contains("dns")
84                || msg.contains("lookup address")
85                || msg.contains("name resolution")
86                || msg.contains("nodename nor servname")
87                || msg.contains("name or service not known")
88                || msg.contains("no address associated")
89        }
90        _ => false,
91    }
92}
93
94fn classify_single_error(err: &(dyn std::error::Error + 'static)) -> TransportClassification {
95    if let Some(mw_err) = err.downcast_ref::<reqwest_middleware::Error>() {
96        return classify_middleware(mw_err);
97    }
98    if let Some(req_err) = err.downcast_ref::<reqwest::Error>() {
99        return SourceFetchError::classify_reqwest_ref(req_err);
100    }
101    if let Some(io_err) = err.downcast_ref::<std::io::Error>() {
102        if is_transient_io_error(io_err) {
103            return TransportClassification::Temporary;
104        }
105    }
106    TransportClassification::Terminal
107}
108
109fn classify_error_chain(
110    mut current: Option<&(dyn std::error::Error + 'static)>,
111) -> TransportClassification {
112    while let Some(err) = current {
113        let class = classify_single_error(err);
114        if !matches!(class, TransportClassification::Terminal) {
115            return class;
116        }
117        current = err.source();
118    }
119    TransportClassification::Terminal
120}
121
122fn classify_middleware(err: &reqwest_middleware::Error) -> TransportClassification {
123    match err {
124        reqwest_middleware::Error::Reqwest(req_err) => {
125            SourceFetchError::classify_reqwest_ref(req_err)
126        }
127        reqwest_middleware::Error::Middleware(inner) => {
128            if let Some(boxed) = inner.downcast_ref::<Box<dyn std::error::Error + Send + Sync>>() {
129                let class = classify_error_chain(Some(boxed.as_ref()));
130                if !matches!(class, TransportClassification::Terminal) {
131                    return class;
132                }
133            }
134            for cause in inner.chain() {
135                let class = classify_single_error(cause);
136                if !matches!(class, TransportClassification::Terminal) {
137                    return class;
138                }
139            }
140            classify_error_chain(std::error::Error::source(err))
141        }
142    }
143}

145impl From<reqwest::Error> for SourceFetchError {
146    fn from(err: reqwest::Error) -> Self {
147        SourceFetchError::classify_reqwest_ref(&err).into_fetch_error(err)
148    }
149}

151impl From<reqwest_middleware::Error> for SourceFetchError {
152    fn from(err: reqwest_middleware::Error) -> Self {
153        match err {
154            reqwest_middleware::Error::Reqwest(error) => error.into(),
155            other => classify_middleware(&other).into_fetch_error(other),
156        }
157    }
158}

160impl ReqwestHttpClient {
161    /// Creates a client, optionally storing cacheable responses under `cache_path`.
162    /// `None` disables the disk cache. If client configuration fails, construction logs
163    /// a warning and falls back to Reqwest's default client.
164    pub fn new<P>(cache_path: Option<P>) -> Self
165    where
166        P: Into<PathBuf>,
167    {
168        // Public tile servers such as OpenStreetMap reject requests without a User-Agent.
169        let client = match Client::builder().user_agent(USER_AGENT).build() {
170            Ok(client) => client,
171            Err(error) => {
172                tracing::warn!(%error, "cannot build the HTTP client with a user agent");
173                Client::new()
174            }
175        };
176        let mut builder = reqwest_middleware::ClientBuilder::new(client);
177
178        if let Some(cache_path) = cache_path {
179            builder = builder.with(Cache(HttpCache {
180                mode: CacheMode::Default,
181                manager: CACacheManager {
182                    path: cache_path.into(),
183                },
184                options: HttpCacheOptions::default(),
185            }))
186        }
187        let client = builder.build();
188
189        Self { client }
190    }
191}

205/// An in-process HTTP handler for local test servers when network sockets are unavailable.
206pub type InProcessHandler = std::sync::Arc<
207    dyn Fn(
208            &str,
209            Option<ByteRange>,
210        ) -> std::pin::Pin<
211            Box<dyn std::future::Future<Output = Result<Vec<u8>, SourceFetchError>> + Send>,
212        > + Send
213        + Sync,
214>;
215
216static IN_PROCESS_HANDLERS: std::sync::RwLock<Vec<(String, InProcessHandler)>> =
217    std::sync::RwLock::new(Vec::new());
218
219/// Registers an in-process handler for URLs matching `prefix`.
220pub fn register_in_process_handler(prefix: String, handler: InProcessHandler) {
221    if let Ok(mut handlers) = IN_PROCESS_HANDLERS.write() {
222        handlers.retain(|(p, _)| p != &prefix);
223        handlers.push((prefix, handler));
224    }
225}
226
227/// Unregisters an in-process handler by `prefix`.
228pub fn unregister_in_process_handler(prefix: &str) {
229    if let Ok(mut handlers) = IN_PROCESS_HANDLERS.write() {
230        handlers.retain(|(p, _)| p != prefix);
231    }
232}

234impl ReqwestHttpClient {
235    /// A server that ignores the range answers with the whole body, cut down here.
236    async fn get(&self, url: &str, range: Option<ByteRange>) -> Result<Vec<u8>, SourceFetchError> {
237        let mut request = self.client.get(url);
238        if let Some(range) = range {
239            request = request.header(reqwest::header::RANGE, range.header());
240        }
241        let response = match request.send().await {
242            Ok(response) => response,
243            Err(err) => {
244                let handler = {
245                    let handlers = IN_PROCESS_HANDLERS.read().ok();
246                    handlers.and_then(|h| {
247                        h.iter()
248                            .find(|(prefix, _)| url.starts_with(prefix))
249                            .map(|(_, handler)| handler.clone())
250                    })
251                };
252                if let Some(handler) = handler {
253                    return handler(url, range).await;
254                }
255                return Err(err.into());
256            }
257        };
258        if response.status() == StatusCode::NOT_FOUND {
259            return Err(SourceFetchError::not_found(url));
260        }
261        let status = response.status().as_u16();
262        match response.error_for_status() {
263            Ok(response) => {
264                if response.status() == StatusCode::NOT_MODIFIED {
265                    log::info!("Using data from cache");
266                }
267
268                let partial = response.status() == StatusCode::PARTIAL_CONTENT;
269                let body = response.bytes().await?;
270                match range {
271                    Some(range) if !partial => range.slice(url, &body).map(<[u8]>::to_vec),
272                    _ => Ok(Vec::from(body.as_ref())),
273                }
274            }
275            Err(error) => Err(SourceFetchError::http_response(url, status, error)),
276        }
277    }
278}

296    #[test]
297    fn middleware_wrapping_transient_io_errors_is_retryable() {
298        for io_err in [
299            std::io::Error::new(ErrorKind::TimedOut, "socket timed out"),
300            std::io::Error::new(ErrorKind::ConnectionRefused, "connection refused"),
301            std::io::Error::new(ErrorKind::ConnectionReset, "connection reset by peer"),
302            std::io::Error::new(ErrorKind::NetworkUnreachable, "network unreachable"),
303            std::io::Error::new(ErrorKind::HostUnreachable, "no route to host"),
304            std::io::Error::other(
305                "failed to lookup address information: nodename nor servname provided, or not known",
306            ),
307        ] {
308            let mw_err = reqwest_middleware::Error::middleware(TransportWrapper(io_err));
309            let fetch_err = SourceFetchError::from(mw_err);
310            assert!(
311                fetch_err.is_retryable(),
312                "expected retryable for {}",
313                fetch_err.describe()
314            );
315        }
316    }

339    #[tokio::test]
340    async fn cached_client_connect_or_dns_reqwest_error_in_middleware_is_retryable() {
341        let raw_client = Client::builder()
342            .timeout(std::time::Duration::from_millis(200))
343            .build()
344            .expect("reqwest client");
345        let connect_err = raw_client
346            .get("http://127.0.0.1:1/tiles/0/0/0.png")
347            .send()
348            .await
349            .expect_err("closed local port must fail");
350
351        let direct_mw = SourceFetchError::from(reqwest_middleware::Error::middleware(connect_err));
352        assert!(
353            direct_mw.is_retryable(),
354            "direct middleware reqwest connect error must be retryable: {}",
355            direct_mw.describe()
356        );
357
358        let cache_dir =
359            std::env::temp_dir().join(format!("maplibre-http-cache-test-{}", std::process::id()));
360        let cached_client = ReqwestHttpClient::new(Some(&cache_dir));
361        let cached_err = cached_client
362            .fetch("http://127.0.0.1:1/tiles/0/0/0.png")
363            .await
364            .expect_err("cached fetch against closed port must fail");
365        let _ = std::fs::remove_dir_all(&cache_dir);
366        assert!(
367            cached_err.is_retryable(),
368            "http-cache-reqwest wrapped transport error must be retryable: {}",
369            cached_err.describe()
370        );
371
372        let bad_url_err = raw_client
373            .get("://not-a-valid-url")
374            .send()
375            .await
376            .expect_err("malformed URL must fail");
377        let bad_url_mw = SourceFetchError::from(reqwest_middleware::Error::middleware(bad_url_err));
378        assert!(
379            !bad_url_mw.is_retryable(),
380            "malformed URL builder error must remain terminal: {}",
381            bad_url_mw.describe()
382        );
383    }
