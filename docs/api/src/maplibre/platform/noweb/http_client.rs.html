38#[derive(Debug, thiserror::Error)]
39#[error("invalid HTTP request header `{name}`: {reason}")]
40struct InvalidHeaderError {
41    name: String,
42    reason: String,
43}
44
45fn build_header_map(headers: &[(String, String)]) -> Result<HeaderMap, SourceFetchError> {
46    let mut map = HeaderMap::with_capacity(headers.len());
47    for (name, value) in headers {
48        let header_name = HeaderName::from_bytes(name.as_bytes()).map_err(|err| {
49            SourceFetchError(Box::new(InvalidHeaderError {
50                name: name.clone(),
51                reason: err.to_string(),
52            }))
53        })?;
54        let header_val = HeaderValue::from_str(value).map_err(|err| {
55            SourceFetchError(Box::new(InvalidHeaderError {
56                name: name.clone(),
57                reason: err.to_string(),
58            }))
59        })?;
60        map.append(header_name, header_val);
61    }
62    Ok(map)
63}

65fn format_cache_key(url: &str, range: Option<ByteRange>, headers: &[(String, String)]) -> String {
66    let mut key = match range {
67        Some(r) => format!("{url}#{}", r.header()),
68        None => url.to_owned(),
69    };
70    if !headers.is_empty() {
71        for (k, v) in headers {
72            key.push('|');
73            key.push_str(&k.to_ascii_lowercase());
74            key.push(':');
75            key.push_str(v);
76        }
77    }
78    key
79}

84            &str,

86        ) -> Pin<Box<dyn Future<Output = Result<Vec<u8>, SourceFetchError>> + Send>>
87        + Send
88        + Sync,

91/// Instance-scoped registry of prefix-matched in-process fallback HTTP handlers.
92#[derive(Clone, Default)]
93pub struct InProcessRoutes(Arc<RwLock<Vec<(String, InProcessHandler)>>>);
94
95impl InProcessRoutes {
96    /// Creates an empty route table.
97    pub fn new() -> Self {
98        Self::default()
99    }
100
101    /// Registers an in-process handler for URLs starting with `prefix`, replacing any prior
102    /// handler registered for the exact same `prefix`.
103    pub fn register(&self, prefix: impl Into<String>, handler: InProcessHandler) {
104        let prefix = prefix.into();
105        if let Ok(mut handlers) = self.0.write() {
106            handlers.retain(|(existing, _)| existing != &prefix);
107            handlers.push((prefix, handler));
108        }
109    }
110
111    /// Removes any handler registered for `prefix`.
112    pub fn unregister(&self, prefix: &str) {
113        if let Ok(mut handlers) = self.0.write() {
114            handlers.retain(|(existing, _)| existing != prefix);
115        }
116    }
117
118    fn lookup(&self, url: &str) -> Option<InProcessHandler> {
119        let handlers = self.0.read().ok()?;
120        handlers
121            .iter()
122            .find(|(prefix, _)| url.starts_with(prefix))
123            .map(|(_, handler)| Arc::clone(handler))
124    }
125}

127static SHARED_DIRECT_CLIENT: LazyLock<Client> = LazyLock::new(|| {
128    // Public tile servers such as OpenStreetMap reject requests without a User-Agent.
129    match Client::builder().user_agent(USER_AGENT).build() {
130        Ok(client) => client,
131        Err(error) => {
132            tracing::warn!(%error, "cannot build the HTTP client with a user agent");
133            Client::new()
134        }
135    }
136});
137
138static SHARED_RANGE_CLIENT: LazyLock<Client> = LazyLock::new(|| {
139    // Byte-range requests (e.g. PMTiles archives) must read exact uncompressed byte offsets
140    // (`Accept-Encoding: identity` with automatic gunzip disabled) and bypass `http-cache`.
141    match Client::builder().user_agent(USER_AGENT).no_gzip().build() {
142        Ok(client) => client,
143        Err(error) => {
144            tracing::warn!(%error, "cannot build the range HTTP client with no_gzip");
145            Client::new()
146        }
147    }
148});

150/// Cloneable HTTP transport that fetches complete bodies and preserves HTTP or network causes.
151/// A configured cache directory enables the HTTP cache middleware; clones share the client.
152#[derive(Clone)]
153pub struct ReqwestHttpClient {
154    cache_manager: Option<CACacheManager>,
155    client: Arc<OnceLock<ClientWithMiddleware>>,
156    offline_cache_client: Arc<OnceLock<ClientWithMiddleware>>,
157    frontend: Option<Arc<SharedCacheFrontend>>,
158    in_process: Option<InProcessRoutes>,
159}
160
161impl ReqwestHttpClient {
162    /// Creates a client, optionally storing cacheable responses under `cache_path`.
163    /// `None` disables the disk cache. If client configuration fails, construction logs
164    /// a warning and falls back to Reqwest's default client.
165    pub fn new<P>(cache_path: Option<P>) -> Self
166    where
167        P: Into<PathBuf>,
168    {
169        let cache_path = cache_path.map(Into::into);
170        let frontend = cache_path
171            .as_ref()
172            .map(|_| Arc::new(SharedCacheFrontend::new(MEMORY_CACHE_CAP_BYTES)));
173        Self::build(cache_path, frontend, None)
174    }
175
176    /// Creates a client whose in-memory cache and disk-I/O gate are shared through `frontend`.
177    pub fn with_cache_frontend<P>(cache_path: Option<P>, frontend: HttpCacheFrontend) -> Self
178    where
179        P: Into<PathBuf>,
180    {
181        Self::build(cache_path.map(Into::into), Some(frontend.0), None)
182    }
183
184    /// Creates a client that consults `routes` when a network request cannot connect.
185    pub fn with_in_process_routes<P>(cache_path: Option<P>, routes: InProcessRoutes) -> Self
186    where
187        P: Into<PathBuf>,
188    {
189        let cache_path = cache_path.map(Into::into);
190        let frontend = cache_path
191            .as_ref()
192            .map(|_| Arc::new(SharedCacheFrontend::new(MEMORY_CACHE_CAP_BYTES)));
193        Self::build(cache_path, frontend, Some(routes))
194    }
195
196    fn build(
197        cache_path: Option<PathBuf>,
198        frontend: Option<Arc<SharedCacheFrontend>>,
199        in_process: Option<InProcessRoutes>,
200    ) -> Self {
201        let cache_manager = cache_path.map(|path| CACacheManager::new(path, false));
202        Self {
203            cache_manager,
204            client: Arc::new(OnceLock::new()),
205            offline_cache_client: Arc::new(OnceLock::new()),
206            frontend,
207            in_process,
208        }
209    }
210
211    fn cached_client(&self) -> &ClientWithMiddleware {
212        self.client.get_or_init(|| {
213            let mut builder = reqwest_middleware::ClientBuilder::new(SHARED_DIRECT_CLIENT.clone());
214            if let Some(manager) = self.cache_manager.clone() {
215                builder = builder.with(Cache(HttpCache {
216                    mode: CacheMode::ForceCache,
217                    manager,
218                    options: HttpCacheOptions::default(),
219                }));
220            }
221            builder.build()
222        })
223    }
224
225    fn offline_client(&self) -> Option<&ClientWithMiddleware> {
226        let manager = self.cache_manager.as_ref()?;
227        Some(self.offline_cache_client.get_or_init(|| {
228            reqwest_middleware::ClientBuilder::new(SHARED_DIRECT_CLIENT.clone())
229                .with(Cache(HttpCache {
230                    mode: CacheMode::OnlyIfCached,
231                    manager: manager.clone(),
232                    options: HttpCacheOptions::default(),
233                }))
234                .build()
235        }))
236    }
237
238    async fn try_disk_cache(
239        &self,
240        url: &str,
241        cache_key: &str,
242    ) -> Option<Result<Vec<u8>, SourceFetchError>> {
243        let manager = self.cache_manager.as_ref()?;
244        let _io_permit = match &self.frontend {
245            Some(frontend) => frontend.io_gate.acquire().await.ok(),
246            None => None,
247        };
248        let normalized = reqwest::Url::parse(url)
249            .map(|parsed| parsed.to_string())
250            .unwrap_or_else(|_| url.to_owned());
251        let disk_key = format!("GET:{normalized}");
252        let Ok(Some((cached, _))) = manager.get(&disk_key).await else {
253            return None;
254        };
255        if cached.status == StatusCode::NOT_FOUND.as_u16() {
256            return Some(Err(SourceFetchError::not_found(url)));
257        }
258        if (200..300).contains(&cached.status) {
259            if let Some(frontend) = &self.frontend {
260                frontend.insert(cache_key, &cached.body);
261            }
262            return Some(Ok(cached.body));
263        }
264        None
265    }
266
267    fn lookup_in_process_handler(&self, url: &str) -> Option<InProcessHandler> {
268        self.in_process.as_ref()?.lookup(url)
269    }

271    /// Executes `req` under `status`. A server that ignores `Range` answers with the whole body,
272    /// sliced down here.
273    async fn execute(
274        &self,
275        req: &HttpRequest,
276        status: NetworkStatus,
277    ) -> Result<Vec<u8>, SourceFetchError> {
278        let url = req.url.as_str();
279        let range = req.range;
280        let volatile = req.cache_policy.is_volatile();
281        let header_map = build_header_map(&req.headers)?;
282        let cache_key = format_cache_key(url, range, &req.headers);
283
284        if !volatile {
285            if let Some(body) = self
286                .frontend
287                .as_deref()
288                .and_then(|frontend| frontend.get(&cache_key))
289            {
290                return Ok(body.to_vec());
291            }
292            if range.is_none() && req.headers.is_empty() {
293                if let Some(cached) = self.try_disk_cache(url, &cache_key).await {
294                    return cached;
295                }
296            }
297        }
298
299        if status.is_offline() {
300            return self.execute_offline(req, &header_map, &cache_key).await;
301        }
302
303        let response = match (range, volatile) {
304            (Some(range), _) => {
305                let request_builder = SHARED_RANGE_CLIENT
306                    .get(url)
307                    .headers(header_map)
308                    .header(reqwest::header::RANGE, range.header())
309                    .header(reqwest::header::ACCEPT_ENCODING, "identity");
310                match request_builder.send().await {
311                    Ok(response) => response,
312                    Err(err) => match self.lookup_in_process_handler(url) {
313                        Some(handler) => return handler(url, Some(range)).await,
314                        None => return Err(err.into()),
315                    },
316                }
317            }
318            (None, true) => {
319                // Volatile sources bypass `CACacheManager` and `SharedCacheFrontend` completely.
320                match SHARED_DIRECT_CLIENT
321                    .get(url)
322                    .headers(header_map)
323                    .send()
324                    .await
325                {
326                    Ok(response) => response,
327                    Err(err) => match self.lookup_in_process_handler(url) {
328                        Some(handler) => return handler(url, None).await,
329                        None => return Err(err.into()),
330                    },
331                }
332            }
333            (None, false) => {
334                let _io_permit = match &self.frontend {
335                    Some(frontend) => frontend.io_gate.acquire().await.ok(),
336                    None => None,
337                };
338                match self
339                    .cached_client()
340                    .get(url)
341                    .headers(header_map)
342                    .send()
343                    .await
344                {
345                    Ok(response) => response,
346                    Err(err) => match self.lookup_in_process_handler(url) {
347                        Some(handler) => return handler(url, None).await,
348                        None => return Err(err.into()),
349                    },
350                }
351            }
352        };
353
354        self.decode_response(url, range, req.cache_policy, &cache_key, response)
355            .await
356    }

358    async fn execute_offline(
359        &self,
360        req: &HttpRequest,
361        header_map: &HeaderMap,
362        cache_key: &str,
363    ) -> Result<Vec<u8>, SourceFetchError> {
364        let url = req.url.as_str();
365        if !req.cache_policy.is_volatile() && req.range.is_none() {
366            if let Some(cached) = self.try_disk_cache(url, cache_key).await {
367                return cached;
368            }
369            if let Some(offline_client) = self.offline_client() {
370                let _io_permit = match &self.frontend {
371                    Some(frontend) => frontend.io_gate.acquire().await.ok(),
372                    None => None,
373                };
374                if let Ok(response) = offline_client
375                    .get(url)
376                    .headers(header_map.clone())
377                    .send()
378                    .await
379                {
380                    if response.status() != StatusCode::GATEWAY_TIMEOUT {
381                        return self
382                            .decode_response(url, None, req.cache_policy, cache_key, response)
383                            .await;
384                    }
385                }
386            }
387        }
388        if is_remote_http_url(url) {
389            return Err(SourceFetchError::offline(url));
390        }
391        // Non-HTTP custom scheme fallback if registered in-process.
392        if let Some(handler) = self.lookup_in_process_handler(url) {
393            return handler(url, req.range).await;
394        }
395        Err(SourceFetchError::offline(url))
396    }
397
398    async fn decode_response(
399        &self,
400        url: &str,
401        range: Option<ByteRange>,
402        policy: CachePolicy,
403        cache_key: &str,
404        response: reqwest::Response,
405    ) -> Result<Vec<u8>, SourceFetchError> {
406        if response.status() == StatusCode::NOT_FOUND {
407            return Err(SourceFetchError::not_found(url));
408        }
409        let status = response.status().as_u16();
410        match response.error_for_status() {
411            Ok(response) => {
412                if response.status() == StatusCode::NOT_MODIFIED {
413                    log::info!("Using data from cache");
414                }
415
416                let partial = response.status() == StatusCode::PARTIAL_CONTENT;
417                let body = response.bytes().await?;
418                let out = match range {
419                    Some(range) if !partial => range.slice(url, &body).map(<[u8]>::to_vec)?,
420                    Some(_) | None => Vec::from(body.as_ref()),
421                };
422                if !policy.is_volatile() {
423                    if let Some(frontend) = &self.frontend {
424                        frontend.insert(cache_key, &out);
425                    }
426                }
427                Ok(out)
428            }
429            Err(error) => Err(SourceFetchError::http_response(url, status, error)),
430        }
431    }
432}

434#[cfg_attr(not(feature = "thread-safe-futures"), async_trait(?Send))]
435#[cfg_attr(feature = "thread-safe-futures", async_trait)]
436impl HttpClient for ReqwestHttpClient {
437    async fn fetch(&self, url: &str) -> Result<Vec<u8>, SourceFetchError> {
438        let req = HttpRequest::new(url, ResourceKind::infer_from_url(url));
439        self.execute(&req, NetworkStatus::Online).await
440    }
441
442    async fn fetch_range(&self, url: &str, range: ByteRange) -> Result<Vec<u8>, SourceFetchError> {
443        let req = HttpRequest::new(url, ResourceKind::infer_from_url(url)).with_range(range);
444        self.execute(&req, NetworkStatus::Online).await
445    }
446
447    async fn fetch_request(&self, req: &HttpRequest) -> Result<Vec<u8>, SourceFetchError> {
448        self.execute(req, NetworkStatus::Online).await
449    }
450
451    async fn fetch_request_with_status(
452        &self,
453        req: &HttpRequest,
454        status: NetworkStatus,
455    ) -> Result<Vec<u8>, SourceFetchError> {
456        self.execute(req, status).await
457    }
458}
