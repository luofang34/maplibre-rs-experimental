18/// Cloneable HTTP transport that fetches complete bodies and preserves HTTP or network causes.
19/// A configured cache directory enables the HTTP cache middleware; clones share the client.
20#[derive(Clone)]
21pub struct ReqwestHttpClient {
22    client: ClientWithMiddleware,
23}
24
25impl From<reqwest::Error> for SourceFetchError {
26    fn from(err: reqwest::Error) -> Self {
27        if err.is_timeout() || err.is_connect() || err.is_request() || err.is_body() {
28            SourceFetchError::temporary(err)
29        } else {
30            SourceFetchError(Box::new(err))
31        }
32    }
33}
34
35impl From<reqwest_middleware::Error> for SourceFetchError {
36    fn from(err: reqwest_middleware::Error) -> Self {
37        match err {
38            reqwest_middleware::Error::Reqwest(error) => error.into(),
39            other => SourceFetchError(Box::new(other)),
40        }
41    }
42}

44impl ReqwestHttpClient {
45    /// Creates a client, optionally storing cacheable responses under `cache_path`.
46    /// `None` disables the disk cache. If client configuration fails, construction logs
47    /// a warning and falls back to Reqwest's default client.
48    pub fn new<P>(cache_path: Option<P>) -> Self
49    where
50        P: Into<PathBuf>,
51    {
52        // Public tile servers such as OpenStreetMap reject requests without a User-Agent.
53        let client = match Client::builder().user_agent(USER_AGENT).build() {
54            Ok(client) => client,
55            Err(error) => {
56                tracing::warn!(%error, "cannot build the HTTP client with a user agent");
57                Client::new()
58            }
59        };
60        let mut builder = reqwest_middleware::ClientBuilder::new(client);
61
62        if let Some(cache_path) = cache_path {
63            builder = builder.with(Cache(HttpCache {
64                mode: CacheMode::Default,
65                manager: CACacheManager {
66                    path: cache_path.into(),
67                },
68                options: HttpCacheOptions::default(),
69            }))
70        }
71        let client = builder.build();
72
73        Self { client }
74    }
75}

79impl HttpClient for ReqwestHttpClient {
80    async fn fetch(&self, url: &str) -> Result<Vec<u8>, SourceFetchError> {
81        self.get(url, None).await
82    }
83
84    async fn fetch_range(&self, url: &str, range: ByteRange) -> Result<Vec<u8>, SourceFetchError> {
85        self.get(url, Some(range)).await
86    }
87}

89impl ReqwestHttpClient {
90    /// A server that ignores the range answers with the whole body, cut down here.
91    async fn get(&self, url: &str, range: Option<ByteRange>) -> Result<Vec<u8>, SourceFetchError> {
92        let mut request = self.client.get(url);
93        if let Some(range) = range {
94            request = request.header(reqwest::header::RANGE, range.header());
95        }
96        let response = request.send().await?;
97        if response.status() == StatusCode::NOT_FOUND {
98            return Err(SourceFetchError::not_found(url));
99        }
100        let status = response.status().as_u16();
101        match response.error_for_status() {
102            Ok(response) => {
103                if response.status() == StatusCode::NOT_MODIFIED {
104                    log::info!("Using data from cache");
105                }
106
107                let partial = response.status() == StatusCode::PARTIAL_CONTENT;
108                let body = response.bytes().await?;
109                match range {
110                    Some(range) if !partial => range.slice(url, &body).map(<[u8]>::to_vec),
111                    _ => Ok(Vec::from(body.as_ref())),
112                }
113            }
114            Err(error) => Err(SourceFetchError::http_response(url, status, error)),
115        }
116    }
