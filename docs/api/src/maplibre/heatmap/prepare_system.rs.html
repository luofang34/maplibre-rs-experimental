1//! Sizes each visible heatmap layer's targets and writes its ramp and opacity for the frame.
2
3use std::collections::HashSet;
4
5use crate::{
6    context::MapContext,
7    heatmap::resources::HeatmapResources,
8    render::{
9        eventually::{Eventually, Eventually::Initialized},
10        Renderer,
11    },
12    style::layer::LayerPaint,
13    tcs::system::{SystemError, SystemResult},
14};
15
16pub fn prepare_system(
17    MapContext {
18        world,
19        style,
20        view_state,
21        renderer: Renderer { device, queue, .. },
22        ..
23    }: &mut MapContext,
24) -> SystemResult {
25    let Some(Initialized(resources)) = world
26        .resources
27        .query_mut::<&mut Eventually<HeatmapResources>>()
28    else {
29        return Err(SystemError::Dependencies);
30    };
31    let zoom = view_state.zoom().value();
32    let size = (
33        view_state.width().round() as u32,
34        view_state.height().round() as u32,
35    );
36    let mut written = HashSet::new();
37    for layer in &style.layers {
38        let Some(LayerPaint::Heatmap(paint)) = &layer.paint else {
39            continue;
40        };
41        if !layer.is_visible_at(view_state.style_zoom().value()) {
42            continue;
43        }
44        resources.write_layer(
45            device,
46            queue,
47            &layer.id,
48            size,
49            &paint.ramp(),
50            paint.opacity_at(zoom),
51        );
52        written.insert(layer.id.as_str());
53    }
54    resources.retain_layers(&written);
55    Ok(())
56}
