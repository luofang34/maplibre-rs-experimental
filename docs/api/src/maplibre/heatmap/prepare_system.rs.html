1//! Sizes each visible heatmap layer's targets and writes its ramp and opacity for the frame.

3use std::collections::HashSet;
4
5use crate::{
6    context::MapContext,
7    heatmap::resources::HeatmapResources,
8    render::{
9        eventually::{Eventually, Eventually::Initialized},
10        Renderer,
11    },
12    style::layer::LayerPaint,
13    tcs::system::{SystemError, SystemResult},
14};

16/// Calculates the downscaled (1/4x) density target dimensions for a given viewport size,
17/// ensuring both width and height are quarter-resolution and never 0.
18#[inline]
19pub fn density_target_size(width: f64, height: f64) -> (u32, u32) {
20    (
21        (width.round() as u32 >> 2).max(1),
22        (height.round() as u32 >> 2).max(1),
23    )
24}
25
26pub fn prepare_system(
27    MapContext {
28        world,
29        style,
30        view_state,
31        renderer: Renderer { device, queue, .. },
32        ..
33    }: &mut MapContext,
34) -> SystemResult {
35    let Some(Initialized(resources)) = world
36        .resources
37        .query_mut::<&mut Eventually<HeatmapResources>>()
38    else {
39        return Err(SystemError::Dependencies);
40    };
41    let zoom = view_state.zoom().value();
42    let size = density_target_size(view_state.width(), view_state.height());
43    let mut written = HashSet::new();
44    for layer in &style.layers {
45        let Some(LayerPaint::Heatmap(paint)) = &layer.paint else {
46            continue;
47        };
48        if !layer.is_visible_at(view_state.style_zoom().value()) {
49            continue;
50        }
51        resources.write_layer(
52            device,
53            queue,
54            &layer.id,
55            size,
56            &paint.ramp(),
57            paint.opacity_at(zoom),
58        );
59        written.insert(layer.id.as_str());
60    }
61    resources.retain_layers(&written);
62    Ok(())
63}

65#[cfg(test)]
66mod tests {
67    use super::*;
68
69    #[test]
70    fn density_target_dimensions_are_quarter_resolution_and_never_zero() {
71        // Full HD viewport is downscaled to 1/4x (480x270)
72        assert_eq!(density_target_size(1920.0, 1080.0), (480, 270));
73
74        // 4K viewport
75        assert_eq!(density_target_size(3840.0, 2160.0), (960, 540));
76
77        // Sub-4 pixel dimensions must clamp to at least 1 (never zero)
78        assert_eq!(density_target_size(4.0, 4.0), (1, 1));
79        assert_eq!(density_target_size(3.0, 3.0), (1, 1));
80        assert_eq!(density_target_size(2.0, 2.0), (1, 1));
81        assert_eq!(density_target_size(1.0, 1.0), (1, 1));
82        assert_eq!(density_target_size(0.0, 0.0), (1, 1));
83
84        // Negative dimensions clamp to at least 1
85        assert_eq!(density_target_size(-100.0, -100.0), (1, 1));
86    }
87}
