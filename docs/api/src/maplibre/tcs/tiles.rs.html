15static EMPTY_PYRAMID: TilePyramid<()> = TilePyramid::new();

17/// Coordinates identifying one tile record in a [`Tiles`] store.
18#[derive(Copy, Clone, Debug)]
19pub struct Tile {
20    /// Canonical coordinates used to build the store's quadkey.
21    pub coords: WorldTileCoords,
22}

24/// Data associated with a [`Tile`], with at most one component of each type per tile.
25pub trait TileComponent: Downcast + 'static {
26    /// Concrete [`TypeId`] of this component value, used for per-component pyramid indexing.
27    fn component_type_id(&self) -> TypeId {
28        TypeId::of::<Self>()
29    }
30}
31impl_downcast!(TileComponent);

33/// Owns tile records, typed components and their feature-query index.
34/// Component references borrow this store and exclude mutation of its tile membership.
35#[derive(Default)]
36pub struct Tiles {
37    /// Tile records keyed by canonical quadkey.
38    pub tiles: BTreeMap<Quadkey, Tile>,
39    /// Component slots under each tile's key. Use [`TileSpawnResult::insert`] to preserve
40    /// one component per type and release replaced values.
41    pub components: FxHashMap<Quadkey, Vec<UnsafeCell<Box<dyn TileComponent>>>>,
42    /// Geometry available to rendered-feature queries; cleared with the owning tiles.
43    pub geometry_index: GeometryIndex,
44    pyramid: TilePyramid<()>,
45    component_pyramids: FxHashMap<TypeId, TilePyramid<()>>,
46}
47
48impl Tiles {
49    /// Hierarchical pyramid of every resident tile in this store.
50    #[inline]
51    pub fn pyramid(&self) -> &TilePyramid<()> {
52        &self.pyramid
53    }
54
55    /// Hierarchical pyramid of resident tiles carrying component `T`.
56    #[inline]
57    pub fn pyramid_of<T: TileComponent>(&self) -> &TilePyramid<()> {
58        self.component_pyramids
59            .get(&TypeId::of::<T>())
60            .unwrap_or(&EMPTY_PYRAMID)
61    }
62
63    /// Borrows `&T` or a pair of shared component references from one tile.
64    /// Built-in queries return `None` for invalid coordinates or missing components.
65    pub fn query<Q: ComponentQuery>(&self, coords: WorldTileCoords) -> Option<Q::Item<'_>> {
66        let mut global_state = GlobalQueryState::default();
67        let state = <Q::State<'_> as QueryState>::create(&mut global_state);
68        Q::query(self, Tile { coords }, state)
69    }
70
71    /// Borrows components of one tile, or returns `None` for missing data or conflicting
72    /// reference types. Repeated shared references are allowed; mutable ones must be disjoint.
73    pub fn query_mut<Q: ComponentQueryMut>(
74        &mut self,
75        coords: WorldTileCoords,
76    ) -> Option<Q::MutItem<'_>> {
77        let mut global_state = GlobalQueryState::default();
78        let state = <Q::State<'_> as QueryState>::create(&mut global_state);
79        Q::query_mut(self, Tile { coords }, state)
80    }
81
82    /// Whether a record exists at these coordinates; invalid coordinates return `false`.
83    #[inline]
84    pub fn exists(&self, coords: WorldTileCoords) -> bool {
85        coords
86            .build_quad_key()
87            .is_some_and(|key| self.components.contains_key(&key))
88    }
89
90    /// Opens a tile for component insertion, preserving its components if it already exists.
91    /// Coordinates outside the canonical tile grid return `None` without changing the store.
92    pub fn spawn_mut(&mut self, coords: WorldTileCoords) -> Option<TileSpawnResult<'_>> {
93        if let Some(key) = coords.build_quad_key() {
94            if self.components.contains_key(&key) {
95                Some(TileSpawnResult {
96                    tiles: self,
97                    tile: Tile { coords },
98                })
99            } else {
100                let tile = Tile { coords };
101                self.tiles.insert(key, tile);
102                self.components.insert(key, Vec::new());
103                self.pyramid.insert(coords, ());
104                Some(TileSpawnResult { tiles: self, tile })
105            }
106        } else {
107            None
108        }
109    }
110
111    /// Drops every tile, component and feature-query geometry.
112    pub fn clear(&mut self) {
113        self.tiles.clear();
114        self.components.clear();
115        self.pyramid.clear();
116        for pyramid in self.component_pyramids.values_mut() {
117            pyramid.clear();
118        }
119        self.geometry_index = GeometryIndex::default();
120    }
121
122    /// Drops a tile with its components and query index; `false` when it was not present.
123    pub fn remove(&mut self, coords: WorldTileCoords) -> bool {
124        let Some(key) = coords.build_quad_key() else {
125            return false;
126        };
127        if let Some(removed_components) = self.components.remove(&key) {
128            for cell in removed_components {
129                let type_id = cell.into_inner().component_type_id();
130                if let Some(pyramid) = self.component_pyramids.get_mut(&type_id) {
131                    pyramid.remove(coords);
132                }
133            }
134        }
135        self.pyramid.remove(coords);
136        self.geometry_index.remove(&coords);
137        self.tiles.remove(&key).is_some()
138    }
139}

141/// Exclusive insertion handle for a valid tile; chained inserts share the same tile record.
142pub struct TileSpawnResult<'t> {
143    tiles: &'t mut Tiles,
144    tile: Tile,
145}
146
147impl<'w> TileSpawnResult<'w> {
148    /// Replaces this tile's component of the same type, dropping its old value immediately.
149    pub fn insert<T: TileComponent>(&mut self, component: T) -> &mut Self {
150        if let Some(key) = self.tile.coords.build_quad_key() {
151            let components = self.tiles.components.entry(key).or_default();
152            if let Some(value) = components
153                .iter_mut()
154                .find_map(|stored| stored.get_mut().downcast_mut::<T>())
155            {
156                *value = component;
157            } else {
158                components.push(UnsafeCell::new(Box::new(component)));
159                self.tiles
160                    .component_pyramids
161                    .entry(TypeId::of::<T>())
162                    .or_default()
163                    .insert(self.tile.coords, ());
164            }
165        }
166        self
167    }
168}

170/// Shared component lookup; built-in queries are references and pairs of references.
171pub trait ComponentQuery {
172    /// Value returned with a lifetime bounded by the tile store.
173    type Item<'t>;
174
175    /// Access ledger handle shared by the members of a pair query.
176    type State<'s>: QueryState<'s>;
177
178    /// Looks up components at `tile` using the common query state.
179    fn query<'t, 's>(
180        tiles: &'t Tiles,
181        tile: Tile,
182        state: Self::State<'s>,
183    ) -> Option<Self::Item<'t>>;
184}
185
186impl<T: TileComponent> ComponentQuery for &T {
187    type Item<'t> = &'t T;
188    type State<'s> = EphemeralQueryState<'s>;
189
190    fn query<'t, 's>(
191        tiles: &'t Tiles,
192        tile: Tile,
193        mut state: Self::State<'s>,
194    ) -> Option<Self::Item<'t>> {
195        state.borrow_shared::<T>()?;
196        let components = tiles.components.get(&tile.coords.build_quad_key()?)?;
197        for (index, component) in components.iter().enumerate() {
198            if state.state.mutably_borrowed_components.contains(&index) {
199                continue;
200            }
201            // Even checking a trait object's type borrows its value. Mutable slots must
202            // be skipped before dereferencing, including slots of a different type.
203            if let Some(value) = unsafe { (&*component.get()).downcast_ref::<T>() } {
204                return Some(value);
205            }
206        }
207        None
208    }
209}

211/// Component lookup with exclusive access to the tile store while constructing the result.
212/// Built-in pairs may mix shared and mutable references to disjoint component types.
213pub trait ComponentQueryMut {
214    /// Value returned with a lifetime bounded by the exclusive tile-store borrow.
215    type MutItem<'t>;
216
217    /// Access ledger handle shared by pair members.
218    type State<'s>: QueryState<'s>;
219
220    /// Looks up components; built-in queries return `None` on absence or conflicting access.
221    fn query_mut<'t, 's>(
222        tiles: &'t mut Tiles,
223        tile: Tile,
224        state: Self::State<'s>,
225    ) -> Option<Self::MutItem<'t>>;
226}
227
228impl<T: TileComponent> ComponentQueryMut for &T {
229    type MutItem<'t> = &'t T;
230    type State<'s> = EphemeralQueryState<'s>;
231
232    fn query_mut<'t, 's>(
233        tiles: &'t mut Tiles,
234        tile: Tile,
235        state: Self::State<'s>,
236    ) -> Option<Self::MutItem<'t>> {
237        <&T as ComponentQuery>::query(tiles, tile, state)
238    }
239}
240
241impl<T: TileComponent> ComponentQueryMut for &mut T {
242    type MutItem<'t> = &'t mut T;
243    type State<'s> = EphemeralQueryState<'s>;
244
245    fn query_mut<'t, 's>(
246        tiles: &'t mut Tiles,
247        tile: Tile,
248        _state: Self::State<'s>,
249    ) -> Option<Self::MutItem<'t>> {
250        let components = tiles.components.get_mut(&tile.coords.build_quad_key()?)?;
251
252        components
253            .iter_mut()
254            .find_map(|component| component.get_mut().downcast_mut())
255    }
256}

258/// Sealed component references used by mutable tuple queries.
259///
260/// Implementations share the tuple's borrow state and cannot inspect slots already borrowed
261/// mutably by another member of the tuple.
262///
263/// ```compile_fail
264/// use maplibre::tcs::tiles::{EphemeralQueryState, Tiles, Tile, ComponentQueryMut, ComponentQueryUnsafe};
265/// struct Custom;
266/// impl ComponentQueryMut for Custom {
267///     type MutItem<'t> = &'t u32;
268///     type State<'s> = EphemeralQueryState<'s>;
269///     fn query_mut<'t, 's>(_: &'t mut Tiles, _: Tile, _: Self::State<'s>) -> Option<&'t u32> { None }
270/// }
271/// impl ComponentQueryUnsafe for Custom {
272///     unsafe fn query_unsafe<'t, 's>(_: &'t Tiles, _: Tile, _: Self::State<'s>) -> Option<&'t u32> { None }
273/// }
274/// ```
275pub trait ComponentQueryUnsafe: ComponentQueryMut + sealed::Reference {
276    /// # Safety
277    /// The caller must prevent overlapping mutable borrows of the queried tile components
278    /// for the lifetime of the returned references, including across query states.
279    unsafe fn query_unsafe<'t, 's>(
280        tiles: &'t Tiles,
281        tile: Tile,
282        state: Self::State<'s>,
283    ) -> Option<Self::MutItem<'t>>;
284}
285
286impl<T: TileComponent> ComponentQueryUnsafe for &T {
287    unsafe fn query_unsafe<'t, 's>(
288        tiles: &'t Tiles,
289        tile: Tile,
290        state: Self::State<'s>,
291    ) -> Option<Self::MutItem<'t>> {
292        <&T as ComponentQuery>::query(tiles, tile, state)
293    }
294}
295
296impl<T: TileComponent> ComponentQueryUnsafe for &mut T {
297    unsafe fn query_unsafe<'t, 's>(
298        tiles: &'t Tiles,
299        tile: Tile,
300        mut state: Self::State<'s>,
301    ) -> Option<Self::MutItem<'t>> {
302        state.borrow_mut::<T>()?;
303        let components = tiles.components.get(&tile.coords.build_quad_key()?)?;
304        for (index, component) in components.iter().enumerate() {
305            if state.state.mutably_borrowed_components.contains(&index) {
306                continue;
307            }
308            // Type inspection is shared only until the test ends. No prior mutable slot
309            // is inspected, and the type check excludes prior shared references to this slot.
310            if unsafe { (&*component.get()).is::<T>() } {
311                state.state.mutably_borrowed_components.insert(index);
312                return unsafe { (&mut *component.get()).downcast_mut() };
313            }
314        }
315        None
316    }
317}

319impl<CQ1: ComponentQuery, CQ2: ComponentQuery> ComponentQuery for (CQ1, CQ2) {
320    type Item<'t> = (CQ1::Item<'t>, CQ2::Item<'t>);
321    type State<'s> = EphemeralQueryState<'s>;
322
323    fn query<'t, 's>(
324        tiles: &'t Tiles,
325        tile: Tile,
326        mut state: Self::State<'s>,
327    ) -> Option<Self::Item<'t>> {
328        Some((
329            CQ1::query(tiles, tile, state.clone_to::<CQ1::State<'_>>())?,
330            CQ2::query(tiles, tile, state.clone_to::<CQ2::State<'_>>())?,
331        ))
332    }
333}

335impl<
336        CQ1: ComponentQueryMut + ComponentQueryUnsafe + 'static,
337        CQ2: ComponentQueryMut + ComponentQueryUnsafe + 'static,
338    > ComponentQueryMut for (CQ1, CQ2)
339{
340    type MutItem<'t> = (CQ1::MutItem<'t>, CQ2::MutItem<'t>);
341    type State<'s> = EphemeralQueryState<'s>;
342
343    fn query_mut<'t, 's>(
344        tiles: &'t mut Tiles,
345        tile: Tile,
346        mut state: Self::State<'s>,
347    ) -> Option<Self::MutItem<'t>> {
348        // Sealed reference queries share one access ledger under an exclusive store borrow.
349        unsafe {
350            Some((
351                <CQ1 as ComponentQueryUnsafe>::query_unsafe(
352                    tiles,
353                    tile,
354                    state.clone_to::<CQ1::State<'_>>(),
355                )?,
356                <CQ2 as ComponentQueryUnsafe>::query_unsafe(
357                    tiles,
358                    tile,
359                    state.clone_to::<CQ2::State<'_>>(),
360                )?,
361            ))
362        }
363    }
364}
