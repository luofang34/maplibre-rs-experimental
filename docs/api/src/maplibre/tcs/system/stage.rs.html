1//! Ordered execution of map systems, with timing and heap-growth measurements.
2
3use crate::{
4    context::MapContext,
5    schedule::{Stage, StageResult},
6    tcs::system::{heap::live_bytes, timings::FrameTimings, IntoSystemContainer, SystemContainer},
7};
8
9/// Runs systems in insertion order, stopping at the first error without rollback.
10#[derive(Default)]
11pub struct SystemStage {
12    systems: Vec<SystemContainer>,
13}
14
15impl SystemStage {
16    /// Appends a system and returns the stage for builder-style construction.
17    #[must_use]
18    pub fn with_system(mut self, system: impl IntoSystemContainer) -> Self {
19        self.add_system(system);
20        self
21    }
22
23    /// Appends a system after every update already registered in this stage.
24    pub fn add_system(&mut self, system: impl IntoSystemContainer) -> &mut Self {
25        self.systems.push(system.into_container());
26        self
27    }
28}
29
30impl Stage for SystemStage {
31    fn run(&mut self, context: &mut MapContext) -> StageResult {
32        for container in &mut self.systems {
33            #[cfg(feature = "trace")]
34            let _span =
35                tracing::info_span!("system", name = container.system.name().as_ref()).entered();
36            let started = instant::Instant::now();
37            let heap_before = live_bytes();
38            container.system.run(context)?;
39            let spent = started.elapsed();
40            let grown = live_bytes() - heap_before;
41            let timings = context.world.resources.get_or_init_mut::<FrameTimings>();
42            timings.record(container.system.name(), spent);
43            timings.record_growth(container.system.name(), grown);
44        }
45        Ok(())
46    }
47}
