38/// Width of the entire Mercator world in pixels at zoom zero.
39pub const TILE_SIZE: f64 = 512.0;
40/// Number of discrete zoom levels represented by [`ZOOM_BOUNDS`], not an inclusive zoom limit.
41pub const MAX_ZOOM: usize = 32;
42
43/// Number of tile columns or rows at each supported grid level: `2^z` for `z` in `0..MAX_ZOOM`.
44pub const ZOOM_BOUNDS: [u32; MAX_ZOOM] = create_zoom_bounds::<MAX_ZOOM>();
45
46const fn create_zoom_bounds<const DIM: usize>() -> [u32; DIM] {
47    let mut result: [u32; DIM] = [0; DIM];
48    let mut i = 0;
49    while i < DIM {
50        result[i] = 2u32.pow(i as u32);
51        i += 1;
52    }
53    result
54}

56/// Ordered tile key containing its zoom level and the quadrant digit for each coordinate bit pair.
57#[derive(Ord, PartialOrd, Eq, PartialEq, Hash, Clone, Copy)]
58pub struct Quadkey(pub(crate) u128);
59
60impl Quadkey {
61    /// Builds a key from quadrant digits, starting with the least-significant coordinate bits.
62    /// The slice length determines the tile zoom; digit values are stored without validation.
63    ///
64    /// # Panics
65    /// Panics when the slice contains [`MAX_ZOOM`] or more digits.
66    pub fn new(quad_encoded: &[ZoomLevel]) -> Self {
67        assert!(
68            quad_encoded.len() < MAX_ZOOM,
69            "quadkey length {} exceeds MAX_ZOOM - 1",
70            quad_encoded.len()
71        );
72        let mut bits = u128::from(quad_encoded.len() as u8) << 120;
73        for (i, part) in quad_encoded.iter().enumerate() {
74            bits |= u128::from(part.0 & 0b11) << (118 - 2 * i);
75        }
76        Self(bits)
77    }
78}
79
80impl fmt::Debug for Quadkey {
81    fn fmt(&self, f: &mut Formatter<'_>) -> fmt::Result {
82        let level = (self.0 >> 120) as u8;
83        if level > 0 {
84            write!(f, "{:?}", ZoomLevel(level))?;
85            for i in 0..(usize::from(level) - 1) {
86                let digit = ((self.0 >> (118 - 2 * i)) & 0b11) as u8;
87                write!(f, "{:?}", ZoomLevel(digit))?;
88            }
89        }
90        Ok(())
91    }
92}

94/// Exact `2.0_f64.powi(i32::from(z))` via IEEE-754 binary64 exponent construction (`z` in `0..=255`).
95#[inline(always)]
96pub(crate) const fn exp2_u8(z: u8) -> f64 {
97    f64::from_bits((1023_u64 + (z as u64)) << 52)
98}

116#[repr(C)]
117pub struct ZoomLevel(u8);

119impl ZoomLevel {
120    /// Stores a level without range checking; grid-table lookups require `z < MAX_ZOOM`.
121    pub const fn new(z: u8) -> Self {
122        ZoomLevel(z)
123    }
124    /// Whether this level consists of the single zoom-zero world tile.
125    pub fn is_root(self) -> bool {
126        self.0 == 0
127    }
128    /// Exact grid dimension `2.0^z` as `f64` without calling `libm::pow`.
129    #[inline(always)]
130    pub const fn grid_scale(self) -> f64 {
131        exp2_u8(self.0)
132    }
133}

172/// Geographic coordinates in degrees, ordered latitude before longitude.
173pub struct LatLon {
174    /// Latitude in degrees north of the equator; no range is enforced by this type.
175    pub latitude: f64,
176    /// Longitude in degrees east of the prime meridian; no wrapping is performed by this type.
177    pub longitude: f64,
178}
179
180impl LatLon {
181    /// Stores coordinates in degrees without clamping latitude or wrapping longitude.
182    pub fn new(latitude: f64, longitude: f64) -> Self {
183        LatLon {
184            latitude,
185            longitude,
186        }
187    }
188
189    /// Projects this geographic location to normalized Mercator coordinates in `[0..1] x [0..1]`.
190    pub fn to_mercator(&self) -> Point2<f64> {
191        let x = self.longitude / 360.0 + 0.5;
192        let y = (1.0 - self.latitude.to_radians().tan().asinh() / PI) * 0.5;
193        Point2::new(x, y.clamp(0.0, 1.0 - 1e-9))
194    }
195
196    /// Converts normalized Mercator coordinates in `[0..1] x [0..1]` back to geographic coordinates.
197    pub fn from_mercator(mercator: Point2<f64>) -> Self {
198        let lat = (PI * (1.0 - 2.0 * mercator.y)).sinh().atan().to_degrees();
199        let lon = mercator.x * 360.0 - 180.0;
200        LatLon::new(lat, lon)
201    }
202}
203
204impl Default for LatLon {
205    fn default() -> Self {
206        LatLon {
207            latitude: 0.0,
208            longitude: 0.0,
209        }
210    }
211}

213impl Display for LatLon {
214    fn fmt(&self, f: &mut Formatter<'_>) -> fmt::Result {
215        write!(f, "{},{}", self.latitude, self.longitude)
216    }
217}

219/// Continuous map zoom: increasing it by one doubles the world-pixel scale.
220#[derive(Copy, Clone, Debug)]
221pub struct Zoom(f64);
222
223impl Zoom {
224    /// Stores a continuous zoom without validating finiteness or a source's supported range.

275impl Zoom {
276    /// Scale factor `2^(tile.z - self)` from this zoom to the tile's grid level.
277    pub fn scale_to_tile(&self, coords: &WorldTileCoords) -> f64 {
278        2.0_f64.powf(coords.z.0 as f64 - self.0)
279    }
280
281    /// Scale factor `2^(z - self)` from this zoom to a discrete grid level.
282    pub fn scale_to_zoom_level(&self, z: ZoomLevel) -> f64 {
283        2.0_f64.powf(z.0 as f64 - self.0)
284    }
285
286    /// Scale factor `2^(zoom - self)` for positions when changing to the supplied zoom.
287    pub fn scale_delta(&self, zoom: &Zoom) -> f64 {
288        2.0_f64.powf(zoom.0 - self.0)
289    }
290
291    /// Selects a tile level using `floor(zoom + log2(512 / tile_size))`.
292    ///
293    /// `tile_size` is the positive source tile width in pixels. A 256-pixel source selects one
294    /// level higher than a 512-pixel source. Clamped to `0..MAX_ZOOM`, not source min/max zoom.
295    pub fn zoom_level(&self, tile_size: f64) -> ZoomLevel {
296        // TODO: Also support round() instead of floor() here
297        let z = (self.0 + (TILE_SIZE / tile_size).log2())
298            .floor()
299            .clamp(0.0, (MAX_ZOOM - 1) as f64) as u8;
300        ZoomLevel(z)
301    }
302}

312/// Mercator coordinates measured in world pixels at a particular zoom.
313///
314/// # Coordinate System Origin
315///
316/// The origin of the coordinate system is in the upper-left corner.
317#[derive(Clone, Copy, Debug, PartialEq, Default)]
318pub struct WorldCoords {
319    /// Pixel distance east from the Mercator world's western edge at the associated zoom.
320    pub x: f64,
321    /// Pixel distance south from the Mercator world's northern edge at the associated zoom.
322    pub y: f64,
323}
324
325impl WorldCoords {
326    /// Projects degrees to Mercator pixels at `zoom` without clamping latitude or wrapping longitude.
327    /// Callers must supply finite longitude and latitude strictly between -90 and 90 degrees.
328    pub fn from_lat_lon(lat_lon: LatLon, zoom: Zoom) -> WorldCoords {
329        let tile_size = TILE_SIZE * 2.0_f64.powf(zoom.0);
330        // Get x value
331        let x = (lat_lon.longitude + 180.0) * (tile_size / 360.0);
332
333        // Convert from degrees to radians
334        let lat_rad = (lat_lon.latitude * PI) / 180.0;
335
336        // get y value
337        let merc_n = f64::ln(f64::tan((PI / 4.0) + (lat_rad / 2.0)));
338        let y = (tile_size / 2.0) - (tile_size * merc_n / (2.0 * PI));
339
340        WorldCoords { x, y }
341    }
342
343    /// Stores a ground-plane pixel position; the associated zoom remains the caller's responsibility.
344    pub fn at_ground(x: f64, y: f64) -> Self {
345        Self { x, y }
346    }
347
348    /// Converts pixels at `zoom` to tile indices at `z`, flooring fractional indices, so a point
349    /// left of the world lies in tile `-1`. The returned indices are not checked against
350    /// canonical tile bounds.
351    pub fn into_world_tile(self, z: ZoomLevel, zoom: Zoom) -> WorldTileCoords {
352        let tile_scale = zoom.scale_to_zoom_level(z) / TILE_SIZE; // TODO: Deduplicate
353        let x = self.x * tile_scale;
354        let y = self.y * tile_scale;
355
356        WorldTileCoords {
357            x: x.floor() as i32,
358            y: y.floor() as i32,
359            z,
360        }
361    }
362}
