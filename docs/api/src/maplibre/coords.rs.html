28/// Unsigned tile-local extent used when normalizing decoded geometry.
29pub const EXTENT_UINT: u32 = 4096;

37/// Width of the entire Mercator world in pixels at zoom zero.
38pub const TILE_SIZE: f64 = 512.0;
39/// Number of discrete zoom levels represented by [`ZOOM_BOUNDS`], not an inclusive zoom limit.
40pub const MAX_ZOOM: usize = 32;
41
42/// Number of tile columns or rows at each supported grid level: `2^z` for `z` in `0..MAX_ZOOM`.
43pub const ZOOM_BOUNDS: [u32; MAX_ZOOM] = create_zoom_bounds::<MAX_ZOOM>();
44
45const fn create_zoom_bounds<const DIM: usize>() -> [u32; DIM] {
46    let mut result: [u32; DIM] = [0; DIM];
47    let mut i = 0;
48    while i < DIM {
49        result[i] = 2u32.pow(i as u32);
50        i += 1;
51    }
52    result
53}

55/// Ordered tile key containing its zoom level and the quadrant digit for each coordinate bit pair.
56#[derive(Ord, PartialOrd, Eq, PartialEq, Clone, Copy)]
57pub struct Quadkey([ZoomLevel; MAX_ZOOM]);
58
59impl Quadkey {
60    /// Builds a key from quadrant digits, starting with the least-significant coordinate bits.
61    /// The slice length determines the tile zoom; digit values are stored without validation.
62    ///
63    /// # Panics
64    /// Panics when the slice contains [`MAX_ZOOM`] or more digits.
65    pub fn new(quad_encoded: &[ZoomLevel]) -> Self {
66        let mut key = [ZoomLevel::default(); MAX_ZOOM];
67        key[0] = (quad_encoded.len() as u8).into();
68        for (i, part) in quad_encoded.iter().enumerate() {
69            key[i + 1] = *part;
70        }
71        Self(key)
72    }
73}
74
75impl fmt::Debug for Quadkey {
76    fn fmt(&self, f: &mut Formatter<'_>) -> fmt::Result {
77        let key = self.0;
78        let ZoomLevel(level) = key[0];
79        let len = level as usize;
80        for part in &self.0[0..len] {
81            write!(f, "{part:?}")?;
82        }
83        Ok(())
84    }
85}

106impl ZoomLevel {

243impl Zoom {
244    /// Scale factor `2^(tile.z - self)` from this zoom to the tile's grid level.
245    pub fn scale_to_tile(&self, coords: &WorldTileCoords) -> f64 {
246        2.0_f64.powf(coords.z.0 as f64 - self.0)
247    }
248
249    /// Scale factor `2^(z - self)` from this zoom to a discrete grid level.
250    pub fn scale_to_zoom_level(&self, z: ZoomLevel) -> f64 {
251        2.0_f64.powf(z.0 as f64 - self.0)
252    }
253
254    /// Scale factor `2^(zoom - self)` for positions when changing to the supplied zoom.
255    pub fn scale_delta(&self, zoom: &Zoom) -> f64 {
256        2.0_f64.powf(zoom.0 - self.0)
257    }
258
259    /// Selects a tile level using `floor(zoom + log2(512 / tile_size))`.
260    ///
261    /// `tile_size` is the positive source tile width in pixels. A 256-pixel source selects one
262    /// level higher than a 512-pixel source. The result is not clamped to source min/max zoom.
263    pub fn zoom_level(&self, tile_size: f64) -> ZoomLevel {
264        // TODO: Also support round() instead of floor() here
265        let z = (self.0 + (TILE_SIZE / tile_size).ln() / 2.0_f64.ln()).floor() as u8;
266        ZoomLevel(z)
267    }
268}

278/// Mercator coordinates measured in world pixels at a particular zoom.
279///
280/// # Coordinate System Origin
281///
282/// The origin of the coordinate system is in the upper-left corner.
283#[derive(Clone, Copy, Debug, PartialEq, Default)]
284pub struct WorldCoords {
285    /// Pixel distance east from the Mercator world's western edge at the associated zoom.
286    pub x: f64,
287    /// Pixel distance south from the Mercator world's northern edge at the associated zoom.
288    pub y: f64,
289}
290
291impl WorldCoords {
292    /// Projects degrees to Mercator pixels at `zoom` without clamping latitude or wrapping longitude.
293    /// Callers must supply finite longitude and latitude strictly between -90 and 90 degrees.
294    pub fn from_lat_lon(lat_lon: LatLon, zoom: Zoom) -> WorldCoords {
295        let tile_size = TILE_SIZE * 2.0_f64.powf(zoom.0);
296        // Get x value
297        let x = (lat_lon.longitude + 180.0) * (tile_size / 360.0);
298
299        // Convert from degrees to radians
300        let lat_rad = (lat_lon.latitude * PI) / 180.0;
301
302        // get y value
303        let merc_n = f64::ln(f64::tan((PI / 4.0) + (lat_rad / 2.0)));
304        let y = (tile_size / 2.0) - (tile_size * merc_n / (2.0 * PI));
305
306        WorldCoords { x, y }
307    }
308
309    /// Stores a ground-plane pixel position; the associated zoom remains the caller's responsibility.
310    pub fn at_ground(x: f64, y: f64) -> Self {
311        Self { x, y }
312    }
313
314    /// Converts pixels at `zoom` to tile indices at `z`, flooring fractional indices, so a point
315    /// left of the world lies in tile `-1`. The returned indices are not checked against
316    /// canonical tile bounds.
317    pub fn into_world_tile(self, z: ZoomLevel, zoom: Zoom) -> WorldTileCoords {
318        let tile_scale = zoom.scale_to_zoom_level(z) / TILE_SIZE; // TODO: Deduplicate
319        let x = self.x * tile_scale;
320        let y = self.y * tile_scale;
321
322        WorldTileCoords {
323            x: x.floor() as i32,
324            y: y.floor() as i32,
325            z,
326        }
327    }
328}

357/// Defines a bounding box on a tiled map with a [`ZoomLevel`] and a padding.
358#[derive(Debug)]
359pub struct ViewRegion {
360    min_tile: WorldTileCoords,
361    max_tile: WorldTileCoords,
362    explicit_tiles: Option<Vec<WorldTileCoords>>,
363    /// At which zoom level does this region exist
364    zoom_level: ZoomLevel,
365    /// Padding around this view region
366    padding: i32,
367    /// The maximum amount of tiles this view region contains
368    max_n_tiles: usize,
369}
370
371impl ViewRegion {
372    /// Covers a world-pixel box at `zoom` with tiles at `z` and an integer tile padding.
373    /// `max_n_tiles` limits iteration, not the membership test in [`Self::is_in_view`].
374    pub fn new(
375        view_region: Aabb2<f64>,
376        padding: i32,
377        max_n_tiles: usize,
378        zoom: Zoom,
379        z: ZoomLevel,
380    ) -> Self {
381        let min_world: WorldCoords = WorldCoords::at_ground(view_region.min.x, view_region.min.y);
382        let min_world_tile: WorldTileCoords = min_world.into_world_tile(z, zoom);
383        let max_world: WorldCoords = WorldCoords::at_ground(view_region.max.x, view_region.max.y);
384        let max_world_tile: WorldTileCoords = max_world.into_world_tile(z, zoom);
385
386        Self {
387            min_tile: min_world_tile,
388            max_tile: max_world_tile,
389            explicit_tiles: None,
390            zoom_level: z,
391            max_n_tiles,
392            padding,
393        }
394    }
395
396    /// Creates a region from an explicit projection-aware tile selection.
397    pub fn from_tiles(
398        tiles: Vec<WorldTileCoords>,
399        zoom_level: ZoomLevel,
400        max_n_tiles: usize,
401    ) -> Self {
402        Self {
403            min_tile: WorldTileCoords::default(),
404            max_tile: WorldTileCoords::default(),
405            explicit_tiles: Some(tiles),
406            zoom_level,
407            max_n_tiles,
408            padding: 0,
409        }
410    }
411
412    /// Returns the grid level supplied when constructing this region.
413    pub fn zoom_level(&self) -> ZoomLevel {
414        self.zoom_level
415    }
416
417    /// Tests the padded bounds or explicit selection without applying the iteration limit.
418    pub fn is_in_view(&self, &world_coords: &WorldTileCoords) -> bool {
419        if let Some(tiles) = &self.explicit_tiles {
420            return tiles.contains(&world_coords);
421        }
422        world_coords.x <= self.max_tile.x + self.padding
423            && world_coords.y <= self.max_tile.y + self.padding
424            && world_coords.x >= self.min_tile.x - self.padding
425            && world_coords.y >= self.min_tile.y - self.padding
426            && world_coords.z == self.zoom_level
427    }
428
429    /// The tiles a view sees, with those seen in copies of the world as the tiles they repeat,
430    /// each once: up to the configured limit, in explicit selection order or column-major
431    /// grid order.
432    pub fn iter(&self) -> Box<dyn Iterator<Item = WorldTileCoords> + '_> {
433        let mut seen = std::collections::HashSet::new();
434        Box::new(
435            self.copies()
436                .map(|coords| coords.wrapped().map_or(coords, |(tile, _)| tile))
437                .filter(move |tile| seen.insert(*tile)),
438        )
439    }
440
441    /// The tiles of the view as drawn, a tile seen again in another copy of the world once more
442    /// at coordinates beyond the grid.
443    pub fn copies(&self) -> Box<dyn Iterator<Item = WorldTileCoords> + '_> {
444        if let Some(tiles) = &self.explicit_tiles {
445            return Box::new(tiles.iter().copied().take(self.max_n_tiles));
446        }
447        Box::new(
448            (self.min_tile.x - self.padding..self.max_tile.x + 1 + self.padding)
449                .flat_map(move |x| {
450                    (self.min_tile.y - self.padding..self.max_tile.y + 1 + self.padding)
451                        .map(move |y| (x, y, self.zoom_level).into())
452                })
453                .take(self.max_n_tiles),
454        )
455    }
456}
