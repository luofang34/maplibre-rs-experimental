27/// Unsigned tile-local extent used when normalizing decoded geometry.
28pub const EXTENT_UINT: u32 = 4096;
29/// Signed tile-local extent for geometry that can extend beyond a tile boundary.

36/// Width of the entire Mercator world in pixels at zoom zero.
37pub const TILE_SIZE: f64 = 512.0;
38/// Number of discrete zoom levels represented by [`ZOOM_BOUNDS`], not an inclusive zoom limit.
39pub const MAX_ZOOM: usize = 32;
40
41/// Number of tile columns or rows at each supported grid level: `2^z` for `z` in `0..MAX_ZOOM`.
42pub const ZOOM_BOUNDS: [u32; MAX_ZOOM] = create_zoom_bounds::<MAX_ZOOM>();
43
44const fn create_zoom_bounds<const DIM: usize>() -> [u32; DIM] {
45    let mut result: [u32; DIM] = [0; DIM];
46    let mut i = 0;
47    while i < DIM {
48        result[i] = 2u32.pow(i as u32);
49        i += 1;
50    }
51    result
52}

54/// Ordered tile key containing its zoom level and the quadrant digit for each coordinate bit pair.
55#[derive(Ord, PartialOrd, Eq, PartialEq, Clone, Copy)]
56pub struct Quadkey([ZoomLevel; MAX_ZOOM]);
57
58impl Quadkey {
59    /// Builds a key from quadrant digits, starting with the least-significant coordinate bits.
60    /// The slice length determines the tile zoom; digit values are stored without validation.
61    ///
62    /// # Panics
63    /// Panics when the slice contains [`MAX_ZOOM`] or more digits.
64    pub fn new(quad_encoded: &[ZoomLevel]) -> Self {
65        let mut key = [ZoomLevel::default(); MAX_ZOOM];
66        key[0] = (quad_encoded.len() as u8).into();
67        for (i, part) in quad_encoded.iter().enumerate() {
68            key[i + 1] = *part;
69        }
70        Self(key)
71    }
72}
73
74impl fmt::Debug for Quadkey {
75    fn fmt(&self, f: &mut Formatter<'_>) -> fmt::Result {
76        let key = self.0;
77        let ZoomLevel(level) = key[0];
78        let len = level as usize;
79        for part in &self.0[0..len] {
80            write!(f, "{part:?}")?;
81        }
82        Ok(())
83    }
84}

102#[repr(C)]
103pub struct ZoomLevel(u8);
104
105impl ZoomLevel {
106    /// Stores a level without range checking; grid-table lookups require `z < MAX_ZOOM`.
107    pub const fn new(z: u8) -> Self {
108        ZoomLevel(z)
109    }
110    /// Whether this level consists of the single zoom-zero world tile.
111    pub fn is_root(self) -> bool {
112        self.0 == 0
113    }
114}

186/// Continuous map zoom: increasing it by one doubles the world-pixel scale.
187#[derive(Copy, Clone, Debug)]
188pub struct Zoom(f64);
189
190impl Zoom {
191    /// Stores a continuous zoom without validating finiteness or a source's supported range.
192    pub fn new(zoom: f64) -> Self {
193        Zoom(zoom)
194    }
195

198        self.0
199    }
200
201    /// Returns the continuous zoom at GPU precision; it does not round to a tile-grid level.
202    pub fn level(&self) -> f32 {

205}

242impl Zoom {
243    /// Scale factor `2^(tile.z - self)` from this zoom to the tile's grid level.
244    pub fn scale_to_tile(&self, coords: &WorldTileCoords) -> f64 {
245        2.0_f64.powf(coords.z.0 as f64 - self.0)
246    }
247
248    /// Scale factor `2^(z - self)` from this zoom to a discrete grid level.
249    pub fn scale_to_zoom_level(&self, z: ZoomLevel) -> f64 {
250        2.0_f64.powf(z.0 as f64 - self.0)
251    }
252
253    /// Scale factor `2^(zoom - self)` for positions when changing to the supplied zoom.
254    pub fn scale_delta(&self, zoom: &Zoom) -> f64 {
255        2.0_f64.powf(zoom.0 - self.0)
256    }
257
258    /// Selects a tile level using `floor(zoom + log2(512 / tile_size))`.
259    ///
260    /// `tile_size` is the positive source tile width in pixels. A 256-pixel source selects one
261    /// level higher than a 512-pixel source. The result is not clamped to source min/max zoom.
262    pub fn zoom_level(&self, tile_size: f64) -> ZoomLevel {
263        // TODO: Also support round() instead of floor() here
264        let z = (self.0 + (TILE_SIZE / tile_size).ln() / 2.0_f64.ln()).floor() as u8;
265        ZoomLevel(z)
266    }
267}

277/// Mercator coordinates measured in world pixels at a particular zoom.
278///
279/// # Coordinate System Origin
280///
281/// The origin of the coordinate system is in the upper-left corner.
282#[derive(Clone, Copy, Debug, PartialEq, Default)]
283pub struct WorldCoords {
284    /// Pixel distance east from the Mercator world's western edge at the associated zoom.
285    pub x: f64,
286    /// Pixel distance south from the Mercator world's northern edge at the associated zoom.
287    pub y: f64,
288}
289
290impl WorldCoords {
291    /// Projects degrees to Mercator pixels at `zoom` without clamping latitude or wrapping longitude.
292    /// Callers must supply finite longitude and latitude strictly between -90 and 90 degrees.
293    pub fn from_lat_lon(lat_lon: LatLon, zoom: Zoom) -> WorldCoords {
294        let tile_size = TILE_SIZE * 2.0_f64.powf(zoom.0);
295        // Get x value
296        let x = (lat_lon.longitude + 180.0) * (tile_size / 360.0);
297
298        // Convert from degrees to radians
299        let lat_rad = (lat_lon.latitude * PI) / 180.0;
300
301        // get y value
302        let merc_n = f64::ln(f64::tan((PI / 4.0) + (lat_rad / 2.0)));
303        let y = (tile_size / 2.0) - (tile_size * merc_n / (2.0 * PI));
304
305        WorldCoords { x, y }
306    }
307
308    /// Stores a ground-plane pixel position; the associated zoom remains the caller's responsibility.
309    pub fn at_ground(x: f64, y: f64) -> Self {
310        Self { x, y }
311    }
312
313    /// Converts pixels at `zoom` to tile indices at `z`, flooring fractional indices, so a point
314    /// left of the world lies in tile `-1`. The returned indices are not checked against
315    /// canonical tile bounds.
316    pub fn into_world_tile(self, z: ZoomLevel, zoom: Zoom) -> WorldTileCoords {
317        let tile_scale = zoom.scale_to_zoom_level(z) / TILE_SIZE; // TODO: Deduplicate
318        let x = self.x * tile_scale;
319        let y = self.y * tile_scale;
320
321        WorldTileCoords {
322            x: x.floor() as i32,
323            y: y.floor() as i32,
324            z,
325        }
326    }
327}

334        }
335    }

356/// Defines a bounding box on a tiled map with a [`ZoomLevel`] and a padding.
357#[derive(Debug)]
358pub struct ViewRegion {
359    min_tile: WorldTileCoords,
360    max_tile: WorldTileCoords,
361    explicit_tiles: Option<Vec<WorldTileCoords>>,
362    /// At which zoom level does this region exist
363    zoom_level: ZoomLevel,
364    /// Padding around this view region
365    padding: i32,
366    /// The maximum amount of tiles this view region contains
367    max_n_tiles: usize,
368}
369
370impl ViewRegion {
371    /// Covers a world-pixel box at `zoom` with tiles at `z` and an integer tile padding.
372    /// `max_n_tiles` limits iteration, not the membership test in [`Self::is_in_view`].
373    pub fn new(
374        view_region: Aabb2<f64>,
375        padding: i32,
376        max_n_tiles: usize,
377        zoom: Zoom,
378        z: ZoomLevel,
379    ) -> Self {
380        let min_world: WorldCoords = WorldCoords::at_ground(view_region.min.x, view_region.min.y);
381        let min_world_tile: WorldTileCoords = min_world.into_world_tile(z, zoom);
382        let max_world: WorldCoords = WorldCoords::at_ground(view_region.max.x, view_region.max.y);
383        let max_world_tile: WorldTileCoords = max_world.into_world_tile(z, zoom);
384
385        Self {
386            min_tile: min_world_tile,
387            max_tile: max_world_tile,
388            explicit_tiles: None,
389            zoom_level: z,
390            max_n_tiles,
391            padding,
392        }
393    }
394
395    /// Creates a region from an explicit projection-aware tile selection.
396    pub fn from_tiles(
397        tiles: Vec<WorldTileCoords>,
398        zoom_level: ZoomLevel,
399        max_n_tiles: usize,
400    ) -> Self {
401        Self {
402            min_tile: WorldTileCoords::default(),
403            max_tile: WorldTileCoords::default(),
404            explicit_tiles: Some(tiles),
405            zoom_level,
406            max_n_tiles,
407            padding: 0,
408        }
409    }
410
411    /// Returns the grid level supplied when constructing this region.
412    pub fn zoom_level(&self) -> ZoomLevel {
413        self.zoom_level
414    }
415
416    /// Tests the padded bounds or explicit selection without applying the iteration limit.
417    pub fn is_in_view(&self, &world_coords: &WorldTileCoords) -> bool {
418        if let Some(tiles) = &self.explicit_tiles {
419            return tiles.contains(&world_coords);
420        }
421        world_coords.x <= self.max_tile.x + self.padding
422            && world_coords.y <= self.max_tile.y + self.padding
423            && world_coords.x >= self.min_tile.x - self.padding
424            && world_coords.y >= self.min_tile.y - self.padding
425            && world_coords.z == self.zoom_level
426    }
427
428    /// The tiles a view sees, with those seen in copies of the world as the tiles they repeat,
429    /// each once: up to the configured limit, in explicit selection order or column-major
430    /// grid order.
431    pub fn iter(&self) -> Box<dyn Iterator<Item = WorldTileCoords> + '_> {
432        let mut seen = std::collections::HashSet::new();
433        Box::new(
434            self.copies()
435                .map(|coords| coords.wrapped().map_or(coords, |(tile, _)| tile))
436                .filter(move |tile| seen.insert(*tile)),
437        )
438    }
439
440    /// The tiles of the view as drawn, a tile seen again in another copy of the world once more
441    /// at coordinates beyond the grid.
442    pub fn copies(&self) -> Box<dyn Iterator<Item = WorldTileCoords> + '_> {
443        if let Some(tiles) = &self.explicit_tiles {
444            return Box::new(tiles.iter().copied().take(self.max_n_tiles));
445        }
446        Box::new(
447            (self.min_tile.x - self.padding..self.max_tile.x + 1 + self.padding)
448                .flat_map(move |x| {
449                    (self.min_tile.y - self.padding..self.max_tile.y + 1 + self.padding)
450                        .map(move |y| (x, y, self.zoom_level).into())
451                })
452                .take(self.max_n_tiles),
453        )
454    }
455}
