16/// Initializes missing pipelines using the surface format and supported multisampling.
17/// Returns `Dependencies` unless all pipeline slots and initialized projection resources
18/// are present. Already initialized pipelines are retained.
19pub fn resource_system(
20    MapContext {
21        world,
22        style,
23        view_state,
24        renderer:
25            crate::render::Renderer {
26                device,
27                queue,
28                resources: crate::render::RenderResources { surface, .. },
29                settings,
30                ..
31            },
32        ..
33    }: &mut MapContext,
34) -> crate::tcs::system::SystemResult {
35    let style_zoom = view_state.style_zoom().value();
36    let needs_bg_pattern = style.layers.iter().any(|layer| {
37        layer.type_ == "background"
38            && layer
39                .paint
40                .as_ref()
41                .is_some_and(|p| crate::vector::pattern::pattern_name(p, style_zoom).is_some())
42    });
43    let needs_globe = style.projection.as_ref().is_some_and(|spec| {
44        spec.projection_type
45            .globe_transition(view_state.zoom().value())
46            > 0.0
47    });
48    let needs_sky = style.sky.is_some();
49    if needs_bg_pattern
50        && world
51            .resources
52            .get::<super::pattern::BackgroundPatternGpu>()
53            .is_none()
54    {
55        let shader = BackgroundPatternShader {
56            format: surface.surface_format(),
57        };
58        let layouts = super::pattern::layouts(device);
59        let pipeline = TilePipeline::new(
60            "background_pattern_pipeline".into(),
61            *settings,
62            shader.describe_vertex(),
63            shader.describe_fragment(),
64            crate::render::resource::TilePipelineOptions {
65                depth_stencil_enabled: true,
66                update_stencil: false,
67                debug_stencil: true,
68                wireframe: false,
69                multisampling: surface.is_multisampling_supported(settings.msaa),
70                textured: false,
71            },
72        )
73        .describe_render_pipeline()
74        .initialize_with_prefix_layouts(device, &[&layouts[0], &layouts[1]]);
75        if world
76            .resources
77            .get::<crate::vector::pattern::PatternResources>()
78            .is_none()
79        {
80            world
81                .resources
82                .insert(crate::vector::pattern::PatternResources::new(
83                    device,
84                    layouts[0].clone(),
85                ));
86        }
87        world
88            .resources
89            .insert(super::pattern::BackgroundPatternGpu::new(
90                device,
91                queue,
92                pipeline,
93                &layouts[1],
94            ));
95    }
96    if let Some(patterns) = world
97        .resources
98        .get_mut::<crate::vector::pattern::PatternResources>()
99    {
100        patterns.update(device, queue, style, view_state.style_zoom().value());
101    }
102    if needs_globe
103        && world
104            .resources
105            .get::<super::pattern::BackgroundPatternGpu>()
106            .is_some_and(|gpu| !gpu.has_globe())
107    {
108        let projection_layout = match world.resources.get::<Eventually<ProjectionGpuResources>>() {
109            Some(Initialized(resources)) => Some(resources.bind_group_layout().clone()),
110            _ => None,
111        };
112        if let Some(projection_layout) = projection_layout {
113            let shader = GlobeBackgroundPatternShader {
114                format: surface.surface_format(),
115            };
116            let image_layouts = super::pattern::layouts(device);
117            let world_layout = super::pattern::world_layout(device);
118            let mut descriptor = TilePipeline::new(
119                "globe_background_pattern_pipeline".into(),
120                *settings,
121                shader.describe_vertex(),
122                shader.describe_fragment(),
123                crate::render::resource::TilePipelineOptions {
124                    depth_stencil_enabled: true,
125                    update_stencil: false,
126                    debug_stencil: true,
127                    wireframe: false,
128                    multisampling: surface.is_multisampling_supported(settings.msaa),
129                    textured: false,
130                },
131            )
132            .with_depth_write()
133            .describe_render_pipeline();
134            descriptor.primitive.cull_mode = Some(wgpu::Face::Back);
135            let pipeline = descriptor.initialize_with_prefix_layouts(
136                device,
137                &[&projection_layout, &image_layouts[0], &world_layout],
138            );
139            if let Some(gpu) = world
140                .resources
141                .get_mut::<super::pattern::BackgroundPatternGpu>()
142            {
143                gpu.with_globe(device, queue, pipeline, &world_layout);
144            }
145        }
146    }
147    let needs_atmosphere = super::queue_system::atmosphere_blend(style, view_state, world) > 0.0;
148    let Some((
149        background_pipeline,
150        globe_background_pipeline,
151        atmosphere_pipeline,
152        sky_pipeline,
153        Initialized(projection_resources),
154    )) = world.resources.query_mut::<(
155        &mut Eventually<BackgroundRenderPipeline>,
156        &mut Eventually<GlobeBackgroundRenderPipeline>,
157        &mut Eventually<AtmosphereRenderPipeline>,
158        &mut Eventually<SkyRenderPipeline>,
159        &mut Eventually<ProjectionGpuResources>,
160    )>()
161    else {
162        return Err(crate::tcs::system::SystemError::Dependencies);
163    };
164
165    let zoom = view_state.zoom().value();
166    let (mut has_opaque_bg, mut has_blended_bg) = (false, false);
167    for layer in style
168        .layers
169        .iter()
170        .filter(|layer| layer.type_ == "background" && layer.is_visible_at(zoom))
171    {
172        if layer
173            .paint
174            .as_ref()
175            .is_some_and(|p| crate::vector::pattern::pattern_name(p, style_zoom).is_some())
176        {
177            continue;
178        }
179        if super::buffers::background_color(layer, zoom)[3] >= 1.0 {
180            has_opaque_bg = true;
181        } else {
182            has_blended_bg = true;
183        }
184    }
185
186    let bg_pipelines = background_pipeline.initialize(BackgroundRenderPipeline::default);
187    if !needs_globe {
188        let size = surface.size();
189        let covers = super::queue_system::flat_viewport_covered(
190            view_state,
191            style.terrain.is_none(),
192            [size.width() as f32, size.height() as f32],
193        );
194        let shader = BackgroundShader {
195            format: surface.surface_format(),
196        };
197        let options = crate::render::resource::TilePipelineOptions {
198            depth_stencil_enabled: true,
199            update_stencil: false,
200            debug_stencil: true,
201            wireframe: false,
202            multisampling: surface.is_multisampling_supported(settings.msaa),
203            textured: false,
204        };
205        if covers {
206            if has_opaque_bg {
207                bg_pipelines.solid_opaque.get_or_insert_with(|| {
208                    TilePipeline::new(
209                        "background_solid_opaque_pipeline".into(),
210                        *settings,
211                        shader.describe_vertex(),
212                        shader.describe_solid_opaque_fragment(),
213                        options,
214                    )
215                    .describe_render_pipeline()
216                    .initialize(device)
217                });
218            }
219            if has_blended_bg {
220                bg_pipelines.solid.get_or_insert_with(|| {
221                    TilePipeline::new(
222                        "background_solid_pipeline".into(),
223                        *settings,
224                        shader.describe_vertex(),
225                        shader.describe_solid_fragment(),
226                        options,
227                    )
228                    .describe_render_pipeline()
229                    .initialize(device)
230                });
231            }
232        } else {
233            if has_opaque_bg {
234                bg_pipelines.clipped_opaque.get_or_insert_with(|| {
235                    TilePipeline::new(
236                        "background_clipped_opaque_pipeline".into(),
237                        *settings,
238                        shader.describe_vertex(),
239                        shader.describe_clipped_opaque_fragment(),
240                        options,
241                    )
242                    .describe_render_pipeline()
243                    .initialize(device)
244                });
245            }
246            if has_blended_bg {
247                bg_pipelines.clipped.get_or_insert_with(|| {
248                    TilePipeline::new(
249                        "background_pipeline".into(),
250                        *settings,
251                        shader.describe_vertex(),
252                        shader.describe_fragment(),
253                        options,
254                    )
255                    .describe_render_pipeline()
256                    .initialize(device)
257                });
258            }
259        }
260    }
261
262    if needs_globe {
263        let globe_bg = globe_background_pipeline.initialize(GlobeBackgroundRenderPipeline::default);
264        let shader = GlobeBackgroundShader {
265            format: surface.surface_format(),
266        };
267        let options = crate::render::resource::TilePipelineOptions {
268            depth_stencil_enabled: true,
269            update_stencil: false,
270            debug_stencil: true,
271            wireframe: false,
272            multisampling: surface.is_multisampling_supported(settings.msaa),
273            textured: false,
274        };
275        if has_opaque_bg {
276            globe_bg.opaque.get_or_insert_with(|| {
277                let mut descriptor = TilePipeline::new(
278                    "globe_background_opaque_pipeline".into(),
279                    *settings,
280                    shader.describe_vertex(),
281                    shader.describe_opaque_fragment(),
282                    options,
283                )
284                .with_depth_write()
285                .describe_render_pipeline();
286                descriptor.primitive.cull_mode = Some(wgpu::Face::Back);
287                descriptor.initialize_with_prefix_layouts(
288                    device,
289                    &[projection_resources.bind_group_layout()],
290                )
291            });
292        }
293        if has_blended_bg {
294            globe_bg.blended.get_or_insert_with(|| {
295                let mut descriptor = TilePipeline::new(
296                    "globe_background_pipeline".into(),
297                    *settings,
298                    shader.describe_vertex(),
299                    shader.describe_fragment(),
300                    options,
301                )
302                .with_depth_write()
303                .describe_render_pipeline();
304                descriptor.primitive.cull_mode = Some(wgpu::Face::Back);
305                descriptor.initialize_with_prefix_layouts(
306                    device,
307                    &[projection_resources.bind_group_layout()],
308                )
309            });
310        }
311
312        if needs_atmosphere {
313            atmosphere_pipeline.initialize(|| {
314                let shader = AtmosphereShader {
315                    format: surface.surface_format(),
316                };
317                let pipeline = TilePipeline::new(
318                    "atmosphere_pipeline".into(),
319                    *settings,
320                    shader.describe_vertex(),
321                    shader.describe_fragment(),
322                    crate::render::resource::TilePipelineOptions {
323                        depth_stencil_enabled: true,
324                        update_stencil: false,
325                        debug_stencil: true,
326                        wireframe: false,
327                        multisampling: surface.is_multisampling_supported(settings.msaa),
328                        textured: false,
329                    },
330                )
331                // The scattering lies behind the globe, its terrain and everything else that writes
332                // depth, as GL JS draws it with a read-only depth test.
333                .with_depth_test()
334                .describe_render_pipeline()
335                .initialize_with_prefix_layouts(
336                    device,
337                    &[projection_resources.bind_group_layout()],
338                );
339                AtmosphereRenderPipeline(pipeline)
340            });
341        }
342    }
343
344    if needs_sky {
345        sky_pipeline.initialize(|| {
346            let shader = SkyShader {
347                format: surface.surface_format(),
348            };
349            // The host compositor needs depth for the opaque sky as well as terrain.
350            let pipeline = TilePipeline::new(
351                "sky_pipeline".into(),
352                *settings,
353                shader.describe_vertex(),
354                shader.describe_fragment(),
355                crate::render::resource::TilePipelineOptions {
356                    depth_stencil_enabled: true,
357                    update_stencil: false,
358                    debug_stencil: true,
359                    wireframe: false,
360                    multisampling: surface.is_multisampling_supported(settings.msaa),
361                    textured: false,
362                },
363            )
364            .with_depth_write()
365            .describe_render_pipeline()
366            .initialize_with_prefix_layouts(device, &[projection_resources.bind_group_layout()]);
367            SkyRenderPipeline(pipeline)
368        });
369    }
370
371    Ok(())
372}

374/// Pipelines drawing a flat fullscreen background (`solid` omits fragment `discard`).
375#[derive(Default)]
376pub struct BackgroundRenderPipeline {
377    /// Horizon/world-edge-clipped opaque (`blend: None`) background pipeline.
378    pub clipped_opaque: Option<wgpu::RenderPipeline>,
379    /// Horizon/world-edge-clipped background pipeline for tilted or partial flat views.
380    pub clipped: Option<wgpu::RenderPipeline>,
381    /// Zero-`discard` opaque (`blend: None`) background pipeline when the flat world fills viewport.
382    pub solid_opaque: Option<wgpu::RenderPipeline>,
383    /// Zero-`discard` background pipeline when the flat world covers the entire viewport.
384    pub solid: Option<wgpu::RenderPipeline>,
385}
386
387/// Pipelines drawing background paint on a globe mesh (`opaque` skips color blending).
388#[derive(Default)]
389pub struct GlobeBackgroundRenderPipeline {
390    /// Opaque (`blend: None`) globe background pipeline for alpha `1.0` backgrounds.
391    pub opaque: Option<wgpu::RenderPipeline>,
392    /// Premultiplied-alpha blended globe background pipeline.
393    pub blended: Option<wgpu::RenderPipeline>,
394}
395
396/// Pipeline drawing optional atmospheric scattering.
397pub struct AtmosphereRenderPipeline(pub wgpu::RenderPipeline);
398
399/// Pipeline filling the screen above the horizon with the sky.
400pub struct SkyRenderPipeline(pub wgpu::RenderPipeline);
