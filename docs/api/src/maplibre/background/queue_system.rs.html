22/// How strongly the atmosphere is drawn this frame: as the style's `sky` blends it, or in full
23/// where the host turned it on, faded with the projection's globe transition and camera altitude below 100 km.
24pub fn atmosphere_blend(
25    style: &crate::style::Style,
26    view_state: &crate::render::view_state::ViewState,
27    world: &crate::tcs::world::World,
28) -> f32 {
29    let zoom = view_state.zoom().value();
30    let transition = style.projection.as_ref().map_or(0.0, |specification| {
31        specification.projection_type.globe_transition(zoom)
32    });
33    let sky_blend = style
34        .sky
35        .as_ref()
36        .map_or(0.0, |sky| sky.atmosphere_blend_at_zoom(zoom));
37    let host_blend = f32::from(u8::from(
38        world
39            .resources
40            .get::<crate::render::SceneLighting>()
41            .is_some_and(|l| l.atmosphere),
42    ));
43    let sky_faded = if sky_blend > 0.0 {
44        let altitude_blend = globe_camera_for_view(view_state)
45            .map_or(0.0, |camera| super::altitude::altitude_blend_for_camera(&camera));
46        sky_blend * altitude_blend
47    } else {
48        0.0
49    };
50    sky_faded.max(host_blend) * transition
51}

53/// Appends visible background, sky and atmosphere items and uploads their frame metadata.
54/// Returns `Dependencies` if the required phases are absent, or `Setup` if atmosphere
55/// camera or shader metadata cannot be constructed. Existing phase items are not cleared.
56pub fn queue_system(
57    MapContext {
58        world,
59        style,
60        view_state,
61        renderer,
62        ..
63    }: &mut MapContext,
64) -> SystemResult {
65    let mut metadatas = Vec::new();
66    let frame = {
67        let size = renderer.state().surface().size();
68        [size.width() as f32, size.height() as f32]
69    };
70    let world_rows = world_rows(view_state, style.terrain.is_none());
71    let mut instances = std::collections::HashMap::new();
72    let projection_transition = style.projection.as_ref().map_or(0.0, |specification| {
73        specification
74            .projection_type
75            .globe_transition(view_state.zoom().value())
76    });
77    let uses_globe = projection_transition > 0.0;
78    let atmosphere_blend = atmosphere_blend(style, view_state, world);
79    let sky_metadata = style
80        .sky
81        .as_ref()
82        .filter(|_| projection_transition < 1.0 || view_state.opaque_environment())
83        .map(|sky| sky_metadata(sky, view_state, projection_transition));
84
85    if let Some(gpu) = world
86        .resources
87        .get::<super::pattern::BackgroundPatternGpu>()
88    {
89        let size = renderer.state().surface().size();
90        gpu.write(
91            &renderer.queue,
92            view_state,
93            [size.width() as f32, size.height() as f32],
94        );
95    }
96    {
97        let Some((layer_item_phase, translucent_phase)) = world.resources.query_mut::<(
98            &mut RenderPhase<LayerItem>,
99            &mut RenderPhase<TranslucentItem>,
100        )>() else {
101            return Err(SystemError::Dependencies);
102        };
103
104        let patterned: std::collections::HashSet<&str> = style
105            .layers
106            .iter()
107            .filter(|layer| {
108                layer.paint.as_ref().is_some_and(|paint| {
109                    crate::vector::pattern::pattern_name(paint, view_state.style_zoom().value())
110                        .is_some()
111                })
112            })
113            .map(|layer| layer.id.as_str())
114            .collect();
115        let mut background_index = 0;
116        for layer in &style.layers {
117            if layer.type_ != "background" || !layer.is_visible_at(view_state.zoom().value()) {
118                continue;
119            }
120            // Under terrain the drape of each terrain tile carries the background, so nothing
121            // paints outside the world.
122            if style.terrain.is_some() && !uses_globe {
123                continue;
124            }
125            // A layer that names an image it does not have draws nothing, not its colour.
126            if crate::vector::pattern::names_missing_image(
127                layer.paint.as_ref(),
128                style,
129                view_state.style_zoom().value(),
130            ) {
131                continue;
132            }
133            background_index = background_index.max(layer.index);
134            let c = background_color(layer, view_state.zoom().value());
135            let z_index = layer.index as f32;
136            instances.insert(layer.id.clone(), metadatas.len() as u32);
137            metadatas.push(BackgroundLayerMetadata {
138                color: c,
139                z_index,
140                padding: [0.0; 3],
141                horizon: view_state.horizon_line().to_shader(),
142                viewport: [
143                    view_state.height() as f32,
144                    f32::from(style.terrain.is_some()),
145                    frame[0],
146                    frame[1],
147                ],
148                world_rows,
149            });
150
151            let draw_function: Box<dyn crate::render::render_phase::Draw<LayerItem>> =
152                if uses_globe && patterned.contains(layer.id.as_str()) {
153                    Box::new(DrawState::<LayerItem, DrawGlobeBackgroundPattern>::new())
154                } else if uses_globe {
155                    Box::new(DrawState::<LayerItem, DrawGlobeBackground>::new())
156                } else if patterned.contains(layer.id.as_str()) {
157                    Box::new(DrawState::<LayerItem, DrawBackgroundPattern>::new())
158                } else {
159                    Box::new(DrawState::<LayerItem, DrawBackground>::new())
160                };
161            layer_item_phase.add(LayerItem {
162                projection: ProjectionBinding::View,
163                draw_function,
164                index: layer.index,
165                generate_borders: false,
166                style_layer: layer.id.clone(),
167                source_shape: crate::render::tile_view_pattern::TileShape::default(),
168
169                // We provide a dummy tile for background.
170                tile: crate::tcs::tiles::Tile {
171                    coords: crate::coords::WorldTileCoords::default(),
172                },
173                run: None,
174            });
175        }
176        // The sky follows the background layers and precedes everything else, as GL JS draws it
177        // under the map; the flat map covers it below the horizon.
178        if sky_metadata.is_some() {
179            layer_item_phase.add(LayerItem {
180                projection: ProjectionBinding::View,
181                draw_function: Box::new(DrawState::<LayerItem, DrawSky>::new()),
182                index: background_index,
183                generate_borders: false,
184                style_layer: "sky".to_string(),
185                source_shape: crate::render::tile_view_pattern::TileShape::default(),
186                tile: crate::tcs::tiles::Tile {
187                    coords: crate::coords::WorldTileCoords::default(),
188                },
189                run: None,
190            });
191        }
192        if atmosphere_blend > 0.0 {
193            translucent_phase.add(TranslucentItem {
194                draw_function: Box::new(DrawState::<TranslucentItem, DrawAtmosphere>::new()),
195                index: u32::MAX,
196                style_layer: "atmosphere".to_string(),
197                tile: crate::tcs::tiles::Tile {
198                    coords: crate::coords::WorldTileCoords::default(),
199                },
200                source_shape: crate::render::tile_view_pattern::TileShape::default(),
201            });
202        }
203    }
204
205    if !metadatas.is_empty() || atmosphere_blend > 0.0 || sky_metadata.is_some() {
206        if metadatas.is_empty() {
207            metadatas.push(BackgroundLayerMetadata {
208                color: [0.0; 4],
209                z_index: 0.0,
210                padding: [0.0; 3],
211                horizon: view_state.horizon_line().to_shader(),
212                viewport: [
213                    view_state.height() as f32,
214                    f32::from(style.terrain.is_some()),
215                    frame[0],
216                    frame[1],
217                ],
218                world_rows,
219            });
220        }
221        let globe_tiles = globe_tiles(world);
222        let tile_metadata: Vec<ShaderTileMetadata> = globe_tiles
223            .iter()
224            .map(|coords| globe_tile_metadata(view_state, *coords))
225            .collect();
226        let atmosphere_metadata = if atmosphere_blend > 0.0 {
227            let camera = globe_camera_for_view(view_state).map_err(|error| {
228                tracing::error!(?error, "failed to build atmosphere globe camera");
229                SystemError::Setup
230            })?;
231            let light = style.light.clone().unwrap_or_default();
232            let scene_lighting = world
233                .resources
234                .get::<crate::render::SceneLighting>()
235                .copied()
236                .unwrap_or_default();
237            AtmosphereLayerMetadata::from_view(
238                &camera,
239                &light,
240                scene_lighting,
241                view_state.zoom().value(),
242                atmosphere_blend,
243            )
244            .map_err(|error| {
245                tracing::error!(?error, "failed to build atmosphere shader metadata");
246                SystemError::Setup
247            })?
248        } else {
249            AtmosphereLayerMetadata::disabled()
250        };
251
252        let (buffer, tile_metadata_buffer, atmosphere_metadata_buffer, sky_metadata_buffer) = {
253            let gpu_buffers = world
254                .resources
255                .get_or_init_mut::<BackgroundGpuBuffers>();
256            gpu_buffers.write_metadata(&renderer.device, &renderer.queue, &metadatas);
257            gpu_buffers.write_tile_metadata(&renderer.device, &renderer.queue, &tile_metadata);
258            gpu_buffers.write_atmosphere(&renderer.device, &renderer.queue, &atmosphere_metadata);
259            if let Some(sky) = &sky_metadata {
260                gpu_buffers.write_sky(&renderer.device, &renderer.queue, sky);
261            }
262            (
263                gpu_buffers
264                    .metadata_buffer()
265                    .expect("metadata buffer")
266                    .clone(),
267                gpu_buffers
268                    .tile_metadata_buffer()
269                    .expect("tile metadata buffer")
270                    .clone(),
271                gpu_buffers
272                    .atmosphere_metadata_buffer()
273                    .expect("atmosphere metadata buffer")
274                    .clone(),
275                sky_metadata.as_ref().map(|_| {
276                    gpu_buffers
277                        .sky_metadata_buffer()
278                        .expect("sky metadata buffer")
279                        .clone()
280                }),
281            )
282        };
283        world.resources.insert(BackgroundBuffers {
284            metadata_buffer: buffer,
285            instances,
286            tile_metadata_buffer,
287            globe_tiles,
288            atmosphere_metadata_buffer,
289            sky_metadata_buffer,
290        });
291    }
292
293    Ok(())
294}
