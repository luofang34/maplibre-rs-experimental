33/// How strongly the atmosphere is drawn this frame: as the style's `sky` blends it, or in full
34/// where the host turned it on, faded with the projection's globe transition.
35pub fn atmosphere_blend(
36    style: &crate::style::Style,
37    view_state: &crate::render::view_state::ViewState,
38    world: &crate::tcs::world::World,
39) -> f32 {
40    let zoom = view_state.zoom().value();
41    let transition = style.projection.as_ref().map_or(0.0, |specification| {
42        specification.projection_type.globe_transition(zoom)
43    });
44    let sky_blend = style
45        .sky
46        .as_ref()
47        .map_or(0.0, |sky| sky.atmosphere_blend_at_zoom(zoom));
48    let host_blend = f32::from(u8::from(crate::celestial::lighting(world).atmosphere));
49    sky_blend.max(host_blend) * transition
50}

52/// Appends visible background, sky and atmosphere items and uploads their frame metadata.
53/// Returns `Dependencies` if the required phases are absent, or `Setup` if atmosphere
54/// camera or shader metadata cannot be constructed. Existing phase items are not cleared.
55pub fn queue_system(
56    MapContext {
57        world,
58        style,
59        view_state,
60        renderer,
61        ..
62    }: &mut MapContext,
63) -> SystemResult {
64    let mut metadatas = Vec::new();
65    let frame = {
66        let size = renderer.state().surface().size();
67        [size.width() as f32, size.height() as f32]
68    };
69    let world_rows = world_rows(view_state, style.terrain.is_none());
70    let mut instances = std::collections::HashMap::new();
71    let projection_transition = style.projection.as_ref().map_or(0.0, |specification| {
72        specification
73            .projection_type
74            .globe_transition(view_state.zoom().value())
75    });
76    let uses_globe = projection_transition > 0.0;
77    let atmosphere_blend = atmosphere_blend(style, view_state, world);
78    let sky_metadata = style
79        .sky
80        .as_ref()
81        .filter(|_| projection_transition < 1.0 || view_state.opaque_environment())
82        .map(|sky| sky_metadata(sky, view_state, projection_transition));
83
84    if let Some(gpu) = world
85        .resources
86        .get::<super::pattern::BackgroundPatternGpu>()
87    {
88        let size = renderer.state().surface().size();
89        gpu.write(
90            &renderer.queue,
91            view_state,
92            [size.width() as f32, size.height() as f32],
93        );
94    }
95    {
96        let Some((layer_item_phase, translucent_phase)) = world.resources.query_mut::<(
97            &mut RenderPhase<LayerItem>,
98            &mut RenderPhase<TranslucentItem>,
99        )>() else {
100            return Err(SystemError::Dependencies);
101        };
102
103        let patterned: std::collections::HashSet<&str> = style
104            .layers
105            .iter()
106            .filter(|layer| {
107                layer.paint.as_ref().is_some_and(|paint| {
108                    crate::vector::pattern::pattern_name(paint, view_state.style_zoom().value())
109                        .is_some()
110                })
111            })
112            .map(|layer| layer.id.as_str())
113            .collect();
114        let mut background_index = 0;
115        for layer in &style.layers {
116            if layer.type_ != "background" || !layer.is_visible_at(view_state.zoom().value()) {
117                continue;
118            }
119            // Under terrain the drape of each terrain tile carries the background, so nothing
120            // paints outside the world.
121            if style.terrain.is_some() && !uses_globe {
122                continue;
123            }
124            // A layer that names an image it does not have draws nothing, not its colour.
125            if crate::vector::pattern::names_missing_image(
126                layer.paint.as_ref(),
127                style,
128                view_state.style_zoom().value(),
129            ) {
130                continue;
131            }
132            background_index = background_index.max(layer.index);
133            let c = background_color(layer, view_state.zoom().value());
134            let z_index = layer.index as f32;
135            instances.insert(layer.id.clone(), metadatas.len() as u32);
136            metadatas.push(BackgroundLayerMetadata {
137                color: c,
138                z_index,
139                padding: [0.0; 3],
140                horizon: view_state.horizon_line().to_shader(),
141                viewport: [
142                    view_state.height() as f32,
143                    f32::from(style.terrain.is_some()),
144                    frame[0],
145                    frame[1],
146                ],
147                world_rows,
148            });
149
150            let draw_function: Box<dyn crate::render::render_phase::Draw<LayerItem>> =
151                if uses_globe && patterned.contains(layer.id.as_str()) {
152                    Box::new(DrawState::<LayerItem, DrawGlobeBackgroundPattern>::new())
153                } else if uses_globe {
154                    Box::new(DrawState::<LayerItem, DrawGlobeBackground>::new())
155                } else if patterned.contains(layer.id.as_str()) {
156                    Box::new(DrawState::<LayerItem, DrawBackgroundPattern>::new())
157                } else {
158                    Box::new(DrawState::<LayerItem, DrawBackground>::new())
159                };
160            layer_item_phase.add(LayerItem {
161                projection: ProjectionBinding::View,
162                draw_function,
163                index: layer.index,
164                generate_borders: false,
165                style_layer: layer.id.clone(),
166                source_shape: crate::render::tile_view_pattern::TileShape::default(),
167
168                // We provide a dummy tile for background.
169                tile: crate::tcs::tiles::Tile {
170                    coords: crate::coords::WorldTileCoords::default(),
171                },
172                run: None,
173            });
174        }
175        // The sky follows the background layers and precedes everything else, as GL JS draws it
176        // under the map; the flat map covers it below the horizon.
177        if sky_metadata.is_some() {
178            layer_item_phase.add(LayerItem {
179                projection: ProjectionBinding::View,
180                draw_function: Box::new(DrawState::<LayerItem, DrawSky>::new()),
181                index: background_index,
182                generate_borders: false,
183                style_layer: "sky".to_string(),
184                source_shape: crate::render::tile_view_pattern::TileShape::default(),
185                tile: crate::tcs::tiles::Tile {
186                    coords: crate::coords::WorldTileCoords::default(),
187                },
188                run: None,
189            });
190        }
191        if atmosphere_blend > 0.0 {
192            translucent_phase.add(TranslucentItem {
193                draw_function: Box::new(DrawState::<TranslucentItem, DrawAtmosphere>::new()),
194                index: u32::MAX,
195                style_layer: "atmosphere".to_string(),
196                tile: crate::tcs::tiles::Tile {
197                    coords: crate::coords::WorldTileCoords::default(),
198                },
199                source_shape: crate::render::tile_view_pattern::TileShape::default(),
200            });
201        }
202    }
203
204    if !metadatas.is_empty() || atmosphere_blend > 0.0 || sky_metadata.is_some() {
205        if metadatas.is_empty() {
206            metadatas.push(BackgroundLayerMetadata {
207                color: [0.0; 4],
208                z_index: 0.0,
209                padding: [0.0; 3],
210                horizon: view_state.horizon_line().to_shader(),
211                viewport: [
212                    view_state.height() as f32,
213                    f32::from(style.terrain.is_some()),
214                    frame[0],
215                    frame[1],
216                ],
217                world_rows,
218            });
219        }
220        let buffer = renderer.queue.create_buffer_init(
221            &renderer.device,
222            &wgpu::util::BufferInitDescriptor {
223                label: Some("Background Metadata Buffer"),
224                contents: bytemuck::cast_slice(&metadatas),
225                usage: wgpu::BufferUsages::VERTEX | wgpu::BufferUsages::COPY_DST,
226            },
227        );
228        let coords = crate::coords::WorldTileCoords::default();
229        let transform = view_state
230            .gpu_view_projection()
231            .to_model_view_projection(coords.transform_for_zoom(view_state.zoom()))
232            .downcast()
233            .into();
234        let tile_metadata = ShaderTileMetadata {
235            transform,
236            zoom_factor: view_state.zoom().scale_to_tile(&coords) as f32,
237            viewport_width: view_state.width() as f32,
238            viewport_height: view_state.height() as f32,
239            tile_mercator_coords: crate::projection::renderer_data::tile_mercator_coordinates(
240                Some(crate::coords::TileCoords::default()),
241            )
242            .into(),
243            clip_antimeridian: 1,
244            line_width_scale: 1.0,
245            line_units_per_pixel: 8.0 * view_state.zoom().scale_to_tile(&coords) as f32,
246            clip_region: crate::render::shaders::WHOLE_TILE,
247        };
248        let tile_metadata_buffer = renderer.queue.create_buffer_init(
249            &renderer.device,
250            &wgpu::util::BufferInitDescriptor {
251                label: Some("Globe Background Tile Metadata Buffer"),
252                contents: bytemuck::bytes_of(&tile_metadata),
253                usage: wgpu::BufferUsages::VERTEX | wgpu::BufferUsages::COPY_DST,
254            },
255        );
256        let atmosphere_metadata = if atmosphere_blend > 0.0 {
257            let camera = globe_camera_for_view(view_state).map_err(|error| {
258                tracing::error!(?error, "failed to build atmosphere globe camera");
259                SystemError::Setup
260            })?;
261            let light = style.light.clone().unwrap_or_default();
262            AtmosphereLayerMetadata::from_view(
263                &camera,
264                &light,
265                crate::celestial::sun_direction(world),
266                view_state.zoom().value(),
267                atmosphere_blend,
268            )
269            .map_err(|error| {
270                tracing::error!(?error, "failed to build atmosphere shader metadata");
271                SystemError::Setup
272            })?
273        } else {
274            AtmosphereLayerMetadata::disabled()
275        };
276        let atmosphere_metadata_buffer = renderer.queue.create_buffer_init(
277            &renderer.device,
278            &wgpu::util::BufferInitDescriptor {
279                label: Some("Atmosphere Metadata Buffer"),
280                contents: bytemuck::bytes_of(&atmosphere_metadata),
281                usage: wgpu::BufferUsages::VERTEX | wgpu::BufferUsages::COPY_DST,
282            },
283        );
284        let sky_metadata_buffer = sky_metadata.map(|sky| {
285            renderer.queue.create_buffer_init(
286                &renderer.device,
287                &wgpu::util::BufferInitDescriptor {
288                    label: Some("Sky Metadata Buffer"),
289                    contents: bytemuck::bytes_of(&sky),
290                    usage: wgpu::BufferUsages::VERTEX | wgpu::BufferUsages::COPY_DST,
291                },
292            )
293        });
294        world.resources.insert(BackgroundBuffers {
295            metadata_buffer: buffer,
296            instances,
297            tile_metadata_buffer,
298            atmosphere_metadata_buffer,
299            sky_metadata_buffer,
300        });
301    }
302
303    Ok(())
304}

306/// The background colour at a zoom, with its opacity multiplied into premultiplied-free alpha.
307fn background_color(layer: &crate::style::layer::StyleLayer, zoom: f64) -> [f32; 4] {
308    let Some(LayerPaint::Background(paint)) = &layer.paint else {
309        return [0.0, 0.0, 0.0, 1.0];
310    };
311    let mut color: [f32; 4] = paint
312        .background_color
313        .as_ref()
314        .and_then(|property| property.evaluate_at_zoom(zoom))
315        .map_or([0.0, 0.0, 0.0, 1.0], |color| {
316            cint::Alpha::<cint::EncodedSrgb<f32>>::from(color).into()
317        });
318    let opacity = paint
319        .background_opacity
320        .as_ref()
321        .and_then(|property| property.evaluate_at_zoom(zoom))
322        .unwrap_or(1.0);
323    color[3] *= opacity.clamp(0.0, 1.0);
324    color
325}

327/// The sky draw's values for the frame: colours at the zoom and the horizon line on screen,
328/// as GL JS `skyUniformValues` computes them.
329fn sky_metadata(
330    sky: &crate::style::sky::SkySpecification,
331    view_state: &crate::render::view_state::ViewState,
332    projection_transition: f32,
333) -> crate::render::shaders::SkyLayerMetadata {
334    let colors = sky.colors_at(view_state.zoom().value());
335    let height = view_state.height();
336    // Zero depth denotes absent content to the host compositor. This distant positive
337    // depth keeps sky coverage valid without visible parallax at the headset near plane.
338    crate::render::shaders::SkyLayerMetadata {
339        sky_color: colors.sky,
340        horizon_color: colors.horizon,
341        horizon: view_state.horizon_line().to_shader(),
342        blend: [
343            (f64::from(colors.sky_horizon_blend) * height / 2.0) as f32,
344            if view_state.opaque_environment() {
345                0.0
346            } else {
347                projection_transition
348            },
349            height as f32,
350            if view_state.has_external_view() {
351                1.0e-8
352            } else {
353                0.0
354            },
355        ],
356    }
357}

359/// The rows that tell a pixel's map row for the northern and southern edges of the flat map,
360/// or ones that cut nothing where the map has no such edges.
361fn world_rows(view_state: &crate::render::view_state::ViewState, edges: bool) -> [[f32; 4]; 2] {
362    use cgmath::{Matrix3, Matrix4, SquareMatrix, Vector3};
363
364    const OPEN: [[f32; 4]; 2] = [[0.0, 0.0, 0.0, -1e30], [0.0, 0.0, 0.0, 1e30]];
365    if !edges {
366        return OPEN;
367    }
368    let center = view_state.camera().position();
369    let relative = view_state.view_projection().0
370        * Matrix4::from_translation(Vector3::new(center.x, center.y, 0.0));
371    // A point of the map plane is (x, y, 0, 1); the clip x, y and w it produces come from
372    // the matrix columns of x, y and the constant.
373    let plane = Matrix3::new(
374        relative.x.x,
375        relative.x.y,
376        relative.x.w,
377        relative.y.x,
378        relative.y.y,
379        relative.y.w,
380        relative.w.x,
381        relative.w.y,
382        relative.w.w,
383    );
384    let Some(inverse) = plane.invert() else {
385        return OPEN;
386    };
387    let world = crate::coords::TILE_SIZE * 2_f64.powf(view_state.zoom().value());
388    // Rows of the inverse give the map x, y and the scale a pixel's clip position carries.
389    [
390        [
391            inverse.x.y as f32,
392            inverse.y.y as f32,
393            inverse.z.y as f32,
394            (-center.y) as f32,
395        ],
396        [
397            inverse.x.z as f32,
398            inverse.y.z as f32,
399            inverse.z.z as f32,
400            (world - center.y) as f32,
401        ],
402    ]
403}

405#[cfg(test)]
406mod tests {
407    use super::background_color;
408    use crate::style::layer::StyleLayer;
409
410    fn layer(paint: serde_json::Value) -> StyleLayer {
411        let layer = serde_json::json!({"id": "b", "type": "background", "paint": paint});
412        serde_json::from_value(layer).expect("valid background layer")
413    }
414
415    #[test]
416    fn opacity_scales_alpha() {
417        let color = background_color(
418            &layer(serde_json::json!({"background-color": "blue", "background-opacity": 0.5})),
419            0.0,
420        );
421        assert_eq!(color, [0.0, 0.0, 1.0, 0.5]);
422    }
423
424    #[test]
425    fn colour_follows_zoom() {
426        let paint = serde_json::json!({"background-color": {
427            "stops": [[0, "red"], [10, "blue"]], "base": 1.0
428        }});
429        let start = background_color(&layer(paint.clone()), 0.0);
430        let end = background_color(&layer(paint), 10.0);
431        assert_eq!(start[..3], [1.0, 0.0, 0.0]);
432        assert_eq!(end[..3], [0.0, 0.0, 1.0]);
433    }
