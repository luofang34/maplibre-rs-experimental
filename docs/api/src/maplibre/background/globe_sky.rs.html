1//! The sky behind a globe seen from an eye in or above the atmosphere: an analytic gradient
2//! from the horizon colour at the eye's sea-level horizon to the sky colour above it.
3//!
4//! Unlike the atmosphere's scattering, the gradient follows only the elevation angle of each
5//! pixel's ray above the eye's local horizon, so it does not change with heading, sun or
6//! altitude: what an avionics synthetic-vision display needs for constant attitude cueing.
7
8use cgmath::{InnerSpace, Vector3};
9
10use crate::{
11    render::{projection::globe_camera_for_view, shaders::SkyLayerMetadata, view_state::ViewState},
12    style::sky::SkySpecification,
13};
14
15/// Smallest sky-horizon blend, as a share of the half field of view, so the horizon colour
16/// always shows as a band rather than a line.
17const MIN_BLEND: f32 = 0.05;
18/// Smallest half field of view the blend is measured against, in radians.
19const MIN_HALF_FIELD_OF_VIEW: f32 = 0.25;
20
21/// Whether the sky is drawn as [`globe_sky_metadata`] does: on a full globe without
22/// atmosphere, where the host asked for an opaque environment.
23pub fn draws_globe_sky(
24    view_state: &ViewState,
25    projection_transition: f32,
26    atmosphere_blend: f32,
27) -> bool {
28    projection_transition >= 1.0
29        && atmosphere_blend <= 0.0
30        && view_state.opaque_environment_requested()
31}
32
33/// The globe sky's values for the frame, or `None` when the view has no globe camera.
34///
35/// The sky is drawn at depth zero, so under reversed-Z it fills only pixels no globe
36/// background or terrain covered: above the limb at any altitude, with no gap and without
37/// hiding the ground colour below it.
38pub fn globe_sky_metadata(
39    sky: &SkySpecification,
40    view_state: &ViewState,
41) -> Option<SkyLayerMetadata> {
42    let camera = globe_camera_for_view(view_state).ok()?;
43    let colors = sky.colors_at(view_state.zoom().value());
44    let inverse_projection = camera.inverse_projection().cast::<f32>()?;
45    let eye = camera.camera_position();
46    let radius = eye.magnitude();
47    let up_world = if radius > 1e-9 {
48        eye / radius
49    } else {
50        Vector3::new(0.0, 0.0, 1.0)
51    };
52    let up_view = camera
53        .world_direction_to_view(up_world)
54        .and_then(|up| up.cast::<f32>())
55        .unwrap_or(Vector3::new(0.0, 1.0, 0.0));
56    // The tangent from an eye at `radius` (unit globe) meets the sphere this far below the
57    // eye's horizontal; inside the sphere the eye's own horizontal is the horizon.
58    let horizon_elevation = -((1.0 / radius.max(1.0)).clamp(0.0, 1.0).acos() as f32);
59    let half_field_of_view =
60        (camera.field_of_view_degrees().to_radians() as f32 * 0.5).max(MIN_HALF_FIELD_OF_VIEW);
61    let blend_angle = colors.sky_horizon_blend.max(MIN_BLEND) * half_field_of_view;
62    Some(SkyLayerMetadata {
63        sky_color: colors.sky,
64        horizon_color: colors.horizon,
65        horizon: view_state.horizon_line().to_shader(),
66        blend: [blend_angle, 0.0, -1.0, 0.0],
67        inverse_projection: inverse_projection.into(),
68        up_and_horizon: [up_view.x, up_view.y, up_view.z, horizon_elevation],
69    })
70}
