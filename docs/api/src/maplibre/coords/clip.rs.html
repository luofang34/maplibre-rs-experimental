1//! Cutting rings and lines down to a rectangle, as GL JS's `geojson-vt` cuts a GeoJSON feature
2//! for each tile it reaches: one axis at a time, each ring on its own.
3//!
4//! A cut ring follows the rectangle's edges where it left it, so a ring that encloses the whole
5//! rectangle becomes the rectangle itself.
6
7use geo_types::{Coord, Rect};
8
9/// One side of the clipping rectangle: the axis it bounds and the range kept on it.
10#[derive(Clone, Copy)]
11struct Slab {
12    axis: usize,
13    min: f64,
14    max: f64,
15}
16
17impl Slab {
18    fn of(coord: Coord<f64>, axis: usize) -> f64 {
19        if axis == 0 {
20            coord.x
21        } else {
22            coord.y
23        }
24    }
25
26    /// Where the segment `a`–`b` crosses `at` on this slab's axis.
27    fn crossing(&self, a: Coord<f64>, b: Coord<f64>, at: f64) -> Coord<f64> {
28        let (from, to) = (Self::of(a, self.axis), Self::of(b, self.axis));
29        let t = (at - from) / (to - from);
30        let point = Coord {
31            x: a.x + (b.x - a.x) * t,
32            y: a.y + (b.y - a.y) * t,
33        };
34        // The crossing lies exactly on the edge, whatever rounding the interpolation does.
35        if self.axis == 0 {
36            Coord { x: at, ..point }
37        } else {
38            Coord { y: at, ..point }
39        }
40    }
41
42    /// A closed ring cut to the slab, with the slab's edges closing what it cut off.
43    fn ring(&self, ring: &[Coord<f64>]) -> Vec<Coord<f64>> {
44        let mut kept = Vec::with_capacity(ring.len());
45        for pair in ring.windows(2) {
46            let (a, b) = (pair[0], pair[1]);
47            let (from, to) = (Self::of(a, self.axis), Self::of(b, self.axis));
48            let inside = |value: f64| (self.min..=self.max).contains(&value);
49            if inside(from) {
50                kept.push(a);
51            }
52            // A segment can leave across one edge and come back across the other.
53            let mut crossings = [self.min, self.max]
54                .into_iter()
55                .filter(|edge| (from - edge) * (to - edge) < 0.0)
56                .collect::<Vec<_>>();
57            if from > to {
58                crossings.reverse();
59            }
60            kept.extend(crossings.into_iter().map(|edge| self.crossing(a, b, edge)));
61        }
62        if let Some(&first) = kept.first() {
63            kept.push(first);
64        }
65        kept
66    }
67
68    /// The runs of a line within the slab.
69    fn line(&self, line: &[Coord<f64>]) -> Vec<Vec<Coord<f64>>> {
70        let inside =
71            |coord: Coord<f64>| (self.min..=self.max).contains(&Self::of(coord, self.axis));
72        let mut runs = Vec::new();
73        let mut run: Vec<Coord<f64>> = Vec::new();
74        for (index, &point) in line.iter().enumerate() {
75            if index > 0 {
76                let a = line[index - 1];
77                let (from, to) = (Self::of(a, self.axis), Self::of(point, self.axis));
78                let mut crossings = [self.min, self.max]
79                    .into_iter()
80                    .filter(|edge| (from - edge) * (to - edge) < 0.0)
81                    .collect::<Vec<_>>();
82                if from > to {
83                    crossings.reverse();
84                }
85                for edge in crossings {
86                    let crossing = self.crossing(a, point, edge);
87                    if run.is_empty() {
88                        run.push(crossing);
89                    } else {
90                        run.push(crossing);
91                        runs.push(std::mem::take(&mut run));
92                    }
93                }
94            }
95            if inside(point) {
96                run.push(point);
97            }
98        }
99        runs.push(run);
100        runs.retain(|run| run.len() >= 2);
101        runs
102    }
103}
104
105fn slabs(area: Rect<f64>) -> [Slab; 2] {
106    let (min, max) = (area.min(), area.max());
107    [
108        Slab {
109            axis: 0,
110            min: min.x,
111            max: max.x,
112        },
113        Slab {
114            axis: 1,
115            min: min.y,
116            max: max.y,
117        },
118    ]
119}
120
121/// The closed ring `ring` cut to `area`, closed again; fewer than four points when nothing of
122/// it encloses any of `area`.
123pub(crate) fn ring_within(ring: &[Coord<f64>], area: Rect<f64>) -> Vec<Coord<f64>> {
124    let [across, down] = slabs(area);
125    down.ring(&across.ring(ring))
126}
127
128/// The runs of `line` within `area`, each at least two points long.
129pub(crate) fn line_within(line: Vec<Coord<f64>>, area: Rect<f64>) -> Vec<Vec<Coord<f64>>> {
130    slabs(area).iter().fold(vec![line], |runs, slab| {
131        runs.iter().flat_map(|run| slab.line(run)).collect()
132    })
133}
