1//! Visible tile regions across canonical and wrapped worlds.
2
3use std::collections::HashSet;
4
5use crate::{
6    coords::{WorldCoords, WorldTileCoords, Zoom, ZoomLevel},
7    util::math::Aabb2,
8};
9
10/// Defines a bounding box on a tiled map with a [`ZoomLevel`] and a padding.
11#[derive(Clone, Debug)]
12pub struct ViewRegion {
13    min_tile: WorldTileCoords,
14    max_tile: WorldTileCoords,
15    explicit_tiles: Option<Vec<WorldTileCoords>>,
16    /// At which zoom level does this region exist
17    zoom_level: ZoomLevel,
18    /// Padding around this view region
19    padding: i32,
20    /// The maximum amount of tiles this view region contains
21    max_n_tiles: usize,
22}
23
24impl ViewRegion {
25    /// Covers a world-pixel box at `zoom` with tiles at `z` and an integer tile padding.
26    /// `max_n_tiles` limits iteration, not the membership test in [`Self::is_in_view`].
27    pub fn new(
28        view_region: Aabb2<f64>,
29        padding: i32,
30        max_n_tiles: usize,
31        zoom: Zoom,
32        z: ZoomLevel,
33    ) -> Self {
34        let min_world: WorldCoords = WorldCoords::at_ground(view_region.min.x, view_region.min.y);
35        let min_world_tile: WorldTileCoords = min_world.into_world_tile(z, zoom);
36        let max_world: WorldCoords = WorldCoords::at_ground(view_region.max.x, view_region.max.y);
37        let max_world_tile: WorldTileCoords = max_world.into_world_tile(z, zoom);
38
39        Self {
40            min_tile: min_world_tile,
41            max_tile: max_world_tile,
42            explicit_tiles: None,
43            zoom_level: z,
44            max_n_tiles,
45            padding,
46        }
47    }
48
49    /// Creates a region from an explicit projection-aware tile selection.
50    pub fn from_tiles(
51        tiles: Vec<WorldTileCoords>,
52        zoom_level: ZoomLevel,
53        max_n_tiles: usize,
54    ) -> Self {
55        Self {
56            min_tile: WorldTileCoords::default(),
57            max_tile: WorldTileCoords::default(),
58            explicit_tiles: Some(tiles),
59            zoom_level,
60            max_n_tiles,
61            padding: 0,
62        }
63    }
64
65    /// Returns the grid level supplied when constructing this region.
66    pub fn zoom_level(&self) -> ZoomLevel {
67        self.zoom_level
68    }
69
70    /// Tests the padded bounds or explicit selection without applying the iteration limit.
71    pub fn is_in_view(&self, &world_coords: &WorldTileCoords) -> bool {
72        if let Some(tiles) = &self.explicit_tiles {
73            return tiles.contains(&world_coords);
74        }
75        world_coords.x <= self.max_tile.x + self.padding
76            && world_coords.y <= self.max_tile.y + self.padding
77            && world_coords.x >= self.min_tile.x - self.padding
78            && world_coords.y >= self.min_tile.y - self.padding
79            && world_coords.z == self.zoom_level
80    }
81
82    /// The tiles a view sees, with those seen in copies of the world as the tiles they repeat,
83    /// each once: up to the configured limit, in explicit selection order or column-major
84    /// grid order.
85    pub fn iter(&self) -> Box<dyn Iterator<Item = WorldTileCoords> + '_> {
86        let mut seen = HashSet::new();
87        Box::new(
88            self.copies()
89                .map(|coords| coords.wrapped().map_or(coords, |(tile, _)| tile))
90                .filter(move |tile| seen.insert(*tile)),
91        )
92    }
93
94    /// The tiles of the view as drawn, a tile seen again in another copy of the world once more
95    /// at coordinates beyond the grid.
96    pub fn copies(&self) -> Box<dyn Iterator<Item = WorldTileCoords> + '_> {
97        if let Some(tiles) = &self.explicit_tiles {
98            return Box::new(tiles.iter().copied().take(self.max_n_tiles));
99        }
100        Box::new(
101            (self.min_tile.x - self.padding..self.max_tile.x + 1 + self.padding)
102                .flat_map(move |x| {
103                    (self.min_tile.y - self.padding..self.max_tile.y + 1 + self.padding)
104                        .map(move |y| (x, y, self.zoom_level).into())
105                })
106                .take(self.max_n_tiles),
107        )
108    }
109}
