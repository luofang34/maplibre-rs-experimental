1//! Cross-platform GPU navigation HUD overlay and tactical selection rectangle.
2//!
3//! Renders a compact floating control capsule in the top-right safe area of the viewport
4//! (3D compass with north-lock, `3D/2D` + `AUTO/MAN` tilt toggle, `+`/`-` zoom buttons, `Home`
5//! reset, and `MAP/RTS` pointer profile switch), plus an optional rubber-band selection box
6//! when dragging a box selection in RTS or `Shift`+drag mode.
7

31/// Logical pixel width of the floating HUD capsule.
32pub const HUD_WIDTH: f64 = 38.0;
33/// Logical pixel height of the floating HUD capsule.
34pub const HUD_HEIGHT: f64 = 186.0;
35/// Right margin from the viewport safe area in logical pixels.
36pub const HUD_MARGIN_RIGHT: f64 = 12.0;
37/// Top margin from the viewport safe area in logical pixels.
38pub const HUD_MARGIN_TOP: f64 = 12.0;
39
40/// Interactive slots inside the floating navigation HUD capsule.
41#[derive(Clone, Copy, Debug, PartialEq, Eq)]
42pub enum HudButton {
43    /// 3D compass needle: click resets north or toggles north-lock; drag orbits bearing/pitch.
44    Compass = 0,
45    /// Cycles `3D AUTO` -> `2D` -> `3D AUTO` (or restores `3D AUTO` from `3D MAN`).
46    Mode3D = 1,
47    /// Smoothly zooms in by one level around the viewport center.
48    ZoomIn = 2,
49    /// Smoothly zooms out by one level around the viewport center.
50    ZoomOut = 3,
51    /// Resets the camera to the style's home view and restores `Auto-Tilt`.
52    Home = 4,
53    /// Switches between `Map` (left-drag pan) and `RTS` (left-drag box select, right-drag pan).
54    PointerMode = 5,
55}
56
57impl HudButton {
58    fn slot_index(self) -> f32 {
59        match self {
60            Self::Compass => 0.0,
61            Self::Mode3D => 1.0,
62            Self::ZoomIn => 2.0,
63            Self::ZoomOut => 3.0,
64            Self::Home => 4.0,
65            Self::PointerMode => 5.0,
66        }
67    }
68}

70/// Shared state between the input controller and the GPU HUD renderer.
71#[derive(Clone, Copy, Debug, PartialEq)]
72pub struct NavigationHudState {
73    /// Whether the HUD overlay is drawn.
74    pub visible: bool,
75    /// Whether oblique 3D pitch is permitted (`true` = 3D, `false` = 2D overhead).
76    pub three_dimensional: bool,
77    /// Whether zoom automatically couples with camera tilt (`Auto-Tilt`).
78    pub auto_tilt: bool,
79    /// Whether bearing is locked to North (`0°`).
80    pub north_locked: bool,
81    /// Whether RTS pointer semantics (`Left-drag` box select, `Right-drag` pan, edge scroll) are active.
82    pub rts_mode: bool,
83    /// Multiplier on scroll-wheel / trackpad zoom rate (`1.0` = default).
84    pub zoom_sensitivity: f64,
85    /// Screen-edge scrolling speed in logical pixels per second.
86    pub edge_scroll_speed_px_per_sec: f64,
87    /// Whether screen-edge scrolling is enabled even in standard Map pointer profile.
88    pub edge_scroll_enabled: bool,
89    /// Button currently under the pointer, if any.
90    pub hovered: Option<HudButton>,
91    /// Button currently held down by the pointer, if any.
92    pub pressed: Option<HudButton>,
93    /// Active box-selection rectangle `[min_x, min_y, max_x, max_y]` in logical pixels.
94    pub selection_box: Option<[f32; 4]>,
95}
96
97impl Default for NavigationHudState {
98    fn default() -> Self {
99        Self {
100            visible: false,
101            three_dimensional: true,
102            auto_tilt: false,
103            north_locked: false,
104            rts_mode: false,
105            zoom_sensitivity: 1.0,
106            edge_scroll_speed_px_per_sec: 1100.0,
107            edge_scroll_enabled: false,
108            hovered: None,
109            pressed: None,
110            selection_box: None,
111        }
112    }
113}

115impl NavigationHudState {
116    /// Top-left logical pixel coordinates `(x, y)` of the HUD panel for the given viewport and insets.
117    pub fn panel_origin(
118        viewport_width: f64,
119        _viewport_height: f64,
120        insets: &EdgeInsets,
121    ) -> Vector2<f64> {
122        let x = (viewport_width - insets.right - HUD_MARGIN_RIGHT - HUD_WIDTH).max(8.0);
123        let y = (insets.top + HUD_MARGIN_TOP).max(8.0);
124        Vector2::new(x, y)
125    }
126
127    /// Tests whether a logical window coordinate hits one of the HUD capsule buttons.
128    pub fn hit_test(
129        logical_pos: Vector2<f64>,
130        viewport_width: f64,
131        viewport_height: f64,
132        insets: &EdgeInsets,
133    ) -> Option<HudButton> {
134        let origin = Self::panel_origin(viewport_width, viewport_height, insets);
135        let rel = logical_pos - origin;
136        if rel.x < 0.0 || rel.x > HUD_WIDTH || rel.y < 0.0 || rel.y > HUD_HEIGHT {
137            return None;
138        }
139        if rel.y < 40.0 {
140            Some(HudButton::Compass)
141        } else if rel.y < 74.0 {
142            Some(HudButton::Mode3D)
143        } else if rel.y < 104.0 {
144            Some(HudButton::ZoomIn)
145        } else if rel.y < 133.0 {
146            Some(HudButton::ZoomOut)
147        } else if rel.y < 160.0 {
148            Some(HudButton::Home)
149        } else {
150            Some(HudButton::PointerMode)
151        }
152    }
153}

155/// The view slots that draw the HUD. The controls steer the map's own camera and are
156/// hit-tested against its viewport, so a slot, which has a camera of its own, shows them only
157/// where its host asks.
158#[derive(Default)]
159pub(crate) struct HudViewSlots(FxHashSet<ViewSlotId>);
160
161// Only a map that draws view slots has any to list.
162#[cfg(feature = "headless")]
163impl HudViewSlots {
164    pub(crate) fn set(&mut self, slot: ViewSlotId, shown: bool) {
165        if shown {
166            self.0.insert(slot);
167        } else {
168            self.0.remove(&slot);
169        }
170    }
171}
172
173/// Whether the view being drawn shows the HUD: the map's own view, or a slot asked to.
174fn shown_in_current_view(world: &World) -> bool {
175    let Some(slot) = world.resources.get::<CurrentViewSlot>().and_then(|s| s.0) else {
176        return true;
177    };
178    world
179        .resources
180        .get::<HudViewSlots>()
181        .is_some_and(|slots| slots.0.contains(&slot))
182}

204struct HudShader {
205    format: wgpu::TextureFormat,
206}

208impl Shader for HudShader {
209    fn describe_vertex(&self) -> VertexState {
210        VertexState {
211            source: include_str!("hud/hud.vertex.wgsl"),
212            entry_point: "main",
213            buffers: Vec::new(),
214        }
215    }
216
217    fn describe_fragment(&self) -> FragmentState {
218        FragmentState {
219            source: include_str!("hud/hud.fragment.wgsl"),
220            entry_point: "main",
221            targets: vec![Some(wgpu::ColorTargetState {
222                format: self.format,
223                blend: Some(wgpu::BlendState::PREMULTIPLIED_ALPHA_BLENDING),
224                write_mask: wgpu::ColorWrites::ALL,
225            })],
226        }
227    }
228}

230/// Plugin that draws the cross-platform navigation HUD at the end of the translucent pass.
231#[derive(Default)]
232pub struct NavigationHudPlugin;
233
234impl<E: Environment> Plugin<E> for NavigationHudPlugin {
235    fn build(
236        &self,
237        schedule: &mut crate::schedule::Schedule,
238        _kernel: std::rc::Rc<crate::kernel::Kernel<E>>,
239        world: &mut World,
240        _renderer: &mut crate::render::Renderer,
241    ) -> Result<(), crate::plugin::PluginError> {
242        world
243            .resources
244            .insert(Eventually::<NavigationHudPipeline>::Uninitialized);
245        schedule.add_system_to_stage(RenderStageLabel::Prepare, resource_system)?;
246        schedule.add_system_to_stage(RenderStageLabel::Queue, queue_system)?;
247        if let Some(hooks) = world.resources.get_mut::<crate::render::PassDrawHooks>() {
248            hooks.register(crate::render::PassHookPoint::TranslucentPassEnd, draw);
249        } else {
250            let mut hooks = crate::render::PassDrawHooks::new();
251            hooks.register(crate::render::PassHookPoint::TranslucentPassEnd, draw);
252            world.resources.insert(hooks);
253        }
254        Ok(())
255    }
256}

258fn resource_system(
259    MapContext {
260        world, renderer, ..
261    }: &mut MapContext,
262) -> SystemResult {
263    let Some(pipeline) = world
264        .resources
265        .get_mut::<Eventually<NavigationHudPipeline>>()
266    else {
267        return Err(SystemError::Dependencies);
268    };
269    let device = &renderer.device;
270    let settings = renderer.settings;
271    let surface = &renderer.resources.surface;
272    pipeline.initialize(|| {
273        let shader = HudShader {
274            format: surface.surface_format(),
275        };
276        let frame_layout = device.create_bind_group_layout(&wgpu::BindGroupLayoutDescriptor {
277            label: Some("navigation hud frame layout"),
278            entries: &[wgpu::BindGroupLayoutEntry {
279                binding: 0,
280                visibility: wgpu::ShaderStages::VERTEX_FRAGMENT,
281                ty: wgpu::BindingType::Buffer {
282                    ty: wgpu::BufferBindingType::Uniform,
283                    has_dynamic_offset: false,
284                    min_binding_size: None,
285                },
286                count: None,
287            }],
288        });
289        NavigationHudPipeline(
290            TilePipeline::new(
291                "navigation_hud_pipeline".into(),
292                settings,
293                shader.describe_vertex(),
294                shader.describe_fragment(),
295                TilePipelineOptions {
296                    depth_stencil_enabled: true,
297                    update_stencil: false,
298                    debug_stencil: true,
299                    wireframe: false,
300                    multisampling: surface.is_multisampling_supported(settings.msaa),
301                    textured: false,
302                },
303            )
304            .describe_render_pipeline()
305            .initialize_with_prefix_layouts(device, &[&frame_layout]),
306        )
307    });
308    Ok(())
309}

311fn queue_system(
312    MapContext {
313        world,
314        view_state,
315        renderer,
316        ..
317    }: &mut MapContext,
318) -> SystemResult {
319    let Some(Initialized(NavigationHudPipeline(pipeline))) =
320        world.resources.get::<Eventually<NavigationHudPipeline>>()
321    else {
322        return Ok(());
323    };
324    let hud_state = world
325        .resources
326        .get::<NavigationHudState>()
327        .copied()
328        .unwrap_or_default();
329    let frame_layout = pipeline.get_bind_group_layout(0);
330    let drawn = hud_state.visible && shown_in_current_view(world);
331    let gpu = world.resources.get_or_init_mut::<NavigationHudGpu>();
332    gpu.drawn = drawn;
333    if !drawn {
334        return Ok(());
335    }
336
337    let width = view_state.width();
338    let height = view_state.height();
339    let origin = NavigationHudState::panel_origin(width, height, view_state.edge_insets());
340    let mut flags = 0u32;
341    if hud_state.three_dimensional {
342        flags |= 1;
343    }
344    if hud_state.auto_tilt {
345        flags |= 2;
346    }
347    if hud_state.north_locked {
348        flags |= 4;
349    }
350    if hud_state.rts_mode {
351        flags |= 8;
352    }
353    let selection_active = u32::from(hud_state.selection_box.is_some());
354    gpu.vertex_count = if selection_active != 0 { 12 } else { 6 };
355
356    let frame = HudFrame {
357        viewport_and_panel: [
358            width as f32,
359            height as f32,
360            origin.x as f32,
361            origin.y as f32,
362        ],
363        angles_and_interaction: [
364            view_state.camera().get_bearing().0 as f32,
365            view_state.camera().get_pitch().0 as f32,
366            hud_state.hovered.map_or(-1.0, HudButton::slot_index),
367            hud_state.pressed.map_or(-1.0, HudButton::slot_index),
368        ],
369        state_flags: [flags, selection_active, 0, 0],
370        selection_rect: hud_state.selection_box.unwrap_or([0.0; 4]),
371    };
372
373    let (buffer, _) = gpu.frame.get_or_insert_with(|| {
374        let buffer = renderer.device.create_buffer(&wgpu::BufferDescriptor {
375            label: Some("navigation hud frame"),
376            size: std::mem::size_of::<HudFrame>() as u64,
377            usage: wgpu::BufferUsages::UNIFORM | wgpu::BufferUsages::COPY_DST,
378            mapped_at_creation: false,
379        });
380        let bind_group = renderer
381            .device
382            .create_bind_group(&wgpu::BindGroupDescriptor {
383                label: Some("navigation hud bind group"),
384                layout: &frame_layout,
385                entries: &[wgpu::BindGroupEntry {
386                    binding: 0,
387                    resource: buffer.as_entire_binding(),
388                }],
389            });
390        (buffer, bind_group)
391    });
392    renderer
393        .queue
394        .write_buffer(buffer, 0, bytemuck::bytes_of(&frame));
395    Ok(())
396}

398/// Draws the navigation HUD at the end of the translucent pass.
399pub fn draw<'w>(pass: &mut TrackedRenderPass<'w>, world: &'w World) {
400    let Some((Initialized(NavigationHudPipeline(pipeline)), gpu)) = world
401        .resources
402        .query::<(&Eventually<NavigationHudPipeline>, &NavigationHudGpu)>()
403    else {
404        return;
405    };
406    // Checked here as well as when queueing: a view drawn without queueing must not inherit
407    // the answer of the view queued before it.
408    let drawn = gpu.drawn && shown_in_current_view(world);
409    let Some((_, bind_group)) = gpu.frame.as_ref().filter(|_| drawn) else {
410        return;
411    };
412    pass.set_pipeline(pipeline);
413    pass.set_bind_group(0, bind_group, &[]);
414    pass.draw(0..gpu.vertex_count, 0..1);
