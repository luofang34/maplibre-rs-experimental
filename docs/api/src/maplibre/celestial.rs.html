1//! Sun and sky inputs a host supplies in Earth-centred, Earth-fixed axes.
2//!
3//! The renderer computes no astronomy: a host works out where the sun and the stars are and
4//! hands their directions over as [`EcefDirection`]s. While no sun is set, the style's root
5//! `light` drives the atmosphere as it does without this module.
6
7use cgmath::{InnerSpace, Vector3};
8use thiserror::Error;
9
10use crate::tcs::world::World;

12const MIN_LENGTH_SQUARED: f64 = 1e-24;
13
14/// A direction refused as an [`EcefDirection`].
15#[derive(Clone, Copy, Debug, Error, PartialEq)]
16pub enum CelestialError {
17    /// A component is NaN or infinite, or the vector is too short to have a direction.
18    #[error("direction ({x}, {y}, {z}) is not a finite, non-zero vector")]
19    InvalidDirection {
20        /// ECEF x component as given.
21        x: f64,
22        /// ECEF y component as given.
23        y: f64,
24        /// ECEF z component as given.
25        z: f64,
26    },
27    /// The twilight band does not run upwards between finite elevations, or the night
28    /// brightness lies outside 0..=1.
29    #[error(
30        "twilight from {night_end_degrees}° to {day_start_degrees}° with night brightness \
31         {night_brightness} is not an ascending band of finite elevations with a brightness \
32         in 0..=1"
33    )]
34    InvalidTwilight {
35        /// Solar elevation given for the end of night, in degrees.
36        night_end_degrees: f64,
37        /// Solar elevation given for the start of full day, in degrees.
38        day_start_degrees: f64,
39        /// Brightness given for the night side.
40        night_brightness: f32,
41    },
42    /// A star's magnitude is not finite, or its colour leaves 0..=1.
43    #[error("star of magnitude {magnitude} and colour {color:?} cannot be drawn")]
44    InvalidStar {
45        /// Magnitude given.
46        magnitude: f32,
47        /// Colour given.
48        color: Option<[f32; 3]>,
49    },
50    /// A disc's angular radius is not above zero and at most five degrees.
51    #[error("disc of angular radius {angular_radius_degrees}° cannot be drawn")]
52    InvalidDisc {
53        /// Angular radius given, in degrees.
54        angular_radius_degrees: f64,
55    },
56}

58/// A unit vector in ECEF (ITRF) axes: x towards 0°N 0°E, y towards 0°N 90°E, z towards the
59/// north pole.
60#[derive(Clone, Copy, Debug, PartialEq)]
61pub struct EcefDirection(Vector3<f64>);
62
63impl EcefDirection {
64    /// Normalizes the vector given, which must be finite and not zero.
65    pub fn new(x: f64, y: f64, z: f64) -> Result<Self, CelestialError> {
66        let vector = Vector3::new(x, y, z);
67        let length_squared = vector.magnitude2();
68        if length_squared.is_finite() && length_squared > MIN_LENGTH_SQUARED {
69            Ok(Self(vector / length_squared.sqrt()))
70        } else {
71            Err(CelestialError::InvalidDirection { x, y, z })
72        }
73    }
74
75    /// The unit vector in ECEF axes.
76    pub fn ecef(self) -> Vector3<f64> {
77        self.0
78    }
79
80    /// The unit vector in the axes the globe is drawn in; see [`ecef_to_globe_world`].
81    pub fn globe_world(self) -> Vector3<f64> {
82        ecef_to_globe_world(self.0)
83    }
84}
85
86/// Turns ECEF axes into the globe's world axes, in which latitude and longitude zero lie on
87/// +z, 90°E on +x and the north pole on +y, as
88/// [`crate::projection::globe::lat_lon_to_unit_sphere`] places them.
89pub fn ecef_to_globe_world(ecef: Vector3<f64>) -> Vector3<f64> {
90    Vector3::new(ecef.y, ecef.z, ecef.x)
91}

93/// The solar elevations over which the ground turns from night to day, and how bright the
94/// night side stays.
95#[derive(Clone, Copy, Debug, PartialEq)]
96pub struct TwilightBand {
97    night_end_degrees: f64,
98    day_start_degrees: f64,
99    night_brightness: f32,
100}
101
102impl Default for TwilightBand {
103    /// Night ends at civil dusk, 6° below the horizon, and full day begins with the sun 2°
104    /// above it.
105    fn default() -> Self {
106        Self {
107            night_end_degrees: -6.0,
108            day_start_degrees: 2.0,
109            night_brightness: 0.1,
110        }
111    }
112}
113
114impl TwilightBand {
115    /// A band from `night_end_degrees` to the higher `day_start_degrees` of solar elevation,
116    /// with the night side at `night_brightness` of the day side's light.
117    pub fn new(
118        night_end_degrees: f64,
119        day_start_degrees: f64,
120        night_brightness: f32,
121    ) -> Result<Self, CelestialError> {
122        let ascending = night_end_degrees.is_finite()
123            && day_start_degrees.is_finite()
124            && (-90.0..day_start_degrees).contains(&night_end_degrees)
125            && day_start_degrees <= 90.0;
126        if ascending && (0.0..=1.0).contains(&night_brightness) {
127            Ok(Self {
128                night_end_degrees,
129                day_start_degrees,
130                night_brightness,
131            })
132        } else {
133            Err(CelestialError::InvalidTwilight {
134                night_end_degrees,
135                day_start_degrees,
136                night_brightness,
137            })
138        }
139    }
140
141    /// The sines of the band's two elevations and the night brightness, as the shaders compare
142    /// them with the cosine between the surface normal and the sun.
143    fn uniform(self) -> [f32; 4] {
144        [
145            self.night_end_degrees.to_radians().sin() as f32,
146            self.day_start_degrees.to_radians().sin() as f32,
147            self.night_brightness,
148            0.0,
149        ]
150    }
151}

153/// What the host has said about the sky, kept as a world resource for the render systems.
154#[derive(Clone, Copy, Debug, Default, PartialEq)]
155pub struct CelestialLighting {
156    /// Direction from the Earth towards the sun; the style's light applies while it is unset.
157    pub sun: Option<EcefDirection>,
158    /// Where the ground turns from night to day.
159    pub twilight: TwilightBand,
160    /// Whether the globe's atmosphere is drawn in full whatever the style's `sky` says.
161    pub atmosphere: bool,
162}
163
164impl CelestialLighting {
165    /// The sun direction in globe world axes, with w 1 while a host has set one and all zero
166    /// otherwise, followed by the twilight band, as the terrain shader takes them.
167    pub fn daylight_uniforms(self) -> [[f32; 4]; 2] {
168        let sun = self.sun.map_or([0.0; 4], |sun| {
169            let world = sun.globe_world();
170            [world.x as f32, world.y as f32, world.z as f32, 1.0]
171        });
172        [sun, self.twilight.uniform()]
173    }
174}
175
176/// The host's sun and twilight band, or the defaults while it has set none.
177pub fn lighting(world: &World) -> CelestialLighting {
178    world
179        .resources
180        .get::<CelestialLighting>()
181        .copied()
182        .unwrap_or_default()
183}
184
185/// The sun direction the host set, if any.
186pub fn sun_direction(world: &World) -> Option<EcefDirection> {
187    world
188        .resources
189        .get::<CelestialLighting>()
190        .and_then(|lighting| lighting.sun)
191}
192
193/// Sets the band over which the ground turns from night to day.
194pub fn set_twilight_band(world: &mut World, twilight: TwilightBand) {
195    world
196        .resources
197        .get_or_init_mut::<CelestialLighting>()
198        .twilight = twilight;
199}
200
201/// Draws the globe's atmosphere without a style `sky`, or leaves it to the style.
202pub fn set_atmosphere(world: &mut World, atmosphere: bool) {
203    world
204        .resources
205        .get_or_init_mut::<CelestialLighting>()
206        .atmosphere = atmosphere;
207}
208
209/// Sets or clears the sun direction the render systems read.
210pub fn set_sun_direction(world: &mut World, sun: Option<EcefDirection>) {
211    world.resources.get_or_init_mut::<CelestialLighting>().sun = sun;
212}
