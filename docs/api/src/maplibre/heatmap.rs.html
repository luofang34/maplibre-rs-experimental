1//! Heatmap layers: point kernels accumulated into an offscreen density texture, then coloured
2//! through the layer's ramp and composited in style order.
3//!
4//! The vector tile path tessellates, uploads and buffers heatmap points like circles. Each
5//! frame a density pass draws every visible layer's points into that layer's half-float
6//! target with additive blending, and a fullscreen composite in the main pass turns the
7//! density into colour at the layer's place in the style. With terrain the heatmap still
8//! draws to the screen, and globe rendering is not supported.
9

36/// Registers the `heatmap` layer type; the vector plugin must be registered as well, since it
37/// fetches, tessellates and uploads the points.
38#[derive(Default)]
39pub struct HeatmapPlugin;
40
41impl<E: Environment> Plugin<E> for HeatmapPlugin {
42    fn build(
43        &self,
44        schedule: &mut Schedule,
45        _kernel: Rc<Kernel<E>>,
46        world: &mut World,
47        graph: &mut RenderGraph,
48    ) {
49        world
50            .resources
51            .insert(Eventually::<HeatmapResources>::Uninitialized);
52        world.resources.init::<HeatmapDensityPhase>();
53        schedule.add_system_to_stage(RenderStageLabel::Prepare, resource_system::resource_system);
54        schedule.add_system_to_stage(RenderStageLabel::Prepare, prepare_system::prepare_system);
55        schedule.add_system_to_stage(RenderStageLabel::Queue, queue_system::queue_system);
56
57        let Some(draw_graph) = graph.get_sub_graph_mut(draw_graph::NAME) else {
58            tracing::error!("draw graph is missing; heatmaps will not be drawn");
59            return;
60        };
61        draw_graph.add_node(density_pass::DENSITY_PASS, density_pass::DensityPassNode);
62        let input = draw_graph.input_node().map(|node| node.id);
63        let edges = [
64            input.map(|input| {
65                draw_graph.add_node_edge(NodeLabel::Id(input), density_pass::DENSITY_PASS)
66            }),
67            Some(draw_graph.add_node_edge(density_pass::DENSITY_PASS, draw_graph::node::MAIN_PASS)),
68        ];
69        for edge in edges.into_iter().flatten() {
70            if let Err(error) = edge {
71                tracing::error!(
72                    ?error,
73                    "unable to order the density pass before the main pass"
74                );
75            }
76        }
77    }
78}
