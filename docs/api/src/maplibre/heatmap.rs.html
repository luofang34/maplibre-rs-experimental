1//! Heatmap layers: point kernels accumulated into an offscreen density texture, then coloured
2//! through the layer's ramp and composited in style order.
3//!
4//! The vector tile path tessellates, uploads and buffers heatmap points like circles. Each
5//! frame a density pass draws every visible layer's points into that layer's half-float
6//! target with additive blending, and a fullscreen composite in the main pass turns the
7//! density into colour at the layer's place in the style. With terrain the heatmap still
8//! draws to the screen, and globe rendering is not supported.
9
10use std::rc::Rc;
11

13    environment::Environment,
14    kernel::Kernel,
15    plugin::Plugin,
16    render::{draw_graph, eventually::Eventually, RenderStageLabel},
17    schedule::Schedule,
18    tcs::world::World,
19};

31/// Registers the `heatmap` layer type; the vector plugin must be registered as well, since it
32/// fetches, tessellates and uploads the points.
33#[derive(Default)]
34pub struct HeatmapPlugin;
35
36impl<E: Environment> Plugin<E> for HeatmapPlugin {
37    fn build(
38        &self,
39        schedule: &mut Schedule,
40        _kernel: Rc<Kernel<E>>,
41        world: &mut World,
42        renderer: &mut crate::render::Renderer,
43    ) -> Result<(), crate::plugin::PluginError> {
44        world
45            .resources
46            .insert(Eventually::<HeatmapResources>::Uninitialized);
47        world.resources.init::<HeatmapDensityPhase>();
48        schedule
49            .add_system_to_stage(RenderStageLabel::Prepare, resource_system::resource_system)?;
50        schedule.add_system_to_stage(RenderStageLabel::Prepare, prepare_system::prepare_system)?;
51        schedule.add_system_to_stage(RenderStageLabel::Queue, queue_system::queue_system)?;
52
53        let draw_graph = &mut renderer.render_graph;
54        let edges = [
55            draw_graph.add_node(density_pass::DENSITY_PASS, density_pass::DensityPassNode),
56            draw_graph.add_node_edge(density_pass::DENSITY_PASS, draw_graph::node::MAIN_PASS),
57        ];
58        for edge in edges {
59            edge?;
60        }
61        Ok(())
62    }
63}
