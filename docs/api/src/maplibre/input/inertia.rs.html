1//! Drag-pan inertia with the velocity model and easing curve of GL JS.
2
3use std::{collections::VecDeque, time::Duration};
4
5use cgmath::{InnerSpace, Vector2, Zero};
6use instant::Instant;
7
8/// Samples older than this no longer contribute to the release velocity.
9const BUFFER_CUTOFF: Duration = Duration::from_millis(160);
10const LINEARITY: f64 = 0.3;
11/// Pixels per second squared.
12const DECELERATION: f64 = 2500.0;
13/// Pixels per second.
14const MAX_SPEED: f64 = 1400.0;
15
16struct Motion {
17    started: Instant,
18    duration: Duration,
19    amount: Vector2<f64>,
20    applied: Vector2<f64>,
21}
22
23/// Records pointer movement during a drag and eases the pan out after release.
24#[derive(Default)]
25pub struct PanInertia {
26    samples: VecDeque<(Instant, Vector2<f64>)>,
27    motion: Option<Motion>,
28}
29
30impl PanInertia {
31    /// Whether an inertial ease-out animation is currently running.
32    pub fn is_active(&self) -> bool {
33        self.motion.is_some()
34    }
35
36    /// Records one pointer movement while dragging.
37    pub fn record(&mut self, now: Instant, delta: Vector2<f64>) {
38        self.motion = None;
39        self.drain(now);
40        self.samples.push_back((now, delta));
41    }
42
43    /// Drops recorded movement and any running ease-out.
44    pub fn cancel(&mut self) {
45        self.samples.clear();
46        self.motion = None;
47    }
48
49    /// Starts easing out from the recorded movement; returns whether any motion follows.
50    pub fn release(&mut self, now: Instant) -> bool {
51        self.drain(now);
52        let (Some((first, _)), true) = (self.samples.front(), self.samples.len() >= 2) else {
53            self.samples.clear();
54            return false;
55        };
56        let elapsed = now.saturating_duration_since(*first).as_secs_f64();
57        let travelled: Vector2<f64> = self.samples.iter().map(|(_, delta)| *delta).sum();
58        let distance = travelled.magnitude();
59        self.samples.clear();
60        if elapsed <= 0.0 || distance <= 0.0 {
61            return false;
62        }
63        let Some(easing) = ease_out(distance, elapsed) else {
64            return false;
65        };
66        self.motion = Some(Motion {
67            started: now,
68            duration: easing.duration,
69            amount: travelled * (easing.distance / distance),
70            applied: Vector2::zero(),
71        });
72        true
73    }
74
75    /// Returns the pixel delta to apply this frame, or `None` once the ease-out has finished.
76    pub fn step(&mut self, now: Instant) -> Option<Vector2<f64>> {
77        let motion = self.motion.as_mut()?;
78        let progress = (now.saturating_duration_since(motion.started).as_secs_f64()
79            / motion.duration.as_secs_f64())
80        .min(1.0);
81        let target = motion.amount * ease(progress);
82        let delta = target - motion.applied;
83        motion.applied = target;
84        if progress >= 1.0 {
85            self.motion = None;
86        }
87        Some(delta)
88    }
89
90    fn drain(&mut self, now: Instant) {
91        while self
92            .samples
93            .front()
94            .is_some_and(|(time, _)| now.saturating_duration_since(*time) > BUFFER_CUTOFF)
95        {
96            self.samples.pop_front();
97        }
98    }
99}

101struct EaseOut {
102    duration: Duration,
103    distance: f64,
104}
105
106/// GL JS `calculateEasing`: release speed clamped to `MAX_SPEED`, decelerated to a stop.
107fn ease_out(distance: f64, elapsed_seconds: f64) -> Option<EaseOut> {
108    let speed = (distance * LINEARITY / elapsed_seconds).min(MAX_SPEED);
109    let duration = speed / (DECELERATION * LINEARITY);
110    if !duration.is_finite() || duration <= 0.0 {
111        return None;
112    }
113    Some(EaseOut {
114        duration: Duration::from_secs_f64(duration),
115        distance: speed * duration / 2.0,
116    })
117}

119/// GL JS pan inertia easing: the cubic Bézier `(0, 0, 0.3, 1)`.
120fn ease(progress: f64) -> f64 {
121    unit_bezier(0.0, 0.0, 0.3, 1.0, progress)
122}
123
124fn unit_bezier(p1x: f64, p1y: f64, p2x: f64, p2y: f64, x: f64) -> f64 {
125    let cx = 3.0 * p1x;
126    let bx = 3.0 * (p2x - p1x) - cx;
127    let ax = 1.0 - cx - bx;
128    let cy = 3.0 * p1y;
129    let by = 3.0 * (p2y - p1y) - cy;
130    let ay = 1.0 - cy - by;
131    let sample_x = |t: f64| ((ax * t + bx) * t + cx) * t;
132    let sample_y = |t: f64| ((ay * t + by) * t + cy) * t;
133    let derivative_x = |t: f64| (3.0 * ax * t + 2.0 * bx) * t + cx;
134
135    let mut t = x;
136    for _ in 0..8 {
137        let error = sample_x(t) - x;
138        if error.abs() < 1e-6 {
139            return sample_y(t);
140        }
141        let slope = derivative_x(t);
142        if slope.abs() < 1e-6 {
143            break;
144        }
145        t -= error / slope;
146    }
147    let (mut low, mut high) = (0.0, 1.0);
148    t = x;
149    while low < high {
150        let error = sample_x(t) - x;
151        if error.abs() < 1e-6 {
152            break;
153        }
154        if error > 0.0 {
155            high = t;
156        } else {
157            low = t;
158        }
159        t = (low + high) / 2.0;
160    }
161    sample_y(t)
162}
