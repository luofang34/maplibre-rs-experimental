15pub(crate) struct ShiftHandler {
16    camera_translate: Vector2<f64>,
17    held_up: bool,
18    held_down: bool,
19    held_left: bool,
20    held_right: bool,
21    held_orbit_ccw: bool,
22    held_orbit_cw: bool,
23    speed: f64,

27impl UpdateState for ShiftHandler {
28    fn update_state(&mut self, map_context: &mut MapContext, dt: Duration) {
29        let mut dummy_nav = NavigationState::default();
30        self.update_state_with_nav(map_context, dt, &mut dummy_nav);
31    }
32}

34impl ShiftHandler {
35    pub(crate) fn new(speed: f64, sensitivity: f64) -> Self {
36        Self {
37            camera_translate: Vector2::zero(),
38            held_up: false,
39            held_down: false,
40            held_left: false,
41            held_right: false,
42            held_orbit_ccw: false,
43            held_orbit_cw: false,
44            speed,
45            sensitivity,
46        }
47    }
48
49    /// Whether any directional or orbit key is currently held down or decaying.
50    pub(crate) fn is_active(&self) -> bool {
51        self.held_up
52            || self.held_down
53            || self.held_left
54            || self.held_right
55            || self.held_orbit_ccw
56            || self.held_orbit_cw
57            || self.camera_translate.magnitude2() > 0.01
58    }
59
60    pub(crate) fn update_state_with_nav(
61        &mut self,
62        state: &mut MapContext,
63        dt: Duration,
64        nav: &mut NavigationState,
65    ) {
66        self.update_parts_with_nav(&state.style, &mut state.view_state, dt, nav);
67    }
68
69    pub(crate) fn update_parts_with_nav(
70        &mut self,
71        style: &crate::style::Style,
72        view_state: &mut crate::render::view_state::ViewState,
73        dt: Duration,
74        nav: &mut NavigationState,
75    ) {
76        let dt_secs = dt.as_secs_f64();
77        let boost = if nav.shift_held { 1.85 } else { 1.0 };
78
79        // Continuous held-key velocity (panning or Alt+Arrow orbiting/tilting).
80        let mut dir = Vector2::zero();
81        if self.held_up {
82            dir.y -= 1.0;
83        }
84        if self.held_down {
85            dir.y += 1.0;
86        }
87        if self.held_left {
88            dir.x -= 1.0;
89        }
90        if self.held_right {
91            dir.x += 1.0;
92        }
93
94        if nav.alt_held && (dir.x != 0.0 || dir.y != 0.0) {
95            if dir.x != 0.0 && !nav.north_locked {
96                nav.on_manual_bearing();
97                let current = view_state.camera().get_bearing();
98                let delta = Rad::from(Deg(dir.x * 95.0 * boost * dt_secs));
99                view_state.camera_mut().set_bearing(current + delta);
100            }
101            if dir.y != 0.0 && nav.three_dimensional {
102                nav.on_manual_tilt();
103                let max_pitch: f64 = view_state.camera().max_pitch().0.to_degrees();
104                let current: f64 = view_state.camera().get_pitch().0.to_degrees();
105                let dy: f64 = dir.y;
106                let next = (current - dy * 55.0 * boost * dt_secs).clamp(0.0, max_pitch);
107                set_pitch_keeping_center(view_state, Deg(next));
108            }
109            self.camera_translate = Vector2::zero();
110        } else if dir.magnitude2() > 0.0 {
111            nav.on_manual_pan_or_zoom();
112            let unit = dir.normalize();
113            let continuous_px = unit * (6.2 * self.sensitivity * boost * dt_secs);
114            let center = center_pixel(view_state);
115            let plane = view_state.center_elevation();
116            if !pan_globe_by_pixels(style, view_state, center, -continuous_px) {
117                pan_plane_by_pixels(view_state, center, -continuous_px, plane);
118            }
119            self.camera_translate = Vector2::zero();
120        } else {
121            let decay = (dt_secs / self.speed).min(1.0);
122            let delta = self.camera_translate * decay;
123            self.camera_translate -= delta;
124            if self.camera_translate.magnitude2() <= 0.01 {
125                self.camera_translate = Vector2::zero();
126            }
127            if !delta.is_zero() {
128                let center = center_pixel(view_state);
129                let plane = view_state.center_elevation();
130                if !pan_globe_by_pixels(style, view_state, center, -delta) {
131                    pan_plane_by_pixels(view_state, center, -delta, plane);
132                }
133            }
134        }
135
136        // Continuous Q / E bearing rotation.
137        let mut orbit_dir = 0.0;
138        if self.held_orbit_ccw {
139            orbit_dir -= 1.0;
140        }
141        if self.held_orbit_cw {
142            orbit_dir += 1.0;
143        }
144        if orbit_dir != 0.0 && !nav.north_locked {
145            nav.on_manual_bearing();
146            let current = view_state.camera().get_bearing();
147            let delta = Rad::from(Deg(orbit_dir * 95.0 * boost * dt_secs));
148            view_state.camera_mut().set_bearing(current + delta);
149        }
150    }

152    pub(crate) fn process_key(&mut self, key: NavKey, state: ButtonState) -> bool {
153        let pressed = state == ButtonState::Pressed;
154        let amount = if pressed {
155            10.0 * self.sensitivity
156        } else {
157            0.0
158        };
159        match key {
160            NavKey::Up => {
161                self.held_up = pressed;
162                self.camera_translate.y -= amount;
163                true
164            }
165            NavKey::Down => {
166                self.held_down = pressed;
167                self.camera_translate.y += amount;
168                true
169            }
170            NavKey::Left => {
171                self.held_left = pressed;
172                self.camera_translate.x -= amount;
173                true
174            }
175            NavKey::Right => {
176                self.held_right = pressed;
177                self.camera_translate.x += amount;
178                true
179            }
180            NavKey::OrbitCcw => {
181                self.held_orbit_ccw = pressed;
182                true
183            }
184            NavKey::OrbitCw => {
185                self.held_orbit_cw = pressed;
186                true
187            }
188            _ => false,
189        }
190    }
191}

193#[cfg(test)]
194mod tests {
195    use super::*;
196
197    #[test]
198    fn long_idle_gap_clamps_decay_fraction_without_overshoot_or_sign_flip() {
199        let mut handler = ShiftHandler::new(0.2, 100.0);
200        assert!(handler.process_key(NavKey::Right, ButtonState::Pressed));
201        let initial_x = handler.camera_translate.x;
202        assert!(initial_x > 0.0);
203
204        let factor = (Duration::from_secs(60).as_secs_f64() / handler.speed).min(1.0);
205        let delta = handler.camera_translate * factor;
206        handler.camera_translate -= delta;
207        assert_eq!(delta.x, initial_x);
208        assert_eq!(handler.camera_translate.x, 0.0);
209    }
