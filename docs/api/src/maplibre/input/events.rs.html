3/// Press or release state of a pointer button or keyboard key.
4#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
5pub enum ButtonState {
6    /// The button or key was pressed down.
7    Pressed,
8    /// The button or key was released.
9    Released,
10}
11
12/// Logical pointer or mouse button identifier.
13#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
14pub enum PointerButton {
15    /// Primary pointer button (left mouse button).
16    Primary,
17    /// Secondary pointer button (right mouse button).
18    Secondary,
19    /// Auxiliary / middle pointer button (wheel click).
20    Middle,
21    /// Additional device-specific button index.
22    Other(u16),
23}
24
25/// Scroll-wheel or two-finger trackpad scroll delta in logical coordinates.
26#[derive(Clone, Copy, Debug, PartialEq)]
27pub enum MouseScrollDelta {
28    /// Discrete wheel notch movement measured in lines (`y > 0` scrolls up / zooms in).
29    LineDelta {
30        /// Horizontal line delta.
31        x: f32,
32        /// Vertical line delta.
33        y: f32,
34    },
35    /// Continuous or pixel-stepped scroll delta in logical window pixels.
36    PixelDelta {
37        /// Horizontal logical pixel delta.
38        x: f64,
39        /// Vertical logical pixel delta (`y > 0` zooms in / scrolls down-content).
40        y: f64,
41    },
42}
43
44/// Lifecycle phase of a touch contact or trackpad multi-touch gesture.
45#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
46pub enum TouchPhase {
47    /// Contact or gesture began.
48    Started,
49    /// Contact moved or gesture updated.
50    Moved,
51    /// Contact lifted or gesture ended normally.
52    Ended,
53    /// Contact or gesture was interrupted by the system.
54    Cancelled,
55}

57/// Active keyboard modifier state accompanying pointer, wheel, and key events.
58#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Hash)]
59pub struct ModifiersState {
60    /// Whether `Shift` is held.
61    pub shift: bool,
62    /// Whether `Alt` / `Option` is held.
63    pub alt: bool,
64    /// Whether `Control` / `Command` / `Super` is held.
65    pub ctrl: bool,
66}

68/// Window-system-independent keyboard navigation action.
69#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
70pub enum NavKey {
71    /// Pan north/up (or tilt up when `Alt` is held): `ArrowUp` or `W`.
72    Up,
73    /// Pan south/down (or tilt down when `Alt` is held): `ArrowDown` or `S`.
74    Down,
75    /// Pan west/left (or orbit counter-clockwise when `Alt` is held): `ArrowLeft` or `A`.
76    Left,
77    /// Pan east/right (or orbit clockwise when `Alt` is held): `ArrowRight` or `D`.
78    Right,
79    /// Continuous counter-clockwise bearing orbit (`Q`).
80    OrbitCcw,
81    /// Continuous clockwise bearing orbit (`E`).
82    OrbitCw,
83    /// Smooth zoom in (`I`, `+`, `=`).
84    ZoomIn,
85    /// Smooth zoom out (`K`, `-`, `_`).
86    ZoomOut,
87    /// Toggle between `Map` and `Rts` pointer profiles (`Tab`).
88    TogglePointerProfile,
89    /// Cycle 2D / 3D / Auto-Tilt camera mode (`T`).
90    Cycle3dMode,
91    /// Reset bearing to north or toggle North-Lock (`N`).
92    Compass,
93    /// Fly back to the home camera (`R` or `Home`).
94    Home,
95    /// Show or hide the navigation HUD overlay (`H`).
96    ToggleHud,
97    /// Expand the debug left viewport inset (`V`).
98    DebugInsetLeft,
99    /// Expand the debug top viewport inset (`B`).
100    DebugInsetTop,
101    /// Expand the debug bottom viewport inset.
102    DebugInsetBottom,
103    /// Expand the debug right viewport inset (`M`).
104    DebugInsetRight,
105}

107/// Window-system-independent input event consumed by [`super::InputController`].
108///
109/// All pointer, touch, and pixel-scroll coordinates are expressed in **logical** window pixels
110/// (physical pixels divided by the display scale factor).
111#[derive(Clone, Copy, Debug, PartialEq)]
112pub enum InputEvent {
113    /// Active keyboard modifier flags changed.
114    ModifiersChanged(ModifiersState),
115    /// Pointer / cursor moved to `position` in logical window pixels.
116    PointerMoved {
117        /// Logical `(x, y)` coordinates in window pixels.
118        position: [f64; 2],
119    },
120    /// Pointer / mouse button pressed or released.
121    PointerButton {
122        /// Button that changed state.
123        button: PointerButton,
124        /// Press or release state.
125        state: ButtonState,
126    },
127    /// Scroll wheel or two-finger trackpad scroll delta.
128    MouseWheel {
129        /// Scroll delta in lines or logical pixels.
130        delta: MouseScrollDelta,
131    },
132    /// Multi-touch contact event in logical window pixels.
133    Touch {
134        /// Unique identifier of the touch contact.
135        id: u64,
136        /// Lifecycle phase of the touch contact.
137        phase: TouchPhase,
138        /// Logical `(x, y)` coordinates in window pixels.
139        position: [f64; 2],
140    },
141    /// Native trackpad pinch magnification gesture.
142    PinchGesture {
143        /// Incremental magnification delta (`> 0` spreading apart, `< 0` pinching together).
144        delta: f64,
145        /// Lifecycle phase of the pinch gesture.
146        phase: TouchPhase,
147    },
148    /// Native trackpad two-finger rotation gesture in degrees.
149    RotationGesture {
150        /// Incremental rotation angle in degrees.
151        delta_deg: f64,
152    },
153    /// Smart-zoom / double-tap gesture at the current pointer position.
154    DoubleTapGesture,
155    /// Continuous trackpad pinch magnification step without explicit start/end phases
156    /// (used for WebKit `GestureEvent` and browser `ctrlKey + wheel` streams).
157    WebPinchWheel {
158        /// Incremental magnification delta.
159        magnification: f64,
160    },
161    /// Logical navigation key pressed or released.
162    Key {
163        /// Logical navigation key.
164        key: NavKey,
165        /// Press or release state.
166        state: ButtonState,
167    },
168}
