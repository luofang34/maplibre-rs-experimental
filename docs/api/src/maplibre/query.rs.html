49/// A feature drawn by a style layer, in the shape of GL JS `MapGeoJSONFeature`.
50#[derive(Clone, Debug, PartialEq, Serialize)]
51pub struct QueriedFeature {
52    /// Style layer id.
53    pub layer: String,
54    /// Style source id.
55    pub source: Option<String>,
56    /// Layer inside the source.
57    pub source_layer: String,
58    /// The feature's id, when the source assigned a whole-number one.
59    pub id: Option<u64>,
60    /// The feature's attributes.
61    pub properties: BTreeMap<String, serde_json::Value>,
62    /// Runtime `feature-state` entries currently assigned to this feature.
63    pub state: BTreeMap<String, serde_json::Value>,
64    /// `Point` for a symbol, `Polygon` for a fill and `LineString` for a line.
65    pub geometry_type: &'static str,
66    /// The rendered text of a symbol.
67    pub text: Option<String>,
68}

70impl From<RenderedSymbol> for QueriedFeature {
71    fn from(symbol: RenderedSymbol) -> Self {
72        Self {
73            layer: symbol.layer,
74            source: symbol.source,
75            source_layer: symbol.source_layer,
76            id: symbol.id,
77            properties: symbol.properties,
78            state: symbol.state,
79            geometry_type: symbol.geometry_type.name(),
80            text: Some(symbol.text),
81        }
82    }
83}

87/// Features under `geometry`, topmost style layer first, across flat Mercator, 3D terrain, and
88/// 3D globe / Mercator-to-globe transition views.
89///
90/// Symbols use their screen placement; vector layers use the per-tile [`crate::io::geometry_index::GeometryIndex`]
91/// and resolve surface intersections through [`ResolvedFrameView`]. A point in the sky or beyond
92/// the globe limb returns nothing.
93pub fn query_rendered_features(
94    world: &World,
95    style: &Style,
96    view_state: &ViewState,
97    geometry: QueryGeometry,
98    options: &QueryOptions,
99) -> Result<Vec<QueriedFeature>, QueryError> {
100    let symbols = query_rendered_symbols_in(world, style, geometry, options)?;
101    let mut found: Vec<(u32, Option<f64>, QueriedFeature)> = symbols
102        .into_iter()
103        .map(|symbol| {
104            let index = style
105                .layers
106                .iter()
107                .find(|layer| layer.id == symbol.layer)
108                .map_or(0, |layer| layer.index);
109            (index, None, QueriedFeature::from(symbol))
110        })
111        .collect();
112    found.extend(vector_features(
113        world, style, view_state, geometry, options,
114    )?);
115    Ok(in_drawing_order(found))
116}

118/// A feature with its layer's index and, for a fill extrusion, its depth on screen.
119type Found = (u32, Option<f64>, QueriedFeature);
120
121/// Where a feature comes in GL JS's top-down order within its layer.
122type TopDown = (u8, i32, i32, i32, Reverse<u32>);
123
124/// What one query keeps while it walks tiles.
125struct Search<'a> {
126    candidates: Vec<Candidate<'a>>,
127    query_filter: Option<Filter>,
128    /// Features already reported, per layer and per tile: GL JS reports a feature once for each
129    /// tile it is found in, merging only the copies of one tile in different worlds.
130    seen: HashSet<(String, WorldTileCoords, Identity)>,
131    /// Each feature with its layer's index and, for a fill extrusion, its depth on screen.
132    found: Vec<(TopDown, Found)>,
133    /// The query in window pixels, which extrusions and viewport circles are met against.
134    screen: extrusion::ScreenQuery,
135    /// The corners of the ground the query covers, in world pixels.
136    ground: Vec<[f64; 2]>,
137    /// Whether the query box surrounds the entire visible globe disc in space.
138    whole_visible_hemisphere: bool,
139    world: &'a World,
140    resolved: &'a ResolvedFrameView,
141    terrain_index: Option<&'a TerrainCoverageIndex>,
142}
143
144fn vector_features(
145    world: &World,
146    style: &Style,
147    view_state: &ViewState,
148    geometry: QueryGeometry,
149    options: &QueryOptions,
150) -> Result<Vec<(u32, Option<f64>, QueriedFeature)>, QueryError> {
151    let zoom = view_state.zoom();
152    let bearing = view_state.camera().get_bearing().0;
153    let candidates: Vec<Candidate> = style
154        .layers
155        .iter()
156        .filter(|layer| {
157            matches!(
158                layer.type_.as_str(),
159                "fill" | "line" | "fill-extrusion" | "circle" | "heatmap"
160            ) && layer.is_visible_at(zoom.value())
161                && options
162                    .layers
163                    .as_ref()
164                    .is_none_or(|layers| layers.contains(&layer.id))
165        })
166        .filter_map(|layer| Candidate::new(layer, zoom.value(), bearing))
167        .collect();
168    if candidates.is_empty() {
169        return Ok(Vec::new());
170    }
171    let query_filter = options
172        .filter
173        .as_ref()
174        .map(Filter::parse)
175        .transpose()
176        .map_err(QueryError::InvalidFilter)?;
177    let Some(bounds) = geometry.bounds() else {
178        return Err(QueryError::InvalidGeometry);
179    };
180    let Ok(resolved) = ResolvedFrameView::get_or_resolve(world, style, view_state) else {
181        return Ok(Vec::new());
182    };
183    let terrain_index = resolve_terrain_index(world, style, &resolved);
184
185    // An extrusion stands up from the ground, so it can meet the query far from where the
186    // query meets the ground: every tile between the camera nadir and the query is searched.
187    let extrudes = candidates
188        .iter()
189        .any(|candidate| candidate.layer.type_ == "fill-extrusion");
190    let margin = candidates
191        .iter()
192        .map(Candidate::margin_pixels)
193        .fold(0.0, f64::max);
194
195    let (region, ground, whole_visible_hemisphere) = if extrudes {
196        let searched = camera_query_bounds(view_state, &resolved, bounds);
197        let Some(searched_fp) = surface_footprint(
198            world,
199            view_state,
200            &resolved,
201            terrain_index.as_deref(),
202            searched,
203        ) else {
204            return Ok(Vec::new());
205        };
206        let query_fp = surface_footprint(
207            world,
208            view_state,
209            &resolved,
210            terrain_index.as_deref(),
211            bounds,
212        );
213        let (corners, whole_hemi) = query_fp
214            .map(|fp| (fp.corners, fp.whole_visible_hemisphere))
215            .unwrap_or_default();
216        (searched_fp.region, corners, whole_hemi)
217    } else {
218        let Some(fp) = surface_footprint(
219            world,
220            view_state,
221            &resolved,
222            terrain_index.as_deref(),
223            bounds,
224        ) else {
225            return Ok(Vec::new());
226        };
227        (fp.region, fp.corners, fp.whole_visible_hemisphere)
228    };
229
230    let screen = match geometry {
231        QueryGeometry::Point(point) => extrusion::ScreenQuery::Point(point),
232        QueryGeometry::Box { .. } => extrusion::ScreenQuery::Box(bounds),
233    };
234    let mut search = Search {
235        candidates,
236        query_filter,
237        seen: HashSet::new(),
238        found: Vec::new(),
239        screen,
240        ground,
241        whole_visible_hemisphere,
242        world,
243        resolved: &resolved,
244        terrain_index: terrain_index.as_deref(),
245    };
246    search_tiles(world, (region, margin), &mut search);
247    let mut found = search.found;
248    found.sort_by_key(|(order, _)| *order);
249    Ok(found.into_iter().map(|(_, feature)| feature).collect())
250}

252/// Collects from the finest level of tiles that holds any index for the region, looking
253/// `margin` screen pixels beyond the query for features a layer draws within reach of it.
254fn search_tiles(world: &World, (region, margin): ([f64; 4], f64), search: &mut Search) {
255    let zoom = search.resolved.zoom;
256    // One level above the view zoom covers sources whose tiles are finer than the view's.
257    let top = (zoom.value().floor() as i32 + 1).clamp(0, MAX_QUERY_ZOOM);
258    let Some(z) = (0..=top).rev().find(|&z| {
259        tiles_in(region, margin, search.resolved, z as u8)
260            .iter()
261            .any(|tile| {
262                world
263                    .tiles
264                    .geometry_index
265                    .tile_indexes(&tile.coords)
266                    .is_some()
267            })
268    }) else {
269        return;
270    };
271    // A source that stops before the view's zoom is queried as GL JS queries it: through the
272    // tiles of the view's zoom cut out of the coarser one.
273    let view_z = (zoom.value().floor() as i32).clamp(0, MAX_QUERY_ZOOM);
274    let tile_z = z.max(view_z) as u8;
275    for mut tile in tiles_in(region, margin, search.resolved, tile_z) {
276        let depth = tile_z - z as u8;
277        let source_coords = WorldTileCoords {
278            x: tile.coords.x >> depth,
279            y: tile.coords.y >> depth,
280            z: ZoomLevel::new(z as u8),
281        };
282        let Some(indexes) = world.tiles.geometry_index.tile_indexes(&source_coords) else {
283            continue;
284        };
285        tile.footprint = if search.whole_visible_hemisphere {
286            ground::Footprint::of(&[[0.0, 0.0], [EXTENT, 0.0], [EXTENT, EXTENT], [0.0, EXTENT]])
287        } else {
288            tile.footprint_of(&search.ground)
289        };
290        let tile = &tile;
291        let scale = f64::from(1_u32 << depth);
292        let offset = [
293            f64::from(tile.coords.x - (source_coords.x << depth)) * EXTENT,
294            f64::from(tile.coords.y - (source_coords.y << depth)) * EXTENT,
295        ];
296        let widen = margin * tile.units_per_pixel;
297        let [x0, y0, x1, y1] = tile.local;
298        // The window in the units of the tile the index belongs to.
299        let child = [x0 - widen, y0 - widen, x1 + widen, y1 + widen];
300        let window = [0, 1, 2, 3].map(|i| (child[i] + offset[i % 2]) / scale);
301        for (source, index) in indexes {
302            for geometry in candidates_in(index, window) {
303                if depth == 0 {
304                    collect(geometry, source, tile, search);
305                    continue;
306                }
307                for part in overzoomed::slice(geometry, scale, offset) {
308                    collect(&part, source, tile, search);
309                }
310            }
311        }
312    }
313}

315/// Whether a fill or line feature meets the query in this tile.
316fn flat_hit(
317    geometry: &IndexedGeometry<f64>,
318    tile: &QueryTile,
319    candidate: &Candidate,
320    zoom: f64,
321) -> bool {
322    let Some(footprint) = &tile.footprint else {
323        return false;
324    };
325    let line = match &candidate.layer.paint {
326        Some(LayerPaint::Line(paint)) => ground::line_reach(paint, &geometry.properties, zoom)
327            .map(|pixels| pixels * tile.units_per_pixel),
328        _ => [0.0; 2],
329    };
330    let translate = candidate
331        .translate
332        .map(|pixels| pixels * tile.units_per_pixel);
333    ground::touches(
334        geometry,
335        footprint,
336        (candidate.layer.type_.as_str(), translate),
337        line,
338    )
339}

341fn collect(
342    geometry: &IndexedGeometry<f64>,
343    source: Option<&str>,
344    tile: &QueryTile,
345    search: &mut Search,
346) {
347    let (kind, geometry_type) = match &geometry.exact {
348        ExactGeometry::Polygon(_) => (GeometryType::Polygon, "Polygon"),
349        ExactGeometry::LineString(_) => (GeometryType::LineString, "LineString"),
350        ExactGeometry::Point(_) => (GeometryType::Point, "Point"),
351    };
352    let id = geometry.id.map(|id| Value::Number(id as f64));
353    let context = FeatureContext {
354        properties: &geometry.properties,
355        geometry_type: kind,
356        id,
357        zoom: f64::from(tile.zoom_level),
358    };
359    let Search {
360        candidates,
361        query_filter,
362        seen,
363        found,
364        screen,
365        whole_visible_hemisphere,
366        world,
367        resolved,
368        terrain_index,
369        ..
370    } = search;
371    let zoom_val = resolved.zoom.value();
372    for candidate in candidates.iter() {
373        let layer = candidate.layer;
374        if layer.source.as_deref() != source
375            || layer.source_layer.as_deref().unwrap_or(GEOJSON_LAYER) != &*geometry.source_layer
376            || candidate
377                .filter
378                .as_ref()
379                .is_some_and(|filter| !filter.evaluate(&context))
380            || query_filter
381                .as_ref()
382                .is_some_and(|filter| !filter.evaluate(&context))
383        {
384            continue;
385        }
386        let visible_check = |cand: &Candidate| {
387            !*whole_visible_hemisphere
388                || visible_in_screen_box(
389                    geometry,
390                    tile.origin,
391                    tile.world_per_unit,
392                    cand.translate,
393                    (world, resolved, *terrain_index, *screen),
394                )
395        };
396        let depth = match (&geometry.exact, candidate.layer.type_.as_str()) {
397            (ExactGeometry::Polygon(polygon), "fill-extrusion") => {
398                match extrusion::extruded_depth(
399                    polygon,
400                    tile,
401                    candidate,
402                    (
403                        &geometry.properties,
404                        *screen,
405                        world,
406                        resolved,
407                        *terrain_index,
408                    ),
409                ) {
410                    Some(depth) => Some(depth),
411                    None => continue,
412                }
413            }
414            (exact, "heatmap") => match &candidate.layer.paint {
415                Some(LayerPaint::Heatmap(paint))
416                    if circle::heatmap_hit(
417                        exact,
418                        tile,
419                        (paint, &geometry.properties),
420                        zoom_val,
421                    ) && visible_check(candidate) =>
422                {
423                    None
424                }
425                _ => continue,
426            },
427            (exact, "circle") => match &candidate.layer.paint {
428                Some(LayerPaint::Circle(paint))
429                    if circle::geometry_hit(
430                        exact,
431                        (tile, candidate),
432                        (paint, &geometry.properties),
433                        (*screen, world, resolved, *terrain_index),
434                    ) =>
435                {
436                    None
437                }
438                _ => continue,
439            },
440            _ if flat_hit(geometry, tile, candidate, zoom_val) && visible_check(candidate) => None,
441            _ => continue,
442        };
443        let identity = match geometry.id {
444            Some(id) => Identity::Id(id),
445            None => Identity::Part(std::sync::Arc::as_ptr(&geometry.properties) as usize),
446        };
447        if !seen.insert((layer.id.clone(), tile.coords, identity)) {
448            continue;
449        }
450        let order = (
451            u8::from(tile.coords.z),
452            tile.coords.y,
453            tile.wrap,
454            tile.coords.x,
455            Reverse(geometry.feature_index),
456        );
457        found.push((
458            order,
459            (
460                layer.index,
461                depth,
462                QueriedFeature {
463                    layer: layer.id.clone(),
464                    source: layer.source.clone(),
465                    source_layer: geometry.source_layer.to_string(),
466                    id: geometry.id,
467                    properties: source_properties(&geometry.properties),
468                    state: crate::style::feature_state::extract_feature_state(&geometry.properties),
469                    geometry_type,
470                    text: None,
471                },
472            ),
473        ));
474    }
