46/// A feature drawn by a style layer, in the shape of GL JS `MapGeoJSONFeature`.
47#[derive(Clone, Debug, PartialEq, Serialize)]
48pub struct QueriedFeature {
49    /// Style layer id.
50    pub layer: String,
51    /// Style source id.
52    pub source: Option<String>,
53    /// Layer inside the source.
54    pub source_layer: String,
55    /// The feature's id, when the source assigned a whole-number one.
56    pub id: Option<u64>,
57    /// The feature's attributes.
58    pub properties: BTreeMap<String, serde_json::Value>,
59    /// `Point` for a symbol, `Polygon` for a fill and `LineString` for a line.
60    pub geometry_type: &'static str,
61    /// The rendered text of a symbol.
62    pub text: Option<String>,
63}

65impl From<RenderedSymbol> for QueriedFeature {
66    fn from(symbol: RenderedSymbol) -> Self {
67        Self {
68            layer: symbol.layer,
69            source: symbol.source,
70            source_layer: symbol.source_layer,
71            id: symbol.id,
72            properties: symbol.properties,
73            geometry_type: symbol.geometry_type.name(),
74            text: Some(symbol.text),
75        }
76    }
77}

79/// A feature's properties as its source gives them, without the feature state an expression
80/// reads through them.
81pub(crate) fn source_properties(
82    properties: &crate::style::expression::FeatureProperties,
83) -> BTreeMap<String, serde_json::Value> {
84    properties
85        .iter()
86        .filter(|(key, _)| !key.starts_with(crate::style::expression::FEATURE_STATE_PREFIX))
87        .map(|(key, value)| (key.clone(), value.to_json()))
88        .collect()
89}

91/// Features under `geometry`, topmost style layer first.
92///
93/// Symbols use their placement; fill and line features are located on the ground plane of a flat
94/// view and need the geometry index, which only native workers send. A fill or line query in a
95/// view with terrain or an active globe is refused rather than answered wrongly, and a point in
96/// the sky returns nothing.
97pub fn query_rendered_features(
98    world: &World,
99    style: &Style,
100    view_state: &ViewState,
101    geometry: QueryGeometry,
102    options: &QueryOptions,
103) -> Result<Vec<QueriedFeature>, QueryError> {
104    let symbols = query_rendered_symbols_in(world, style, geometry, options)?;
105    let mut found: Vec<(u32, Option<f64>, QueriedFeature)> = symbols
106        .into_iter()
107        .map(|symbol| {
108            let index = style
109                .layers
110                .iter()
111                .find(|layer| layer.id == symbol.layer)
112                .map_or(0, |layer| layer.index);
113            (index, None, QueriedFeature::from(symbol))
114        })
115        .collect();
116    found.extend(vector_features(
117        world, style, view_state, geometry, options,
118    )?);
119    Ok(in_drawing_order(found))
120}

122/// Features topmost first, as GL JS `Style.queryRenderedFeatures` orders them: layer by layer
123/// from the top, where each fill extrusion layer gives out the nearest remaining extrusions
124/// for as long as the nearest is in that layer or above it.
125fn in_drawing_order(found: Vec<(u32, Option<f64>, QueriedFeature)>) -> Vec<QueriedFeature> {
126    let (mut extruded, flat): (Vec<_>, Vec<_>) =
127        found.into_iter().partition(|(_, depth, _)| depth.is_some());
128    // Nearest last, so the nearest is popped first.
129    extruded.sort_by(|a, b| b.1.unwrap_or(0.0).total_cmp(&a.1.unwrap_or(0.0)));
130    let extrusion_layers: HashSet<u32> = extruded.iter().map(|(layer, _, _)| *layer).collect();
131    let mut layers: Vec<u32> = flat
132        .iter()
133        .map(|(layer, _, _)| *layer)
134        .chain(extrusion_layers.iter().copied())
135        .collect();
136    layers.sort_unstable_by_key(|layer| Reverse(*layer));
137    layers.dedup();
138    let mut flat_by_layer: BTreeMap<u32, Vec<QueriedFeature>> = BTreeMap::new();
139    for (layer, _, feature) in flat {
140        flat_by_layer.entry(layer).or_default().push(feature);
141    }
142    let mut ordered = Vec::new();
143    for layer in layers {
144        if extrusion_layers.contains(&layer) {
145            while extruded
146                .last()
147                .is_some_and(|(nearest, _, _)| *nearest >= layer)
148            {
149                if let Some((_, _, feature)) = extruded.pop() {
150                    ordered.push(feature);
151                }
152            }
153        }
154        if let Some(features) = flat_by_layer.remove(&layer) {
155            ordered.extend(features);
156        }
157    }
158    ordered
159}

161/// A feature with its layer's index and, for a fill extrusion, its depth on screen.
162type Found = (u32, Option<f64>, QueriedFeature);
163
164/// Where a feature comes in GL JS's top-down order within its layer.
165type TopDown = (u8, i32, i32, i32, Reverse<u32>);

167/// A layer a fill or line feature can be reported for, with its filter parsed once.
168struct Candidate<'a> {
169    layer: &'a StyleLayer,
170    filter: Option<Filter>,
171    /// How far from a line, in screen pixels, its features can be drawn and still be met: half
172    /// its width and its offset.
173    reach_pixels: f64,
174    /// Where the layer draws its features from where they are, in screen pixels along the
175    /// map's axes: its `*-translate`.
176    translate: [f64; 2],
177}
178
179impl<'a> Candidate<'a> {
180    /// A layer whose filter cannot be parsed draws nothing, so it also matches nothing.
181    fn new(layer: &'a StyleLayer, zoom: f64, bearing: f64) -> Option<Self> {
182        let filter = match &layer.filter {
183            Some(filter) => Some(Filter::parse(filter).ok()?),
184            None => None,
185        };
186        let reach_pixels = match &layer.paint {
187            Some(LayerPaint::Line(paint)) => line_margin_pixels(paint, zoom),
188            Some(LayerPaint::Circle(paint)) => circle::margin_pixels(paint, zoom),
189            Some(LayerPaint::Heatmap(paint)) => circle::heatmap_margin_pixels(paint, zoom),
190            _ => 0.0,
191        };
192        Some(Self {
193            layer,
194            filter,
195            reach_pixels,
196            translate: layer_translate_pixels(layer.paint.as_ref(), zoom, bearing),
197        })
198    }
199
200    /// How far from the query, in screen pixels, a feature the layer draws there can lie.
201    fn margin_pixels(&self) -> f64 {
202        self.reach_pixels + self.translate[0].hypot(self.translate[1])
203    }
204}

206/// The farthest a line layer reaches from its features: half its width and its offset, at their
207/// widest where they vary by feature.
208fn line_margin_pixels(paint: &LinePaint, zoom: f64) -> f64 {
209    // A width or offset set by feature is not known before the feature is; this much covers
210    // what styles give them.
211    const BY_FEATURE: f64 = 64.0;
212    let [half_width, offset] = ground::line_reach(paint, &Default::default(), zoom);
213    let varies = [&paint.line_width, &paint.line_gap_width, &paint.line_offset]
214        .into_iter()
215        .flatten()
216        .any(|property| !property.is_feature_constant());
217    half_width + offset.abs() + if varies { BY_FEATURE } else { 0.0 }
218}

220/// What one query keeps while it walks tiles.
221struct Search<'a> {
222    candidates: Vec<Candidate<'a>>,
223    query_filter: Option<Filter>,
224    /// Features already reported, per layer and per tile: GL JS reports a feature once for each
225    /// tile it is found in, merging only the copies of one tile in different worlds.
226    seen: HashSet<(String, WorldTileCoords, Identity)>,
227    /// Each feature with its layer's index and, for a fill extrusion, its depth on screen.
228    /// Each with GL JS's order within its layer: tiles from the lowest zoom, then top to bottom,
229    /// west copy of the world first, and west to east; then the features of a tile from the last
230    /// one in its data.
231    found: Vec<(TopDown, Found)>,
232    /// The query in window pixels, which extrusions are met against as they stand.
233    screen: extrusion::ScreenQuery,
234    /// The corners of the ground the query covers, in world pixels.
235    ground: Vec<[f64; 2]>,
236    view_state: &'a ViewState,
237}

239/// Fill and line features are located on a flat ground plane.
240fn refuse_unsupported_view(style: &Style, zoom: f64) -> Result<(), QueryError> {
241    if style.terrain.is_some() {
242        return Err(QueryError::UnsupportedView { reason: "terrain" });
243    }
244    if style
245        .projection
246        .as_ref()
247        .is_some_and(|projection| projection.projection_type.globe_transition(zoom) != 0.0)
248    {
249        return Err(QueryError::UnsupportedView {
250            reason: "an active globe",
251        });
252    }
253    Ok(())
254}

256fn vector_features(
257    world: &World,
258    style: &Style,
259    view_state: &ViewState,
260    geometry: QueryGeometry,
261    options: &QueryOptions,
262) -> Result<Vec<(u32, Option<f64>, QueriedFeature)>, QueryError> {
263    let zoom = view_state.zoom();
264    let bearing = view_state.camera().get_bearing().0;
265    let candidates: Vec<Candidate> = style
266        .layers
267        .iter()
268        .filter(|layer| {
269            matches!(
270                layer.type_.as_str(),
271                "fill" | "line" | "fill-extrusion" | "circle" | "heatmap"
272            ) && layer.is_visible_at(zoom.value())
273                && options
274                    .layers
275                    .as_ref()
276                    .is_none_or(|layers| layers.contains(&layer.id))
277        })
278        .filter_map(|layer| Candidate::new(layer, zoom.value(), bearing))
279        .collect();
280    if candidates.is_empty() {
281        return Ok(Vec::new());
282    }
283    refuse_unsupported_view(style, zoom.value())?;
284    let query_filter = options
285        .filter
286        .as_ref()
287        .map(Filter::parse)
288        .transpose()
289        .map_err(QueryError::InvalidFilter)?;
290    let Some(bounds) = geometry.bounds() else {
291        return Err(QueryError::InvalidGeometry);
292    };
293    // An extrusion stands up from the ground, so it can meet the query far from where the
294    // query meets the ground: every tile in view is searched.
295    let extrudes = candidates
296        .iter()
297        .any(|candidate| candidate.layer.type_ == "fill-extrusion");
298    let margin = candidates
299        .iter()
300        .map(Candidate::margin_pixels)
301        .fold(0.0, f64::max);
302    let searched = if extrudes {
303        camera_query_bounds(view_state, bounds)
304    } else {
305        bounds
306    };
307    let Some(region) = ground_region(view_state, searched) else {
308        return Ok(Vec::new());
309    };
310    let screen = match geometry {
311        QueryGeometry::Point(point) => extrusion::ScreenQuery::Point(point),
312        QueryGeometry::Box { .. } => extrusion::ScreenQuery::Box(bounds),
313    };
314    let mut search = Search {
315        candidates,
316        query_filter,
317        seen: HashSet::new(),
318        found: Vec::new(),
319        screen,
320        ground: ground_corners(view_state, bounds),
321        view_state,
322    };
323    search_tiles(world, (region, zoom, margin), &mut search);
324    let mut found = search.found;
325    found.sort_by_key(|(order, _)| *order);
326    Ok(found.into_iter().map(|(_, feature)| feature).collect())
327}

329/// Collects from the finest level of tiles that holds any index for the region, looking
330/// `margin` screen pixels beyond the query for features a layer draws within reach of it.
331fn search_tiles(world: &World, (region, zoom, margin): ([f64; 4], Zoom, f64), search: &mut Search) {
332    // One level above the view zoom covers sources whose tiles are finer than the view's.
333    let top = (zoom.value().floor() as i32 + 1).clamp(0, MAX_QUERY_ZOOM);
334    let Some(z) = (0..=top).rev().find(|&z| {
335        tiles_in(region, margin, zoom, z as u8).iter().any(|tile| {
336            world
337                .tiles
338                .geometry_index
339                .tile_indexes(&tile.coords)
340                .is_some()
341        })
342    }) else {
343        return;
344    };
345    // A source that stops before the view's zoom is queried as GL JS queries it: through the
346    // tiles of the view's zoom cut out of the coarser one.
347    let view_z = (zoom.value().floor() as i32).clamp(0, MAX_QUERY_ZOOM);
348    let tile_z = z.max(view_z) as u8;
349    for mut tile in tiles_in(region, margin, zoom, tile_z) {
350        let depth = tile_z - z as u8;
351        let source_coords = WorldTileCoords {
352            x: tile.coords.x >> depth,
353            y: tile.coords.y >> depth,
354            z: ZoomLevel::new(z as u8),
355        };
356        let Some(indexes) = world.tiles.geometry_index.tile_indexes(&source_coords) else {
357            continue;
358        };
359        tile.footprint = tile.footprint_of(&search.ground);
360        let tile = &tile;
361        let scale = f64::from(1_u32 << depth);
362        let offset = [
363            f64::from(tile.coords.x - (source_coords.x << depth)) * EXTENT,
364            f64::from(tile.coords.y - (source_coords.y << depth)) * EXTENT,
365        ];
366        let widen = margin * tile.units_per_pixel;
367        let [x0, y0, x1, y1] = tile.local;
368        // The window in the units of the tile the index belongs to.
369        let child = [x0 - widen, y0 - widen, x1 + widen, y1 + widen];
370        let window = [0, 1, 2, 3].map(|i| (child[i] + offset[i % 2]) / scale);
371        for (source, index) in indexes {
372            for geometry in candidates_in(index, window) {
373                if depth == 0 {
374                    collect(geometry, source, tile, search);
375                    continue;
376                }
377                for part in overzoomed::slice(geometry, scale, offset) {
378                    collect(&part, source, tile, search);
379                }
380            }
381        }
382    }
383}

385/// Whether a fill or line feature meets the query in this tile.
386fn flat_hit(
387    geometry: &IndexedGeometry<f64>,
388    tile: &QueryTile,
389    candidate: &Candidate,
390    zoom: f64,
391) -> bool {
392    let Some(footprint) = &tile.footprint else {
393        return false;
394    };
395    let line = match &candidate.layer.paint {
396        Some(LayerPaint::Line(paint)) => ground::line_reach(paint, &geometry.properties, zoom)
397            .map(|pixels| pixels * tile.units_per_pixel),
398        _ => [0.0; 2],
399    };
400    let translate = candidate
401        .translate
402        .map(|pixels| pixels * tile.units_per_pixel);
403    ground::touches(
404        geometry,
405        footprint,
406        (candidate.layer.type_.as_str(), translate),
407        line,
408    )
409}

411fn collect(
412    geometry: &IndexedGeometry<f64>,
413    source: Option<&str>,
414    tile: &QueryTile,
415    search: &mut Search,
416) {
417    let (kind, geometry_type) = match &geometry.exact {
418        ExactGeometry::Polygon(_) => (GeometryType::Polygon, "Polygon"),
419        ExactGeometry::LineString(_) => (GeometryType::LineString, "LineString"),
420        ExactGeometry::Point(_) => (GeometryType::Point, "Point"),
421    };
422    let id = geometry.id.map(|id| Value::Number(id as f64));
423    let context = FeatureContext {
424        properties: &geometry.properties,
425        geometry_type: kind,
426        id,
427        zoom: f64::from(tile.zoom_level),
428    };
429    let Search {
430        candidates,
431        query_filter,
432        seen,
433        found,
434        screen,
435        view_state,
436        ..
437    } = search;
438    for candidate in candidates.iter() {
439        let layer = candidate.layer;
440        if layer.source.as_deref() != source
441            || layer.source_layer.as_deref().unwrap_or(GEOJSON_LAYER) != &*geometry.source_layer
442            || candidate
443                .filter
444                .as_ref()
445                .is_some_and(|filter| !filter.evaluate(&context))
446            || query_filter
447                .as_ref()
448                .is_some_and(|filter| !filter.evaluate(&context))
449        {
450            continue;
451        }
452        let depth = match (&geometry.exact, candidate.layer.type_.as_str()) {
453            (ExactGeometry::Polygon(polygon), "fill-extrusion") => {
454                match extrusion::extruded_depth(
455                    polygon,
456                    tile,
457                    candidate,
458                    (&geometry.properties, *screen, view_state),
459                ) {
460                    Some(depth) => Some(depth),
461                    None => continue,
462                }
463            }
464            (exact, "heatmap") => match &candidate.layer.paint {
465                Some(LayerPaint::Heatmap(paint))
466                    if circle::heatmap_hit(
467                        exact,
468                        tile,
469                        (paint, &geometry.properties),
470                        view_state.zoom().value(),
471                    ) =>
472                {
473                    None
474                }
475                _ => continue,
476            },
477            (exact, "circle") => match &candidate.layer.paint {
478                Some(LayerPaint::Circle(paint))
479                    if circle::geometry_hit(
480                        exact,
481                        (tile, candidate),
482                        (paint, &geometry.properties),
483                        (*screen, view_state),
484                    ) =>
485                {
486                    None
487                }
488                _ => continue,
489            },
490            _ if flat_hit(geometry, tile, candidate, view_state.zoom().value()) => None,
491            _ => continue,
492        };
493        let identity = match geometry.id {
494            Some(id) => Identity::Id(id),
495            None => Identity::Part(std::sync::Arc::as_ptr(&geometry.properties) as usize),
496        };
497        if !seen.insert((layer.id.clone(), tile.coords, identity)) {
498            continue;
499        }
500        let order = (
501            u8::from(tile.coords.z),
502            tile.coords.y,
503            tile.wrap,
504            tile.coords.x,
505            Reverse(geometry.feature_index),
506        );
507        found.push((
508            order,
509            (
510                layer.index,
511                depth,
512                QueriedFeature {
513                    layer: layer.id.clone(),
514                    source: layer.source.clone(),
515                    source_layer: geometry.source_layer.to_string(),
516                    id: geometry.id,
517                    properties: source_properties(&geometry.properties),
518                    geometry_type,
519                    text: None,
520                },
521            ),
522        ));
523    }
524}
