15/// Features topmost first, as GL JS `Style.queryRenderedFeatures` orders them: layer by layer
16/// from the top, where each fill extrusion layer gives out the nearest remaining extrusions
17/// for as long as the nearest is in that layer or above it.
18pub(super) fn in_drawing_order(
19    found: Vec<(u32, Option<f64>, QueriedFeature)>,
20) -> Vec<QueriedFeature> {
21    let (mut extruded, flat): (Vec<_>, Vec<_>) =
22        found.into_iter().partition(|(_, depth, _)| depth.is_some());
23    // Nearest last, so the nearest is popped first.
24    extruded.sort_by(|a, b| b.1.unwrap_or(0.0).total_cmp(&a.1.unwrap_or(0.0)));
25    let extrusion_layers: HashSet<u32> = extruded.iter().map(|(layer, _, _)| *layer).collect();
26    let mut layers: Vec<u32> = flat
27        .iter()
28        .map(|(layer, _, _)| *layer)
29        .chain(extrusion_layers.iter().copied())
30        .collect();
31    layers.sort_unstable_by_key(|layer| Reverse(*layer));
32    layers.dedup();
33    let mut flat_by_layer: BTreeMap<u32, Vec<QueriedFeature>> = BTreeMap::new();
34    for (layer, _, feature) in flat {
35        flat_by_layer.entry(layer).or_default().push(feature);
36    }
37    let mut ordered = Vec::new();
38    for layer in layers {
39        if extrusion_layers.contains(&layer) {
40            while extruded
41                .last()
42                .is_some_and(|(nearest, _, _)| *nearest >= layer)
43            {
44                if let Some((_, _, feature)) = extruded.pop() {
45                    ordered.push(feature);
46                }
47            }
48        }
49        if let Some(features) = flat_by_layer.remove(&layer) {
50            ordered.extend(features);
51        }
52    }
53    ordered
54}

56/// A layer a fill, line, circle, heatmap or extrusion feature can be reported for, with its
57/// filter parsed once.
58pub(super) struct Candidate<'a> {
59    pub(super) layer: &'a StyleLayer,
60    pub(super) filter: Option<Filter>,
61    /// How far from a line or point, in screen pixels, its features can be drawn and still be met.
62    pub(super) reach_pixels: f64,
63    /// Where the layer draws its features from where they are, in screen pixels along the
64    /// map's axes: its `*-translate`.
65    pub(super) translate: [f64; 2],
66}
67
68impl<'a> Candidate<'a> {
69    /// A layer whose filter cannot be parsed draws nothing, so it also matches nothing.
70    pub(super) fn new(layer: &'a StyleLayer, zoom: f64, bearing: f64) -> Option<Self> {
71        let filter = match &layer.filter {
72            Some(filter) => Some(Filter::parse(filter).ok()?),
73            None => None,
74        };
75        let reach_pixels = match &layer.paint {
76            Some(LayerPaint::Line(paint)) => line_margin_pixels(paint, zoom),
77            Some(LayerPaint::Circle(paint)) => circle::margin_pixels(paint, zoom),
78            Some(LayerPaint::Heatmap(paint)) => circle::heatmap_margin_pixels(paint, zoom),
79            _ => 0.0,
80        };
81        Some(Self {
82            layer,
83            filter,
84            reach_pixels,
85            translate: layer_translate_pixels(layer.paint.as_ref(), zoom, bearing),
86        })
87    }
88
89    /// How far from the query, in screen pixels, a feature the layer draws there can lie.
90    pub(super) fn margin_pixels(&self) -> f64 {
91        self.reach_pixels + self.translate[0].hypot(self.translate[1])
92    }
93}

95/// The farthest a line layer reaches from its features: half its width and its offset, at their
96/// widest where they vary by feature.
97fn line_margin_pixels(paint: &LinePaint, zoom: f64) -> f64 {
98    // A width or offset set by feature is not known before the feature is; this much covers
99    // what styles give them.
100    const BY_FEATURE: f64 = 64.0;
101    let [half_width, offset] = ground::line_reach(paint, &Default::default(), zoom);
102    let varies = [&paint.line_width, &paint.line_gap_width, &paint.line_offset]
103        .into_iter()
104        .flatten()
105        .any(|property| !property.is_feature_constant());
106    half_width + offset.abs() + if varies { BY_FEATURE } else { 0.0 }
107}
