1//! The features of a tile cut out of a coarser one, as GL JS `sliceVectorTileLayer` cuts an
2//! overzoomed tile out of the last tile its source has.
3
4use crate::{coords::EXTENT, io::geometry_index::IndexedGeometry};
5
6/// The buffer GL JS keeps around an overzoomed tile, in the units of its 8192-unit extent.
7const GL_BUFFER: f64 = 128.0;
8
9/// The parts of a coarser tile's geometry within a finer tile `scale` times smaller whose top
10/// left corner lies at `offset` of the finer tile's units: moved into its units, cut to it and its
11/// buffer, and left out where only the buffer holds them, as GL JS `FeatureIndex.insert` leaves
12/// them to the neighbouring tile.
13pub(super) fn slice(
14    geometry: &IndexedGeometry<f64>,
15    scale: f64,
16    offset: [f64; 2],
17) -> Vec<IndexedGeometry<f64>> {
18    let buffer = GL_BUFFER * EXTENT / 8192.0;
19    let area = geo_types::Rect::new(
20        geo_types::Coord {
21            x: -buffer,
22            y: -buffer,
23        },
24        geo_types::Coord {
25            x: EXTENT + buffer,
26            y: EXTENT + buffer,
27        },
28    );
29    geometry
30        .rescaled(scale, offset)
31        .clipped_to(area)
32        .into_iter()
33        .filter(|part| {
34            let (lower, upper) = (part.bounds.lower(), part.bounds.upper());
35            lower.x() < EXTENT && lower.y() < EXTENT && upper.x() >= 0.0 && upper.y() >= 0.0
36        })
37        .collect()
38}
