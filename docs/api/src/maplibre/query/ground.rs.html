1//! Whether a fill or line feature meets a query on the ground plane, as GL JS
2//! `FillStyleLayer` and `LineStyleLayer.queryIntersectsFeature` decide it.
3//!
4//! The query is the ground the screen point or box covers, which a pitched view turns into a
5//! trapezoid; a translated layer moves it back by its translation, and a line is met within half
6//! its width of it once it is moved sideways by its offset.
7
8use geo::prelude::*;
9use geo_types::{Coord, LineString, Point, Polygon};
10
11use crate::{
12    io::geometry_index::{ExactGeometry, IndexedGeometry},
13    style::{expression::FeatureProperties, layer::LinePaint},
14};
15
16/// The ground a query covers, in the units of one tile.
17#[derive(Clone, Debug)]
18pub(super) enum Footprint {
19    /// The ground under a screen point.
20    Point(Point<f64>),
21    /// The ground under a screen box, a trapezoid when the view is pitched.
22    Polygon(Polygon<f64>),
23}
24
25impl Footprint {
26    /// The footprint of ground corners: one for a point, four for a box.
27    pub(super) fn of(corners: &[[f64; 2]]) -> Option<Self> {
28        match corners {
29            [] => None,
30            [point] => Some(Self::Point(Point::new(point[0], point[1]))),
31            _ if corners.iter().all(|corner| corner == &corners[0]) => {
32                Some(Self::Point(Point::new(corners[0][0], corners[0][1])))
33            }
34            _ => Some(Self::Polygon(Polygon::new(
35                LineString::from(corners.iter().map(|&[x, y]| (x, y)).collect::<Vec<_>>()),
36                Vec::new(),
37            ))),
38        }
39    }
40
41    /// The footprint moved by `[dx, dy]`.
42    fn shifted(&self, [dx, dy]: [f64; 2]) -> Self {
43        let shift = |coord: Coord<f64>| Coord {
44            x: coord.x + dx,
45            y: coord.y + dy,
46        };
47        match self {
48            Self::Point(point) => Self::Point(Point::from(shift(point.0))),
49            Self::Polygon(polygon) => Self::Polygon(polygon.map_coords(shift)),
50        }
51    }
52
53    fn meets(&self, polygon: &Polygon<f64>) -> bool {
54        match self {
55            Self::Point(point) => polygon.intersects(point),
56            Self::Polygon(footprint) => polygon.intersects(footprint),
57        }
58    }
59
60    /// Whether the footprint comes nearer than `radius` to `point`, as GL JS
61    /// `polygonIntersectsBufferedPoint`.
62    pub(super) fn reaches(&self, point: Point<f64>, radius: f64) -> bool {
63        match self {
64            Self::Point(at) => at.euclidean_distance(&point) < radius,
65            Self::Polygon(footprint) => footprint.euclidean_distance(&point) < radius,
66        }
67    }
68
69    fn distance(&self, line: &LineString<f64>) -> f64 {
70        match self {
71            Self::Point(point) => point.euclidean_distance(line),
72            Self::Polygon(footprint) => footprint.euclidean_distance(line),
73        }
74    }
75}

77/// Half a line's drawn width and its sideways offset, in screen pixels, as GL JS
78/// `getLineWidth` and `line-offset` give them for one feature.
79pub(super) fn line_reach(paint: &LinePaint, properties: &FeatureProperties, zoom: f64) -> [f64; 2] {
80    let value = |property: &Option<crate::style::property::StyleProperty<f32>>, default: f32| {
81        property
82            .as_ref()
83            .and_then(|value| value.evaluate_for(properties, zoom))
84            .map_or(f64::from(default), f64::from)
85    };
86    let width = value(&paint.line_width, 1.0);
87    let gap = value(&paint.line_gap_width, 0.0);
88    let drawn = if gap > 0.0 { gap + 2.0 * width } else { width };
89    [drawn / 2.0, value(&paint.line_offset, 0.0)]
90}

92/// A line moved `offset` units to its right, each vertex along the bisector of its corner, as
93/// GL JS `offsetLine`.
94fn offset_line(line: &LineString<f64>, offset: f64) -> LineString<f64> {
95    let mut points: Vec<Coord<f64>> = line.coords().copied().collect();
96    points.dedup();
97    let normal = |from: Coord<f64>, to: Coord<f64>| {
98        let (dx, dy) = (to.x - from.x, to.y - from.y);
99        let length = dx.hypot(dy);
100        // GL JS's perpendicular of (x, y) is (-y, x).
101        [-dy / length, dx / length]
102    };
103    let shifted = (0..points.len())
104        .map(|index| {
105            let before = index.checked_sub(1).map_or([0.0, 0.0], |previous| {
106                normal(points[previous], points[index])
107            });
108            let after = points
109                .get(index + 1)
110                .map_or([0.0, 0.0], |&next| normal(points[index], next));
111            let sum = [before[0] + after[0], before[1] + after[1]];
112            let length = sum[0].hypot(sum[1]);
113            let mut bisector = if length > 0.0 {
114                [sum[0] / length, sum[1] / length]
115            } else {
116                [0.0, 0.0]
117            };
118            let cos_half_angle = bisector[0] * after[0] + bisector[1] * after[1];
119            if cos_half_angle != 0.0 {
120                bisector = [bisector[0] / cos_half_angle, bisector[1] / cos_half_angle];
121            }
122            Coord {
123                x: points[index].x + bisector[0] * offset,
124                y: points[index].y + bisector[1] * offset,
125            }
126        })
127        .collect();
128    LineString::new(shifted)
129}

131/// Whether `geometry`, drawn by a layer of `kind` moved by `translate` tile units, meets the
132/// query's `footprint`; `line` is a line layer's half width and offset in tile units.
133pub(super) fn touches(
134    geometry: &IndexedGeometry<f64>,
135    footprint: &Footprint,
136    (kind, translate): (&str, [f64; 2]),
137    line: [f64; 2],
138) -> bool {
139    // A translated layer draws a feature away from where it is, so the query is moved back.
140    let query = footprint.shifted([-translate[0], -translate[1]]);
141    let [half_width, offset] = line;
142    let near = |ring: &LineString<f64>| {
143        let ring = if offset == 0.0 {
144            ring.clone()
145        } else {
146            offset_line(ring, offset)
147        };
148        query.distance(&ring) < half_width
149    };
150    match (&geometry.exact, kind) {
151        (ExactGeometry::Polygon(polygon), "fill") => query.meets(polygon),
152        (ExactGeometry::Polygon(polygon), "line") => {
153            near(polygon.exterior()) || polygon.interiors().iter().any(near)
154        }
155        (ExactGeometry::LineString(line), "line") => near(line),
156        _ => false,
157    }
158}
