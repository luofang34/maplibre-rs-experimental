11/// Tiles a single query reads at one zoom before it uses a coarser one.
12const MAX_TILES: i64 = 256;

14/// The screen box with the point under the camera added, as GL JS `getCameraQueryGeometry`:
15/// the base of any extrusion that stands up into the query lies between the two.
16pub(super) fn camera_query_bounds(view_state: &ViewState, [x0, y0, x1, y1]: [f64; 4]) -> [f64; 4] {
17    let offset = view_state.camera().get_pitch().0.tan() * view_state.camera_to_center_distance();
18    let camera = [view_state.width() / 2.0, view_state.height() / 2.0 + offset];
19    [
20        x0.min(camera[0]),
21        y0.min(camera[1]),
22        x1.max(camera[0]),
23        y1.max(camera[1]),
24    ]
25}

27/// The rectangle of world pixels a screen box covers on the ground plane, or `None` when none of
28/// its corners reaches the ground (it is all sky).
29pub(super) fn ground_region(view_state: &ViewState, bounds: [f64; 4]) -> Option<[f64; 4]> {
30    ground_corners(view_state, bounds)
31        .into_iter()
32        .fold(None, |region, [x, y]| {
33            Some(match region {
34                None => [x, y, x, y],
35                Some(r) => [r[0].min(x), r[1].min(y), r[2].max(x), r[3].max(y)],
36            })
37        })
38}

40/// The corners of a screen box on the ground plane in world pixels, leaving out those in the
41/// sky.
42pub(super) fn ground_corners(view_state: &ViewState, bounds: [f64; 4]) -> Vec<[f64; 2]> {
43    let Ok(inverted) = view_state.inverted_view_projection() else {
44        return Vec::new();
45    };
46    let mut corners = Vec::with_capacity(4);
47    for (x, y) in [
48        (bounds[0], bounds[1]),
49        (bounds[2], bounds[1]),
50        (bounds[2], bounds[3]),
51        (bounds[0], bounds[3]),
52    ] {
53        let Some(point) =
54            view_state.window_to_world_at_ground(&Vector2::new(x, y), &inverted, true)
55        else {
56            continue;
57        };
58        if point.x.is_finite() && point.y.is_finite() {
59            corners.push([point.x, point.y]);
60        }
61    }
62    corners
63}

65/// One tile a query touches: its canonical coordinates and the query rectangle in its grid.
66pub(super) struct QueryTile {
67    pub(super) coords: WorldTileCoords,
68    /// Which copy of the world the tile is seen in: zero for the one at the antimeridian's east.
69    pub(super) wrap: i32,
70    /// Query rectangle in tile units, as `[min x, min y, max x, max y]`.
71    pub(super) local: [f64; 4],
72    /// Tile units that make one screen pixel.
73    pub(super) units_per_pixel: f64,
74    pub(super) zoom_level: u8,
75    /// World pixels of the tile's top-left corner, in the copy of the world it is seen in.
76    pub(super) origin: [f64; 2],
77    /// World pixels in one tile unit.
78    pub(super) world_per_unit: f64,
79    /// The ground the query covers, in tile units, for features on the ground plane.
80    pub(super) footprint: Option<Footprint>,
81}
82
83impl QueryTile {
84    /// The footprint of ground corners given in world pixels.
85    pub(super) fn footprint_of(&self, corners: &[[f64; 2]]) -> Option<Footprint> {
86        let local: Vec<[f64; 2]> = corners
87            .iter()
88            .map(|corner| {
89                [
90                    (corner[0] - self.origin[0]) / self.world_per_unit,
91                    (corner[1] - self.origin[1]) / self.world_per_unit,
92                ]
93            })
94            .collect();
95        Footprint::of(&local)
96    }
97}

99/// The tiles at grid level `z` that a region of world pixels at view zoom `zoom` covers, or
100/// comes within `margin` world pixels of.
101pub(super) fn tiles_in(region: [f64; 4], margin: f64, zoom: Zoom, z: u8) -> Vec<QueryTile> {
102    let scale = zoom.scale_to_zoom_level(ZoomLevel::new(z));
103    let to_grid = |world: f64| world / TILE_SIZE * scale;
104    let (x0, y0) = (to_grid(region[0]), to_grid(region[1]));
105    let (x1, y1) = (to_grid(region[2]), to_grid(region[3]));
106    let reach = to_grid(margin);
107    let (tx0, tx1) = ((x0 - reach).floor() as i64, (x1 + reach).floor() as i64);
108    let (ty0, ty1) = ((y0 - reach).floor() as i64, (y1 + reach).floor() as i64);
109    let tiles_wide = 1_i64 << z;
110    if (tx1 - tx0 + 1) * (ty1 - ty0 + 1) > MAX_TILES {
111        return Vec::new();
112    }
113    let units_per_pixel = EXTENT * scale / TILE_SIZE;
114    let mut tiles = Vec::new();
115    for ty in ty0.max(0)..=ty1.min(tiles_wide - 1) {
116        for tx in tx0..=tx1 {
117            let canonical = tx.rem_euclid(tiles_wide);
118            tiles.push(QueryTile {
119                coords: WorldTileCoords {
120                    x: canonical as i32,
121                    y: ty as i32,
122                    z: ZoomLevel::new(z),
123                },
124                wrap: tx.div_euclid(tiles_wide) as i32,
125                local: [
126                    (x0 - tx as f64) * EXTENT,
127                    (y0 - ty as f64) * EXTENT,
128                    (x1 - tx as f64) * EXTENT,
129                    (y1 - ty as f64) * EXTENT,
130                ],
131                units_per_pixel,
132                zoom_level: z,
133                origin: [tx as f64 * TILE_SIZE / scale, ty as f64 * TILE_SIZE / scale],
134                world_per_unit: TILE_SIZE / scale / EXTENT,
135                footprint: None,
136            });
137        }
138    }
139    tiles
140}

142/// The geometries whose bounds meet the query rectangle.
143pub(super) fn candidates_in(index: &TileIndex, local: [f64; 4]) -> Vec<&IndexedGeometry<f64>> {
144    let window = rstar::AABB::from_corners(
145        geo_types::Point::new(local[0], local[1]),
146        geo_types::Point::new(local[2], local[3]),
147    );
148    match index {
149        TileIndex::Spatial { tree } => tree.locate_in_envelope_intersecting(&window).collect(),
150        TileIndex::Linear { list } => list
151            .iter()
152            .filter(|geometry| {
153                use rstar::Envelope;
154                geometry.bounds.intersects(&window)
155            })
156            .collect(),
157    }
