10/// Tiles a single query reads at one zoom before it uses a coarser one.
11const MAX_TILES: i64 = 256;
12
13/// One tile a query touches: its canonical coordinates and the query rectangle in its grid.
14pub(super) struct QueryTile {
15    pub(super) coords: WorldTileCoords,
16    /// Which copy of the world the tile is seen in: zero for the one at the antimeridian's east.
17    pub(super) wrap: i32,
18    /// Query rectangle in tile units, as `[min x, min y, max x, max y]`.
19    pub(super) local: [f64; 4],
20    /// Tile units that make one screen pixel.
21    pub(super) units_per_pixel: f64,
22    pub(super) zoom_level: u8,
23    /// World pixels of the tile's top-left corner, in the copy of the world it is seen in.
24    pub(super) origin: [f64; 2],
25    /// World pixels in one tile unit.
26    pub(super) world_per_unit: f64,
27    /// The ground the query covers, in tile units, for features on the ground plane.
28    pub(super) footprint: Option<Footprint>,
29}
30
31impl QueryTile {
32    /// The footprint of ground corners given in world pixels.
33    pub(super) fn footprint_of(&self, corners: &[[f64; 2]]) -> Option<Footprint> {
34        let local: Vec<[f64; 2]> = corners
35            .iter()
36            .map(|corner| {
37                [
38                    (corner[0] - self.origin[0]) / self.world_per_unit,
39                    (corner[1] - self.origin[1]) / self.world_per_unit,
40                ]
41            })
42            .collect();
43        Footprint::of(&local)
44    }
45}

47/// The tiles at grid level `z` that a region of world pixels covers, or comes within `margin`
48/// screen pixels of.
49pub(super) fn tiles_in(
50    region: [f64; 4],
51    margin: f64,
52    resolved: &ResolvedFrameView,
53    z: u8,
54) -> Vec<QueryTile> {
55    let scale = resolved.zoom.scale_to_zoom_level(ZoomLevel::new(z));
56    let to_grid = |world: f64| world / TILE_SIZE * scale;
57    let (x0, y0) = (to_grid(region[0]), to_grid(region[1]));
58    let (x1, y1) = (to_grid(region[2]), to_grid(region[3]));
59    let mid_my = (((region[1] + region[3]) * 0.5) / resolved.world_size).clamp(0.0, 1.0);
60    let reach = to_grid(margin * resolved.globe_scale_at_mercator_y(mid_my));
61    let (tx0, tx1) = ((x0 - reach).floor() as i64, (x1 + reach).floor() as i64);
62    let (ty0, ty1) = ((y0 - reach).floor() as i64, (y1 + reach).floor() as i64);
63    let tiles_wide = 1_i64 << z;
64    if (tx1 - tx0 + 1) * (ty1 - ty0 + 1) > MAX_TILES {
65        return Vec::new();
66    }
67    let base_units_per_pixel = EXTENT * scale / TILE_SIZE;
68    let mut tiles = Vec::new();
69    for ty in ty0.max(0)..=ty1.min(tiles_wide - 1) {
70        let row_my = ((ty as f64 + 0.5) / tiles_wide as f64).clamp(0.0, 1.0);
71        let units_per_pixel = base_units_per_pixel * resolved.globe_scale_at_mercator_y(row_my);
72        for tx in tx0..=tx1 {
73            let canonical = tx.rem_euclid(tiles_wide);
74            tiles.push(QueryTile {
75                coords: WorldTileCoords {
76                    x: canonical as i32,
77                    y: ty as i32,
78                    z: ZoomLevel::new(z),
79                },
80                wrap: tx.div_euclid(tiles_wide) as i32,
81                local: [
82                    (x0 - tx as f64) * EXTENT,
83                    (y0 - ty as f64) * EXTENT,
84                    (x1 - tx as f64) * EXTENT,
85                    (y1 - ty as f64) * EXTENT,
86                ],
87                units_per_pixel,
88                zoom_level: z,
89                origin: [tx as f64 * TILE_SIZE / scale, ty as f64 * TILE_SIZE / scale],
90                world_per_unit: TILE_SIZE / scale / EXTENT,
91                footprint: None,
92            });
93        }
94    }
95    tiles
96}

98/// The geometries whose bounds meet the query rectangle.
99pub(super) fn candidates_in(index: &TileIndex, local: [f64; 4]) -> Vec<&IndexedGeometry<f64>> {
100    let window = rstar::AABB::from_corners(
101        geo_types::Point::new(local[0], local[1]),
102        geo_types::Point::new(local[2], local[3]),
103    );
104    match index {
105        TileIndex::Spatial { tree } => tree.locate_in_envelope_intersecting(&window).collect(),
106        TileIndex::Linear { list } => list
107            .iter()
108            .filter(|geometry| {
109                use rstar::Envelope;
110                geometry.bounds.intersects(&window)
111            })
112            .collect(),
113    }
