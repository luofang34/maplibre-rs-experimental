1//! What an eclipse takes from the light: the share of the sun's disc the moon leaves uncovered.
2
3use super::{CelestialError, CelestialLighting};
4use crate::tcs::world::World;
5
6/// The share of the sun's disc left uncovered where the eye stands, from 0 in totality to 1
7/// outside an eclipse, in linear light.
8#[derive(Clone, Copy, Debug, PartialEq)]
9pub struct VisibleFraction(f32);
10
11impl Default for VisibleFraction {
12    /// The whole disc, as outside an eclipse.
13    fn default() -> Self {
14        Self(1.0)
15    }
16}
17
18impl VisibleFraction {
19    /// The share `fraction`, in 0..=1.
20    pub fn new(fraction: f32) -> Result<Self, CelestialError> {
21        if (0.0..=1.0).contains(&fraction) {
22            Ok(Self(fraction))
23        } else {
24            Err(CelestialError::InvalidVisibleFraction { fraction })
25        }
26    }
27
28    /// The share in 0..=1.
29    pub fn get(self) -> f32 {
30        self.0
31    }
32}
33
34/// Sets the share of the sun's disc left uncovered where the eye stands. The sky's scattered
35/// sunlight and the day side's light scale with it; the sun's disc and its glare do not, as the
36/// moon's disc covers the one and the other stands in for the corona. The ground has a day
37/// side only under a host's sun direction, so without one only the sky dims. The share is the
38/// eye's alone and the whole day side takes it, so it suits a view from the ground; a view of
39/// the globe leaves it at 1.
40pub fn set_sun_visible_fraction(world: &mut World, fraction: VisibleFraction) {
41    world
42        .resources
43        .get_or_init_mut::<CelestialLighting>()
44        .sun_visible = fraction;
45}
