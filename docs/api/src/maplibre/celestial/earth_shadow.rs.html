1//! The Earth's shadow where the moon crosses it, as a host computes it for a lunar eclipse.
2//!
3//! The shadow's size is a convention a host chooses, such as the enlargement an eclipse canon
4//! applies, so the host gives its axis and radii rather than the renderer deriving them.
5
6use cgmath::InnerSpace;
7
8use super::{CelestialDiscs, CelestialError, EcefDirection};
9use crate::tcs::world::World;
10
11/// Widest angular radius the shadow's penumbra may have.
12const MAX_RADIUS_DEGREES: f64 = 5.0;
13/// Danjon's scale runs from L0, a moon almost lost in totality, to L4, a bright copper one.
14const MAX_DANJON: f32 = 4.0;
15/// A moon of middling darkness, deep red or rust, when the host gives no Danjon value.
16const DEFAULT_DANJON: f32 = 2.0;
17
18/// The Earth's shadow at the moon's distance, as seen from the eye: its axis and the angular
19/// radii of its umbra and penumbra, and how dark the umbra is on Danjon's scale.
20#[derive(Clone, Copy, Debug, PartialEq)]
21pub struct EarthShadow {
22    axis: EcefDirection,
23    umbra_degrees: f64,
24    penumbra_degrees: f64,
25    danjon: f32,
26}
27
28impl EarthShadow {
29    /// The shadow along `axis`, from the eye towards the shadow's centre at the moon's distance,
30    /// with an umbra smaller than its penumbra, both above zero and at most five degrees, and a
31    /// Danjon value in 0..=4, L2 when none is given.
32    pub fn new(
33        axis: EcefDirection,
34        umbra_radius_degrees: f64,
35        penumbra_radius_degrees: f64,
36        danjon: Option<f32>,
37    ) -> Result<Self, CelestialError> {
38        let danjon_value = danjon.unwrap_or(DEFAULT_DANJON);
39        let nested = umbra_radius_degrees > 0.0
40            && umbra_radius_degrees < penumbra_radius_degrees
41            && penumbra_radius_degrees <= MAX_RADIUS_DEGREES;
42        if nested && (0.0..=MAX_DANJON).contains(&danjon_value) {
43            Ok(Self {
44                axis,
45                umbra_degrees: umbra_radius_degrees,
46                penumbra_degrees: penumbra_radius_degrees,
47                danjon: danjon_value,
48            })
49        } else {
50            Err(CelestialError::InvalidEarthShadow {
51                umbra_radius_degrees,
52                penumbra_radius_degrees,
53                danjon,
54            })
55        }
56    }
57
58    /// Direction from the eye to the shadow's centre.
59    pub fn axis(&self) -> EcefDirection {
60        self.axis
61    }
62
63    /// Angular radius of the umbra, in degrees.
64    pub fn umbra_radius_degrees(&self) -> f64 {
65        self.umbra_degrees
66    }
67
68    /// Angular radius of the penumbra, in degrees.
69    pub fn penumbra_radius_degrees(&self) -> f64 {
70        self.penumbra_degrees
71    }
72
73    /// How dark the umbra is, from 0 to 4 on Danjon's scale.
74    pub fn danjon(&self) -> f32 {
75        self.danjon
76    }
77
78    /// The share of the sun's light that reaches a point `direction` from the eye: none in the
79    /// umbra, all beyond the penumbra, and a smooth ramp between, as the sun's disc emerges
80    /// from behind the Earth.
81    pub fn sunlit_share(&self, direction: EcefDirection) -> f64 {
82        let angle = direction
83            .ecef()
84            .cross(self.axis.ecef())
85            .magnitude()
86            .atan2(direction.ecef().dot(self.axis.ecef()))
87            .to_degrees();
88        let t = ((angle - self.umbra_degrees) / (self.penumbra_degrees - self.umbra_degrees))
89            .clamp(0.0, 1.0);
90        t * t * (3.0 - 2.0 * t)
91    }
92}

94/// Sets or clears the Earth's shadow the moon is lit through.
95pub fn set_earth_shadow(world: &mut World, shadow: Option<EarthShadow>) {
96    world
97        .resources
98        .get_or_init_mut::<CelestialDiscs>()
99        .earth_shadow = shadow;
