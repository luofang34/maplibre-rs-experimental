14#[derive(Clone)]
15/// Collects optional configuration without creating a window, GPU device or surface.
16pub struct RendererBuilder {
17    wgpu_settings: Option<WgpuSettings>,
18    renderer_settings: Option<RendererSettings>,
19}
20
21impl RendererBuilder {
22    /// Starts with renderer and device settings that will use their defaults unless replaced.
23    pub fn new() -> Self {
24        Self {
25            wgpu_settings: None,
26            renderer_settings: None,
27        }
28    }
29
30    /// Replaces render target and pool settings; later calls replace the entire value.
31    pub fn with_renderer_settings(mut self, renderer_settings: RendererSettings) -> Self {
32        self.renderer_settings = Some(renderer_settings);
33        self
34    }
35
36    /// Replaces the device request configuration; later calls replace the entire value.
37    pub fn with_wgpu_settings(mut self, wgpu_settings: WgpuSettings) -> Self {
38        self.wgpu_settings = Some(wgpu_settings);
39        self
40    }
41
42    /// Resolves missing settings to defaults without acquiring any GPU resources.
43    pub fn build(self) -> UninitializedRenderer {
44        UninitializedRenderer {
45            wgpu_settings: self.wgpu_settings.unwrap_or_default(),
46            renderer_settings: self.renderer_settings.unwrap_or_default(),
47        }
48    }
49}
50
51impl Default for RendererBuilder {
52    fn default() -> Self {
53        Self::new()
54    }
55}

57/// Resolved configuration that can be consumed to acquire GPU resources for an existing window.
58pub struct UninitializedRenderer {
59    /// Backend selection, adapter preference, features and limits for the device request.
60    pub wgpu_settings: WgpuSettings,
61    /// Render target properties and tile buffer capacities.
62    pub renderer_settings: RendererSettings,
63}
64
65impl UninitializedRenderer {
66    /// Acquires a window-compatible adapter/device and configures its presentation surface.
67    /// Returns a renderer with an empty render graph; plugins supply the rendering pipeline.
68    /// Window-handle, surface and device-request failures are propagated as [`RenderError`].
69    pub async fn initialize_renderer<MWC>(
70        self,
71        existing_window: &MWC::MapWindow,
72    ) -> Result<Renderer, RenderError>
73    where
74        MWC: MapWindowConfig,
75        <MWC as MapWindowConfig>::MapWindow: HeadedMapWindow,
76    {
77        Renderer::initialize(existing_window, self.wgpu_settings, self.renderer_settings).await
78    }
79}
80
81#[cfg(feature = "headless")]
82impl UninitializedRenderer {
83    pub(crate) async fn initialize_headless<MWC>(
84        self,
85        existing_window: &MWC::MapWindow,
86    ) -> Result<Renderer, RenderError>
87    where
88        MWC: MapWindowConfig,
89    {
90        Renderer::initialize_headless(existing_window, self.wgpu_settings, self.renderer_settings)
91            .await
92    }
