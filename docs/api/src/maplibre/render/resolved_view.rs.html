32/// Transition threshold at which globe horizon clipping begins (`GLOBE_Z_CLIPPING_START` in WGSL).
33const GLOBE_Z_CLIPPING_START: f64 = 0.2;

35/// Active geometric projection mode of a [`ResolvedFrameView`].
36#[derive(Clone, Debug)]
37pub enum ProjectionMode {
38    /// Planar Web Mercator projection (`transition == 0.0`).
39    FlatMercator,
40    /// Vertical-perspective globe or Mercator-to-globe transition (`transition > 0.0`).
41    Globe {
42        /// Interpolation weight in `(0.0, 1.0]` (`1.0` for pure globe).
43        transition: f32,
44        /// Pre-built vertical-perspective globe camera state.
45        camera: Box<GlobeCameraState>,
46    },
47}

49/// Window-space coordinates and clip depth of a projected 3D world point.
50#[derive(Clone, Copy, Debug, PartialEq)]
51pub struct ProjectedWindowPoint {
52    /// Viewport pixel coordinates `[x, y]` with `y` pointing downward.
53    pub window: [f64; 2],
54    /// Unreversed normalized clip depth (`clip.z / clip.w`) for front-to-back ordering.
55    pub depth: f64,
56    /// Homogeneous perspective distance `clip.w`.
57    pub clip_w: f64,
58}

60/// Immutable per-frame camera and projection snapshot derived once from [`Style`] and [`ViewState`].
61#[derive(Clone, Debug)]
62pub struct ResolvedFrameView {
63    key: ViewSnapshotKey,
64    /// Authoritative camera representation driving this frame.
65    pub authority: CameraAuthority,
66    /// Active geometric projection mode (flat Mercator or globe / transition).
67    pub projection: ProjectionMode,
68    /// Precomputed GPU uniform data for projection-aware shaders.
69    pub shader_projection: ShaderProjectionData,
70    /// Unreversed clip-space view-projection matrix (`[0, w]` depth) for CPU math.
71    pub view_projection: ViewProjection,
72    /// Reversed-Z GPU clip-space view-projection matrix.
73    pub gpu_view_projection: ViewProjection,
74    /// Pre-inverted Mercator view-projection matrix when invertible.
75    pub inverted_view_projection: Option<InvertedViewProjection>,
76    /// Transform mapping normalized device coordinates to window pixels.
77    pub clip_to_window: Matrix4<f64>,
78    /// Continuous map zoom level.
79    pub zoom: Zoom,
80    /// Style evaluation zoom level.
81    pub style_zoom: Zoom,
82    /// Web Mercator world width in pixels at `zoom`.
83    pub world_size: f64,
84    /// Mercator pixels per metre at the map center.
85    pub pixels_per_meter: f64,
86    /// Orbit distance in pixels from eye to map center.
87    pub camera_to_center_distance: f64,
88    /// Elevation in metres of the orbit target at the map center.
89    pub center_elevation: f64,
90    /// Eye position in Mercator world pixels (`x, y`) and metres above sea level (`z`).
91    pub eye_position: Vector3<f64>,
92    /// Viewport logical dimensions `(width, height)` in pixels.
93    pub viewport_size: (f64, f64),
94    /// Clockwise map bearing in radians.
95    pub bearing_radians: f64,
96    /// Camera pitch in radians.
97    pub pitch_radians: f64,
98    /// Planetary body geometry.
99    pub body: Body,
100    /// Whether 3D terrain is declared on the active style.
101    pub terrain_enabled: bool,
102}

104impl ResolvedFrameView {
105    /// Resolves all per-frame camera matrices, globe state, and shader projection data in one pass.
106    pub fn resolve(style: &Style, view_state: &ViewState) -> Result<Self, ProjectionStateError> {
107        let zoom = view_state.zoom();
108        let transition = style.projection.as_ref().map_or(0.0, |spec| {
109            spec.projection_type.globe_transition(zoom.value())
110        });
111        let key = ViewSnapshotKey::capture(style, view_state, transition);
112        let (projection, shader_projection) = if transition == 0.0 {
113            (
114                ProjectionMode::FlatMercator,
115                flat_shader_projection(view_state),
116            )
117        } else {
118            let globe = globe_camera_for_view(view_state)?;
119            let shader = globe_shader_projection(view_state, transition, &globe)?;
120            (
121                ProjectionMode::Globe {
122                    transition,
123                    camera: Box::new(globe),
124                },
125                shader,
126            )
127        };
128
129        let view_projection = view_state.view_projection();
130        let gpu_view_projection = view_state.gpu_view_projection();
131        let inverted_view_projection = view_state.inverted_view_projection().ok();
132
133        Ok(Self {
134            key,
135            authority: view_state.camera_authority(),
136            projection,
137            shader_projection,
138            view_projection,
139            gpu_view_projection,
140            inverted_view_projection,
141            clip_to_window: view_state.clip_to_window_transform(),
142            zoom,
143            style_zoom: view_state.style_zoom(),
144            world_size: TILE_SIZE * 2.0_f64.powf(zoom.value()),
145            pixels_per_meter: view_state.pixels_per_meter(),
146            camera_to_center_distance: view_state.camera_to_center_distance(),
147            center_elevation: view_state.center_elevation(),
148            eye_position: view_state.eye_position(),
149            viewport_size: view_state.viewport_size(),
150            bearing_radians: view_state.camera().get_bearing().0,
151            pitch_radians: view_state.camera().get_pitch().0,
152            body: view_state.body(),
153            terrain_enabled: style.terrain.is_some(),
154        })
155    }

157    /// Returns `true` when this snapshot already matches every relevant input of `style` and
158    /// `view_state`.
159    #[inline]
160    pub fn matches(&self, style: &Style, view_state: &ViewState) -> bool {
161        let transition = style.projection.as_ref().map_or(0.0, |spec| {
162            spec.projection_type
163                .globe_transition(view_state.zoom().value())
164        });
165        self.key == ViewSnapshotKey::capture(style, view_state, transition)
166    }
167
168    /// Borrows the current snapshot in `world.resources` when up to date, or resolves a fresh one.
169    pub fn get_or_resolve<'a>(
170        world: &'a World,
171        style: &Style,
172        view_state: &ViewState,
173    ) -> Result<std::borrow::Cow<'a, Self>, ProjectionStateError> {
174        if let Some(cached) = world.resources.get::<Self>() {
175            if cached.matches(style, view_state) {
176                return Ok(std::borrow::Cow::Borrowed(cached));
177            }
178        }
179        Self::resolve(style, view_state).map(std::borrow::Cow::Owned)
180    }

182    /// Returns the active Mercator-to-globe transition weight in `0.0..=1.0`.
183    #[inline]
184    pub fn globe_transition(&self) -> f32 {
185        match &self.projection {
186            ProjectionMode::FlatMercator => 0.0,
187            ProjectionMode::Globe { transition, .. } => *transition,
188        }
189    }
190
191    /// Borrows the precomputed [`GlobeCameraState`] when globe rendering is active.
192    #[inline]
193    pub fn globe_camera(&self) -> Option<&GlobeCameraState> {
194        match &self.projection {
195            ProjectionMode::FlatMercator => None,
196            ProjectionMode::Globe { camera, .. } => Some(camera),
197        }
198    }
199
200    /// Clip-space `w` of the view center in `f64`, blended across Mercator-to-globe transitions.
201    #[inline]
202    pub fn center_clip_w(&self) -> f64 {
203        if self.authority != CameraAuthority::ConstrainedMap {
204            return f64::from(self.shader_projection.center_clip_w);
205        }
206        match &self.projection {
207            ProjectionMode::FlatMercator => self.camera_to_center_distance,
208            ProjectionMode::Globe { transition, camera } => {
209                let globe_w = (OPENGL_TO_WGPU_MATRIX
210                    * camera.view_projection()
211                    * camera.target().extend(1.0))
212                .w;
213                if *transition >= 1.0 {
214                    globe_w
215                } else {
216                    let t = f64::from(*transition);
217                    self.camera_to_center_distance * (1.0 - t) + globe_w * t
218                }
219            }
220        }
221    }
222
223    /// Tile-unit scale factor for map-aligned widths/radii at normalized Mercator `y` (`0..=1`),
224    /// matching `project_symbol_scale` in `projection.vertex.wgsl`.
225    #[inline]
226    pub fn globe_scale_at_mercator_y(&self, mercator_y: f64) -> f64 {
227        let transition = self.globe_transition();
228        if transition <= 0.0 {
229            return 1.0;
230        }
231        let (_, lat) = mercator_to_angular_radians(0.0, mercator_y.clamp(0.0, 1.0));
232        let cos_lat = lat.cos().max(1e-3);
233        let globe_circle_x = f64::from(self.shader_projection.globe_circle[0]);
234        1.0 + (globe_circle_x / cos_lat - 1.0) * f64::from(transition)
235    }
236
237    /// Signed horizon distance of normalized Mercator `(x, y)` matching `globe_horizon_distance`
238    /// in `projection.vertex.wgsl` (`>= 0.0` on the visible hemisphere, `< 0.0` occluded).
239    #[inline]
240    pub fn horizon_distance_at_mercator(&self, mercator_x: f64, mercator_y: f64) -> f64 {
241        let ProjectionMode::Globe { transition, camera } = &self.projection else {
242            return 1.0;
243        };
244        let (lon, lat) = mercator_to_angular_radians(mercator_x, mercator_y.clamp(0.0, 1.0));
245        let surface = angular_radians_to_unit_sphere(lon, lat);
246        let clipping_t = ((f64::from(*transition) - GLOBE_Z_CLIPPING_START)
247            / (1.0 - GLOBE_Z_CLIPPING_START))
248            .clamp(0.0, 1.0);
249        let plane = camera.clipping_plane();
250        let dot = plane.truncate().dot(surface) + plane.w;
251        (1.0 - clipping_t) + clipping_t * dot
252    }

254    /// Projects a world-pixel position `[x, y]` elevated by `elevation_meters` above sea level
255    /// into window coordinates, returning `None` behind the camera or occluded behind the globe
256    /// horizon.
257    pub fn project_world_point(
258        &self,
259        [x, y]: [f64; 2],
260        elevation_meters: f64,
261    ) -> Option<ProjectedWindowPoint> {
262        let clip = match &self.projection {
263            ProjectionMode::FlatMercator => {
264                self.view_projection
265                    .project(Vector4::new(x, y, elevation_meters, 1.0))
266            }
267            ProjectionMode::Globe { transition, camera } => {
268                let mx = x / self.world_size;
269                let my = (y / self.world_size).clamp(0.0, 1.0);
270                let (lon, lat) = mercator_to_angular_radians(mx, my);
271                let surface = angular_radians_to_unit_sphere(lon, lat);
272                let clipping_t = ((f64::from(*transition) - GLOBE_Z_CLIPPING_START)
273                    / (1.0 - GLOBE_Z_CLIPPING_START))
274                    .clamp(0.0, 1.0);
275                let plane = camera.clipping_plane();
276                let horizon_dist =
277                    (1.0 - clipping_t) + clipping_t * (plane.truncate().dot(surface) + plane.w);
278                if horizon_dist < 0.0 {
279                    return None;
280                }
281                let elevated = elevate_surface_point(surface, elevation_meters, self.body);
282                let globe_clip =
283                    OPENGL_TO_WGPU_MATRIX * camera.view_projection() * elevated.extend(1.0);
284                if *transition >= 1.0 {
285                    globe_clip
286                } else {
287                    let flat_clip =
288                        self.view_projection
289                            .project(Vector4::new(x, y, elevation_meters, 1.0));
290                    let t = f64::from(*transition);
291                    flat_clip * (1.0 - t) + globe_clip * t
292                }
293            }
294        };
295
296        if !clip.w.is_finite()
297            || clip.w <= 0.0
298            || !clip.x.is_finite()
299            || !clip.y.is_finite()
300            || !clip.z.is_finite()
301        {
302            return None;
303        }
304        let ndc = Vector4::new(clip.x / clip.w, clip.y / clip.w, clip.z / clip.w, 1.0);
305        let window = self.clip_to_window * ndc;
306        Some(ProjectedWindowPoint {
307            window: [window.x, window.y],
308            depth: ndc.z,
309            clip_w: clip.w,
310        })
311    }

314/// Publishes or refreshes [`ResolvedFrameView`] in [`World::resources`] when camera or projection
315/// inputs have changed.
316pub fn resolve_frame_view_system(
317    MapContext {
318        world,
319        style,
320        view_state,
321        ..
322    }: &mut MapContext,
323) -> SystemResult {
324    if world
325        .resources
326        .get::<ResolvedFrameView>()
327        .is_some_and(|cached| cached.matches(style, view_state))
328    {
329        return Ok(());
330    }
331    match ResolvedFrameView::resolve(style, view_state) {
332        Ok(resolved) => {
333            world.resources.insert(resolved);
334        }
335        Err(_) => {
336            world.resources.remove::<ResolvedFrameView>();
337        }
338    }
339    Ok(())
340}

353    fn base_view(zoom: f64) -> ViewState {
354        let z = Zoom::new(zoom);
355        let center = WorldCoords::from_lat_lon(LatLon::new(0.0, 0.0), z);
356        ViewState::new(
357            PhysicalSize::new(512, 512).expect("size"),
358            center,
359            z,
360            Deg(0.0),
361            Rad(0.6435011087932844),
362        )
363    }

365    #[test]
366    fn snapshot_matches_until_camera_or_projection_changes() {
367        let mut style = Style::default();
368        let mut view = base_view(4.0);
369        let resolved = ResolvedFrameView::resolve(&style, &view).expect("resolve flat");
370        assert_eq!(resolved.authority, CameraAuthority::ConstrainedMap);
371        assert!(matches!(resolved.projection, ProjectionMode::FlatMercator));
372        assert!(resolved.matches(&style, &view));
373
374        view.set_center_elevation(250.0);
375        assert!(!resolved.matches(&style, &view));
376
377        let resolved_elev = ResolvedFrameView::resolve(&style, &view).expect("resolve elevated");
378        assert!(resolved_elev.matches(&style, &view));
379
380        style.projection = Some(ProjectionSpecification {
381            projection_type: ProjectionType::Globe,
382        });
383        assert!(!resolved_elev.matches(&style, &view));
384        let globe = ResolvedFrameView::resolve(&style, &view).expect("resolve globe");
385        assert_eq!(globe.globe_transition(), 1.0);
386        assert!(globe.globe_camera().is_some());
387    }

389    #[test]
390    fn globe_projection_projects_front_center_and_occludes_antipode() {
391        let style = Style {
392            projection: Some(ProjectionSpecification {
393                projection_type: ProjectionType::Globe,
394            }),
395            ..Style::default()
396        };
397        let view = base_view(2.0);
398        let resolved = ResolvedFrameView::resolve(&style, &view).expect("resolve globe");
399        let front = WorldCoords::from_lat_lon(LatLon::new(0.0, 0.0), view.zoom());
400        let projected = resolved
401            .project_world_point([front.x, front.y], 0.0)
402            .expect("front of globe visible");
403        assert!((projected.window[0] - 256.0).abs() < 1e-6);
404        assert!((projected.window[1] - 256.0).abs() < 1e-6);
405
406        let antipode = WorldCoords::from_lat_lon(LatLon::new(0.0, 180.0), view.zoom());
407        assert!(
408            resolved
409                .project_world_point([antipode.x, antipode.y], 0.0)
410                .is_none(),
411            "antipode behind globe horizon must be occluded"
412        );
