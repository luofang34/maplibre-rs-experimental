1//! The frame's render passes and the order constraints between them.
2//!
3//! Plugins add passes under names and declare which pass must run before which. The order is
4//! resolved once after the passes change: every constraint holds, and passes the constraints
5//! leave unordered run in the order they were added, so a frame records the same commands on
6//! every run and every platform.
7
8#![deny(missing_docs)]
9
10use std::{borrow::Cow, collections::BTreeSet};
11
12use thiserror::Error;
13
14use crate::{render::RenderResources, tcs::world::World};
15
16/// The device and the frame's command encoder, which every pass records into.
17pub struct RenderContext<'d> {
18    /// Device that owns the resources the passes use.
19    pub device: &'d wgpu::Device,
20    /// The frame's single command stream; passes append to it and the runner submits it.
21    pub command_encoder: wgpu::CommandEncoder,
22}
23
24/// One render pass of the frame.
25pub trait Node: Send + Sync + 'static {
26    /// Records the pass's commands. A pass whose inputs are not ready records nothing.
27    fn run(&self, render_context: &mut RenderContext, resources: &RenderResources, world: &World);
28}
29
30/// A pass that records nothing, used as a named anchor other passes order themselves around.
31pub struct EmptyNode;
32
33impl Node for EmptyNode {
34    fn run(&self, _: &mut RenderContext, _: &RenderResources, _: &World) {}
35}
36
37/// A pass could not be added, ordered or resolved.
38#[derive(Error, Debug, Clone, Eq, PartialEq)]
39pub enum RenderGraphError {
40    /// No pass has this name.
41    #[error("render pass {0:?} does not exist")]
42    InvalidNode(Cow<'static, str>),
43    /// A pass with this name already exists.
44    #[error("render pass {0:?} already exists")]
45    DuplicateNode(Cow<'static, str>),
46    /// The order constraints form a cycle through these passes, so none of them can run first.
47    #[error("render passes {0:?} require each other to run first")]
48    Cycle(Vec<Cow<'static, str>>),
49}
50
51struct Pass {
52    name: Cow<'static, str>,
53    node: Box<dyn Node>,
54}
55
56/// The frame's passes in the order they run.
57#[derive(Default)]
58pub struct RenderGraph {
59    passes: Vec<Pass>,
60    /// `(before, after)` pass indices.
61    constraints: BTreeSet<(usize, usize)>,
62    /// Pass indices in run order, resolved after the last change.
63    order: Option<Vec<usize>>,
64}
65
66impl RenderGraph {
67    /// Adds a pass that runs after the passes added before it, unless constraints say
68    /// otherwise.
69    pub fn add_node(
70        &mut self,
71        name: impl Into<Cow<'static, str>>,
72        node: impl Node,
73    ) -> Result<(), RenderGraphError> {
74        let name = name.into();
75        if self.index(&name).is_some() {
76            return Err(RenderGraphError::DuplicateNode(name));
77        }
78        self.passes.push(Pass {
79            name,
80            node: Box::new(node),
81        });
82        self.order = None;
83        Ok(())
84    }
85
86    /// Requires the pass `before` to run before the pass `after`.
87    pub fn add_node_edge(
88        &mut self,
89        before: impl Into<Cow<'static, str>>,
90        after: impl Into<Cow<'static, str>>,
91    ) -> Result<(), RenderGraphError> {
92        let before = self.existing(before.into())?;
93        let after = self.existing(after.into())?;
94        self.constraints.insert((before, after));
95        self.order = None;
96        Ok(())
97    }
98
99    /// Whether a pass with this name exists.
100    pub fn contains(&self, name: &str) -> bool {
101        self.index(name).is_some()
102    }
103
104    /// Resolves the run order after passes or constraints changed.
105    pub fn prepare(&mut self) -> Result<(), RenderGraphError> {
106        if self.order.is_none() {
107            self.order = Some(self.resolve()?);
108        }
109        Ok(())
110    }
111
112    /// The pass names in the order they run.
113    pub fn node_order(&self) -> Result<Vec<&str>, RenderGraphError> {
114        let order = match &self.order {
115            Some(order) => order.clone(),
116            None => self.resolve()?,
117        };
118        Ok(order
119            .into_iter()
120            .map(|index| self.passes[index].name.as_ref())
121            .collect())
122    }
123
124    /// Records every pass in run order, resolving the order first if it is stale.
125    pub(crate) fn run(
126        &self,
127        render_context: &mut RenderContext,
128        resources: &RenderResources,
129        world: &World,
130    ) -> Result<(), RenderGraphError> {
131        let order = match &self.order {
132            Some(order) => Cow::Borrowed(order),
133            None => Cow::Owned(self.resolve()?),
134        };
135        for &index in order.iter() {
136            let pass = &self.passes[index];
137            #[cfg(feature = "trace")]
138            let _span = tracing::info_span!("render_pass", name = pass.name.as_ref()).entered();
139            pass.node.run(render_context, resources, world);
140        }
141        Ok(())
142    }
143
144    fn index(&self, name: &str) -> Option<usize> {
145        self.passes.iter().position(|pass| pass.name == name)
146    }
147
148    fn existing(&self, name: Cow<'static, str>) -> Result<usize, RenderGraphError> {
149        self.index(&name).ok_or(RenderGraphError::InvalidNode(name))
150    }
151
152    /// Kahn's algorithm, always taking the earliest-added pass that is ready.
153    fn resolve(&self) -> Result<Vec<usize>, RenderGraphError> {
154        let mut waiting_on = vec![0usize; self.passes.len()];
155        for &(_, after) in &self.constraints {
156            waiting_on[after] += 1;
157        }
158        let mut ready: BTreeSet<usize> = (0..self.passes.len())
159            .filter(|&index| waiting_on[index] == 0)
160            .collect();
161        let mut order = Vec::with_capacity(self.passes.len());
162        while let Some(next) = ready.pop_first() {
163            order.push(next);
164            for &(_, after) in self.constraints.range((next, 0)..=(next, usize::MAX)) {
165                waiting_on[after] -= 1;
166                if waiting_on[after] == 0 {
167                    ready.insert(after);
168                }
169            }
170        }
171        if order.len() < self.passes.len() {
172            let cycle = (0..self.passes.len())
173                .filter(|&index| waiting_on[index] > 0)
174                .map(|index| self.passes[index].name.clone())
175                .collect();
176            return Err(RenderGraphError::Cycle(cycle));
177        }
178        Ok(order)
179    }
180}
