1//! The frame's render passes and the order constraints between them.
2//!
3//! Plugins add passes under names and declare which pass must run before which. The order is
4//! resolved once after the passes change: every constraint holds, and passes the constraints
5//! leave unordered run in the order they were added, so a frame records the same commands on
6//! every run and every platform.
7
8#![deny(missing_docs)]
9
10use std::{borrow::Cow, collections::BTreeSet};

12use thiserror::Error;

16/// The device and the frame's command encoder, which every pass records into.
17pub struct RenderContext<'d> {
18    /// Device that owns the resources the passes use.
19    pub device: &'d wgpu::Device,
20    /// The frame's single command stream; passes append to it and the runner submits it.
21    pub command_encoder: wgpu::CommandEncoder,
22    /// Whether `main_pass` should inline `translucent_pass` draws directly into its own render pass
23    /// because no intermediate pass or symbol depth copy runs between them this frame.
24    pub(crate) inline_translucent_pass: bool,
25}
26
27impl<'d> RenderContext<'d> {
28    /// Creates a frame render context around `device` and `command_encoder`.
29    pub fn new(device: &'d wgpu::Device, command_encoder: wgpu::CommandEncoder) -> Self {
30        Self {
31            device,
32            command_encoder,
33            inline_translucent_pass: false,
34        }
35    }
36}

38/// One render pass of the frame.
39pub trait Node: Send + Sync + 'static {
40    /// Records the pass's commands. A pass whose inputs are not ready records nothing.
41    fn run(&self, render_context: &mut RenderContext, resources: &RenderResources, world: &World);
42
43    /// Whether this pass may write 3D occluders into the shared scene depth attachment outside
44    /// `main_pass` (defaults to `true` so external custom 3D scene passes preserve symbol depth
45    /// occlusion automatically).
46    fn writes_custom_scene_depth(&self) -> bool {
47        true
48    }
49
50    /// Whether this pass will record GPU commands on the current frame.
51    ///
52    /// Defaults to `true` so external host passes are never bypassed when fusing built-in passes.
53    fn has_active_work(&self, _resources: &RenderResources, _world: &World) -> bool {
54        true
55    }
56}

58/// A pass that records nothing, used as a named anchor other passes order themselves around.
59pub struct EmptyNode;
60
61impl Node for EmptyNode {
62    fn run(&self, _: &mut RenderContext, _: &RenderResources, _: &World) {}
63
64    fn writes_custom_scene_depth(&self) -> bool {
65        false
66    }
67
68    fn has_active_work(&self, _: &RenderResources, _: &World) -> bool {
69        false
70    }
71}

73/// A pass could not be added, ordered or resolved.
74#[derive(Error, Debug, Clone, Eq, PartialEq)]
75pub enum RenderGraphError {
76    /// No pass has this name.
77    #[error("render pass {0:?} does not exist")]
78    InvalidNode(Cow<'static, str>),
79    /// A pass with this name already exists.
80    #[error("render pass {0:?} already exists")]
81    DuplicateNode(Cow<'static, str>),
82    /// The order constraints form a cycle through these passes, so none of them can run first.
83    #[error("render passes {0:?} require each other to run first")]
84    Cycle(Vec<Cow<'static, str>>),
85}
86
87struct Pass {
88    name: Cow<'static, str>,
89    node: Box<dyn Node>,
90}
91
92/// The frame's passes in the order they run.
93#[derive(Default)]
94pub struct RenderGraph {
95    passes: Vec<Pass>,
96    /// `(before, after)` pass indices.
97    constraints: BTreeSet<(usize, usize)>,
98    /// Pass indices in run order, resolved after the last change.
99    order: Option<Vec<usize>>,
100}
101
102impl RenderGraph {
103    /// Adds a pass that runs after the passes added before it, unless constraints say
104    /// otherwise.
105    pub fn add_node(
106        &mut self,
107        name: impl Into<Cow<'static, str>>,
108        node: impl Node,
109    ) -> Result<(), RenderGraphError> {
110        let name = name.into();
111        if self.index(&name).is_some() {
112            return Err(RenderGraphError::DuplicateNode(name));
113        }
114        self.passes.push(Pass {
115            name,
116            node: Box::new(node),
117        });
118        self.order = None;
119        Ok(())
120    }
121
122    /// Requires the pass `before` to run before the pass `after`.
123    pub fn add_node_edge(
124        &mut self,
125        before: impl Into<Cow<'static, str>>,
126        after: impl Into<Cow<'static, str>>,
127    ) -> Result<(), RenderGraphError> {
128        let before = self.existing(before.into())?;
129        let after = self.existing(after.into())?;
130        self.constraints.insert((before, after));
131        self.order = None;
132        Ok(())
133    }
134
135    /// Whether a pass with this name exists.
136    pub fn contains(&self, name: &str) -> bool {
137        self.index(name).is_some()
138    }
139
140    /// Whether any registered pass may write custom 3D scene depth outside `main_pass`.
141    pub(crate) fn has_custom_scene_depth_writers(&self) -> bool {
142        self.passes
143            .iter()
144            .any(|pass| pass.node.writes_custom_scene_depth())
145    }
146
147    /// Resolves the run order after passes or constraints changed.
148    pub fn prepare(&mut self) -> Result<(), RenderGraphError> {
149        if self.order.is_none() {
150            self.order = Some(self.resolve()?);
151        }
152        Ok(())
153    }
154
155    /// The pass names in the order they run.
156    pub fn node_order(&self) -> Result<Vec<&str>, RenderGraphError> {
157        let order = match &self.order {
158            Some(order) => order.clone(),
159            None => self.resolve()?,
160        };
161        Ok(order
162            .into_iter()
163            .map(|index| self.passes[index].name.as_ref())
164            .collect())
165    }
166
167    /// Records every pass in run order, resolving the order first if it is stale.
168    pub(crate) fn run(
169        &self,
170        render_context: &mut RenderContext,
171        resources: &RenderResources,
172        world: &World,
173    ) -> Result<(), RenderGraphError> {
174        let order = match &self.order {
175            Some(order) => Cow::Borrowed(order),
176            None => Cow::Owned(self.resolve()?),
177        };
178        let main_pos = order.iter().position(|&idx| {
179            self.passes[idx].name.as_ref() == crate::render::draw_graph::node::MAIN_PASS
180        });
181        let trans_pos = order.iter().position(|&idx| {
182            self.passes[idx].name.as_ref() == crate::render::draw_graph::node::TRANSLUCENT_PASS
183        });
184        render_context.inline_translucent_pass = match (main_pos, trans_pos) {
185            (Some(m), Some(t))
186                if m < t && !crate::render::translucent_pass::needs_symbol_depth_copy(world) =>
187            {
188                order[m + 1..t]
189                    .iter()
190                    .all(|&idx| !self.passes[idx].node.has_active_work(resources, world))
191            }
192            _ => false,
193        };
194        for &index in order.iter() {
195            let pass = &self.passes[index];
196            #[cfg(feature = "trace")]
197            let _span = tracing::info_span!("render_pass", name = pass.name.as_ref()).entered();
198            pass.node.run(render_context, resources, world);
199        }
200        Ok(())
201    }
202
203    fn index(&self, name: &str) -> Option<usize> {
204        self.passes.iter().position(|pass| pass.name == name)
205    }
206
207    fn existing(&self, name: Cow<'static, str>) -> Result<usize, RenderGraphError> {
208        self.index(&name).ok_or(RenderGraphError::InvalidNode(name))
209    }
210
211    /// Kahn's algorithm, always taking the earliest-added pass that is ready.
212    fn resolve(&self) -> Result<Vec<usize>, RenderGraphError> {
213        let mut waiting_on = vec![0usize; self.passes.len()];
214        for &(_, after) in &self.constraints {
215            waiting_on[after] += 1;
216        }
217        let mut ready: BTreeSet<usize> = (0..self.passes.len())
218            .filter(|&index| waiting_on[index] == 0)
219            .collect();
220        let mut order = Vec::with_capacity(self.passes.len());
221        while let Some(next) = ready.pop_first() {
222            order.push(next);
223            for &(_, after) in self.constraints.range((next, 0)..=(next, usize::MAX)) {
224                waiting_on[after] -= 1;
225                if waiting_on[after] == 0 {
226                    ready.insert(after);
227                }
228            }
229        }
230        if order.len() < self.passes.len() {
231            let cycle = (0..self.passes.len())
232                .filter(|&index| waiting_on[index] > 0)
233                .map(|index| self.passes[index].name.clone())
234                .collect();
235            return Err(RenderGraphError::Cycle(cycle));
236        }
237        Ok(order)
238    }
239}
