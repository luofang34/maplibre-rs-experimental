10/// GPU argument layout for non-indexed `draw_indirect` and `multi_draw_indirect` commands (16 bytes).
11///
12/// Matches the WebGPU / Vulkan `VkDrawIndirectCommand` binary layout and Bevy's
13/// `IndirectParametersNonIndexed`.
14#[repr(C)]
15#[derive(Copy, Clone, Debug, Default, PartialEq, Eq, Pod, Zeroable)]
16pub struct DrawIndirectArgs {
17    /// Number of vertices to draw per instance.
18    pub vertex_count: u32,
19    /// Number of instances to draw (`0` when culled by a GPU compute pass).
20    pub instance_count: u32,
21    /// Index of the first vertex read from bound vertex buffers.
22    pub first_vertex: u32,
23    /// Instance index of the first instance drawn (`0` unless `INDIRECT_FIRST_INSTANCE` is enabled).
24    pub first_instance: u32,
25}
26
27impl DrawIndirectArgs {
28    /// Byte stride between packed [`DrawIndirectArgs`] entries (`16` bytes).
29    pub const STRIDE: wgpu::BufferAddress = size_of::<Self>() as wgpu::BufferAddress;
30
31    /// Returns the raw byte representation of this indirect command.
32    #[inline]
33    pub fn as_bytes(&self) -> &[u8] {
34        bytemuck::bytes_of(self)
35    }
36}

114/// GPU/CPU batch-set header tracking indirect command count and base slot (8 bytes).
115///
116/// Mirrors Bevy's `IndirectBatchSet`. Placing `indirect_parameters_count` at byte offset `0`
117/// allows a GPU compute culling pass to atomically compact surviving draws and pass the same
118/// buffer directly as `count_buffer` to `multi_draw_[indexed_]indirect_count`.
119#[repr(C)]
120#[derive(Copy, Clone, Debug, Default, PartialEq, Eq, Pod, Zeroable)]
121pub struct IndirectBatchSet {
122    /// Number of indirect draw commands active in this batch set.
123    pub indirect_parameters_count: u32,
124    /// Starting slot index in the indexed or non-indexed indirect parameter buffer.
125    pub indirect_parameters_base: u32,
126}
127
128impl IndirectBatchSet {
129    /// Byte stride between packed [`IndirectBatchSet`] entries (`8` bytes).
130    pub const STRIDE: wgpu::BufferAddress = size_of::<Self>() as wgpu::BufferAddress;
131}

145/// Mesh geometry bounds used to construct [`DrawIndirectArgs`] or [`DrawIndexedIndirectArgs`].
146#[derive(Clone, Debug, PartialEq, Eq, Hash)]
147pub enum IndirectMeshMetadata {
148    /// Indexed mesh bounds (`first_index .. first_index + index_count` plus `base_vertex`).
149    Indexed {
150        /// Contiguous index range within the bound index buffer.
151        index_range: Range<u32>,
152        /// Signed vertex offset added to each index.
153        base_vertex: i32,
154    },
155    /// Non-indexed mesh bounds (`first_vertex .. first_vertex + vertex_count`).
156    NonIndexed {
157        /// Contiguous vertex range within bound vertex buffers.
158        vertex_range: Range<u32>,
159    },
160}
161
162impl IndirectMeshMetadata {
163    /// Whether this metadata describes an indexed draw.
164    #[inline]
165    pub const fn is_indexed(&self) -> bool {
166        matches!(self, Self::Indexed { .. })
167    }
168
169    /// Builds [`DrawIndexedIndirectArgs`] for `instance_range` if this metadata is [`Self::Indexed`].
170    #[inline]
171    pub fn to_indexed_args(&self, instance_range: &Range<u32>) -> Option<DrawIndexedIndirectArgs> {
172        match self {
173            Self::Indexed {
174                index_range,
175                base_vertex,
176            } => Some(DrawIndexedIndirectArgs {
177                index_count: index_range.end.saturating_sub(index_range.start),
178                instance_count: instance_range.end.saturating_sub(instance_range.start),
179                first_index: index_range.start,
180                base_vertex: *base_vertex,
181                first_instance: instance_range.start,
182            }),
183            Self::NonIndexed { .. } => None,
184        }
185    }
186
187    /// Builds [`DrawIndirectArgs`] for `instance_range` if this metadata is [`Self::NonIndexed`].
188    #[inline]
189    pub fn to_non_indexed_args(&self, instance_range: &Range<u32>) -> Option<DrawIndirectArgs> {
190        match self {
191            Self::NonIndexed { vertex_range } => Some(DrawIndirectArgs {
192                vertex_count: vertex_range.end.saturating_sub(vertex_range.start),
193                instance_count: instance_range.end.saturating_sub(instance_range.start),
194                first_vertex: vertex_range.start,
195                first_instance: instance_range.start,
196            }),
197            Self::Indexed { .. } => None,
198        }
199    }
