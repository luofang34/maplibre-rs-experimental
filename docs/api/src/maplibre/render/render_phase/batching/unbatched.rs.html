14fn sorted_active_bins<BPI: BinnedPhaseItem>(
15    bins_map: &HashMap<(BPI::BatchSetKey, BPI::BinKey), RenderBin>,
16) -> Vec<(&BPI::BatchSetKey, &BPI::BinKey, &RenderBin)> {
17    let mut active: Vec<(&BPI::BatchSetKey, &BPI::BinKey, &RenderBin)> = bins_map
18        .iter()
19        .filter(|(_, bin)| !bin.entries.is_empty())
20        .map(|((bsk, bk), bin)| (bsk, bk, bin))
21        .collect();
22    active.sort_unstable_by(|a, b| (a.0, a.1).cmp(&(b.0, b.1)));
23    active
24}

26pub(super) fn prepare_unbatchable_bins<BPI: BinnedPhaseItem, GBD: GetBatchData<BPI>>(
27    world: &World,
28    bins_map: &HashMap<(BPI::BatchSetKey, BPI::BinKey), RenderBin>,
29    use_dynamic_uniforms: bool,
30    use_vertex_instance: bool,
31    min_uniform_alignment: u32,
32    instance_buffer: &mut BatchedInstanceBuffer<GBD::BufferData>,
33    out: &mut PreparedBinGroupList<BPI>,
34) {
35    let active_bins = sorted_active_bins::<BPI>(bins_map);
36
37    for (bsk, bk, bin) in active_bins {
38        let group_start = out.entries.len();
39        for entry in &bin.entries {
40            let count = entry.instance_count.get();
41            for offset in 0..count {
42                let input = InputDataIndex(entry.input_index.0.wrapping_add(offset));
43                let has_explicit_extra = entry.extra_index != PhaseItemExtraIndex::None;
44                match GBD::get_batch_data(world, entry.item, input) {
45                    Some((bd, _)) => {
46                        if has_explicit_extra {
47                            if entry.extra_index.dynamic_offset().is_some() {
48                                if !use_dynamic_uniforms && !use_vertex_instance {
49                                    let _ = instance_buffer.push(bd);
50                                }
51                                out.entries.push(PreparedSingleEntry {
52                                    item: entry.item,
53                                    instance_range: 0..1,
54                                    extra_index: entry.extra_index.clone(),
55                                });
56                            } else {
57                                let idx = instance_buffer.push(bd);
58                                out.entries.push(PreparedSingleEntry {
59                                    item: entry.item,
60                                    instance_range: idx..(idx + 1),
61                                    extra_index: entry.extra_index.clone(),
62                                });
63                            }
64                        } else if use_dynamic_uniforms {
65                            let (_, dyn_off) =
66                                instance_buffer.push_dynamic_uniform(bd, min_uniform_alignment);
67                            out.entries.push(PreparedSingleEntry {
68                                item: entry.item,
69                                instance_range: 0..1,
70                                extra_index: PhaseItemExtraIndex::DynamicOffset(dyn_off),
71                            });
72                        } else {
73                            let idx = instance_buffer.push(bd);
74                            let extra_index = if use_vertex_instance {
75                                PhaseItemExtraIndex::VertexInstance
76                            } else {
77                                PhaseItemExtraIndex::None
78                            };
79                            out.entries.push(PreparedSingleEntry {
80                                item: entry.item,
81                                instance_range: idx..(idx + 1),
82                                extra_index,
83                            });
84                        }
85                    }
86                    None if has_explicit_extra => {
87                        out.entries.push(PreparedSingleEntry {
88                            item: entry.item,
89                            instance_range: 0..1,
90                            extra_index: entry.extra_index.clone(),
91                        });
92                    }
93                    None => {}
94                }
95            }
96        }
97        let group_end = out.entries.len();
98        if group_end > group_start {
99            out.groups
100                .push((bsk.clone(), bk.clone(), group_start..group_end));
101        }
102    }
103}

105pub(super) fn prepare_non_mesh_bins<BPI: BinnedPhaseItem, GBD: GetBatchData<BPI>>(
106    world: &World,
107    bins_map: &HashMap<(BPI::BatchSetKey, BPI::BinKey), RenderBin>,
108    use_dynamic_uniforms: bool,
109    use_vertex_instance: bool,
110    min_uniform_alignment: u32,
111    instance_buffer: &mut BatchedInstanceBuffer<GBD::BufferData>,
112    out: &mut PreparedBinGroupList<BPI>,
113) {
114    let active_bins = sorted_active_bins::<BPI>(bins_map);
115
116    for (bsk, bk, bin) in active_bins {
117        let group_start = out.entries.len();
118        for entry in &bin.entries {
119            let count = entry.instance_count.get();
120            if use_dynamic_uniforms {
121                let mut matched_any = false;
122                for offset in 0..count {
123                    let input = InputDataIndex(entry.input_index.0.wrapping_add(offset));
124                    if let Some((bd, _)) = GBD::get_batch_data(world, entry.item, input) {
125                        matched_any = true;
126                        let (_, dyn_off) =
127                            instance_buffer.push_dynamic_uniform(bd, min_uniform_alignment);
128                        out.entries.push(PreparedSingleEntry {
129                            item: entry.item,
130                            instance_range: 0..1,
131                            extra_index: PhaseItemExtraIndex::DynamicOffset(dyn_off),
132                        });
133                    }
134                }
135                if !matched_any {
136                    out.entries.push(PreparedSingleEntry {
137                        item: entry.item,
138                        instance_range: 0..count,
139                        extra_index: entry.extra_index.clone(),
140                    });
141                }
142            } else {
143                let mut start_idx: Option<u32> = None;
144                let mut end_idx = 0u32;
145                for offset in 0..count {
146                    let input = InputDataIndex(entry.input_index.0.wrapping_add(offset));
147                    if let Some((bd, _)) = GBD::get_batch_data(world, entry.item, input) {
148                        let idx = instance_buffer.push(bd);
149                        if start_idx.is_none() {
150                            start_idx = Some(idx);
151                        }
152                        end_idx = idx + 1;
153                    }
154                }
155                let (instance_range, extra_index) = match start_idx {
156                    Some(s) => {
157                        let extra = if entry.extra_index == PhaseItemExtraIndex::None
158                            && use_vertex_instance
159                        {
160                            PhaseItemExtraIndex::VertexInstance
161                        } else {
162                            entry.extra_index.clone()
163                        };
164                        (s..end_idx, extra)
165                    }
166                    None => (0..count, entry.extra_index.clone()),
167                };
168                out.entries.push(PreparedSingleEntry {
169                    item: entry.item,
170                    instance_range,
171                    extra_index,
172                });
173            }
174        }
175        let group_end = out.entries.len();
176        if group_end > group_start {
177            out.groups
178                .push((bsk.clone(), bk.clone(), group_start..group_end));
179        }
180    }
