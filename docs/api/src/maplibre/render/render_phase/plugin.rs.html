27/// Installs [`BinnedRenderPhase<BPI>`], [`DrawFunctions<BPI>`], [`BatchedInstanceBuffer<GBD::BufferData>`],
28/// shared [`IndirectParametersBuffers`], and their `Prepare` / `PhaseSort` / `Cleanup` systems.
29///
30/// Install after [`RenderPlugin`](crate::render::RenderPlugin) so the standard render stages exist.
31pub struct BinnedRenderPhasePlugin<BPI, GBD> {
32    fallback_mode: FallbackInstanceBufferMode,
33    max_inactive_generations: u64,
34    _marker: PhantomData<fn() -> (BPI, GBD)>,
35}
36
37impl<BPI, GBD> Default for BinnedRenderPhasePlugin<BPI, GBD> {
38    fn default() -> Self {
39        Self::new()
40    }
41}
42
43impl<BPI, GBD> BinnedRenderPhasePlugin<BPI, GBD> {
44    /// Creates a [`BinnedRenderPhasePlugin`] with [`FallbackInstanceBufferMode::DynamicUniform`]
45    /// and [`DEFAULT_MAX_INACTIVE_GENERATIONS`].
46    pub const fn new() -> Self {
47        Self {
48            fallback_mode: FallbackInstanceBufferMode::DynamicUniform,
49            max_inactive_generations: DEFAULT_MAX_INACTIVE_GENERATIONS,
50            _marker: PhantomData,
51        }
52    }
53
54    /// Configures the fallback instance buffer strategy used on backends without vertex storage buffers.
55    pub const fn with_fallback_mode(mut self, mode: FallbackInstanceBufferMode) -> Self {
56        self.fallback_mode = mode;
57        self
58    }
59
60    /// Configures how many inactive sweep generations empty bins remain cached before eviction.
61    pub const fn with_max_inactive_generations(mut self, max_inactive_generations: u64) -> Self {
62        self.max_inactive_generations = max_inactive_generations;
63        self
64    }
65
66    /// Initializes the phase's resources in `world` and registers its `Prepare`, `PhaseSort`, and
67    /// `Cleanup` systems onto `schedule`.
68    pub fn install(&self, schedule: &mut Schedule, world: &mut World) -> Result<(), PluginError>
69    where
70        BPI: BinnedPhaseItem,
71        GBD: GetBatchData<BPI>,
72    {
73        world.resources.get_or_init_mut::<DrawFunctions<BPI>>();
74        world
75            .resources
76            .get_or_init_mut::<BinnedRenderPhase<BPI>>()
77            .set_max_inactive_generations(self.max_inactive_generations);
78        world
79            .resources
80            .get_or_init_mut::<BatchedInstanceBuffer<GBD::BufferData>>()
81            .set_fallback_mode(self.fallback_mode);
82        world
83            .resources
84            .get_or_init_mut::<IndirectParametersBuffers>();
85
86        schedule.add_system_to_stage(
87            RenderStageLabel::Prepare,
88            prepare_binned_render_phase_system::<BPI, GBD>,
89        )?;
90        schedule.add_system_to_stage(
91            RenderStageLabel::PhaseSort,
92            batch_and_prepare_binned_render_phase::<BPI, GBD>,
93        )?;
94        schedule.add_system_to_stage(
95            RenderStageLabel::Cleanup,
96            cleanup_binned_render_phase_system::<BPI>,
97        )?;
98        Ok(())
99    }
100}

119/// Resets per-pass binned phase queues, instance buffers, and indirect parameter staging at the
120/// start of `Prepare`, ensuring multi-viewport slots never inherit a prior slot's queued items.
121pub fn prepare_binned_render_phase_system<BPI, GBD>(
122    MapContext { world, .. }: &mut MapContext,
123) -> SystemResult
124where
125    BPI: BinnedPhaseItem,
126    GBD: GetBatchData<BPI>,
127{
128    prepare_binned_phase_world::<BPI, GBD>(world)
129}
130
131pub(super) fn prepare_binned_phase_world<BPI, GBD>(world: &mut World) -> SystemResult
132where
133    BPI: BinnedPhaseItem,
134    GBD: GetBatchData<BPI>,
135{
136    let Some(phase) = world.resources.get_mut::<BinnedRenderPhase<BPI>>() else {
137        return Err(SystemError::Dependencies);
138    };
139    phase.clear_entries();
140
141    if let Some(instance_buffer) = world
142        .resources
143        .get_mut::<BatchedInstanceBuffer<GBD::BufferData>>()
144    {
145        instance_buffer.clear();
146    }
147    if let Some(indirect_buffers) = world.resources.get_mut::<IndirectParametersBuffers>() {
148        indirect_buffers.clear();
149    }
150    Ok(())
151}

153/// Clears current-frame binned entries and sweeps cold bins during `Cleanup` (including after
154/// mid-frame stage failures via `Schedule::run_after_failure`).
155pub fn cleanup_binned_render_phase_system<BPI>(
156    MapContext { world, .. }: &mut MapContext,
157) -> SystemResult
158where
159    BPI: BinnedPhaseItem,
160{
161    cleanup_binned_phase_world::<BPI>(world)
162}
163
164pub(super) fn cleanup_binned_phase_world<BPI>(world: &mut World) -> SystemResult
165where
166    BPI: BinnedPhaseItem,
167{
168    let eye = world.resources.get::<EyeInFrame>().copied();
169    let Some(phase) = world.resources.get_mut::<BinnedRenderPhase<BPI>>() else {
170        return Err(SystemError::Dependencies);
171    };
172
173    // Always empty queued entries and prepared batches immediately so failed frames or subsequent
174    // viewports never re-render stale items.
175    phase.clear_entries();
176
177    let same_multi_eye_frame = eye.is_some_and(|e| {
178        e.index > 0 || (e.frame > 0 && phase.last_swept_eye_frame == Some(e.frame))
179    });
180    if !same_multi_eye_frame {
181        if let Some(e) = eye.filter(|e| e.frame > 0) {
182            phase.last_swept_eye_frame = Some(e.frame);
183        }
184        phase.sweep();
185    }
186    Ok(())
