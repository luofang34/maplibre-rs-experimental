1//! Compilation of owned shader and fixed-function descriptions into GPU render pipelines.
2
3use std::borrow::Cow;
4
5use crate::render::resource::shader::{FragmentState, VertexState};
6
7/// Consumes a pipeline configuration into a descriptor without allocating GPU resources.
8pub trait RenderPipeline {
9    /// Produces shader, binding and fixed-function state for later device initialization.
10    fn describe_render_pipeline(self) -> RenderPipelineDescriptor;
11}
12
13/// Owned pipeline state, including WGSL sources compiled at initialization.
14pub struct RenderPipelineDescriptor {
15    /// Debug label of the pipeline. This will show up in graphics debuggers for easy identification.
16    pub label: Option<Cow<'static, str>>,
17    /// Descriptor-owned bind groups in shader group-index order, after any prefix layouts.
18    /// `None` adds no groups; it does not request wgpu's automatic layout inference.
19    pub layout: Option<Vec<Vec<wgpu::BindGroupLayoutEntry>>>,
20    /// Vertex shader source, entry point and ordered input buffer layouts.
21    pub vertex: VertexState,
22    /// The properties of the pipeline at the primitive assembly and rasterization level.
23    pub primitive: wgpu::PrimitiveState,
24    /// The effect of draw calls on the depth and stencil aspects of the output target, if any.
25    pub depth_stencil: Option<wgpu::DepthStencilState>,
26    /// The multi-sampling properties of the pipeline.
27    pub multisample: wgpu::MultisampleState,
28    /// Fragment shader source, entry point and color attachment formats and blend states.
29    pub fragment: FragmentState,
30}
31
32impl RenderPipelineDescriptor {
33    /// Compiles the shaders and creates a pipeline with only the descriptor-owned bind groups.
34    /// Invalid WGSL, incompatible bindings or unsupported states use wgpu's validation mechanism.
35    pub fn initialize(&self, device: &wgpu::Device) -> wgpu::RenderPipeline {
36        self.initialize_with_prefix_layouts(device, &[])
37    }
38
39    /// Creates a pipeline with shared bind-group layouts before descriptor-owned layouts.
40    /// Shader group indices must account for the prefix length. Prefix layouts and shader
41    /// interfaces must be compatible with `device`; validation errors use wgpu's error scopes
42    /// or uncaptured-error handler. Every call creates fresh shader modules and a pipeline.
43    pub fn initialize_with_prefix_layouts(
44        &self,
45        device: &wgpu::Device,
46        prefix_layouts: &[&wgpu::BindGroupLayout],
47    ) -> wgpu::RenderPipeline {
48        let bind_group_layouts = if let Some(layout) = &self.layout {
49            layout
50                .iter()
51                .map(|entries| {
52                    device.create_bind_group_layout(&wgpu::BindGroupLayoutDescriptor {
53                        label: None,
54                        entries: entries.as_ref(),
55                    })
56                })
57                .collect::<Vec<_>>()
58        } else {
59            vec![]
60        };
61
62        let all_bind_group_layouts = prefix_layouts
63            .iter()
64            .copied()
65            .chain(bind_group_layouts.iter())
66            .map(Some)
67            .collect::<Vec<_>>();
68
69        let pipeline_layout = device.create_pipeline_layout(&wgpu::PipelineLayoutDescriptor {
70            bind_group_layouts: &all_bind_group_layouts,
71            ..Default::default()
72        });
73
74        let vertex_shader_module = device.create_shader_module(wgpu::ShaderModuleDescriptor {
75            label: None,
76            source: wgpu::ShaderSource::Wgsl(self.vertex.source.into()),
77        });
78        let fragment_shader_module = device.create_shader_module(wgpu::ShaderModuleDescriptor {
79            label: None,
80            source: wgpu::ShaderSource::Wgsl(self.fragment.source.into()),
81        });
82
83        let pipeline = device.create_render_pipeline(&wgpu::RenderPipelineDescriptor {
84            label: self.label.as_ref().map(|label| label.as_ref()),
85            layout: Some(&pipeline_layout),
86            vertex: wgpu::VertexState {
87                module: &vertex_shader_module,
88                entry_point: Some(self.vertex.entry_point),
89                compilation_options: Default::default(),
90                buffers: self
91                    .vertex
92                    .buffers
93                    .iter()
94                    .map(|layout| {
95                        Some(wgpu::VertexBufferLayout {
96                            array_stride: layout.array_stride,
97                            step_mode: layout.step_mode,
98                            attributes: layout.attributes.as_slice(),
99                        })
100                    })
101                    .collect::<Vec<_>>()
102                    .as_slice(),
103            },
104            fragment: Some(wgpu::FragmentState {
105                module: &fragment_shader_module,
106                entry_point: Some(self.fragment.entry_point),
107                compilation_options: Default::default(),
108                targets: self.fragment.targets.as_slice(),
109            }),
110            primitive: self.primitive,
111            depth_stencil: self.depth_stencil.clone(),
112            multisample: self.multisample,
113
114            multiview_mask: None,
115            cache: None,
116        });
117
118        pipeline
119    }
120}
