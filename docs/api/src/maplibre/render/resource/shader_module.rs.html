1//! Shader module creation helper with `SharedPipelines` caching and a render-free dummy fallback for `web-tile-worker`.
2
3/// Creates a `wgpu::ShaderModule` directly from static WGSL source without cache lookup.
4#[cfg(any(not(target_arch = "wasm32"), feature = "webgpu", feature = "web-webgl"))]
5pub(super) fn create_shader_module_uncached(
6    device: &wgpu::Device,
7    label: Option<&str>,
8    source: &'static str,
9) -> wgpu::ShaderModule {
10    device.create_shader_module(wgpu::ShaderModuleDescriptor {
11        label,
12        source: wgpu::ShaderSource::Wgsl(std::borrow::Cow::Borrowed(source)),
13    })
14}
15
16/// Dummy fallback for render-free `wasm32` worker builds where `wgsl` is not enabled.
17#[cfg(all(
18    target_arch = "wasm32",
19    not(feature = "webgpu"),
20    not(feature = "web-webgl")
21))]
22pub(super) fn create_shader_module_uncached(
23    device: &wgpu::Device,
24    label: Option<&str>,
25    _source: &'static str,
26) -> wgpu::ShaderModule {
27    device.create_shader_module(wgpu::ShaderModuleDescriptor {
28        label,
29        source: wgpu::ShaderSource::Dummy(std::marker::PhantomData),
30    })
31}
32
33/// Returns a cached [`wgpu::ShaderModule`] from the active [`SharedPipelines`](super::SharedPipelines)
34/// scope when entered, or creates one directly otherwise.
35pub(crate) fn create_shader_module_from_wgsl(
36    device: &wgpu::Device,
37    label: Option<&str>,
38    source: &'static str,
39) -> wgpu::ShaderModule {
40    if let Some(cache) = super::pipeline_cache::SharedPipelines::active() {
41        return cache.shader_module(device, label, source);
42    }
43    create_shader_module_uncached(device, label, source)
44}
