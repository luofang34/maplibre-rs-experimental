12/// A presentation surface with its retained window owner, configured dimensions and format.
13pub struct WindowHead {
14    pub(super) surface: wgpu::Surface<'static>,
15    pub(super) size: PhysicalSize,
16
17    pub(super) texture_format: wgpu::TextureFormat,
18    /// Non-sRGB variant of texture_format used for rendering.
19    /// Prevents automatic linear→sRGB conversion by the GPU, since our colors
20    /// (from CSS) are already in sRGB space.
21    pub(super) render_format: wgpu::TextureFormat,
22    pub(super) present_mode: wgpu::PresentMode,
23    pub(super) texture_format_features: TextureFormatFeatures,
24}
25
26/// Returns the non-sRGB variant of a texture format.
27/// This prevents the GPU from applying automatic linear→sRGB gamma conversion,
28/// which would double-gamma colors that are already in sRGB space (e.g., CSS colors).
29pub(super) fn strip_srgb(format: wgpu::TextureFormat) -> wgpu::TextureFormat {
30    match format {
31        wgpu::TextureFormat::Rgba8UnormSrgb => wgpu::TextureFormat::Rgba8Unorm,
32        wgpu::TextureFormat::Bgra8UnormSrgb => wgpu::TextureFormat::Bgra8Unorm,
33        other => other,
34    }
35}
36
37impl WindowHead {
38    /// Configures a new physical-pixel size using the selected format and presentation mode.
39    ///
40    /// # Panics
41    /// Panics if either dimension is zero. Wgpu also rejects unsupported dimensions or
42    /// configurations, and outstanding acquired frames must be dropped before this call.
43    pub fn resize_and_configure(&mut self, width: u32, height: u32, device: &wgpu::Device) {
44        self.size = PhysicalSize::new(width, height).unwrap();
45        self.configure(device);
46    }
47
48    /// Applies the stored dimensions and presentation settings to the platform surface.
49    /// Drop all acquired frames first. The device must belong to an adapter compatible with
50    /// this surface and support its format, dimensions and presentation mode.
51    pub fn configure(&self, device: &wgpu::Device) {
52        // The base format is implicit; listing it would require unsupported WebGL view formats.
53        let mut view_formats = Vec::new();
54        if self.render_format != self.texture_format {
55            view_formats.push(self.render_format);
56        }
57
58        let surface_config = wgpu::SurfaceConfiguration {
59            alpha_mode: wgpu::CompositeAlphaMode::Auto,
60            usage: wgpu::TextureUsages::RENDER_ATTACHMENT,
61            format: self.texture_format,
62            width: self.size.width(),
63            height: self.size.height(),
64            present_mode: self.present_mode,
65            view_formats,
66            desired_maximum_frame_latency: 2,
67            color_space: Default::default(),
68        };
69
70        self.surface.configure(device, &surface_config);
71    }
72
73    /// Replaces the platform surface while retaining size and presentation settings.
74    ///
75    /// Returns window-handle or surface-creation failures without changing the existing surface.
76    /// Call [`Self::configure`] on success before acquiring frames. The new surface retains
77    /// a cloned window-handle owner independently of the supplied host wrapper.
78    pub fn recreate_surface<MW>(
79        &mut self,
80        window: &MW,
81        instance: &wgpu::Instance,
82    ) -> Result<(), RenderError>
83    where
84        MW: MapWindow + HeadedMapWindow,
85    {
86        self.surface = instance.create_surface(window.handle().clone())?;
87        Ok(())
88    }
89
90    /// Platform surface for acquisition; its configuration is managed by this window head.
91    pub fn surface(&self) -> &wgpu::Surface<'_> {
92        &self.surface
93    }
94}
95
96impl HasChanged for WindowHead {
97    /// Tuple of width and height
98    type Criteria = (u32, u32);
99
100    fn has_changed(&self, criteria: &Self::Criteria) -> bool {
101        self.size.width() != criteria.0 || self.size.height() != criteria.1
102    }
103}
