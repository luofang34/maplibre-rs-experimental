1//! Surface acquisition with one bounded reconfiguration attempt.
2
3use thiserror::Error;
4use wgpu::{CurrentSurfaceTexture, SurfaceTexture};
5
6/// Why a window cannot supply a frame.
7#[derive(Debug, Error, Clone, Copy, PartialEq, Eq)]
8pub enum SurfaceAcquireError {
9    /// The compositor did not supply an image before its deadline.
10    #[error("surface acquisition timed out")]
11    Timeout,
12    /// The compositor cannot display the window.
13    #[error("surface is occluded")]
14    Occluded,
15    /// The surface still needs configuration after a retry.
16    #[error("surface configuration remains outdated")]
17    Outdated,
18    /// The host must recreate the surface.
19    #[error("surface was lost")]
20    Lost,
21    /// The device reported a validation error during acquisition.
22    #[error("surface acquisition failed validation")]
23    Validation,
24}
25
26pub(super) fn acquire(
27    mut next: impl FnMut() -> CurrentSurfaceTexture,
28    mut configure: impl FnMut(),
29) -> Result<SurfaceTexture, SurfaceAcquireError> {
30    for attempt in 0..2 {
31        match next() {
32            CurrentSurfaceTexture::Success(texture) => return Ok(texture),
33            CurrentSurfaceTexture::Timeout => return Err(SurfaceAcquireError::Timeout),
34            CurrentSurfaceTexture::Occluded => return Err(SurfaceAcquireError::Occluded),
35            CurrentSurfaceTexture::Lost => return Err(SurfaceAcquireError::Lost),
36            CurrentSurfaceTexture::Validation => return Err(SurfaceAcquireError::Validation),
37            CurrentSurfaceTexture::Suboptimal(texture) => {
38                // Reconfiguration requires every outstanding surface texture to be dropped.
39                drop(texture);
40            }
41            CurrentSurfaceTexture::Outdated => {}
42        }
43        if attempt == 0 {
44            configure();
45        }
46    }
47    Err(SurfaceAcquireError::Outdated)
48}
