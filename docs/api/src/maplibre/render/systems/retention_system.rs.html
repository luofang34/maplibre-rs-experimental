1//! Evicts tiles that left the view, keeping a recently used cache sized from the viewport.
2//!
3//! The tile store would otherwise grow for the whole session. GL JS keeps the tiles its
4//! covering needs plus a cache of `MAX_TILE_CACHE_ZOOM_LEVELS` times the tiles in view and
5//! unloads the rest; this system does the same once per frame in the Cleanup stage. A tile
6//! whose worker request is still in flight is never evicted, so its result cannot land on a
7//! re-requested tile and duplicate its layers. Tiles the frame requests but does not draw,
8//! such as those around an external eye, count as in use too: evicting one would only have
9//! the request systems fetch it again on the next frame. Only drawn tiles size the cache,
10//! as GL JS sizes it from the viewport; a wide request would otherwise keep every tile a
11//! flight passes resident.
12

40/// Out-of-view tiles kept per tile in use, as GL JS `MAX_TILE_CACHE_ZOOM_LEVELS`.
41const CACHE_TILES_PER_VIEW_TILE: usize = 5;
42/// Smallest cache, so a tiny viewport still keeps some history for panning back.
43const MIN_CACHE_TILES: usize = 64;
44/// Tessellated geometry the out-of-view cache may hold. The tile count is GL JS's, sized for
45/// a flat viewport of a few dozen tiles; an eye on terrain draws hundreds, and a planet tile
46/// at low zoom weighs tens of megabytes, so the cache is bounded in bytes as well.
47const CACHE_BYTES: usize = 128 << 20;
48/// Frames between resident-memory summaries for hosts driving the map from an eye, which
49/// run on devices with a memory limit.
50const SUMMARY_EVERY_FRAMES: u64 = 90;
51
52/// What the out-of-view cache may hold.
53#[derive(Clone, Copy, Debug)]
54pub(crate) struct CacheBudget {
55    pub tiles: usize,
56    pub bytes: usize,
57}

69/// Drops the least recently used tiles beyond the cache budget and releases their GPU data.
70pub fn retention_system(
71    MapContext {
72        world,
73        style,
74        view_state,
75        #[cfg(not(target_arch = "wasm32"))]
76        renderer,
77        ..
78    }: &mut MapContext,
79) -> SystemResult {
80    if crate::render::eye_covering::EyeInFrame::reuses_content(world) {
81        return Ok(());
82    }
83    let drawn = drawn_tiles(world).len();
84    let in_use = tiles_in_use(world, style, view_state);
85    drop_cancelled(world, &in_use);
86    let evicted = evict_stale_tiles(world, &in_use, drawn);
87    if !evicted.is_empty() {
88        tracing::debug!(count = evicted.len(), "evicted tiles that left the view");
89        drop_gpu_data(world, &evicted);
90    }
91    if view_state.has_external_view() {
92        summarize_residency(world, drawn, in_use.len(), view_state.eye_settled());
93        #[cfg(not(target_arch = "wasm32"))]
94        summarize_gpu_objects(world, renderer);
95        summarize_timings(world);
96    }
97    Ok(())
98}

100/// Once a second, how many GPU objects the frame keeps alive, for a host whose footprint
101/// climbs while nothing loads: the kind whose count climbs with it is the one leaking.
102#[cfg(not(target_arch = "wasm32"))]
103fn summarize_gpu_objects(world: &World, renderer: &crate::render::Renderer) {
104    let frame = world
105        .resources
106        .get::<TileRetention>()
107        .map_or(0, |retention| retention.frame);
108    if !frame.is_multiple_of(SUMMARY_EVERY_FRAMES) {
109        return;
110    }
111    let Some(report) = renderer.instance.generate_report() else {
112        return;
113    };
114    let hub = report.hub_report();
115    tracing::info!(
116        buffers = hub.buffers.num_allocated,
117        textures = hub.textures.num_allocated,
118        texture_views = hub.texture_views.num_allocated,
119        bind_groups = hub.bind_groups.num_allocated,
120        samplers = hub.samplers.num_allocated,
121        command_buffers = hub.command_buffers.num_allocated,
122        pipelines = hub.render_pipelines.num_allocated,
123        "gpu objects"
124    );
125}

127/// Once a second, the costliest systems and stages of the frame, for a host that cannot
128/// attach a profiler.
129fn summarize_timings(world: &mut World) {
130    let frame = world
131        .resources
132        .get::<TileRetention>()
133        .map_or(0, |retention| retention.frame);
134    if !frame.is_multiple_of(SUMMARY_EVERY_FRAMES) {
135        return;
136    }
137    let timings = world.resources.get_or_init_mut::<FrameTimings>();
138    let growers: Vec<String> = timings
139        .top_growth(6)
140        .iter()
141        .filter(|(_, kb)| *kb > 0.5)
142        .map(|(name, kb)| format!("{name}={kb:.1}"))
143        .collect();
144    let top = timings.take_top(8);
145    let costliest: Vec<String> = top
146        .iter()
147        .map(|(name, ms)| format!("{name}={ms:.2}"))
148        .collect();
149    tracing::info!(ms_per_frame = %costliest.join(" "), "frame time by system");
150    if !growers.is_empty() {
151        tracing::info!(
152            kb_per_frame = %growers.join(" "),
153            rust_heap_mb = live_bytes() >> 20,
154            "heap growth by system"
155        );
156    }
157}

159fn summarize_residency(world: &World, drawn: usize, in_use: usize, settled: bool) {
160    let frame = world
161        .resources
162        .get::<TileRetention>()
163        .map_or(0, |retention| retention.frame);
164    if !frame.is_multiple_of(SUMMARY_EVERY_FRAMES) {
165        return;
166    }
167    let tiles = world.tiles.tiles.len();
168    let bytes: usize = world
169        .tiles
170        .tiles
171        .values()
172        .map(|tile| tile_bytes(&world.tiles, tile.coords))
173        .sum();
174    let index_bytes = world.tiles.geometry_index.approximate_bytes();
175    let pool_revision = match world.resources.get::<Eventually<VectorBufferPool>>() {
176        Some(Initialized(pool)) => pool.revision(),
177        _ => 0,
178    };
179    let (drapes, free_drapes, drape_bytes, dem_textures, dem_bytes) =
180        match world.resources.get::<Eventually<TerrainResources>>() {
181            Some(Initialized(terrain)) => {
182                let (held, free) = terrain.drape_counts();
183                let (drape_bytes, dem_bytes) = terrain.texture_bytes();
184                (
185                    held,
186                    free,
187                    drape_bytes,
188                    terrain.dem_texture_count(),
189                    dem_bytes,
190                )
191            }
192            _ => (0, 0, 0, 0, 0),
193        };
194    tracing::info!(
195        tiles,
196        pending = crate::io::tile_backpressure::tiles_in_flight(&world.tiles),
197        settled,
198        geometry_mb = bytes >> 20,
199        index_mb = index_bytes >> 20,
200        pool_revision,
201        drawn,
202        in_use,
203        drapes,
204        free_drapes,
205        drape_mb = drape_bytes >> 20,
206        dem_textures,
207        dem_mb = dem_bytes >> 20,
208        finest_dem_zoom = finest_dem_zoom(world),
209        symbol_revision = symbol_revision(world),
210        "resident tiles"
211    );
212}

214/// The tiles the frame draws from, with the tiles whose shapes stand in for them.
215pub(crate) fn drawn_tiles(world: &World) -> HashSet<WorldTileCoords> {
216    let mut drawn = HashSet::new();
217    if let Some(Initialized(pattern)) = world.resources.get::<Eventually<WgpuTileViewPattern>>() {
218        for view_tile in pattern.iter() {
219            drawn.insert(view_tile.coords());
220            view_tile.render(|shape| {
221                drawn.insert(shape.coords());
222            });
223        }
224    }
225    if let Some(covering) = world
226        .resources
227        .get::<crate::sdf::covering::SymbolCovering>()
228    {
229        drawn.extend(covering.tiles.iter().copied());
230    }
231    drawn
232}

234/// Every tile the current frame draws from or requests, with the DEM tiles and ancestors
235/// terrain reads.
236pub(crate) fn tiles_in_use(
237    world: &World,
238    style: &Style,
239    view_state: &ViewState,
240) -> HashSet<WorldTileCoords> {
241    let mut in_use = drawn_tiles(world);
242    match view_region_for_projection(
243        style,
244        view_state,
245        world,
246        view_state.zoom().zoom_level(DEFAULT_TILE_SIZE),
247        ViewStatePadding::Loose,
248    ) {
249        Ok(Some(requested)) => {
250            // The request systems fetch the ancestor at the source's maximum zoom in place of
251            // a finer tile, so that is the tile to keep.
252            let max_zoom = source_max_zoom(style, TileKind::Vector);
253            for coords in requested.iter() {
254                in_use.insert(coords);
255                in_use.insert(clamp_to_max_zoom(coords, max_zoom));
256            }
257        }
258        Ok(None) => {}
259        Err(error) => tracing::warn!(%error, "cannot select the requested tiles to keep"),
260    }
261    if let Some(requests) = world
262        .resources
263        .get::<crate::terrain::request_system::DrapeRequests>()
264    {
265        in_use.extend(requests.0.iter().copied());
266    }
267    if let Some(prefetch) = world
268        .resources
269        .get::<crate::terrain::request_system::DrapePrefetchRequests>()
270    {
271        in_use.extend(prefetch.0.iter().copied());
272    }
273    if let Some(dem) = dem_source(style) {
274        let view: Vec<WorldTileCoords> = in_use.iter().copied().collect();
275        for coords in view {
276            let Some(mut dem_coords) = dem_tile_coords(coords, dem.minzoom, dem.maxzoom) else {
277                continue;
278            };
279            // The coverage index falls back through every ancestor while finer tiles load.
280            loop {
281                in_use.insert(dem_coords);
282                match dem_coords.get_parent() {
283                    Some(parent) if u8::from(parent.z) >= dem.minzoom => dem_coords = parent,
284                    _ => break,
285                }
286            }
287        }
288    }
289    in_use
290}

292/// Removes the least recently used settled tiles beyond the cache budget and returns them;
293/// the tile budget follows `view_tiles`, the tiles the frame draws.
294pub(crate) fn evict_stale_tiles(
295    world: &mut World,
296    in_use: &HashSet<WorldTileCoords>,
297    view_tiles: usize,
298) -> Vec<WorldTileCoords> {
299    if world.resources.get::<RetainLoadedTiles>().is_some() {
300        return evict_beyond(
301            world,
302            in_use,
303            CacheBudget {
304                tiles: usize::MAX,
305                bytes: usize::MAX,
306            },
307        );
308    }
309    // A host nearly out of memory keeps nothing beyond what the frame draws from.
310    let critical = world
311        .resources
312        .get::<MemoryBudget>()
313        .is_some_and(|budget| budget.pressure() == MemoryPressure::Critical);
314    let budget = if critical {
315        CacheBudget { tiles: 0, bytes: 0 }
316    } else {
317        CacheBudget {
318            tiles: (view_tiles * CACHE_TILES_PER_VIEW_TILE).max(MIN_CACHE_TILES),
319            bytes: CACHE_BYTES,
320        }
321    };
322    evict_beyond(world, in_use, budget)
323}

325/// Removes the least recently used settled tiles until the rest fit `budget`, in tiles and
326/// in bytes of geometry, and returns them.
327pub(crate) fn evict_beyond(
328    world: &mut World,
329    in_use: &HashSet<WorldTileCoords>,
330    budget: CacheBudget,
331) -> Vec<WorldTileCoords> {
332    let World { resources, tiles } = world;
333    let retention = resources.get_or_init_mut::<TileRetention>();
334    retention.frame = retention.frame.wrapping_add(1);
335    let frame = retention.frame;
336    for coords in in_use {
337        retention.last_used.insert(*coords, frame);
338    }
339
340    let mut candidates: Vec<(u64, WorldTileCoords, usize)> = tiles
341        .tiles
342        .values()
343        .map(|tile| tile.coords)
344        .filter(|coords| !in_use.contains(coords) && is_settled(tiles, *coords))
345        .map(|coords| {
346            (
347                retention.last_used.get(&coords).copied().unwrap_or(0),
348                coords,
349                tile_bytes(tiles, coords),
350            )
351        })
352        .collect();
353    let mut kept = candidates.len();
354    let mut kept_bytes: usize = candidates.iter().map(|(_, _, bytes)| bytes).sum();
355    if kept <= budget.tiles && kept_bytes <= budget.bytes {
356        return Vec::new();
357    }
358    candidates.sort_by_key(|(used, _, _)| *used);
359    let mut evicted = Vec::new();
360    for (_, coords, bytes) in candidates {
361        if kept <= budget.tiles && kept_bytes <= budget.bytes {
362            break;
363        }
364        kept -= 1;
365        kept_bytes -= bytes;
366        evicted.push(coords);
367    }
368    for coords in &evicted {
369        tiles.remove(*coords);
370        retention.last_used.remove(coords);
371    }
372    evicted
373}

375/// Drops the tiles whose requests no one wants any more, freeing their slots for tiles in view.
376fn drop_cancelled(world: &mut World, in_use: &HashSet<WorldTileCoords>) {
377    let cancelled = crate::io::tile_retry::cancel_unwanted(world, in_use);
378    if cancelled.is_empty() {
379        return;
380    }
381    tracing::debug!(
382        count = cancelled.len(),
383        "cancelled requests for tiles out of view"
384    );
385    let retention = world.resources.get_or_init_mut::<TileRetention>();
386    for coords in &cancelled {
387        retention.last_used.remove(coords);
388    }
389    for coords in &cancelled {
390        world.tiles.remove(*coords);
391    }
392    drop_gpu_data(world, &cancelled);
393}

395fn drop_gpu_data(world: &mut World, evicted: &[WorldTileCoords]) {
396    if let Some(Initialized(pool)) = world.resources.get_mut::<Eventually<VectorBufferPool>>() {
397        for coords in evicted {
398            pool.remove_tile(*coords);
399        }
400    }
401    if let Some(Initialized(pool)) = world
402        .resources
403        .get_mut::<Eventually<crate::sdf::SymbolBufferPool>>()
404    {
405        for coords in evicted {
406            pool.remove_tile(*coords);
407        }
408    }
409    if let Some(Initialized(raster)) = world.resources.get_mut::<Eventually<RasterResources>>() {
410        for coords in evicted {
411            raster.remove_texture(*coords);
412        }
413    }
414    if let Some(Initialized(terrain)) = world.resources.get_mut::<Eventually<TerrainResources>>() {
415        for coords in evicted {
416            terrain.drop_dem(*coords);
417        }
418    }
419}

424fn finest_dem_zoom(world: &World) -> u8 {
425    world
426        .tiles
427        .tiles
428        .values()
429        .filter_map(|tile| {
430            matches!(
431                world
432                    .tiles
433                    .query::<&crate::terrain::DemTileComponent>(tile.coords),
434                Some(crate::terrain::DemTileComponent::Loaded(_))
435            )
436            .then_some(u8::from(tile.coords.z))
437        })
438        .max()
439        .unwrap_or(0)
440}
