1//! Uploads data to the GPU which is needed for rendering.
2use crate::{
3    context::MapContext,
4    render::{
5        eventually::{Eventually, Eventually::Initialized},
6        projection::{projection_data_for_view, ProjectionGpuResources},
7        tile_mesh::{GlobeTileMeshCache, TileMeshUsage},
8        tile_view_pattern::WgpuTileViewPattern,
9        Renderer,
10    },
11    tcs::system::{SystemError, SystemResult},
12};
13
14pub fn upload_system(
15    MapContext {
16        world,
17        style,
18        view_state,
19        renderer: Renderer { device, queue, .. },
20        ..
21    }: &mut MapContext,
22) -> SystemResult {
23    let background_tiles = crate::background::queue_system::globe_tiles(world);
24    let Some((Initialized(tile_view_pattern), Initialized(projection_resources), tile_mesh_cache)) =
25        world.resources.query_mut::<(
26            &mut Eventually<WgpuTileViewPattern>,
27            &mut Eventually<ProjectionGpuResources>,
28            &mut GlobeTileMeshCache,
29        )>()
30    else {
31        return Err(SystemError::Dependencies);
32    };
33
34    let view_proj = view_state.gpu_view_projection();
35    tile_view_pattern.upload_pattern(
36        queue,
37        &view_proj,
38        view_state.width() as f32,
39        view_state.height() as f32,
40        view_state.style_zoom(),
41    );
42    let projection_data = projection_data_for_view(style, view_state).map_err(|error| {
43        tracing::error!(%error, "unable to prepare projection state");
44        SystemError::Setup
45    })?;
46    projection_resources.upload(queue, projection_data);
47
48    if projection_data.transition > 0.0 {
49        for coords in background_tiles {
50            tile_mesh_cache
51                .prepare(device, queue, coords, TileMeshUsage::Raster, false)
52                .map_err(|error| {
53                    tracing::error!(%error, %coords, "unable to prepare globe background mesh");
54                    SystemError::Setup
55                })?;
56        }
57    }
58
59    let mut visible_tiles = Vec::new();
60    for view_tile in tile_view_pattern.iter() {
61        view_tile.render(|shape| visible_tiles.push(shape.coords()));
62    }
63    for coords in visible_tiles {
64        for (usage, generate_borders) in [
65            (TileMeshUsage::Stencil, true),
66            (TileMeshUsage::Stencil, false),
67            (TileMeshUsage::Raster, true),
68            (TileMeshUsage::Raster, false),
69        ] {
70            tile_mesh_cache
71                .prepare(device, queue, coords, usage, generate_borders)
72                .map_err(|error| {
73                    tracing::error!(%error, %coords, "unable to prepare globe tile mesh");
74                    SystemError::Setup
75                })?;
76        }
77    }
78
79    Ok(())
80}
