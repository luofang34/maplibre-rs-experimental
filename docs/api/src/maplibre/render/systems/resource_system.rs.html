1//! Prepares GPU-owned resources by initializing them if they are uninitialized or out-of-date.
2
3use std::{borrow::Cow, mem};
4
5use crate::{
6    context::MapContext,
7    render::{
8        depth_copy::DepthCopyPipeline,
9        eventually::Eventually,
10        projection::ProjectionGpuResources,
11        resource::{BackingBufferDescriptor, RenderPipeline, Texture, TilePipeline},
12        settings::Msaa,
13        shaders,
14        shaders::{Shader, ShaderTileMetadata},
15        tile_view_pattern::{TileViewPattern, WgpuTileViewPattern, DEFAULT_TILE_VIEW_PATTERN_SIZE},
16        MaskPipeline, Renderer,
17    },
18    tcs::system::{System, SystemError, SystemResult},
19};
20
21#[derive(Default)]
22pub struct ResourceSystem;
23
24impl System for ResourceSystem {
25    fn name(&self) -> Cow<'static, str> {
26        "resource_system".into()
27    }
28
29    fn run(
30        &mut self,
31        MapContext {
32            renderer:
33                Renderer {
34                    settings,
35                    device,
36                    queue,
37                    adapter,
38                    resources: state,
39                    ..
40                },
41            world,
42            ..
43        }: &mut MapContext,
44    ) -> SystemResult {
45        if world
46            .resources
47            .get::<crate::render::tracked_pass::DrawCapabilities>()
48            .is_none()
49        {
50            world
51                .resources
52                .insert(crate::render::tracked_pass::DrawCapabilities::of(adapter));
53        }
54        let Some((tile_view_pattern, mask_pipeline, projection_resources, depth_copy)) =
55            world.resources.query_mut::<(
56                &mut Eventually<WgpuTileViewPattern>,
57                &mut Eventually<MaskPipeline>,
58                &mut Eventually<ProjectionGpuResources>,
59                &mut Eventually<DepthCopyPipeline>,
60            )>()
61        else {
62            return Err(SystemError::Dependencies);
63        };
64
65        let surface = &mut state.surface;
66
67        let size = surface.size();
68
69        surface.reconfigure(device);
70
71        let projection_resources =
72            projection_resources.initialize(|| ProjectionGpuResources::new(device, queue));
73
74        if matches!(state.render_target, Eventually::Uninitialized) {
75            let view = surface
76                .create_view(device)
77                .map_err(crate::render::error::RenderError::from)?;
78            state.render_target = Eventually::Initialized(view);
79        }
80
81        let depth_msaa = if surface.is_multisampling_supported(settings.msaa) {
82            settings.msaa
83        } else {
84            Msaa { samples: 1 }
85        };
86        state.depth_texture.reinitialize(
87            || {
88                Texture::new(
89                    Some("depth texture"),
90                    device,
91                    settings.depth_texture_format,
92                    size.width(),
93                    size.height(),
94                    depth_msaa,
95                    // A host compositor reads the depth back through the depth copy pass.
96                    wgpu::TextureUsages::RENDER_ATTACHMENT | wgpu::TextureUsages::TEXTURE_BINDING,
97                )
98            },
99            &(size.width(), size.height()),
100        );
101        if state.eye_depth_target.is_some() {
102            depth_copy.initialize(|| DepthCopyPipeline::new(device, depth_msaa.samples));
103        }
104
105        state.multisampling_texture.reinitialize(
106            || {
107                if settings.msaa.is_multisampling()
108                    && surface.is_multisampling_supported(settings.msaa)
109                {
110                    Some(Texture::new(
111                        Some("multisampling texture"),
112                        device,
113                        surface.surface_format(),
114                        size.width(),
115                        size.height(),
116                        settings.msaa,
117                        wgpu::TextureUsages::RENDER_ATTACHMENT,
118                    ))
119                } else {
120                    None
121                }
122            },
123            &(size.width(), size.height()),
124        );
125
126        tile_view_pattern.initialize(|| {
127            let tile_view_buffer_desc = wgpu::BufferDescriptor {
128                label: Some("tile view buffer"),
129                size: mem::size_of::<ShaderTileMetadata>() as wgpu::BufferAddress
130                    * DEFAULT_TILE_VIEW_PATTERN_SIZE,
131                usage: wgpu::BufferUsages::VERTEX | wgpu::BufferUsages::COPY_DST,
132                mapped_at_creation: false,
133            };
134
135            TileViewPattern::new(BackingBufferDescriptor::new(
136                device.create_buffer(&tile_view_buffer_desc),
137                tile_view_buffer_desc.size,
138            ))
139        });
140
141        mask_pipeline.initialize(|| {
142            let mask_shader = shaders::TileMaskShader {
143                format: surface.surface_format(),
144                draw_colors: false,
145                debug_lines: false,
146            };
147
148            let pipeline = TilePipeline::new(
149                "mask_pipeline".into(),
150                *settings,
151                mask_shader.describe_vertex(),
152                mask_shader.describe_fragment(),
153                crate::render::resource::TilePipelineOptions {
154                    depth_stencil_enabled: true,
155                    update_stencil: true,
156                    debug_stencil: false,
157                    wireframe: false,
158                    multisampling: surface.is_multisampling_supported(settings.msaa),
159                    textured: false,
160                },
161            )
162            .describe_render_pipeline()
163            .initialize_with_prefix_layouts(device, &[projection_resources.bind_group_layout()]);
164            MaskPipeline(pipeline)
165        });
166
167        Ok(())
168    }
169}
