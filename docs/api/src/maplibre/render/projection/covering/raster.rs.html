15/// The tiles each raster source used by a visible layer covers, following that source's tile
16/// size, rounding and zoom range through the same covering as the view.
17pub fn raster_source_regions(
18    style: &Style,
19    view_state: &ViewState,
20    world: &World,
21    padding: ViewStatePadding,
22) -> Result<RasterCoverings, ProjectionStateError> {
23    if !style
24        .layers
25        .iter()
26        .any(|layer| RASTER_LAYER_TYPES.contains(&layer.type_.as_str()))
27    {
28        return Ok(Vec::new());
29    }
30    let zoom = view_state.zoom().value();
31    let style_zoom = view_state.style_zoom().value();
32    let mut regions = Vec::new();
33    for group in source_layer_groups(style, TileKind::Raster) {
34        if !group
35            .layers
36            .iter()
37            .any(|layer| layer.is_visible_at(style_zoom))
38        {
39            continue;
40        }
41        let source = RasterSourceId::new(group.source_name);
42        let definition = source.name().and_then(|name| style.sources.get(name));
43        let mut request = raster_covering_request(zoom, definition);
44        if !request.zoom_range.serves(request.level) && style_zoom > zoom {
45            request.level = raster_covering_request(style_zoom, definition).level;
46        }
47        let history = world
48            .resources
49            .get::<FrameLodHistory>()
50            .and_then(|history| history.raster.get(&source));
51        let tiles =
52            covering_region_with_history(style, view_state, world, request, padding, history)?
53                .map_or_else(Vec::new, |region| {
54                    region
55                        .iter()
56                        .filter(|coords| coords.build_quad_key().is_some())
57                        .collect()
58                });
59        regions.push((source, tiles));
60    }
61    Ok(regions)
62}
63
64fn raster_covering_request(zoom: f64, source: Option<&Source>) -> super::CoveringRequest {
65    let (tile_size, minzoom, maxzoom) = match source {
66        Some(Source::Raster(source)) => (
67            source.tile_size.map_or(TILE_SIZE, f64::from),
68            source.minzoom,
69            source.maxzoom,
70        ),
71        Some(Source::RasterDem(source)) => (
72            f64::from(source.tile_size()),
73            source.minzoom,
74            source.maxzoom,
75        ),
76        _ => (TILE_SIZE, None, None),
77    };
78    super::CoveringRequest::raster_source(zoom, tile_size, minzoom, maxzoom)
79}
