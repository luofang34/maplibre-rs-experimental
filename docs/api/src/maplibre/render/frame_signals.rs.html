1//! What a host needs to render on demand: whether the map needs another frame, which resources
2//! became ready, and what each frame cost.
3//!
4//! The map is dirty after data arrives or the style changes, stays animating while labels fade
5//! or terrain drapes wait for their redraw budget, needs a frame while tile requests are in
6//! flight, and needs one after the camera moves. A frame's statistics are gathered as its stages
7//! run and closed at the end of the schedule.
8
9use std::time::Duration;

11use crate::{
12    context::MapContext, coords::WorldTileCoords, io::tile_retry::RequestKind, tcs::world::World,
13};

15/// A resource that finished loading, as a host learns it.
16#[derive(Clone, Copy, Debug, PartialEq, Eq)]
17pub enum ResourceReady {
18    /// A tile request of `kind` finished; `loaded` is false when it failed and will be retried.
19    Tile {
20        /// The tile.
21        coords: WorldTileCoords,
22        /// What was requested for it.
23        kind: RequestKind,
24        /// Whether the request delivered its data.
25        loaded: bool,
26    },
27    /// A tile's glyphs and icons reached the GPU, so its labels can draw.
28    SymbolAtlas {
29        /// The tile.
30        coords: WorldTileCoords,
31    },
32}
33
34/// What one frame cost.
35#[derive(Clone, Debug, Default, PartialEq)]
36pub struct FrameStats {
37    /// The frame's number, counting from one.
38    pub frame: u64,
39    /// The host's frame this belongs to: one per frame, or one per XR frame whatever its eyes,
40    /// which a frame timeline records the frame under.
41    pub host_frame: u64,
42    /// CPU time of each schedule stage, in the order they ran.
43    pub stages: Vec<(String, Duration)>,
44    /// GPU time of the latest main pass whose timestamps came back, usually the previous
45    /// frame's; absent when the device cannot time passes or none came back this frame.
46    pub gpu: Option<Duration>,
47    /// Bytes written to GPU buffers and textures.
48    pub upload_bytes: u64,
49    /// Terrain drapes drawn again.
50    pub drape_redraws: u32,
51    /// Draw calls recorded.
52    pub draws: u32,
53}
54
55impl FrameStats {
56    /// The CPU time of all stages together.
57    pub fn cpu(&self) -> Duration {
58        self.stages.iter().map(|(_, spent)| *spent).sum()
59    }
60}

62/// The signals a frame leaves for its host.
63#[derive(Default)]
64pub struct FrameSignals {
65    current: FrameStats,
66    last: FrameStats,
67    frame: u64,
68    events: Vec<ResourceReady>,
69    dirty: bool,
70    animating: bool,
71    host_frame: u64,
72    /// The camera each eye drew last, by eye index; a map without eyes is eye 0.
73    cameras: Vec<[f64; 24]>,
74}
75
76fn signals(world: &mut World) -> &mut FrameSignals {
77    world.resources.get_or_init_mut::<FrameSignals>()
78}
79
80/// The map changed in a way the next frame shows.
81pub(crate) fn mark_dirty(world: &mut World) {
82    signals(world).dirty = true;
83}
84
85/// Something on screen is still moving, so the frame after this one differs from it.
86pub(crate) fn keep_animating(world: &mut World) {
87    signals(world).animating = true;
88}
89
90/// A resource finished loading; the map is dirty and the host is told.
91pub(crate) fn resource_ready(world: &mut World, ready: ResourceReady) {
92    let signals = signals(world);
93    signals.dirty = true;
94    signals.events.push(ready);
95}

97/// Whether the camera differs from the one the last frame drew, as while a gesture runs.
98pub(crate) fn camera_moved(
99    world: &World,
100    view_state: &crate::render::view_state::ViewState,
101) -> bool {
102    world
103        .resources
104        .get::<FrameSignals>()
105        .and_then(|signals| signals.cameras.get(eye_index(world)))
106        .is_some_and(|drawn| *drawn != camera_of(view_state))
107}
108
109/// The eye being drawn: its index in an XR frame, 0 otherwise.
110fn eye_index(world: &World) -> usize {
111    world
112        .resources
113        .get::<crate::render::eye_covering::EyeInFrame>()
114        .map_or(0, |eye| eye.index)
115}

117/// Charges a stage's CPU time to the current frame.
118pub(crate) fn record_stage(world: &mut World, stage: String, spent: Duration) {
119    signals(world).current.stages.push((stage, spent));
120}

122/// Counts terrain drapes drawn again this frame.
123pub(crate) fn count_drape_redraws(world: &mut World, redraws: usize) {
124    let current = &mut signals(world).current;
125    current.drape_redraws = current
126        .drape_redraws
127        .saturating_add(u32::try_from(redraws).unwrap_or(u32::MAX));
128}

130/// Records the GPU time of the latest measured main pass, which reaches the CPU a frame or more
131/// after its own frame.
132pub(crate) fn record_gpu_time(world: &mut World, host_frame: u64, spent: Duration) {
133    signals(world).current.gpu = Some(spent);
134    // The timeline files the time under the frame whose pass it measured.
135    if let Some(trace) = world
136        .resources
137        .get_mut::<crate::render::frame_trace::FrameTraceSlot>()
138        .and_then(|slot| slot.0.as_mut())
139    {
140        trace.record_span(
141            host_frame,
142            "map",
143            crate::render::frame_trace::Clock::Gpu,
144            spent,
145        );
146    }
147}

149/// The host frame the frame being drawn belongs to: a new one, unless it is a later eye of
150/// an XR frame.
151pub(crate) fn host_frame_in_progress(world: &World) -> u64 {
152    let last = world
153        .resources
154        .get::<FrameSignals>()
155        .map_or(0, |signals| signals.host_frame);
156    let later_eye = world
157        .resources
158        .get::<crate::render::eye_covering::EyeInFrame>()
159        .is_some_and(|eye| eye.index > 0);
160    if later_eye {
161        last
162    } else {
163        last.wrapping_add(1)
164    }
165}

167fn camera(context: &MapContext) -> [f64; 24] {
168    camera_of(&context.view_state)
169}
170
171fn camera_of(view_state: &crate::render::view_state::ViewState) -> [f64; 24] {
172    let matrix: [[f64; 4]; 4] = view_state.view_projection().0.into();
173    let mut flat = [0.0; 24];
174    for (column, values) in matrix.iter().enumerate() {
175        flat[column * 4..column * 4 + 4].copy_from_slice(values);
176    }
177    // Over a cap the flat camera stops at the last row of tiles while a free pose moves on.
178    if let Some(pose) = view_state.globe_pose() {
179        flat[16..19].copy_from_slice(&pose.target);
180        flat[19..23].copy_from_slice(&pose.orientation);
181        flat[23] = pose.distance_meters;
182    }
183    flat
184}

186/// Closes the frame: its uploads and draws are counted, its statistics become the last frame's,
187/// and the camera it drew is remembered.
188pub(crate) fn finish_frame(context: &mut MapContext) {
189    let upload_bytes = context.renderer.queue.take_written_bytes();
190    let draws = context
191        .world
192        .resources
193        .get::<crate::render::tracked_pass::RenderStats>()
194        .map_or(0, |stats| stats.peek().draws);
195    let drawn = camera(context);
196    let eye = eye_index(&context.world);
197    let host_frame = host_frame_in_progress(&context.world);
198    let signals = signals(&mut context.world);
199    signals.host_frame = host_frame;
200    signals.frame = signals.frame.wrapping_add(1);
201    let mut finished = std::mem::take(&mut signals.current);
202    finished.frame = signals.frame;
203    finished.host_frame = host_frame;
204    finished.upload_bytes = upload_bytes;
205    finished.draws = draws;
206    signals.last = finished;
207    signals.dirty = std::mem::take(&mut signals.animating);
208    if signals.cameras.len() <= eye {
209        signals.cameras.resize(eye + 1, drawn);
210    }
211    signals.cameras[eye] = drawn;
212    if let Some((signals, slot)) = context.world.resources.query_mut::<(
213        &mut FrameSignals,
214        &mut crate::render::frame_trace::FrameTraceSlot,
215    )>() {
216        if let Some(trace) = slot.0.as_mut() {
217            trace.record_map_frame(&signals.last);
218        }
219    }
220}

222impl MapContext {
223    /// Whether the map should draw another frame: something changed or still animates since
224    /// the last one, tiles are still loading, or the camera moved.
225    pub fn needs_redraw(&self) -> bool {
226        let loading = crate::io::tile_retry::needs_frame(&self.world)
227            || crate::sdf::provided::awaiting(&self.world);
228        let Some(signals) = self.world.resources.get::<FrameSignals>() else {
229            return true;
230        };
231        signals.dirty
232            || loading
233            || signals.cameras.get(eye_index(&self.world)) != Some(&camera(self))
234    }
235
236    /// The statistics of the last finished frame.
237    pub fn last_frame_stats(&self) -> FrameStats {
238        self.world
239            .resources
240            .get::<FrameSignals>()
241            .map(|signals| signals.last.clone())
242            .unwrap_or_default()
243    }
244
245    /// The resources that became ready since the last call.
246    pub fn take_ready_resources(&mut self) -> Vec<ResourceReady> {
247        std::mem::take(&mut signals(&mut self.world).events)
248    }
249}
