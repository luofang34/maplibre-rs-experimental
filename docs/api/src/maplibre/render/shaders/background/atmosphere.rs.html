16/// Per-draw inputs for GL JS-compatible atmospheric scattering.
17#[repr(C)]
18#[derive(Copy, Clone, Debug, Pod, Zeroable)]
19pub struct AtmosphereLayerMetadata {
20    pub(super) inverse_projection: Mat4x4f32,
21    pub(super) globe_position: [f32; 4],
22    pub(super) sun_direction: [f32; 4],
23    pub(super) radius_blend_padding: [f32; 4],
24}
25
26impl AtmosphereLayerMetadata {
27    /// Creates inert metadata for frames where no atmosphere draw is queued.
28    pub fn disabled() -> Self {
29        Self {
30            inverse_projection: cgmath::Matrix4::from_scale(1.0).into(),
31            globe_position: [0.0; 4],
32            sun_direction: [0.0, 0.0, 1.0, 1.0],
33            radius_blend_padding: [1.0, 0.0, Body::EARTH.radius_meters as f32, 0.0],
34        }
35    }
36
37    /// The globe's centre in view axes, the sun's direction in view axes with the share of the
38    /// sun an eclipse leaves as w, and the globe's radius in pixels, the blend and the body's
39    /// radius in metres, as the scattering takes them.
40    pub fn sky_inputs(&self) -> [[f32; 4]; 3] {
41        [
42            self.globe_position,
43            self.sun_direction,
44            self.radius_blend_padding,
45        ]
46    }
47
48    /// Creates atmosphere inputs from the current globe camera and the host's sky: its sun, or
49    /// the root light while the host has set none, and the share of the sun an eclipse leaves.
50    pub fn from_view(
51        camera: &GlobeCameraState,
52        light: &LightSpecification,
53        sky: SceneLighting,
54        zoom: f64,
55        blend: f32,
56    ) -> Result<Self, AtmosphereMetadataError> {
57        let inverse_projection = camera
58            .inverse_projection()
59            .cast::<f32>()
60            .ok_or(AtmosphereMetadataError::FloatConversion)?;
61        let globe_position = camera
62            .globe_center_in_view()
63            .cast::<f32>()
64            .ok_or(AtmosphereMetadataError::FloatConversion)?;
65        let sun_direction = match sky.sun_direction {
66            Some(sun) => camera
67                .world_direction_to_view(sun)
68                .ok_or(LightError::InvalidViewDirection),
69            None => light.sun_direction_in_view(camera, zoom),
70        }
71        .map_err(|source| AtmosphereMetadataError::Light { source })?
72        .cast::<f32>()
73        .ok_or(AtmosphereMetadataError::FloatConversion)?;
74        let radius = camera.globe_radius_pixels() as f32;
75        if !radius.is_finite() || radius <= 0.0 || !blend.is_finite() {
76            return Err(AtmosphereMetadataError::InvalidScalar { radius, blend });
77        }
78        Ok(Self {
79            inverse_projection: inverse_projection.into(),
80            globe_position: [globe_position.x, globe_position.y, globe_position.z, 0.0],
81            sun_direction: [
82                sun_direction.x,
83                sun_direction.y,
84                sun_direction.z,
85                sky.sun_visible(),
86            ],
87            radius_blend_padding: [
88                radius,
89                blend.clamp(0.0, 1.0),
90                camera.body().radius_meters as f32,
91                0.0,
92            ],
93        })
94    }
95}

97/// Invalid CPU inputs for the atmosphere shader.
98#[derive(Clone, Copy, Debug, Error, PartialEq)]
99pub enum AtmosphereMetadataError {
100    /// The root light cannot be evaluated for this view.
101    #[error("failed to evaluate atmosphere light")]
102    Light {
103        /// Underlying light error.
104        #[source]
105        source: LightError,
106    },
107    /// Camera values cannot be represented by the shader's f32 interface.
108    #[error("atmosphere camera data cannot be represented as f32")]
109    FloatConversion,
110    /// Radius and blend must be finite and radius must be positive.
111    #[error("invalid atmosphere radius {radius} or blend {blend}")]
112    InvalidScalar {
113        /// Globe radius in view-space pixels.
114        radius: f32,
115        /// Evaluated atmosphere blend.
116        blend: f32,
117    },
118}

120/// Per-frame values of the sky draw: colours, the horizon line on screen and the blends.
121///
122/// A negative `blend[2]` selects the globe sky, shaded by each pixel's elevation angle above
123/// the eye's sea-level horizon instead of its distance from a screen-space horizon line.
124#[repr(C)]
125#[derive(Copy, Clone, Debug, Pod, Zeroable)]
126pub struct SkyLayerMetadata {
127    /// Premultiplied sky colour.
128    pub sky_color: [f32; 4],
129    /// Premultiplied horizon colour.
130    pub horizon_color: [f32; 4],
131    /// Horizon point in screen pixels with y up, then the unit normal pointing into the sky.
132    pub horizon: [f32; 4],
133    /// Blend width (pixels, or radians on the globe sky), globe transition, viewport height
134    /// (negative for the globe sky), and reversed sky depth.
135    pub blend: [f32; 4],
136    /// Inverse OpenGL-convention projection turning a clip position into a view-space ray.
137    pub inverse_projection: Mat4x4f32,
138    /// The eye's local zenith in view axes, then the elevation of its sea-level horizon in
139    /// radians (zero or negative).
140    pub up_and_horizon: [f32; 4],
141}
142
143impl SkyLayerMetadata {
144    /// Shader location of [`Self::up_and_horizon`]; the sky's other attributes take 8..=15.
145    const UP_AND_HORIZON_LOCATION: u32 = 7;
146}

148/// Shader filling the screen above the horizon with the sky, as GL JS `drawSky` does.
149pub struct SkyShader {
150    /// Render-target format.
151    pub format: wgpu::TextureFormat,
152}
153
154impl Shader for SkyShader {
155    fn describe_vertex(&self) -> VertexState {
156        VertexState {
157            source: include_str!(concat!(
158                env!("OUT_DIR"),
159                "/wgsl/render/shaders/sky.vertex.wgsl"
160            )),
161            entry_point: "main",
162            buffers: vec![VertexBufferLayout {
163                array_stride: std::mem::size_of::<SkyLayerMetadata>() as u64,
164                step_mode: wgpu::VertexStepMode::Instance,
165                attributes: (0..9)
166                    .map(|index| wgpu::VertexAttribute {
167                        offset: index * wgpu::VertexFormat::Float32x4.size(),
168                        format: wgpu::VertexFormat::Float32x4,
169                        shader_location: if index < 8 {
170                            8 + index as u32
171                        } else {
172                            SkyLayerMetadata::UP_AND_HORIZON_LOCATION
173                        },
174                    })
175                    .collect(),
176            }],
177        }
178    }
179
180    fn describe_fragment(&self) -> FragmentState {
181        FragmentState {
182            source: include_str!(concat!(
183                env!("OUT_DIR"),
184                "/wgsl/render/shaders/sky.fragment.wgsl"
185            )),
186            entry_point: "main",
187            targets: vec![Some(wgpu::ColorTargetState {
188                format: self.format,
189                blend: Some(wgpu::BlendState::PREMULTIPLIED_ALPHA_BLENDING),
190                write_mask: wgpu::ColorWrites::ALL,
191            })],
192        }
193    }
194}

196/// Shader drawing a translucent atmospheric shell.
197pub struct AtmosphereShader {
198    /// Render-target format.
199    pub format: wgpu::TextureFormat,
200}
201
202impl Shader for AtmosphereShader {
203    fn describe_vertex(&self) -> VertexState {
204        VertexState {
205            source: include_str!(concat!(
206                env!("OUT_DIR"),
207                "/wgsl/render/shaders/atmosphere.vertex.wgsl"
208            )),
209            entry_point: "main",
210            buffers: atmosphere_buffers(),
211        }
212    }
213
214    fn describe_fragment(&self) -> FragmentState {
215        FragmentState {
216            source: concat!(
217                include_str!(concat!(
218                    env!("OUT_DIR"),
219                    "/wgsl/render/shaders/atmosphere_scattering.wgsl.part"
220                )),
221                include_str!(concat!(
222                    env!("OUT_DIR"),
223                    "/wgsl/render/shaders/atmosphere.fragment.wgsl"
224                ))
225            ),
226            entry_point: "main",
227            targets: vec![Some(wgpu::ColorTargetState {
228                format: self.format,
229                // The scattering shader emits premultiplied RGB, matching GL JS's
230                // ColorMode.alphaBlended (`ONE`, `ONE_MINUS_SRC_ALPHA`).
231                blend: Some(wgpu::BlendState::PREMULTIPLIED_ALPHA_BLENDING),
232                write_mask: wgpu::ColorWrites::ALL,
233            })],
234        }
235    }
236}
