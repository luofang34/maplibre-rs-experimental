1//! Compact 16-byte GPU vertex encoding (`GpuShaderVertex`) for vector tile pipelines.
2//!
3//! CPU tessellation and spatial-elevation systems operate on 32-byte `f32` [`ShaderVertex`]
4//! records, while [`BufferPool`](crate::vector::BufferPool) packs them into 16-byte
5//! [`GpuShaderVertex`] words on upload (`-50%` vertex VRAM and PCIe/WebGPU `writeBuffer` bytes).

14/// Compact 16-byte GPU vertex record stored in [`VectorBufferPool`](crate::vector::VectorBufferPool).
15/// Vertex encoding family matching WGSL vertex buffer decoding for vector geometry pipelines.
16#[derive(Clone, Copy, PartialEq, Eq)]
17pub(crate) enum GpuVertexKind {
18    /// Fill/polygon pipeline vertex encoding.
19    Fill,
20    /// Line/stroke pipeline vertex encoding.
21    Line,
22    /// Fill-extrusion pipeline vertex encoding.
23    Extrusion,
24}
25
26impl GpuVertexKind {
27    pub(crate) fn from_layer_type(layer_type: &str) -> Option<Self> {
28        match layer_type {
29            "line" => Some(Self::Line),
30            "fill-extrusion" => Some(Self::Extrusion),
31            "circle" | "heatmap" => None,
32            _ => Some(Self::Fill),
33        }
34    }
35
36    #[inline]
37    pub(crate) fn encode(self, v: &ShaderVertex) -> GpuShaderVertex {
38        match self {
39            Self::Fill => encode_fill_vertex(v),
40            Self::Line => encode_line_vertex(v),
41            Self::Extrusion => encode_extrusion_vertex(v),
42        }
43    }
44}

57/// Maps a CPU vertex type `Self` to its GPU-resident storage representation `Self::Gpu`.
58pub trait PoolVertex: PodTrait {
59    /// GPU vertex record uploaded to the vertex slab.
60    type Gpu: PodTrait + std::fmt::Debug;
61
62    /// Returns a compact tag identifying which GPU bit-layout `layer_type` produces, so shared
63    /// source-layer vertex slabs only alias across compatible shader families.
64    #[inline]
65    fn encoding_family(layer_type: &str) -> u8 {
66        let _ = layer_type;
67        0
68    }
69
70    /// Required alignment of `first_vertex` in elements (`4` for quad-indexed shaders that decode
71    /// corners via `@builtin(vertex_index) % 4u`, `1` otherwise).
72    #[inline]
73    fn vertex_element_alignment(layer_type: &str) -> u64 {
74        let _ = layer_type;
75        1
76    }
77
78    /// Returns `true` when `src` fits losslessly in the compact 8-byte extrusion format.
79    #[inline]
80    fn can_encode_compact_extrusion(layer_type: &str, src: &[Self]) -> bool {
81        let _ = (layer_type, src);
82        false
83    }
84
85    /// Packs `src` into two 8-byte compact extrusion vertices per `Self::Gpu` word when eligible.
86    #[inline]
87    fn try_encode_compact_extrusion<'a>(
88        layer_type: &str,
89        src: &'a [Self],
90        scratch: &'a mut Vec<Self::Gpu>,
91    ) -> Option<&'a [Self::Gpu]> {
92        let _ = (layer_type, src, scratch);
93        None
94    }
95
96    /// Returns `src` in `Self::Gpu` form, either by zero-copy slice borrow or by packing into `scratch`.
97    fn encode_vertices<'a>(
98        layer_type: &str,
99        src: &'a [Self],
100        scratch: &'a mut Vec<Self::Gpu>,
101    ) -> &'a [Self::Gpu];
102}

104impl PoolVertex for ShaderSymbolVertex {
105    type Gpu = Self;

107    #[inline]
108    fn encode_vertices<'a>(
109        _layer_type: &str,
110        src: &'a [Self],
111        _scratch: &'a mut Vec<Self::Gpu>,
112    ) -> &'a [Self::Gpu] {
113        src
114    }
115}

117impl PoolVertex for ShaderVertex {
118    type Gpu = GpuShaderVertex;
119
120    #[inline]
121    fn encoding_family(layer_type: &str) -> u8 {
122        match layer_type {
123            "line" => 1,
124            "fill-extrusion" => 2,
125            "circle" | "heatmap" => 3,
126            _ => 0,
127        }
128    }
129
130    #[inline]
131    fn vertex_element_alignment(layer_type: &str) -> u64 {
132        match layer_type {
133            "circle" | "heatmap" => 4,
134            _ => 1,
135        }
136    }
137
138    fn encode_vertices<'a>(
139        layer_type: &str,
140        src: &'a [Self],
141        scratch: &'a mut Vec<GpuShaderVertex>,
142    ) -> &'a [GpuShaderVertex] {
143        scratch.clear();
144        scratch.reserve(src.len());
145        match Self::encoding_family(layer_type) {
146            1 => scratch.extend(src.iter().map(encode_line_vertex)),
147            2 => scratch.extend(src.iter().map(encode_extrusion_vertex)),
148            3 => scratch.extend(src.iter().map(encode_point_vertex)),
149            _ => scratch.extend(src.iter().map(encode_fill_vertex)),
150        }
151        scratch.as_slice()
152    }
153
154    #[inline]
155    fn can_encode_compact_extrusion(layer_type: &str, src: &[Self]) -> bool {
156        layer_type == "fill-extrusion" && can_encode_compact_extrusion(src)
157    }
158
159    #[inline]
160    fn try_encode_compact_extrusion<'a>(
161        layer_type: &str,
162        src: &'a [Self],
163        scratch: &'a mut Vec<GpuShaderVertex>,
164    ) -> Option<&'a [GpuShaderVertex]> {
165        if layer_type != "fill-extrusion" || !can_encode_compact_extrusion(src) {
166            return None;
167        }
168        encode_compact_extrusion_vertices(src, scratch);
169        Some(scratch.as_slice())
170    }
171}

179        v.normal[1].to_bits(),
180    ])

183#[inline]
184fn pack_tile_pos_q2(pos: [f32; 2]) -> u32 {
185    let q = |c: f32| ((c - 2048.0) * 4.0).round().clamp(-32768.0, 32767.0) as i16 as u16 as u32;
186    q(pos[0]) | (q(pos[1]) << 16)
187}

189#[inline]
190fn encode_line_vertex(v: &ShaderVertex) -> GpuShaderVertex {
191    let w0 = pack_tile_pos_q2(v.position);
192    let w1 = pack_f16_pair(v.normal[0], v.normal[1]);
193    let dist_i24 = (v.distance * 8.0).round().clamp(-8_388_608.0, 8_388_607.0) as i32;
194    let dist_u24 = (dist_i24 as u32) & 0x00ff_ffff;
195    let spatial = v.elevation > -1e20;
196    let style_bits = v.edge_distance.to_bits() & 0x7f_ffff;
197    let per_feature = style_bits != 0;
198    let width_q4 = ((style_bits >> 11) & 0xfff).min(0x7ff);
199    let offset_q3 = (((style_bits & 0x7ff) as i32 - 1024 + 512).clamp(0, 1023)) as u32;
200
201    if !spatial {
202        let stroke_bits = v.elevation.to_bits() & 0x7f_ffff;
203        let neg_side = ((stroke_bits >> 22) & 1) != 0;
204        let gap_raw = (stroke_bits >> 11) & 0x7ff;
205        let blur_raw = stroke_bits & 0x7ff;
206        let gap_u8 = if gap_raw == 0x7ff {
207            0xff
208        } else {
209            gap_raw.min(0xfe)
210        };
211        let blur_u8 = if blur_raw == 0x7ff {
212            0xff
213        } else {
214            blur_raw.min(0xfe)
215        };
216        let w2 = dist_u24 | (blur_u8 << 24);
217        let w3 = gap_u8
218            | (offset_q3 << 8)
219            | (width_q4 << 18)
220            | (u32::from(per_feature) << 29)
221            | (u32::from(neg_side) << 30);
222        GpuShaderVertex([w0, w1, w2, w3])
223    } else {
224        let neg_side = (v.corner_code - v.corner_code.floor()) > 0.25;
225        let elev_u16 = ((v.elevation + 1000.0).clamp(0.0, 16383.75) * 4.0).round() as u32 & 0xffff;
226        let w2 = dist_u24 | (((elev_u16 >> 8) & 0xff) << 24);
227        let w3 = (elev_u16 & 0xff)
228            | (offset_q3 << 8)
229            | (width_q4 << 18)
230            | (u32::from(per_feature) << 29)
231            | (u32::from(neg_side) << 30)
232            | (1_u32 << 31);
233        GpuShaderVertex([w0, w1, w2, w3])
234    }
235}

237#[inline]
238fn encode_extrusion_vertex(v: &ShaderVertex) -> GpuShaderVertex {
239    let w0 = pack_tile_pos_q2(v.position);
240    let len = v.normal[0].hypot(v.normal[1]);
241    let kind = if len <= 0.1 {
242        0_u32
243    } else if len <= 0.5 {
244        1_u32
245    } else if len <= 1.5 {
246        2_u32
247    } else {
248        3_u32
249    };
250    let angle_u14 = if kind >= 2 {
251        (v.normal[1].atan2(v.normal[0]).rem_euclid(TAU) * (16384.0 / TAU)).round() as u32 & 0x3fff
252    } else {
253        0
254    };
255    let datum_code = (v.corner_code / 1_048_576.0).floor();
256    let zoom_code = v.corner_code - datum_code * 1_048_576.0;
257    let overscale_f = (zoom_code / 65536.0).floor();
258    let overscale = overscale_f.clamp(0.0, 7.0) as u32;
259    let rise = zoom_code - overscale_f * 65536.0 - 32768.0;
260    let edge_u13 = (v.edge_distance.clamp(0.0, 16382.0) * 0.5).round() as u32 & 0x1fff;
261    let w1 = kind | (angle_u14 << 2) | (overscale << 16) | (edge_u13 << 19);
262
263    let base_i24 = (v.distance * 16.0).round().clamp(-8_388_608.0, 8_388_607.0) as i32;
264    let height_i24 = (v.elevation * 16.0)
265        .round()
266        .clamp(-8_388_608.0, 8_388_607.0) as i32;
267    let rise_f16 = u32::from(f32_to_f16_bits(rise));
268    let w2 = ((base_i24 as u32) & 0x00ff_ffff) | ((rise_f16 & 0xff) << 24);
269    let w3 = ((height_i24 as u32) & 0x00ff_ffff) | (((rise_f16 >> 8) & 0xff) << 24);
270    GpuShaderVertex([w0, w1, w2, w3])
271}

273#[inline]
274fn extrusion_kind_and_angle(normal: [f32; 2]) -> (u32, f32) {
275    let len = normal[0].hypot(normal[1]);
276    let kind = if len <= 0.1 {
277        0_u32
278    } else if len <= 0.5 {
279        1_u32
280    } else if len <= 1.5 {
281        2_u32
282    } else {
283        3_u32
284    };
285    let angle = if kind >= 2 {
286        normal[1].atan2(normal[0]).rem_euclid(TAU)
287    } else {
288        0.0
289    };
290    (kind, angle)
291}

293#[inline]
294fn extrusion_zoom_params(corner_code: f32) -> (f32, f32, f32) {
295    let datum_code = (corner_code / 1_048_576.0).floor();
296    let zoom_code = corner_code - datum_code * 1_048_576.0;
297    let overscale_f = (zoom_code / 65536.0).floor();
298    let rise = zoom_code - overscale_f * 65536.0 - 32768.0;
299    (datum_code, overscale_f, rise)
300}

302/// Returns `true` when `v` fits in the compact 8-byte `fill-extrusion` vertex format.
303#[inline]
304pub(crate) fn is_compact_extrusion_vertex(v: &ShaderVertex) -> bool {
305    let (datum_code, _overscale_f, rise) = extrusion_zoom_params(v.corner_code);
306    v.edge_distance.abs() <= 1e-3
307        && datum_code.abs() <= 1e-3
308        && rise.abs() <= 1e-3
309        && (0.0..=255.5).contains(&v.distance)
310        && (0.0..=1023.5).contains(&v.elevation)
311}
312
313/// Returns `true` when every vertex in `src` fits in the compact 8-byte `fill-extrusion`
314/// vertex format (base in `0..=255.5` m, height in `0..=1023.5` m, zero `edge_distance`,
315/// default ground-datum `0`, and zero zoom-interpolated `rise`).
316pub fn can_encode_compact_extrusion(src: &[ShaderVertex]) -> bool {
317    !src.is_empty() && src.iter().all(is_compact_extrusion_vertex)
318}

320#[inline]
321pub(crate) fn encode_compact_extrusion_words(v: &ShaderVertex) -> [u32; 2] {
322    let w0 = pack_tile_pos_q2(v.position);
323    let (kind, angle) = extrusion_kind_and_angle(v.normal);
324    let angle_u10 = if kind >= 2 {
325        (angle * (1024.0 / TAU)).round() as u32 & 0x3ff
326    } else {
327        0
328    };
329    let base_u9 = (v.distance * 2.0).round().clamp(0.0, 511.0) as u32;
330    let height_u11 = (v.elevation * 2.0).round().clamp(0.0, 2047.0) as u32;
331    let w1 = kind | (angle_u10 << 2) | (base_u9 << 12) | (height_u11 << 21);
332    [w0, w1]
333}

335/// Expands a single 8-byte compact extrusion record into a standard 16-byte [`GpuShaderVertex`].
336#[inline]
337pub(crate) fn unpack_compact_extrusion_word([w0, compact_w1]: [u32; 2]) -> GpuShaderVertex {
338    let rise_f16 = u32::from(f32_to_f16_bits(0.0));
339    let kind = compact_w1 & 3;
340    let angle_u14 = ((compact_w1 >> 2) & 0x3ff) << 4;
341    let base_i24 = (((compact_w1 >> 12) & 0x1ff) * 8) as i32;
342    let height_i24 = ((compact_w1 >> 21) * 8) as i32;
343    let w1 = kind | (angle_u14 << 2);
344    let w2 = ((base_i24 as u32) & 0x00ff_ffff) | ((rise_f16 & 0xff) << 24);
345    let w3 = ((height_i24 as u32) & 0x00ff_ffff) | (((rise_f16 >> 8) & 0xff) << 24);
346    GpuShaderVertex([w0, w1, w2, w3])
347}

349/// Packs two 8-byte compact extrusion vertices per 16-byte [`GpuShaderVertex`] word.
350pub fn encode_compact_extrusion_vertices(src: &[ShaderVertex], out: &mut Vec<GpuShaderVertex>) {
351    out.clear();
352    out.reserve(src.len().div_ceil(2));
353    for chunk in src.chunks(2) {
354        let a = encode_compact_extrusion_words(&chunk[0]);
355        let b = chunk.get(1).map_or([0, 0], encode_compact_extrusion_words);
356        out.push(GpuShaderVertex([a[0], a[1], b[0], b[1]]));
357    }
358}

360/// Expands `logical_count` compact 8-byte extrusion vertices back into standard 16-byte
361/// [`GpuShaderVertex`] records for fallback non-compact pipelines.
362pub fn unpack_compact_extrusion_vertices(
363    packed: &[GpuShaderVertex],
364    logical_count: usize,
365    out: &mut Vec<GpuShaderVertex>,
366) {
367    out.clear();
368    out.reserve(logical_count);
369    let mut emitted = 0_usize;
370    for word in packed {
371        for half in [[word.0[0], word.0[1]], [word.0[2], word.0[3]]] {
372            if emitted >= logical_count {
373                return;
374            }
375            out.push(unpack_compact_extrusion_word(half));
376            emitted += 1;
377        }
378    }
379}

381#[inline]
382fn encode_point_vertex(v: &ShaderVertex) -> GpuShaderVertex {
383    let w0 = pack_tile_pos_q2(v.position);
384    let w1 = pack_f16_pair(v.normal[0], v.normal[1]);
385    let w2 = pack_f16_pair(v.distance, v.edge_distance);
386    let elev_f16 = if v.elevation > -1e20 {
387        f32_to_f16_bits(v.elevation.clamp(-30000.0, 65504.0))
388    } else {
389        f32_to_f16_bits(-65504.0)
390    };
391    let corner_floor = v.corner_code.floor();
392    let cell = corner_floor.clamp(0.0, 64.0) as u32;
393    let frac = v.corner_code - corner_floor;
394    let has_stroke_op = frac > 0.125;
395    let stroke_op_u8 = if has_stroke_op {
396        (((frac - 0.25) * 2.0).clamp(0.0, 1.0) * 255.0).round() as u32
397    } else {
398        0
399    };
400    let w3 = u32::from(elev_f16)
401        | ((cell | (u32::from(has_stroke_op) << 7)) << 16)
402        | (stroke_op_u8 << 24);
403    GpuShaderVertex([w0, w1, w2, w3])
404}

406/// Packs two finite `f32` values into IEEE-754 binary16 lanes matching WGSL `unpack2x16float`.
407#[inline]
408pub fn pack_f16_pair(low: f32, high: f32) -> u32 {
409    u32::from(f32_to_f16_bits(low)) | (u32::from(f32_to_f16_bits(high)) << 16)
410}

412/// Converts a finite `f32` (clamped to `[-65504.0, 65504.0]`) into IEEE-754 `binary16` bits.
413#[inline]
414pub fn f32_to_f16_bits(val: f32) -> u16 {
415    if !val.is_finite() {
416        return if val.is_nan() {
417            0
418        } else if val.is_sign_negative() {
419            0xfbff
420        } else {
421            0x7bff
422        };
423    }
424    let clamped = val.clamp(-65504.0, 65504.0);
425    let bits = clamped.to_bits();
426    let sign = ((bits >> 16) & 0x8000) as u16;
427    let abs_bits = bits & 0x7fff_ffff;
428    if abs_bits < 0x3300_0000 {
429        return sign;
430    }
431    if abs_bits < 0x3880_0000 {
432        // Subnormal f16
433        let exp = (abs_bits >> 23) as i32 - 127;
434        let mant = (abs_bits & 0x7f_ffff) | 0x80_0000;
435        let shift = (14 - exp) as u32 + 13;
436        let rounded = (mant + (1 << (shift - 1)) - 1 + ((mant >> shift) & 1)) >> shift;
437        return sign | (rounded as u16);
438    }
439    let exp16 = (((abs_bits >> 23) as i32 - 127 + 15) as u32) << 10;
440    let mant = abs_bits & 0x7f_ffff;
441    let round_add = 0x0fff + ((mant >> 13) & 1);
442    let half = exp16 + ((mant + round_add) >> 13);
443    sign | (half.min(0x7bff) as u16)
444}

446/// Converts IEEE-754 `binary16` bits back to `f32`.
447#[inline]
448pub fn f16_bits_to_f32(bits: u16) -> f32 {
449    let sign = u32::from(bits & 0x8000) << 16;
450    let exp = u32::from((bits >> 10) & 0x1f);
451    let mant = u32::from(bits & 0x03ff);
452    let f32_bits = if exp == 0 {
453        if mant == 0 {
454            sign
455        } else {
456            let mut m = mant;
457            let mut e = 127_u32 - 14;
458            while (m & 0x0400) == 0 {
459                m <<= 1;
460                e -= 1;
461            }
462            sign | (e << 23) | ((m & 0x03ff) << 13)
463        }
464    } else if exp == 0x1f {
465        sign | 0x7f80_0000 | (mant << 13)
466    } else {
467        sign | ((exp + 127 - 15) << 23) | (mant << 13)
468    };
469    f32::from_bits(f32_bits)
470}
