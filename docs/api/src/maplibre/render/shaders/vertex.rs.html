7/// Tile-space geometry shared by fills, lines and circles.
8#[repr(C)]
9#[derive(Copy, Clone, Pod, Zeroable)]
10pub struct ShaderVertex {
11    /// Tile-local position in the renderer's 4096-unit grid.
12    pub position: Vec2f32,
13    /// Extrusion direction for polygon/line vertices; circle vertices store radius and stroke width in pixels.
14    pub normal: Vec2f32,
15    /// Distance along a stroked path, in tile units.
16    pub distance: f32,
17    /// Height above the body datum for spatial lines; a float below -1e20 keeps cartographic
18    /// draping and then carries a stroke's side, gap width and blur.
19    pub elevation: f32,
20    /// Distance along the ring of an extruded wall, in tile units; a stroke packs its
21    /// per-feature width and offset here instead. Zero elsewhere.
22    pub edge_distance: f32,
23    /// For a vertex of a subdivided circle or heatmap kernel, one plus the packed position of
24    /// its corner on an eight-by-eight grid; zero where the shader decodes the corner from the
25    /// vertex index.
26    pub corner_code: f32,
27}
28
29impl ShaderVertex {
30    /// Creates a vertex with zero path distance and the sentinel for cartographic draping.
31    pub fn new(position: Vec2f32, normal: Vec2f32) -> Self {
32        Self {
33            position,
34            normal,
35            distance: 0.0,
36            elevation: -1e30,
37            edge_distance: 0.0,
38            corner_code: 0.0,
39        }
40    }
41}
42
43impl Default for ShaderVertex {
44    fn default() -> Self {
45        ShaderVertex::new([0.0, 0.0], [0.0, 0.0])
46    }
47}

50#[repr(C)]
51#[derive(Debug, Copy, Clone, Pod, Zeroable)]
52pub struct FillShaderFeatureMetadata {
53    /// Encoded-sRGB color with straight alpha; feature opacity is folded into alpha.

57/// What the fourth lane of [`SDFShaderFeatureMetadata::pose`] holds for a glyph with a pose,
58/// before the glyph's rise in metres is added; a vertex without a pose holds 0. At this
59/// magnitude an `f32` keeps any rise on Earth to within a centimetre.
60pub const POSED_GLYPH: f32 = 65536.0;
61
62/// Per-vertex collision visibility and sampled terrain height for symbols.
63#[repr(C)]
64#[derive(Debug, Copy, Clone, Pod, Zeroable, Default)]
65pub struct SDFShaderFeatureMetadata {
66    /// Collision and fade opacity in 0..=1, repeated for the symbol's vertices.
67    pub opacity: f32,
68    /// Sampled ground elevation in meters; symbol height offsets are carried in the geometry.
69    pub elevation: f32,
70    /// Where a glyph placed along a line sits: the offset of its centre from the vertex anchor
71    /// in tile units, its direction in radians, and [`POSED_GLYPH`] plus how far the ground
72    /// under the glyph rises above the ground under the label's anchor, in metres, when the
73    /// glyph has such a pose.
74    pub pose: [f32; 4],
75    /// Fill colour of the symbol's text or icon, straight alpha, evaluated for its feature.
76    pub color: [f32; 4],
77    /// Halo colour of the text or icon.
78    pub halo: [f32; 4],
79    /// Size, halo width, halo blur and opacity of the text or icon.
80    pub params: [f32; 4],
81}

83/// Per-layer instance record shared across tile pipelines.
84#[repr(C)]
85#[derive(Copy, Clone, Pod, Zeroable)]
86pub struct ShaderLayerMetadata {
87    /// Style painter-order index carried by the shared instance layout.
88    pub z_index: f32,
89    /// Evaluated line width in style pixels, before per-tile drape scaling.
90    pub line_width: f32,
91    /// Layer translation converted to tile units in map axes.
92    pub translate: Vec2f32,
93    /// Circle stroke colour as straight RGBA; other layers leave it black.
94    pub stroke_color: Vec4f32,
95    /// Circle opacity, stroke opacity and blur ratio; the last slot is padding.
96    pub circle_params: Vec4f32,
97    /// Circle pitch-scale (x) and pitch-alignment (y), 1.0 for `map`; the rest is padding.
98    pub circle_flags: Vec4f32,
99}
100
101impl ShaderLayerMetadata {
102    /// Metadata for a layer without circle paint.
103    pub fn new(z_index: f32, line_width: f32, translate: Vec2f32) -> Self {
104        Self {
105            z_index,
106            line_width,
107            translate,
108            stroke_color: [0.0, 0.0, 0.0, 1.0],
109            circle_params: [1.0, 1.0, 0.0, 0.0],
110            circle_flags: [0.0; 4],
111        }
112    }
113}

115/// Per-tile instance record with projection, viewport and line scaling inputs.
116#[repr(C)]
117#[derive(Copy, Clone, Pod, Zeroable)]
118pub struct ShaderTileMetadata {
119    /// Column-major tile-to-clip transform used by the flat projection path.
120    pub transform: Mat4x4f32,
121    /// Scale from view zoom to tile zoom, normally `2^(tile_zoom - view_zoom)`.
122    pub zoom_factor: f32,
123    /// Render viewport width in pixels.
124    pub viewport_width: f32,
125    /// Render viewport height in pixels.
126    pub viewport_height: f32,
127    /// Mercator world origin in XY and per-tile-unit scales in ZW.
128    pub tile_mercator_coords: Vec4f32,
129    /// One clips geometry outside the root tile at the antimeridian; zero disables clipping.
130    pub clip_antimeridian: u32,
131    /// Converts style pixels to pixels in an offscreen drape texture.
132    pub line_width_scale: f32,
133    /// Tile units per texture or viewport pixel for line dash lengths.
134    pub line_units_per_pixel: f32,
135}
136
137impl ShaderTileMetadata {
138    /// Creates root-tile metadata for a 512-pixel viewport without antimeridian clipping.
139    /// Callers drawing another tile or viewport must replace those fields before upload.
140    pub fn new(transform: Mat4x4f32, zoom_factor: f32) -> Self {
141        Self {
142            transform,
143            zoom_factor,
144            viewport_width: 512.0,
145            viewport_height: 512.0,
146            tile_mercator_coords: [0.0, 0.0, 1.0 / 4096.0, 1.0 / 4096.0],
147            clip_antimeridian: 0,
148            line_width_scale: 1.0,
149            line_units_per_pixel: 8.0 * zoom_factor,
150        }
151    }
152}
