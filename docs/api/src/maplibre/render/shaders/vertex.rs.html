8/// Tile-space geometry shared by fills, lines and circles.
9#[repr(C)]
10#[derive(Copy, Clone, Pod, Zeroable)]
11pub struct ShaderVertex {
12    /// Tile-local position in the renderer's 4096-unit grid.
13    pub position: Vec2f32,
14    /// Extrusion direction for polygon/line vertices; circle vertices store radius and stroke width in pixels.
15    pub normal: Vec2f32,
16    /// Distance along a stroked path, in tile units.
17    pub distance: f32,
18    /// Height above the body datum for spatial lines, and for circles and heatmap points the
19    /// ground under them; a float below -1e20 keeps cartographic draping and then carries a
20    /// stroke's side, gap width and blur.
21    pub elevation: f32,
22    /// Distance along the ring of an extruded wall, in tile units; a stroke packs its
23    /// per-feature width and offset here instead. Zero elsewhere.
24    pub edge_distance: f32,
25    /// For a vertex of a subdivided circle or heatmap kernel, one plus the packed position of
26    /// its corner on an eight-by-eight grid; zero where the shader decodes the corner from the
27    /// vertex index. A circle's fractional part carries a per-feature stroke opacity.
28    pub corner_code: f32,
29}
30
31impl ShaderVertex {
32    /// Creates a vertex with zero path distance and the sentinel for cartographic draping.
33    pub fn new(position: Vec2f32, normal: Vec2f32) -> Self {
34        Self {
35            position,
36            normal,
37            distance: 0.0,
38            elevation: -1e30,
39            edge_distance: 0.0,
40            corner_code: 0.0,
41        }
42    }
43}
44
45impl Default for ShaderVertex {
46    fn default() -> Self {
47        ShaderVertex::new([0.0, 0.0], [0.0, 0.0])
48    }
49}

51/// Per-vertex feature color (`Unorm8x4`, 4 bytes) for vector geometry.
52#[repr(C)]
53#[derive(Debug, Copy, Clone, Default, PartialEq, Eq, Pod, Zeroable)]
54pub struct FillShaderFeatureMetadata {
55    /// Quantized RGBA8 color (or 24-bit fixed-point `line-gradient` length + opacity in `a`).
56    pub color: [u8; 4],
57}
58
59impl FillShaderFeatureMetadata {
60    /// Quantizes floating-point `[r, g, b, a]` in `0.0..=1.0` into `Unorm8x4`.
61    #[inline]
62    pub fn from_rgba(color: [f32; 4]) -> Self {
63        Self {
64            color: color.map(|c| (c.clamp(0.0, 1.0) * 255.0).round() as u8),
65        }
66    }
67
68    /// Packs a `line-gradient` feature length in `1/16` tile units (`24-bit` unsigned fixed-point
69    /// in `rgb`, up to `1,048,575.9375` tile units) and straight opacity in `a`.
70    #[inline]
71    pub fn from_line_gradient(length: f32, opacity: f32) -> Self {
72        let len_q4 = (length.max(0.0) * 16.0).round().min(16_777_215.0) as u32;
73        Self {
74            color: [
75                (len_q4 & 0xFF) as u8,
76                ((len_q4 >> 8) & 0xFF) as u8,
77                ((len_q4 >> 16) & 0xFF) as u8,
78                (opacity.clamp(0.0, 1.0) * 255.0).round() as u8,
79            ],
80        }
81    }
82
83    /// Unpacks a `pack_circle_colors` negative-lane `[f32; 4]` pair into `[fill, stroke]`.
84    #[inline]
85    pub fn unpack_circle_pair(packed: [f32; 4]) -> [Self; 2] {
86        let pair = |lane: f32| -> [u8; 2] {
87            let v = (-lane - 1.0).max(0.0).round() as u32;
88            [((v >> 8) & 0xFF) as u8, (v & 0xFF) as u8]
89        };
90        let [fr, fg] = pair(packed[0]);
91        let [fb, fa] = pair(packed[1]);
92        let [sr, sg] = pair(packed[2]);
93        let [sb, sa] = pair(packed[3]);
94        [
95            Self {
96                color: [fr, fg, fb, fa],
97            },
98            Self {
99                color: [sr, sg, sb, sa],
100            },
101        ]
102    }
103}

105/// What the fourth lane of [`SDFShaderFeatureMetadata::decoded_pose`] holds for a glyph with a pose,
106/// before the glyph's rise in metres is added; a vertex without a pose holds 0. At this
107/// magnitude an `f32` keeps any rise on Earth to within a centimetre.
108pub const POSED_GLYPH: f32 = 65536.0;

110/// Per-vertex collision visibility, GPU fade ramp, and sampled terrain height for symbols (`32 B`).
111///
112/// Field offsets (`16 B`, `8 B`, `4 B`, `4 B`) preserve natural hardware alignment across
113/// WebGPU, Metal, and mobile Vulkan drivers while using only 4 vertex attribute locations.
114#[repr(C)]
115#[derive(Debug, Copy, Clone, PartialEq, Pod, Zeroable, Default)]
116pub struct SDFShaderFeatureMetadata {
117    /// Bit-packed symbol metrics, elevation, fade timestamp, and along-line angle/rise (`@location(15)`):
118    /// - `packed[0]`: `(elev_q4_i20 & 0x000f_ffff) | (halo_w_q4 << 20) | (has_pose << 28) | (start_hi3 << 29)`
119    /// - `packed[1]`: `size_f16 | (rise_f16 << 16)`
120    /// - `packed[2]`: `angle_u16 | (start_lo16 << 16)`
121    /// - `packed[3]`: `target_u8 | (start_u8 << 8) | (style_u8 << 16) | (halo_b_q4 << 24)`
122    pub packed: [u32; 4],
123    /// Full-precision `[shift_x, shift_y]` in layout px when `!has_pose`, or `[dx, dy]` in tile
124    /// units relative to the label's integer anchor when `has_pose` (`@location(12)`).
125    pub pose_xy: [f32; 2],
126    /// Fill colour of the symbol's text or icon (`Unorm8x4`, `@location(10)`).
127    pub color: [u8; 4],
128    /// Halo colour of the text or icon (`Unorm8x4`, `@location(13)`).
129    pub halo: [u8; 4],
130}
131
132impl SDFShaderFeatureMetadata {
133    /// Packs placement visibility, fade ramp, anchor shift, and paint style into 32 bytes.
134    #[inline]
135    #[allow(clippy::too_many_arguments)]
136    pub fn from_placement(
137        target_opacity: f32,
138        elevation: f32,
139        start_opacity: f32,
140        start_time_ms: f32,
141        shift: [f32; 2],
142        color: [f32; 4],
143        halo: [f32; 4],
144        params: [f32; 4],
145    ) -> Self {
146        let u8_norm = |v: f32| (v.clamp(0.0, 1.0) * 255.0).round() as u32;
147        let target_u8 = u8_norm(target_opacity);
148        let start_u8 = u8_norm(start_opacity);
149        let style_op_u8 = u8_norm(params[3]);
150        let size_f16 = u32::from(super::f32_to_f16_bits(params[0].max(0.0)));
151        let halo_w_q4 = (params[1].max(0.0) * 16.0).round().min(255.0) as u32;
152        let halo_b_q4 = (params[2].max(0.0) * 16.0).round().min(255.0) as u32;
153        let elev_i20 = (elevation * 16.0).round().clamp(-524_288.0, 524_287.0) as i32;
154        let elev_bits = (elev_i20 as u32) & 0x000f_ffff;
155        let start_ticks = (start_time_ms.max(0.0) * 0.25).round().min(524_287.0) as u32;
156        Self {
157            packed: [
158                elev_bits | (halo_w_q4 << 20) | (((start_ticks >> 16) & 0x7) << 29),
159                size_f16,
160                (start_ticks & 0xffff) << 16,
161                target_u8 | (start_u8 << 8) | (style_op_u8 << 16) | (halo_b_q4 << 24),
162            ],
163            pose_xy: shift,
164            color: color.map(|c| (c.clamp(0.0, 1.0) * 255.0).round() as u8),
165            halo: halo.map(|c| (c.clamp(0.0, 1.0) * 255.0).round() as u8),
166        }
167    }
168
169    /// Encodes a glyph's curved-line pose (`dx, dy` in tile units, `angle` in radians, `rise` in metres).
170    #[inline]
171    pub fn set_glyph_pose(&mut self, dx: f32, dy: f32, angle: f32, rise: f32) {
172        let tau = std::f32::consts::TAU;
173        let angle_u16 = if angle.is_finite() {
174            (angle.rem_euclid(tau) * (65536.0 / tau)).round() as u32 & 0xffff
175        } else {
176            0
177        };
178        let rise_f16 = u32::from(super::f32_to_f16_bits(rise));
179        self.packed[0] |= 1_u32 << 28;
180        self.packed[1] = (self.packed[1] & 0xffff) | (rise_f16 << 16);
181        self.packed[2] = (self.packed[2] & 0xffff_0000) | angle_u16;
182        self.pose_xy = [dx, dy];
183    }
184
185    /// Whether target collision opacity is greater than zero.
186    #[inline]
187    pub const fn has_target_opacity(&self) -> bool {
188        (self.packed[3] & 0xff) != 0
189    }
190
191    /// Whether starting fade opacity is greater than zero.
192    #[inline]
193    pub const fn has_start_opacity(&self) -> bool {
194        ((self.packed[3] >> 8) & 0xff) != 0
195    }
196
197    /// Decodes the quantized fade start timestamp in milliseconds.
198    #[inline]
199    pub fn start_time_ms(&self) -> f32 {
200        let ticks = (self.packed[2] >> 16) | (((self.packed[0] >> 29) & 0x7) << 16);
201        (ticks as f32) * 4.0
202    }
203
204    /// Sets quantized target collision opacity (`0.0..=1.0`), used by unit tests.
205    #[inline]
206    pub fn set_target_opacity(&mut self, opacity: f32) {
207        let u = (opacity.clamp(0.0, 1.0) * 255.0).round() as u32;
208        self.packed[3] = (self.packed[3] & !0xff) | u;
209    }
210
211    /// Whether this vertex carries an along-line glyph pose (`true`) rather than an anchor shift.
212    #[inline]
213    pub const fn has_pose(&self) -> bool {
214        ((self.packed[0] >> 28) & 1) != 0
215    }
216
217    /// Decodes `[dx, dy, angle, POSED_GLYPH + rise]` (or `[shift_x, shift_y, 0.0, 0.0]` when unposed).
218    #[inline]
219    pub fn decoded_pose(&self) -> [f32; 4] {
220        if self.has_pose() {
221            let rise = super::f16_bits_to_f32((self.packed[1] >> 16) as u16);
222            let angle = (self.packed[2] & 0xffff) as f32 * (std::f32::consts::TAU / 65536.0);
223            [self.pose_xy[0], self.pose_xy[1], angle, POSED_GLYPH + rise]
224        } else {
225            [self.pose_xy[0], self.pose_xy[1], 0.0, 0.0]
226        }
227    }
228}

265/// Per-tile instance record with projection, viewport and line scaling inputs.
266///
267/// Every `Vec4f32` field sits on a 16-byte boundary, `Float32x2` scalar pairs sit on 8-byte
268/// boundaries, and explicit tail padding rounds the record to 128 bytes so per-tile buffer slices
269/// preserve 16-byte vector attribute alignment on mobile Vulkan drivers.
270#[repr(C)]
271#[derive(Copy, Clone, Pod, Zeroable)]
272pub struct ShaderTileMetadata {
273    /// Column-major tile-to-clip transform used by the flat projection path.
274    pub transform: Mat4x4f32,
275    /// Mercator world origin in XY and per-tile-unit scales in ZW.
276    pub tile_mercator_coords: Vec4f32,
277    /// The part of the tile the shape draws, as minimum and maximum tile coordinates: a tile
278    /// drawn over a finer view tile covers only that tile's square of it. A line on the
279    /// terrain clips itself to it, where a flat shape takes it from the stencil mask.
280    pub clip_region: Vec4f32,
281    /// Render viewport width in pixels.
282    pub viewport_width: f32,
283    /// Render viewport height in pixels.
284    pub viewport_height: f32,
285    /// Converts style pixels to pixels in an offscreen drape texture.
286    pub line_width_scale: f32,
287    /// Tile units per texture or viewport pixel for line dash lengths.
288    pub line_units_per_pixel: f32,
289    /// Scale from view zoom to tile zoom, normally `2^(tile_zoom - view_zoom)`.
290    pub zoom_factor: f32,
291    /// One clips geometry outside the root tile at the antimeridian; zero disables clipping.
292    pub clip_antimeridian: u32,
293    /// Explicit tail padding keeping `size_of::<ShaderTileMetadata>() == 128`.
294    #[doc(hidden)]
295    pub _pad: [u32; 2],
296}
297
298/// The [`ShaderTileMetadata::clip_region`] of a shape that draws its whole tile.
299pub const WHOLE_TILE: Vec4f32 = [0.0, 0.0, 4096.0, 4096.0];
300
301impl ShaderTileMetadata {
302    /// Creates root-tile metadata for a 512-pixel viewport without antimeridian clipping.
303    /// Callers drawing another tile or viewport must replace those fields before upload.
304    pub fn new(transform: Mat4x4f32, zoom_factor: f32) -> Self {
305        Self {
306            transform,
307            tile_mercator_coords: [0.0, 0.0, 1.0 / 4096.0, 1.0 / 4096.0],
308            clip_region: WHOLE_TILE,
309            viewport_width: 512.0,
310            viewport_height: 512.0,
311            line_width_scale: 1.0,
312            line_units_per_pixel: 8.0 * zoom_factor,
313            zoom_factor,
314            clip_antimeridian: 0,
315            _pad: [0; 2],
316        }
317    }
318
319    /// Builds an instance-stepped [`VertexBufferLayout`] over `ShaderTileMetadata` with `attributes`.
320    pub fn instance_layout(attributes: Vec<wgpu::VertexAttribute>) -> VertexBufferLayout {
321        VertexBufferLayout {
322            array_stride: std::mem::size_of::<Self>() as u64,
323            step_mode: wgpu::VertexStepMode::Instance,
324            attributes,
325        }
326    }
327
328    /// Column attributes (`@location(4..=7)`) for [`Self::transform`].
329    pub fn transform_attributes() -> [wgpu::VertexAttribute; 4] {
330        let base = std::mem::offset_of!(Self, transform) as u64;
331        let col = wgpu::VertexFormat::Float32x4.size();
332        [
333            attribute(base, wgpu::VertexFormat::Float32x4, 4),
334            attribute(base + col, wgpu::VertexFormat::Float32x4, 5),
335            attribute(base + 2 * col, wgpu::VertexFormat::Float32x4, 6),
336            attribute(base + 3 * col, wgpu::VertexFormat::Float32x4, 7),
337        ]
338    }
339
340    /// Shared instance layout binding `transform` (`4..=7`), `tile_mercator_coords` (`2`), and
341    /// `zoom_factor` (`9`), used by fill, fill-extrusion, heatmap, and tile-mask shaders.
342    pub fn basic_layout() -> VertexBufferLayout {
343        let mut attrs = Self::transform_attributes().to_vec();
344        attrs.push(attribute(
345            std::mem::offset_of!(Self, zoom_factor) as u64,
346            wgpu::VertexFormat::Float32,
347            9,
348        ));
349        attrs.push(attribute(
350            std::mem::offset_of!(Self, tile_mercator_coords) as u64,
351            wgpu::VertexFormat::Float32x4,
352            2,
353        ));
354        Self::instance_layout(attrs)
355    }
