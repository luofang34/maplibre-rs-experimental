1//! Shader descriptors of the heatmap density pass and its colourising composite.
2
3use super::{Shader, ShaderLayerMetadata, ShaderTileMetadata};
4use crate::render::{
5    resource::{FragmentState, VertexBufferLayout, VertexState},
6    ShaderVertex,
7};
8
9/// Format of the density target. Half floats render and blend without device features.
10pub const DENSITY_FORMAT: wgpu::TextureFormat = wgpu::TextureFormat::R16Float;
11
12/// Draws each point as a Gaussian kernel, added into the density target.
13pub struct HeatmapDensityShader;
14
15impl Shader for HeatmapDensityShader {
16    fn describe_vertex(&self) -> VertexState {
17        VertexState {
18            source: concat!(
19                include_str!("projection.vertex.wgsl"),
20                include_str!("heatmap_density.vertex.wgsl")
21            ),
22            entry_point: "main",
23            buffers: vec![
24                VertexBufferLayout {
25                    array_stride: std::mem::size_of::<ShaderVertex>() as u64,
26                    step_mode: wgpu::VertexStepMode::Vertex,
27                    attributes: vec![
28                        wgpu::VertexAttribute {
29                            offset: 0,
30                            format: wgpu::VertexFormat::Float32x2,
31                            shader_location: 0,
32                        },
33                        wgpu::VertexAttribute {
34                            offset: std::mem::offset_of!(ShaderVertex, normal) as u64,
35                            format: wgpu::VertexFormat::Float32x4,
36                            shader_location: 1,
37                        },
38                        wgpu::VertexAttribute {
39                            offset: std::mem::offset_of!(ShaderVertex, corner_code) as u64,
40                            format: wgpu::VertexFormat::Float32,
41                            shader_location: 10,
42                        },
43                    ],
44                },
45                VertexBufferLayout {
46                    array_stride: std::mem::size_of::<ShaderTileMetadata>() as u64,
47                    step_mode: wgpu::VertexStepMode::Instance,
48                    // The viewport size sits between the zoom factor and the mercator extent, and
49                    // only listing it puts the extent at its offset.
50                    attributes: wgpu::vertex_attr_array![
51                        4 => Float32x4, 5 => Float32x4, 6 => Float32x4, 7 => Float32x4,
52                        9 => Float32, 11 => Float32, 12 => Float32, 2 => Float32x4
53                    ]
54                    .to_vec(),
55                },
56                VertexBufferLayout {
57                    array_stride: std::mem::size_of::<ShaderLayerMetadata>() as u64,
58                    step_mode: wgpu::VertexStepMode::Instance,
59                    attributes: vec![
60                        wgpu::VertexAttribute {
61                            offset: std::mem::offset_of!(ShaderLayerMetadata, line_width) as u64,
62                            format: wgpu::VertexFormat::Float32,
63                            shader_location: 13,
64                        },
65                        wgpu::VertexAttribute {
66                            offset: std::mem::offset_of!(ShaderLayerMetadata, circle_params) as u64,
67                            format: wgpu::VertexFormat::Float32x4,
68                            shader_location: 14,
69                        },
70                    ],
71                },
72            ],
73        }
74    }
75
76    fn describe_fragment(&self) -> FragmentState {
77        FragmentState {
78            source: include_str!("heatmap_density.fragment.wgsl"),
79            entry_point: "main",
80            targets: vec![Some(wgpu::ColorTargetState {
81                format: DENSITY_FORMAT,
82                // Overlapping kernels add up; that sum is the density.
83                blend: Some(wgpu::BlendState {
84                    color: wgpu::BlendComponent {
85                        src_factor: wgpu::BlendFactor::One,
86                        dst_factor: wgpu::BlendFactor::One,
87                        operation: wgpu::BlendOperation::Add,
88                    },
89                    alpha: wgpu::BlendComponent {
90                        src_factor: wgpu::BlendFactor::One,
91                        dst_factor: wgpu::BlendFactor::One,
92                        operation: wgpu::BlendOperation::Add,
93                    },
94                }),
95                write_mask: wgpu::ColorWrites::ALL,
96            })],
97        }
98    }
99}

101/// Looks each density up in the colour ramp and draws the result over the layers below.
102pub struct HeatmapCompositeShader {
103    /// Colour format of the render target the heatmap composites into.
104    pub format: wgpu::TextureFormat,
105}
106
107impl Shader for HeatmapCompositeShader {
108    fn describe_vertex(&self) -> VertexState {
109        VertexState {
110            source: include_str!("heatmap_composite.vertex.wgsl"),
111            entry_point: "main",
112            buffers: vec![],
113        }
114    }
115
116    fn describe_fragment(&self) -> FragmentState {
117        FragmentState {
118            source: include_str!("heatmap_composite.fragment.wgsl"),
119            entry_point: "main",
120            targets: vec![Some(wgpu::ColorTargetState {
121                format: self.format,
122                blend: Some(wgpu::BlendState::PREMULTIPLIED_ALPHA_BLENDING),
123                write_mask: wgpu::ColorWrites::ALL,
124            })],
125        }
126    }
127}
