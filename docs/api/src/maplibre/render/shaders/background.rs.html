13/// Background color, painter index and viewport horizon supplied per draw.
14#[repr(C)]
15#[derive(Copy, Clone, Debug, Pod, Zeroable)]
16pub struct BackgroundLayerMetadata {
17    /// Encoded-sRGB background color with straight alpha.
18    pub color: [f32; 4],
19    /// Style painter-order index carried by the instance layout.
20    pub z_index: f32,
21    /// Keeps the horizon sixteen-byte aligned.
22    pub padding: [f32; 3],
23    /// Horizon point in screen pixels with y up, then the unit normal pointing into the sky;
24    /// the flat map ends there, as the tiles GL JS draws the background on do.
25    pub horizon: [f32; 4],
26    /// Viewport height in X, terrain-enabled flag in Y, then the frame's width and height in
27    /// pixels.
28    pub viewport: [f32; 4],
29    /// Where the flat map ends north and south, relative to the view center: the row that
30    /// turns a pixel into its numerator of the map row under it with the northern edge's map
31    /// row behind it, then the denominator's row with the southern edge's.
32    pub world_rows: [[f32; 4]; 2],
33}

101/// Per-draw inputs for GL JS-compatible atmospheric scattering.
102#[repr(C)]
103#[derive(Copy, Clone, Debug, Pod, Zeroable)]
104pub struct AtmosphereLayerMetadata {
105    inverse_projection: Mat4x4f32,
106    globe_position: [f32; 4],
107    sun_direction: [f32; 4],
108    radius_blend_padding: [f32; 4],
109}
110
111impl AtmosphereLayerMetadata {
112    /// Creates inert metadata for frames where no atmosphere draw is queued.
113    pub fn disabled() -> Self {
114        Self {
115            inverse_projection: cgmath::Matrix4::from_scale(1.0).into(),
116            globe_position: [0.0; 4],
117            sun_direction: [0.0, 0.0, 1.0, 0.0],
118            radius_blend_padding: [1.0, 0.0, Body::EARTH.radius_meters as f32, 0.0],
119        }
120    }
121
122    /// Creates atmosphere inputs from the current globe camera and root light.
123    pub fn from_view(
124        camera: &GlobeCameraState,
125        light: &LightSpecification,
126        zoom: f64,
127        blend: f32,
128    ) -> Result<Self, AtmosphereMetadataError> {
129        let inverse_projection = camera
130            .inverse_projection()
131            .cast::<f32>()
132            .ok_or(AtmosphereMetadataError::FloatConversion)?;
133        let globe_position = camera
134            .globe_center_in_view()
135            .cast::<f32>()
136            .ok_or(AtmosphereMetadataError::FloatConversion)?;
137        let sun_direction = light
138            .sun_direction_in_view(camera, zoom)
139            .map_err(|source| AtmosphereMetadataError::Light { source })?
140            .cast::<f32>()
141            .ok_or(AtmosphereMetadataError::FloatConversion)?;
142        let radius = camera.globe_radius_pixels() as f32;
143        if !radius.is_finite() || radius <= 0.0 || !blend.is_finite() {
144            return Err(AtmosphereMetadataError::InvalidScalar { radius, blend });
145        }
146        Ok(Self {
147            inverse_projection: inverse_projection.into(),
148            globe_position: [globe_position.x, globe_position.y, globe_position.z, 0.0],
149            sun_direction: [sun_direction.x, sun_direction.y, sun_direction.z, 0.0],
150            radius_blend_padding: [
151                radius,
152                blend.clamp(0.0, 1.0),
153                camera.body().radius_meters as f32,
154                0.0,
155            ],
156        })
157    }
158}

160/// Invalid CPU inputs for the atmosphere shader.
161#[derive(Clone, Copy, Debug, Error, PartialEq)]
162pub enum AtmosphereMetadataError {
163    /// The root light cannot be evaluated for this view.
164    #[error("failed to evaluate atmosphere light")]
165    Light {
166        /// Underlying light error.
167        #[source]
168        source: LightError,
169    },
170    /// Camera values cannot be represented by the shader's f32 interface.
171    #[error("atmosphere camera data cannot be represented as f32")]
172    FloatConversion,
173    /// Radius and blend must be finite and radius must be positive.
174    #[error("invalid atmosphere radius {radius} or blend {blend}")]
175    InvalidScalar {
176        /// Globe radius in view-space pixels.
177        radius: f32,
178        /// Evaluated atmosphere blend.
179        blend: f32,
180    },
181}

197/// Shader filling the screen above the horizon with the sky, as GL JS `drawSky` does.
198pub struct SkyShader {
199    /// Render-target format.
200    pub format: wgpu::TextureFormat,
201}
202
203impl Shader for SkyShader {
204    fn describe_vertex(&self) -> VertexState {
205        VertexState {
206            source: include_str!("sky.vertex.wgsl"),
207            entry_point: "main",
208            buffers: vec![VertexBufferLayout {
209                array_stride: std::mem::size_of::<SkyLayerMetadata>() as u64,
210                step_mode: wgpu::VertexStepMode::Instance,
211                attributes: (0..4)
212                    .map(|index| wgpu::VertexAttribute {
213                        offset: index * wgpu::VertexFormat::Float32x4.size(),
214                        format: wgpu::VertexFormat::Float32x4,
215                        shader_location: 8 + index as u32,
216                    })
217                    .collect(),
218            }],
219        }
220    }
221
222    fn describe_fragment(&self) -> FragmentState {
223        FragmentState {
224            source: include_str!("sky.fragment.wgsl"),
225            entry_point: "main",
226            targets: vec![Some(wgpu::ColorTargetState {
227                format: self.format,
228                blend: Some(wgpu::BlendState::PREMULTIPLIED_ALPHA_BLENDING),
229                write_mask: wgpu::ColorWrites::ALL,
230            })],
231        }
232    }
233}
