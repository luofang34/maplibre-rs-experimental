1//! Deferred resource construction and replacement when resource criteria change.
2
3#![deny(missing_docs)]

5use std::mem;
6
7use crate::{coords::WorldTileCoords, render::tile_view_pattern::HasTile, tcs::world::World};
8
9/// Wrapper around a resource which can be initialized or uninitialized.
10/// Uninitialized resources can be constructed once with [`Eventually::initialize`].
11#[derive(Default)]
12pub enum Eventually<T> {
13    /// A resource is available for use and can be taken or replaced.
14    Initialized(T),
15    /// No resource is stored; initialization has not occurred or the value was taken.
16    #[default]
17    Uninitialized,
18}
19
20/// Determines whether a resource must be replaced to satisfy caller-supplied criteria.
21pub trait HasChanged {
22    /// Comparable criteria such as viewport dimensions or resource configuration.
23    type Criteria: Eq;
24
25    /// Whether the current resource no longer satisfies the supplied criteria.
26    fn has_changed(&self, criteria: &Self::Criteria) -> bool;
27}
28
29impl<T> HasChanged for Option<T>
30where
31    T: HasChanged,
32{
33    type Criteria = T::Criteria;
34
35    fn has_changed(&self, criteria: &Self::Criteria) -> bool {
36        match self {
37            None => true,
38            Some(value) => value.has_changed(criteria),
39        }
40    }
41}
42
43impl<T> Eventually<T>
44where
45    T: HasChanged,
46{
47    #[tracing::instrument(name = "reinitialize", skip_all)]
48    /// Constructs a replacement only when uninitialized or when the current resource has changed.
49    pub fn reinitialize(&mut self, f: impl FnOnce() -> T, criteria: &T::Criteria) {
50        let should_replace = match &self {
51            Eventually::Initialized(current) => current.has_changed(criteria),
52            Eventually::Uninitialized => true,
53        };
54
55        if should_replace {
56            *self = Eventually::Initialized(f());
57        }
58    }
59}
60impl<T> Eventually<T> {
61    #[tracing::instrument(name = "initialize", skip_all)]
62    /// Initializes on first use and returns the stored resource; later calls do not invoke `f`.
63    pub fn initialize(&mut self, f: impl FnOnce() -> T) -> &mut T {
64        if let Eventually::Uninitialized = self {
65            *self = Eventually::Initialized(f());
66        }
67
68        match self {
69            Eventually::Initialized(data) => data,
70            Eventually::Uninitialized => panic!("not initialized"),
71        }
72    }
73
74    /// Moves out the stored state, leaving this wrapper uninitialized.
75    pub fn take(&mut self) -> Eventually<T> {
76        mem::replace(self, Eventually::Uninitialized)
77    }
78
79    /// Borrows the resource mutably when the caller has already established initialization.
80    ///
81    /// # Panics
82    /// Panics with `message` if this wrapper is uninitialized.
83    pub fn expect_initialized_mut(&mut self, message: &str) -> &mut T {
84        match self {
85            Eventually::Initialized(value) => value,
86            Eventually::Uninitialized => panic!("{message}"),
87        }
88    }
89}

91impl<T> HasTile for Eventually<T>
92where
93    T: HasTile,
94{
95    fn has_source_tile(
96        &self,
97        source: &crate::raster::RasterSourceId,
98        coords: WorldTileCoords,
99        world: &World,
100    ) -> bool {
101        match self {
102            Eventually::Initialized(value) => value.has_source_tile(source, coords, world),
103            Eventually::Uninitialized => false,
104        }
105    }
106
107    fn has_tile(&self, coords: WorldTileCoords, world: &World) -> bool {
108        match self {
109            Eventually::Initialized(value) => value.has_tile(coords, world),
110            Eventually::Uninitialized => false,
111        }
112    }
