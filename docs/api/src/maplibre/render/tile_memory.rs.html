20fn capacity<T>(values: &Vec<T>) -> usize {
21    values.capacity().saturating_mul(size_of::<T>())
22}

24/// Bytes owned by this tile, including an estimate for nested metadata and its query index.
25pub(crate) fn tile_bytes(tiles: &Tiles, coords: WorldTileCoords) -> usize {
26    let vector = tiles
27        .query::<&VectorLayerBucketComponent>(coords)
28        .map_or(0, |bucket| {
29            bucket
30                .layers
31                .iter()
32                .map(|layer| match layer {
33                    VectorLayerBucket::AvailableLayer(layer) => vector_bytes(layer),
34                    VectorLayerBucket::Missing(_) => 0,
35                })
36                .sum::<usize>()
37        });
38    let raster = tiles
39        .query::<&RasterLayersDataComponent>(coords)
40        .map_or(0, |bucket| {
41            bucket
42                .layers
43                .iter()
44                .map(|layer| match layer {
45                    RasterLayerData::Available(layer) => capacity(layer.image.as_raw()),
46                    RasterLayerData::Missing(_) => 0,
47                })
48                .sum::<usize>()
49        });
50    let mut seen_dems = HashSet::new();
51    let mut dem = 0;
52    if let Some(DemTileComponent::Loaded(loaded)) = tiles.query::<&DemTileComponent>(coords) {
53        if seen_dems.insert(Arc::as_ptr(&loaded.tile)) {
54            dem += loaded.tile.pixels().len();
55        }
56    }
57    if let Some(layers) = tiles.query::<&DemLayersComponent>(coords) {
58        for (_, loaded) in layers.iter() {
59            if seen_dems.insert(Arc::as_ptr(&loaded.tile)) {
60                dem += loaded.tile.pixels().len();
61            }
62        }
63    }
64    let pending = tiles
65        .query::<&LayerReplacements>(coords)
66        .map_or(0, |pending| {
67            pending.vector.iter().map(vector_bytes).sum::<usize>()
68        });
69    vector
70        + pending
71        + raster
72        + dem
73        + symbols(tiles, coords)
74        + tiles.geometry_index.tile_bytes(coords)
75        + feature_paint_bytes(tiles, coords)
76}

78fn feature_paint_bytes(tiles: &Tiles, coords: WorldTileCoords) -> usize {
79    let Some(paint) = tiles.query::<&TileFeaturePaintComponent>(coords) else {
80        return 0;
81    };
82    let mut seen_props: FxHashSet<*const FeatureProperties> = FxHashSet::default();
83    tiles
84        .geometry_index
85        .record_tile_property_ptrs(coords, &mut seen_props);
86    const ENTRY_OVERHEAD: usize = 2 * size_of::<String>() + 16;
87    let mut total = 0_usize;
88    for (layer_id, entries) in paint.committed.iter().chain(paint.pending.iter()) {
89        total = total
90            .saturating_add(size_of::<(String, Vec<BucketFeatureEntry>)>())
91            .saturating_add(layer_id.capacity())
92            .saturating_add(capacity(entries));
93        for entry in entries {
94            if seen_props.insert(Arc::as_ptr(&entry.properties)) {
95                let props_bytes: usize = entry
96                    .properties
97                    .iter()
98                    .map(|(k, v)| k.capacity() + value_bytes(v) + ENTRY_OVERHEAD)
99                    .sum();
100                total = total
101                    .saturating_add(size_of::<FeatureProperties>())
102                    .saturating_add(props_bytes);
103            }
104        }
105    }
106    total
107}

109fn symbols(tiles: &Tiles, coords: WorldTileCoords) -> usize {
110    let mut atlases = HashSet::new();
111    tiles
112        .query::<&SymbolLayersDataComponent>(coords)
113        .into_iter()
114        .flat_map(|bucket| &bucket.layers)
115        .chain(
116            tiles
117                .query::<&LayerReplacements>(coords)
118                .into_iter()
119                .flat_map(|pending| &pending.symbols),
120        )
121        .map(|layer| {
122            let geometry = capacity(&layer.buffer.buffer.vertices)
123                + capacity(&layer.buffer.buffer.indices)
124                + capacity(&layer.features)
125                + layer
126                    .features
127                    .iter()
128                    .map(|feature| {
129                        feature.str.capacity()
130                            + feature.data.properties.capacity()
131                                * size_of::<(String, crate::style::expression::Value)>()
132                            + feature
133                                .data
134                                .properties
135                                .iter()
136                                .map(|(key, value)| key.capacity() + value_bytes(value))
137                                .sum::<usize>()
138                    })
139                    .sum::<usize>();
140            let atlas = layer
141                .atlas
142                .as_ref()
143                .filter(|atlas| atlases.insert(Arc::as_ptr(atlas)))
144                .map_or(0, |atlas| atlas.approximate_bytes());
145            geometry + atlas
146        })
147        .sum()
148}

153fn value_bytes(value: &crate::style::expression::Value) -> usize {
154    use crate::style::expression::Value;
155    match value {
156        Value::String(value) => value.capacity(),
157        Value::Array(values) => capacity(values) + values.iter().map(value_bytes).sum::<usize>(),
158        Value::Object(values) => values
159            .iter()
160            .map(|(key, value)| size_of::<(String, Value)>() + key.capacity() + value_bytes(value))
161            .sum(),
162        _ => 0,
163    }
164}

166fn vector_bytes(layer: &AvailableVectorLayerBucket) -> usize {
167    capacity(&layer.buffer.buffer.vertices)
168        + capacity(&layer.buffer.buffer.indices)
169        + capacity(&layer.feature_indices)
170        + capacity(&layer.feature_colors)
171        + capacity(&layer.feature_sort_keys)
172}
