23// Valid tiles use `0..=123` (`(z % 31) * 4 + quadrant`). `0x7f` (`127`) stays outside all tile
24// references while keeping bit 7 (`0x80`, `STENCIL_MARK`) clear so 3D extrusions rising above
25// unmasked horizon/background pixels are not rejected by `extrusion_color_pipeline`.
26pub(crate) const EMPTY_STENCIL_REFERENCE: u32 = 0x7f;
27
28/// High stencil bit (`0x80`) marking pixels already drawn by the extrusion color pass.
29/// Both tile references (`0..=123`) and `EMPTY_STENCIL_REFERENCE` (`0x7f`) keep this bit zero.
30pub(crate) const STENCIL_MARK: u32 = 0x80;

40impl Node for MainPassNode {
41    fn run(&self, render_context: &mut RenderContext, state: &RenderResources, world: &World) {
42        let Initialized(render_target) = &state.render_target else {
43            return;
44        };
45        let Initialized(multisampling_texture) = &state.multisampling_texture else {
46            return;
47        };
48        let Initialized(depth_texture) = &state.depth_texture else {
49            return;
50        };
51        let color = color_attachment(render_target.deref(), multisampling_texture.as_ref());
52        let pass = render_context
53            .command_encoder
54            .begin_render_pass(&wgpu::RenderPassDescriptor {
55                multiview_mask: None,
56                label: Some("main_pass"),
57                color_attachments: &[Some(color)],
58                depth_stencil_attachment: Some(wgpu::RenderPassDepthStencilAttachment {
59                    view: &depth_texture.view,
60                    depth_ops: Some(wgpu::Operations {
61                        load: wgpu::LoadOp::Clear(0.0),
62                        store: StoreOp::Store,
63                    }),
64                    stencil_ops: Some(wgpu::Operations {
65                        load: wgpu::LoadOp::Clear(EMPTY_STENCIL_REFERENCE),
66                        store: StoreOp::Store,
67                    }),
68                }),
69                timestamp_writes: world
70                    .resources
71                    .get::<crate::render::gpu_timer::GpuTimerSlot>()
72                    .and_then(|slot| slot.0.as_ref())
73                    .and_then(crate::render::gpu_timer::GpuTimer::pass_writes),
74                occlusion_query_set: None,
75            });
76        let mut pass = crate::render::tracked_pass::TrackedRenderPass::new(pass);
77        draw_layers(&mut pass, world);
78        pass.finish(world);
79    }
80}

82/// Active stencil contents left in the current render target.
83#[derive(Clone, Copy, Default)]
84pub(crate) enum StencilState<'w> {
85    /// Target holds [`EMPTY_STENCIL_REFERENCE`] everywhere.
86    #[default]
87    Cleared,
88    /// All `TileKind::Vector` masks are stamped coarse-to-fine and bit 7 (`STENCIL_MARK`) is zero.
89    Vector,
90    /// Masks belonging to this mutating layer group were stamped and possibly bitwise-inverted.
91    Mutated(&'w [LayerItem]),
92    /// Arbitrary stencil values may remain after a custom layer draw.
93    DirtyAll,
94}
95
96fn unique_vector_stencils(masks: &[TileMaskItem]) -> bool {
97    let mut seen = [0_u64; 2];
98    masks
99        .iter()
100        .filter(|mask| mask.is_vector && !mask.generate_borders)
101        .all(|mask| {
102            let reference = usize::from(mask.source_shape.stencil_reference());
103            let Some(word) = seen.get_mut(reference / 64) else {
104                return false;
105            };
106            let bit = 1_u64 << (reference % 64);
107            let fresh = *word & bit == 0;
108            *word |= bit;
109            fresh
110        })
111}

113fn next_stencil_state<'w>(
114    state: StencilState<'w>,
115    masks: &[TileMaskItem],
116    group: &'w [LayerItem],
117) -> StencilState<'w> {
118    let Some(first) = group.first() else {
119        return state;
120    };
121    match first.mask_group {
122        MaskGroup::Unmasked => state,
123        MaskGroup::Vector if unique_vector_stencils(masks) => StencilState::Vector,
124        MaskGroup::Vector | MaskGroup::Mutating => StencilState::Mutated(group),
125        MaskGroup::Custom => StencilState::DirtyAll,
126    }
127}

129/// The layers that draw after the first symbol layer, with the stencil state left before it.
130///
131/// Symbols draw in their own pass so they can read the depth the layers below them left; the
132/// layers above the first symbol layer join that pass in order, so they cover symbols as the
133/// style's order says. A frame with a ground surface draws everything in the main pass.
134pub(super) struct AboveSymbols<'w> {
135    /// Stencil state left at the end of the main pass.
136    pub stencil_state: StencilState<'w>,
137    /// Items of the layers that draw after the first symbol layer.
138    pub items: &'w [LayerItem],
139}
140
141pub(super) fn above_symbols(world: &World) -> Option<AboveSymbols<'_>> {
142    if world
143        .resources
144        .get::<crate::render::ground::FrameGround>()
145        .is_some_and(|ground| ground.0.is_some())
146    {
147        return None;
148    }
149    let cutoff = world
150        .resources
151        .get::<RenderPhase<TranslucentItem>>()?
152        .into_iter()
153        .map(|item| item.index)
154        .filter(|index| *index != u32::MAX)
155        .min()?;
156    let layers = world.resources.get::<RenderPhase<LayerItem>>()?;
157    let items = layers.into_iter().as_slice();
158    let split = items.partition_point(|item| item.index < cutoff);
159    let below = &items[..split];
160    let masks = world
161        .resources
162        .get::<RenderPhase<TileMaskItem>>()
163        .map(|phase| phase.into_iter().as_slice());
164    let masks_ready = masks.is_some()
165        && matches!(
166            world.resources.query::<(
167                &Eventually<MaskPipeline>,
168                &Eventually<crate::render::projection::ProjectionGpuResources>,
169            )>(),
170            Some((Initialized(_), Initialized(_)))
171        );
172    let mut stencil_state = StencilState::Cleared;
173    if let (true, Some(masks)) = (masks_ready, masks) {
174        for group in below.chunk_by(same_layer) {
175            stencil_state = next_stencil_state(stencil_state, masks, group);
176        }
177    }
178    Some(AboveSymbols {
179        stencil_state,
180        items: &items[split..],
181    })
182}

188/// Draws one layer's items inside its tile masks, updating `stencil_state`.
189pub(super) fn draw_group<'w>(
190    pass: &mut crate::render::tracked_pass::TrackedRenderPass<'w>,
191    world: &'w World,
192    stencil_state: &mut StencilState<'w>,
193    group: &'w [LayerItem],
194) {
195    if let Some(masks) = world.resources.get::<RenderPhase<TileMaskItem>>() {
196        apply_layer_masks(
197            pass,
198            world,
199            masks.into_iter().as_slice(),
200            stencil_state,
201            group,
202        );
203    }
204    for layer in group {
205        layer.draw_function.draw(pass, world, layer);
206    }
207}

209fn draw_layers<'w>(
210    pass: &mut crate::render::tracked_pass::TrackedRenderPass<'w>,
211    world: &'w World,
212) {
213    let ground = world
214        .resources
215        .get::<crate::render::ground::FrameGround>()
216        .and_then(|ground| ground.0);
217    let mut ground_pending = ground.is_some();
218    // Pass hooks at the beginning of main pass (e.g. starfield).
219    let pass_hooks = world
220        .resources
221        .get::<crate::render::PassDrawHooks>()
222        .map(|hooks| hooks.get_hooks(crate::render::PassHookPoint::MainPassBegin));
223    if let Some(hooks) = pass_hooks {
224        for hook in hooks {
225            hook.draw(pass, world);
226        }
227    }
228    let above = above_symbols(world);
229    let masks = world
230        .resources
231        .get::<RenderPhase<TileMaskItem>>()
232        .map(|phase| phase.into_iter().as_slice());
233    if let Some(layers) = world.resources.get::<RenderPhase<LayerItem>>() {
234        let mut stencil_state = StencilState::Cleared;
235        let all = layers.into_iter().as_slice();
236        let drawn = all.len() - above.as_ref().map_or(0, |above| above.items.len());
237        for group in all[..drawn].chunk_by(same_layer) {
238            if let Some(masks) = masks {
239                apply_layer_masks(pass, world, masks, &mut stencil_state, group);
240            }
241            for layer in group {
242                // The ground preserves depth for screen-space layers while remaining above the
243                // background.
244                if let Some(surface) = ground.filter(|surface| {
245                    ground_pending && layer.index > surface.draw_after_layer_index
246                }) {
247                    (surface.draw)(pass, world);
248                    ground_pending = false;
249                }
250                layer.draw_function.draw(pass, world, layer);
251            }
252        }
253    }
254    if let Some(surface) = ground.filter(|_| ground_pending) {
255        (surface.draw)(pass, world);
256    }
257}

259pub(crate) fn apply_layer_masks<'w>(
260    pass: &mut crate::render::tracked_pass::TrackedRenderPass<'w>,
261    world: &'w World,
262    masks: &[TileMaskItem],
263    state: &mut StencilState<'w>,
264    current: &'w [LayerItem],
265) {
266    let Some(first) = current.first() else {
267        return;
268    };
269    match first.mask_group {
270        MaskGroup::Unmasked => return,
271        MaskGroup::Custom => {
272            *state = StencilState::DirtyAll;
273            return;
274        }
275        MaskGroup::Vector if matches!(*state, StencilState::Vector) => return,
276        MaskGroup::Vector | MaskGroup::Mutating => {}
277    }
278
279    let Some(first_mask) = masks.first() else {
280        *state = next_stencil_state(*state, masks, current);
281        return;
282    };
283    if matches!(
284        SetMaskPipeline::render(world, first_mask, pass),
285        RenderCommandResult::Failure
286    ) {
287        return;
288    }
289
290    let owns_mask = |mask: &TileMaskItem, layers: &[LayerItem]| {
291        layers
292            .iter()
293            .any(|layer| layer.source_shape.buffer_range() == mask.source_shape.buffer_range())
294    };
295
296    // Clear only masks left dirty or mismatched by a preceding pyramid.
297    match *state {
298        StencilState::Cleared => {}
299        StencilState::Vector => {
300            for mask in masks.iter().filter(|mask| mask.is_vector) {
301                DrawMask::render_with_reference(world, mask, pass, EMPTY_STENCIL_REFERENCE);
302            }
303        }
304        StencilState::Mutated(previous) => {
305            for mask in masks.iter().filter(|mask| owns_mask(mask, previous)) {
306                DrawMask::render_with_reference(world, mask, pass, EMPTY_STENCIL_REFERENCE);
307            }
308        }
309        StencilState::DirtyAll => {
310            for mask in masks {
311                DrawMask::render_with_reference(world, mask, pass, EMPTY_STENCIL_REFERENCE);
312            }
313        }
314    }
315
316    match first.mask_group {
317        MaskGroup::Vector if unique_vector_stencils(masks) => {
318            for mask in masks.iter().filter(|mask| mask.is_vector) {
319                mask.draw_function.draw(pass, world, mask);
320            }
321            *state = StencilState::Vector;
322        }
323        MaskGroup::Vector | MaskGroup::Mutating => {
324            for mask in masks.iter().filter(|mask| owns_mask(mask, current)) {
325                mask.draw_function.draw(pass, world, mask);
326            }
327            *state = StencilState::Mutated(current);
328        }
329        MaskGroup::Unmasked | MaskGroup::Custom => {}
330    }
331}

333fn color_attachment<'a>(
334    target: &'a wgpu::TextureView,
335    multisampled: Option<&'a Texture>,
336) -> wgpu::RenderPassColorAttachment<'a> {
337    let (view, resolve_target) = match multisampled {
338        Some(texture) => (texture.view.deref(), Some(target)),
339        None => (target, None),
340    };
341    wgpu::RenderPassColorAttachment {
342        depth_slice: None,
343        view,
344        resolve_target,
345        ops: wgpu::Operations {
346            load: wgpu::LoadOp::Clear(wgpu::Color::TRANSPARENT),
347            store: StoreOp::Store,
348        },
349    }
