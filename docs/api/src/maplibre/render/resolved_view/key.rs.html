1//! Bitwise cache key tracking all style and [`ViewState`] inputs of a [`ResolvedFrameView`](super::ResolvedFrameView).
2
3use cgmath::Matrix4;
4
5use crate::{
6    render::view_state::{CameraAuthority, ViewState},
7    style::Style,
8};
9
10#[derive(Clone, Copy, Debug, PartialEq, Eq)]
11pub(super) struct ViewSnapshotKey {
12    width_bits: u64,
13    height_bits: u64,
14    insets_bits: [u64; 4],
15    fovy_bits: u64,
16    zoom_bits: u64,
17    style_zoom_bits: u64,
18    cam_pos_bits: [u64; 2],
19    cam_angles_bits: [u64; 3],
20    center_elev_bits: u64,
21    min_elev_bits: u64,
22    body_radius_bits: u64,
23    globe_orbits_center: bool,
24    terrain_enabled: bool,
25    transition_bits: u32,
26    authority: CameraAuthority,
27    pose_bits: Option<[u64; 9]>,
28    external_proj_bits: Option<[u64; 16]>,
29    external_cam_bits: Option<[u64; 16]>,
30}
31
32fn matrix_bits(m: Matrix4<f64>) -> [u64; 16] {
33    [
34        m.x.x.to_bits(),
35        m.x.y.to_bits(),
36        m.x.z.to_bits(),
37        m.x.w.to_bits(),
38        m.y.x.to_bits(),
39        m.y.y.to_bits(),
40        m.y.z.to_bits(),
41        m.y.w.to_bits(),
42        m.z.x.to_bits(),
43        m.z.y.to_bits(),
44        m.z.z.to_bits(),
45        m.z.w.to_bits(),
46        m.w.x.to_bits(),
47        m.w.y.to_bits(),
48        m.w.z.to_bits(),
49        m.w.w.to_bits(),
50    ]
51}
52
53impl ViewSnapshotKey {
54    pub(super) fn capture(style: &Style, view_state: &ViewState, transition: f32) -> Self {
55        let insets = view_state.edge_insets();
56        let pos = view_state.camera().position();
57        let pose_bits = view_state.globe_pose().map(|pose| {
58            [
59                pose.target[0].to_bits(),
60                pose.target[1].to_bits(),
61                pose.target[2].to_bits(),
62                pose.target_elevation_meters.to_bits(),
63                pose.orientation[0].to_bits(),
64                pose.orientation[1].to_bits(),
65                pose.orientation[2].to_bits(),
66                pose.orientation[3].to_bits(),
67                pose.distance_meters.to_bits(),
68            ]
69        });
70        Self {
71            width_bits: view_state.width().to_bits(),
72            height_bits: view_state.height().to_bits(),
73            insets_bits: [
74                insets.top.to_bits(),
75                insets.bottom.to_bits(),
76                insets.left.to_bits(),
77                insets.right.to_bits(),
78            ],
79            fovy_bits: view_state.field_of_view().0.to_bits(),
80            zoom_bits: view_state.zoom().value().to_bits(),
81            style_zoom_bits: view_state.style_zoom().value().to_bits(),
82            cam_pos_bits: [pos.x.to_bits(), pos.y.to_bits()],
83            cam_angles_bits: [
84                view_state.camera().get_bearing().0.to_bits(),
85                view_state.camera().get_pitch().0.to_bits(),
86                view_state.camera().get_roll().0.to_bits(),
87            ],
88            center_elev_bits: view_state.center_elevation().to_bits(),
89            min_elev_bits: view_state.min_elevation().to_bits(),
90            body_radius_bits: view_state.body().radius_meters.to_bits(),
91            globe_orbits_center: view_state.globe_orbits_center(),
92            terrain_enabled: style.terrain.is_some(),
93            transition_bits: transition.to_bits(),
94            authority: view_state.camera_authority(),
95            pose_bits,
96            external_proj_bits: view_state.external_projection().map(matrix_bits),
97            external_cam_bits: view_state.external_camera_matrix().map(matrix_bits),
98        }
99    }
100}
