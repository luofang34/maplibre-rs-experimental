1//! Native persistent `MAP_WRITE | COPY_SRC` staging ring for [`UploadQueue`](super::UploadQueue).
2//!
3//! Avoids per-frame `wgpu::Queue::write_buffer` staging allocations and `(wgpu internal)
4//! PendingWrites` blit command buffers on native non-GL backends by writing staged bytes directly
5//! into mapped ring chunks and encoding `copy_buffer_to_buffer` transfers into the frame encoder.
6
7use std::sync::{
8    atomic::{AtomicU8, Ordering},
9    Arc,
10};
11
12use super::StagedCopy;
13
14const IN_FLIGHT: u8 = 0;
15const READY_MAPPED: u8 = 1;
16const MAP_FAILED: u8 = 2;
17
18/// Minimum chunk capacity (`64 KiB`).
19const MIN_CHUNK_CAPACITY: u64 = 64 * 1024;
20
21/// Steady-state chunk ceiling retained across quiet frames (`64 KiB`).
22pub(super) const STEADY_CHUNK_CAPACITY: u64 = 64 * 1024;
23
24/// Maximum chunk capacity kept in the ring across frames (`256 KiB`).
25const MAX_RETAINED_CHUNK_CAPACITY: u64 = 256 * 1024;
26
27/// Maximum number of ready/mapping chunks retained in the steady-state ring.
28const MAX_RING_CHUNKS: usize = 3;
29
30/// Consecutive low-volume frames before oversized (`> 64 KiB`) chunks are pruned.
31const QUIET_FRAMES_BEFORE_SHRINK: u8 = 8;
32
33#[derive(Debug)]
34struct MappedChunk {
35    buffer: wgpu::Buffer,
36    capacity: u64,
37    map_state: Arc<AtomicU8>,
38}
39
40impl Drop for MappedChunk {
41    fn drop(&mut self) {
42        self.buffer.destroy();
43    }
44}
45
46#[derive(Debug, Default)]
47pub(super) struct NativeStagingRing {
48    ring: Vec<MappedChunk>,
49    in_flight: Vec<MappedChunk>,
50    quiet_frames: u8,
51}
52
53impl NativeStagingRing {
54    /// Records one low- or zero-volume frame and prunes oversized idle chunks once settled.
55    pub(super) fn record_quiet_frame(&mut self) {
56        self.quiet_frames = self.quiet_frames.saturating_add(1);
57        if self.quiet_frames >= QUIET_FRAMES_BEFORE_SHRINK {
58            self.prune_oversized();
59        }
60    }
61
62    /// Resets the quiet-frame counter after a high-volume transfer burst.
63    pub(super) fn reset_quiet_frames(&mut self) {
64        self.quiet_frames = 0;
65    }
66
67    fn prune_oversized(&mut self) {
68        let mut idx = 0;
69        while idx < self.ring.len() {
70            let chunk = &self.ring[idx];
71            if chunk.capacity > STEADY_CHUNK_CAPACITY
72                && chunk.map_state.load(Ordering::Acquire) != IN_FLIGHT
73            {
74                drop(self.ring.swap_remove(idx));
75            } else {
76                idx += 1;
77            }
78        }
79    }
80
81    fn drop_failed_chunks(&mut self) {
82        let mut idx = 0;
83        while idx < self.ring.len() {
84            if self.ring[idx].map_state.load(Ordering::Acquire) == MAP_FAILED {
85                drop(self.ring.swap_remove(idx));
86            } else {
87                idx += 1;
88            }
89        }
90    }
91
92    fn take_ready_chunk(&mut self, needed: u64) -> Option<MappedChunk> {
93        self.drop_failed_chunks();
94        let mut best: Option<(usize, u64)> = None;
95        for (idx, chunk) in self.ring.iter().enumerate() {
96            if chunk.capacity >= needed
97                && chunk.map_state.load(Ordering::Acquire) == READY_MAPPED
98                && best.is_none_or(|(_, best_cap)| chunk.capacity < best_cap)
99            {
100                best = Some((idx, chunk.capacity));
101            }
102        }
103        best.map(|(idx, _)| self.ring.swap_remove(idx))
104    }
105
106    fn allocate_chunk(device: &wgpu::Device, needed: u64, max_buf: u64) -> MappedChunk {
107        let capacity = needed
108            .max(MIN_CHUNK_CAPACITY)
109            .checked_next_power_of_two()
110            .unwrap_or(max_buf)
111            .min(max_buf);
112        let buffer = device.create_buffer(&wgpu::BufferDescriptor {
113            label: Some("upload_queue_mapped_staging"),
114            size: capacity,
115            usage: wgpu::BufferUsages::MAP_WRITE | wgpu::BufferUsages::COPY_SRC,
116            mapped_at_creation: true,
117        });
118        MappedChunk {
119            buffer,
120            capacity,
121            map_state: Arc::new(AtomicU8::new(READY_MAPPED)),
122        }
123    }
124
125    /// Writes `payload` directly into a mapped staging chunk, unmaps it, encodes
126    /// `copy_buffer_to_buffer` transfers onto `encoder`, and returns `true` when any copies were
127    /// recorded. Falls back to `false` without draining `copies` if mapping access fails.
128    pub(super) fn encode_into(
129        &mut self,
130        device: &wgpu::Device,
131        encoder: &mut wgpu::CommandEncoder,
132        payload: &[u8],
133        copies: &mut Vec<StagedCopy>,
134        max_buf: u64,
135    ) -> bool {
136        let payload_len = payload.len() as u64;
137        if payload_len == 0 || copies.is_empty() || payload_len > max_buf {
138            return false;
139        }
140        if payload_len <= STEADY_CHUNK_CAPACITY {
141            self.record_quiet_frame();
142        } else {
143            self.reset_quiet_frames();
144        }
145
146        let chunk = self
147            .take_ready_chunk(payload_len)
148            .unwrap_or_else(|| Self::allocate_chunk(device, payload_len, max_buf));
149
150        let write_ok = match chunk.buffer.slice(0..payload_len).get_mapped_range_mut() {
151            Ok(mut view) => {
152                view.copy_from_slice(payload);
153                true
154            }
155            Err(_) => {
156                chunk.map_state.store(MAP_FAILED, Ordering::Release);
157                false
158            }
159        };
160        if !write_ok {
161            drop(chunk);
162            return false;
163        }
164        chunk.buffer.unmap();
165
166        let mut encoded = false;
167        for copy in copies.drain(..) {
168            encoder.copy_buffer_to_buffer(
169                &chunk.buffer,
170                copy.src_offset,
171                &copy.dst,
172                copy.dst_offset,
173                copy.size,
174            );
175            encoded = true;
176        }
177        chunk.map_state.store(IN_FLIGHT, Ordering::Release);
178        self.in_flight.push(chunk);
179        encoded
180    }
181
182    /// Re-maps submitted ring chunks for future frames and drops transient burst allocations.
183    pub(super) fn recall_after_submit(&mut self) {
184        let should_shrink = self.quiet_frames >= QUIET_FRAMES_BEFORE_SHRINK;
185        for chunk in self.in_flight.drain(..) {
186            let over_ceiling = chunk.capacity > MAX_RETAINED_CHUNK_CAPACITY
187                || (should_shrink && chunk.capacity > STEADY_CHUNK_CAPACITY);
188            if over_ceiling || self.ring.len() >= MAX_RING_CHUNKS {
189                drop(chunk);
190                continue;
191            }
192            let state = Arc::clone(&chunk.map_state);
193            state.store(IN_FLIGHT, Ordering::Release);
194            chunk
195                .buffer
196                .slice(..)
197                .map_async(wgpu::MapMode::Write, move |res| {
198                    let next = if res.is_ok() {
199                        READY_MAPPED
200                    } else {
201                        MAP_FAILED
202                    };
203                    state.store(next, Ordering::Release);
204                });
205            self.ring.push(chunk);
206        }
207    }
208}
