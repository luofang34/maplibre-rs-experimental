32#[derive(Debug)]
33struct BackingBuffer<B> {
34    /// The internal structure which is used for storage
35    inner: B,
36    /// The size of the `inner` buffer
37    inner_size: wgpu::BufferAddress,
38}

41    fn new(inner: B, inner_size: wgpu::BufferAddress) -> Self {
42        Self { inner, inner_size }
43    }

46/// The tile mask pattern assigns each tile a value which can be used for stencil testing.
47pub struct TileViewPattern<Q, B> {
48    view_tiles: Vec<ViewTile>,
49    view_tiles_buffer: BackingBuffer<B>,
50    /// Occupied entries, including appended metadata; a view-pattern upload starts a new frame.
51    uploaded: u64,
52    phantom_q: PhantomData<Q>,
53}

55/// The metadata buffer cannot hold every requested entry.
56#[derive(Clone, Copy, Debug, thiserror::Error, PartialEq, Eq)]
57#[error("tile metadata buffer holds {capacity} entries, {requested} were requested")]
58pub struct TileMetadataOverflow {
59    /// Entries the buffer can hold.
60    pub capacity: u64,
61    /// Entries needed this frame.
62    pub requested: u64,
63}
64
65impl<Q: Queue<B>, B> TileViewPattern<Q, B> {
66    /// Takes ownership of a metadata buffer whose descriptor capacity is measured in bytes.
67    pub fn new(view_tiles_buffer: BackingBufferDescriptor<B>) -> Self {
68        Self {
69            view_tiles: Vec::with_capacity(64),
70            view_tiles_buffer: BackingBuffer::new(
71                view_tiles_buffer.buffer,
72                view_tiles_buffer.inner_size,
73            ),
74            uploaded: 0,
75            phantom_q: Default::default(),
76        }
77    }
78
79    /// Number of metadata entries still available in this frame.
80    pub fn remaining_metadata_capacity(&self) -> usize {
81        let capacity = self.view_tiles_buffer.inner_size / size_of::<ShaderTileMetadata>() as u64;
82        capacity.saturating_sub(self.uploaded) as usize
83    }
84
85    /// Appends entries without changing ranges returned by earlier appends in this frame.
86    /// Call after [`Self::upload_pattern`]; the next pattern upload invalidates these ranges.
87    /// Returns an overflow error without writing or reserving entries if they do not all fit.
88    pub fn upload_extra_metadata(
89        &mut self,
90        queue: &Q,
91        entries: &[ShaderTileMetadata],
92    ) -> Result<Vec<std::ops::Range<wgpu::BufferAddress>>, TileMetadataOverflow> {
93        const STRIDE: u64 = size_of::<ShaderTileMetadata>() as u64;
94        let capacity = self.view_tiles_buffer.inner_size / STRIDE;
95        let requested = self.uploaded + entries.len() as u64;
96        if requested > capacity {
97            return Err(TileMetadataOverflow {
98                capacity,
99                requested,
100            });
101        }
102        if entries.is_empty() {
103            return Ok(Vec::new());
104        }
105        let offset = self.uploaded * STRIDE;
106        queue.write_buffer(
107            &self.view_tiles_buffer.inner,
108            offset,
109            bytemuck::cast_slice(entries),
110        );
111        self.uploaded = requested;
112        Ok((0..entries.len() as u64)
113            .map(|index| offset + index * STRIDE..offset + (index + 1) * STRIDE)
114            .collect())
115    }
116
117    /// Pairs every view tile with the loaded sources that draw it. Vector shapes come from
118    /// the pyramid nearest the tile; raster shapes follow each raster source's own covering,
119    /// as GL JS pairs a source's visible tiles with the view, and fall back to the pyramid
120    /// while those tiles load.
121    #[tracing::instrument(skip_all)]
122    #[must_use]
123    pub fn generate_pattern(
124        &self,
125        view_region: &ViewRegion,
126        sources: &ViewTileSources,
127        raster_coverings: &RasterCoverings,
128        zoom: Zoom,
129        world: &World,
130    ) -> Vec<ViewTile> {
131        let mut view_tiles = Vec::with_capacity(self.view_tiles.len());
132        // A parent stands in once for each copy of the world it is seen in.
133        let mut vector_parents: HashMap<i32, HashSet<WorldTileCoords>> = HashMap::new();
134        let mut raster_parents: HashMap<(RasterSourceId, i32), HashSet<WorldTileCoords>> =
135            HashMap::new();
136        let covering: Vec<_> = if raster_coverings.is_empty() {
137            vec![(RasterSourceId::default(), Vec::new())]
138        } else {
139            raster_coverings.to_vec()
140        };
141
142        for seen in view_region.copies() {
143            let Some((coords, wrap)) = seen.wrapped() else {
144                continue;
145            };
146            let mut vector = source_shapes(
147                &sources.of_kind(TileKind::Vector),
148                coords,
149                zoom,
150                world,
151                vector_parents.entry(wrap).or_default(),
152            );
153            vector.for_each_mut(&mut |shape| shape.wrapped(wrap, zoom));
154            let raster = covering
155                .iter()
156                .map(|(source, covering)| {
157                    let availability = super::SourceTiles {
158                        source,
159                        availability: sources.of_kind(TileKind::Raster),
160                    };
161                    let covered = covering_shapes_for(coords, covering)
162                        .into_iter()
163                        .filter(|tile| availability.has_tile(*tile, world))
164                        .collect();
165                    let mut shapes = raster_shapes(
166                        &availability,
167                        coords,
168                        covered,
169                        zoom,
170                        world,
171                        raster_parents.entry((source.clone(), wrap)).or_default(),
172                    );
173                    if let Some(faded) =
174                        with_departing((&availability, source, &shapes), zoom, world)
175                    {
176                        shapes = faded;
177                    }
178                    shapes.for_each_mut(&mut |shape| shape.wrapped(wrap, zoom));
179                    (source.clone(), shapes)
180                })
181                .collect();
182            view_tiles.push(ViewTile {
183                target: coords,
184                vector,
185                raster,
186            });
187        }
188
189        view_tiles
190    }
191
192    /// Replaces the CPU-side view tiles; call [`Self::upload_pattern`] before drawing them.
193    pub fn update_pattern(&mut self, mut view_tiles: Vec<ViewTile>) {
194        self.view_tiles.clear();
195        self.view_tiles.append(&mut view_tiles)
196    }
197
198    /// Iterates view tiles in covering order; source shapes may reference shared ancestors.
199    pub fn iter(&self) -> impl Iterator<Item = &ViewTile> + '_ {
200        self.view_tiles.iter()
201    }
202
203    /// Borrows the backing metadata buffer, including space reserved for extra entries.
204    pub fn buffer(&self) -> &B {
205        &self.view_tiles_buffer.inner
206    }
207
208    /// Writes this frame's view metadata and resets the append cursor.
209    /// Shapes exceeding capacity lose their buffer ranges and are skipped for this frame.
210    /// Viewport dimensions and `style_zoom` determine screen-pixel line widths.
211    #[tracing::instrument(skip_all)]
212    pub fn upload_pattern(
213        &mut self,
214        queue: &Q,
215        view_proj: &ViewProjection,
216        viewport_width: f32,
217        viewport_height: f32,
218        style_zoom: Zoom,
219    ) {
220        let capacity = (self.view_tiles_buffer.inner_size
221            / std::mem::size_of::<ShaderTileMetadata>() as wgpu::BufferAddress)
222            as usize;
223        let mut buffer = Vec::with_capacity(self.view_tiles.len());
224        let mut skipped = 0_usize;
225
226        let mut add_to_buffer = |shape: &mut TileShape| {
227            if buffer.len() >= capacity {
228                // A pitched view falling back to many small child tiles can exceed the
229                // buffer; shapes past the end stay unrendered until their own tiles load.
230                shape.clear_buffer_range();
231                skipped += 1;
232                return;
233            }
234            shape.set_buffer_range(buffer.len() as u64);
235            let transform = view_proj
236                .to_model_view_projection(shape.transform)
237                .downcast()
238                .into();
239            buffer.push(ShaderTileMetadata {
240                transform,
241                zoom_factor: shape.zoom_factor as f32,
242                viewport_width,
243                viewport_height,
244                tile_mercator_coords: {
245                    let mut mercator = tile_mercator_coordinates(
246                        shape
247                            .coords()
248                            .into_tile(crate::style::source::TileAddressingScheme::XYZ),
249                    );
250                    // A copy of the world lies whole worlds over in Mercator units.
251                    mercator.x += shape.wrap as f32;
252                    mercator.into()
253                },
254                line_width_scale: 1.0,
255                line_units_per_pixel: 8.0 * style_zoom.scale_to_tile(&shape.coords()) as f32,
256                clip_antimeridian: u32::from(u8::from(shape.coords().z) == 0),
257                clip_region: clip_region(shape),
258            });
259        };
260
261        for view_tile in &mut self.view_tiles {
262            view_tile.vector.for_each_mut(&mut add_to_buffer);
263            for (_, shapes) in &mut view_tile.raster {
264                shapes.for_each_mut(&mut add_to_buffer);
265            }
266        }
267
268        if skipped > 0 {
269            tracing::warn!(
270                skipped,
271                capacity,
272                "tile pattern exceeds its buffer; distant shapes are skipped this frame"
273            );
274        }
275        self.uploaded = buffer.len() as u64;
276        let raw_buffer = bytemuck::cast_slice(buffer.as_slice());
277        queue.write_buffer(&self.view_tiles_buffer.inner, 0, raw_buffer);
278    }
279}

281/// The shown tiles with the loaded descendants their source is fading out drawn over each of
282/// them, as GL JS draws departing tiles over the tile that replaces them until the fade ends;
283/// `None` when nothing under these shapes is departing.
284fn with_departing<T: HasTile>(
285    (availability, source, shapes): (&T, &RasterSourceId, &SourceShapes),
286    zoom: Zoom,
287    world: &World,
288) -> Option<SourceShapes> {
289    let shown: Vec<WorldTileCoords> = match shapes {
290        SourceShapes::SourceEqTarget(shape) => vec![shape.coords()],
291        SourceShapes::Children(shapes) => shapes.iter().map(TileShape::coords).collect(),
292        SourceShapes::Parent(_) | SourceShapes::None => return None,
293    };
294    let departing = world
295        .resources
296        .get::<crate::raster::cross_fade::RasterCrossFade>()?
297        .departing(source.name())?;
298    let mut tiles = Vec::new();
299    let mut any = false;
300    for tile in shown {
301        tiles.push(TileShape::new(tile, zoom));
302        let mut under: Vec<_> = departing
303            .tiles
304            .iter()
305            .copied()
306            .filter(|departed| {
307                departed.z > tile.z
308                    && ancestor_at(*departed, tile.z) == Some(tile)
309                    && availability.has_tile(*departed, world)
310            })
311            .collect();
312        under.sort_by_key(|departed| (departed.z, departed.x, departed.y));
313        any |= !under.is_empty();
314        tiles.extend(
315            under
316                .into_iter()
317                .map(|departed| TileShape::new(departed, zoom).within(tile)),
318        );
319    }
320    any.then_some(SourceShapes::Children(tiles))
321}

323fn raster_shapes<T: HasTile>(
324    sources: &T,
325    coords: WorldTileCoords,
326    covered: Vec<WorldTileCoords>,
327    zoom: Zoom,
328    world: &World,
329    used_parents: &mut HashSet<WorldTileCoords>,
330) -> SourceShapes {
331    let covered = super::coverage::disjoint_tiles(coords, covered);
332    let complete = super::coverage::covers_target(coords, &covered).then(|| covered.clone());
333    if let Some(tiles) = complete {
334        selected_shapes(coords, tiles, zoom, used_parents)
335    } else if covered.is_empty() {
336        match super::coverage::loaded_cover(sources, coords, world) {
337            Some(tiles) => selected_shapes(coords, tiles, zoom, used_parents),
338            None => source_shapes(sources, coords, zoom, world, used_parents),
339        }
340    } else {
341        // What the view needs of this tile is loaded in part: the nearest loaded tile above or
342        // at it stays underneath, and the part's own tiles are drawn over it.
343        let mut tiles = super::coverage::loaded_cover(sources, coords, world).unwrap_or_default();
344        tiles.retain(|tile| !covered.contains(tile));
345        tiles.extend(covered);
346        tiles.sort_by_key(|tile| tile.z);
347        selected_shapes(coords, tiles, zoom, used_parents)
348    }
349}

351/// The loaded shapes nearest to `coords` in the pyramid: the tile itself, complete children,
352/// a parent, or whatever children exist. A parent already standing in for another view tile
353/// is not drawn twice.
354fn source_shapes<T: HasTile>(
355    container: &T,
356    coords: WorldTileCoords,
357    zoom: Zoom,
358    world: &World,
359    used_parents: &mut HashSet<WorldTileCoords>,
360) -> SourceShapes {
361    match super::coverage::loaded_cover(container, coords, world)
362        .or_else(|| container.get_available_children(coords, world, CHILDREN_SEARCH_DEPTH))
363    {
364        Some(tiles) => selected_shapes(coords, tiles, zoom, used_parents),
365        None => SourceShapes::None,
366    }
367}
368
369fn selected_shapes(
370    coords: WorldTileCoords,
371    tiles: Vec<WorldTileCoords>,
372    zoom: Zoom,
373    used_parents: &mut HashSet<WorldTileCoords>,
374) -> SourceShapes {
375    if let [parent] = tiles.as_slice() {
376        // Adjacent view tiles sharing an ancestor must not blend that ancestor repeatedly.
377        if parent.z < coords.z && !used_parents.insert(*parent) {
378            return SourceShapes::None;
379        }
380    }
381    shapes_from_tiles(coords, tiles, zoom)
382}

384/// Shapes for covering tiles that overlap `coords`: the tile itself, its descendants, or the
385/// one ancestor standing in for it.
386fn shapes_from_tiles(
387    coords: WorldTileCoords,
388    tiles: Vec<WorldTileCoords>,
389    zoom: Zoom,
390) -> SourceShapes {
391    match tiles.as_slice() {
392        [tile] if *tile == coords => SourceShapes::SourceEqTarget(TileShape::new(coords, zoom)),
393        [tile] if tile.z < coords.z => SourceShapes::Parent(TileShape::new(*tile, zoom)),
394        _ => SourceShapes::Children(
395            tiles
396                .into_iter()
397                .map(|tile| TileShape::new(tile, zoom))
398                .collect(),
399        ),
400    }
401}

403/// The tiles of a covering that overlap a view tile: the tile itself, its descendants, or the
404/// ancestor standing in for it, as GL JS `getTerrainCoords` pairs them.
405pub fn covering_shapes_for(
406    target: WorldTileCoords,
407    covering: &[WorldTileCoords],
408) -> Vec<WorldTileCoords> {
409    covering
410        .iter()
411        .copied()
412        .filter(|source| overlaps(target, *source))
413        .collect()
414}
415
416fn overlaps(target: WorldTileCoords, source: WorldTileCoords) -> bool {
417    if source.z >= target.z {
418        ancestor_at(source, target.z) == Some(target)
419    } else {
420        ancestor_at(target, source.z) == Some(source)
421    }
422}
423
424fn ancestor_at(tile: WorldTileCoords, level: ZoomLevel) -> Option<WorldTileCoords> {
425    let mut current = tile;
426    while current.z > level {
427        current = current.get_parent()?;
428    }
429    Some(current)
430}

432/// The square of its tile a shape draws, in tile coordinates: that of the finer tile it is
433/// drawn within, or the whole tile.
434fn clip_region(shape: &TileShape) -> crate::render::shaders::Vec4f32 {
435    let Some(within) = shape.within else {
436        return crate::render::shaders::WHOLE_TILE;
437    };
438    let levels = i32::from(u8::from(within.z)) - i32::from(u8::from(shape.coords.z));
439    if levels <= 0 || within.x >> levels != shape.coords.x || within.y >> levels != shape.coords.y {
440        return crate::render::shaders::WHOLE_TILE;
441    }
442    let size = crate::coords::EXTENT / f64::from(1_u32 << levels);
443    let x = f64::from(within.x - (shape.coords.x << levels)) * size;
444    let y = f64::from(within.y - (shape.coords.y << levels)) * size;
445    [x as f32, y as f32, (x + size) as f32, (y + size) as f32]
446}
