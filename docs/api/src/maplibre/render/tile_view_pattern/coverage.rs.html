1//! Complete, disjoint source coverage shared by screen and terrain rendering.
2
3use super::{HasTile, COMPLETE_CHILDREN_SEARCH_DEPTH};
4use crate::{coords::WorldTileCoords, projection::tile_covering::covers, tcs::world::World};

6/// The tile itself, complete descendants, or its nearest loaded ancestor.
7pub(crate) fn loaded_cover(
8    sources: &impl HasTile,
9    coords: WorldTileCoords,
10    world: &World,
11) -> Option<Vec<WorldTileCoords>> {
12    if sources.has_tile(coords, world) {
13        Some(vec![coords])
14    } else {
15        sources
16            .get_complete_children(coords, world, COMPLETE_CHILDREN_SEARCH_DEPTH)
17            .or_else(|| {
18                sources
19                    .get_available_parent(coords, world)
20                    .map(|parent| vec![parent])
21            })
22    }
23}

25/// [`loaded_cover`], or else whichever descendants have loaded: GL JS draws the vector tiles it
26/// has over the parts they cover and leaves the rest empty rather than drawing none of them.
27pub(crate) fn loaded_or_partial_cover(
28    sources: &impl HasTile,
29    coords: WorldTileCoords,
30    world: &World,
31) -> Option<Vec<WorldTileCoords>> {
32    loaded_cover(sources, coords, world).or_else(|| {
33        let partial = loaded_descendants(sources, coords, world, COMPLETE_CHILDREN_SEARCH_DEPTH);
34        (!partial.is_empty()).then_some(partial)
35    })
36}

38/// The loaded tiles under `coords`, the shallowest wherever one is loaded, up to `depth` levels.
39fn loaded_descendants(
40    sources: &impl HasTile,
41    coords: WorldTileCoords,
42    world: &World,
43    depth: usize,
44) -> Vec<WorldTileCoords> {
45    if depth == 0 {
46        return Vec::new();
47    }
48    coords
49        .get_children()
50        .into_iter()
51        .flat_map(|child| {
52            if sources.has_tile(child, world) {
53                vec![child]
54            } else {
55                loaded_descendants(sources, child, world, depth - 1)
56            }
57        })
58        .collect()
59}

61/// A disjoint covering, or no replacement while some part of the target is missing.
62pub(crate) fn complete_cover(
63    target: WorldTileCoords,
64    tiles: Vec<WorldTileCoords>,
65) -> Option<Vec<WorldTileCoords>> {
66    target.build_quad_key()?;
67    let disjoint = disjoint_tiles(target, tiles);
68    covers_target(target, &disjoint).then_some(disjoint)
69}

71pub(super) fn disjoint_tiles(
72    target: WorldTileCoords,
73    mut tiles: Vec<WorldTileCoords>,
74) -> Vec<WorldTileCoords> {
75    tiles.retain(|tile| {
76        tile.build_quad_key().is_some() && (covers(*tile, target) || covers(target, *tile))
77    });
78    tiles.sort_by_key(|tile| tile.z);
79    let mut disjoint = Vec::new();
80    for tile in tiles {
81        // Drawing both an ancestor and a descendant would blend translucent layers twice.
82        if !disjoint.iter().any(|parent| covers(*parent, tile)) {
83            disjoint.push(tile);
84        }
85    }
86    disjoint
87}

89/// Checks the union, so duplicates and unrelated tiles cannot fill a missing quadrant.
90pub(crate) fn covers_target(target: WorldTileCoords, tiles: &[WorldTileCoords]) -> bool {
91    if target.build_quad_key().is_none() {
92        return false;
93    }
94    let tiles: Vec<_> = tiles
95        .iter()
96        .copied()
97        .filter(|tile| tile.build_quad_key().is_some())
98        .collect();
99    covers_subtree(target, &tiles)
100}
101
102fn covers_subtree(target: WorldTileCoords, tiles: &[WorldTileCoords]) -> bool {
103    if tiles.iter().any(|tile| covers(*tile, target)) {
104        return true;
105    }
106    let descendants: Vec<_> = tiles
107        .iter()
108        .copied()
109        .filter(|tile| covers(target, *tile))
110        .collect();
111    // Recursion stops at the deepest supplied tile, including the maximum supported zoom.
112    !descendants.is_empty()
113        && target
114            .get_children()
115            .into_iter()
116            .all(|child| covers_subtree(child, &descendants))
117}
