1//! Frame-to-frame memoization for [`TileViewPattern::generate_pattern`] during sub-tile camera
2//! motion where the visible covering, GPU pools, and loaded tile states did not change.

33/// Caches the most recent [`TileViewPattern`] key across frames so smooth panning or zooming
34/// inside an unchanged covering only updates per-shape zoom transforms in place.
35#[derive(Default)]
36pub struct TileViewPatternMemo {
37    key: Option<PatternMemoKey>,
38    last_zoom_bits: u64,
39}
40
41impl TileViewPatternMemo {
42    /// Returns `true` if `tile_view_pattern` already holds the exact covering/source selection
43    /// for `(view_region, raster_coverings, style_rev, world)`.
44    pub(crate) fn matches(
45        &mut self,
46        view_region: &ViewRegion,
47        raster_coverings: &[(RasterSourceId, Vec<WorldTileCoords>)],
48        style_rev: u64,
49        world: &World,
50    ) -> bool {
51        if has_active_cross_fade(world) {
52            self.invalidate();
53            return false;
54        }
55        let next_key = build_key(view_region, raster_coverings, style_rev, world);
56        if self.key.as_ref() == Some(&next_key) {
57            return true;
58        }
59        self.invalidate();
60        false
61    }
62
63    /// Returns `true` if `zoom` changed since the previous frame and records the new zoom.
64    pub(crate) fn needs_zoom_update(&mut self, zoom: Zoom) -> bool {
65        let zoom_bits = zoom.value().to_bits();
66        if self.last_zoom_bits != zoom_bits {
67            self.last_zoom_bits = zoom_bits;
68            true
69        } else {
70            false
71        }
72    }
73
74    pub(crate) fn record(
75        &mut self,
76        view_region: &ViewRegion,
77        raster_coverings: &[(RasterSourceId, Vec<WorldTileCoords>)],
78        style_rev: u64,
79        zoom: Zoom,
80        world: &World,
81    ) {
82        if has_active_cross_fade(world) {
83            self.invalidate();
84            return;
85        }
86        self.key = Some(build_key(view_region, raster_coverings, style_rev, world));
87        self.last_zoom_bits = zoom.value().to_bits();
88    }
89
90    fn invalidate(&mut self) {
91        self.key = None;
92    }
93}

95impl<Q, B> TileViewPattern<Q, B> {
96    pub(crate) fn is_empty(&self) -> bool {
97        self.view_tiles.is_empty()
98    }
99
100    pub(crate) fn update_zoom_in_place(&mut self, zoom: Zoom) {
101        for view_tile in &mut self.view_tiles {
102            view_tile
103                .vector
104                .for_each_mut(&mut |shape| shape.set_zoom(zoom));
105            for (_, shapes) in &mut view_tile.raster {
106                shapes.for_each_mut(&mut |shape| shape.set_zoom(zoom));
107            }
108        }
109    }
110}

112fn has_active_cross_fade(world: &World) -> bool {
113    #[cfg(feature = "raster")]
114    {
115        world
116            .resources
117            .get::<crate::raster::cross_fade::RasterCrossFade>()
118            .is_some_and(|fade| !fade.is_empty())
119    }
120    #[cfg(not(feature = "raster"))]
121    {
122        let _ = world;
123        false
124    }
125}
126
127fn build_key(
128    view_region: &ViewRegion,
129    raster_coverings: &[(RasterSourceId, Vec<WorldTileCoords>)],
130    style_rev: u64,
131    world: &World,
132) -> PatternMemoKey {
133    let mut vh = FxHasher::default();
134    let mut view_region_len = 0_usize;
135    let mut tv = FxHasher::default();
136    for seen in view_region.copies() {
137        view_region_len += 1;
138        seen.hash(&mut vh);
139        if let Some((canonical, _)) = seen.wrapped() {
140            if let Some(buckets) = world.tiles.query::<&VectorLayerBucketComponent>(canonical) {
141                (
142                    canonical,
143                    buckets.done,
144                    buckets.failed,
145                    buckets.uploaded_stamp,
146                )
147                    .hash(&mut tv);
148            }
149        }
150    }
151    let mut rh = FxHasher::default();
152    for (source, tiles) in raster_coverings {
153        source.hash(&mut rh);
154        tiles.len().hash(&mut rh);
155        for coords in tiles {
156            coords.hash(&mut rh);
157        }
158    }
159    let (vector_pool_rev, vector_pool_eviction_gen, vector_pool_has_capacity) =
160        match world.resources.get::<Eventually<VectorBufferPool>>() {
161            Some(Initialized(pool)) => (
162                pool.revision(),
163                pool.eviction_generation(),
164                pool.has_capacity(),
165            ),
166            _ => (0, 0, false),
167        };
168    #[cfg(feature = "raster")]
169    let raster_mutation_gen = match world
170        .resources
171        .get::<Eventually<crate::raster::resource::RasterResources>>()
172    {
173        Some(Initialized(raster)) => raster.mutation_generation(),
174        _ => 0,
175    };
176    #[cfg(not(feature = "raster"))]
177    let raster_mutation_gen = 0_u64;
178
179    PatternMemoKey {
180        style_rev,
181        view_region_digest: vh.finish(),
182        view_region_len,
183        raster_coverings_digest: rh.finish(),
184        raster_coverings_len: raster_coverings.len(),
185        vector_pool_rev,
186        vector_pool_eviction_gen,
187        vector_pool_has_capacity,
188        vector_arrival_epoch: crate::vector::arrival_epoch(world),
189        view_vector_digest: tv.finish(),
190        tiles_len: world.tiles.tiles.len(),
191        raster_mutation_gen,
192    }
193}
