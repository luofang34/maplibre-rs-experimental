12/// The loaded shapes nearest to `coords` in the pyramid: the tile itself, complete children,
13/// a parent, or whatever children exist. A parent already standing in for another view tile
14/// is not drawn twice.
15pub(super) fn source_shapes<T: HasTile>(
16    container: &T,
17    coords: WorldTileCoords,
18    zoom: Zoom,
19    world: &World,
20    used_parents: &mut FxHashSet<WorldTileCoords>,
21) -> SourceShapes {
22    source_shapes_bounded(container, coords, zoom, world, used_parents, None)
23}
24
25pub(super) fn source_shapes_bounded<T: HasTile>(
26    container: &T,
27    coords: WorldTileCoords,
28    zoom: Zoom,
29    world: &World,
30    used_parents: &mut FxHashSet<WorldTileCoords>,
31    max_zoom: Option<u8>,
32) -> SourceShapes {
33    let max_child_depth = max_zoom.map_or(usize::MAX, |max_z| {
34        usize::from(max_z.saturating_sub(u8::from(coords.z)))
35    });
36    let cover = if container.has_tile(coords, world) {
37        Some(vec![coords])
38    } else {
39        container
40            .get_complete_children(
41                coords,
42                world,
43                COMPLETE_CHILDREN_SEARCH_DEPTH.min(max_child_depth),
44            )
45            .or_else(|| {
46                container
47                    .get_available_parent(coords, world)
48                    .map(|parent| vec![parent])
49            })
50    };
51    match cover.or_else(|| {
52        container
53            .get_available_children(coords, world, CHILDREN_SEARCH_DEPTH.min(max_child_depth))
54            .filter(|tiles| !tiles.is_empty())
55    }) {
56        Some(tiles) if !tiles.is_empty() => selected_shapes(coords, tiles, zoom, used_parents),
57        _ => SourceShapes::None,
58    }
59}

61pub(super) fn selected_shapes(
62    coords: WorldTileCoords,
63    tiles: Vec<WorldTileCoords>,
64    zoom: Zoom,
65    used_parents: &mut FxHashSet<WorldTileCoords>,
66) -> SourceShapes {
67    if let [parent] = tiles.as_slice() {
68        // Adjacent view tiles sharing an ancestor must not blend that ancestor repeatedly.
69        if parent.z < coords.z && !used_parents.insert(*parent) {
70            return SourceShapes::None;
71        }
72    }
73    shapes_from_tiles(coords, tiles, zoom)
74}

76/// Shapes for covering tiles that overlap `coords`: the tile itself, its descendants, or the
77/// one ancestor standing in for it.
78fn shapes_from_tiles(
79    coords: WorldTileCoords,
80    tiles: Vec<WorldTileCoords>,
81    zoom: Zoom,
82) -> SourceShapes {
83    match tiles.as_slice() {
84        [tile] if *tile == coords => SourceShapes::SourceEqTarget(TileShape::new(coords, zoom)),
85        [tile] if tile.z < coords.z => SourceShapes::Parent(TileShape::new(*tile, zoom)),
86        _ => SourceShapes::Children(
87            tiles
88                .into_iter()
89                .map(|tile| TileShape::new(tile, zoom))
90                .collect(),
91        ),
92    }
93}

95/// The tiles of a covering that overlap a view tile: the tile itself, its descendants, or the
96/// ancestor standing in for it, as GL JS `getTerrainCoords` pairs them.
97pub fn covering_shapes_for(
98    target: WorldTileCoords,
99    covering: &[WorldTileCoords],
100) -> Vec<WorldTileCoords> {
101    covering
102        .iter()
103        .copied()
104        .filter(|source| overlaps(target, *source))
105        .collect()
106}

108fn overlaps(target: WorldTileCoords, source: WorldTileCoords) -> bool {
109    if source.z >= target.z {
110        ancestor_at(source, target.z) == Some(target)
111    } else {
112        ancestor_at(target, source.z) == Some(source)
113    }
114}

116pub(super) fn ancestor_at(tile: WorldTileCoords, level: ZoomLevel) -> Option<WorldTileCoords> {
117    let mut current = tile;
118    while current.z > level {
119        current = current.get_parent()?;
120    }
121    Some(current)
122}

124/// The square of its tile a shape draws, in tile coordinates: that of the finer tile it is
125/// drawn within, or the whole tile.
126pub(super) fn clip_region(shape: &TileShape) -> crate::render::shaders::Vec4f32 {
127    let Some(within) = shape.within else {
128        return crate::render::shaders::WHOLE_TILE;
129    };
130    let levels = i32::from(u8::from(within.z)) - i32::from(u8::from(shape.coords.z));
131    if levels <= 0 || within.x >> levels != shape.coords.x || within.y >> levels != shape.coords.y {
132        return crate::render::shaders::WHOLE_TILE;
133    }
134    let size = crate::coords::EXTENT / f64::from(1_u32 << levels);
135    let x = f64::from(within.x - (shape.coords.x << levels)) * size;
136    let y = f64::from(within.y - (shape.coords.y << levels)) * size;
137    [x as f32, y as f32, (x + size) as f32, (y + size) as f32]
138}
