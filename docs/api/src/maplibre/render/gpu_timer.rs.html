1//! The GPU time of the main pass, measured with timestamp queries where the device has them.
2//!
3//! A frame writes timestamps at the start and end of its main pass, resolves them into a buffer
4//! and maps it after submission. A later frame reads the mapped values; frames in between leave
5//! the queries alone, so one measurement is in flight at a time.
6
7use std::{
8    sync::{
9        atomic::{AtomicU8, Ordering},
10        Arc,
11    },
12    time::Duration,
13};
14
15/// Bytes of the two resolved timestamps.
16const RESOLVED_BYTES: u64 = 2 * std::mem::size_of::<u64>() as u64;
17
18const IDLE: u8 = 0;
19const WRITTEN: u8 = 1;
20const MAPPING: u8 = 2;
21const MAPPED: u8 = 3;
22
23/// The timer of a renderer, absent on a device that cannot write timestamps.
24#[derive(Default)]
25pub struct GpuTimerSlot(pub Option<GpuTimer>);
26
27/// Timestamp queries around the main pass and the buffers their values travel through.
28pub struct GpuTimer {
29    queries: wgpu::QuerySet,
30    resolved: wgpu::Buffer,
31    readback: wgpu::Buffer,
32    state: Arc<AtomicU8>,
33    /// The host frame whose passes the measurement in flight timed.
34    frame: std::sync::atomic::AtomicU64,
35}
36
37impl GpuTimer {
38    /// A timer for a device that can write timestamps in render passes, or `None`.
39    pub fn new(device: &wgpu::Device) -> Option<Self> {
40        if !device.features().contains(wgpu::Features::TIMESTAMP_QUERY) {
41            return None;
42        }
43        let buffer = |usage| {
44            device.create_buffer(&wgpu::BufferDescriptor {
45                label: Some("main pass timestamps"),
46                size: RESOLVED_BYTES,
47                usage,
48                mapped_at_creation: false,
49            })
50        };
51        Some(Self {
52            queries: device.create_query_set(&wgpu::QuerySetDescriptor {
53                label: Some("main pass timestamps"),
54                ty: wgpu::QueryType::Timestamp,
55                count: 2,
56            }),
57            resolved: buffer(wgpu::BufferUsages::QUERY_RESOLVE | wgpu::BufferUsages::COPY_SRC),
58            readback: buffer(wgpu::BufferUsages::MAP_READ | wgpu::BufferUsages::COPY_DST),
59            state: Arc::new(AtomicU8::new(IDLE)),
60            frame: std::sync::atomic::AtomicU64::new(0),
61        })
62    }
63
64    /// Names the host frame that the next measurement will time, unless one is in flight.
65    pub fn tag(&self, frame: u64) {
66        if self.state.load(Ordering::Acquire) == IDLE {
67            self.frame.store(frame, Ordering::Release);
68        }
69    }
70
71    /// The host frame of the measurement [`Self::take`] returns.
72    pub fn frame(&self) -> u64 {
73        self.frame.load(Ordering::Acquire)
74    }
75
76    /// The timestamp writes for this frame's main pass, unless an earlier measurement is still
77    /// on its way back.
78    pub fn pass_writes(&self) -> Option<wgpu::RenderPassTimestampWrites<'_>> {
79        self.span_writes(true, true)
80    }
81
82    /// The timestamp writes of one pass in a run of passes timed together: the first pass
83    /// writes the start, claiming the timer unless a measurement is still on its way back, and
84    /// the last writes the end.
85    pub fn span_writes(
86        &self,
87        first: bool,
88        last: bool,
89    ) -> Option<wgpu::RenderPassTimestampWrites<'_>> {
90        if first {
91            self.state
92                .compare_exchange(IDLE, WRITTEN, Ordering::AcqRel, Ordering::Acquire)
93                .ok()?;
94        } else if self.state.load(Ordering::Acquire) != WRITTEN || !last {
95            return None;
96        }
97        Some(wgpu::RenderPassTimestampWrites {
98            query_set: &self.queries,
99            beginning_of_pass_write_index: first.then_some(0),
100            end_of_pass_write_index: last.then_some(1),
101        })
102    }
103
104    /// Copies this frame's timestamps out, after its main pass has been encoded.
105    pub fn resolve(&self, encoder: &mut wgpu::CommandEncoder) {
106        if self.state.load(Ordering::Acquire) != WRITTEN {
107            return;
108        }
109        encoder.resolve_query_set(&self.queries, 0..2, &self.resolved, 0);
110        encoder.copy_buffer_to_buffer(&self.resolved, 0, &self.readback, 0, RESOLVED_BYTES);
111    }
112
113    /// Asks for the copied timestamps once the frame has been submitted.
114    pub fn after_submit(&self) {
115        if self
116            .state
117            .compare_exchange(WRITTEN, MAPPING, Ordering::AcqRel, Ordering::Acquire)
118            .is_err()
119        {
120            return;
121        }
122        let state = self.state.clone();
123        self.readback
124            .map_async(wgpu::MapMode::Read, .., move |result| {
125                state.store(
126                    if result.is_ok() { MAPPED } else { IDLE },
127                    Ordering::Release,
128                );
129            });
130    }
131
132    /// The main pass time of the last measured frame, once its timestamps are back.
133    pub fn take(&self, device: &wgpu::Device, period_nanos: f32) -> Option<Duration> {
134        if self.state.load(Ordering::Acquire) == MAPPING {
135            // Drives the mapping callback without waiting for the GPU.
136            if let Err(error) = device.poll(wgpu::PollType::Poll) {
137                tracing::debug!(%error, "polling for GPU timestamps failed");
138            }
139        }
140        if self.state.load(Ordering::Acquire) != MAPPED {
141            return None;
142        }
143        // A pass that draws nothing may never reach the stage that writes its end on GPUs that
144        // sample at stage boundaries, so an end not after the start measures nothing.
145        let elapsed = self.readback.get_mapped_range(..).ok().and_then(|bytes| {
146            let start = u64::from_le_bytes(bytes[..8].try_into().unwrap_or_default());
147            let end = u64::from_le_bytes(bytes[8..16].try_into().unwrap_or_default());
148            (end > start).then(|| {
149                Duration::from_nanos(((end - start) as f64 * f64::from(period_nanos)) as u64)
150            })
151        });
152        self.readback.unmap();
153        self.state.store(IDLE, Ordering::Release);
154        elapsed
155    }
156}
