1//! A renderer on the host's own GPU objects, drawing into a texture the host samples.
2//!
3//! A host that composites the map itself hands over the wgpu instance, adapter, device and
4//! queue it renders with. The map draws each frame into its offscreen texture, in the format
5//! and sample count the host asks for and resolved to one sample, and the host binds that
6//! texture in its own passes: nothing is read back to the CPU. The handles are reference
7//! counted, so the host keeps using its own, and any texture it still holds, after the map
8//! resizes or is dropped.
9
10use super::{
11    resource::{share_gpu, Surface},
12    settings::{RendererSettings, WgpuSettings},
13    upload_queue, RenderResources, Renderer,
14};
15use crate::window::MapWindow;
16
17/// The GPU objects a host renders with.
18#[derive(Clone, Debug)]
19pub struct HostGpu {
20    /// The instance the adapter came from.
21    pub instance: wgpu::Instance,
22    /// The adapter the device was requested from.
23    pub adapter: wgpu::Adapter,
24    /// The device both the host and the map draw with.
25    pub device: wgpu::Device,
26    /// The queue both submit to.
27    pub queue: wgpu::Queue,
28}
29
30impl Renderer {
31    /// A renderer drawing with the host's device into an offscreen texture of `window`'s size,
32    /// in `settings.texture_format` (RGBA8 when unset) at `settings.msaa` samples where the
33    /// format allows them.
34    pub fn on_host_gpu<MW: MapWindow>(
35        window: &MW,
36        gpu: HostGpu,
37        settings: RendererSettings,
38    ) -> Self {
39        let HostGpu {
40            instance,
41            adapter,
42            device,
43            queue,
44        } = gpu;
45        // The device is already made, so its features and limits are what the map gets.
46        let wgpu_settings = WgpuSettings {
47            features: device.features(),
48            limits: device.limits(),
49            ..Default::default()
50        };
51        let settings = settings
52            .with_float_depth_if_supported(device.features())
53            .with_backend_msaa(adapter.get_info().backend);
54        let surface = Surface::from_image(&device, &adapter, window, &settings);
55        Self {
56            instance,
57            device: share_gpu(device),
58            queue: upload_queue::UploadQueue::new(queue),
59            adapter,
60            wgpu_settings,
61            settings,
62            resources: RenderResources::new(surface),
63            render_graph: Default::default(),
64        }
65    }
66}
