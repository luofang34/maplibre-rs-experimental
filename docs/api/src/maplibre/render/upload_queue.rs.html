1//! The renderer's queue, counting the bytes each frame writes to buffers and textures, and
2//! coalescing small aligned buffer writes into one per-frame staging upload.
3//!
4//! Bytes reach the GPU through [`UploadQueue::write_buffer`], [`UploadQueue::write_texture`],
5//! [`UploadQueue::create_buffer_init`] and [`UploadQueue::create_texture_with_data`]. The queue
6//! is not reachable as a [`wgpu::Queue`] except through [`UploadQueue::inner`], and the
7//! crate denies the uncounted wgpu methods that its clippy.toml lists, so clippy rejects an
8//! upload that skips the count.
9
10use std::sync::{
11    atomic::{AtomicU64, Ordering},
12    Mutex,
13};
14
15use wgpu::util::DeviceExt;
16
17/// Maximum single `write_buffer` payload (exclusive) staged into the per-frame coalescing belt.
18const MAX_STAGED_WRITE_BYTES: usize = 65_536;
19
20/// Minimum persistent GPU staging buffer capacity (`64 KiB`).
21const MIN_STAGING_BUFFER_CAPACITY: u64 = 64 * 1024;
22
23#[derive(Debug)]
24struct StagedCopy {
25    dst: wgpu::Buffer,
26    src_offset: u64,
27    dst_offset: u64,
28    size: u64,
29}
30
31#[derive(Debug, Default)]
32struct StagingBeltState {
33    payload: Vec<u8>,
34    copies: Vec<StagedCopy>,
35    buffer: Option<(wgpu::Buffer, u64)>,
36}
37
38impl Drop for StagingBeltState {
39    fn drop(&mut self) {
40        if let Some((buf, _)) = self.buffer.take() {
41            buf.destroy();
42        }
43    }
44}
45
46/// A [`wgpu::Queue`] that counts the bytes written through it and optionally coalesces small
47/// 4-byte-aligned buffer writes into a single per-frame staging upload.
48#[derive(Debug)]
49pub struct UploadQueue {
50    queue: wgpu::Queue,
51    written: AtomicU64,
52    staging_enabled: bool,
53    staging: Mutex<StagingBeltState>,
54}
55
56impl UploadQueue {
57    /// Wraps a queue without write coalescing (used for standalone/custom temporary queues).

61            written: AtomicU64::new(0),
62            staging_enabled: false,
63            staging: Mutex::new(StagingBeltState::default()),

77    /// Writes `data` into `buffer` at `offset`, counting its bytes.
78    ///
79    /// When staging is enabled and `data` is `< 64 KiB` with 4-byte-aligned `offset` and length,
80    /// the write is appended into CPU staging memory and flushed in one `writeBuffer` + GPU
81    /// `copy_buffer_to_buffer` command buffer right before frame submission.
82    pub fn write_buffer(&self, buffer: &wgpu::Buffer, offset: wgpu::BufferAddress, data: &[u8]) {
83        self.written.fetch_add(data.len() as u64, Ordering::Relaxed);
84        if data.is_empty() {
85            return;
86        }
87        let size = data.len() as u64;
88        let can_stage = self.staging_enabled
89            && data.len() < MAX_STAGED_WRITE_BYTES
90            && offset.is_multiple_of(wgpu::COPY_BUFFER_ALIGNMENT)
91            && size.is_multiple_of(wgpu::COPY_BUFFER_ALIGNMENT);
92
93        if can_stage {
94            let mut staging = crate::platform::lock_unpoisoned(&self.staging);
95            if let Some(last) = staging.copies.last() {
96                if &last.dst == buffer && last.dst_offset == offset && last.size == size {
97                    let start = last.src_offset as usize;
98                    staging.payload[start..start + data.len()].copy_from_slice(data);
99                    return;
100                }
101            }
102            let src_offset = staging.payload.len() as u64;
103            staging.payload.extend_from_slice(data);
104            staging.copies.push(StagedCopy {
105                dst: buffer.clone(),
106                src_offset,
107                dst_offset: offset,
108                size,
109            });
110            return;
111        }
112
113        if self.staging_enabled {
114            self.flush_staged_for_buffer(buffer);
115        }
116        #[allow(clippy::disallowed_methods)] // The counted path itself.
117        self.queue.write_buffer(buffer, offset, data);
118    }

120    /// Cancels any pending staged writes targeting `buffer` before the caller destroys it.
121    pub fn cancel_staged_buffer(&self, buffer: &wgpu::Buffer) {
122        if !self.staging_enabled {
123            return;
124        }
125        let mut staging = crate::platform::lock_unpoisoned(&self.staging);
126        staging.copies.retain(|copy| &copy.dst != buffer);
127        if staging.copies.is_empty() {
128            staging.payload.clear();
129        }
130    }

132    /// Uploads all staged `< 64 KiB` writes via a single `wgpu::Queue::write_buffer` into the
133    /// persistent staging buffer and returns a command buffer encoding the `copy_buffer_to_buffer`
134    /// transfers to submit ahead of the frame's render passes.
135    pub fn flush_staged(&self, device: &wgpu::Device) -> Option<wgpu::CommandBuffer> {
136        if !self.staging_enabled {
137            return None;
138        }
139        let mut staging = crate::platform::lock_unpoisoned(&self.staging);
140        if staging.copies.is_empty() {
141            staging.payload.clear();
142            return None;
143        }
144
145        let payload_bytes = staging.payload.len() as u64;
146        let max_buf = device.limits().max_buffer_size;
147        if payload_bytes > max_buf {
148            let StagingBeltState {
149                payload, copies, ..
150            } = &mut *staging;
151            for copy in copies.drain(..) {
152                let start = copy.src_offset as usize;
153                let end = start + copy.size as usize;
154                #[allow(clippy::disallowed_methods)]
155                self.queue
156                    .write_buffer(&copy.dst, copy.dst_offset, &payload[start..end]);
157            }
158            payload.clear();
159            return None;
160        }
161
162        let need_alloc = staging
163            .buffer
164            .as_ref()
165            .is_none_or(|(_, cap)| *cap < payload_bytes);
166        if need_alloc {
167            let next_cap = payload_bytes
168                .max(MIN_STAGING_BUFFER_CAPACITY)
169                .checked_next_power_of_two()
170                .unwrap_or(max_buf)
171                .min(max_buf);
172            let buf = device.create_buffer(&wgpu::BufferDescriptor {
173                label: Some("upload_queue_staging_buffer"),
174                size: next_cap,
175                usage: wgpu::BufferUsages::COPY_SRC | wgpu::BufferUsages::COPY_DST,
176                mapped_at_creation: false,
177            });
178            if let Some((old_buf, _)) = staging.buffer.replace((buf, next_cap)) {
179                old_buf.destroy();
180            }
181        }
182
183        let (staging_buf, _) = staging.buffer.as_ref()?;
184        #[allow(clippy::disallowed_methods)]
185        self.queue.write_buffer(staging_buf, 0, &staging.payload);
186
187        let mut encoder = device.create_command_encoder(&wgpu::CommandEncoderDescriptor {
188            label: Some("upload_queue_staging_encoder"),
189        });
190        let StagingBeltState {
191            payload,
192            copies,
193            buffer,
194        } = &mut *staging;
195        if let Some((staging_buf, _)) = buffer.as_ref() {
196            for copy in copies.drain(..) {
197                encoder.copy_buffer_to_buffer(
198                    staging_buf,
199                    copy.src_offset,
200                    &copy.dst,
201                    copy.dst_offset,
202                    copy.size,
203                );
204            }
205        }
206        payload.clear();
207        Some(encoder.finish())
208    }

210    fn flush_staged_for_buffer(&self, target: &wgpu::Buffer) {
211        let mut staging = crate::platform::lock_unpoisoned(&self.staging);
212        if !staging.copies.iter().any(|copy| &copy.dst == target) {
213            return;
214        }
215        let StagingBeltState {
216            payload, copies, ..
217        } = &mut *staging;
218        copies.retain(|copy| {
219            if &copy.dst == target {
220                let start = copy.src_offset as usize;
221                let end = start + copy.size as usize;
222                #[allow(clippy::disallowed_methods)]
223                self.queue
224                    .write_buffer(&copy.dst, copy.dst_offset, &payload[start..end]);
225                false
226            } else {
227                true
228            }
229        });
230        if copies.is_empty() {
231            payload.clear();
232        }
233    }

235    /// Writes `data` into a texture, counting its bytes.
236    pub fn write_texture(
237        &self,
238        texture: wgpu::TexelCopyTextureInfo<'_>,
239        data: &[u8],
240        layout: wgpu::TexelCopyBufferLayout,
241        size: wgpu::Extent3d,
242    ) {
243        self.written.fetch_add(data.len() as u64, Ordering::Relaxed);
244        #[allow(clippy::disallowed_methods)] // The counted path itself.
245        self.queue.write_texture(texture, data, layout, size);
246    }

273    /// Submits command buffers, flushing any remaining direct-submission staged buffer writes first.
274    pub fn submit<I: IntoIterator<Item = wgpu::CommandBuffer>>(
275        &self,
276        command_buffers: I,
277    ) -> wgpu::SubmissionIndex {
278        if self.staging_enabled {
279            let mut staging = crate::platform::lock_unpoisoned(&self.staging);
280            if !staging.copies.is_empty() {
281                let StagingBeltState {
282                    payload, copies, ..
283                } = &mut *staging;
284                for copy in copies.drain(..) {
285                    let start = copy.src_offset as usize;
286                    let end = start + copy.size as usize;
287                    #[allow(clippy::disallowed_methods)]
288                    self.queue
289                        .write_buffer(&copy.dst, copy.dst_offset, &payload[start..end]);
290                }
291                payload.clear();
292            }
293        }
294        self.queue.submit(command_buffers)
295    }
