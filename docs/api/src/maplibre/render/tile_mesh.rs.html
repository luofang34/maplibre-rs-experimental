17/// 32-bit float GPU vertex position uploaded for [`GpuTileMesh`] draws.
18///
19/// Uploading 32-bit float `[x, y]` (`Float32x2`, 8-byte stride) with `Uint32` indices via
20/// `COPY_DST` avoids sub-aligned 16-bit vertex/index stream corruption on mobile Vulkan drivers.
21#[repr(C)]
22#[derive(Clone, Copy, Debug, PartialEq, Pod, Zeroable)]
23pub struct GpuTileMeshVertex {
24    /// Tile-space `[x, y]` coordinates (`0.0..=4096.0` on body, `-32768.0`/`32767.0` on poles).
25    pub position: [f32; 2],
26}
27
28impl GpuTileMeshVertex {
29    /// Vertex buffer layout at `@location(0)` (`Float32x2`, stride 8).
30    pub fn vertex_buffer_layout() -> VertexBufferLayout {
31        VertexBufferLayout {
32            array_stride: std::mem::size_of::<Self>() as u64,
33            step_mode: wgpu::VertexStepMode::Vertex,
34            attributes: vec![wgpu::VertexAttribute {
35                offset: 0,
36                format: wgpu::VertexFormat::Float32x2,
37                shader_location: 0,
38            }],
39        }
40    }
41}

90/// Failure while preparing a GPU tile mesh.
91#[derive(Debug, Error)]
92pub enum GpuTileMeshError {
93    /// CPU mesh generation failed.
94    #[error("failed to generate tile mesh")]
95    Generate {
96        /// Underlying mesh error.
97        #[source]
98        source: TileMeshError,
99    },
100    /// Index count cannot be represented by a WebGPU draw range.
101    #[error("tile mesh index count exceeds u32")]
102    IndexCount {
103        /// Failed integer conversion.
104        #[source]
105        source: TryFromIntError,
106    },
107}

109/// Lazily populated GPU cache keyed by subdivision, borders, and pole extensions.
110#[derive(Default)]
111pub struct GlobeTileMeshCache {
112    meshes: HashMap<TileMeshKey, GpuTileMesh>,
113    scratch: TileMeshScratch,
114    globe_active: bool,
115}
116
117impl GlobeTileMeshCache {
118    /// Creates a new empty tile mesh cache.
119    pub fn new() -> Self {
120        Self::default()
121    }
122
123    /// Updates whether spherical globe projection (`transition > 0.0`) is active this frame.
124    ///
125    /// Flat Mercator frames (`!globe_active`) reuse a minimal `1x1` quad (`granularity = 1`, no
126    /// pole caps) instead of uploading dense `128x128` globe grids.
127    pub fn set_globe_active(&mut self, globe_active: bool) {
128        self.globe_active = globe_active;
129    }
130
131    /// Returns a reference to the reusable scratch buffer.
132    pub fn scratch(&self) -> &TileMeshScratch {
133        &self.scratch
134    }
135
136    fn key_for(
137        &self,
138        coords: WorldTileCoords,
139        usage: TileMeshUsage,
140        generate_borders: bool,
141    ) -> TileMeshKey {
142        if self.globe_active {
143            mesh_key(coords, usage, generate_borders)
144        } else {
145            flat_mesh_key(generate_borders)
146        }
147    }
148
149    /// Creates the mesh variant needed by a draw if it is not cached.
150    pub fn prepare(
151        &mut self,
152        device: &wgpu::Device,
153        queue: &crate::render::upload_queue::UploadQueue,
154        coords: WorldTileCoords,
155        usage: TileMeshUsage,
156        generate_borders: bool,
157    ) -> Result<(), GpuTileMeshError> {
158        let key = self.key_for(coords, usage, generate_borders);
159        if self.meshes.contains_key(&key) {
160            return Ok(());
161        }
162        let views = create_tile_mesh_into(
163            &mut self.scratch,
164            TileMeshOptions {
165                granularity: key.granularity,
166                generate_borders: key.generate_borders,
167                extend_to_north_pole: key.extend_to_north_pole,
168                extend_to_south_pole: key.extend_to_south_pole,
169            },
170            TileIndexType::U32,
171        )
172        .map_err(|source| GpuTileMeshError::Generate { source })?;
173        let index_count = u32::try_from(views.indices.len())
174            .map_err(|source| GpuTileMeshError::IndexCount { source })?;
175        let gpu_vertices: Vec<GpuTileMeshVertex> = views
176            .vertices
177            .iter()
178            .map(|v| GpuTileMeshVertex {
179                position: [f32::from(v.x), f32::from(v.y)],
180            })
181            .collect();
182        let vertex_buffer = upload_buffer(
183            device,
184            queue,
185            "globe tile mesh vertices",
186            wgpu::BufferUsages::VERTEX,
187            bytemuck::cast_slice(&gpu_vertices),
188        );
189        let (index_buffer, index_format) = upload_indices(device, queue, views.indices);
190        self.meshes.insert(
191            key,
192            GpuTileMesh {
193                vertex_buffer,
194                index_buffer,
195                index_format,
196                index_count,
197            },
198        );
199        Ok(())
200    }
201
202    /// Returns a previously prepared mesh variant.
203    pub fn get(
204        &self,
205        coords: WorldTileCoords,
206        usage: TileMeshUsage,
207        generate_borders: bool,
208    ) -> Option<&GpuTileMesh> {
209        self.meshes
210            .get(&self.key_for(coords, usage, generate_borders))
211    }
212}

214fn upload_buffer(
215    device: &wgpu::Device,
216    queue: &crate::render::upload_queue::UploadQueue,
217    label: &'static str,
218    usage: wgpu::BufferUsages,
219    bytes: &[u8],
220) -> wgpu::Buffer {
221    let size = (bytes.len().max(16).next_multiple_of(16)) as wgpu::BufferAddress;
222    let buffer = device.create_buffer(&wgpu::BufferDescriptor {
223        label: Some(label),
224        size,
225        usage: usage | wgpu::BufferUsages::COPY_DST,
226        mapped_at_creation: false,
227    });
228    if !bytes.is_empty() {
229        queue.write_buffer(&buffer, 0, bytes);
230    }
231    buffer
232}

234fn upload_indices(
235    device: &wgpu::Device,
236    queue: &crate::render::upload_queue::UploadQueue,
237    indices: TileMeshIndicesRef<'_>,
238) -> (wgpu::Buffer, wgpu::IndexFormat) {
239    let (bytes, format) = match indices {
240        TileMeshIndicesRef::U16(values) => {
241            (bytemuck::cast_slice(values), wgpu::IndexFormat::Uint16)
242        }
243        TileMeshIndicesRef::U32(values) => {
244            (bytemuck::cast_slice(values), wgpu::IndexFormat::Uint32)
245        }
246    };
247    let buffer = upload_buffer(
248        device,
249        queue,
250        "globe tile mesh indices",
251        wgpu::BufferUsages::INDEX,
252        bytes,
253    );
254    (buffer, format)
255}

257fn flat_mesh_key(generate_borders: bool) -> TileMeshKey {
258    TileMeshKey {
259        granularity: 1,
260        generate_borders,
261        extend_to_north_pole: false,
262        extend_to_south_pole: false,
263    }
264}

266fn mesh_key(coords: WorldTileCoords, usage: TileMeshUsage, generate_borders: bool) -> TileMeshKey {
267    let zoom = u32::from(u8::from(coords.z));
268    let minimum = match usage {
269        TileMeshUsage::Raster => 32,
270        TileMeshUsage::Stencil => 1,
271    };
272    let granularity = 128_u32.checked_shr(zoom).unwrap_or(0).max(minimum);
273    let last_row = i64::from(ZOOM_BOUNDS[usize::from(u8::from(coords.z))]) - 1;
274    TileMeshKey {
275        granularity,
276        generate_borders,
277        extend_to_north_pole: coords.y == 0,
278        extend_to_south_pole: i64::from(coords.y) == last_row,
279    }
280}
