1//! A surface, such as terrain, that the main pass draws between the style's layers.
2
3use crate::{render::tracked_pass::TrackedRenderPass, tcs::world::World};
4
5/// Records a ground surface's draws into the main pass.
6pub type DrawGround = for<'w> fn(&mut TrackedRenderPass<'w>, &'w World);
7
8/// The ground surface a frame draws.
9#[derive(Clone, Copy)]
10pub struct GroundSurface {
11    /// The surface draws before the first layer whose index is above this one, so layers
12    /// beneath it, such as the background, stay beneath it.
13    pub draw_after_layer_index: u32,
14    /// Records the surface.
15    pub draw: DrawGround,
16}
17
18/// The ground surface of the frame, set each frame by the plugin that owns the surface.
19///
20/// While a frame has a ground surface, every layer draws in the main pass, so the surface's
21/// depth covers the layers it hides.
22#[derive(Clone, Copy, Default)]
23pub struct FrameGround(pub Option<GroundSurface>);
