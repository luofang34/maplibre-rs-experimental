7/// Names of the render passes that draw map layers and export their depth.
8pub mod draw_graph {
9    /// Nodes ordered from opaque drawing through 3D scene entities and translucency to depth export.
10    pub mod node {
11        /// Draws tile masks and map-layer items into the frame attachments.
12        pub const MAIN_PASS: &str = "main_pass";
13        /// Canonical 3D scene pass slot for dynamic 3D entities (e.g. glTF aircraft and ground
14        /// vehicles), ordered after `MAIN_PASS` (and `GLOBE_DAYLIGHT`) and before `OVERLAY_PASS`,
15        /// `DEPTH_PROBE_PASS`, `SUN_DEPTH_COPY`, `TRANSLUCENT_PASS`, and `DEPTH_COPY`.
16        pub const SCENE_3D_PASS: &str = "scene_3d_pass";
17        /// Draws translucent items using the attachments populated by the main pass.
18        pub const TRANSLUCENT_PASS: &str = "translucent_pass";
19        /// Copies rendered depth to the optional host depth target.
20        pub const DEPTH_COPY: &str = "depth_copy";
21    }
22}

34/// Registers the frame stages, draw graph, render phases and projection resource slots.
35/// Install before plugins that add systems to these stages or nodes to the draw graph.
36#[derive(Default)]
37pub struct RenderPlugin;
38
39impl<E: Environment> Plugin<E> for RenderPlugin {
40    fn build(
41        &self,
42        schedule: &mut Schedule,
43        _kernel: Rc<Kernel<E>>,
44        world: &mut World,
45        renderer: &mut crate::render::Renderer,
46    ) -> Result<(), crate::plugin::PluginError> {
47        let resources = &mut world.resources;
48
49        add_draw_passes(&mut renderer.render_graph)?;
50
51        // render graph dependency
52        resources.init::<RenderPhase<LayerItem>>();
53        resources.init::<super::tracked_pass::RenderStats>();
54        resources.init::<RenderPhase<TileMaskItem>>();
55        resources.init::<RenderPhase<TranslucentItem>>();
56        // tile_view_pattern:
57        resources.insert(Eventually::<WgpuTileViewPattern>::Uninitialized);
58        resources.insert(Eventually::<projection::ProjectionGpuResources>::Uninitialized);
59        resources.init::<tile_mesh::GlobeTileMeshCache>();
60        resources.init::<ViewTileSources>();
61        // masks
62        resources.insert(Eventually::<MaskPipeline>::Uninitialized);
63        resources.insert(Eventually::<DepthCopyPipeline>::Uninitialized);
64        resources.init::<super::ground::FrameGround>();
65        resources
66            .get_or_init_mut::<super::BeforeSubmitHooks>()
67            .register(super::gpu_timer::resolve_before_submit);
68        resources
69            .get_or_init_mut::<super::AfterSubmitCallbacks>()
70            .register(super::gpu_timer::map_after_submit);
71
72        // The frame input comes first: the request systems the plugins add to Request must
73        // already see the frame's view.
74        schedule.add_stage(
75            RenderStageLabel::Input,
76            SystemStage::default()
77                .with_system(frame_input::frame_input_system)
78                .with_system(crate::context::camera_flight_system)
79                .with_system(crate::terrain::elevation::center_target_system)
80                .with_system(resolved_view::resolve_frame_view_system),
81        )?;
82        schedule.add_stage(RenderStageLabel::Request, SystemStage::default())?;
83        schedule.add_stage(
84            RenderStageLabel::Prepare,
85            SystemStage::default().with_system(SystemContainer::new(ResourceSystem)),
86        )?;
87        schedule.add_stage(
88            RenderStageLabel::Queue,
89            SystemStage::default()
90                .with_system(crate::vector::upload_system::upload_system)
91                .with_system(crate::raster::upload_system::upload_system)
92                .with_system(tile_view_pattern_system)
93                .with_system(upload_system),
94        )?;
95        schedule.add_stage(
96            RenderStageLabel::PhaseSort,
97            SystemStage::default().with_system(sort_phase_system),
98        )?;
99        schedule.add_stage(
100            RenderStageLabel::Render,
101            SystemStage::default()
102                .with_system(super::gpu_timer::gpu_timer_system)
103                .with_system(SystemContainer::new(GraphRunnerSystem)),
104        )?;
105        schedule.add_stage(
106            RenderStageLabel::Cleanup,
107            SystemStage::default()
108                .with_system(cleanup_system)
109                .with_system(retention_system),
110        )?;
111        // A frame that fails part-way still clears its render phases, or the next frame
112        // draws its items twice.
113        schedule.run_after_failure(RenderStageLabel::Cleanup)?;
114        Ok(())
115    }
116}

118/// Adds the main, scene, translucent and depth-copy passes in that order.
119fn add_draw_passes(graph: &mut RenderGraph) -> Result<(), RenderGraphError> {
120    graph.add_node(draw_graph::node::MAIN_PASS, MainPassNode::new())?;
121    graph.add_node(draw_graph::node::SCENE_3D_PASS, EmptyNode)?;
122    graph.add_node(
123        draw_graph::node::TRANSLUCENT_PASS,
124        TranslucentPassNode::new(),
125    )?;
126    graph.add_node(draw_graph::node::DEPTH_COPY, DepthCopyNode)?;
127    graph.add_node_edge(draw_graph::node::MAIN_PASS, draw_graph::node::SCENE_3D_PASS)?;
128    graph.add_node_edge(
129        draw_graph::node::SCENE_3D_PASS,
130        draw_graph::node::TRANSLUCENT_PASS,
131    )?;
132    graph.add_node_edge(
133        draw_graph::node::TRANSLUCENT_PASS,
134        draw_graph::node::DEPTH_COPY,
135    )
136}
