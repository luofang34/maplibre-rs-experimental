7/// Labels in the root graph, which drives the draw subgraph after its dependencies.
8pub mod main_graph {
9    /// Input-label namespace; the root graph requires no named external slots.
10    pub mod input {}
11    /// Root nodes used to order work before drawing.
12    pub mod node {
13        /// Empty dependency node that must run before the draw-subgraph driver.
14        pub const MAIN_PASS_DEPENDENCIES: &str = "main_pass_dependencies";
15        /// Driver that invokes the draw subgraph with the frame's render context.
16        pub const MAIN_PASS_DRIVER: &str = "main_pass_driver";
17    }
18}
19
20/// Labels for the subgraph that draws map layers and exports their depth.
21pub mod draw_graph {
22    /// Name used to register and look up this subgraph in the root graph.
23    pub const NAME: &str = "draw";
24    /// Input-label namespace; the draw subgraph declares no external input slots.
25    pub mod input {}
26    /// Nodes ordered from opaque drawing through translucency to depth export.
27    pub mod node {
28        /// Draws tile masks and map-layer items into the frame attachments.
29        pub const MAIN_PASS: &str = "main_pass";
30        /// Draws translucent items using the attachments populated by the main pass.
31        pub const TRANSLUCENT_PASS: &str = "translucent_pass";
32        /// Copies rendered depth to the optional host depth target.
33        pub const DEPTH_COPY: &str = "depth_copy";
34    }
35}

37/// GPU pipeline that writes tile identities into stencil for later clipped draws.
38pub struct MaskPipeline(pub wgpu::RenderPipeline);
39impl Deref for MaskPipeline {
40    type Target = wgpu::RenderPipeline;
41
42    fn deref(&self) -> &Self::Target {
43        &self.0
44    }
45}

47/// Registers the frame stages, draw graph, render phases and projection resource slots.
48/// Install before plugins that add systems to these stages or nodes to the draw graph.
49#[derive(Default)]
50pub struct RenderPlugin;
51
52impl<E: Environment> Plugin<E> for RenderPlugin {
53    fn build(
54        &self,
55        schedule: &mut Schedule,
56        _kernel: Rc<Kernel<E>>,
57        world: &mut World,
58        graph: &mut RenderGraph,
59    ) {
60        let resources = &mut world.resources;
61
62        let mut draw_graph = RenderGraph::default();
63        // Draw nodes
64        draw_graph.add_node(draw_graph::node::MAIN_PASS, MainPassNode::new());
65        // Draw nodes
66        draw_graph.add_node(
67            draw_graph::node::TRANSLUCENT_PASS,
68            TranslucentPassNode::new(),
69        );
70        // Input node
71        let input_node_id = draw_graph.set_input(vec![]);
72        // Edges
73        draw_graph
74            .add_node_edge(input_node_id, draw_graph::node::MAIN_PASS)
75            .expect("main pass or draw node does not exist");
76        draw_graph
77            .add_node_edge(
78                draw_graph::node::MAIN_PASS,
79                draw_graph::node::TRANSLUCENT_PASS,
80            )
81            .expect("main pass or draw node does not exist");
82        draw_graph.add_node(draw_graph::node::DEPTH_COPY, DepthCopyNode);
83        draw_graph
84            .add_node_edge(
85                draw_graph::node::TRANSLUCENT_PASS,
86                draw_graph::node::DEPTH_COPY,
87            )
88            .expect("translucent pass or depth copy node does not exist");
89
90        graph.add_sub_graph(draw_graph::NAME, draw_graph);
91        graph.add_node(main_graph::node::MAIN_PASS_DEPENDENCIES, EmptyNode);
92        graph.add_node(main_graph::node::MAIN_PASS_DRIVER, MainPassDriverNode);
93        graph
94            .add_node_edge(
95                main_graph::node::MAIN_PASS_DEPENDENCIES,
96                main_graph::node::MAIN_PASS_DRIVER,
97            )
98            .expect("main pass driver or dependencies do not exist");
99
100        // render graph dependency
101        resources.init::<RenderPhase<LayerItem>>();
102        resources.init::<super::tracked_pass::RenderStats>();
103        resources.init::<RenderPhase<TileMaskItem>>();
104        resources.init::<RenderPhase<TranslucentItem>>();
105        // tile_view_pattern:
106        resources.insert(Eventually::<WgpuTileViewPattern>::Uninitialized);
107        resources.insert(Eventually::<projection::ProjectionGpuResources>::Uninitialized);
108        resources.init::<tile_mesh::GlobeTileMeshCache>();
109        resources.init::<ViewTileSources>();
110        // masks
111        resources.insert(Eventually::<MaskPipeline>::Uninitialized);
112        resources.insert(Eventually::<DepthCopyPipeline>::Uninitialized);
113
114        // The frame input comes first: the request systems the plugins add to Extract must
115        // already see the frame's view.
116        schedule.add_stage(
117            RenderStageLabel::Extract,
118            SystemStage::default()
119                .with_system(frame_input::frame_input_system)
120                .with_system(crate::terrain::elevation::center_target_system),
121        );
122        schedule.add_stage(
123            RenderStageLabel::Prepare,
124            SystemStage::default().with_system(SystemContainer::new(ResourceSystem)),
125        );
126        schedule.add_stage(
127            RenderStageLabel::Queue,
128            SystemStage::default()
129                .with_system(tile_view_pattern_system)
130                .with_system(upload_system),
131        );
132        schedule.add_stage(
133            RenderStageLabel::PhaseSort,
134            SystemStage::default().with_system(sort_phase_system),
135        );
136        schedule.add_stage(
137            RenderStageLabel::Render,
138            SystemStage::default().with_system(SystemContainer::new(GraphRunnerSystem)),
139        );
140        schedule.add_stage(
141            RenderStageLabel::Cleanup,
142            SystemStage::default()
143                .with_system(cleanup_system)
144                .with_system(retention_system),
145        );
146    }
147}
