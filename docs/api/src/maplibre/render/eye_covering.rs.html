1//! One tile covering for both eyes of a frame.
2//!
3//! Tile selection and content ingestion happen once per stereo frame. Each eye then draws
4//! that selection with its own projection, so a tile cannot change level or content between
5//! the left and right images.

21/// Identifier of a registered viewport slot in a multi-view frame.

25impl ViewSlotId {

31    /// Raw zero-based slot index.

83/// Which eye of a frame the schedule is running for.
84#[derive(Clone, Copy, Debug, PartialEq, Eq)]
85pub struct EyeInFrame {
86    /// Position of the eye in the frame; the first eye selects the tiles.
87    pub index: usize,
88    /// The frame, so a selection is never reused across frames.
89    pub frame: u64,
90}

92impl EyeInFrame {
93    /// Whether this eye must reuse the content selected for an earlier eye of the frame.
94    pub(crate) fn reuses_content(world: &World) -> bool {
95        world
96            .resources
97            .get::<Self>()
98            .is_some_and(|eye| eye.index > 0)
99    }
100}

102/// Tiles per raster source, as [`raster_source_regions`] selects them.
103pub type RasterCoverings = Vec<(RasterSourceId, Vec<WorldTileCoords>)>;
104
105/// The tiles the previous frame drew, per covering, as one immutable snapshot every request
106/// and draw of this frame selects against. Each covering keeps its own history: the terrain
107/// surface, the view and each raster source split the quadtree at different levels, so one
108/// covering's leaves say nothing about where another split.
109#[derive(Clone, Default)]
110pub(crate) struct FrameLodHistory {
111    pub(crate) view: LodHistory,
112    pub(crate) raster: FxHashMap<RasterSourceId, LodHistory>,
113    pub(crate) terrain: LodHistory,
114}
115
116/// Takes the snapshot for the frame about to select tiles. Every camera gets it, not only a
117/// host's eye: a map camera that moves a little each frame would otherwise switch a tile
118/// sitting on a level threshold between its fine and coarse version frame after frame.
119pub(crate) fn snapshot_lod_history(world: &mut World) {
120    if EyeInFrame::reuses_content(world) {
121        return;
122    }
123    let shared = world.resources.get::<SharedCovering>();
124    let history = FrameLodHistory {
125        view: LodHistory::new(
126            shared
127                .and_then(|shared| shared.tiles.as_ref())
128                .map_or(&[], |(tiles, _)| tiles.as_slice()),
129        ),
130        raster: shared
131            .map(|shared| {
132                shared
133                    .raster
134                    .iter()
135                    .map(|(name, tiles)| (name.clone(), LodHistory::new(tiles)))
136                    .collect()
137            })
138            .unwrap_or_default(),
139        terrain: LodHistory::new(crate::terrain::drawn_surface_tiles(world)),
140    };
141    world.resources.insert(history);
142}

144/// The drawn tiles a frame selected: the first eye's, which later eyes of the frame reuse, or
145/// the map camera's own.
146#[derive(Clone, Debug)]
147pub struct SharedCovering {
148    /// The stereo frame whose eyes share the selection; none for a map camera's frame.
149    frame: Option<u64>,
150    host_frame: u64,
151    style_rev: u64,
152    level: ZoomLevel,
153    view_proj_bits: [[u64; 4]; 4],
154    terrain_key: (u64, u64, usize),
155    tiles: Option<(Vec<WorldTileCoords>, ZoomLevel)>,
156    raster: RasterCoverings,
157}
158
159impl SharedCovering {
160    /// Binds a cloned covering snapshot to `frame` for `CoveringPolicy::StereoReuse`.
161    #[cfg(feature = "headless")]
162    pub(crate) fn with_frame(&self, frame: u64) -> Self {
163        Self {
164            frame: Some(frame),
165            ..self.clone()
166        }
167    }
168
169    /// The tiles the first eye selected, or `None` when it selected no region.
170    pub(crate) fn tiles(&self) -> Option<&[WorldTileCoords]> {
171        self.tiles.as_ref().map(|(tiles, _)| tiles.as_slice())
172    }
173
174    fn region(&self) -> Option<ViewRegion> {
175        self.tiles
176            .as_ref()
177            .map(|(tiles, level)| ViewRegion::from_tiles(tiles.clone(), *level, tiles.len()))
178    }
179}

187/// Returns already-selected tight-padding tiles for this host frame when view projection and
188/// terrain state match [`SharedCovering`], avoiding duplicate quadtree walks in Cleanup.
189pub(crate) fn matching_tight_tiles<'a>(
190    world: &'a World,
191    style: &Style,
192    view_state: &ViewState,
193    level: ZoomLevel,
194) -> Option<&'a [WorldTileCoords]> {
195    let eye = world.resources.get::<EyeInFrame>().copied();
196    if let Some(eye) = eye.filter(|e| e.index > 0) {
197        return world
198            .resources
199            .get::<SharedCovering>()
200            .filter(|shared| shared.frame == Some(eye.frame))
201            .map(|shared| {
202                shared
203                    .tiles
204                    .as_ref()
205                    .map_or(&[][..], |(tiles, _)| tiles.as_slice())
206            });
207    }
208    let host_frame = crate::render::frame_signals::host_frame_in_progress(world);
209    let view_proj_bits: [[u64; 4]; 4] =
210        (*AsRef::<[[f64; 4]; 4]>::as_ref(&view_state.view_projection().0))
211            .map(|col| col.map(f64::to_bits));
212    let style_rev = style.revision();
213    let terrain_key = terrain_index_key(world);
214    world
215        .resources
216        .get::<SharedCovering>()
217        .filter(|shared| {
218            shared.frame == eye.map(|e| e.frame)
219                && shared.host_frame == host_frame
220                && shared.style_rev == style_rev
221                && shared.level == level
222                && shared.view_proj_bits == view_proj_bits
223                && shared.terrain_key == terrain_key
224        })
225        .map(|shared| {
226            shared
227                .tiles
228                .as_ref()
229                .map_or(&[][..], |(tiles, _)| tiles.as_slice())
230        })
231}

233pub(crate) fn terrain_index_key(world: &World) -> (u64, u64, usize) {
234    world
235        .resources
236        .get::<crate::terrain::coverage::TerrainCoverageIndex>()
237        .map_or((0, 0, 0), |idx| {
238            (
239                idx.min_elevation().to_bits(),
240                idx.max_elevation().to_bits(),
241                idx.dem_bounds_len(),
242            )
243        })
244}

246/// The tiles the frame draws at `level`, with the raster tiles each raster source draws:
247/// the first eye's selection when the schedule runs for a later eye of the same frame, and
248/// otherwise this view's own, kept as the next frame's level history.
249pub(crate) fn drawn_covering(
250    style: &Style,
251    view_state: &ViewState,
252    world: &mut World,
253    level: ZoomLevel,
254) -> Result<(Option<ViewRegion>, RasterCoverings), ProjectionStateError> {
255    let eye = world.resources.get::<EyeInFrame>().copied();
256    if let Some(eye) = eye.filter(|eye| eye.index > 0) {
257        if let Some(shared) = world
258            .resources
259            .get::<SharedCovering>()
260            .filter(|shared| shared.frame == Some(eye.frame))
261        {
262            return Ok((shared.region(), shared.raster.clone()));
263        }
264    }
265    let host_frame = crate::render::frame_signals::host_frame_in_progress(world);
266    let multi_active = world
267        .resources
268        .get::<MultiViewExtractCovering>()
269        .is_some_and(|m| m.active);
270    let view_proj_bits: [[u64; 4]; 4] =
271        (*AsRef::<[[f64; 4]; 4]>::as_ref(&view_state.view_projection().0))
272            .map(|col| col.map(f64::to_bits));
273    let style_rev = style.revision();
274    let terrain_key = terrain_index_key(world);
275    if !multi_active && eye.is_none_or(|e| e.index == 0) {
276        if let Some(shared) = world.resources.get::<SharedCovering>().filter(|shared| {
277            shared.frame == eye.map(|e| e.frame)
278                && shared.host_frame == host_frame
279                && shared.style_rev == style_rev
280                && shared.level == level
281                && shared.view_proj_bits == view_proj_bits
282                && shared.terrain_key == terrain_key
283        }) {
284            return Ok((shared.region(), shared.raster.clone()));
285        }
286    }
287    let cached_tight_region = if !multi_active && eye.is_none_or(|eye| eye.index == 0) {
288        world
289            .resources
290            .get::<FrameLooseCovering>()
291            .filter(|cache| cache.host_frame == host_frame)
292            .and_then(|cache| cache.view_tiles.as_ref())
293            .filter(|(_, padding, cached_level)| {
294                *padding == ViewStatePadding::Tight && *cached_level == level
295            })
296            .map(|(tiles, _, _)| {
297                tiles
298                    .as_ref()
299                    .map(|tiles| ViewRegion::from_tiles(tiles.clone(), level, tiles.len().max(1)))
300            })
301    } else {
302        None
303    };
304    let region = match cached_tight_region {
305        Some(cached) => cached,
306        None => {
307            view_region_for_projection(style, view_state, world, level, ViewStatePadding::Tight)?
308        }
309    };
310    let cached_raster = if !multi_active && eye.is_none_or(|eye| eye.index == 0) {
311        world
312            .resources
313            .get::<FrameLooseCovering>()
314            .filter(|cache| cache.host_frame == host_frame)
315            .and_then(|cache| cache.raster_regions.as_ref())
316            .filter(|(_, padding, cached_terrain)| {
317                *padding == ViewStatePadding::Tight && *cached_terrain == terrain_key
318            })
319            .map(|(regions, _, _)| regions.clone())
320    } else {
321        None
322    };
323    let raster = match cached_raster {
324        Some(cached) => cached,
325        None => raster_source_regions(style, view_state, world, ViewStatePadding::Tight)?,
326    };
327    if eye.is_none_or(|eye| eye.index == 0) {
328        world.resources.insert(SharedCovering {
329            frame: eye.map(|eye| eye.frame),
330            host_frame,
331            style_rev,
332            level,
333            view_proj_bits,
334            terrain_key,
335            tiles: region
336                .as_ref()
337                .map(|region| (region.iter().collect(), region.zoom_level())),
338            raster: raster.clone(),
339        });
340    }
341    Ok((region, raster))
342}
