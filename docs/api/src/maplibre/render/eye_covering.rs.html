81/// Which eye of a frame the schedule is running for.
82#[derive(Clone, Copy, Debug, PartialEq, Eq)]
83pub struct EyeInFrame {
84    /// Position of the eye in the frame; the first eye selects the tiles.
85    pub index: usize,
86    /// The frame, so a selection is never reused across frames.
87    pub frame: u64,
88}
89
90impl EyeInFrame {
91    /// Whether this eye must reuse the content selected for an earlier eye of the frame.
92    pub(crate) fn reuses_content(world: &World) -> bool {
93        world
94            .resources
95            .get::<Self>()
96            .is_some_and(|eye| eye.index > 0)
97    }
98}

100/// Tiles per raster source, as [`raster_source_regions`] selects them.
101pub type RasterCoverings = Vec<(RasterSourceId, Vec<WorldTileCoords>)>;
102
103/// The tiles the previous frame drew, per covering, as one immutable snapshot every request
104/// and draw of this frame selects against. Each covering keeps its own history: the terrain
105/// surface, the view and each raster source split the quadtree at different levels, so one
106/// covering's leaves say nothing about where another split.
107#[derive(Clone, Default)]
108pub(crate) struct FrameLodHistory {
109    pub(crate) view: LodHistory,
110    pub(crate) raster: FxHashMap<RasterSourceId, LodHistory>,
111    pub(crate) terrain: LodHistory,
112}
113
114/// Takes the snapshot for the frame about to select tiles. Every camera gets it, not only a
115/// host's eye: a map camera that moves a little each frame would otherwise switch a tile
116/// sitting on a level threshold between its fine and coarse version frame after frame.
117pub(crate) fn snapshot_lod_history(world: &mut World) {
118    if EyeInFrame::reuses_content(world) {
119        return;
120    }
121    let shared = world.resources.get::<SharedCovering>();
122    let history = FrameLodHistory {
123        view: LodHistory::new(
124            shared
125                .and_then(|shared| shared.tiles.as_ref())
126                .map_or(&[], |(tiles, _)| tiles.as_slice()),
127        ),
128        raster: shared
129            .map(|shared| {
130                shared
131                    .raster
132                    .iter()
133                    .map(|(name, tiles)| (name.clone(), LodHistory::new(tiles)))
134                    .collect()
135            })
136            .unwrap_or_default(),
137        terrain: LodHistory::new(crate::terrain::drawn_surface_tiles(world)),
138    };
139    world.resources.insert(history);
140}

142/// The drawn tiles a frame selected: the first eye's, which later eyes of the frame reuse, or
143/// the map camera's own.
144#[derive(Clone, Debug)]
145pub struct SharedCovering {
146    /// The stereo frame whose eyes share the selection; none for a map camera's frame.
147    frame: Option<u64>,
148    tiles: Option<(Vec<WorldTileCoords>, ZoomLevel)>,
149    raster: RasterCoverings,
150}
151
152impl SharedCovering {
153    /// Binds a cloned covering snapshot to `frame` for `CoveringPolicy::StereoReuse`.
154    #[cfg(feature = "headless")]
155    pub(crate) fn with_frame(&self, frame: u64) -> Self {
156        Self {
157            frame: Some(frame),
158            tiles: self.tiles.clone(),
159            raster: self.raster.clone(),
160        }
161    }
162
163    /// The tiles the first eye selected, or `None` when it selected no region.
164    pub(crate) fn tiles(&self) -> Option<&[WorldTileCoords]> {
165        self.tiles.as_ref().map(|(tiles, _)| tiles.as_slice())
166    }
167
168    fn region(&self) -> Option<ViewRegion> {
169        self.tiles
170            .as_ref()
171            .map(|(tiles, level)| ViewRegion::from_tiles(tiles.clone(), *level, tiles.len()))
172    }
173}

175/// The tiles the frame's covering selected, each once, or `None` before a covering is selected
176/// or when it selected no region.
177pub(crate) fn drawn_tiles(world: &World) -> Option<&[WorldTileCoords]> {
178    world.resources.get::<SharedCovering>()?.tiles()
179}

181/// The tiles the frame draws at `level`, with the raster tiles each raster source draws:
182/// the first eye's selection when the schedule runs for a later eye of the same frame, and
183/// otherwise this view's own, kept as the next frame's level history.
184pub(crate) fn drawn_covering(
185    style: &Style,
186    view_state: &ViewState,
187    world: &mut World,
188    level: ZoomLevel,
189) -> Result<(Option<ViewRegion>, RasterCoverings), ProjectionStateError> {
190    let eye = world.resources.get::<EyeInFrame>().copied();
191    if let Some(eye) = eye.filter(|eye| eye.index > 0) {
192        if let Some(shared) = world
193            .resources
194            .get::<SharedCovering>()
195            .filter(|shared| shared.frame == Some(eye.frame))
196        {
197            return Ok((shared.region(), shared.raster.clone()));
198        }
199    }
200    let region =
201        view_region_for_projection(style, view_state, world, level, ViewStatePadding::Tight)?;
202    let raster = raster_source_regions(style, view_state, world, ViewStatePadding::Tight)?;
203    if eye.is_none_or(|eye| eye.index == 0) {
204        world.resources.insert(SharedCovering {
205            frame: eye.map(|eye| eye.frame),
206            tiles: region
207                .as_ref()
208                .map(|region| (region.iter().collect(), region.zoom_level())),
209            raster: raster.clone(),
210        });
211    }
212    Ok((region, raster))
213}
