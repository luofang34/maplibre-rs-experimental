1//! Frames for a head-mounted display: a scene the host placed in its world, seen by one or
2//! more eyes, each drawn into a target the host presents.
3//!
4//! The host's world is whatever its tracker reports, in metres. The scene is the map's local
5//! frame: metres east, north and up from an anchor. A placement joins the two with a
6//! rotation, a translation and one uniform scale, so the same map can stand on a table as a
7//! model or surround the viewer at full size, and moving between the two is a change of
8//! placement rather than a change of renderer.
9
10use std::time::Duration;
11
12use cgmath::{Matrix4, SquareMatrix};
13
14use crate::render::{
15    camera::EyeFrustum,
16    view_state::{ExternalAnchor, ExternalView, ViewState},
17};
18
19/// Where the host put the scene in its world.
20#[derive(Clone, Copy, Debug, PartialEq)]
21pub struct ScenePlacement {
22    /// The geographic point and altitude the scene's origin stands for.
23    pub anchor: ExternalAnchor,
24    /// Scene frame to the host's world: a rotation, a translation and one uniform scale, the
25    /// host's metres per scene metre.
26    pub world_from_scene: Matrix4<f64>,
27}
28
29/// Where an eye's image is drawn.
30#[derive(Debug, Default)]
31pub struct EyeTarget {
32    /// Colour target in the renderer's surface format and size; `None` draws into the map's
33    /// own texture.
34    pub color: Option<wgpu::TextureView>,
35    /// A `Depth32Float` texture of the surface size the eye's depth is written to for the
36    /// host's compositor, in the map's convention: one at the near plane, zero at the far
37    /// plane.
38    pub depth: Option<wgpu::TextureView>,
39}
40
41/// One eye of a frame.
42#[derive(Debug)]
43pub struct XrEye {
44    /// The eye's pose in the host's world: eye space, x right, y up and the eye looking along
45    /// negative z, to world.
46    pub world_from_eye: Matrix4<f64>,
47    /// The eye's frustum with clip distances in the host's world units.
48    pub frustum: EyeFrustum,
49    /// Where the eye's image goes.
50    pub target: EyeTarget,
51}
52
53/// A frame for a head-mounted display.
54#[derive(Debug)]
55pub struct XrFrame {
56    /// Draw continuous sky for full immersion; false preserves passthrough around a globe.
57    pub opaque_environment: bool,
58    /// Time since the host started; animated properties read it.
59    pub timestamp: Duration,
60    /// Where the scene stands in the host's world.
61    pub placement: ScenePlacement,
62    /// The eyes to draw, each into its own target.
63    pub eyes: Vec<XrEye>,
64    /// How far beyond each eye's frustum tiles are requested, as a factor on its tangents;
65    /// one requests what the frame shows.
66    pub request_overscan: f64,
67    /// Where the scene will stand when a flight in progress ends; the tiles that frame needs
68    /// are requested now, so they are there when it arrives, and the levels passed on the
69    /// way are not requested at all.
70    pub prefetch: Option<ScenePlacement>,
71}
72
73/// The view a flight is heading for, kept by the frame so tile requests can cover it.
74///
75/// The view is built once per destination, from the eye's pose when the flight begins, and
76/// kept while the destination stands: rebuilding it from every frame's head pose would move
77/// its covering a little each frame and, over a flight, request many times the tiles any one
78/// frame needs.
79#[derive(Default)]
80pub struct PrefetchView {
81    /// The view, or none while no flight is in progress.
82    pub view_state: Option<ViewState>,
83    /// The destination the view was built for.
84    pub placement: Option<ScenePlacement>,
85}
86
87impl ScenePlacement {
88    /// The view an eye at `world_from_eye` has of the scene; `None` when the eye's pose has
89    /// no inverse.
90    pub fn view_from(
91        &self,
92        world_from_eye: Matrix4<f64>,
93        frustum: EyeFrustum,
94    ) -> Option<ExternalView> {
95        let eye_from_world = world_from_eye.invert()?;
96        Some(ExternalView {
97            anchor: self.anchor,
98            view: eye_from_world * self.world_from_scene,
99            frustum,
100        })
101    }
102}

132    #[test]
133    fn a_scaled_placement_puts_the_eye_where_the_scene_sees_it() {
134        // A scene a thousand times smaller than the world, moved a metre up, seen by an eye
135        // two metres above the world origin looking straight down.
136        let placement = ScenePlacement {
137            anchor: ExternalAnchor {
138                position: LatLon::new(47.0, 11.0),
139                altitude_meters: 500.0,
140            },
141            world_from_scene: Matrix4::from_translation(Vector3::new(0.0, 0.0, 1.0))
142                * Matrix4::from_scale(1e-3),
143        };
144        let world_from_eye = Matrix4::from_translation(Vector3::new(0.0, 0.0, 2.0));
145        let frustum = EyeFrustum::symmetric(Rad(1.0), 1.0, 0.05, 50.0);
146
147        let view = placement
148            .view_from(world_from_eye, frustum)
149            .expect("a translation is invertible");
150        let mut view_state = view_state();
151        view_state
152            .set_external_view(view, &ProjectionType::Mercator)
153            .expect("the eye is above the scene");
154
155        let pose = view_state.camera_pose();
156        // One metre above the scene origin in the world is a thousand scene metres.
157        assert!((pose.altitude_meters - 1500.0).abs() < 1e-6, "{pose:?}");
158        assert!(pose.pitch.0.abs() < 1e-6, "{pose:?}");
159        assert!(
160            pose.bearing.0.abs() < 1e-6 && pose.roll.0.abs() < 1e-6,
161            "{pose:?}"
162        );
163        assert!(
164            (pose.position.latitude - 47.0).abs() < 1e-9
165                && (pose.position.longitude - 11.0).abs() < 1e-9,
166            "{pose:?}"
167        );
168        let projection = view_state
169            .external_projection()
170            .expect("an external view is in effect");
171        // Clip distances arrive in world metres and reach the map in its pixels: the near
172        // plane is 50 scene metres away, at the map's pixels per metre.
173        let near = 0.05 / 1e-3 * view_state.pixels_per_meter();
174        let expected = EyeFrustum::symmetric(Rad(1.0), 1.0, near, near * 1000.0).projection();
175        assert!((projection.x.x - expected.x.x).abs() < 1e-9);
176        assert!((projection.w.z - expected.w.z).abs() < 1e-6 * expected.w.z.abs());
177    }

179    #[test]
180    fn a_globe_standing_on_a_table_is_seen_from_where_the_viewer_stands() {
181        // A globe fifteen centimetres in radius, its polar axis vertical (world y) and the
182        // anchor turned towards the viewer at +z, its center a metre ahead of the eye and
183        // thirty centimetres below it.
184        let radius_meters = 6_371_008.8;
185        let radius = 0.15;
186        let scale = radius / radius_meters;
187        let latitude = 47.26_f64.to_radians();
188        let east = Vector3::new(1.0, 0.0, 0.0);
189        let north = Vector3::new(0.0, latitude.cos(), -latitude.sin());
190        let up = Vector3::new(0.0, latitude.sin(), latitude.cos());
191        let rotation = cgmath::Matrix3::from_cols(east, north, up);
192        let center = Vector3::new(0.0, -0.3, -1.0);
193        let world_from_scene = Matrix4::from_translation(center + up * radius)
194            * Matrix4::from(rotation)
195            * Matrix4::from_scale(scale);
196        let placement = ScenePlacement {
197            anchor: ExternalAnchor {
198                position: LatLon::new(47.26, 11.39),
199                altitude_meters: 0.0,
200            },
201            world_from_scene,
202        };
203        let view = placement
204            .view_from(
205                Matrix4::identity(),
206                EyeFrustum::symmetric(Rad(1.2), 16.0 / 9.0, 0.1, 100.0),
207            )
208            .expect("an identity pose is invertible");
209        let mut view_state = view_state();
210        view_state
211            .set_external_view(view, &ProjectionType::Globe)
212            .expect("the eye is beside the globe");
213
214        let eye = view_state
215            .external_globe_eye()
216            .expect("the globe camera takes the eye");
217        // The eye is a metre and a bit from the center, seven radii out.
218        let distance = (center.magnitude()) / radius;
219        assert!(
220            (eye.position.magnitude() - distance).abs() < 1e-6,
221            "{eye:?}"
222        );
223        // The view axis misses the globe, so the map center is the point beneath the eye: the
224        // eye's direction from the globe center, thirty degrees south of the anchor.
225        let toward_eye = -center / center.magnitude();
226        let expected_latitude = 47.26 - toward_eye.dot(up).acos().to_degrees();
227        let beneath_eye = unit_sphere_to_lat_lon(eye.position.normalize());
228        assert!(
229            (beneath_eye.latitude - expected_latitude).abs() < 1e-6,
230            "{beneath_eye:?} vs latitude {expected_latitude}"
231        );
232        assert!(
233            (beneath_eye.longitude - 11.39).abs() < 1e-6,
234            "{beneath_eye:?}"
235        );
236        let map_center = view_state.external_view().anchor.position;
237        assert!(
238            (map_center.latitude - expected_latitude).abs() < 1e-6
239                && (map_center.longitude - 11.39).abs() < 1e-6,
240            "{map_center:?}"
241        );
242        // The zoom follows the eye's distance to that point: six radii out.
243        let zoom_distance = view_state.camera_to_center_distance() / view_state.pixels_per_meter();
244        assert!(
245            (zoom_distance - (distance - 1.0) * radius_meters).abs() < 1.0,
246            "{zoom_distance} metres to the center"
247        );
248    }
