1//! Horizon, fog and near/far clip-plane depth calculations for [`ViewState`].
2
3use std::f64::consts;
4
5use cgmath::{Point2, Rad, Vector2};
6
7use super::{NavigationMode, ViewState};
8use crate::coords::TILE_SIZE;
9
10/// Pitch beyond which the Mercator plane has no usable horizon.
11pub(super) const MAX_MERCATOR_HORIZON_ANGLE: Rad<f64> = Rad(89.25 * consts::PI / 180.0);
12/// Keeps some scene below the camera renderable when terrain dips under it.
13const MIN_RENDER_DISTANCE_BELOW_CAMERA_METERS: f64 = 100.0;
14
15impl ViewState {
16    /// Distance in pixels from the screen center to the Mercator horizon, positive above the
17    /// center, as GL JS `getMercatorHorizon`.
18    pub fn mercator_horizon(&self) -> f64 {
19        let pitch = self.camera.get_pitch().0.abs();
20        self.camera_to_center_distance()
21            * ((consts::FRAC_PI_2 - pitch).tan() * 0.85)
22                .min((MAX_MERCATOR_HORIZON_ANGLE.0 - pitch).tan())
23    }
24
25    /// Returns a view shifted backward toward the camera's ground nadir when the camera is
26    /// pitched (`> 20°`), covering the high-detail foreground strip immediately below the
27    /// bottom edge of the screen so backward edge-scrolling never outruns tile loading.
28    pub fn pitched_nadir_apron(&self) -> Option<Self> {
29        if self.has_external_view() || self.navigation_mode() != NavigationMode::Constrained {
30            return None;
31        }
32        let pitch_deg = self.camera.get_pitch().0.to_degrees().abs();
33        if pitch_deg <= 20.0 {
34            return None;
35        }
36        let center = self.camera.position();
37        let eye = self.eye_position();
38        let to_nadir = Vector2::new(eye.x - center.x, eye.y - center.y);
39        if !to_nadir.x.is_finite()
40            || !to_nadir.y.is_finite()
41            || (to_nadir.x * to_nadir.x + to_nadir.y * to_nadir.y) < 1.0
42        {
43            return None;
44        }
45        let world_size = TILE_SIZE * 2_f64.powf(self.zoom.value());
46        let mut apron = self.clone();
47        let new_x = (center.x + to_nadir.x * 0.68).rem_euclid(world_size);
48        let new_y = (center.y + to_nadir.y * 0.68).clamp(0.0, world_size);
49        apron.camera.move_to(Point2::new(new_x, new_y));
50        Some(apron)
51    }
52
53    /// Distances in pixels the fog depth spans: from the camera's distance to sea level, where
54    /// the map center sits, to the far plane, as the GL JS fog matrix takes them. An external
55    /// eye's fog is fixed by its height instead.
56    pub fn fog_depth_range(&self) -> (f64, f64) {
57        if let Some(range) = self.eye_fog_depth_range() {
58            return range;
59        }
60        let distance = self.camera_to_center_distance();
61        let limited_pitch = self
62            .camera
63            .get_pitch()
64            .0
65            .abs()
66            .min(MAX_MERCATOR_HORIZON_ANGLE.0);
67        let camera_to_sea_level = (distance / 2.0)
68            .max(distance + self.center.elevation * self.pixels_per_meter() / limited_pitch.cos());
69        let (_, far_z) = self.depth_range(self.center_offset());
70        (camera_to_sea_level, far_z)
71    }
72
73    /// How much of the fog shows: GL JS fades it in as the horizon comes into view between 60
74    /// and 70 degrees of pitch, while an external eye's wide frame has the horizon in view at
75    /// almost any pitch, so its fog is always on.
76    pub fn fog_opacity(&self) -> f32 {
77        if self.external_eye.is_some() {
78            return 1.0;
79        }
80        crate::style::sky::SkySpecification::fog_blend_opacity(
81            self.camera.get_pitch().0.to_degrees(),
82        )
83    }
84
85    /// Near and far clip distances in pixels, following GL JS `_calculateNearFarZIfNeeded`.
86    ///
87    /// The far plane reaches the top of the screen on the lowest visible plane, capped at the
88    /// Mercator horizon so a pitched camera never asks for an infinite range.
89    pub fn depth_range(&self, center_offset: Point2<f64>) -> (f64, f64) {
90        let distance = self.camera_to_center_distance();
91        let pixels_per_meter = self.pixels_per_meter();
92        let pitch = self.camera.get_pitch().0.abs();
93        let limited_pitch = pitch.min(MAX_MERCATOR_HORIZON_ANGLE.0);
94        let camera_to_sea_level = (distance / 2.0)
95            .max(distance + self.center.elevation * pixels_per_meter / limited_pitch.cos());
96        let camera_altitude = pitch.cos() * distance / pixels_per_meter + self.center.elevation;
97        let min_elevation = self
98            .center
99            .elevation
100            .min(self.min_elevation)
101            .min(camera_altitude - MIN_RENDER_DISTANCE_BELOW_CAMERA_METERS);
102        let lowest_plane = if min_elevation < 0.0 {
103            camera_to_sea_level - min_elevation * pixels_per_meter / limited_pitch.cos()
104        } else {
105            camera_to_sea_level
106        };
107
108        let ground_angle = consts::FRAC_PI_2 + pitch;
109        // A rolled view reaches further along the screen diagonal than the vertical field of
110        // view alone, so the far plane widens the field of view with the roll as GL JS does.
111        let roll = self.camera.get_roll().0;
112        let rolled_fov = self.perspective.fovy().0
113            * (roll.cos().abs() * self.height + roll.sin().abs() * self.width)
114            / self.height;
115        let fov_above_center = rolled_fov * (0.5 + center_offset.y / self.height);
116        let surface_distance = |fov: f64| {
117            fov.sin() * lowest_plane
118                / (consts::PI - ground_angle - fov)
119                    .clamp(0.01, consts::PI - 0.01)
120                    .sin()
121        };
122        let top_half_surface_distance = surface_distance(fov_above_center);
123
124        let horizon = distance
125            * ((consts::FRAC_PI_2 - pitch).tan() * 0.85)
126                .min((MAX_MERCATOR_HORIZON_ANGLE.0 - pitch).tan());
127        let horizon_angle = (horizon / distance).atan();
128        let min_fov_center_to_horizon = consts::FRAC_PI_2 - MAX_MERCATOR_HORIZON_ANGLE.0;
129        let fov_center_to_horizon = if horizon_angle > min_fov_center_to_horizon {
130            2.0 * horizon_angle * (0.5 + center_offset.y / (horizon * 2.0))
131        } else {
132            min_fov_center_to_horizon
133        };
134        let top_half_horizon_distance = surface_distance(fov_center_to_horizon);
135
136        let top_half = top_half_surface_distance.min(top_half_horizon_distance);
137        let far_z = ((consts::FRAC_PI_2 - limited_pitch).cos() * top_half + lowest_plane) * 1.01;
138        let near_z = self.height / 50.0;
139        (near_z, far_z)
140    }
