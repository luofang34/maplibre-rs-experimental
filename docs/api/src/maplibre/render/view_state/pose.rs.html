1//! Camera pose: where the camera stands and where it looks.
2//!
3//! The view state keeps the map center, zoom and angles, as GL JS does. A pose gives the
4//! camera's own ground position and altitude instead, the form a tracker or a flight path
5//! provides. Converting a pose into center and zoom keeps tile covering and pixel scale
6//! consistent with the map's own gestures, as GL JS `calculateCenterFromCameraLngLatAlt` and
7//! `getCameraLngLat` do.
8

22/// Where the camera stands and where it looks.
23#[derive(Clone, Copy, Debug)]
24pub struct CameraPose {
25    /// Ground position under the camera.
26    pub position: LatLon,
27    /// Altitude of the camera in metres above sea level.
28    pub altitude_meters: f64,
29    /// Bearing, clockwise from north.
30    pub bearing: Deg<f64>,
31    /// Pitch away from looking straight down; beyond 90 degrees the camera looks upwards.
32    pub pitch: Deg<f64>,
33    /// Roll about the view axis.
34    pub roll: Deg<f64>,
35}

37/// What a jump sets, as GL JS `CameraOptions`; what is left out keeps its value.
38#[derive(Clone, Copy, Debug, Default, PartialEq)]
39pub struct CameraOptions {
40    /// The map center.
41    pub center: Option<LatLon>,
42    /// The zoom.
43    pub zoom: Option<f64>,
44    /// Bearing, clockwise from north.
45    pub bearing: Option<Deg<f64>>,
46    /// Pitch away from looking straight down.
47    pub pitch: Option<Deg<f64>>,
48    /// Roll about the view axis.
49    pub roll: Option<Deg<f64>>,
50    /// Elevation of the center in metres above sea level, as GL JS `elevation`. Terrain that
51    /// holds the center stands it on the loaded ground from the next frame.
52    pub elevation: Option<f64>,
53}

55/// The center, zoom and angles a jump resolves to.
56#[derive(Clone, Copy, Debug)]
57pub(crate) struct ResolvedJump {
58    pub center: LatLon,
59    pub zoom: Zoom,
60    pub bearing: Deg<f64>,
61    pub pitch: Deg<f64>,
62    pub roll: Deg<f64>,
63}

65impl ResolvedJump {
66    /// Whether two cameras show the same, but for rounding.
67    pub(crate) fn same_as(&self, other: &Self) -> bool {
68        const TOLERANCE: f64 = 1e-9;
69        [
70            self.center.latitude - other.center.latitude,
71            self.center.longitude - other.center.longitude,
72            self.zoom.value() - other.zoom.value(),
73            self.bearing.0 - other.bearing.0,
74            self.pitch.0 - other.pitch.0,
75            self.roll.0 - other.roll.0,
76        ]
77        .iter()
78        .all(|difference| difference.abs() <= TOLERANCE)
79    }
80}
81
82/// Bound on the center search, as GL JS sets it; the search converges in about five passes.
83const MAX_CENTER_ITERATIONS: usize = 10;
84/// Distance error in metres below which the center search stops.
85const CENTER_TOLERANCE_METERS: f64 = 1e-12;
86
87/// How many heights ahead an eye's center may lie; beyond that the ground near the horizon
88/// would sling the center and the zoom by kilometres per degree of head pitch, and a glance
89/// up would request tiles at a wildly different zoom.
90const EYE_CENTER_REACH_HEIGHTS: f64 = 5.0;
91/// The reach in metres is kept within these bounds: near the ground the center stays far
92/// enough ahead that the covering reaches the horizon, and from high up it stays near enough
93/// that the center's latitude remains on the map.
94const MIN_EYE_CENTER_REACH_METERS: f64 = 10_000.0;
95const MAX_EYE_CENTER_REACH_METERS: f64 = 50_000.0;
96
97impl ViewState {
98    /// Places the camera at `pose`.
99    ///
100    /// The map center moves to where the view ray meets the center elevation and the zoom
101    /// follows the camera's distance to it, so a pose beyond the pitch limit is clamped to it.
102    pub fn set_camera_pose(&mut self, pose: CameraPose) {
103        let pitch = Rad::from(pose.pitch).0;
104        let (distance_meters, elevation) =
105            distance_to_center_from_altitude(pose.altitude_meters, self.center_elevation(), pitch);
106        self.place_camera(pose, distance_meters, elevation);
107    }
108
109    /// Places a host's eye at `pose`. The center is where the gaze meets the anchor elevation,
110    /// at most `EYE_CENTER_REACH_HEIGHTS` heights ahead; a gaze nearer the horizon than that
111    /// keeps the center at that reach and leaves the center elevation alone, so a head
112    /// turning up to the horizon moves the center smoothly instead of snapping it to a fixed
113    /// distance in the air.
114    pub(super) fn set_eye_pose(&mut self, pose: CameraPose, anchor_elevation: f64) {
115        let pitch = Rad::from(pose.pitch).0;
116        let dz = -pitch.cos();
117        let elevation = anchor_elevation;
118        // Sampled terrain at the gaze center must not feed back into the next frame's
119        // center search: steep slopes can make a stationary eye oscillate between LODs.
120        let above_ground = pose.altitude_meters - anchor_elevation;
121        let reach = (EYE_CENTER_REACH_HEIGHTS * above_ground.abs())
122            .clamp(MIN_EYE_CENTER_REACH_METERS, MAX_EYE_CENTER_REACH_METERS);
123        let distance_meters = if dz * above_ground < 0.0 {
124            (-above_ground / dz).min(reach)
125        } else {
126            reach
127        };
128        self.place_camera(pose, distance_meters, elevation);
129    }
130
131    fn place_camera(&mut self, pose: CameraPose, distance_meters: f64, elevation: f64) {
132        let pitch = Rad::from(pose.pitch).0;
133        let bearing = Rad::from(pose.bearing).0;
134        let (dx, dy, _) = camera_direction(pitch, bearing);
135        let camera = mercator_from_lat_lon(pose.position);
136        let circumference = self.body().circumference_meters();
137        // The Mercator scale changes with latitude, and the center's latitude is the one that
138        // matters; starting from the camera's scale, every pass re-evaluates it at the center
139        // found so far.
140        let mut meters_per_unit = circumference * latitude_at_mercator_y(camera.y).cos();
141        let mut center = camera;
142        let mut distance_units = distance_meters / meters_per_unit;
143        for _ in 0..MAX_CENTER_ITERATIONS {
144            distance_units = distance_meters / meters_per_unit;
145            center = Point2::new(
146                camera.x + dx * distance_units,
147                camera.y + dy * distance_units,
148            );
149            meters_per_unit = circumference * latitude_at_mercator_y(center.y).cos();
150            if (distance_meters - distance_units * meters_per_unit).abs() <= CENTER_TOLERANCE_METERS
151            {
152                break;
153            }
154        }
155        let zoom = self.zoom_for_center_distance_units(distance_units);
156        self.set_center_elevation(elevation);
157        self.set_center_and_angles(
158            lat_lon_at_mercator(center),
159            zoom,
160            pose.bearing,
161            pose.pitch,
162            pose.roll,
163        );
164    }
165
166    /// The zoom at which the camera sits `distance_meters` from a center at `center`, the
167    /// definition a pose and an external eye share.
168    pub(super) fn zoom_for_center_distance(&self, distance_meters: f64, center: LatLon) -> Zoom {
169        let meters_per_unit =
170            self.body().circumference_meters() * center.latitude.to_radians().cos();
171        self.zoom_for_center_distance_units(distance_meters / meters_per_unit)
172    }
173
174    /// The zoom at which the camera sits `distance_units` Mercator units from the center.
175    fn zoom_for_center_distance_units(&self, distance_units: f64) -> Zoom {
176        Zoom::new(
177            (self.height()
178                / 2.0
179                / (self.field_of_view().0 / 2.0).tan()
180                / distance_units
181                / TILE_SIZE)
182                .log2(),
183        )
184    }
185
186    /// Moves the map center and the camera angles, leaving the center elevation as it is.
187    pub(super) fn set_center_and_angles(
188        &mut self,
189        center: LatLon,
190        zoom: Zoom,
191        bearing: Deg<f64>,
192        pitch: Deg<f64>,
193        roll: Deg<f64>,
194    ) {
195        let world_size = TILE_SIZE * 2f64.powf(zoom.value());
196        let center = mercator_from_lat_lon(center);
197        self.update_zoom(zoom);
198        let camera = self.camera_mut();
199        camera.move_to(Point2::new(center.x * world_size, center.y * world_size));
200        camera.set_bearing(bearing);
201        camera.set_pitch(pitch);
202        camera.set_roll(roll);
203    }
204
205    /// Moves the map at once, as GL JS `jumpTo`: to the center, zoom and angles given, with
206    /// the others kept as they are. A zoom that is not a number is left out, as GL JS
207    /// constrains a jump.
208    ///
209    /// Constrained navigation keeps the center's latitude within the Mercator limit; free-globe
210    /// navigation places the pose [`Self::globe_pose_for`] gives. The center rests at the
211    /// height [`Self::jump_elevation`] gives, until terrain holding it stands it on the ground
212    /// under it, as [`crate::context::MapContext::jump_to`] does at once.
213    pub fn jump_to(&mut self, options: CameraOptions) -> Result<(), NavigationError> {
214        if self.navigation_mode() == NavigationMode::FreeGlobe {
215            return self.set_globe_pose(self.globe_pose_for(options));
216        }
217        let jump = self.resolve_jump(options);
218        let center = LatLon::new(
219            jump.center
220                .latitude
221                .clamp(-MERCATOR_LATITUDE_LIMIT, MERCATOR_LATITUDE_LIMIT),
222            jump.center.longitude,
223        );
224        self.set_center_and_angles(center, jump.zoom, jump.bearing, jump.pitch, jump.roll);
225        if let Some(elevation) = self.jump_elevation(options) {
226            self.rest_center_at(elevation);
227        }
228        Ok(())
229    }
230
231    /// The height a jump to `options` rests its center at, `None` keeping the current one:
232    /// the elevation given, or, for a center that moves without one, sea level where terrain
233    /// holds it, as GL JS reads ground it has not loaded, and the center altitude otherwise.
234    /// A new center never inherits the ground of the one it left.
235    pub fn jump_elevation(&self, options: CameraOptions) -> Option<f64> {
236        options
237            .elevation
238            .filter(|meters| meters.is_finite())
239            .or_else(|| {
240                options.center.map(|_| {
241                    if self.center_held_by_terrain() {
242                        0.0
243                    } else {
244                        self.center_altitude()
245                    }
246                })
247            })
248    }
249
250    /// The center, zoom and angles a jump to `options` shows, the camera's own standing in for
251    /// those left out.
252    pub(crate) fn resolve_jump(&self, options: CameraOptions) -> ResolvedJump {
253        let camera = self.camera();
254        ResolvedJump {
255            // A free-globe camera may look past the Mercator world the flat center stops at.
256            center: options.center.unwrap_or_else(|| {
257                self.pose_view()
258                    .map_or_else(|| self.center_lat_lon(), |view| view.center)
259            }),
260            zoom: options
261                .zoom
262                .filter(|zoom| zoom.is_finite())
263                .map_or_else(|| self.zoom(), Zoom::new),
264            bearing: options
265                .bearing
266                .unwrap_or_else(|| camera.get_bearing().into()),
267            pitch: options.pitch.unwrap_or_else(|| camera.get_pitch().into()),
268            roll: options.roll.unwrap_or_else(|| camera.get_roll().into()),
269        }
270    }
271
272    /// The map center in latitude and longitude.
273    pub fn center_lat_lon(&self) -> LatLon {
274        let world_size = TILE_SIZE * 2f64.powf(self.zoom().value());
275        let position = self.camera().position();
276        lat_lon_at_mercator(Point2::new(
277            position.x / world_size,
278            position.y / world_size,
279        ))
280    }
281
282    /// Where the camera stands and where it looks.
283    pub fn camera_pose(&self) -> CameraPose {
284        let (position, altitude_meters) = camera_ground_position(self);
285        let world_size = TILE_SIZE * 2f64.powf(self.zoom().value());
286        let camera = self.camera();
287        CameraPose {
288            position: lat_lon_at_mercator(Point2::new(
289                position.x / world_size,
290                position.y / world_size,
291            )),
292            altitude_meters,
293            bearing: camera.get_bearing().into(),
294            pitch: camera.get_pitch().into(),
295            roll: camera.get_roll().into(),
296        }
297    }
298}

300/// Position on the unit Mercator square, with y pointing south.
301pub(super) fn mercator_from_lat_lon(position: LatLon) -> Point2<f64> {
302    let x = (position.longitude + 180.0) / 360.0;
303    let latitude = position.latitude.to_radians();
304    let y = 0.5 - (PI / 4.0 + latitude / 2.0).tan().ln() / (2.0 * PI);
305    Point2::new(x, y)
306}
307
308pub(super) fn lat_lon_at_mercator(point: Point2<f64>) -> LatLon {
309    LatLon::new(
310        latitude_at_mercator_y(point.y).to_degrees(),
311        point.x * 360.0 - 180.0,
312    )
313}
314
315/// Latitude in radians at a unit Mercator y.
316fn latitude_at_mercator_y(y: f64) -> f64 {
317    (PI * (1.0 - 2.0 * y)).sinh().atan()
318}
