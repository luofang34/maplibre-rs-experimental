1//! Camera pose: where the camera stands and where it looks.
2//!
3//! The view state keeps the map center, zoom and angles, as GL JS does. A pose gives the
4//! camera's own ground position and altitude instead, the form a tracker or a flight path
5//! provides. Converting a pose into center and zoom keeps tile covering and pixel scale
6//! consistent with the map's own gestures, as GL JS `calculateCenterFromCameraLngLatAlt` and
7//! `getCameraLngLat` do.
8
9use std::f64::consts::PI;
10
11use cgmath::{Deg, Point2, Rad};

13use crate::{
14    coords::{LatLon, Zoom, TILE_SIZE},

16    render::view_state::{NavigationError, NavigationMode, ViewState},
17    terrain::interaction::{
18        camera_direction, camera_ground_position, distance_to_center_from_altitude,
19    },
20};

22/// Where the camera stands and where it looks.
23#[derive(Clone, Copy, Debug)]
24pub struct CameraPose {
25    /// Ground position under the camera.
26    pub position: LatLon,
27    /// Altitude of the camera in metres above sea level.
28    pub altitude_meters: f64,
29    /// Bearing, clockwise from north.
30    pub bearing: Deg<f64>,
31    /// Pitch away from looking straight down; beyond 90 degrees the camera looks upwards.
32    pub pitch: Deg<f64>,
33    /// Roll about the view axis.
34    pub roll: Deg<f64>,
35}
36
37/// What a jump sets, as GL JS `CameraOptions`; what is left out keeps its value.
38#[derive(Clone, Copy, Debug, Default, PartialEq)]
39pub struct CameraOptions {
40    /// The map center.
41    pub center: Option<LatLon>,
42    /// The zoom.
43    pub zoom: Option<f64>,
44    /// Bearing, clockwise from north.
45    pub bearing: Option<Deg<f64>>,
46    /// Pitch away from looking straight down.
47    pub pitch: Option<Deg<f64>>,
48    /// Roll about the view axis.
49    pub roll: Option<Deg<f64>>,
50    /// Elevation of the center in metres above sea level, as GL JS `elevation`. Terrain that
51    /// holds the center stands it on the loaded ground from the next frame.
52    pub elevation: Option<f64>,
53}

55/// The center, zoom and angles a jump resolves to.
56pub(super) struct ResolvedJump {
57    pub center: LatLon,
58    pub zoom: Zoom,
59    pub bearing: Deg<f64>,
60    pub pitch: Deg<f64>,
61    pub roll: Deg<f64>,
62}
63
64/// Bound on the center search, as GL JS sets it; the search converges in about five passes.
65const MAX_CENTER_ITERATIONS: usize = 10;
66/// Distance error in metres below which the center search stops.
67const CENTER_TOLERANCE_METERS: f64 = 1e-12;
68
69/// How many heights ahead an eye's center may lie; beyond that the ground near the horizon
70/// would sling the center and the zoom by kilometres per degree of head pitch, and a glance
71/// up would request tiles at a wildly different zoom.
72const EYE_CENTER_REACH_HEIGHTS: f64 = 5.0;
73/// The reach in metres is kept within these bounds: near the ground the center stays far
74/// enough ahead that the covering reaches the horizon, and from high up it stays near enough
75/// that the center's latitude remains on the map.
76const MIN_EYE_CENTER_REACH_METERS: f64 = 10_000.0;
77const MAX_EYE_CENTER_REACH_METERS: f64 = 50_000.0;
78
79impl ViewState {
80    /// Places the camera at `pose`.
81    ///
82    /// The map center moves to where the view ray meets the center elevation and the zoom
83    /// follows the camera's distance to it, so a pose beyond the pitch limit is clamped to it.
84    pub fn set_camera_pose(&mut self, pose: CameraPose) {
85        let pitch = Rad::from(pose.pitch).0;
86        let (distance_meters, elevation) =
87            distance_to_center_from_altitude(pose.altitude_meters, self.center_elevation(), pitch);
88        self.place_camera(pose, distance_meters, elevation);
89    }
90
91    /// Places a host's eye at `pose`. The center is where the gaze meets the anchor elevation,
92    /// at most `EYE_CENTER_REACH_HEIGHTS` heights ahead; a gaze nearer the horizon than that
93    /// keeps the center at that reach and leaves the center elevation alone, so a head
94    /// turning up to the horizon moves the center smoothly instead of snapping it to a fixed
95    /// distance in the air.
96    pub(super) fn set_eye_pose(&mut self, pose: CameraPose, anchor_elevation: f64) {
97        let pitch = Rad::from(pose.pitch).0;
98        let dz = -pitch.cos();
99        let elevation = anchor_elevation;
100        // Sampled terrain at the gaze center must not feed back into the next frame's
101        // center search: steep slopes can make a stationary eye oscillate between LODs.
102        let above_ground = pose.altitude_meters - anchor_elevation;
103        let reach = (EYE_CENTER_REACH_HEIGHTS * above_ground.abs())
104            .clamp(MIN_EYE_CENTER_REACH_METERS, MAX_EYE_CENTER_REACH_METERS);
105        let distance_meters = if dz * above_ground < 0.0 {
106            (-above_ground / dz).min(reach)
107        } else {
108            reach
109        };
110        self.place_camera(pose, distance_meters, elevation);
111    }
112
113    fn place_camera(&mut self, pose: CameraPose, distance_meters: f64, elevation: f64) {
114        let pitch = Rad::from(pose.pitch).0;
115        let bearing = Rad::from(pose.bearing).0;
116        let (dx, dy, _) = camera_direction(pitch, bearing);
117        let camera = mercator_from_lat_lon(pose.position);
118        let circumference = self.body().circumference_meters();
119        // The Mercator scale changes with latitude, and the center's latitude is the one that
120        // matters; starting from the camera's scale, every pass re-evaluates it at the center
121        // found so far.
122        let mut meters_per_unit = circumference * latitude_at_mercator_y(camera.y).cos();
123        let mut center = camera;
124        let mut distance_units = distance_meters / meters_per_unit;
125        for _ in 0..MAX_CENTER_ITERATIONS {
126            distance_units = distance_meters / meters_per_unit;
127            center = Point2::new(
128                camera.x + dx * distance_units,
129                camera.y + dy * distance_units,
130            );
131            meters_per_unit = circumference * latitude_at_mercator_y(center.y).cos();
132            if (distance_meters - distance_units * meters_per_unit).abs() <= CENTER_TOLERANCE_METERS
133            {
134                break;
135            }
136        }
137        let zoom = self.zoom_for_center_distance_units(distance_units);
138        self.set_center_elevation(elevation);
139        self.set_center_and_angles(
140            lat_lon_at_mercator(center),
141            zoom,
142            pose.bearing,
143            pose.pitch,
144            pose.roll,
145        );
146    }
147
148    /// The zoom at which the camera sits `distance_meters` from a center at `center`, the
149    /// definition a pose and an external eye share.
150    pub(super) fn zoom_for_center_distance(&self, distance_meters: f64, center: LatLon) -> Zoom {
151        let meters_per_unit =
152            self.body().circumference_meters() * center.latitude.to_radians().cos();
153        self.zoom_for_center_distance_units(distance_meters / meters_per_unit)
154    }
155
156    /// The zoom at which the camera sits `distance_units` Mercator units from the center.
157    fn zoom_for_center_distance_units(&self, distance_units: f64) -> Zoom {
158        Zoom::new(
159            (self.height()
160                / 2.0
161                / (self.field_of_view().0 / 2.0).tan()
162                / distance_units
163                / TILE_SIZE)
164                .log2(),
165        )
166    }
167
168    /// Moves the map center and the camera angles, leaving the center elevation as it is.
169    pub(super) fn set_center_and_angles(
170        &mut self,
171        center: LatLon,
172        zoom: Zoom,
173        bearing: Deg<f64>,
174        pitch: Deg<f64>,
175        roll: Deg<f64>,
176    ) {
177        let world_size = TILE_SIZE * 2f64.powf(zoom.value());
178        let center = mercator_from_lat_lon(center);
179        self.update_zoom(zoom);
180        let camera = self.camera_mut();
181        camera.move_to(Point2::new(center.x * world_size, center.y * world_size));
182        camera.set_bearing(bearing);
183        camera.set_pitch(pitch);
184        camera.set_roll(roll);
185    }
186
187    /// Moves the map at once, as GL JS `jumpTo`: to the center, zoom and angles given, with
188    /// the others kept as they are. A zoom that is not a number is left out, as GL JS
189    /// constrains a jump.
190    ///
191    /// Constrained navigation keeps the center's latitude within the Mercator limit; free-globe
192    /// navigation places the pose [`Self::globe_pose_for`] gives. The center rests at the
193    /// height [`Self::jump_elevation`] gives, until terrain holding it stands it on the ground
194    /// under it, as [`crate::context::MapContext::jump_to`] does at once.
195    pub fn jump_to(&mut self, options: CameraOptions) -> Result<(), NavigationError> {
196        if self.navigation_mode() == NavigationMode::FreeGlobe {
197            return self.set_globe_pose(self.globe_pose_for(options));
198        }
199        let jump = self.resolve_jump(options);
200        let center = LatLon::new(
201            jump.center
202                .latitude
203                .clamp(-MERCATOR_LATITUDE_LIMIT, MERCATOR_LATITUDE_LIMIT),
204            jump.center.longitude,
205        );
206        self.set_center_and_angles(center, jump.zoom, jump.bearing, jump.pitch, jump.roll);
207        if let Some(elevation) = self.jump_elevation(options) {
208            self.rest_center_at(elevation);
209        }
210        Ok(())
211    }
212
213    /// The height a jump to `options` rests its center at, `None` keeping the current one:
214    /// the elevation given, or, for a center that moves without one, sea level where terrain
215    /// holds it, as GL JS reads ground it has not loaded, and the center altitude otherwise.
216    /// A new center never inherits the ground of the one it left.
217    pub fn jump_elevation(&self, options: CameraOptions) -> Option<f64> {
218        options
219            .elevation
220            .filter(|meters| meters.is_finite())
221            .or_else(|| {
222                options.center.map(|_| {
223                    if self.center_held_by_terrain() {
224                        0.0
225                    } else {
226                        self.center_altitude()
227                    }
228                })
229            })
230    }
231
232    /// The center, zoom and angles a jump to `options` shows, the camera's own standing in for
233    /// those left out.
234    pub(super) fn resolve_jump(&self, options: CameraOptions) -> ResolvedJump {
235        let camera = self.camera();
236        ResolvedJump {
237            // A free-globe camera may look past the Mercator world the flat center stops at.
238            center: options.center.unwrap_or_else(|| {
239                self.pose_view()
240                    .map_or_else(|| self.center_lat_lon(), |view| view.center)
241            }),
242            zoom: options
243                .zoom
244                .filter(|zoom| zoom.is_finite())
245                .map_or_else(|| self.zoom(), Zoom::new),
246            bearing: options
247                .bearing
248                .unwrap_or_else(|| camera.get_bearing().into()),
249            pitch: options.pitch.unwrap_or_else(|| camera.get_pitch().into()),
250            roll: options.roll.unwrap_or_else(|| camera.get_roll().into()),
251        }
252    }
253
254    /// The map center in latitude and longitude.
255    pub fn center_lat_lon(&self) -> LatLon {
256        let world_size = TILE_SIZE * 2f64.powf(self.zoom().value());
257        let position = self.camera().position();
258        lat_lon_at_mercator(Point2::new(
259            position.x / world_size,
260            position.y / world_size,
261        ))
262    }
263
264    /// Where the camera stands and where it looks.
265    pub fn camera_pose(&self) -> CameraPose {
266        let (position, altitude_meters) = camera_ground_position(self);
267        let world_size = TILE_SIZE * 2f64.powf(self.zoom().value());
268        let camera = self.camera();
269        CameraPose {
270            position: lat_lon_at_mercator(Point2::new(
271                position.x / world_size,
272                position.y / world_size,
273            )),
274            altitude_meters,
275            bearing: camera.get_bearing().into(),
276            pitch: camera.get_pitch().into(),
277            roll: camera.get_roll().into(),
278        }
279    }
280}

282/// Position on the unit Mercator square, with y pointing south.
283pub(super) fn mercator_from_lat_lon(position: LatLon) -> Point2<f64> {
284    let x = (position.longitude + 180.0) / 360.0;
285    let latitude = position.latitude.to_radians();
286    let y = 0.5 - (PI / 4.0 + latitude / 2.0).tan().ln() / (2.0 * PI);
287    Point2::new(x, y)
288}
289
290pub(super) fn lat_lon_at_mercator(point: Point2<f64>) -> LatLon {
291    LatLon::new(
292        latitude_at_mercator_y(point.y).to_degrees(),
293        point.x * 360.0 - 180.0,
294    )
295}
296
297/// Latitude in radians at a unit Mercator y.
298fn latitude_at_mercator_y(y: f64) -> f64 {
299    (PI * (1.0 - 2.0 * y)).sinh().atan()
300}
