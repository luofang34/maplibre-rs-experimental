1//! Camera pose: where the camera stands and where it looks.
2//!
3//! The view state keeps the map center, zoom and angles, as GL JS does. A pose gives the
4//! camera's own ground position and altitude instead, the form a tracker or a flight path
5//! provides. Converting a pose into center and zoom keeps tile covering and pixel scale
6//! consistent with the map's own gestures, as GL JS `calculateCenterFromCameraLngLatAlt` and
7//! `getCameraLngLat` do.
8
9use std::f64::consts::PI;
10
11use cgmath::{Deg, Point2, Rad};
12
13use crate::{
14    coords::{LatLon, Zoom, TILE_SIZE},
15    render::view_state::ViewState,
16    terrain::interaction::{
17        camera_direction, camera_ground_position, distance_to_center_from_altitude,
18    },
19};
20
21/// Where the camera stands and where it looks.
22#[derive(Clone, Copy, Debug)]
23pub struct CameraPose {
24    /// Ground position under the camera.
25    pub position: LatLon,
26    /// Altitude of the camera in metres above sea level.
27    pub altitude_meters: f64,
28    /// Bearing, clockwise from north.
29    pub bearing: Deg<f64>,
30    /// Pitch away from looking straight down; beyond 90 degrees the camera looks upwards.
31    pub pitch: Deg<f64>,
32    /// Roll about the view axis.
33    pub roll: Deg<f64>,
34}
35
36/// Bound on the center search, as GL JS sets it; the search converges in about five passes.
37const MAX_CENTER_ITERATIONS: usize = 10;
38/// Distance error in metres below which the center search stops.
39const CENTER_TOLERANCE_METERS: f64 = 1e-12;
40
41/// How many heights ahead an eye's center may lie; beyond that the ground near the horizon
42/// would sling the center and the zoom by kilometres per degree of head pitch, and a glance
43/// up would request tiles at a wildly different zoom.
44const EYE_CENTER_REACH_HEIGHTS: f64 = 5.0;
45/// The reach in metres is kept within these bounds: near the ground the center stays far
46/// enough ahead that the covering reaches the horizon, and from high up it stays near enough
47/// that the center's latitude remains on the map.
48const MIN_EYE_CENTER_REACH_METERS: f64 = 10_000.0;
49const MAX_EYE_CENTER_REACH_METERS: f64 = 50_000.0;
50
51impl ViewState {
52    /// Places the camera at `pose`.
53    ///
54    /// The map center moves to where the view ray meets the center elevation and the zoom
55    /// follows the camera's distance to it, so a pose beyond the pitch limit is clamped to it.
56    pub fn set_camera_pose(&mut self, pose: CameraPose) {
57        let pitch = Rad::from(pose.pitch).0;
58        let (distance_meters, elevation) =
59            distance_to_center_from_altitude(pose.altitude_meters, self.center_elevation(), pitch);
60        self.place_camera(pose, distance_meters, elevation);
61    }
62
63    /// Places a host's eye at `pose`. The center is where the gaze meets the anchor elevation,
64    /// at most `EYE_CENTER_REACH_HEIGHTS` heights ahead; a gaze nearer the horizon than that
65    /// keeps the center at that reach and leaves the center elevation alone, so a head
66    /// turning up to the horizon moves the center smoothly instead of snapping it to a fixed
67    /// distance in the air.
68    pub(super) fn set_eye_pose(&mut self, pose: CameraPose, anchor_elevation: f64) {
69        let pitch = Rad::from(pose.pitch).0;
70        let dz = -pitch.cos();
71        let elevation = anchor_elevation;
72        // Sampled terrain at the gaze center must not feed back into the next frame's
73        // center search: steep slopes can make a stationary eye oscillate between LODs.
74        let above_ground = pose.altitude_meters - anchor_elevation;
75        let reach = (EYE_CENTER_REACH_HEIGHTS * above_ground.abs())
76            .clamp(MIN_EYE_CENTER_REACH_METERS, MAX_EYE_CENTER_REACH_METERS);
77        let distance_meters = if dz * above_ground < 0.0 {
78            (-above_ground / dz).min(reach)
79        } else {
80            reach
81        };
82        self.place_camera(pose, distance_meters, elevation);
83    }
84
85    fn place_camera(&mut self, pose: CameraPose, distance_meters: f64, elevation: f64) {
86        let pitch = Rad::from(pose.pitch).0;
87        let bearing = Rad::from(pose.bearing).0;
88        let (dx, dy, _) = camera_direction(pitch, bearing);
89        let camera = mercator_from_lat_lon(pose.position);
90        let circumference = self.body().circumference_meters();
91        // The Mercator scale changes with latitude, and the center's latitude is the one that
92        // matters; starting from the camera's scale, every pass re-evaluates it at the center
93        // found so far.
94        let mut meters_per_unit = circumference * latitude_at_mercator_y(camera.y).cos();
95        let mut center = camera;
96        let mut distance_units = distance_meters / meters_per_unit;
97        for _ in 0..MAX_CENTER_ITERATIONS {
98            distance_units = distance_meters / meters_per_unit;
99            center = Point2::new(
100                camera.x + dx * distance_units,
101                camera.y + dy * distance_units,
102            );
103            meters_per_unit = circumference * latitude_at_mercator_y(center.y).cos();
104            if (distance_meters - distance_units * meters_per_unit).abs() <= CENTER_TOLERANCE_METERS
105            {
106                break;
107            }
108        }
109        let zoom = self.zoom_for_center_distance_units(distance_units);
110        self.set_center_elevation(elevation);
111        self.set_center_and_angles(
112            lat_lon_at_mercator(center),
113            zoom,
114            pose.bearing,
115            pose.pitch,
116            pose.roll,
117        );
118    }
119
120    /// The zoom at which the camera sits `distance_meters` from a center at `center`, the
121    /// definition a pose and an external eye share.
122    pub(super) fn zoom_for_center_distance(&self, distance_meters: f64, center: LatLon) -> Zoom {
123        let meters_per_unit =
124            self.body().circumference_meters() * center.latitude.to_radians().cos();
125        self.zoom_for_center_distance_units(distance_meters / meters_per_unit)
126    }
127
128    /// The zoom at which the camera sits `distance_units` Mercator units from the center.
129    fn zoom_for_center_distance_units(&self, distance_units: f64) -> Zoom {
130        Zoom::new(
131            (self.height()
132                / 2.0
133                / (self.field_of_view().0 / 2.0).tan()
134                / distance_units
135                / TILE_SIZE)
136                .log2(),
137        )
138    }
139
140    /// Moves the map center and the camera angles, leaving the center elevation as it is.
141    pub(super) fn set_center_and_angles(
142        &mut self,
143        center: LatLon,
144        zoom: Zoom,
145        bearing: Deg<f64>,
146        pitch: Deg<f64>,
147        roll: Deg<f64>,
148    ) {
149        let world_size = TILE_SIZE * 2f64.powf(zoom.value());
150        let center = mercator_from_lat_lon(center);
151        self.update_zoom(zoom);
152        let camera = self.camera_mut();
153        camera.move_to(Point2::new(center.x * world_size, center.y * world_size));
154        camera.set_bearing(bearing);
155        camera.set_pitch(pitch);
156        camera.set_roll(roll);
157    }
158
159    /// Where the camera stands and where it looks.
160    pub fn camera_pose(&self) -> CameraPose {
161        let (position, altitude_meters) = camera_ground_position(self);
162        let world_size = TILE_SIZE * 2f64.powf(self.zoom().value());
163        let camera = self.camera();
164        CameraPose {
165            position: lat_lon_at_mercator(Point2::new(
166                position.x / world_size,
167                position.y / world_size,
168            )),
169            altitude_meters,
170            bearing: camera.get_bearing().into(),
171            pitch: camera.get_pitch().into(),
172            roll: camera.get_roll().into(),
173        }
174    }
175}
176
177/// Position on the unit Mercator square, with y pointing south.
178pub(super) fn mercator_from_lat_lon(position: LatLon) -> Point2<f64> {
179    let x = (position.longitude + 180.0) / 360.0;
180    let latitude = position.latitude.to_radians();
181    let y = 0.5 - (PI / 4.0 + latitude / 2.0).tan().ln() / (2.0 * PI);
182    Point2::new(x, y)
183}
184
185pub(super) fn lat_lon_at_mercator(point: Point2<f64>) -> LatLon {
186    LatLon::new(
187        latitude_at_mercator_y(point.y).to_degrees(),
188        point.x * 360.0 - 180.0,
189    )
190}
191
192/// Latitude in radians at a unit Mercator y.
193fn latitude_at_mercator_y(y: f64) -> f64 {
194    (PI * (1.0 - 2.0 * y)).sinh().atan()
195}
