1//! External views: an eye the host places, for head tracking, a stereo pair, or a globe the
2//! host set down in its own world.
3//!
4//! The host hands over the view matrix it renders with and the eye's frustum. The map derives
5//! its center, zoom and angles from the view matrix, so tile covering and pixel scale follow
6//! the eye as they do for the map's own gestures, and draws with matrices built from the eye
7//! itself, in its own units, so the frame matches the host's frustum whatever way the eye
8//! turns.
9

31pub struct ExternalAnchor {
32    /// Ground position of the frame origin.
33    pub position: LatLon,
34    /// Altitude of the frame origin in metres above sea level.
35    pub altitude_meters: f64,
36}

38/// An eye the host renders with.
39#[derive(Clone, Copy, Debug)]
40pub struct ExternalView {
41    /// Origin of the local frame the view matrix starts from.
42    pub anchor: ExternalAnchor,
43    /// Local frame to eye space.
44    ///
45    /// The local frame measures metres east, north and up from the anchor. Eye space follows
46    /// the OpenGL convention: x right, y up, the eye looking along negative z. The matrix may
47    /// scale uniformly, as it does when the host shows the map as a model at another size;
48    /// the frustum's clip distances are then in the eye's units, not the local frame's.
49    pub view: Matrix4<f64>,
50    /// The eye's frustum, with clip distances in eye space units.
51    pub frustum: EyeFrustum,
52}
53
54/// Why an external view cannot drive the map.
55#[derive(Error, Debug, Clone, Copy, PartialEq)]
56pub enum ExternalViewError {
57    /// The view matrix has no inverse, so it places the camera nowhere.
58    #[error("the external view matrix is singular")]
59    SingularView,
60    /// The frustum has no volume: a side or clip distance is not finite, the near plane is
61    /// not in front of the eye, or the far plane is not beyond it. A host whose compositor
62    /// reports an unbounded far plane picks one before handing the frustum over.
63    #[error("the external frustum {frustum:?} encloses no volume")]
64    InvalidFrustum {
65        /// The frustum as supplied.
66        frustum: EyeFrustum,
67    },
68}

70/// What the view state keeps of an external view for the frame.
71#[derive(Clone, Copy, Debug)]
72pub(super) struct ExternalEye {
73    /// Origin of the local frame.
74    anchor: ExternalAnchor,
75    /// Local frame to eye space, as the host supplied it.
76    eye_from_local: Matrix4<f64>,
77    /// Eye space units per local-frame metre.
78    scale: f64,
79    /// The eye's frustum with clip distances in local-frame metres.
80    frustum: EyeFrustum,
81    /// The eye in unit-sphere coordinates, for the globe camera.
82    sphere: SphereEye,
83    /// Pixel focal length of the host frustum, unaffected by request overscan.
84    lod_focal_pixels: f64,
85    /// Physical scale used for cartographic styling, independent of gaze.
86    zoom: f64,
87    /// How much that zoom moved since the previous eye: a flight or a hand zoom in progress.
88    zoom_rate: f64,
89}

91/// Zoom change between consecutive eyes below which the eye counts as settled. A six second
92/// flight across seven zoom levels moves about 0.013 per frame at ninety frames a second.
93const SETTLED_ZOOM_RATE: f64 = 0.002;
94
95impl ViewState {
96    /// Drives the map from a host-supplied eye.
97    ///
98    /// The view matrix becomes the map center, zoom and angles. Where the globe is drawn, the
99    /// center is the point the eye's view axis meets on the sphere; on the flat map the view
100    /// becomes a [`CameraPose`], applied as [`set_camera_pose`](Self::set_camera_pose)
101    /// would. Those only steer the tile covering and the pixel scale: the frame is drawn with
102    /// matrices built from the eye itself, so a pitch beyond the map's limit, which the pose
103    /// clamps, still renders what the eye sees. The eye replaces the map's own camera and
104    /// perspective until [`clear_external_view`](Self::clear_external_view).
105    pub fn set_external_view(
106        &mut self,
107        external: ExternalView,
108        projection: &ProjectionType,
109    ) -> Result<(), ExternalViewError> {
110        let frustum = external.frustum;
111        let finite = [
112            frustum.left,
113            frustum.right,
114            frustum.top,
115            frustum.bottom,
116            frustum.near,
117            frustum.far,
118        ]
119        .iter()
120        .all(|value| value.is_finite());
121        if !finite
122            || frustum.left + frustum.right <= 0.0
123            || frustum.top + frustum.bottom <= 0.0
124            || frustum.near <= 0.0
125            || frustum.far <= frustum.near
126        {
127            return Err(ExternalViewError::InvalidFrustum { frustum });
128        }
129        let frame = EyeFrame::of(&external.view)?;
130        let body = self.body();
131        let sphere = SphereEye::place(&frame, external.anchor, body);
132        let on_sphere = sphere.pose(body);
133        let zoom = self.zoom_for_center_distance(on_sphere.distance_meters, on_sphere.center);
134        if projection.uses_globe_rendering(zoom.value()) {
135            self.set_center_and_angles(
136                on_sphere.center,
137                zoom,
138                on_sphere.bearing,
139                on_sphere.pitch,
140                on_sphere.roll,
141            );
142        } else {
143            self.set_eye_pose(
144                flat_pose_of(&frame, external.anchor, body),
145                external.anchor.altitude_meters,
146            );
147        }
148        let nadir = crate::projection::globe::unit_sphere_to_lat_lon(sphere.position.normalize());
149        let height = ((sphere.position.magnitude() - 1.0) * body.radius_meters).max(1.0);
150        let zoom = self.zoom_for_center_distance(height, nadir).value();
151        let zoom_rate = self
152            .external_eye
153            .map_or(0.0, |previous| (zoom - previous.zoom).abs());
154        self.external_eye = Some(ExternalEye {
155            anchor: external.anchor,
156            eye_from_local: external.view,
157            scale: frame.scale,
158            frustum: external.frustum.scaled(1.0 / frame.scale),
159            sphere,
160            lod_focal_pixels: self.height / (frustum.top + frustum.bottom),
161            zoom,
162            zoom_rate,
163        });
164        Ok(())
165    }

167    /// Whether a host-supplied eye drives the map.
168    pub fn has_external_view(&self) -> bool {
169        self.external_eye.is_some()

172    /// Whether the eye's zoom has come to rest; the map's own camera always counts as
173    /// settled. While the zoom moves, every level passed would get requests for ground that
174    /// is never looked at.
175    pub fn eye_settled(&self) -> bool {
176        self.external_eye
177            .is_none_or(|eye| eye.zoom_rate <= SETTLED_ZOOM_RATE)
178    }

180    /// Returns to the map's own perspective; the pose the external view left stays.
181    pub fn clear_external_view(&mut self) {
182        self.external_eye = None;
183    }

185    /// The projection an external eye gives the map, in the map's camera units, if an

188        self.external_frustum().map(|frustum| frustum.projection())

191    /// The external eye's frustum with clip distances in world pixels at the anchor.
192    pub(super) fn external_frustum(&self) -> Option<EyeFrustum> {
193        self.external_eye
194            .map(|eye| eye.frustum.scaled(self.anchor_pixels_per_meter(eye.anchor)))
195    }

197    /// World pixels per metre at a point, the Mercator scale of its latitude.
198    fn anchor_pixels_per_meter(&self, anchor: ExternalAnchor) -> f64 {
199        let world_size = TILE_SIZE * 2f64.powf(self.zoom().value());
200        world_size
201            / self
202                .body()
203                .circumference_at_latitude(anchor.position.latitude)
204    }
205
206    /// World space, x and y in pixels and z in metres, to the map's camera space, built from
207    /// the external eye rather than the map's angles; `None` without an external view.
208    ///
209    /// The camera space is in world pixels at the anchor, with y down as the map's own camera
210    /// space has it, so the same projection conventions apply on top.
211    pub(super) fn external_camera_matrix(&self) -> Option<Matrix4<f64>> {
212        let eye = self.external_eye?;
213        let world_size = TILE_SIZE * 2f64.powf(self.zoom().value());
214        let pixels_per_meter = self.anchor_pixels_per_meter(eye.anchor);
215        let anchor = mercator_from_lat_lon(eye.anchor.position);
216        let local_from_world =
217            Matrix4::from_nonuniform_scale(1.0 / pixels_per_meter, -1.0 / pixels_per_meter, 1.0)
218                * Matrix4::from_translation(Vector3::new(
219                    -anchor.x * world_size,
220                    -anchor.y * world_size,
221                    -eye.anchor.altitude_meters,
222                ));
223        Some(
224            FLIP_Y
225                * Matrix4::from_scale(pixels_per_meter / eye.scale)
226                * eye.eye_from_local
227                * local_from_world,
228        )
229    }

231    /// The view state with the external eye's frustum widened by `factor` on every side, so
232    /// a tile covering taken from it reaches beyond the frame; unchanged without an external
233    /// eye or with a factor of one.
234    pub fn overscanned(&self, factor: f64) -> Option<Self> {
235        let eye = self.external_eye?;
236        if factor == 1.0 {
237            return None;
238        }
239        let mut widened = self.clone();
240        widened.external_eye = Some(ExternalEye {
241            frustum: EyeFrustum {
242                left: eye.frustum.left * factor,
243                right: eye.frustum.right * factor,
244                top: eye.frustum.top * factor,
245                bottom: eye.frustum.bottom * factor,
246                ..eye.frustum
247            },
248            ..eye
249        });
250        Some(widened)
251    }

253    /// Fog distances for an external eye, in world pixels at the anchor: fixed by the eye's
254    /// height above the anchor's ground rather than by where it looks, so the haze on a
255    /// distant ridge does not change as the head turns. The far distance is the geometric horizon of the
256    /// body seen from that height; the fog starts a share of the way there.
257    pub(super) fn eye_fog_depth_range(&self) -> Option<(f64, f64)> {
258        let eye = self.external_eye?;
259        let position = eye.eye_from_local.invert()? * Vector4::new(0.0, 0.0, 0.0, 1.0);
260        let height = (position.z / position.w).max(MIN_FOG_HEIGHT_METERS);
261        let radius = self.body().circumference_meters() / std::f64::consts::TAU;
262        let far = (2.0 * radius * height)
263            .sqrt()
264            .clamp(MIN_FOG_FAR_METERS, MAX_FOG_FAR_METERS);
265        let pixels_per_meter = self.anchor_pixels_per_meter(eye.anchor);
266        Some((
267            far / FOG_FAR_TO_NEAR * pixels_per_meter,
268            far * pixels_per_meter,
269        ))
270    }

272    /// The view state of an eye at the same place looking straight down over a frame that
273    /// reaches `reach` times its height to every side; `None` without an external eye. A tile
274    /// covering taken from it holds what surrounds the eye, so a turn of the head finds the
275    /// tiles loaded.
276    pub fn surround(&self, reach: f64, projection: &ProjectionType) -> Option<Self> {
277        let eye = self.external_eye?;
278        if !self.eye_settled() {
279            return None;
280        }
281        let local_from_eye = eye.eye_from_local.invert()?;
282        let position = local_from_eye * Vector4::new(0.0, 0.0, 0.0, 1.0);
283        let position = position.truncate() / position.w;
284        // Local east, north and up map onto the eye's right, up and back: the eye faces down.
285        let view = Matrix4::from_scale(eye.scale) * Matrix4::from_translation(-position);
286        let frustum = EyeFrustum {
287            left: reach,
288            right: reach,
289            top: reach,
290            bottom: reach,
291            ..eye.frustum.scaled(eye.scale)
292        };
293        let mut below = self.clone();
294        below
295            .set_external_view(
296                ExternalView {
297                    anchor: eye.anchor,
298                    view,
299                    frustum,
300                },
301                projection,
302            )
303            .ok()?;
304        Some(below)
305    }

318    /// The map's own view in the form a host would supply, anchored at the map center on the
319    /// center elevation.
320    ///
321    /// Feeding it back through [`set_external_view`](Self::set_external_view) reproduces the
322    /// frame, which is also how a host learns the frame the map would render on its own.
323    pub fn external_view(&self) -> ExternalView {
324        let world_size = TILE_SIZE * 2f64.powf(self.zoom().value());
325        let center = self.camera().position();
326        let anchor = ExternalAnchor {
327            position: lat_lon_at_mercator(Point2::new(
328                center.x / world_size,
329                center.y / world_size,
330            )),
331            altitude_meters: self.center_elevation(),
332        };
333        let pixels_per_meter = self.pixels_per_meter();
334        let local_to_world =
335            Matrix4::from_translation(Vector3::new(center.x, center.y, self.center_elevation()))
336                * Matrix4::from_nonuniform_scale(pixels_per_meter, -pixels_per_meter, 1.0);
337        // The view it comes with maps local metres to the map's camera pixels, so the clip
338        // distances are in pixels too.
339        let frustum = self.external_frustum().unwrap_or_else(|| {
340            let (near_z, far_z) = self.depth_range(self.center_offset());
341            EyeFrustum::from_projection(FLIP_Y * self.perspective_matrix() * FLIP_Y, near_z, far_z)
342        });
343        ExternalView {
344            anchor,
345            view: FLIP_Y * self.camera_matrix() * local_to_world,
346            frustum,
347        }
348    }
349}

351/// Sine of the pitch below which a view counts as looking straight down.
352const LEVEL_PITCH_THRESHOLD: f64 = 1e-6;
353/// An eye on the ground still sees this far into the haze.
354const MIN_FOG_HEIGHT_METERS: f64 = 100.0;
355/// Bounds on the far fog distance, so the haze neither closes in nor vanishes at odd heights.
356const MIN_FOG_FAR_METERS: f64 = 20_000.0;
357const MAX_FOG_FAR_METERS: f64 = 600_000.0;
358/// The near fog distance is the far one over this.
359const FOG_FAR_TO_NEAR: f64 = 16.0;
360
361/// Where an eye is and which way it faces, in the local frame of an [`ExternalView`].
362#[derive(Clone, Copy, Debug)]
363pub(super) struct EyeFrame {
364    /// The eye in local-frame metres.
365    pub position: Vector3<f64>,
366    /// Unit vector along the eye's x axis.
367    pub right: Vector3<f64>,
368    /// Unit vector along the eye's y axis.
369    pub up: Vector3<f64>,
370    /// Unit vector along the eye's z axis, which points away from what it looks at.
371    pub back: Vector3<f64>,
372    /// Eye space units per local-frame metre.
373    pub scale: f64,
374}
375
376impl EyeFrame {
377    fn of(view: &Matrix4<f64>) -> Result<Self, ExternalViewError> {
378        if (0..4).any(|column| (0..4).any(|row| !view[column][row].is_finite())) {
379            return Err(ExternalViewError::SingularView);
380        }
381        let inverse = view.invert().ok_or(ExternalViewError::SingularView)?;
382        let eye = inverse * Vector4::new(0.0, 0.0, 0.0, 1.0);
383        let scale = (view * Vector4::new(1.0, 0.0, 0.0, 0.0))
384            .truncate()
385            .magnitude();
386        if eye.w.abs() <= f64::EPSILON || !scale.is_finite() || scale <= 0.0 {
387            return Err(ExternalViewError::SingularView);
388        }
389        let axis =
390            |direction: Vector3<f64>| (inverse * direction.extend(0.0)).truncate().normalize();
391        Ok(Self {
392            position: eye.truncate() / eye.w,
393            right: axis(Vector3::unit_x()),
394            up: axis(Vector3::unit_y()),
395            back: axis(Vector3::unit_z()),
396            scale,
397        })
398    }
399
400    /// Where the eye looks.
401    pub fn forward(&self) -> Vector3<f64> {
402        -self.back
403    }
404}

406/// Bearing, pitch and roll of an eye whose right and forward directions are given in a
407/// frame with x east, y north and z up.
408pub(super) fn view_angles(
409    right: Vector3<f64>,
410    forward: Vector3<f64>,
411) -> (Rad<f64>, Rad<f64>, Rad<f64>) {
412    let pitch = (-forward.z).clamp(-1.0, 1.0).acos();
413    // Looking straight down, the forward direction says nothing about the bearing; the right
414    // vector, which lies on the ground, still does. The threshold sits above the rounding of
415    // an arc cosine near one, which leaves a straight-down view a few hundred nanoradians of
416    // pitch.
417    let bearing = if pitch.sin() > LEVEL_PITCH_THRESHOLD {
418        forward.x.atan2(forward.y)
419    } else {
420        (-right.y).atan2(right.x)
421    };
422    let level_right = Vector3::new(bearing.cos(), -bearing.sin(), 0.0);
423    let level_up = level_right.cross(forward);
424    // The map turns the view by minus the roll about its axis, which is where the sign comes
425    // from.
426    let roll = (-right.dot(level_up)).atan2(right.dot(level_right));
427    (Rad(bearing), Rad(pitch), Rad(roll))
428}

430/// The camera pose an eye describes on the flat map, where the local frame's axes are the
431/// map's.
432fn flat_pose_of(frame: &EyeFrame, anchor: ExternalAnchor, body: Body) -> CameraPose {
433    let (bearing, pitch, roll) = view_angles(frame.right, frame.forward());
434    let units_per_meter =
435        1.0 / (body.circumference_meters() * anchor.position.latitude.to_radians().cos());
436    let anchor_mercator = mercator_from_lat_lon(anchor.position);
437    CameraPose {
438        position: lat_lon_at_mercator(Point2::new(
439            anchor_mercator.x + frame.position.x * units_per_meter,
440            anchor_mercator.y - frame.position.y * units_per_meter,
441        )),
442        altitude_meters: anchor.altitude_meters + frame.position.z,
443        bearing: bearing.into(),
444        pitch: pitch.into(),
445        roll: roll.into(),
446    }
447}
