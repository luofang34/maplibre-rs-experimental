1//! Drag and zoom on a free-globe camera, which turn the camera about the body's center and so
2//! carry it over a pole as anywhere else.
3
4use cgmath::{InnerSpace, Point2, Vector3};
5
6use super::{turn_between, ViewState};
7use crate::{projection::globe::lat_lon_to_unit_sphere, render::projection::globe_camera_for_view};
8
9/// Corrections that bring a zoom's anchor back under its pixel.
10const ANCHOR_PASSES: usize = 6;
11
12/// Times a zoom step that leaves no drawable camera is halved before the zoom stops, which
13/// brings it within a thousandth of the step of the last distance a camera can be drawn from.
14const ZOOM_HALVINGS: usize = 10;
15
16impl ViewState {
17    /// The point of the sphere at the center's elevation under `pixel`, or on its horizon where
18    /// the pixel's ray passes it, in the body's frame.
19    fn grabbed(&self, pixel: Point2<f64>) -> Option<Vector3<f64>> {
20        let camera = globe_camera_for_view(self).ok()?;
21        let elevation = self.center_elevation();
22        let location = camera
23            .screen_point_to_location_at(pixel, elevation)
24            .or_else(|| camera.screen_point_to_location(pixel))?;
25        Some(lat_lon_to_unit_sphere(location))
26    }
27
28    /// Drags the free-globe camera so the ground under `from` moves under `to`, turning the
29    /// whole camera about the body's center. Returns whether it moved.
30    pub fn drag_free_globe(&mut self, from: Point2<f64>, to: Point2<f64>) -> bool {
31        if self.navigation_mode() != super::NavigationMode::FreeGlobe {
32            return false;
33        }
34        let (Some(grabbed), Some(under)) = (self.grabbed(from), self.grabbed(to)) else {
35            return false;
36        };
37        // The camera sees a point turned by the opposite of its own turn; the point under `to`
38        // must turn onto the grabbed one. Eye and target turn together about the body's center,
39        // keeping their heights, so a camera that could be drawn still can.
40        self.turn_free_globe(turn_between(under, grabbed));
41        true
42    }
43
44    /// Zooms the free-globe camera by `zoom_delta` keeping the ground under `pixel` there, or
45    /// by as much of it as leaves a camera that can be drawn, as one looking up at a raised
46    /// target cannot once zooming out takes its eye under the ground. Returns whether it moved.
47    pub fn zoom_free_globe(&mut self, pixel: Point2<f64>, zoom_delta: f64) -> bool {
48        if self.navigation_mode() != super::NavigationMode::FreeGlobe {
49            return false;
50        }
51        let Some(anchor) = self.grabbed(pixel) else {
52            return false;
53        };
54        let mut delta = zoom_delta;
55        for _ in 0..=ZOOM_HALVINGS {
56            let before = self.clone();
57            self.zoom_free_globe_about(anchor, pixel, delta);
58            if self.keep_if_drawable(before) {
59                return true;
60            }
61            delta *= 0.5;
62        }
63        false
64    }
65
66    /// Zooms by `zoom_delta`, then turns the camera until `anchor` lies under `pixel` again.
67    fn zoom_free_globe_about(&mut self, anchor: Vector3<f64>, pixel: Point2<f64>, zoom_delta: f64) {
68        self.scale_free_globe_distance(2_f64.powf(-zoom_delta));
69        for _ in 0..ANCHOR_PASSES {
70            let Some(under) = self.grabbed(pixel) else {
71                break;
72            };
73            if (under - anchor).magnitude() < 1e-15 {
74                break;
75            }
76            self.turn_free_globe(turn_between(under, anchor));
77        }
78    }
79
80    /// Keeps a gesture's result if the globe camera can be drawn from it, the same rule
81    /// [`ViewState::set_globe_pose`] holds a placed pose to; otherwise puts back `before`.
82    pub(super) fn keep_if_drawable(&mut self, before: ViewState) -> bool {
83        if globe_camera_for_view(self).is_ok() {
84            true
85        } else {
86            *self = before;
87            false
88        }
89    }
