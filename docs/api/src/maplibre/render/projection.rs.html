34/// View-wide globe projection values uploaded as a uniform buffer.
35#[repr(C)]
36#[derive(Clone, Copy, Debug, Pod, Zeroable)]
37pub struct ShaderProjectionData {
38    /// Matrix projecting the unit globe to WebGPU clip space.
39    pub main_matrix: Mat4x4f32,
40    /// Unit-sphere horizon plane.
41    pub clipping_plane: Vec4f32,
42    /// Mercator-to-globe interpolation factor.
43    pub transition: f32,
44    /// Clip-space w of the view center: the camera-to-center distance in Mercator pixels,
45    /// blended with the globe's value, so screen-space sizes can scale with distance.
46    pub center_clip_w: f32,
47    /// Radius of the body in metres, which scales elevations onto the unit sphere.
48    pub radius_meters: f32,
49    /// Whether viewport symbols use a world-up basis for a tracked external eye.
50    pub external_view: f32,
51    /// x: the globe's angle per pixel relative to a world of 512 pixels at the zoom, which the
52    /// extent of a circle lying on the globe follows; y: 1 when the globe centre references
53    /// below are set; z: 1 when symbol sizes are independent of target depth; w is padding.
54    pub globe_circle: [f32; 4],
55    /// The Mercator position of the globe centre the shaders project relative to, in x and y,
56    /// exactly representable in f32 so tile origins difference from it without rounding; the
57    /// other lanes are padding.
58    pub globe_center: [f32; 4],
59    /// Clip position of that centre on the unit sphere.
60    pub globe_center_clip: [f32; 4],
61    /// Clip-space vector from the body's centre to it, which a height scales.
62    pub globe_center_radial: [f32; 4],
63}

65impl ShaderProjectionData {
66    /// Creates the view-wide subset of renderer projection data.
67    pub fn from_renderer_data(data: RendererProjectionData) -> Self {
68        Self {
69            main_matrix: data.main_matrix.into(),
70            clipping_plane: data.clipping_plane.into(),
71            transition: data.projection_transition,
72            center_clip_w: 1.0,
73            radius_meters: Body::EARTH.radius_meters as f32,
74            external_view: 0.0,
75            globe_circle: [1.0, 0.0, 0.0, 0.0],
76            globe_center: [0.0; 4],
77            globe_center_clip: [0.0; 4],
78            globe_center_radial: [0.0; 4],
79        }
80    }
81}

100/// GPU buffer and binding shared by projection-aware pipelines.
101pub struct ProjectionGpuResources {
102    bind_group_layout: wgpu::BindGroupLayout,
103    bind_group: wgpu::BindGroup,
104    buffer: wgpu::Buffer,
105    /// Never-updated default projection for draws into drape textures.
106    flat_bind_group: wgpu::BindGroup,
107}
108
109impl ProjectionGpuResources {
110    /// Allocates the projection uniform and its stable bind-group layout.
111    pub fn new(device: &wgpu::Device, queue: &crate::render::upload_queue::UploadQueue) -> Self {
112        let bind_group_layout = device.create_bind_group_layout(&wgpu::BindGroupLayoutDescriptor {
113            label: Some("projection uniform layout"),
114            entries: &[wgpu::BindGroupLayoutEntry {
115                binding: 0,
116                visibility: wgpu::ShaderStages::VERTEX,
117                ty: wgpu::BindingType::Buffer {
118                    ty: wgpu::BufferBindingType::Uniform,
119                    has_dynamic_offset: false,
120                    min_binding_size: wgpu::BufferSize::new(
121                        std::mem::size_of::<ShaderProjectionData>() as u64,
122                    ),
123                },
124                count: None,
125            }],
126        });
127        let initial_data = ShaderProjectionData::default();
128        let buffer = queue.create_buffer_init(
129            device,
130            &wgpu::util::BufferInitDescriptor {
131                label: Some("projection uniform buffer"),
132                contents: bytemuck::bytes_of(&initial_data),
133                usage: wgpu::BufferUsages::UNIFORM | wgpu::BufferUsages::COPY_DST,
134            },
135        );
136        let bind_group = device.create_bind_group(&wgpu::BindGroupDescriptor {
137            label: Some("projection uniform bind group"),
138            layout: &bind_group_layout,
139            entries: &[wgpu::BindGroupEntry {
140                binding: 0,
141                resource: buffer.as_entire_binding(),
142            }],
143        });
144        let flat_buffer = queue.create_buffer_init(
145            device,
146            &wgpu::util::BufferInitDescriptor {
147                label: Some("flat projection uniform buffer"),
148                contents: bytemuck::bytes_of(&initial_data),
149                usage: wgpu::BufferUsages::UNIFORM,
150            },
151        );
152        let flat_bind_group = device.create_bind_group(&wgpu::BindGroupDescriptor {
153            label: Some("flat projection uniform bind group"),
154            layout: &bind_group_layout,
155            entries: &[wgpu::BindGroupEntry {
156                binding: 0,
157                resource: flat_buffer.as_entire_binding(),
158            }],
159        });
160        Self {
161            bind_group_layout,
162            bind_group,
163            buffer,
164            flat_bind_group,
165        }
166    }
167
168    /// Returns the bind group a draw asks for: the camera's projection or the flat default.
169    pub fn bind_group_for(&self, binding: ProjectionBinding) -> &wgpu::BindGroup {
170        match binding {
171            ProjectionBinding::View => &self.bind_group,
172            ProjectionBinding::Flat => &self.flat_bind_group,
173        }
174    }
175
176    /// Returns the layout used when creating projection-aware pipelines.
177    pub fn bind_group_layout(&self) -> &wgpu::BindGroupLayout {
178        &self.bind_group_layout
179    }
180
181    /// Returns the bind group used by projection-aware draw commands.
182    pub fn bind_group(&self) -> &wgpu::BindGroup {
183        &self.bind_group
184    }
185
186    /// Uploads projection state for the current frame.
187    pub fn upload(
188        &self,
189        queue: &crate::render::upload_queue::UploadQueue,
190        data: ShaderProjectionData,
191    ) {
192        queue.write_buffer(&self.buffer, 0, bytemuck::bytes_of(&data));
193    }
194}

196/// Failure while deriving projection state from the current map view.
197#[derive(Debug, Error)]
198pub enum ProjectionStateError {
199    /// Mercator tile selection failed.
200    #[error("failed to select mercator tiles")]
201    MercatorCovering {
202        /// Underlying covering error.
203        #[source]
204        source: MercatorCoveringError,
205    },
206    /// Globe camera state could not be constructed.
207    #[error("failed to construct globe camera state: {source}")]
208    GlobeCamera {
209        /// Underlying camera error.
210        #[source]
211        source: GlobeCameraError,
212    },
213    /// A globe matrix or clipping plane cannot be represented as 32-bit floats.
214    #[error("globe projection state cannot be represented as f32")]
215    FloatConversion,
216    /// Globe tile traversal failed.
217    #[error("failed to select visible globe tiles")]
218    GlobeCovering {
219        /// Underlying covering error.
220        #[source]
221        source: GlobeCoveringError,
222    },
223}

225/// A freely placed camera has no canonical map center for depth-based label scaling.
226pub(crate) fn fixed_symbol_scale(view: &ViewState) -> bool {
227    view.has_external_view() || view.navigation_mode() == NavigationMode::FreeGlobe
228}

230/// Derives the view-wide projection uniform from style and camera state.
231pub fn projection_data_for_view(
232    style: &Style,
233    view_state: &ViewState,
234) -> Result<ShaderProjectionData, ProjectionStateError> {
235    let transition = style.projection.as_ref().map_or(0.0, |specification| {
236        specification
237            .projection_type
238            .globe_transition(view_state.zoom().value())
239    });
240    let mercator_center_w = view_state.camera_to_center_distance() as f32;
241    let radius_meters = view_state.body().radius_meters as f32;
242    if transition == 0.0 {
243        return Ok(ShaderProjectionData {
244            external_view: f32::from(view_state.has_external_view()),
245            center_clip_w: mercator_center_w,
246            globe_circle: [1.0, 0.0, f32::from(fixed_symbol_scale(view_state)), 0.0],
247            radius_meters,
248            ..ShaderProjectionData::default()
249        });
250    }
251
252    let globe = globe_camera_for_view(view_state)?;
253    let globe_center_clip = globe.wgpu_view_projection() * globe.target().extend(1.0);
254    let globe_center_w = globe_center_clip.w as f32;
255    let globe_matrix = globe
256        .wgpu_view_projection()
257        .cast::<f32>()
258        .ok_or(ProjectionStateError::FloatConversion)?;
259    let clipping_plane = globe
260        .clipping_plane()
261        .cast::<f32>()
262        .ok_or(ProjectionStateError::FloatConversion)?;
263    let data = compose_projection_data(
264        ProjectionMatrices {
265            mercator: Matrix4::from_scale(1.0),
266            globe: globe_matrix,
267        },
268        clipping_plane,
269        transition,
270        ProjectionDataParams {
271            apply_globe_matrix: true,
272            ..ProjectionDataParams::default()
273        },
274    );
275    let shader = ShaderProjectionData {
276        external_view: f32::from(view_state.has_external_view()),
277        center_clip_w: mercator_center_w + (globe_center_w - mercator_center_w) * transition,
278        radius_meters,
279        globe_circle: [
280            globe.circle_radius_correction() as f32,
281            0.0,
282            f32::from(fixed_symbol_scale(view_state)),
283            0.0,
284        ],
285        ..ShaderProjectionData::from_renderer_data(data)
286    };
287    // Below this zoom f32 rounding stays far under a pixel, and projecting from the global
288    // coordinates keeps every edge where GL JS, which does the same, draws it.
289    let references = (view_state.zoom().value() >= PRECISE_GLOBE_ZOOM)
290        .then(|| globe_center_references(&globe))
291        .flatten();
292    Ok(match references {
293        Some([center, clip, radial]) => ShaderProjectionData {
294            globe_circle: [shader.globe_circle[0], 1.0, shader.globe_circle[2], 0.0],
295            globe_center: center,
296            globe_center_clip: clip,
297            globe_center_radial: radial,
298            ..shader
299        },
300        None => shader,
301    })
302}

304/// The zoom from which the shaders project globe positions relative to the view's centre: a
305/// pixel is about 20 m there, and the metres global f32 coordinates round to start to show.
306const PRECISE_GLOBE_ZOOM: f64 = 12.0;
307
308/// The centre the shaders project globe positions relative to: the f32 Mercator position
309/// nearest the globe camera's centre on the ground, with its clip position and body-centre
310/// vector computed in f64. Positions built from global f32 coordinates round to metres, which
311/// a close view of the globe draws as geometry that slides; relative to this centre they keep
312/// their separation.
313fn globe_center_references(globe: &GlobeCameraState) -> Option<[[f32; 4]; 3]> {
314    let center = crate::terrain::sightline::lat_lon_to_mercator(globe.center());
315    // The centre need only be near the view; one f32 can hold exactly is differenced exactly.
316    let (x_f32, y_f32) = (center.x as f32, center.y as f32);
317    let (x, y) = (f64::from(x_f32), f64::from(y_f32));
318    let longitude = x * std::f64::consts::TAU + std::f64::consts::PI;
319    let tangent_half_latitude = (std::f64::consts::PI - y * std::f64::consts::TAU).exp();
320    let denominator = tangent_half_latitude * tangent_half_latitude + 1.0;
321    let sin_latitude = (tangent_half_latitude * tangent_half_latitude - 1.0) / denominator;
322    let cos_latitude = 2.0 * tangent_half_latitude / denominator;
323    let surface = cgmath::Vector3::new(
324        longitude.sin() * cos_latitude,
325        sin_latitude,
326        longitude.cos() * cos_latitude,
327    );
328    let matrix = globe.wgpu_view_projection();
329    let clip = |w: f64| {
330        let clip = matrix * surface.extend(w);
331        let lanes = [clip.x as f32, clip.y as f32, clip.z as f32, clip.w as f32];
332        lanes.iter().all(|lane| lane.is_finite()).then_some(lanes)
333    };
334    Some([[x_f32, y_f32, 0.0, 0.0], clip(1.0)?, clip(0.0)?])
335}

337/// Constructs the vertical-perspective camera matching the current map view.
338pub fn globe_camera_for_view(
339    view_state: &ViewState,
340) -> Result<GlobeCameraState, ProjectionStateError> {
341    let external_eye = view_state.external_globe_eye();
342    // A free-globe camera is drawn from its pose; a host's eye replaces either.
343    let pose = view_state.pose_view().filter(|_| external_eye.is_none());
344    let world_size = TILE_SIZE
345        * 2.0_f64.powf(
346            pose.as_ref()
347                .map_or(view_state.zoom().value(), |pose| pose.style_zoom),
348        );
349    let camera_position = view_state.camera().position();
350    let center = pose.as_ref().map_or_else(
351        || mercator_world_to_lat_lon(camera_position.x, camera_position.y, world_size),
352        |pose| pose.center,
353    );
354    let angle = |free: Option<f64>, flat: f64| free.unwrap_or(flat);
355    let options = GlobeCameraOptions {
356        width: view_state.width(),
357        height: view_state.height(),
358        field_of_view_degrees: external_eye.map_or_else(
359            || view_state.field_of_view().0.to_degrees(),
360            |eye| eye.frustum.vertical_field_of_view().0.to_degrees(),
361        ),
362        center,
363        world_size,
364        bearing_degrees: angle(
365            pose.as_ref().map(|pose| pose.bearing_degrees),
366            view_state.camera().get_bearing().0.to_degrees(),
367        ),
368        pitch_degrees: angle(
369            pose.as_ref().map(|pose| pose.pitch_degrees),
370            view_state.camera().get_pitch().0.to_degrees(),
371        ),
372        roll_degrees: angle(
373            pose.as_ref().map(|pose| pose.roll_degrees),
374            view_state.camera().get_roll().0.to_degrees(),
375        ),
376        center_offset: external_eye
377            .map_or_else(|| view_state.center_offset(), |_| Point2::new(0.0, 0.0)),
378        body: view_state.body(),
379        // A host's eye is placed where it is; the center's terrain must not move it.
380        target_elevation_meters: if external_eye.is_none() && view_state.globe_orbits_center() {
381            view_state.center_elevation()
382        } else {
383            0.0
384        },
385        radius_pixels: pose.as_ref().map(|pose| pose.radius_pixels),
386    };
387    match external_eye {
388        Some(eye) => GlobeCameraState::from_external_eye(options, eye),
389        None => GlobeCameraState::new(options),
390    }
391    .map_err(|source| ProjectionStateError::GlobeCamera { source })
392}

394pub(crate) fn mercator_world_to_lat_lon(x: f64, y: f64, world_size: f64) -> LatLon {
395    let longitude = x / world_size * 360.0 - 180.0;
396    let latitude = (std::f64::consts::PI * (1.0 - 2.0 * y / world_size))
397        .sinh()
398        .atan()
399        .to_degrees();
400    LatLon::new(latitude, longitude)
401}
