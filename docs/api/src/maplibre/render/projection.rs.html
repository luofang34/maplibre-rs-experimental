38/// View-wide globe projection values uploaded as a uniform buffer.
39#[repr(C)]
40#[derive(Clone, Copy, Debug, Pod, Zeroable)]
41pub struct ShaderProjectionData {
42    /// Matrix projecting the unit globe to WebGPU clip space.
43    pub main_matrix: Mat4x4f32,
44    /// Unit-sphere horizon plane.
45    pub clipping_plane: Vec4f32,
46    /// Mercator-to-globe interpolation factor.
47    pub transition: f32,
48    /// Clip-space w of the view center: the camera-to-center distance in Mercator pixels,
49    /// blended with the globe's value, so screen-space sizes can scale with distance.
50    pub center_clip_w: f32,
51    /// Radius of the body in metres, which scales elevations onto the unit sphere.
52    pub radius_meters: f32,
53    /// Whether viewport symbols use a world-up basis for a tracked external eye.
54    pub external_view: f32,
55    /// x: the globe's angle per pixel relative to a world of 512 pixels at the zoom, which the
56    /// extent of a circle lying on the globe follows; y: 1 when the globe centre references
57    /// below are set; z: 1 when symbol sizes are independent of target depth; w is padding.
58    pub globe_circle: [f32; 4],
59    /// The Mercator position of the globe centre the shaders project relative to, in x and y,
60    /// exactly representable in f32 so tile origins difference from it without rounding; the
61    /// other lanes are padding.
62    pub globe_center: [f32; 4],
63    /// Clip position of that centre on the unit sphere.
64    pub globe_center_clip: [f32; 4],
65    /// Clip-space vector from the body's centre to it, which a height scales.
66    pub globe_center_radial: [f32; 4],
67}
68
69impl ShaderProjectionData {
70    /// Creates the view-wide subset of renderer projection data.
71    pub fn from_renderer_data(data: RendererProjectionData) -> Self {
72        Self {
73            main_matrix: data.main_matrix.into(),
74            clipping_plane: data.clipping_plane.into(),
75            transition: data.projection_transition,
76            center_clip_w: 1.0,
77            radius_meters: Body::EARTH.radius_meters as f32,
78            external_view: 0.0,
79            globe_circle: [1.0, 0.0, 0.0, 0.0],
80            globe_center: [0.0; 4],
81            globe_center_clip: [0.0; 4],
82            globe_center_radial: [0.0; 4],
83        }
84    }
85}

104/// GPU buffer and binding shared by projection-aware pipelines.
105pub struct ProjectionGpuResources {
106    bind_group_layout: wgpu::BindGroupLayout,
107    bind_group: wgpu::BindGroup,
108    buffer: wgpu::Buffer,
109    /// Never-updated default projection for draws into drape textures.
110    flat_bind_group: wgpu::BindGroup,
111    /// Latest uploaded `ShaderProjectionData::transition` (`f32::to_bits`) for pipeline selection.
112    transition_bits: std::sync::atomic::AtomicU32,
113}
114
115impl ProjectionGpuResources {
116    /// Allocates the projection uniform and its stable bind-group layout.
117    pub fn new(device: &wgpu::Device, queue: &crate::render::upload_queue::UploadQueue) -> Self {
118        let bind_group_layout = crate::render::resource::create_bind_group_layout_cached(
119            device,
120            Some("projection uniform layout"),
121            &[wgpu::BindGroupLayoutEntry {
122                binding: 0,
123                visibility: wgpu::ShaderStages::VERTEX,
124                ty: wgpu::BindingType::Buffer {
125                    ty: wgpu::BufferBindingType::Uniform,
126                    has_dynamic_offset: false,
127                    min_binding_size: wgpu::BufferSize::new(
128                        std::mem::size_of::<ShaderProjectionData>() as u64,
129                    ),
130                },
131                count: None,
132            }],
133        );
134        let initial_data = ShaderProjectionData::default();
135        let buffer = queue.create_buffer_init(
136            device,
137            &wgpu::util::BufferInitDescriptor {
138                label: Some("projection uniform buffer"),
139                contents: bytemuck::bytes_of(&initial_data),
140                usage: wgpu::BufferUsages::UNIFORM | wgpu::BufferUsages::COPY_DST,
141            },
142        );
143        let bind_group = device.create_bind_group(&wgpu::BindGroupDescriptor {
144            label: Some("projection uniform bind group"),
145            layout: &bind_group_layout,
146            entries: &[wgpu::BindGroupEntry {
147                binding: 0,
148                resource: buffer.as_entire_binding(),
149            }],
150        });
151        let flat_buffer = queue.create_buffer_init(
152            device,
153            &wgpu::util::BufferInitDescriptor {
154                label: Some("flat projection uniform buffer"),
155                contents: bytemuck::bytes_of(&initial_data),
156                usage: wgpu::BufferUsages::UNIFORM,
157            },
158        );
159        let flat_bind_group = device.create_bind_group(&wgpu::BindGroupDescriptor {
160            label: Some("flat projection uniform bind group"),
161            layout: &bind_group_layout,
162            entries: &[wgpu::BindGroupEntry {
163                binding: 0,
164                resource: flat_buffer.as_entire_binding(),
165            }],
166        });
167        Self {
168            bind_group_layout,
169            bind_group,
170            buffer,
171            flat_bind_group,
172            transition_bits: std::sync::atomic::AtomicU32::new(initial_data.transition.to_bits()),
173        }
174    }
175
176    /// Returns whether `binding` resolves to an active globe projection (`transition > 0.0`).
177    pub fn is_globe_active(&self, binding: ProjectionBinding) -> bool {
178        matches!(binding, ProjectionBinding::View)
179            && f32::from_bits(
180                self.transition_bits
181                    .load(std::sync::atomic::Ordering::Relaxed),
182            ) > 0.0
183    }
184
185    /// Returns the bind group a draw asks for: the camera's projection or the flat default.
186    pub fn bind_group_for(&self, binding: ProjectionBinding) -> &wgpu::BindGroup {
187        match binding {
188            ProjectionBinding::View => &self.bind_group,
189            ProjectionBinding::Flat => &self.flat_bind_group,
190        }
191    }
192
193    /// Returns the layout used when creating projection-aware pipelines.
194    pub fn bind_group_layout(&self) -> &wgpu::BindGroupLayout {
195        &self.bind_group_layout
196    }
197
198    /// Returns the bind group used by projection-aware draw commands.
199    pub fn bind_group(&self) -> &wgpu::BindGroup {
200        &self.bind_group
201    }
202
203    /// Uploads projection state for the current frame.
204    pub fn upload(
205        &self,
206        queue: &crate::render::upload_queue::UploadQueue,
207        data: ShaderProjectionData,
208    ) {
209        self.transition_bits.store(
210            data.transition.to_bits(),
211            std::sync::atomic::Ordering::Relaxed,
212        );
213        queue.write_buffer(&self.buffer, 0, bytemuck::bytes_of(&data));
214    }
215}

217/// Failure while deriving projection state from the current map view.
218#[derive(Debug, Error)]
219pub enum ProjectionStateError {
220    /// Mercator tile selection failed.
221    #[error("failed to select mercator tiles")]
222    MercatorCovering {
223        /// Underlying covering error.
224        #[source]
225        source: MercatorCoveringError,
226    },
227    /// Globe camera state could not be constructed.
228    #[error("failed to construct globe camera state: {source}")]
229    GlobeCamera {
230        /// Underlying camera error.
231        #[source]
232        source: GlobeCameraError,
233    },
234    /// A globe matrix or clipping plane cannot be represented as 32-bit floats.
235    #[error("globe projection state cannot be represented as f32")]
236    FloatConversion,
237    /// Globe tile traversal failed.
238    #[error("failed to select visible globe tiles")]
239    GlobeCovering {
240        /// Underlying covering error.
241        #[source]
242        source: GlobeCoveringError,
243    },
244}

247pub(crate) fn fixed_symbol_scale(view: &ViewState) -> bool {
248    view.has_external_view() || view.navigation_mode() == NavigationMode::FreeGlobe

251pub(crate) fn flat_shader_projection(view_state: &ViewState) -> ShaderProjectionData {
252    ShaderProjectionData {
253        external_view: f32::from(view_state.has_external_view()),
254        center_clip_w: view_state.camera_to_center_distance() as f32,
255        globe_circle: [1.0, 0.0, f32::from(fixed_symbol_scale(view_state)), 0.0],
256        radius_meters: view_state.body().radius_meters as f32,
257        ..ShaderProjectionData::default()
258    }
259}

261pub(crate) fn globe_shader_projection(
262    view_state: &ViewState,
263    transition: f32,
264    globe: &GlobeCameraState,
265) -> Result<ShaderProjectionData, ProjectionStateError> {
266    let mercator_center_w = view_state.camera_to_center_distance() as f32;
267    let radius_meters = view_state.body().radius_meters as f32;
268    let globe_center_clip = globe.wgpu_view_projection() * globe.target().extend(1.0);
269    let globe_center_w = globe_center_clip.w as f32;
270    let globe_matrix = globe
271        .wgpu_view_projection()
272        .cast::<f32>()
273        .ok_or(ProjectionStateError::FloatConversion)?;
274    let clipping_plane = globe
275        .clipping_plane()
276        .cast::<f32>()
277        .ok_or(ProjectionStateError::FloatConversion)?;
278    let data = compose_projection_data(
279        ProjectionMatrices {
280            mercator: Matrix4::from_scale(1.0),
281            globe: globe_matrix,
282        },
283        clipping_plane,
284        transition,
285        ProjectionDataParams {
286            apply_globe_matrix: true,
287            ..ProjectionDataParams::default()
288        },
289    );
290    let shader = ShaderProjectionData {
291        external_view: f32::from(view_state.has_external_view()),
292        center_clip_w: mercator_center_w + (globe_center_w - mercator_center_w) * transition,
293        radius_meters,
294        globe_circle: [
295            globe.circle_radius_correction() as f32,
296            0.0,
297            f32::from(fixed_symbol_scale(view_state)),
298            0.0,
299        ],
300        ..ShaderProjectionData::from_renderer_data(data)
301    };
302    // Below this zoom f32 rounding stays far under a pixel, and projecting from the global
303    // coordinates keeps every edge where GL JS, which does the same, draws it. Host external
304    // eyes always project relative to the sub-camera point so polar and high-altitude 6-DoF
305    // views never drop out of f64 reference precision.
306    let references = (view_state.zoom().value() >= PRECISE_GLOBE_ZOOM
307        || view_state.has_external_view())
308    .then(|| globe_center_references(globe))
309    .flatten();
310    Ok(match references {
311        Some([center, clip, radial]) => ShaderProjectionData {
312            globe_circle: [shader.globe_circle[0], 1.0, shader.globe_circle[2], 0.0],
313            globe_center: center,
314            globe_center_clip: clip,
315            globe_center_radial: radial,
316            ..shader
317        },
318        None => shader,
319    })
320}

322/// Derives the view-wide projection uniform from style and camera state.
323pub fn projection_data_for_view(
324    style: &Style,
325    view_state: &ViewState,
326) -> Result<ShaderProjectionData, ProjectionStateError> {
327    let transition = style.projection.as_ref().map_or(0.0, |specification| {
328        specification
329            .projection_type
330            .globe_transition(view_state.zoom().value())
331    });
332    if transition == 0.0 {
333        return Ok(flat_shader_projection(view_state));
334    }
335    let globe = globe_camera_for_view(view_state)?;
336    globe_shader_projection(view_state, transition, &globe)
337}

339/// The zoom from which the shaders project globe positions relative to the view's centre: a
340/// pixel is about 20 m there, and the metres global f32 coordinates round to start to show.
341const PRECISE_GLOBE_ZOOM: f64 = 12.0;
342
343/// The centre the shaders project globe positions relative to: the f32 Mercator position
344/// nearest the globe camera's centre on the ground, with its clip position and body-centre
345/// vector computed in f64. Positions built from global f32 coordinates round to metres, which
346/// a close view of the globe draws as geometry that slides; relative to this centre they keep
347/// their separation.
348fn globe_center_references(globe: &GlobeCameraState) -> Option<[[f32; 4]; 3]> {
349    let center_ll = if globe.is_external_eye() {
350        let mag = globe.camera_position().magnitude();
351        if mag > 1e-12 {
352            unit_sphere_to_lat_lon(globe.camera_position() / mag)
353        } else {
354            globe.center()
355        }
356    } else {
357        globe.center()
358    };
359    // Unclamped Web Mercator y keeps the reference right under polar cameras beyond 85.05°
360    // instead of 550 km away at the Mercator rim; clamping only 11 cm from ±90° keeps tan finite.
361    let lat_rad = center_ll
362        .latitude
363        .clamp(-89.999_999, 89.999_999)
364        .to_radians();
365    let x_unclamped = center_ll.longitude / 360.0 + 0.5;
366    let y_unclamped = (1.0 - lat_rad.tan().asinh() / std::f64::consts::PI) * 0.5;
367    // The centre need only be near the view; one f32 can hold exactly is differenced exactly.
368    let (x_f32, y_f32) = (x_unclamped as f32, y_unclamped as f32);
369    let (x, y) = (f64::from(x_f32), f64::from(y_f32));
370    let longitude = x * std::f64::consts::TAU + std::f64::consts::PI;
371    let tangent_half_latitude = (std::f64::consts::PI - y * std::f64::consts::TAU).exp();
372    let denominator = tangent_half_latitude * tangent_half_latitude + 1.0;
373    let sin_latitude = (tangent_half_latitude * tangent_half_latitude - 1.0) / denominator;
374    let cos_latitude = 2.0 * tangent_half_latitude / denominator;
375    let surface = cgmath::Vector3::new(
376        longitude.sin() * cos_latitude,
377        sin_latitude,
378        longitude.cos() * cos_latitude,
379    );
380    let matrix = globe.wgpu_view_projection();
381    let clip = |w: f64| {
382        let clip = matrix * surface.extend(w);
383        let lanes = [clip.x as f32, clip.y as f32, clip.z as f32, clip.w as f32];
384        lanes.iter().all(|lane| lane.is_finite()).then_some(lanes)
385    };
386    Some([[x_f32, y_f32, 0.0, 0.0], clip(1.0)?, clip(0.0)?])
387}

389/// Constructs the vertical-perspective camera matching the current map view.
390pub fn globe_camera_for_view(
391    view_state: &ViewState,
392) -> Result<GlobeCameraState, ProjectionStateError> {
393    let external_eye = view_state.external_globe_eye();
394    // A free-globe camera is drawn from its pose; a host's eye replaces either.
395    let pose = view_state.pose_view().filter(|_| external_eye.is_none());
396    let world_size = TILE_SIZE
397        * 2.0_f64.powf(
398            pose.as_ref()
399                .map_or(view_state.zoom().value(), |pose| pose.style_zoom),
400        );
401    let camera_position = view_state.camera().position();
402    let center = pose.as_ref().map_or_else(
403        || mercator_world_to_lat_lon(camera_position.x, camera_position.y, world_size),
404        |pose| pose.center,
405    );
406    let angle = |free: Option<f64>, flat: f64| free.unwrap_or(flat);
407    let options = GlobeCameraOptions {
408        width: view_state.width(),
409        height: view_state.height(),
410        field_of_view_degrees: external_eye.map_or_else(
411            || view_state.field_of_view().0.to_degrees(),
412            |eye| eye.frustum.vertical_field_of_view().0.to_degrees(),
413        ),
414        center,
415        world_size,
416        bearing_degrees: angle(
417            pose.as_ref().map(|pose| pose.bearing_degrees),
418            view_state.camera().get_bearing().0.to_degrees(),
419        ),
420        pitch_degrees: angle(
421            pose.as_ref().map(|pose| pose.pitch_degrees),
422            view_state.camera().get_pitch().0.to_degrees(),
423        ),
424        roll_degrees: angle(
425            pose.as_ref().map(|pose| pose.roll_degrees),
426            view_state.camera().get_roll().0.to_degrees(),
427        ),
428        center_offset: external_eye
429            .map_or_else(|| view_state.center_offset(), |_| Point2::new(0.0, 0.0)),
430        body: view_state.body(),
431        // A host's eye is placed where it is; the center's terrain must not move it.
432        // A free-globe pose names its target's height, so it orbits that height from the
433        // first frame, before the terrain has decided what the flat camera orbits.
434        target_elevation_meters: if external_eye.is_none()
435            && (pose.is_some() || view_state.globe_orbits_center())
436        {
437            view_state.center_elevation()
438        } else {
439            0.0
440        },
441        radius_pixels: pose.as_ref().map(|pose| pose.radius_pixels),
442    };
443    match external_eye {
444        Some(eye) => GlobeCameraState::from_external_eye(options, eye),
445        None => GlobeCameraState::new(options),
446    }
447    .map_err(|source| ProjectionStateError::GlobeCamera { source })
448}

450pub(crate) fn mercator_world_to_lat_lon(x: f64, y: f64, world_size: f64) -> LatLon {
451    let longitude = x / world_size * 360.0 - 180.0;
452    let latitude = (std::f64::consts::PI * (1.0 - 2.0 * y / world_size))
453        .sinh()
454        .atan()
455        .to_degrees();
456    LatLon::new(latitude, longitude)
457}
