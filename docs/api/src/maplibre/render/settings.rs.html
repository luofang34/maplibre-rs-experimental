10/// Provides configuration for renderer initialization. Use [`Device::features`](wgpu::Device::features),
11/// [`Device::limits`](wgpu::Device::limits), and the [`WgpuAdapterInfo`](wgpu::AdapterInfo)
12/// resource to get runtime information about the actual adapter, backend, features, and limits.
13#[derive(Clone)]
14pub struct WgpuSettings {
15    /// Label included in GPU diagnostics for the requested device.
16    pub device_label: Option<Cow<'static, str>>,
17    /// Backends available for adapter selection; `None` enables all compiled backends.
18    /// The default reads `WGPU_BACKEND` when present.
19    pub backends: Option<Backends>,
20    /// Preference used when choosing a compatible adapter; not a guaranteed device class.
21    pub power_preference: PowerPreference,
22    /// Explicitly required features, added after automatic feature selection and exclusions.
23    /// These take precedence over [`Self::disabled_features`]; unsupported requests fail initialization.
24    pub features: Features,
25    /// Features to remove from automatic selection, unless explicitly required by [`Self::features`].
26    pub disabled_features: Option<Features>,
27    /// Limits used for the device request, rather than the adapter's full capabilities.
28    /// Unsupported values fail initialization unless reduced by [`Self::constrained_limits`].
29    pub limits: Limits,
30    /// Optional ceilings on requested capabilities: maxima decrease and minimum alignments increase.
31    /// This is applied to [`Self::limits`] and does not expand the request toward adapter limits.
32    pub constrained_limits: Option<Limits>,
33
34    /// Requests a GPU trace in `wgpu_trace` under the working directory when the backend supports it.
35    pub record_trace: bool,
36}
37
38impl Default for WgpuSettings {
39    fn default() -> Self {
40        let backends = Some(wgpu::Backends::from_env().unwrap_or(Backends::all()));
41
42        let limits = if cfg!(feature = "web-webgl") {
43            Limits {
44                max_texture_dimension_2d: 4096,
45                ..Limits::downlevel_webgl2_defaults()
46            }
47        } else if cfg!(target_os = "android") {
48            Limits {
49                max_storage_textures_per_shader_stage: 4,
50                max_compute_workgroups_per_dimension: 0,
51                max_compute_workgroup_size_z: 0,
52                max_compute_workgroup_size_y: 0,
53                max_compute_workgroup_size_x: 0,
54                max_compute_workgroup_storage_size: 0,
55                max_compute_invocations_per_workgroup: 0,
56                ..Limits::downlevel_defaults()
57            }
58        } else {
59            Limits {
60                ..Limits::default()
61            }
62        };
63
64        let features = if cfg!(target_arch = "wasm32") {
65            Features::empty()
66        } else {
67            Features::TEXTURE_ADAPTER_SPECIFIC_FORMAT_FEATURES
68        };
69
70        Self {
71            device_label: Default::default(),
72            backends,
73            power_preference: PowerPreference::HighPerformance,
74            features,
75            disabled_features: None,
76            limits,
77            constrained_limits: None,
78            record_trace: false,
79        }
80    }
81}

93/// Configuration resource for [Multi-Sample Anti-Aliasing](https://en.wikipedia.org/wiki/Multisample_anti-aliasing).
94///
95pub struct Msaa {
96    /// The requested number of samples for Multi-Sample Anti-Aliasing. Higher numbers result in
97    /// smoother edges.
98    /// Defaults to 4.
99    /// WebGL uses one sample because symbol occlusion reads the depth attachment.
100    pub samples: u32,
101}
102
103impl Msaa {
104    /// Whether more than one sample is requested; this does not validate format support.
105    pub fn is_multisampling(&self) -> bool {
106        self.samples > 1
107    }
108}
109
110impl Default for Msaa {
111    fn default() -> Self {
112        // By default we are trying to multisample
113        Self { samples: 4 }
114    }
115}

117/// Capacity of each GPU tile buffer pool in elements, not bytes.
118/// Actual allocations are capped by the device's maximum buffer size.
119#[derive(Clone, Copy, Debug, PartialEq, Eq)]
120pub struct BufferPoolSizes {
121    /// Number of geometry vertices in the vertex pool.
122    pub vertices: u64,
123    /// Number of draw indices in the index pool.
124    pub indices: u64,
125    /// Number of per-vertex feature metadata records.
126    pub feature_metadata: u64,
127    /// Number of layer metadata records.
128    pub layer_metadata: u64,
129}
130
131impl BufferPoolSizes {
132    /// Zero-capacity pool configuration that disables geometry uploads for a pool family
133    /// without creating invalid zero-sized GPU buffers.
134    pub const EMPTY: Self = Self {
135        vertices: 0,
136        indices: 0,
137        feature_metadata: 0,
138        layer_metadata: 0,
139    };
140
141    /// Returns `true` when every pool sub-allocation capacity is zero.
142    pub const fn is_empty(&self) -> bool {
143        self.vertices == 0
144            && self.indices == 0
145            && self.feature_metadata == 0
146            && self.layer_metadata == 0
147    }
148}
149
150impl Default for BufferPoolSizes {
151    fn default() -> Self {
152        Self {
153            vertices: 10 * 1_000_000,
154            indices: 10 * 1_000_000,
155            feature_metadata: 10 * 1024 * 1000,
156            layer_metadata: 10 * 1024,
157        }
158    }
159}

161/// Styling parameters for [`TerrainDrapeMode::ProceduralRelief`], shading pure 3D DEM relief
162/// directly in `MAIN_PASS` without allocating `DrapeScratch` or `DrapeCache`.
163#[derive(Clone, Copy, Debug, PartialEq)]
164pub struct ProceduralReliefStyle {
165    /// Optional host reference altitude in metres MSL (e.g. ownship altitude for TAWS
166    /// relative-elevation amber/red terrain awareness bands). `None` disables TAWS bands.
167    pub reference_altitude_meters: Option<f32>,
168    /// Enables hypsometric elevation color ramp based on stitched DEM elevation (metres MSL).
169    pub hypsometric: bool,
170    /// Analytical DEM surface-normal hillshade strength (`0.0..=1.0`, default `0.65`).
171    pub slope_shading: f32,
172    /// Ground metric gridline spacing in metres (`<= 0.0` disables gridlines; default `1000.0`).
173    pub grid_spacing_meters: f32,
174    /// Opacity of procedural metric gridlines (`0.0..=1.0`, default `0.18`).
175    pub grid_opacity: f32,
176}
177
178impl Default for ProceduralReliefStyle {
179    fn default() -> Self {
180        Self {
181            reference_altitude_meters: None,
182            hypsometric: true,
183            slope_shading: 0.65,
184            grid_spacing_meters: 1000.0,
185            grid_opacity: 0.18,
186        }
187    }
188}

190/// Controls whether 3D terrain bakes per-tile 2D style drapes (`DRAPE_PASS`) or shades pure
191/// procedural DEM relief directly in `MAIN_PASS`.
192#[derive(Clone, Copy, Debug, PartialEq)]
193pub enum TerrainDrapeMode {
194    /// Bakes drapeable style layers into mipmapped per-tile textures of `edge_px × edge_px`.
195    Draped {
196        /// Edge length in pixels of each drape texture (clamped to `>= MIN_DRAPE_SIZE`).
197        edge_px: u32,
198    },
199    /// Allocates zero `DrapeScratch` and zero `DrapeCache` textures, shading stitched DEM
200    /// elevation, analytical surface normal, optional TAWS relative-altitude bands, and
201    /// procedural metric gridlines directly in `MAIN_PASS`.
202    ProceduralRelief(ProceduralReliefStyle),
203}
204
205impl TerrainDrapeMode {
206    /// Default [`TerrainDrapeMode::ProceduralRelief`] with hypsometric bands, analytical slope
207    /// shading, and `1 km` gridlines.
208    pub const fn procedural_relief() -> Self {
209        Self::ProceduralRelief(ProceduralReliefStyle {
210            reference_altitude_meters: None,
211            hypsometric: true,
212            slope_shading: 0.65,
213            grid_spacing_meters: 1000.0,
214            grid_opacity: 0.18,
215        })
216    }
217
218    /// Returns `true` when drape baking is disabled in favor of direct procedural DEM relief.
219    pub const fn is_procedural_relief(&self) -> bool {
220        matches!(self, Self::ProceduralRelief(_))
221    }
222}
223
224impl Default for TerrainDrapeMode {
225    fn default() -> Self {
226        Self::Draped {
227            edge_px: crate::terrain::resources::DRAPE_SIZE,
228        }
229    }
230}

232#[derive(Clone, Copy)]
233/// Requested render target properties and pool capacities used during renderer initialization.
234pub struct RendererSettings {
235    /// Requested color/depth sample count; the WebGL backend reduces this to one.
236    pub msaa: Msaa,
237    /// Capacity of the vector buffer pool.
238    pub buffer_pools: BufferPoolSizes,
239    /// Capacity of the symbol buffer pools in elements of their respective vertex/metadata types.
240    pub symbol_pools: BufferPoolSizes,
241    /// Requested surface format; `None` selects a supported window format or offscreen RGBA8.
242    pub texture_format: Option<TextureFormat>,
243    /// Depth/stencil format; initialization selects `Depth32FloatStencil8` when supported.
244    pub depth_texture_format: TextureFormat,
245    /// Present mode for surfaces if a surface is used.
246    pub present_mode: PresentMode,
247    /// Edge, in pixels, of the texture each terrain tile's map is drawn into. Each texture
248    /// costs four bytes a texel and a third more for mipmaps, so the default of
249    /// [`DRAPE_SIZE`](crate::terrain::resources::DRAPE_SIZE) holds about 21 MiB a tile; a
250    /// phone, which shares its memory with the GPU, keeps more tiles in less.
251    pub terrain_drape_size: u32,
252    /// Whether terrain bakes 2D style drapes (`DRAPE_PASS`) or shades procedural DEM relief
253    /// directly in `MAIN_PASS` (`TerrainDrapeMode::ProceduralRelief`).
254    pub terrain_drape_mode: TerrainDrapeMode,
255    /// Draws every layer fragment as an equal step of grey added to what is below, so the
256    /// brightness of a pixel counts how often it was drawn, as GL JS's overdraw inspector does.
257    pub overdraw_inspector: bool,
258    /// Allocates and populates a persistent single-sample reversed-Z `Depth32Float` attachment
259    /// via `DEPTH_COPY` on offscreen targets so hosts can sample or copy depth without `XrFrame`.
260    pub depth_readout: bool,
261}
262
263impl RendererSettings {
264    /// Resolves the active [`TerrainDrapeMode`], honoring legacy `terrain_drape_size` overrides
265    /// when `terrain_drape_mode` was left at its default.
266    pub fn effective_terrain_drape_mode(&self) -> TerrainDrapeMode {
267        match self.terrain_drape_mode {
268            TerrainDrapeMode::ProceduralRelief(style) => TerrainDrapeMode::ProceduralRelief(style),
269            TerrainDrapeMode::Draped { edge_px }
270                if edge_px != crate::terrain::resources::DRAPE_SIZE =>
271            {
272                TerrainDrapeMode::Draped { edge_px }
273            }
274            TerrainDrapeMode::Draped { .. } => TerrainDrapeMode::Draped {
275                edge_px: self.terrain_drape_size,
276            },
277        }
278    }
279
280    pub(super) fn with_backend_msaa(mut self, backend: wgpu::Backend) -> Self {
281        // Symbol occlusion and compositor depth copies sample the depth attachment.
282        // WebGL supports multisampled renderbuffers, but cannot sample their depth.
283        if cfg!(target_arch = "wasm32") && backend == wgpu::Backend::Gl {
284            self.msaa.samples = 1;
285        }
286        self
287    }
288
289    /// Selects 32-bit float depth when the device offers it.
290    ///
291    /// Reversed-Z only recovers precision on a float depth buffer; the 24-bit fallback keeps
292    /// rendering correct but leaves depth precision at fixed-point levels.
293    pub fn with_float_depth_if_supported(mut self, features: Features) -> Self {
294        if features.contains(Features::DEPTH32FLOAT_STENCIL8) {
295            self.depth_texture_format = TextureFormat::Depth32FloatStencil8;
296        }
297        self
298    }
299}

301impl Default for RendererSettings {

305            buffer_pools: BufferPoolSizes::default(),
306            symbol_pools: BufferPoolSizes::default(),
307            terrain_drape_size: crate::terrain::resources::DRAPE_SIZE,
308            terrain_drape_mode: TerrainDrapeMode::default(),

312            present_mode: PresentMode::AutoVsync,
313            overdraw_inspector: false,
314            depth_readout: false,
315        }
