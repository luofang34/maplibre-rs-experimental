15/// Retained node definitions and resource dependencies for a render pass sequence.
16/// Nodes can retain internal state; the runner borrows the topology while executing nodes
17/// in dependency order and recording their commands in a shared encoder.
18///
19/// It consists of three main components: [`Nodes`](Node), [`Edges`](Edge)
20/// and [`Slots`](super::SlotType).
21///
22/// Nodes are responsible for generating draw calls and operating on input and output slots.
23/// Edges specify the order of execution for nodes and connect input and output slots together.
24/// Slots describe the render resources created or used by the nodes.
25///
26/// Additionally a render graph can contain multiple sub graphs, which are run by the
27/// corresponding nodes. Every render graph can have its own optional input node.
28/// The caller must construct an acyclic graph and connect every required node input;
29/// connection insertion does not detect cycles.
30///
31/// ## Example
32/// Here is a simple render graph example with two nodes connected by a node edge.
33/// ```
34/// use maplibre::render::graph::{EmptyNode, RenderGraph, RenderGraphError};
35/// # fn main() -> Result<(), RenderGraphError> {
36/// let mut graph = RenderGraph::default();
37/// graph.add_node("prepare", EmptyNode);
38/// graph.add_node("draw", EmptyNode);
39/// graph.add_node_edge("prepare", "draw")?;
40/// # Ok(())
41/// # }
42/// ```
43#[derive(Default)]
44pub struct RenderGraph {
45    nodes: HashMap<NodeId, NodeState>,
46    node_names: HashMap<Cow<'static, str>, NodeId>,
47    sub_graphs: HashMap<Cow<'static, str>, RenderGraph>,
48    input_node: Option<NodeId>,
49
50    current_id: usize,
51}

57    /// Updates all nodes and sub graphs of the render graph. Should be called before executing it.
58    pub fn update(&mut self, state: &mut RenderResources) {
59        for node in self.nodes.values_mut() {
60            node.node.update(state);
61        }
62
63        for sub_graph in self.sub_graphs.values_mut() {
64            sub_graph.update(state);
65        }
66    }

68    /// Creates the graph input node with matching input and output declarations.
69    ///
70    /// # Panics
71    /// Panics if an input node has already been set, even if it was subsequently removed by name.
72    pub fn set_input(&mut self, inputs: Vec<SlotInfo>) -> NodeId {
73        assert!(self.input_node.is_none(), "Graph already has an input node");
74
75        let id = self.add_node("GraphInputNode", GraphInputNode { inputs });
76        self.input_node = Some(id);
77        id
78    }

80    /// Returns the input node's state if it is still stored in the graph.
81    #[inline]
82    pub fn input_node(&self) -> Option<&NodeState> {
83        self.input_node.and_then(|id| self.get_node_state(id).ok())
84    }

86    /// Allocates an ID for the node and binds its name for subsequent name-based lookups.
87    /// Reusing a name rebinds the lookup; the earlier node and its edges remain under its ID.
88    pub fn add_node<T>(&mut self, name: impl Into<Cow<'static, str>>, node: T) -> NodeId
89    where
90        T: Node,
91    {
92        let id = NodeId::new(self.current_id);
93        self.current_id = self.current_id.wrapping_add(1);
94        let name = name.into();
95        let mut node_state = NodeState::new(id, node);
96        node_state.name = Some(name.clone());
97        self.nodes.insert(id, node_state);
98        self.node_names.insert(name, id);
99        id
100    }
101
102    /// Removes the `node` with the `name` from the graph.
103    /// If the name does not exist, nothing happens.
104    pub fn remove_node(
105        &mut self,
106        name: impl Into<Cow<'static, str>>,
107    ) -> Result<(), RenderGraphError> {
108        let name = name.into();
109        if let Some(id) = self.node_names.remove(&name) {
110            if let Some(node_state) = self.nodes.remove(&id) {
111                // Remove all edges from other nodes to this one. Note that as we're removing this
112                // node, we don't need to remove its input edges
113                for input_edge in node_state.edges.input_edges().iter() {
114                    match input_edge {
115                        Edge::SlotEdge {
116                            output_node,
117                            output_index: _,
118                            input_node: _,
119                            input_index: _,
120                        } => {
121                            if let Ok(output_node) = self.get_node_state_mut(*output_node) {
122                                output_node.edges.remove_output_edge(input_edge.clone())?;
123                            }
124                        }
125                        Edge::NodeEdge {
126                            input_node: _,
127                            output_node,
128                        } => {
129                            if let Ok(output_node) = self.get_node_state_mut(*output_node) {
130                                output_node.edges.remove_output_edge(input_edge.clone())?;
131                            }
132                        }
133                    }
134                }
135                // Remove all edges from this node to other nodes. Note that as we're removing this
136                // node, we don't need to remove its output edges
137                for output_edge in node_state.edges.output_edges().iter() {
138                    match output_edge {
139                        Edge::SlotEdge {
140                            output_node: _,
141                            output_index: _,
142                            input_node,
143                            input_index: _,
144                        } => {
145                            if let Ok(input_node) = self.get_node_state_mut(*input_node) {
146                                input_node.edges.remove_input_edge(output_edge.clone())?;
147                            }
148                        }
149                        Edge::NodeEdge {
150                            output_node: _,
151                            input_node,
152                        } => {
153                            if let Ok(input_node) = self.get_node_state_mut(*input_node) {
154                                input_node.edges.remove_input_edge(output_edge.clone())?;
155                            }
156                        }
157                    }
158                }
159            }
160        }
161
162        Ok(())
163    }

165    /// Retrieves the [`NodeState`] referenced by the `label`.
166    pub fn get_node_state(
167        &self,
168        label: impl Into<NodeLabel>,
169    ) -> Result<&NodeState, RenderGraphError> {
170        let label = label.into();
171        let node_id = self.get_node_id(&label)?;
172        self.nodes
173            .get(&node_id)
174            .ok_or(RenderGraphError::InvalidNode(label))
175    }
176
177    /// Retrieves the [`NodeState`] referenced by the `label` mutably.
178    pub fn get_node_state_mut(
179        &mut self,
180        label: impl Into<NodeLabel>,
181    ) -> Result<&mut NodeState, RenderGraphError> {
182        let label = label.into();
183        let node_id = self.get_node_id(&label)?;
184        self.nodes
185            .get_mut(&node_id)
186            .ok_or(RenderGraphError::InvalidNode(label))
187    }
188
189    /// Resolves a registered name or passes through an explicit ID without checking its presence.
190    pub fn get_node_id(&self, label: impl Into<NodeLabel>) -> Result<NodeId, RenderGraphError> {
191        let label = label.into();
192        match label {
193            NodeLabel::Id(id) => Ok(id),
194            NodeLabel::Name(ref name) => self
195                .node_names
196                .get(name)
197                .cloned()
198                .ok_or(RenderGraphError::InvalidNode(label)),
199        }
200    }

202    /// Retrieves the [`Node`] referenced by the `label`.
203    pub fn get_node<T>(&self, label: impl Into<NodeLabel>) -> Result<&T, RenderGraphError>
204    where
205        T: Node,
206    {
207        self.get_node_state(label).and_then(|n| n.node())
208    }

210    /// Retrieves the [`Node`] referenced by the `label` mutably.

212        &mut self,
213        label: impl Into<NodeLabel>,
214    ) -> Result<&mut T, RenderGraphError>
215    where
216        T: Node,

245    /// Returns an iterator over a tuple of the input edges and the corresponding output nodes
246    /// for the node referenced by the label.
247    pub fn iter_node_inputs(
248        &self,
249        label: impl Into<NodeLabel>,
250    ) -> Result<impl Iterator<Item = (&Edge, &NodeState)>, RenderGraphError> {
251        let node = self.get_node_state(label)?;
252        Ok(node
253            .edges
254            .input_edges()
255            .iter()
256            .map(|edge| (edge, edge.get_output_node()))
257            .map(move |(edge, output_node_id)| {
258                (edge, self.get_node_state(output_node_id).unwrap())
259            }))
260    }
261
262    /// Returns an iterator over a tuple of the output edges and the corresponding input nodes
263    /// for the node referenced by the label.
264    pub fn iter_node_outputs(
265        &self,
266        label: impl Into<NodeLabel>,
267    ) -> Result<impl Iterator<Item = (&Edge, &NodeState)>, RenderGraphError> {
268        let node = self.get_node_state(label)?;
269        Ok(node
270            .edges
271            .output_edges()
272            .iter()
273            .map(|edge| (edge, edge.get_input_node()))
274            .map(move |(edge, input_node_id)| (edge, self.get_node_state(input_node_id).unwrap())))
275    }

277    /// Adds the `sub_graph` with the `name` to the graph.

300impl Debug for RenderGraph {
301    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
302        for node in self.iter_nodes() {
303            writeln!(f, "{:?}", node.id)?;
304            writeln!(f, "  in: {:?}", node.input_slots)?;
305            writeln!(f, "  out: {:?}", node.output_slots)?;
306        }
307
308        Ok(())
309    }

312/// A [`Node`] which acts as an entry point for a [`RenderGraph`] with custom inputs.
313/// It has the same input and output slots and simply copies them over when run.
314pub struct GraphInputNode {
315    inputs: Vec<SlotInfo>,
316}
317
318impl Node for GraphInputNode {
319    fn input(&self) -> Vec<SlotInfo> {
320        self.inputs.clone()
321    }
322
323    fn output(&self) -> Vec<SlotInfo> {
324        self.inputs.clone()
325    }
326
327    fn run(
328        &self,
329        graph: &mut RenderGraphContext,
330        _render_context: &mut RenderContext,
331        _state: &RenderResources,
332        _world: &World,
333    ) -> Result<(), NodeRunError> {
334        for i in 0..graph.inputs().len() {
335            let input = graph.inputs()[i].clone();
336            graph.set_output(i, input)?;
337        }
338        Ok(())
339    }
