32/// Drops a source that left the style: its results on every tile and its textures, so neither
33/// outlives it.
34pub(crate) fn forget_source(world: &mut World, name: &str) {
35    let source = RasterSourceId::from(name);
36    let coords: Vec<WorldTileCoords> = world.tiles.tiles.values().map(|tile| tile.coords).collect();
37    for coords in coords {
38        if let Some(component) = world
39            .tiles
40            .query_mut::<&mut RasterLayersDataComponent>(coords)
41        {
42            component.layers.retain(|layer| layer.source() != &source);
43        }
44        if let Some(dem) = world.tiles.query_mut::<&mut DemLayersComponent>(coords) {
45            dem.remove(&source);
46        }
47    }
48    if let Some(Eventually::Initialized(raster)) = world
49        .resources
50        .get_mut::<Eventually<resource::RasterResources>>()
51    {
52        raster.forget_source(&source);
53    }
54    let keep_for_terrain = world
55        .resources
56        .get::<Eventually<crate::terrain::resources::TerrainResources>>()
57        .is_some();
58    if !keep_for_terrain {
59        if let Some(pool) = world.resources.get_mut::<GpuDemPool>() {
60            pool.drop_source(&source);
61        }
62    }
63}

82/// Registers raster requests, worker results, uploads and draws with the render schedule.
83/// `T` supplies the host-specific worker message representation.
84pub struct RasterPlugin<T>(PhantomData<T>);
85
86impl<T: RasterTransferables> Default for RasterPlugin<T> {
87    fn default() -> Self {
88        Self(Default::default())
89    }
90}
91
92impl<E: Environment, T: RasterTransferables> Plugin<E> for RasterPlugin<T> {
93    fn build(
94        &self,
95        schedule: &mut Schedule,
96        kernel: Rc<Kernel<E>>,
97        world: &mut World,
98        _renderer: &mut crate::render::Renderer,
99    ) -> Result<(), crate::plugin::PluginError> {
100        world
101            .resources
102            .insert(Eventually::<RasterResources>::Uninitialized);
103
104        world
105            .resources
106            .get_or_init_mut::<ViewTileSources>()
107            .add::<RasterTilesDone>(TileKind::Raster);
108
109        schedule.add_system_to_stage(
110            RenderStageLabel::Request,
111            SystemContainer::new(RequestSystem::<E, T>::new(&kernel)),
112        )?;
113        schedule.add_system_to_stage(
114            RenderStageLabel::Request,
115            SystemContainer::new(PopulateWorldSystem::<E, T>::new(&kernel)),
116        )?;
117        schedule.add_system_to_stage(RenderStageLabel::Prepare, resource_system)?;
118        schedule.add_system_to_stage(RenderStageLabel::Queue, queue_system)?;
119        Ok(())
120    }
121}

123/// Decoded tile pixels waiting for upload; availability does not imply GPU residency.
124pub struct AvailableRasterLayerData {
125    /// Tile-grid coordinates covered by the image.
126    pub coords: WorldTileCoords,
127    /// Style source owning these pixels, or the explicit fallback source.
128    pub source: RasterSourceId,
129    /// Decoded RGBA8 texels sampled without sRGB-to-linear conversion.
130    pub image: RgbaImage,
131}
132
133/// A completed raster request that produced no usable image.
134pub struct MissingRasterLayerData {
135    /// Coordinates of the unavailable tile.
136    pub coords: WorldTileCoords,
137    /// Style source whose request yielded no image.
138    pub source: RasterSourceId,
139}
140
141/// A worker result retained on a tile before GPU upload and fallback selection.
142pub enum RasterLayerData {
143    /// Decoded pixels can supply coverage once uploaded.
144    Available(AvailableRasterLayerData),
145    /// Fetching or decoding yielded no image; usable ancestors must remain eligible.
146    Missing(MissingRasterLayerData),
147}
148
149impl RasterLayerData {
150    pub(crate) fn source(&self) -> &RasterSourceId {
151        match self {
152            Self::Available(layer) => &layer.source,
153            Self::Missing(layer) => &layer.source,
154        }
155    }
156}
157
158impl RasterLayersDataComponent {
159    pub(crate) fn record(&mut self, layer: RasterLayerData) -> bool {
160        let has_image = matches!(layer, RasterLayerData::Available(_));
161        if let Some(existing) = self
162            .layers
163            .iter_mut()
164            .find(|existing| existing.source() == layer.source())
165        {
166            // A failed refresh must not discard usable pixels from an earlier response.
167            if !has_image && matches!(existing, RasterLayerData::Available(_)) {
168                return false;
169            }
170            *existing = layer;
171        } else {
172            self.layers.push(layer);
173        }
174        has_image
175    }
176
177    /// Whether this source delivered an image for the tile.
178    pub fn has_source_image(&self, source: &RasterSourceId) -> bool {
179        self.layers.iter().any(
180            |layer| matches!(layer, RasterLayerData::Available(data) if &data.source == source),
181        )
182    }
183
184    /// Whether this source has a completed unavailable result.
185    pub fn source_is_missing(&self, source: &RasterSourceId) -> bool {
186        self.layers
187            .iter()
188            .any(|layer| matches!(layer, RasterLayerData::Missing(data) if &data.source == source))
189    }
190
191    /// Whether a result has arrived for this source.
192    pub fn has_source_result(&self, source: &RasterSourceId) -> bool {
193        self.layers.iter().any(|layer| layer.source() == source)
194    }
195
196    /// Whether any source delivered an image for the tile.
197    pub fn has_image(&self) -> bool {
198        self.layers
199            .iter()
200            .any(|layer| matches!(layer, RasterLayerData::Available(_)))
201    }
202
203    /// Whether every received result is unavailable, including fetch and decode failures.
204    pub fn is_missing(&self) -> bool {
205        !self.layers.is_empty() && !self.has_image()
206    }
207}

221fn source_texture_ready(source: &RasterSourceId, coords: WorldTileCoords, world: &World) -> bool {
222    match world.resources.get::<Eventually<RasterResources>>() {
223        Some(Eventually::Initialized(resources)) if resources.draws(source) => {
224            resources.get_bound_texture(source, &coords).is_some()
225        }
226        _ => true,
227    }
228}

230impl HasTile for RasterTilesDone {
231    fn has_source_tile(
232        &self,
233        source: &RasterSourceId,
234        coords: WorldTileCoords,
235        world: &World,
236    ) -> bool {
237        let cpu_ready = world
238            .tiles
239            .query::<&RasterLayersDataComponent>(coords)
240            .is_some_and(|data| data.has_source_image(source))
241            || world
242                .tiles
243                .query::<&DemLayersComponent>(coords)
244                .is_some_and(|dem| dem.contains(source));
245        cpu_ready && source_texture_ready(source, coords, world)
246    }
247
248    fn has_tile(&self, coords: WorldTileCoords, world: &World) -> bool {
249        let raster_ready = world
250            .tiles
251            .query::<&RasterLayersDataComponent>(coords)
252            .is_some_and(|data| {
253                data.layers.iter().any(|layer| {
254                    matches!(
255                        layer,
256                        RasterLayerData::Available(available)
257                            if source_texture_ready(&available.source, coords, world)
258                    )
259                })
260            });
261        let dem_ready = match world.resources.get::<Eventually<RasterResources>>() {
262            Some(Eventually::Initialized(resources)) => world
263                .tiles
264                .query::<&DemLayersComponent>(coords)
265                .is_some_and(|dem| {
266                    dem.iter().any(|(source, _)| {
267                        resources.draws(source)
268                            && resources.get_bound_texture(source, &coords).is_some()
269                    })
270                }),
271            _ => false,
272        };
273        raster_ready || dem_ready
274    }
275
276    fn nearest_ancestor_candidate(
277        &self,
278        coords: WorldTileCoords,
279        world: &World,
280    ) -> Option<WorldTileCoords> {
281        let raster = world
282            .tiles
283            .pyramid_of::<RasterLayersDataComponent>()
284            .nearest_ancestor(coords)
285            .map(|(ancestor, _)| ancestor);
286        let dem = world
287            .tiles
288            .pyramid_of::<DemLayersComponent>()
289            .nearest_ancestor(coords)
290            .map(|(ancestor, _)| ancestor);
291        raster.into_iter().chain(dem).max_by_key(|c| c.z)
292    }
293
294    fn has_descendant_candidates(
295        &self,
296        coords: WorldTileCoords,
297        max_depth: usize,
298        world: &World,
299    ) -> bool {
300        world
301            .tiles
302            .pyramid_of::<RasterLayersDataComponent>()
303            .has_descendants(coords, max_depth)
304            || world
305                .tiles
306                .pyramid_of::<DemLayersComponent>()
307                .has_descendants(coords, max_depth)
308    }
309}
