35/// Errors returned while unpacking or registering font files.
36#[derive(Clone, Debug, PartialEq, Eq, thiserror::Error)]
37pub enum FontError {
38    /// Leading magic signature is none of `TTF`, `OTF`, `TTC`, `WOFF`, or `WOFF2`.
39    #[error("unrecognized font container format")]
40    UnrecognizedFormat,
41    /// Structural or decompression failure inside a font container.
42    #[error("invalid font container: {0}")]
43    InvalidContainer(&'static str),
44}
45
46static GLOBAL_FONT_SYSTEM: LazyLock<Mutex<FontSystem>> =
47    LazyLock::new(|| Mutex::new(FontSystem::new()));
48
49/// Acquires the shared process-wide [`FontSystem`] used by symbol asset loading.
50pub fn global_font_system() -> MutexGuard<'static, FontSystem> {
51    GLOBAL_FONT_SYSTEM
52        .lock()
53        .unwrap_or_else(std::sync::PoisonError::into_inner)
54}
55
56/// Registers an in-memory `TTF`, `OTF`, `TTC`, `WOFF`, or `WOFF2` font into the shared
57/// [`FontSystem`], optionally binding it under `family_override`.
58pub fn register_font_bytes(
59    family_override: Option<&str>,
60    data: Vec<u8>,
61) -> Result<Vec<u64>, FontError> {
62    global_font_system().register_font_bytes(family_override, data)
63}
