1//! Prepares GPU-owned resources by initializing them if they are uninitialized or out-of-date.
2use crate::{
3    context::MapContext,
4    debug::DebugPipeline,
5    render::{
6        eventually::{Eventually, Eventually::Initialized},
7        projection::ProjectionGpuResources,
8        resource::{RenderPipeline, TilePipeline},
9        shaders,
10        shaders::Shader,
11        RenderResources, Renderer,
12    },
13    tcs::system::{SystemError, SystemResult},
14};
15
16pub fn resource_system(
17    MapContext {
18        world,
19        renderer:
20            Renderer {
21                device,
22                resources: RenderResources { surface, .. },
23                settings,
24                ..
25            },
26        ..
27    }: &mut MapContext,
28) -> SystemResult {
29    let Some((debug_pipeline, Initialized(projection_resources))) = world.resources.query_mut::<(
30        &mut Eventually<DebugPipeline>,
31        &mut Eventually<ProjectionGpuResources>,
32    )>() else {
33        return Err(SystemError::Dependencies);
34    };
35
36    debug_pipeline.initialize(|| {
37        let mask_shader = shaders::TileMaskShader {
38            format: surface.surface_format(),
39            draw_colors: true,
40            debug_lines: true,
41        };
42
43        let pipeline = TilePipeline::new(
44            "debug_pipeline".into(),
45            *settings,
46            mask_shader.describe_vertex(),
47            mask_shader.describe_fragment(),
48            crate::render::resource::TilePipelineOptions {
49                depth_stencil_enabled: false,
50                update_stencil: false,
51                debug_stencil: true,
52                wireframe: false,
53                multisampling: false,
54                textured: false,
55            },
56        )
57        .describe_render_pipeline()
58        .initialize_with_prefix_layouts(device, &[projection_resources.bind_group_layout()]);
59        DebugPipeline(pipeline)
60    });
61
62    Ok(())
63}
